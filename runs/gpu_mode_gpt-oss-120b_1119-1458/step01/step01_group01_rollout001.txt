"""
TriMul (outgoing) – Triton‑accelerated implementation.

The expensive N³ reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is computed by a custom batched‑matrix‑multiply kernel.
We reshape the tensors to (B·H, N, N) so each (batch, hidden) pair becomes
an independent GEMM of size N×N.  The kernel tiles the GEMM in
(M, N, K) blocks, loads tiles in FP16, performs the dot product with
Tensor‑Core MMA (fp16→fp32 accumulation) and writes back the result as
FP16.  All other operations (LayerNorm, Linear projections, sigmoids,
masking and the final output projection) are handled with PyTorch.
The whole pipeline returns a FP32 tensor of shape [B, N, N, dim].
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched per‑hidden‑channel matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # base pointers
    stride_ap, stride_am, stride_ak,          # strides for A: (P, M, K)
    stride_bp, stride_bk, stride_bn,          # strides for B: (P, K, N)
    stride_cp, stride_cm, stride_cn,          # strides for C: (P, M, N)
    M, N, K,                                  # matrix dimensions (all = seq_len)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes for each batch*hidden pair p:
        C[p, i, j] = Σ_k A[p, i, k] * B[p, k, j]
    where A and B are (P, M, K) and (P, K, N) respectively.
    All tensors are fp16; accumulation is carried out in fp32.
    """
    pid_m = tl.program_id(0)      # block row index
    pid_n = tl.program_id(1)      # block column index
    pid_p = tl.program_id(2)      # batch*hidden index

    # ---- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (p) slice
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- K loop -------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (M, K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # fp16 load

        # Load B tile: (K, N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # fp16 load

        # Matmul tile (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)           # result is fp32

    # ---- write back ----------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),      # cast back to fp16
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the batched‑matmul kernel
# ----------------------------------------------------------------------
def _triton_matmul(
    left: torch.Tensor,
    right: torch.Tensor,
    seq_len: int,
    hidden_dim: int,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    left / right : [B, N, N, H] (fp16)
    Returns out   : [B, N, N, H] (fp16)
    """
    B = left.shape[0]

    # (B, N, N, H) -> (B, H, N, N)   (contiguous)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs a transpose on the two N axes before flattening
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, K, J)

    # Collapse batch & hidden into a single matmul batch dimension
    P = B * hidden_dim
    left_flat = left_perm.view(P, seq_len, seq_len)
    right_flat = right_perm.view(P, seq_len, seq_len)

    out = torch.empty_like(left_flat)

    # strides (contiguous tensors)
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # blocks over M
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # blocks over N
        P,                                    # batch*hidden
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out,
        stride_ap,
        stride_am,
        stride_ak,
        stride_bp,
        stride_bk,
        stride_bn,
        stride_cp,
        stride_cm,
        stride_cn,
        seq_len,
        seq_len,
        seq_len,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,          # H100 can handle 16 warps (1024 threads) comfortably
        num_stages=3,
    )

    # reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point – outgoing TriMul
# ----------------------------------------------------------------------

    """

    N³ reduction.

            - mask       : [B, N, N]       (float32 or bool)
            - weights    : dict of model parameters (torch.Tensor)
            - config     : dict with keys dim, hidden_dim, nomask (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]                 # input & output channel dimension
    hidden_dim = config["hidden_dim"]   # internal hidden dimension
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B, N, N, dim]  fp32

    # Cast to fp16 early – the bulk of the work (linear + matmul) benefits
    x_fp16 = x.half()   # fp16, still accurate enough for subsequent ops

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]    # (hidden, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_fp16, left_proj_weight)        # [B, N, N, hidden]  fp16
    right = F.linear(x_fp16, right_proj_weight)      # [B, N, N, hidden]  fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (applied after projection)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates: sigmoid(Linear(x))
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight)).to(left.dtype)   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight)).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))               # keep fp32 for later scaling

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy N³ reduction via Triton (batched matmul)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]   # N
    out_fp16 = _triton_matmul(left, right,
                              seq_len=seq_len,
                              hidden_dim=hidden_dim,
                              BLOCK_M=128,
                              BLOCK_N=128,
                              BLOCK_K=64)

    # ------------------------------------------------------------------
    # 6. Post‑matmul processing (still in fp32 for stability)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # [B, N, N, hidden]

    # LayerNorm over hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # Apply output gate (broadcast over hidden dim)
    out = out * out_gate

    # Final linear projection back to original dimension
    to_out_weight = weights["to_out.weight"]          # (dim, hidden)
    out = F.linear(out, to_out_weight)               # [B, N, N, dim] fp32

    return out
```
