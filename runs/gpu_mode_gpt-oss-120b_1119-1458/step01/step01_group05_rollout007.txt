def custom_kernel(data):
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # 1) LayerNorm
    x_norm = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # 2) Linear projections & gates combined
    # flatten
    x_flat = x_norm.view(-1, dim)               # shape (B*N*N, dim)
    # Cast to fp16
    x_flat_fp16 = x_flat.to(torch.float16)

    # Stack weight matrices
    # Each weight is shape (hidden_dim, dim)
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # Convert weight to fp16
    w_all = torch.cat([w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0).to(torch.float16) # shape (5*hidden_dim, dim)
    # mm
    out_all_fp16 = torch.nn.functional.linear(x_flat_fp16, w_all)   # shape (B*N*N, 5*hidden_dim)
    # split
    splits = torch.split(out_all_fp16, hidden_dim, dim=1)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = splits
    # reshape to (B,N,N,hidden_dim)
    left_proj = left_proj_fp16.view(B, N, N, hidden_dim)
    right_proj = right_proj_fp16.view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_gate_fp16.view(B, N, N, hidden_dim))
    right_gate = torch.sigmoid(right_gate_fp16.view(B, N, N, hidden_dim))
    out_gate = torch.sigmoid(out_gate_fp16.view(B, N, N, hidden_dim))

    # 3) Apply mask & gates
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        left = fused_mul3(left_proj, mask_exp, left_gate)
        right = fused_mul3(right_proj, mask_exp, right_gate)

    # 4) core contraction
    # Cast to fp16 if not already
    left_fp16 = left   # already fp16
    right_fp16 = right

    # reshape for batched matmul
    # left[i,k]: shape (B, N, N, H) where index i dim 1, k dim 2

def fused_mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) – fast forward pass.

The implementation fuses the five linear projections (left/right projections and
the three gates) into a single half‑precision GEMM, uses a tiny Triton kernel to
apply the optional mask together with the two gates, and performs the
core N‑by‑N contraction with a single batched fp16 matrix‑multiply (torch.bmm).
All heavy arithmetic is carried out in fp16 on the H100, while the final
output is returned in the original dtype (float32).  The code respects the
provided weight tensors and the optional ``nomask`` flag.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: three‑way element‑wise product (left * mask * gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,          # pointers to the three inputs and the output
    size,                                   # total number of elements
    BLOCK_SIZE: tl.constexpr,               # compile‑time constant
):
    """out = a * b * c   (all tensors are 1‑D contiguous)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024  # good default for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]   float32

    # ------------------------------------------------------------------
    # 2. Fused linear projections + gates (single half‑precision GEMM)
    # ------------------------------------------------------------------
    # flatten to (B*N*N, dim) and cast to fp16
    x_flat = x_norm.view(-1, dim).contiguous()
    x_fp16 = x_flat.to(torch.float16)

    # pack the five weight matrices (no bias) and cast to fp16
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    w_all = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).to(torch.float16)                       # shape (5*hidden_dim, dim)

    # single matmul: (B*N*N, dim) @ (dim, 5*hidden_dim) -> (B*N*N, 5*hidden_dim)
    all_proj = F.linear(x_fp16, w_all)       # bias is None

    # split the concatenated result
    splits = torch.split(all_proj, hidden_dim, dim=1)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = splits

    # reshape to 4‑D tensors
    left_proj_fp16 = left_proj_fp16.view(B, N, N, hidden_dim)
    right_proj_fp16 = right_proj_fp16.view(B, N, N, hidden_dim)
    left_gate_fp16 = torch.sigmoid(left_gate_fp16.view(B, N, N, hidden_dim))
    right_gate_fp16 = torch.sigmoid(right_gate_fp16.view(B, N, N, hidden_dim))
    out_gate_fp16 = torch.sigmoid(out_gate_fp16.view(B, N, N, hidden_dim))

    # ------------------------------------------------------------------
    # 3. Apply optional mask and the two position gates
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj_fp16 * left_gate_fp16
        right = right_proj_fp16 * right_gate_fp16
    else:
        # Expand mask to hidden_dim and cast to fp16
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous().to(torch.float16)
        left = fused_mul3(left_proj_fp16, mask_exp, left_gate_fp16)
        right = fused_mul3(right_proj_fp16, mask_exp, right_gate_fp16)

    # ------------------------------------------------------------------
    # 4. Core contraction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    (batched fp16 GEMM via torch.bmm)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) then view as a batch of matrices
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  = left[i,k]
    right_mat = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  = right[j,k] (k axis is 2)

    BH = B * hidden_dim
    left_flat = left_mat.view(BH, N, N)   # (B*H, N, K)
    right_flat = right_mat.view(BH, N, N) # (B*H, K, N)

    # batched matrix multiply (fp16)
    out_flat = torch.bmm(left_flat, right_flat)      # (B*H, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out_fp16 = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output layer‑norm (still fp16)
    # ------------------------------------------------------------------
    out_norm_fp16 = F.layer_norm(
        out_fp16,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 6. Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_gated = out_norm_fp16 * out_gate_fp16

    # final linear: (B*N*N, hidden_dim) @ (hidden_dim, dim) -> (B*N*N, dim)
    out_gated_flat = out_gated.view(-1, hidden_dim).contiguous()
    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)
    out_final_fp16 = F.linear(out_gated_flat, to_out_weight_fp16)  # bias = None

    # reshape to original 4‑D layout and cast back to the input dtype (float32)
    out_final = out_final_fp16.view(B, N, N, dim).to(inp.dtype)

    return out_final
