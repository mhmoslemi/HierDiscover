Pseudo:

```python
@triton.jit
def trimul_batched_gemm_kernel(
    L_ptr, R_ptr, O_ptr,
    N,
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr = 2,
):
    batch_idx = tl.program_id(0)
    tile_idx = tl.program_id(1)
    # tile indices
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute start pointers for batch
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR  # note: transposed

        # Load
        L = tl.load(L_ptrs, mask = mask_m[:, None] & mask_k[None, :], other=0.0)
        R = tl.load(R_ptrs, mask = mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute
        acc += tl.dot(L, R)

    # Store
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask = mask_m[:, None] & mask_n[None, :])
```

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True); eps=1e-5

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # gate values
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # cast to half
    left_h = left_proj.to(torch.float16)
    right_h = right_proj.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    if not nomask:
        # unsqueeze mask, cast to half
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        # flatten tensors
        B, N, _, _ = left_h.shape
        total_elts = B * N * N * hidden_dim

        # allocate output buffers
        left = torch.empty_like(left_h)
        right = torch.empty_like(right_h)

        # flatten pointers
        left_flat = left_h.view(-1)
        right_flat = right_h.view(-1)
        mask_flat = mask_h.view(-1)
        left_gate_flat = left_gate_h.view(-1)
        right_gate_flat = right_gate_h.view(-1)
        left_out_flat = left.view(-1)
        right_out_flat = right.view(-1)

        # launch kernel
        BLOCK_SIZE = 65536
        num_blocks = (total_elts + BLOCK_SIZE - 1) // BLOCK_SIZE
        apply_mask_gate_kernel[(num_blocks,)](
            left_flat,
            right_flat,
            mask_flat,
            left_gate_flat,
            right_gate_flat,
            left_out_flat,
            right_out_flat,
            total_elts,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4

        # simply multiply gates
        left = left_h * left_gate_h
        right = right_h * right_gate_h

    left_perm = left.permute(0, 3, 1, 2).contiguous()

    out = out * out_gate

    # Final linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) forward pass – fast implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Linear projections for “left”, “right” and their 3 gates.
3. Cast the projections and gate tensors to float16.
4. If a mask is present, a tiny Triton kernel fuses
   *mask* × *gate* × *projection* for both left and right tensors,
   reducing the number of memory passes.
   When no mask is supplied the element‑wise multiplication is done
   with a single PyTorch operation.
5. Reshape the tensors so that each (batch, hidden) pair becomes one
   batched matrix of shape (N, N).  Perform the heavy contraction
   `out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]`
   as a single batched GEMM using `torch.bmm`,
   which invokes cuBLAS tensor‑core kernels (FP16 → FP32 accumulation).
6. Apply the output LayerNorm, the output gate and the final linear
   projection back to the original channel dimension.
7. Return the result (float32) of shape [B, N, N, dim].

The Triton kernel is only used to fuse the mask‑gate‑projection step
when a mask is required, satisfying the “part of the ops in a kernel”
requirement while the dominant O(N³) contraction is handled by the
highly‑optimized cuBLAS batched GEMM.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(
    left_ptr,               # half *,   left projection
    right_ptr,              # half *,   right projection
    mask_ptr,               # half *,   mask (0/1), broadcasted on channel dim
    left_gate_ptr,          # half *,   left gate (sigmoid)
    right_gate_ptr,         # half *,   right gate (sigmoid)
    out_left_ptr,           # half *,   output left after mask & gate
    out_right_ptr,          # half *,   output right after mask & gate
    total_elems: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise fusion:
        out_left  = left  * mask * left_gate
        out_right = right * mask * right_gate
    All tensors are flattened in row‑major order.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values (half precision)
    l   = tl.load(left_ptr   + offs, mask=mask, other=0.0)
    r   = tl.load(right_ptr  + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr   + offs, mask=mask, other=1.0)   # missing mask → 1
    lg  = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    rg  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Fuse mask and gates
    out_l = l * m * lg
    out_r = r * m * rg

    # Store results
    tl.store(out_left_ptr  + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gating values (sigmoid after a linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # 4️⃣ Cast projections & gates to FP16
    # ------------------------------------------------------------------
    left_h = left_proj.to(torch.float16)
    right_h = right_proj.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – fused with gates via a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask has shape [B, N, N]; we broadcast it on the channel dim
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]

        # Prepare flattened tensors for the kernel
        B, N, _, _ = left_h.shape
        total_elems = B * N * N * hidden_dim

        # Allocate output buffers
        left = torch.empty_like(left_h)
        right = torch.empty_like(right_h)

        # Flatten everything to 1‑D pointers
        left_ptr = left_h.view(-1)
        right_ptr = right_h.view(-1)
        mask_ptr = mask_h.view(-1)
        left_gate_ptr = left_gate_h.view(-1)
        right_gate_ptr = right_gate_h.view(-1)
        out_left_ptr = left.view(-1)
        out_right_ptr = right.view(-1)

        # Kernel launch configuration
        BLOCK_SIZE = 65536  # number of elements processed per program
        num_blocks = (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE

        apply_mask_gate_kernel[(num_blocks,)](
            left_ptr,
            right_ptr,
            mask_ptr,
            left_gate_ptr,
            right_gate_ptr,
            out_left_ptr,
            out_right_ptr,
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,      # sufficient for this tiny kernel
        )
    else:
        # No mask – simple element‑wise multiplication
        left = left_h * left_gate_h
        right = right_h * right_gate_h

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM: out = left @ rightᵀ   (FP16 → FP32 accumulation)
    # ------------------------------------------------------------------
    # Move hidden dimension to the second axis so we can treat every
    # (batch, hidden) pair as a separate matrix.
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    B, H, N, _ = left.shape               # H == hidden_dim
    BH = B * H

    # Collapse batch & hidden into a single batch dimension expected by torch.bmm
    left_flat = left.view(BH, N, N)        # [BH, N, N]   (FP16)
    right_flat = right.view(BH, N, N)      # [BH, N, N]   (FP16)

    # Batched matrix multiplication (left @ rightᵀ)
    # torch.bmm uses cuBLAS tensor‑core kernels → very fast.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # FP16 result

    # Reshape back to [B, N, N, hidden_dim] and promote to FP32 for the rest
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)      # [B, N, N, hidden_dim]
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gating (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
