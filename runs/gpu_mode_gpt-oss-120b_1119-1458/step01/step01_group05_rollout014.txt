@triton.jit
def outgoing_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,  # stride for hidden dimension within batch is 1
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute pointers offsets for left and right base
    left_ptrs = left_ptr + batch_id * stride_left_batch + offs_m[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    # Right pointer: rows correspond to k block (k dimension) and columns correspond to j offsets:
    right_ptrs = right_ptr + batch_id * stride_right_batch + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Load left block
        left = tl.load(left_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0,BLOCK_K) < N), other=0.0)
        # Load right block
        right = tl.load(right_ptrs, mask=(k*BLOCK_K + tl.arange(0,BLOCK_K)[:, None] < N) & mask_n[None, :], other=0.0)

        # dot product
        # left: (BLOCK_M, BLOCK_K), right: (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right)

        # Increment pointers
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    # Write output
    out_ptrs = out_ptr + batch_id * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16) if (tl.float16 == tl.float16) else acc, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # batch id
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guard
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers
    a_ptrs = A_ptr + pid * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = B_ptr + pid * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0,BLOCK_K) < K), other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0,BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0, dtype=tl.float16)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    c_ptrs = C_ptr + pid * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – fast Triton implementation
================================================

The kernel implements the forward pass of the “outgoing” TriMul operator
described in AlphaFold‑3.  The computation can be written as

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where

    left  = (x·W_left  + bias)  ◦ σ(x·W_left_gate)  ◦ mask
    right = (x·W_right + bias)  ◦ σ(x·W_right_gate) ◦ mask

x is first layer‑normed, the result of the contraction is layer‑normed,
gated, and finally projected back to the original `dim`.

* All linear layers and the large contraction are performed in **FP16**
  (Tensor‑core friendly) and the final result is returned in the original
  dtype (usually FP32).
* The element‑wise product ``mask * gate * projection`` is fused in a
  tiny Triton kernel (`fused_mul3_kernel`) to reduce the memory traffic.
* The batched matrix multiplication for the contraction is carried out by
  a custom Triton kernel (`batched_matmul_kernel`) which processes every
  (batch, hidden) pair as an independent GEMM.  The kernel tiles the
  product with block sizes that fit comfortably into H100 shared memory
  and uses an FP32 accumulator (via `tl.dot`) for numerical stability.

The implementation follows the same API as the reference PyTorch module:
``custom_kernel((input, mask, weights, config)) → output``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Element‑wise fused multiply three tensors (mask * proj * gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    size,                           # total number of elements (1‑D)
    BLOCK_SIZE: tl.constexpr,       # compile‑time constant
):
    """out = a * b * c   (all tensors are 1‑D contiguous, dtype=fp16)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors (all FP16)."""
    assert a.shape == b.shape == c.shape
    a, b, c = a.contiguous(), b.contiguous(), c.contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 65536                         # enough work per program
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣ Batched GEMM for the core contraction
#     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched FP16 GEMM (A: [B,M,K], B: [B,K,N], C: [B,M,N])."""
    batch_id = tl.program_id(0)          # which (batch, hidden) pair
    pid_m   = tl.program_id(1)           # row tile
    pid_n   = tl.program_id(2)           # col tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K‑dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            A_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + cur_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # B tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            B_ptr
            + batch_id * stride_b_batch
            + cur_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        acc += tl.dot(a, b)   # dot uses FP32 accumulation

    # write back C
    c = acc.to(tl.float16)
    c_ptrs = (
        C_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Public entry point – the Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weight_dict, config_dict)
            - input_tensor : [B, N, N, dim]  (float32, on CUDA)
            - mask_tensor  : [B, N, N]        (float32/bool, on CUDA)
            - weight_dict  : model weights (see skeleton)
            - config_dict  : {"dim": int, "hidden_dim": int,
                              "nomask": bool (optional), "eps": float (optional)}

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Convert model weights to FP16 (layer‑norms, linears, etc.)
    # ------------------------------------------------------------------
    # Keeping a separate dict avoids mutating the caller's tensors.
    w = {k: v.to(torch.float16) for k, v in weights.items()}

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (performed in FP16 for speed)
    # ------------------------------------------------------------------
    # LayerNorm works with half‑precision tensors on recent GPUs.
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                       # FP16, shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) → hidden dim, also FP16
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, w["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x_norm, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 4️⃣ Gating tensors (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, w["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask and gates (fused Triton kernel when a mask is present)
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask: [B,N,N] → [B,N,N,1] → broadcast to hidden dim
        mask_fp16 = mask.to(torch.float16)
        mask_exp = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        left = fused_mul3(left_proj, mask_exp, left_gate)
        right = fused_mul3(right_proj, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction – batched GEMM in FP16 (Tensor‑cores)
    # ------------------------------------------------------------------
    # Permute to shape (B, H, N, N) so each (b,h) pair is a separate matrix.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]  (k,j)

    # Flatten the leading (B,H) dimensions → batch dimension for the kernel
    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N)   # [batch, M=N, K=N]
    right_flat = right_perm.view(batch, N, N) # [batch, K=N, N=N]
    out_flat = torch.empty_like(left_flat)    # will hold the result (FP16)

    # Strides for the flattened tensors (all are contiguous)
    s_a_batch = left_flat.stride(0)
    s_a_m = left_flat.stride(1)
    s_a_k = left_flat.stride(2)

    s_b_batch = right_flat.stride(0)
    s_b_k = right_flat.stride(1)
    s_b_n = right_flat.stride(2)

    s_c_batch = out_flat.stride(0)
    s_c_m = out_flat.stride(1)
    s_c_n = out_flat.stride(2)

    # Tile sizes – these work well for H100 (fit into 64 KB shared memory)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                      # M
        N,                      # N (output cols)
        N,                      # K (reduction dim)
        s_a_batch,
        s_a_m,
        s_a_k,
        s_b_batch,
        s_b_k,
        s_b_n,
        s_c_batch,
        s_c_m,
        s_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # FP16, shape [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate  # element‑wise gate (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to `dim` and cast to original dtype
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, w["to_out.weight"])   # FP16, [B,N,N,dim]
    out_final = out_final.to(inp.dtype)                  # back to FP32 (or original)

    return out_final
