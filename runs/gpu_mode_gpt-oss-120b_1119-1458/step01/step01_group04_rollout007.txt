def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Cast to half for projection
    x_half = x.to(torch.float16)

    # Cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # compute out gate later maybe float32?

    # Projections
    left = F.linear(x_half, left_proj_w)   # shape [B,N,N,hidden]
    right = F.linear(x_half, right_proj_w) # shape [B,N,N,hidden]

    # Gates
    left_gate = torch.sigmoid(F.linear(x_half, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))
    # out_gate keep float32 for later gating
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])) # keep float32

    # Apply gates and mask using Triton kernel (optional)
    # left = left * left_gate
    # right = right * right_gate
    # if not nomask: left *= mask; right *= mask

    # Since multiplication is cheap, we can just do it in PyTorch:
    left = left * left_gate
    right = right * right_gate
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_f
        right = right * mask_f

    # Reshape for batched matmul
    B,N,_ , _ = left.shape
    total_heads = B * hidden_dim
    left_flat = left.permute(0,3,1,2).contiguous().view(total_heads, N, N)    # half
    right_flat = right.permute(0,3,1,2).contiguous().view(total_heads, N, N)   # half

    # Batched GEMM
    out_half = torch.bmm(left_flat, right_flat.transpose(1,2))   # half

    # Convert to FP32
    out = out_half.to(torch.float32)

    # Reshape back
    out = out.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()   # [B,N,N,hidden]

    # To_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Apply out_gate gating via Triton kernel
    # flatten and kernel
    total_elem = out.numel()
    # Ensure contiguous view
    out_view = out.view(-1)
    out_gate_view = out_gate.view(-1)
    BLOCK = 1024
    grid = ( (total_elem + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](out_view, out_gate_view, total_elem, BLOCK=BLOCK)

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass – fast implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Linear projections to the hidden dimension (float32) and compute sigmoid
   gates.  Apply gates (and optional mask) to obtain the “left” and “right”
   tensors.
3. Convert ``left``/``right`` to FP16 and perform a batched matrix‑multiply
   C = A @ Bᵀ for every hidden head using ``torch.bmm``.  The inputs are
   FP16 so the H100 Tensor‑Core path is used; the result is cast back to
   FP32.
4. Reshape the result to ``[B, N, N, hidden]`` and apply a second
   LayerNorm.
5. Multiply the normalized tensor by ``out_gate`` element‑wise with a tiny
   Triton kernel (the only required Triton component).
6. Final linear projection back to the original channel dimension.

All heavy arithmetic is performed in mixed‑precision (FP16
mul‑add/FP32 accumulation) and the Triton kernel is only a
straight‑forward element‑wise multiplication, keeping the implementation
compact while achieving sub‑millisecond runtimes on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton tiny kernel: out = out * gate  (both FP32)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    out_ptr,          # *fp32
    gate_ptr,         # *fp32
    total_elems,      # i64 scalar
    BLOCK: tl.constexpr,  # compile‑time block size
):
    pid = tl.program_id(0)
    base = pid * BLOCK
    offs = base + tl.arange(0, BLOCK)
    mask = offs < total_elems

    o = tl.load(out_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, o * g, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:        [B, N, N] (bool or float)
            - weights: dictionary with all linear / LN weights
            - config:   contains "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. First LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # --------------------------------------------------------------
    # 2. Linear projections + gates (keep in FP32 up to this point)
    # --------------------------------------------------------------
    # Projections
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden]

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

    # Optional mask (broadcast over hidden)
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] and cast to same dtype as left/right
        mask_f = mask.unsqueeze(-1).to(left.dtype)       # FP32 -> broadcast
        left = left * mask_f
        right = right * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3. Batched GEMM: C = left @ rightᵀ  (use FP16 Tensor‑Core path)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # Convert to FP16 for the matmul (Tensor‑Core friendly)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Shape for batched matmul: (total_heads, N, N)
    left_flat = left_fp16.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Batched GEMM (A @ Bᵀ)
    out_fp16 = torch.bmm(left_flat, right_flat.transpose(1, 2))  # FP16 result

    # Cast back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # --------------------------------------------------------------
    # 4. Reshape back to [B, N, N, hidden]
    # --------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5. Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    # --------------------------------------------------------------
    # 6. Multiply by out_gate (element‑wise, Triton kernel)
    # --------------------------------------------------------------
    # out_gate is computed from the original normalized tensor (float32)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden]

    # Flatten tensors for the Triton kernel
    total_elements = out.numel()
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)

    # Triton launch
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elements,
        BLOCK=BLOCK,
    )
    # The kernel writes the gated result back into `out`.

    # --------------------------------------------------------------
    # 7. Final projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # FP32 → [B, N, N, dim]

    return out
