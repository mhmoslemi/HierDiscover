"""
TriMul (outgoing) – Triton‑accelerated implementation
=====================================================

The forward pass consists of four stages:

1️⃣ **LayerNorm + fused linear projection**  
   The four linear layers (left_proj, right_proj, left_gate,
   right_gate) and the output gate are fused into a *single*
   matrix‑multiplication `x @ W` (where `W` concatenates the five
   weight matrices).  This reduces the number of large GEMM
   launches from five to one and therefore cuts the memory traffic
   of the huge tensor `x` (shape `[B,N,N,dim]`) by a factor of 5.
   All operations are performed in half‑precision (fp16) to profit
   from Tensor‑Core throughput while the final result is cast back
   to fp32.

2️⃣ **Masked gating**  
   The projected tensors are multiplied by their sigmoid‑gates
   and (optionally) by the pair‑wise mask.  These are tiny
   element‑wise kernels that stay in fp16.

3️⃣ **Heavy pair‑wise product**  
   ``out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]``    
   is an independent batched matrix‑multiplication per hidden
   channel (the *heads*).  The Triton kernel computes
   ``C = A @ Bᵀ`` with tiling parameters `BLOCK_M`, `BLOCK_N`,
   `BLOCK_K`.  The kernel works on a *contiguous* layout
   ``(head, i, k)`` / ``(head, j, k)`` that is built by a cheap
   ``permute``+``view`` copy (the copy is far cheaper than the
   O(N³) compute).  Inputs are loaded as fp16, accumulated in
   fp32 and written back as fp32.

4️⃣ **Post‑processing**  
   The result is reshaped to `[B,N,N,hidden]`, a second
   LayerNorm + output‑gate are applied, and a final linear layer
   projects back to the original feature dimension `dim`.
   The whole pipeline returns a `float32` tensor of shape
   `[B, N, N, dim]`.

The implementation is fully self‑contained – only the
`custom_kernel` entry point is required by the evaluator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: out[i,j] = sum_k left[i,k] * right[j,k] (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16  * left = (head, i, k)
    right_ptr,         # fp16  * right = (head, j, k)
    out_ptr,           # fp32  * out   = (head, i, j)
    N,                 # i32   sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left  (head, i, k)
    stride_rh, stride_rj, stride_rk,   # i64 strides for right (head, j, k)
    stride_oh, stride_oi, stride_oj,   # i64 strides for out   (head, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_h = tl.program_id(0)   # which head (B * hidden_dim)
    pid_m = tl.program_id(1)   # block row (i)
    pid_n = tl.program_id(2)   # block col (j)

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current head
    # ------------------------------------------------------------------
    left_head_ptr  = left_ptr  + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr   = out_ptr   + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left tile [M, K] --------
        left_ptrs = left_head_ptr \
            + offs_i[:, None] * stride_li \
            + offs_k[None, :] * stride_lk
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # -------- right tile (transposed) [K, N] --------
        # right is stored as (head, j, k).  We need rows=k, cols=j.
        right_ptrs = right_head_ptr \
            + offs_k[:, None] * stride_rk \
            + offs_j[None, :] * stride_rj
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # -------- dot product --------
        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_head_ptr \
        + offs_i[:, None] * stride_oi \
        + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Triton‑backed TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Args:
        data = (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32)
            - mask         : [B, N, N]       (bool/float, optional)
            - weights      : dict with all learnable parameters
            - config       : {"dim": int, "hidden_dim": int,
                              "nomask": bool (optional)}

    Returns:
        Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm + fused linear projection (all five mats at once)
    # ------------------------------------------------------------------
    # Normalization (float32) – cheap compared to the following matmul
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # cast once to fp16 – all subsequent large matmuls use Tensor‑Cores
    x_h = x.half()

    # concatenate the five weight matrices (each (hidden, dim))
    # order: left_proj, right_proj, left_gate, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                                 # (5*hidden, dim)

    # single huge linear – shape [B,N,N,5*hidden]
    combined_out = F.linear(x_h, combined_weight)   # fp16

    # split back
    splits = torch.split(combined_out, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = splits

    # --------------------------------------------------------------
    # 2️⃣ Gating + optional mask (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw).half()
    right_gate = torch.sigmoid(right_gate_raw).half()
    # out_gate will be used later after the second LayerNorm → keep fp32
    out_gate = torch.sigmoid(out_gate_raw).float()

    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x_h.dtype)   # [B,N,N,1] fp16
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate    # [B,N,N,hidden] fp16
    right = right_proj * right_gate # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Heavy pair‑wise multiplication (Triton GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # Rearrange to contiguous (head, i, k) / (head, j, k)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (fp32)
    out_flat = torch.empty(
        (total_heads, N, N), dtype=torch.float32, device=device
    )

    # ------------------------------------------------------------------
    # Launch configuration – adapt tile size to N
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),  # note swap: (head,j,k)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp32, shape [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 4️⃣ Post‑processing (LayerNorm → out‑gate → final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                          # fp32

    # broadcast out_gate over the hidden dimension (same shape already)
    out = out * out_gate

    # final projection back to `dim` (use fp16 weight to stay in Tensor‑Core)
    out = F.linear(out, weights["to_out.weight"].half()).float()

    return out
