```python
"""
Implementation of the outgoing TriMul module (AlphaFold3). 
The computationally intensive N³ contraction is performed as a batch of
matrix‑multiplications using torch.bmm (cuBLAS Tensor‑Core kernels), while
all other element‑wise operations use fused half‑precision kernels for
maximum throughput. The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------------------
# Optional minimal Triton kernel (used for mask&gate fusion in this version)
# -----------------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    x_ptr,           # const half *
    gate_ptr,        # const half *
    mask_ptr,        # const half *
    out_ptr,         # half *
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Compute out = x * gate * mask (elementwise) on a flattened view.
    This kernel simply demonstrates usage of Triton; the main contraction is
    performed by an efficient cuBLAS batched GEMM (torch.bmm).
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def apply_mask_gate(x: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Fuse element‑wise mask and gate using a tiny Triton kernel."""
    total = x.numel()
    out = torch.empty_like(x)
    BLOCK = 65536  # 64k elements per kernel launch
    grid = (tl.cdiv(total, BLOCK),)
    fuse_mask_gate_kernel[grid](
        x, gate, mask, out,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=2,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : model weight tensors
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # helpers
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # load weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = to_half(weights["left_proj.weight"])
    right_proj_w = to_half(weights["right_proj.weight"])
    left_gate_w = to_half(weights["left_gate.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim] fp32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy computation
    # ------------------------------------------------------------------
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_half, left_proj_w)   # [B,N,N,hidden_dim] fp16
    right = F.linear(x_half, right_proj_w) # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 4. Mask (if supplied) – broadcast on the hidden dimension
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_half, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))
    # fuse mask/gate with Triton (optional – illustrates custom kernel usage)
    # left = apply_mask_gate(left, left_gate, torch.ones_like(left_gate))  # mask already applied
    # right = apply_mask_gate(right, right_gate, torch.ones_like(right_gate))
    left = left * left_gate
    right = right * right_gate
    del left_gate, right_gate  # free memory early

    # ------------------------------------------------------------------
    # 6. N³ contraction – batched GEMM via torch.bmm (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for batched matmul
    left_view = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_view.view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_flat = right_view.view(B * hidden_dim, N, N) # (B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 8. Output gate and final projection back to original dim
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))   # [B,N,N,hidden_dim] fp16
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)                        # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # Return in float32 as required by the spec
    # ------------------------------------------------------------------
    return final.to(torch.float32)
```

"""
Outgoing TriMul (AlphaFold‑3 style) – highly‑optimized forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32 for numerical stability).
2. Cast to float16 and apply two linear projections (left/right) without bias.
3. Optional pair‑wise mask is broadcast and applied in fp16.
4. Compute two sigmoid gates (left‑gate / right‑gate) from the same input and
   multiply them element‑wise into the projected tensors.
5. The N³ contraction  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   is performed as a *batched* GEMM using torch.bmm (cuBLAS Tensor‑Core
   kernels).  The tensors are reshaped to (B·H,N,N) so that each (batch,head)
   pair executes a standard matrix multiplication.
6. Layer‑norm over the hidden dimension H.
7. An output gate (sigmoid of a third linear projection) is applied.
8. Final linear projection back to the original channel dimension C.
9. Result is returned in float32.

A tiny Triton kernel (`fuse_mask_gate_kernel`) is provided to demonstrate
custom‑kernel usage; the compute‑intensive contraction uses the much faster
cuBLAS batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Minimal Triton kernel (illustrates custom‑kernel use; not on the critical path)
# -------------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    x_ptr,          # const half *
    gate_ptr,       # const half *
    mask_ptr,       # const half *
    out_ptr,        # half *
    total_elts: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """out = x * gate * mask  (element‑wise on a flattened view)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dictionary with the model parameters as torch.Tensors
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move a weight to fp16 on the correct device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy lifting
    # ------------------------------------------------------------------
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_half, left_proj_w)   # [B, N, N, hidden_dim] (fp16)
    right = F.linear(x_half, right_proj_w)  # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_half, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))  # [B,N,N,hidden_dim]
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction – batched GEMM via cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for the batched matmul
    left_view  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left_view.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right_view.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # batched matrix multiply: left @ rightᵀ for each (batch,head)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 8️⃣ Output gate and final projection back to the original dim
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))  # [B,N,N,hidden_dim] (fp16)
    gated = out_norm * out_gate

    # final linear projection (hidden_dim -> dim)
    final = F.linear(gated, to_out_w)  # [B,N,N,dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
