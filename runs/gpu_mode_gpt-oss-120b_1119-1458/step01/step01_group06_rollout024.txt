"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ [B,N,N,dim] (float32).
2. Fuse the four linear projections (left_proj, left_gate, right_proj, right_gate)
   into a single matrix‑multiply:
        fused = x @ W_fused.T      # W_fused shape (4·hidden, dim)
   Split the result into the four components, apply sigmoid to the gate
   halves and multiply:  left = left_proj * sigmoid(left_gate) ,
   right = right_proj * sigmoid(right_gate).
   If a mask is present, multiply left/right by the mask (broadcast on the
   hidden dimension).
3. Compute the heavy N³ reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton kernel that treats each hidden channel independently.
   The kernel reads the tensors in their native layout (B,N,N,hidden) – no
   costly permute/contiguous copies – and it can optionally apply the mask
   while loading.
4. Cast the result to float32, apply the second layer‑norm (to_out_norm),
   multiply by the out‑gate (computed from x) and finish with a final linear
   projection to the original dimension.
5. Return the float32 tensor of shape [B, N, N, dim].

The implementation reduces the number of dense GEMMs from four (two proj +
two gate) to a single fused GEMM, avoids the expensive (B,N,N,hidden) →
(B,hidden,N,N) copy, and fuses masking into the Triton reduction.
"""

import torch
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched Gram product (left @ rightᵀ) over the sequence dim.
# Shape (H, N, N) per batch, with optional mask.
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_gram_kernel(
    left_ptr, right_ptr, out_ptr, mask_ptr,          # memory pointers
    N,                                              # sequence length
    stride_li, stride_lk, stride_lh,                # left strides (i, k, h)
    stride_ri, stride_rk, stride_rh,                # right strides (i, k, h)
    stride_oi, stride_oj, stride_oh,                # out strides   (i, j, h)
    stride_mi, stride_mk,                           # mask strides  (i, k)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr                           # 0 = no mask, 1 = use mask
):
    # Program IDs
    h = tl.program_id(0)          # hidden channel index
    pid_i = tl.program_id(1)      # tile row (i)
    pid_j = tl.program_id(2)      # tile col (j)

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k, h]
        left = tl.load(
            left_ptr
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
            + h * stride_lh,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # right[k, j, h]  (note the transpose on the second dim)
        right = tl.load(
            right_ptr
            + offs_k[:, None] * stride_ri
            + offs_j[None, :] * stride_rk
            + h * stride_rh,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )

        if HAS_MASK:
            # mask left[i, k]
            mask_left = tl.load(
                mask_ptr
                + offs_i[:, None] * stride_mi
                + offs_k[None, :] * stride_mk,
                mask=mask_i[:, None] & mask_k[None, :],
                other=1.0
            )
            left = left * mask_left

            # mask right[k, j]
            mask_right = tl.load(
                mask_ptr
                + offs_k[:, None] * stride_mi
                + offs_j[None, :] * stride_mk,
                mask=mask_k[:, None] & mask_j[None, :],
                other=1.0
            )
            right = right * mask_right

        # FP16 dot, accumulate in FP32
        acc += tl.dot(left, right).to(tl.float32)

    # Store FP16 result
    tl.store(
        out_ptr
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (float32 or bool) – may be None
            - weights      : dict of tensors with keys
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight"
                * "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config       : dict with entries
                * "dim" (int), "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Fuse 4 linear projections (proj + gate) into one matmul
    # -----------------------------------------------------------------
    # Concatenate weights: [left_proj; left_gate; right_proj; right_gate]
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    )  # shape (4*hidden_dim, dim)

    # Convert to half for the matmul
    x_fp16 = x.to(torch.float16)
    fused_weight_fp16 = fused_weight.to(torch.float16)

    fused = torch.nn.functional.linear(x_fp16, fused_weight_fp16)  # (B,N,N,4*H)

    # -----------------------------------------------------------------
    # 3. Split, apply sigmoids and masks
    # -----------------------------------------------------------------
    # Slicing along the last dimension
    lp = fused[..., :hidden_dim]                           # left projection
    lg_raw = fused[..., hidden_dim:2 * hidden_dim]         # left gate pre‑sigmoid
    rp = fused[..., 2 * hidden_dim:3 * hidden_dim]         # right projection
    rg_raw = fused[..., 3 * hidden_dim:]                   # right gate pre‑sigmoid

    lg = lg_raw.sigmoid()
    rg = rg_raw.sigmoid()

    left = lp * lg    # shape (B,N,N,hidden_dim), FP16
    right = rp * rg   # shape (B,N,N,hidden_dim), FP16

    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(torch.float16)      # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Compute out_gate (independent linear + sigmoid)
    # -----------------------------------------------------------------
    out_gate = torch.nn.functional.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()                                            # FP16
    out_gate = out_gate.to(torch.float32)                  # will be used in FP32 later

    # -----------------------------------------------------------------
    # 5. Triton Gram product
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    # output buffer (FP16)
    out_buf = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left.device)

    # Block size (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides for the per‑batch tensors (shape N,N,hidden_dim)
    stride_li, stride_lk, stride_lh = left.stride(0), left.stride(1), left.stride(2)
    stride_ri, stride_rk, stride_rh = right.stride(0), right.stride(1), right.stride(2)
    stride_oi, stride_oj, stride_oh = out_buf.stride(0), out_buf.stride(1), out_buf.stride(2)

    # Mask strides (if needed)
    if not nomask and mask is not None:
        mask = mask.to(torch.float16).contiguous()
        stride_mi, stride_mk = mask.stride(0), mask.stride(1)
    else:
        # dummy values – they will never be dereferenced when HAS_MASK==0
        stride_mi = stride_mk = 1
        mask = left  # any tensor works; kernel won't read it when HAS_MASK==0

    grid = (
        hidden_dim,                         # hidden channel dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑tiles
    )

    # Launch one kernel per batch to avoid runtime division on hidden index
    for b in range(B):
        left_b = left[b]    # shape (N, N, hidden_dim)
        right_b = right[b]
        out_b = out_buf[b]
        mask_b = mask[b] if not nomask and mask is not None else mask

        tri_mul_gram_kernel[grid](
            left_b,
            right_b,
            out_b,
            mask_b,
            N,
            stride_li, stride_lk, stride_lh,
            stride_ri, stride_rk, stride_rh,
            stride_oi, stride_oj, stride_oh,
            stride_mi, stride_mk,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            HAS_MASK=int(not nomask and mask is not None),
        )

    # -----------------------------------------------------------------
    # 6. Convert to FP32 for the remaining ops
    # -----------------------------------------------------------------
    out = out_buf.to(torch.float32)

    # -----------------------------------------------------------------
    # 7. Hidden → output LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
