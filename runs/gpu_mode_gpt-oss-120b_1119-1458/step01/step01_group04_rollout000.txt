TriMul (outgoing) – a highly‑optimized Triton implementation

The module implements the forward pass of the “outgoing” Triangle Multiplicative
Update (TriMul) from AlphaFold‑3.  The computational bottleneck is the
einstein sum

    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

which is equivalent to a batch of independent matrix‑multiplications
C = A @ Bᵀ for every hidden channel *d* of every batch element.  In the
kernel each channel (the product of batch size *B* and hidden dimension
*H*) is mapped to a separate Triton program (pid‑0).  Within a program a
2‑D tiling over the (i, j) output plane is performed (pid‑1, pid‑2) and the
inner reduction over *k* is tiled with a small BLOCK_K.  All arithmetic
is carried out in float16 with an fp32 accumulator (Tensor‑Core friendly)
and the final result is stored as fp32.

All surrounding LayerNorm, linear projections and gating are performed in
PyTorch (mixed‑precision) – only the O(N³·H·B) work is off‑loaded to Triton.
The implementation works for any batch size ≤ 2, any sequence length N up to
1024, and hidden dimension 128 (as used in the reference models).  The final
output tensor has shape (B, N, N, dim) and dtype float32.
"""

# Computes  out_slice = left_slice @ right_sliceᵀ
#   left_slice : (N, N)   – indexed as [i, k]
#   right_slice: (N, N)   – indexed as [j, k] (k is the reduction dim)
# ----------------------------------------------------------------------
@triton.jit

    left_ptr,         # *fp16  left  (shape: [heads, N, N])
    right_ptr,        # *fp16  right (shape: [heads, N, N])
    out_ptr,          # *fp32  output (shape: [heads, N, N])

    stride_rh, stride_ri, stride_rk,   # i64 strides for right
    stride_oh, stride_oi, stride_oj,   # i64 strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    # Program IDs (head, i‑tile, j‑tile)
    # ------------------------------------------------------------------
    pid_h = tl.program_id(0)            # which head (B * hidden)
    pid_m = tl.program_id(1)            # block row in i dimension
    pid_n = tl.program_id(2)            # block col in j dimension

    # ------------------------------------------------------------------
    # Offsets for the i / j tile (including OOB mask)
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = j0 + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for the current head
    # ------------------------------------------------------------------
    left_head_ptr  = left_ptr  + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr   = out_ptr   + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in BLOCK_K chunks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k0 = kb * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- load left tile  [M, K] ----------
        left_ptrs = (
            left_head_ptr
            + offs_i[:, None] * stride_li     # i dimension
            + offs_k[None, :] * stride_lk     # k dimension
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        # fp16 → fp32 for accumulation
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # ---------- load right tile transposed to [K, N] ----------
        # right stored as (head, j, k).  We want rows = k, cols = j.
        right_ptrs = (
            right_head_ptr
            + offs_k[:, None] * stride_rk     # k dimension (row)
            + offs_j[None, :] * stride_ri     # j dimension (col)
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ---------- inner product ----------
        # (M,K) @ (K,N)  -> (M,N)
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Store the computed tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_head_ptr
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Entry point – called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N]       (bool/float)
        weights      : dict of model parameters (LayerNorm, Linear, …)
        config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------ unpack ------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # -------------------------- LayerNorm -----------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # -------------------------- Linear proj ---------------------------
    # Cast weights to fp16 once (saves a few cycles)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # Mixed‑precision matmul: (B*N*N, dim) @ (dim, hidden) → (B,N,N,hidden)
    # Using torch.nn.functional.linear which internally calls a GEMM.
    with torch.cuda.amp.autocast(enabled=True):
        left = F.linear(x, left_proj_w)   # fp16 output
        right = F.linear(x, right_proj_w) # fp16 output

    # --------------------------- optional mask -------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden dim
        mask_ = mask.unsqueeze(-1).to(left.dtype)      # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -------------------------- gates (sigmoid) -----------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep fp32 for later

    # -------------------------- apply gates ---------------------------
    left = left * left_gate   # fp16 * fp16 → fp16
    right = right * right_gate

    # -------------------------- prepare for Triton --------------------
    B, N, _, _ = left.shape                # N is the sequence length
    total_heads = B * hidden_dim           # = batch * hidden_dim

    # reshape to (total_heads, N, N) with layout (head, i, k) / (head, j, k)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # output buffer (fp32 accumulator)
    out_flat = torch.empty(
        (total_heads, N, N), dtype=torch.float32, device=device
    )

    # ----------------------- Triton launch config --------------------
    # Empirically good block sizes for H100: 128×128 tiles, K‑tile = 64.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------ reshape back ---------------------------
    # out_flat shape: (total_heads, N, N) → (B, hidden, N, N) → (B, N, N, hidden)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]  fp32

    # ---------------------- post‑processing -------------------------
    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Gating
    out = out * out_gate  # broadcast over hidden dim

    # Final linear projection back to `dim`
    # Using fp16 weight for speed, accumulate in fp32
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # result is fp16 → cast to fp32

    return out.to(torch.float32)
