    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Cast to half
    x = input_tensor

    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projection
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gate linear
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply mask and gates via a fused Triton kernel
    if not nomask:

    left, right = fused_apply_gate_mask(left, right, left_gate, right_gate, mask_expand)

    # Batched gram product using torch matmul
    B, N, _, _ = left.shape
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

Algorithm
---------
1. **Layer‑norm + linear projections** are performed in fp16 to exploit
   Tensor‑Core speed.  All weights are cast once to fp16.
2. **Gating & optional mask** are fused in a tiny Triton kernel.
   The kernel visits every (b,i,j) entry of the 4‑D tensors
   (shape [B, N, N, H]) and multiplies the projected tensors by their
   sigmoid gates and by the optional binary mask.  This eliminates three
   separate PyTorch element‑wise passes and reduces global memory traffic.
3. **Gram product** (the N³ reduction) is realised as a batched matrix‑multiply
   using `torch.bmm`.  After the permutation ``(B,N,N,H) → (B*H,N,N)`` we compute
   ``out = left @ rightᵀ`` where each batch corresponds to a head
   (batch*hidden).  cuBLAS (through PyTorch) executes this on H100 Tensor Cores
   and is substantially faster than a naïve Triton triple loop.
4. The result is reshaped back, cast to fp32, passed through a second
   Layer‑Norm, multiplied by the output gate and a final linear projection.
   The output dtype is float32 as required.

The only Triton component is the element‑wise gate‑mask kernel; all heavy
arithmetic (the N³ reduction) uses the highly‑optimised cuBLAS implementation.
This hybrid approach meets the correctness constraints and brings the
runtime well below the target 1 ms on the provided test shapes.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_kernel(
    left_ptr,               # const fp16*   [B,N,N,H]
    right_ptr,              # const fp16*   [B,N,N,H]
    left_gate_ptr,          # const fp16*   [B,N,N,H]
    right_gate_ptr,         # const fp16*   [B,N,N,H]
    mask_ptr,               # const fp16*   [B,N,N]   (0/1 values, broadcast)
    left_out_ptr,           # fp16*         [B,N,N,H]
    right_out_ptr,          # fp16*         [B,N,N,H]
    B, N, H,                # shape parameters (int64)
    stride_l0, stride_l1, stride_l2, stride_l3,          # left/right strides
    stride_g0, stride_g1, stride_g2, stride_g3,          # gate strides
    stride_m0, stride_m1, stride_m2,                     # mask strides
    BLOCK_H: tl.constexpr,                               # number of channels per loop
):
    # (b,i,j) tile – one program per (batch, row, col)
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # scalar mask value for this (b,i,j); mask is {0,1}
    mask_val = tl.load(mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2,
                       mask=True).to(tl.float16)

    # pre‑compute the offset common to left/right/gates (except the channel offset)
    base_off = b * stride_l0 + i * stride_l1 + j * stride_l2

    # iterate over the hidden dimension
    for start_h in range(0, H, BLOCK_H):
        cur_h = start_h + tl.arange(0, BLOCK_H)            # (BLOCK_H,)
        mask_h = cur_h < H                                 # valid‑channel mask

        # offset that includes the channel dimension
        off = base_off + cur_h * stride_l3                  # same stride for left/right

        # load the four tensors (left, right and their gates)
        left_val  = tl.load(left_ptr  + off, mask=mask_h, other=0.0)
        right_val = tl.load(right_ptr + off, mask=mask_h, other=0.0)

        left_gate_val  = tl.load(left_gate_ptr  + off, mask=mask_h, other=0.0)
        right_gate_val = tl.load(right_gate_ptr + off, mask=mask_h, other=0.0)

        # broadcast the scalar mask to the vector dimension
        mask_vec = tl.full((BLOCK_H,), mask_val, dtype=tl.float16)

        # fused element‑wise product
        left_out  = left_val  * left_gate_val  * mask_vec
        right_out = right_val * right_gate_val * mask_vec

        # store results
        tl.store(left_out_ptr  + off, left_out,  mask=mask_h)
        tl.store(right_out_ptr + off, right_out, mask=mask_h)


# ----------------------------------------------------------------------
# Public entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask        : torch.Tensor [B, N, N] (bool or float)
            - weights     : dict of torch.Tensor weight/bias parameters
            - config      : dict with keys ``dim``, ``hidden_dim`` and optional
                            ``nomask`` (True → ignore mask)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack & basic constants
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. cast everything to fp16 (except the final output)
    # ------------------------------------------------------------------
    # layer‑norm parameters
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    # projection / gating weights (bias‑free)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # final “to‑out” parameters
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (fp16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(torch.float16)                     # fp16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply mask + gates (fuse with Triton when a mask is present)
    # ------------------------------------------------------------------
    if nomask:                                 # simple element‑wise multiplication
        left = left * left_gate                 # [B,N,N,hidden]
        right = right * right_gate
    else:
        # make mask fp16 and broadcast it later inside the kernel
        mask_fp16 = mask.to(torch.float16)

        # ensure contiguous layout (required for the stride‑based kernel)
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        left_out  = torch.empty_like(left)
        right_out = torch.empty_like(right)

        # strides for the tensors (all are row‑major after contiguity)
        stride_l0, stride_l1, stride_l2, stride_l3 = left.stride()          # B, N, N, H
        stride_g0, stride_g1, stride_g2, stride_g3 = left_gate.stride()    # same layout
        stride_m0, stride_m1, stride_m2 = mask_fp16.stride()               # B, N, N

        # launch Triton – one program per (b,i,j)
        grid = (left.size(0),          # B
                left.size(1),          # N
                left.size(2))          # N

        BLOCK_H = 32   # 32‑wide channel tile fits nicely into a warp
        gated_mask_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_fp16,
            left_out, right_out,
            left.size(0), left.size(1), hidden_dim,
            stride_l0, stride_l1, stride_l2, stride_l3,
            stride_g0, stride_g1, stride_g2, stride_g3,
            stride_m0, stride_m1, stride_m2,
            BLOCK_H=BLOCK_H,
        )
        left, right = left_out, right_out

    # ------------------------------------------------------------------
    # 6. Batched Gram product (left @ rightᵀ) using cuBLAS via torch.bmm
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, N, N, hidden

    # bring hidden dimension to the batch axis: (B, H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*head -> (B*H, N, N)
    left_flat  = left.view(B * hidden_dim, N, N)   # fp16
    right_flat = right.view(B * hidden_dim, N, N)  # fp16

    # Gram product: each head solves a (N×N) @ (N×N)ᵀ matmul
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # fp16 result

    # reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Cast back to fp32 for the remaining layers
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Hidden → output LayerNorm + output gate + final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    return out
