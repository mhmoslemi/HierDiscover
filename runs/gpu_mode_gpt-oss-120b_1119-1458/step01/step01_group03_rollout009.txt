def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # 1. LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )

    # 2. Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))

    # 4. Mask (if any)
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype) # float
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 5. Apply gates and cast to half
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    # 6. Prepare for batched matrix multiply
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # 7. Batched GEMM using torch.bmm
    out_h = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_h.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # 8. Output layernorm
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 9. Out gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # 10. Fuse gating via Triton elementwise kernel
    out_gated = torch.empty_like(out_norm)
    total_elements = out_norm.numel()
    BLOCK_SIZE = 1024
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    mul_gate_kernel[grid](
        out_norm,
        out_gate,
        out_gated,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4
    )

    # 11. Final linear projection
    out_final = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return out_final

"""
TriMul (outgoing) forward pass – fast Triton/PyTorch implementation
------------------------------------------------------------------
The heavy tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix multiply: for every hidden channel *batch*
`out_h = left_h @ right_h.T` with `left_h, right_h ∈ ℝ^{N×N}`.
We let cuBLAS (torch.bmm) perform this GEMM because it already
exploits Tensor‑Cores on an H100.  The remaining element‑wise
operations (masking, gating and the final out‑gate multiplication)
are fused in a tiny Triton kernel to guarantee that at least part of
the work runs on a custom kernel (as required by the task).

Steps
-----
1. Input LayerNorm (float32).
2. Linear projections → left / right (float32).
3. Gates (sigmoid of linear(x)).
4. Optional mask is applied before gating.
5. Multiply by gates and cast to float16 for the contraction.
6. Batched GEMM with `torch.bmm` (half‑precision, fp32 accumulation).
7. Output LayerNorm (float32).
8. Compute `out_gate = sigmoid(linear(x))`.
9. Fuse `out_norm * out_gate` with a Triton element‑wise kernel.
10. Final linear projection back to the original channel dimension.

The whole pipeline stays on‑device, uses mixed precision for the
O(N³) work, and respects the original TriMul semantics.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: element‑wise multiplication (out_norm * out_gate)
# --------------------------------------------------------------
@triton.jit
def mul_gate_kernel(
    a_ptr,               # float*  – out_norm (contiguous 1‑D)
    b_ptr,               # float*  – out_gate (contiguous 1‑D)
    c_ptr,               # float*  – output (contiguous 1‑D)
    N,                   # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


# --------------------------------------------------------------
# Entry point: custom Triton‑accelerated TriMul (outgoing) forward
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the channel dimension (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – still float32
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)        # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5. Apply gates and cast to FP16 for the O(N³) contraction
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)   # [B, N, N, hidden_dim]
    right = (right_proj * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the batched GEMM:
    #    treat (batch * hidden_dim) as a single batch dimension.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                               # combined batch size
    left_flat = left.view(BH, N, N)                   # [BH, N, N]  (fp16)
    right_flat = right.view(BH, N, N)                 # [BH, N, N]  (fp16)

    # ------------------------------------------------------------------
    # 7. Batched matrix multiplication using cuBLAS (torch.bmm)
    #    The operation is performed in FP16 with FP32 accumulation.
    # ------------------------------------------------------------------
    out_fp16 = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_fp16.to(torch.float32)                 # back to fp32

    # reshape to the original 4‑D layout
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 9. Compute out‑gate (sigmoid) – same as in the reference model
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 10. Fuse out_norm * out_gate with a tiny Triton kernel
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.contiguous().view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)
    out_gated_flat = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK_SIZE = 1024                     # 1 K elements per program
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    mul_gate_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        out_gated_flat,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    out_gated = out_gated_flat.view_as(out_norm)   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 11. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = torch.nn.functional.linear(
        out_gated, weights["to_out.weight"]
    )  # [B, N, N, dim] – float32

    return out_final
