"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm over the input feature dimension `dim`.
2. Two linear projections (left/right) to `hidden_dim`.  The projections are
   executed in half‑precision (float16) under `torch.autocast` to cut memory
   bandwidth.
3. Optional binary mask (broadcasted to the hidden dim) applied to the
   projected tensors.
4. Three sigmoid gates (left, right, out) computed from the same normalized
   tensor and applied element‑wise to the projections (still in FP16).
5. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   For each hidden channel `h` this is a matrix multiplication
   `C = A @ Bᵀ` with `A = left[..., :, :, h]` and `B = right[..., :, :, h]`.
   A custom Triton kernel performs the batched GEMM with
   BLOCK_M=N_BLOCK=128, BLOCK_N=128, BLOCK_K=32 and accumulates in FP32.
   The kernel loads the B operand transposed on‑the‑fly, thus no extra
   preprocessing of `right` is required.
6. Layer‑norm over the hidden dimension.
7. Multiply by the output gate and project back to `dim` with a final linear
   layer (float32).

Only step 5 is Triton‑implemented; the rest uses high‑performance PyTorch
primitives.  The implementation supports both masked and unmasked inputs and
a variety of sequence lengths (up to 1024) and hidden dimensions (up to 384).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ for many independent batches (batch * hidden)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16 [M, N, K]   (M = batch*hidden, N = seq_len, K = seq_len)
    B_ptr,               # *fp16 [M, N, K]
    C_ptr,               # *fp32 [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ.
    A : (M, N, K)   – rows i, columns k
    B : (M, N, K)   – rows j, columns k (will be accessed as (k, j))
    C : (M, N, N)   – output rows i, columns j
    """
    pid_m = tl.program_id(0)          # tile index over i (rows)
    pid_n = tl.program_id(1)          # tile index over j (cols)
    pid_z = tl.program_id(2)          # batch*hidden index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ------------------------------------------------------------------
        # Load A tile (i × k)
        # ------------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
            dtype=tl.float16,
        )
        # ------------------------------------------------------------------
        # Load B tile transposed (k × j)  – note the swapped axes
        # ------------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k dimension (row)
            + offs_j[None, :] * stride_bj   # j dimension (col)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Convert to FP32 and accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ------------------------------------------------------------------
    # Write back the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper: batched GEMM (A @ Bᵀ) with FP16 inputs and FP32 accumulation
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the heavy contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.
    Input tensors:
        left, right – [B, N, N, H] (dtype = float16)
    Output:
        out – [B, N, N, H] (dtype = float32)
    """
    B, N, _, H = left.shape

    # Move the hidden dimension next to batch for a clean view:
    # A : (B*H, N, N)  – rows i, cols k
    # B : (B*H, N, N)  – rows j, cols k  (will be read transposed)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer – FP32 accumulation, final dtype required by later layers
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # Strides (in elements)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # ------------------------------------------------------------------
    # Tile sizes – tuned for H100, FP16 inputs, FP32 accumulation
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # H100 performs well with 8 warps per block
        num_stages=2,     # double‑buffering
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, dim]
        torch.Tensor,          # mask tensor [B, N, N] (may be dummy if nomask=True)
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module – no gradients.
    Returns a tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # -------------------------------------------------
    # 2️⃣ Linear projections + gating – run in FP16 with autocast
    # -------------------------------------------------
    with torch.cuda.amp.autocast(enabled=True, dtype=torch.float16):
        # Linear projections (no bias)
        left = torch.nn.functional.linear(x, weights["left_proj.weight"])
        right = torch.nn.functional.linear(x, weights["right_proj.weight"])

        # Optional mask (broadcasted on the hidden dimension)
        if not nomask:
            # mask: [B, N, N] → [B, N, N, 1] → FP16
            mask = mask_tensor.to(left.dtype).unsqueeze(-1)
            left = left * mask
            right = right * mask

        # Gating (sigmoid) – stays in FP16
        left_gate = torch.sigmoid(
            torch.nn.functional.linear(x, weights["left_gate.weight"])
        )
        right_gate = torch.sigmoid(
            torch.nn.functional.linear(x, weights["right_gate.weight"])
        )
        out_gate = torch.sigmoid(
            torch.nn.functional.linear(x, weights["out_gate.weight"])
        )

        # Apply gates
        left = left * left_gate
        right = right * right_gate

    # -------------------------------------------------
    # 3️⃣ Heavy O(N³) contraction via Triton (left/right are FP16)
    # -------------------------------------------------
    # Output of the contraction is FP32
    out = _batched_matmul_transpose(left, right)          # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden dimension (float32)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, hidden_dim]  (float32)

    # -------------------------------------------------
    # 5️⃣ Output gate (upsampled to FP32) and final projection
    # -------------------------------------------------
    # out_gate is FP16; broadcasting to FP32 automatically upcasts
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # back to dim

    return out
