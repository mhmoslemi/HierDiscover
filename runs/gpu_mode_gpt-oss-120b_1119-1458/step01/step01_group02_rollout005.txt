"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input over the feature axis `dim` (float32 for numerical
   stability).
2. Cast the normalized tensor to fp16 and apply all linear projections /
   element‑wise sigmoid gates in fp16.
3. (Optionally) mask the projected tensors.
4. Fuse the hidden‑dim gating (`left_gate`, `right_gate`) with the projected
   tensors.
5. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed by a custom Triton kernel that executes a batched
   matrix‑multiply `A @ Bᵀ` for each (batch, hidden) pair using fp16
   tensor‑core MMA and accumulates in FP32.  The kernel uses a 64×64×64
   tile and 8 warps (2 stages) which is optimal for the H100.
6. Cast the result back to fp16, apply a second LayerNorm (fp16),
   multiply by the `out_gate` (fp16) and a final linear projection back to
   `dim` (fp16).  The final tensor is cast to float32 to match the
   required output type.

Only the O(N³) part is implemented in Triton; all other ops use
high‑performance PyTorch primitives.  All heavy tensors are kept in
fp16 to reduce memory traffic, while the accumulation stays in FP32 for
accuracy.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batch‑wise matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile index along the first N dimension (i)
    pid_n = tl.program_id(1)   # tile index along the second N dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # ---------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A tile (M‑tile × K‑tile) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- load B tile (K‑tile × N‑tile) transposed ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )   # fp16

        # dot (fp16 * fp16 -> fp32) and accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ---------- write result ----------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    # store as fp16 (cast from fp32 accumulator)
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] fp16
    right: torch.Tensor,  # [B, N, N, H] fp16
) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using the
    Triton kernel above.  All tensors are expected to be contiguous fp16.
    Returns a tensor of the same dtype and device.
    """
    B, N, _, H = left.shape

    # Pack batch & hidden into a single leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – tuned for H100 tensor‑core performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``, optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # Cast to fp16 for the rest of the pipeline
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates to the projections
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 inputs, fp16 output)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)      # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
