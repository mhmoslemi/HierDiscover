  5️⃣ Dense N³ contraction:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} ⋅ right_{b,j,k,h}
     Implemented as a batched matrix‑multiply (left @ rightᵀ) for each
     head h.  For N ≤ 256 a custom Triton kernel is used; for larger N we
     fall back to cuBLAS via torch.bmm, which is far faster on an H100.
  6️⃣ Layer‑Norm over the hidden dimension (H).
  7️⃣ Output gate (sigmoid) and final linear projection back to C.
  8️⃣ Return the result in float32.

Only the N³ contraction is written in Triton; all other steps rely on
well‑optimized PyTorch kernels.  The mixed‑precision (float16) path keeps
the heavy compute on Tensor Cores while preserving numerical stability
through float32 layer‑norms and the final cast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for one (batch, head) pair
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,            # const float16 *
    right_ptr,           # const float16 *
    out_ptr,             # float16 *
    B, N, H,             # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul: out = left @ rightᵀ for a single (batch, head)."""
    pid_head = tl.program_id(0)   # encodes (batch, hidden) = b*H + h
    pid_m    = tl.program_id(1)   # block row (i)
    pid_n    = tl.program_id(2)   # block column (j)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (i, k)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # rightᵀ tile: (k, j) – right_ptr already points to the transposed view
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store the result
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_triton(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched matmul for the N³ contraction.
    left  : (B, N, N, H)  float16
    right_t: (B, N, N, H)  float16  (right already transposed on the two N axes)
    Returns out with the same shape and dtype.
    """
    assert left.is_cuda and right_t.is_cuda
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides in element units
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right_t.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # block sizes tuned for H100 – generous registers & 8 warps
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                              # (batch, head) dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # col tiles
    )

    _tri_mul_kernel[grid](
        left, right_t, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rj, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing exactly the tensors used by the
                         reference PyTorch module
        - config_dict  : must contain 'dim' (C) and 'hidden_dim' (H)

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move / cast model parameters
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_w            = weights["norm.weight"]
    norm_b            = weights["norm.bias"]
    left_proj_w       = to_half(weights["left_proj.weight"])
    right_proj_w      = to_half(weights["right_proj.weight"])
    left_gate_w       = to_half(weights["left_gate.weight"])
    right_gate_w      = to_half(weights["right_gate.weight"])
    out_gate_w        = to_half(weights["out_gate.weight"])
    to_out_norm_w     = to_half(weights["to_out_norm.weight"])
    to_out_norm_b     = to_half(weights["to_out_norm.bias"])
    to_out_w          = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )   # [B, N, N, C]  (float32)

    # Cast to half for the heavy lifting
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid on linear projections of the original X)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B, N, N, H]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction (left @ rightᵀ per hidden dimension)
    #    – small N → Triton kernel, otherwise cuBLAS via torch.bmm
    # ------------------------------------------------------------------
    if N <= 256:   # Triton is competitive for modest sizes
        # rightᵀ on the two N axes
        right_t = right.permute(0, 2, 1, 3).contiguous()
        out = _tri_mul_triton(left, right_t)          # [B, N, N, H] (half)
    else:
        # Use the highly‑optimized cuBLAS batched GEMM
        #  shape casting: (B, N, N, H) → (B*H, N, N)
        left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
        right_t    = right.permute(0, 2, 1, 3).contiguous()  # (B, N, N, H) → (B, N, N, H) with (k,j) order
        # pack batch*head dimension
        left_flat  = left_perm.view(B * hidden_dim, N, N)    # (B*H, i, k)
        right_flat = right_t.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H, k, j)
        out_flat   = torch.bmm(left_flat, right_flat)       # (B*H, i, j)
        out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm over the hidden dimension (still half)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # [B, N, N, H] (half)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to C
    # ------------------------------------------------------------------
    gated = out * out_gate                      # element‑wise gate
    final = F.linear(gated, to_out_w)           # [B, N, N, C] (half)

    # ------------------------------------------------------------------
    # Return in float32 as required by the test harness
    # ------------------------------------------------------------------
    return final.to(torch.float32)
