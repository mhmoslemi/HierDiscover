"""
TriMul (outgoing) – fast forward pass.

The heavy N³ contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix‑multiply (left @ rightᵀ) for each
pair of (batch, hidden) using a custom Triton kernel.  All other
operations (LayerNorm, linear projections, gating, etc.) are performed
with PyTorch’s highly‑optimised kernels.  Mixed‑precision (FP16) is used
for the bulk of the work; the final result is returned in FP32.

The kernel tiles the output matrix with
    BLOCK_M × BLOCK_N tiles and iterates over the K‑dimension in
    BLOCK_K‑sized chunks.  Accumulation is performed in FP32 to keep
    numerical error low while letting Tensor‑Cores operate on FP16 data.
    The launch uses 8 warps and 3 shared‑memory stages, which is a good
    trade‑off on NVIDIA H100.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: batched matmul for the N³ contraction
# ----------------------------------------------------------------------

    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matmul: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The kernel processes a single (batch, hidden) pair per program instance.
    """
    pid_head = tl.program_id(0)      # encodes (batch, hidden)
    pid_m    = tl.program_id(1)      # block row index (i)
    pid_n    = tl.program_id(2)      # block col index (j)

    # ------------------------------------------------------------------
    # Decode batch and hidden indices
    # ------------------------------------------------------------------
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # ------------------------------------------------------------------
    # Compute tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   (i, k, h)
        left_ptrs = left_ptr + \
            b * stride_l_b + h * stride_l_h + \
            offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # rightᵀ tile: (k, j, h)  – right was pre‑transposed (k,j)
        right_ptrs = right_ptr + \
            b * stride_r_b + h * stride_r_h + \
            offs_k[:, None] * stride_r_i + offs_n[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate inner product over K
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Write back the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_o_b + h * stride_o_h + \
        offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched matmul for the TriMul contraction.
    left  : (B, N, N, H)  FP16
    right_t: (B, N, N, H)  FP16   (right already transposed on the two
                                   N‑axes, i.e. right_t[b, k, j, h])
    Returns out of shape (B, N, N, H) FP16.
    """
    assert left.is_cuda and right_t.is_cuda, "Triton kernel requires CUDA tensors"
    B, N, _, H = left.shape

    out = torch.empty_like(left)   # FP16 output buffer

    # ------------------------------------------------------------------
    # Strides in element units (as expected by Triton)
    # ------------------------------------------------------------------
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right_t.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # ------------------------------------------------------------------
    # Tile sizes – tuned for H100 (occupancy + Tensor‑Core utilisation)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                      # one program per (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left,
        right_t,
        out,
        B,
        N,
        H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rj, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps → 256 threads per block
        num_stages=3,              # double‑buffered shared memory
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Tensors with all linear / norm weights
        - config_dict  : dict containing at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper – move a weight to half‑precision on the correct device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]          # (C,)
    norm_b = weights["norm.bias"]           # (C,)

    left_proj_w   = to_half(weights["left_proj.weight"])    # (hidden_dim, C)
    right_proj_w  = to_half(weights["right_proj.weight"])   # (hidden_dim, C)

    left_gate_w   = to_half(weights["left_gate.weight"])    # (hidden_dim, C)
    right_gate_w  = to_half(weights["right_gate.weight"])   # (hidden_dim, C)
    out_gate_w    = to_half(weights["out_gate.weight"])     # (hidden_dim, C)

    to_out_norm_w = to_half(weights["to_out_norm.weight"])  # (hidden_dim,)
    to_out_norm_b = to_half(weights["to_out_norm.bias"])    # (hidden_dim,)
    to_out_w      = to_half(weights["to_out.weight"])       # (C, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm – keep in float32 for stability
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C]  FP32

    # Cast to FP16 for the heavy work
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_half, left_proj_w)   # [B, N, N, hidden_dim]  FP16
    right = F.linear(x_half, right_proj_w)  # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; convert to half and add channel dim
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating – sigmoid(linear(x))
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_half, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x_half, out_gate_w))    # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via Triton (batched matmul)
    # ------------------------------------------------------------------
    # right is transposed on the two N‑axes: (b, k, j, h)
    right_t = right.permute(0, 2, 1, 3).contiguous()
    out = tri_mul_triton(left, right_t)           # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim (float32 internally)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                   # FP16
    final = F.linear(gated, to_out_w)             # [B,N,N,C] FP16

    # ------------------------------------------------------------------
    # 8. Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
