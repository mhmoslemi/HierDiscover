"""
Optimized outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference forward pass but moves the
computationally dominant part (the N³ “einsum”) to a highly‑optimized
batched matrix multiplication (`torch.bmm`) that runs on H100 Tensor‑Cores.
All linear projections, gates and the final GEMM use FP16 for bandwidth‑
and compute‑efficiency while the surrounding LayerNorms are kept in FP32
(or FP16 when safe) to preserve numerical stability.

Two tiny Triton kernels are used:
* `apply_gates_kernel` – fuses the element‑wise multiplication of the
  left/right tensors with their respective gates (in‑place).
* `elementwise_mul_kernel` – multiplies the post‑norm tensor with the
  output gate before the final projection.

The overall data flow (with shapes) is:

    x (float32)  ──► LayerNorm ──► cast FP16
    ├─► left_proj  ──► left_gate ──► (gate) ──►
    │                                   │   ──► batched GEMM (A @ Bᵀ)  →  out (FP16)
    └─► right_proj ──► right_gate ──► (gate) ──►

    out ──► LayerNorm (FP16) ──► out_gate (FP16) ──► elementwise_mul_kernel
    ──► final linear (FP16) ──► cast to float32 → result [B, N, N, dim]
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,       # fp16,   shape (total_elems,)
    left_gate_ptr,  # fp16,   same shape as left_ptr
    right_ptr,      # fp16,   shape (total_elems,)
    right_gate_ptr, # fp16,   same shape as right_ptr
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place multiplication:
        left  = left  * left_gate
        right = right * right_gate
    The tensors are flattened beforehand.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # left side
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    # right side
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # fp16 input
    b_ptr,          # fp16 input
    out_ptr,        # fp16 output
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    out = a * b   (both flattened)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim]  (float32)
        - mask          : torch.Tensor      [B, N, N]       (float32 or bool)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   contains 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # Cast to half for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to half‑precision
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, w_right_proj)  # same shape

    # ------------------------------------------------------------------
    # 4️⃣ Gating pre‑activations (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Fuse left/right with their gates using a tiny Triton kernel
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536  # ~64k elements per program (tuned for H100)
    grid = ((numel + BLOCK - 1) // BLOCK,)

    # Flatten the tensors – they are contiguous so view(-1) is safe
    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # No need to keep the gate tensors after this point
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ operation: batched matrix multiplication A @ Bᵀ
    #    (equivalent to einsum('... i k d, ... j k d -> ... i j d'))
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    total_heads = B * H

    # Rearrange to (head, i, k) layout expected by torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Batched GEMM in FP16 – leverages H100 Tensor‑Core fast paths
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [head, N, N]

    # Restore the original layout: [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing LayerNorm (half‑precision weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (triton element‑wise kernel)
    # ------------------------------------------------------------------
    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_out](
        out.view(-1),
        out_gate.contiguous().view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (FP16 → FP16)
    # ------------------------------------------------------------------
    final = F.linear(out_mul, w_to_out)  # [B, N, N, dim]  fp16

    # Return in the expected float32 format
    return final.to(torch.float32)
