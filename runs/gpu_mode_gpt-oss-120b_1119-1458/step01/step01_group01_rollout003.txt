@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ap, stride_am, stride_ak,
    stride_bp, stride_bk, stride_bn,
    stride_cp, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block across M
    pid_n = tl.program_id(1)  # block across N
    pid_p = tl.program_id(2)  # block across batch*hidden (P)

    # Compute row/col offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # P-dim base pointers
    a_ptrs_base = A_ptr + pid_p * stride_ap
    b_ptrs_base = B_ptr + pid_p * stride_bp
    c_ptr_base  = C_ptr + pid_p * stride_cp

    # accumulator FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_ptrs = a_ptrs_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        b_ptrs = b_ptrs_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot returns fp32 when inputs are fp16
        acc += tl.dot(a, b)

    c_ptrs = c_ptr_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Triton accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy N³ reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed as a batched half‑precision GEMM using a custom Triton kernel.
Each (batch, hidden) pair executes an independent matrix multiplication
    left_[p] (N×N)  @  right_[p]^T (N×N)  →  out_[p] (N×N)
The kernel tiles the M, N and K dimensions, uses fp16 loads, fp32 accumulation
and Tensor‑Core dot products (tl.dot) for maximal throughput on H100.
All remaining operations (LayerNorms, Linear projections, sigmoids and the
final projection) are done with PyTorch.  The output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (batch * hidden) matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    stride_ap, stride_am, stride_ak,    # strides for A (P, M, K)
    stride_bp, stride_bk, stride_bn,    # strides for B (P, K, N)
    stride_cp, stride_cm, stride_cn,    # strides for C (P, M, N)
    M, N, K,                            # matrix sizes (all = seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[p, i, j] = Σ_k A[p, i, k] * B[p, k, j]
    where A and B are fp16, C is fp16 (accumulated in fp32).
    """
    pid_m = tl.program_id(0)      # block index over M
    pid_n = tl.program_id(1)      # block index over N
    pid_p = tl.program_id(2)      # batch * hidden index (P)

    # -------------------------- offsets -------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (p) slice
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------- K loop --------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (M, K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # fp16 load

        # B tile: (K, N)
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # fp16 load

        # Tensor‑core dot (fp16 → fp32 accumulation)
        acc += tl.dot(a, b)

    # -------------------------- store ---------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _triton_tri_mul(left: torch.Tensor,
                    right: torch.Tensor,
                    seq_len: int,
                    hidden_dim: int,
                    BLOCK_M: int = 128,
                    BLOCK_N: int = 128,
                    BLOCK_K: int = 32):
    """
    left / right : [B, N, N, H]   (dtype = float16)
    Returns out  : [B, N, N, H]   (dtype = float16)
    """
    B = left.shape[0]
    # (B, N, N, H) → (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right must be transposed on the last two dims to realise the "right^T"
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    P = B * hidden_dim                     # total independent matmuls
    # reshape to (P, N, N) where each P corresponds to (b, h)
    left_flat = left_perm.view(P, seq_len, seq_len)
    right_flat = right_perm.view(P, seq_len, seq_len)

    # allocate output
    out = torch.empty_like(left_flat)

    # strides (tensors are contiguous → simple strides)
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # M‑blocks
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # N‑blocks
        P,                                    # one block per (batch, hidden)
    )

    _tri_mul_kernel[grid](
        left_flat, right_flat, out,
        stride_ap, stride_am, stride_ak,
        stride_bp, stride_bk, stride_bn,
        stride_cp, stride_cm, stride_cn,
        seq_len, seq_len, seq_len,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8, num_stages=3
    )

    # reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data (tuple): (input_tensor, mask, weights, config)
    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input & output channels
    hidden_dim = config["hidden_dim"]  # internal hidden dimension
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B, N, N, hidden]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask shape [B, N, N])
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # broadcast to (...,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # keep fp32 for later

    # Apply the element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the N³ reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel:  out = Σ_k left[..., i, k, d] * right[..., j, k, d]
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]   # N
    out_fp16 = _triton_tri_mul(
        left_fp16,
        right_fp16,
        seq_len=seq_len,
        hidden_dim=hidden_dim,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )   # [B, N, N, hidden] (fp16)

    # Back to fp32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate (broadcast over hidden)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
