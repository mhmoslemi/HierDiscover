"""
TriMul (outgoing) forward pass – Triton‑accelerated O(N³) contraction.

Algorithm
---------
1. Layer‑norm over the feature dimension `dim`.
2. Two bias‑free linear projections (`left_proj`, `right_proj`) from `dim` to
   `hidden_dim`.
3. (Optional) mask the projections.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from the
   normalized tensor and apply them element‑wise to the projections.
5. Heavy contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `C = A @ Bᵀ` for each hidden slice.
   It is performed by a custom Triton kernel that tiles the N×N matrices in
   blocks (BLOCK_M × BLOCK_N) and accumulates in FP32 while loading half‑precision
   inputs (leveraging tensor‑cores on H100).
6. Layer‑norm over `hidden_dim`.
7. Multiply by `out_gate` and project back to `dim`.

Only step 5 is Triton‑implemented; the remaining operations use high‑performance
PyTorch primitives.  The kernel works for any batch size (≤2), sequence length
up to 1024 and hidden dimension 128 (as used in the test suite.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (batched over batch*hidden)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                # *fp16  [M, N, K]   (M = batch*hidden)
    B_ptr,                # *fp16  [M, N, K]
    C_ptr,                # *fp32  [M, N, N]
    M, N, K,              # i32: sizes
    stride_am, stride_ai, stride_ak,   # strides for A (dim0, dim1, dim2)
    stride_bm, stride_bj, stride_bk,   # strides for B (dim0, dim1, dim2)
    stride_cm, stride_ci, stride_cj,   # strides for C (dim0, dim1, dim2)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)        # tile along i‑dimension
    pid_n = tl.program_id(1)        # tile along j‑dimension
    pid_z = tl.program_id(2)        # batch*hidden index

    # -------------------------------------------------
    # Tile offsets
    # -------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # -------------------- load A tile --------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # (BLOCK_M, BLOCK_K)  fp16

        # -------------------- load B tile (transposed) --------------------
        # B is accessed as B[k, j] → load with k as first dim, j as second
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # (BLOCK_K, BLOCK_N)  fp16

        # accumulate in FP32 (dot handles fp16->fp32 conversion)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # -------------------- write result --------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using Triton.
    left / right: [B, N, N, H] in fp16.
    Returns: [B, N, N, H] in fp32.
    """
    B, N, _, H = left.shape

    # Move hidden dim next to batch: (B, H, N, N) → (M, N, N) where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP32)
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # Strides in elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Block sizes – these work well on H100 (tensor‑core friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient required).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, on CUDA)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float, may be ignored)
        - weights_dict : mapping from weight name to torch.Tensor
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool) which defaults to True.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (dim)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim]

    # -------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1]
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) and application
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (half‑precision inputs)
    # -------------------------------------------------
    # Cast to fp16 to reduce memory bandwidth; the kernel accumulates in FP32.
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out = _batched_matmul_transpose(left_fp16, right_fp16)  # [B, N, N, hidden_dim] float32

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 7️⃣ Gate and final projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
