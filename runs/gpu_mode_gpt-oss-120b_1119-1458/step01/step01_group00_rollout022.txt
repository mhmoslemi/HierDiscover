def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid_head = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   total_heads = B * H
   b = pid_head // H
   h = pid_head % H

   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   mask_m = offs_m < N
   mask_n = offs_n < N

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: rows i (offs_m), cols k (offs_k)
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile: we need element at (k, j) for original right[b, j, k, h]
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + \
            offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j

"""
TriMul (outgoing) forward pass – highly‑optimised Triton version.

Key ideas
---------
* All five linear projections (left/right and the three gates) are fused into a
  single mat‑mul `combined = x @ W_comb.T`.  This reduces the number of passes
  over the huge input tensor `x` (shape B×N×N×C) from five to one.
* The heavy N³ contraction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  is implemented as a batched GEMM (`left @ rightᵀ`) using a custom Triton
  kernel.  The kernel works on the transposed view of `right` without an
  expensive `permute(...).contiguous()` copy.
* All arithmetic that does not need float32 precision is kept in float16.
  Layer‑norms, gating and the final linear projection are performed in
  float16, with a single float32 conversion at the very end (as required by the
  spec).
* Block sizes and warps are tuned for an NVIDIA H100 (BLOCK_M/N=64, BLOCK_K=64,
  8 warps, 3 stages) to maximise Tensor‑Core utilisation and hide memory latency.

The entry point is `custom_kernel(data)`, where `data` is a tuple
(input, mask, weights, config) as described in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # const __fp16 *
    right_ptr,                    # const __fp16 *
    out_ptr,                      # __fp16 *
    B, N, H,                      # i32 compile‑time scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (right is transposed on the N axes)
    """
    pid_head = tl.program_id(0)   # encodes (batch, head)
    pid_m    = tl.program_id(1)   # block row (i)
    pid_n    = tl.program_id(2)   # block col (j)

    # ------------------------------------------------------------------
    # decode batch index b and hidden index h
    # ------------------------------------------------------------------
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # ------------------------------------------------------------------
    # compute tile offsets (with bounds checking)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K (the third axis of the input)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (i,k)  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile: we need right[b, j, k, h] (i.e. (k,j) view)
        # original right layout is (b, i, k, h); we swap i ↔ j on‑the‑fly.
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # accumulate the product over the K‑tile
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N) <- (BLOCK_M,BLOCK_K)·(BLOCK_K,BLOCK_N)

    # ------------------------------------------------------------------
    # Write back the tile (convert back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left  : torch.Tensor of shape [B, N, N, H] (fp16)
    right : torch.Tensor of shape [B, N, N, H] (fp16) – NOT transposed

    Returns
    -------
    out : torch.Tensor of shape [B, N, N, H] (fp16)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides expressed in element counts (as expected by Triton)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tuned block sizes for H100 (Tensor‑Core friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # one program per (batch, head)
        (N + BLOCK_M - 1) // BLOCK_M,          # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # col tiles
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,        # H100 benefits from 8 warps per block
        num_stages=3        # double‑buffering to hide memory latency
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Tensors containing all linear / norm weights
        - config_dict  : dict with keys 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Helper to move tensors to the correct device/dtype (half for compute)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load and cast model parameters
    # ------------------------------------------------------------------
    norm_w = to_half(weights["norm.weight"])
    norm_b = to_half(weights["norm.bias"])

    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])

    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm directly in fp16 (saves a full‑precision copy)
    # ------------------------------------------------------------------
    # Convert input to fp16 once, then normalise in fp16.
    x_half = input_tensor.to(torch.float16)
    x_norm = F.layer_norm(
        x_half,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )   # [B, N, N, C]  (fp16)

    # ------------------------------------------------------------------
    # 2. Fuse the five linear projections (two projections + three gates)
    # ------------------------------------------------------------------
    # Stack the weight matrices vertically: (5*H, C)
    combined_weight = torch.cat(
        [left_proj_w,
         right_proj_w,
         left_gate_w,
         right_gate_w,
         out_gate_w],
        dim=0,
    )
    # Linear projection of the whole tensor in a single matmul
    combined = F.linear(x_norm, combined_weight)               # [B,N,N,5*H] fp16

    H = hidden_dim  # for brevity
    left = combined[..., :H]               # [B,N,N,H]
    right = combined[..., H:2*H]           # [B,N,N,H]

    left_gate  = torch.sigmoid(combined[..., 2*H:3*H])
    right_gate = torch.sigmoid(combined[..., 3*H:4*H])
    out_gate   = torch.sigmoid(combined[..., 4*H:5*H])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Apply the element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via the tuned Triton kernel (left @ rightᵀ per head)
    # ------------------------------------------------------------------
    out = tri_mul_triton(left, right)      # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (still in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7. Apply output gate and final projection back to the original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)          # [B,N,N,C] fp16

    # ------------------------------------------------------------------
    # 8. Convert back to float32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
