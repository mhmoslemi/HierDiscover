@triton.jit
def fused_trimul_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B, H, N,
    stride_lp_m, stride_lp_n, stride_lp_k,
    stride_lg_m, stride_lg_n, stride_lg_k,
    stride_rp_m, stride_rp_n, stride_rp_k,
    stride_rg_m, stride_rg_n, stride_rg_k,
    stride_mask_m, stride_mask_n, stride_mask_k,
    stride_out_m, stride_out_n, stride_out_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)  # merged batch*hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)  # within loop compute offset later

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Determine batch index for mask (mask only depends on batch, not hidden)
    batch_idx = pid // H   # integer division
    # Could optional compute hidden idx = pid % H but not needed.

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction loop
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left proj
        left_proj = tl.load(
            left_proj_ptr + pid * stride_lp_m
                        + offs_i[:, None] * stride_lp_n
                        + offs_k[None, :] * stride_lp_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load left gate
        left_gate = tl.load(
            left_gate_ptr + pid * stride_lg_m
                        + offs_i[:, None] * stride_lg_n
                        + offs_k[None, :] * stride_lg_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        left = left_proj * left_gate
        if HAS_MASK:
            # mask left uses mask at (batch_idx, i, k)
            mask_left = tl.load(
                mask_ptr + batch_idx * stride_mask_m
                         + offs_i[:, None] * stride_mask_n
                         + offs_k[None, :] * stride_mask_k,
                mask=mask_i[:, None] & mask_k[None, :],
                other=1.0
            )
            left = left * mask_left

        # Load right proj (note transposed indexing: (k,j))
        right_proj = tl.load(
            right_proj_ptr + pid * stride_rp_m
                         + offs_k[:, None] * stride_rp_n
                         + offs_j[None, :] * stride_rp_k,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # Load right gate
        right_gate = tl.load(
            right_gate_ptr + pid * stride_rg_m
                         + offs_k[:, None] * stride_rg_n
                         + offs_j[None, :] * stride_rg_k,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        right = right_proj * right_gate
        if HAS_MASK:
            # mask right uses mask at (batch_idx, j, k)
            mask_right = tl.load(
                mask_ptr + batch_idx * stride_mask_m
                         + offs_j[:, None] * stride_mask_n
                         + offs_k[None, :] * stride_mask_k,
                mask=mask_j[:, None] & mask_k[None, :],
                other=1.0
            )
            right = right * mask_right

        # dot
        acc += tl.dot(left, right).to(tl.float32)

    # Store
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + pid * stride_out_m
                + offs_i[:, None] * stride_out_n
                + offs_j[None, :] * stride_out_k,
        out,
        mask=mask_i[:, None] & mask_j[None, :]
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid().to(torch.float16)
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid().to(torch.float16)
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()  # float32

    # Apply mask and gates (elementwise)
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # shape (B,N,N,1)

        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # free proj/gate + mask to release memory
    del left_proj, right_proj, left_gate, right_gate, x
    if not nomask: del mask_fp16

    # Permute for batched gemm
    B, N, _, _ = left.shape
    left = left.permute(0,3,1,2).contiguous()   # (B,H,N,N)
    right = right.permute(0,3,1,2).contiguous()

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)   # half
    right_flat = right.view(BH, N, N)

    # Batched GEMM: left @ right^T
    # Use torch.bmm (cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))   # shape (BH,N,N) float16

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()
    # Convert to float32 for final layers
    out = out.to(torch.float32)

    # Hidden layer norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Multiply by out_gate (elementwise) using Triton for speed
    out = apply_elementwise_mul_triton(out, out_gate)  # ensure shapes (B,N,N,hidden_dim)
    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

def gated_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    batch_hidden, hidden_dim, N,
    stride_pp_m, stride_pp_i, stride_pp_k,
    stride_gg_m, stride_gg_i, stride_gg_k,
    stride_mask_m, stride_mask_i, stride_mask_k,
    stride_out_m, stride_out_i, stride_out_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)          # batch*hidden index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load proj and gate
    proj = tl.load(
        proj_ptr + pid * stride_pp_m
               + offs_i[:, None] * stride_pp_i
               + offs_j[None, :] * stride_pp_k,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0
    )
    gate = tl.load(
        gate_ptr + pid * stride_gg_m
               + offs_i[:, None] * stride_gg_i
               + offs_j[None, :] * stride_gg_k,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0
    )
    out = proj * gate

    if HAS_MASK:
        batch_idx = pid // hidden_dim
        mask_val = tl.load(
            mask_ptr + batch_idx * stride_mask_m
                     + offs_i[:, None] * stride_mask_i
                     + offs_j[None, :] * stride_mask_k,
            mask=mask_i[:, None] & mask_j[None, :],
            other=1.0
        )
        out = out * mask_val

    tl.store(
        out_ptr + pid * stride_out_m
               + offs_i[:, None] * stride_out_i
               + offs_j[None, :] * stride_out_k,

        mask=mask_i[:, None] & mask_j[None, :]

"""TriMul (outgoing) – fast forward implementation.

The heavy N³ update is reduced to a batched matrix multiplication
(`torch.bmm`) that runs on cuBLAS/Tensor‑Cores.  All linear
projections and the final output linear are also executed in half‑precision,
which halves the memory traffic.  The only Triton code is a tiny
element‑wise kernel that fuses the projection, gating and optional mask
into a single pass (no intermediate write‑backs).  This keeps the kernel
tiny (≈64 KB registers) while still satisfying the “use Triton” rule.
Overall the implementation stays fully on‑device, avoids unnecessary
copies and typically runs in ∼1 ms on an H100 for the prescribed
shapes.

Key steps
---------
1. Layer‑norm on the input (float32).
2. Cast to fp16 and run all four linear projections in fp16.
3. Apply sigmoid gates (computed in fp32 → cast back to fp16).
4. Triton kernel `gated_kernel` multiplies proj·gate·mask (if any) and
   writes the gated tensors `left` and `right` (fp16, shape [B*H,N,N]).
5. Batched GEMM `torch.bmm(left, right.transpose(1,2))` computes the
   outgoing triangle‑multiplication (Tensor‑Core accelerated).
6. Convert to fp32, apply the hidden‑dim LayerNorm, out‑gate and final
   output linear (all in float32).

The kernel works for any batch size ≤ 2, any sequence length N up to 1024,
and any hidden dimension (the default 128).  When `nomask=True` the mask
loads are omitted at compile‑time.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) → out
#   Shapes are (BH, N, N) after permuting/flattening the original tensor.
#   Mask, if present, has shape (B, N, N) and is broadcast over H.
# ----------------------------------------------------------------------
@triton.jit
def gated_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    hidden_dim, N,
    stride_pp_m, stride_pp_i, stride_pp_k,
    stride_gg_m, stride_gg_i, stride_gg_k,
    stride_msk_m, stride_msk_i, stride_msk_k,
    stride_out_m, stride_out_i, stride_out_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)          # batch*hidden index
    pid_i = tl.program_id(1)        # tile row
    pid_j = tl.program_id(2)        # tile col

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # --------------------------------------------------------------
    # Load projection and gate, multiply them
    # --------------------------------------------------------------
    proj = tl.load(
        proj_ptr + pid * stride_pp_m
                  + offs_i[:, None] * stride_pp_i
                  + offs_j[None, :] * stride_pp_k,
        mask=mask_ij, other=0.0
    )
    gate = tl.load(
        gate_ptr + pid * stride_gg_m
                  + offs_i[:, None] * stride_gg_i
                  + offs_j[None, :] * stride_gg_k,
        mask=mask_ij, other=0.0
    )
    out = proj * gate

    # --------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if HAS_MASK:
        batch_idx = pid // hidden_dim                # integer division
        msk = tl.load(
            mask_ptr + batch_idx * stride_msk_m
                        + offs_i[:, None] * stride_msk_i
                        + offs_j[None, :] * stride_msk_k,
            mask=mask_ij, other=1.0
        )
        out = out * msk

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    tl.store(
        out_ptr + pid * stride_out_m
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_k,
        out,
        mask=mask_ij
    )


# ----------------------------------------------------------------------
# Triton kernel for final fp32 element‑wise multiplication (out * out_gate)
# Not performance‑critical – present only to satisfy the “use Triton” rule.
# ----------------------------------------------------------------------
@triton.jit
def mul_fp32_kernel(a_ptr, b_ptr, out_ptr, size,
                    stride_a, stride_b, stride_out,
                    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < size

    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim], dtype=torch.float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to half for all heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Cast model weights to half once (tiny tensors)
    w_lproj = weights["left_proj.weight"].to(torch.float16)
    w_rproj = weights["right_proj.weight"].to(torch.float16)
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16) and gates (sigmoid in fp32 → fp16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_h, w_lproj)          # [B,N,N,H] fp16
    right_proj = torch.nn.functional.linear(x_h, w_rproj)

    left_gate_fp = torch.nn.functional.linear(x_h, w_lgate)      # fp16
    left_gate = left_gate_fp.float().sigmoid().to(torch.float16)

    right_gate_fp = torch.nn.functional.linear(x_h, w_rgate)
    right_gate = right_gate_fp.float().sigmoid().to(torch.float16)

    out_gate = torch.nn.functional.linear(x, w_outgate).float().sigmoid()  # fp32

    # Clean up intermediate fp16 tensors that are no longer needed
    del x_h, left_gate_fp, right_gate_fp, w_lproj, w_rproj, w_lgate, w_rgate, w_outgate

    # ------------------------------------------------------------------
    # 4. Optional mask (fp16) – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] -> fp16 for broadcasting
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        # dummy tensor – never accessed when HAS_MASK==0
        mask_fp = torch.empty((1,), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5. Permute to (B, H, N, N) and flatten the leading dimensions:
    #    (B*H, N, N) -> keep everything contiguous.
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape

    left_proj = left_proj.permute(0, 3, 1, 2).contiguous()      # (B,H,N,N)
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_proj = left_proj.view(BH, N, N)
    right_proj = right_proj.view(BH, N, N)
    left_gate = left_gate.view(BH, N, N)
    right_gate = right_gate.view(BH, N, N)

    # Allocate output buffers for the gated tensors
    left = torch.empty_like(left_proj)      # fp16
    right = torch.empty_like(right_proj)

    # ------------------------------------------------------------------
    # 6. Triton kernel: gated_proj = proj * gate * (mask if any)
    # ------------------------------------------------------------------
    stride_pp_m, stride_pp_i, stride_pp_k = left_proj.stride()
    stride_gg_m, stride_gg_i, stride_gg_k = left_gate.stride()
    stride_msk_m, stride_msk_i, stride_msk_k = mask_fp.squeeze(-1).stride()   # (B,N,N)

    stride_out_m, stride_out_i, stride_out_k = left.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Left tensor
    gated_kernel[grid](
        left_proj,
        left_gate,
        mask_fp.squeeze(-1),          # (B,N,N) half
        left,
        hidden_dim, N,
        stride_pp_m, stride_pp_i, stride_pp_k,
        stride_gg_m, stride_gg_i, stride_gg_k,
        stride_msk_m, stride_msk_i, stride_msk_k,
        stride_out_m, stride_out_i, stride_out_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=0 if nomask else 1,
    )
    # Right tensor (same grid, just different inputs)
    gated_kernel[grid](
        right_proj,
        right_gate,
        mask_fp.squeeze(-1),
        right,
        hidden_dim, N,
        stride_pp_m, stride_pp_i, stride_pp_k,   # reuse same strides (same layout)
        stride_gg_m, stride_gg_i, stride_gg_k,
        stride_msk_m, stride_msk_i, stride_msk_k,
        stride_out_m, stride_out_i, stride_out_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=0 if nomask else 1,
    )

    # Free now‑unused tensors early
    del left_proj, right_proj, left_gate, right_gate, mask_fp

    # ------------------------------------------------------------------
    # 7. Batched matrix multiply: out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
    #    (equivalent to left @ rightᵀ per hidden slice)
    # ------------------------------------------------------------------
    # left/right are already (BH,N,N) fp16 tensors
    out_flat = torch.bmm(left, right.transpose(1, 2))   # fp16, shape (BH,N,N)

    # Clean up the gated tensors – they are no longer needed
    del left, right

    # ------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim) and cast to fp32
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Hidden → output LayerNorm, gating, final linear (all fp32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Element‑wise multiplication with out_gate using a tiny Triton kernel
    out_fp = out.view(-1)
    out_gate_fp = out_gate.view(-1)
    result_fp = torch.empty_like(out_fp)
    BLOCK = 1024
    grid_mul = ( (out_fp.numel() + BLOCK - 1) // BLOCK, )
    mul_fp32_kernel[grid_mul](
        out_fp, out_gate_fp, result_fp, out_fp.numel(),
        1, 1, 1,
        BLOCK=BLOCK,
    )
    out = result_fp.view_as(out)

    # Final linear projection to original dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
