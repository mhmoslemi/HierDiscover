    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_batch, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # The batch id is pid0, i tile is pid1, j tile is pid2
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_batch, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    hid = pid_batch % H
    batch = pid_batch // H

    # base offset for this (batch, hid)
    batch_offset = batch * stride_batch + hid

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k0 = k_iter * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # load left tile
        a_ptrs = left_ptr + batch_offset + i_range[:, None] * stride_i + k_range[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile (transposed)
        b_ptrs = right_ptr + batch_offset + k_range[:, None] * stride_k + j_range[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_ptr + batch_offset + i_range[:, None] * stride_i + j_range[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Cast to FP16 and apply the linear projections (no bias) to obtain the
   “left” and “right” hidden‑dim tensors.
3. Compute per‑position gates (sigmoid of linear layers) in FP16.
4. Fuse element‑wise multiplication of the projections, the gates and the
   optional mask with a tiny Triton kernel.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each (batch, head) pair.
   We implement it with a custom Triton kernel that works directly on the
   original layout (no costly permute+contiguous copies) and accumulates in
   FP32 for numerical stability.  The result is written back as FP32.
6. Layer‑norm over the hidden dimension, gating with `out_gate` and a final
   linear projection back to the original channel dimension.

All heavy compute (step 4 & 5) is performed by Triton kernels; the rest
uses native PyTorch ops (layer‑norm, linear, sigmoid, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,      # pointers
    size,                               # total number of elements
    BLOCK_SIZE: tl.constexpr,           # compile‑time constant
):
    """Element‑wise product out = a * b * c (all tensors 1‑D contiguous)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    c = tl.load(c_ptr + offs, mask=mask, other=tl.float16(0.0))

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton (FP16)."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                        # pointers
    B, H, N,                                            # dimensions
    stride_batch, stride_i, stride_k,                   # strides (in elements)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    with left/right stored as [B, N, N, H] (contiguous, last dim = H).
    The kernel loops over the K dimension in blocks and accumulates in FP32.
    """
    pid_batch = tl.program_id(0)   # combined batch*head index
    pid_i = tl.program_id(1)       # tile row
    pid_j = tl.program_id(2)       # tile col

    # De‑compose batch & head from linear pid
    hid = pid_batch % H
    batch = pid_batch // H

    # Base offset for this (batch,head) pair
    base = batch * stride_batch + hid          # stride_h == 1

    # Tile start coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Ranges for the current tile
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k0 = kk * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ----- load left tile (i,k) -----
        a_ptrs = left_ptr + base \
                 + i_range[:, None] * stride_i \
                 + k_range[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0)).to(tl.float32)

        # ----- load right tile (k,j) – transposed view -----
        b_ptrs = right_ptr + base \
                 + k_range[:, None] * stride_k \
                 + j_range[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0.0)).to(tl.float32)

        # ----- MAC -----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ----- write result (FP32) -----
    c_ptrs = out_ptr + base \
             + i_range[:, None] * stride_i \
             + j_range[None, :] * stride_k
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor – [B, N, N, C]   (float32, CUDA)
            mask_tensor  – [B, N, N]      (bool/float32, CUDA) – may be ignored
            weights_dict – model weights (see skeleton)
            config_dict  – must contain "dim", "hidden_dim", optional "eps", "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x_norm.half()                      # [B, N, N, dim] – FP16

    # Cast weights once to FP16
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) – stay in FP16
    # ------------------------------------------------------------------
    left_proj = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right_proj = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))

    # ------------------------------------------------------------------
    # 5. Apply mask + gates (fused Triton kernel if mask is present)
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask: [B, N, N] -> [B, N, N, hidden_dim] (contiguous)
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        mask_h = mask_exp.half()
        left = fused_mul3(left_proj, left_gate, mask_h)
        right = fused_mul3(right_proj, right_gate, mask_h)

    # ------------------------------------------------------------------
    # 6. Core contraction – custom Triton GEMM across the K dimension
    # ------------------------------------------------------------------
    # Allocate output tensor in FP32 (the kernel writes FP32 directly)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=inp.device)

    # Strides of the FP16 tensors (in elements)
    stride_b = left.stride(0)          # B stride = N*N*hidden_dim
    stride_i = left.stride(1)          # i stride = N*hidden_dim
    stride_k = left.stride(2)          # k stride = hidden_dim
    # (The hidden‑dim stride is 1 and is baked into the offsets.)

    # Tiling parameters – tuned for H100 (shared‑memory friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trmul_outgoing_kernel[grid](
        left,                         # FP16 left
        right,                        # FP16 right (transposed on‑the‑fly)
        out,                          # FP32 output
        B, hidden_dim, N,
        stride_b, stride_i, stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output layer‑norm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim] – FP32

    # ------------------------------------------------------------------
    # 8. Output gate (sigmoid) – computed in FP32 for stability
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))   # [B,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9. Final projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, to_out_w)   # [B, N, N, dim] – FP32

    return out_final
