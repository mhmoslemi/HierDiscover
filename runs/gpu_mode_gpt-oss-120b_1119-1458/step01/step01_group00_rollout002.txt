"""
TriMul (outgoing) forward pass – Triton‑accelerated N³ contraction.

Algorithm
---------
1. Layer‑norm the input (float32) → x_norm.
2. Cast to float16 for the heavy computation.
3. Linear projections (left/right) without bias.
4. Optional mask (broadcast on hidden dim) and gating (sigmoid of linear
   projections) are applied element‑wise – still in float16.
5. The core N³ contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is realised as a batched matrix‑multiply `left @ rightᵀ` for each
   (b,h) pair.  This is implemented in Triton (half‑precision inputs,
   FP32 accumulation).
6. Layer‑norm over the hidden dimension, apply the output gate and the
   final linear projection back to the original channel dimension.
7. Return the result in float32.

Only steps 1‑4 and 6‑7 use native PyTorch kernels (highly optimised);
step 5 – the N³ contraction – is the only custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # const float16 *
    right_ptr,           # const float16 *
    out_ptr,             # float16 *
    B, N, H,             # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    Each (b, h) pair is processed by a distinct program instance.
    """
    pid_head = tl.program_id(0)          # encodes (batch, hidden)
    pid_m = tl.program_id(1)             # block row (i)
    pid_n = tl.program_id(2)             # block col (j)

    # ------------------------------------------------------------------
    # Decode batch and head indices
    # ------------------------------------------------------------------
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]  shape (BLOCK_M, BLOCK_K)
        left_tile_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_tile = tl.load(
            left_tile_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )  # float16

        # right[b, j, k, h]  (note the swapped i/j order → we read the transposed view)
        right_tile_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right_tile = tl.load(
            right_tile_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )  # float16

        # Cast to FP32 for accumulation (Tensor‑core does this automatically)
        acc += tl.dot(left_tile.to(tl.float32), right_tile.to(tl.float32))

    # ------------------------------------------------------------------
    # Store the tile (convert back to float16)
    # ------------------------------------------------------------------
    out_tile_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(
        out_tile_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :]
    )


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor
        Shape (B, N, N, H), dtype float16, CUDA.
    right : torch.Tensor
        Same shape as ``left``, dtype float16, CUDA.
        (The kernel treats the second N axis of ``right`` as the
        transposed dimension.)

    Returns
    -------
    torch.Tensor
        Shape (B, N, N, H), dtype float16, on the same device.
    """

    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # Strides in element units (not bytes)

    s_rb, s_ri, s_rj, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tiling parameters – tuned for Hopper (H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,                                # each (batch, head) pair
        (N + BLOCK_M - 1) // BLOCK_M,         # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,         # tiles along j
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rj, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps per block works well on H100
        num_stages=2,              # double‑buffering
    )
    return out


# ----------------------------------------------------------------------
# Public entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Tensors containing all linear / norm weights
        - config_dict  : dict with at least ``dim`` and ``hidden_dim`` entries

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move a weight to the target device and dtype
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16, non_blocking=True)

    # ------------------------------------------------------------------
    # Extract weights (biases are kept in float32 for the LayerNorms)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C]  float32

    # Cast to half for the compute‑heavy part
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, right_proj_w)  # [B, N, N, hidden_dim]  fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via Triton
    #    Compute left @ rightᵀ for each (b, hidden) pair.
    # ------------------------------------------------------------------
    # rightᵀ view: swap the two sequence axes.
    right_t = right.permute(0, 2, 1, 3).contiguous()  # shape (B, N, N, H)
    out_h = tri_mul_triton(left, right_t)            # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Apply the output gate and the final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate
    final_h = F.linear(gated, to_out_w)   # [B,N,N,C]  fp16

    # ------------------------------------------------------------------
    # 8. Cast back to float32 as required by the spec
    # ------------------------------------------------------------------
    return final_h.to(torch.float32)
