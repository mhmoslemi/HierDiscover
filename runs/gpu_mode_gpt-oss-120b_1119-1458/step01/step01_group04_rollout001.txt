uses `tl.dot` (Tensor‑Core friendly) for the inner product.  Half‑precision
loads, fp32 accumulation and a small auto‑tuner give a large speed‑up over
the reference PyTorch implementation while keeping the final output in
float32.

All surrounding LayerNorms, linear projections and gating are performed
with PyTorch (cuBLAS) – they are memory‑bound and already highly optimized.
Only the cubic contraction is off‑loaded to Triton.

The entry point `custom_kernel` matches the required signature:
    (input, mask, weights, config) → output tensor of shape [B,N,N,dim] (fp32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------------
# Triton kernel: one GEMM per head (head = batch * hidden_dim)
# computes Cₕ = Aₕ @ Bₕᵀ   where Aₕ[i,k] = left[head,i,k] (fp16)
#                           Bₕ[j,k] = right[head,j,k] (fp16)
# ------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32}, num_warps=4),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32}, num_warps=8),
        triton.Config({"BLOCK_M": 96, "BLOCK_N": 96, "BLOCK_K": 32}, num_warps=8),
    ],
    key=["N"],
)
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16*  (head, i, k)
    right_ptr,         # fp16*  (head, j, k)
    out_ptr,           # fp32*  (head, i, j)
    N,                 # i32    sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left
    stride_rh, stride_rj, stride_rk,   # i64 strides for right (note: rj = row=j)
    stride_oh, stride_oi, stride_oj,   # i64 strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_h = tl.program_id(0)   # which head (batch * hidden)
    pid_m = tl.program_id(1)   # block row (i)
    pid_n = tl.program_id(2)   # block col (j)

    # ------------------------------------------------------------------
    # Compute offsets and masks for the M/N dimensions
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current head
    # ------------------------------------------------------------------
    left_head_ptr  = left_ptr  + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr   = out_ptr   + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for tk in range(num_tiles_k):
        k0 = tk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- load left tile A[i,k] ----------
        left_ptrs = (
            left_head_ptr
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        a = tl.load(left_ptrs, mask=left_mask, other=0.0)   # fp16

        # ---------- load right tile B[j,k] (transposed) ----------
        # right stored as (head, j, k). We need (k, j) to compute A @ Bᵀ.
        right_ptrs = (
            right_head_ptr
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        b = tl.load(right_ptrs, mask=right_mask, other=0.0)  # fp16

        # ---------- dot product (fp16 * fp16 → fp32) ----------
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_head_ptr
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ------------------------------------------------------------------
# Triton‑based TriMul forward pass
# ------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Performs the outgoing Triangle Multiplicative Update.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5  # LayerNorm epsilon

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) → fp16 for memory bandwidth
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)  # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 3) Optional masking (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gating (sigmoid after a linear)  → fp16 (except out_gate)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # fp32

    # --------------------------------------------------------------
    # 5) Apply gates
    # --------------------------------------------------------------
    left = left * left_gate    # fp16
    right = right * right_gate  # fp16

    # --------------------------------------------------------------
    # 6) Prepare tensors for the Triton kernel:
    #    reshape to (total_heads, N, N)  where total_heads = B * hidden_dim
    #    layout = (head, i, k) for left and (head, j, k) for right
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # (B, N, N, hidden) → (B, hidden, N, N) → (total_heads, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (fp32)
    out_flat = torch.empty(
        (total_heads, N, N), dtype=torch.float32, device=input_tensor.device
    )

    # --------------------------------------------------------------
    # 7) Triton launch configuration
    # --------------------------------------------------------------
    # Tile sizes are auto‑tuned; fallback values are defined in the kernel.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_heads,                         # program_id(0) – head dimension
        (N + BLOCK_M - 1) // BLOCK_M,        # program_id(1) – i blocks
        (N + BLOCK_N - 1) // BLOCK_N,        # program_id(2) – j blocks
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),   # stride for row dimension (j)
        right_flat.stride(2),   # stride for col dimension (k)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # make the compiler aware of the dtype of the pointer arguments
        # (left/right fp16, out fp32)
    )

    # --------------------------------------------------------------
    # 8) Reshape back to (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 9) Post‑processing: LayerNorm → out_gate → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden_dim (fp32)

    out = F.linear(out, weights["to_out.weight"])   # final projection to `dim`

    return out
```
