@triton.jit
def trimul_batch_gemm_kernel(
    L_ptr, R_ptr, O_ptr,
    N,
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to starting locations for this batch
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load L tile
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0).to(tl.float32)

        # Load R tile transposed (K x N)
        # Because we need R[k,j]; original layout R[j,k] at dims (N,N)
        R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(R_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0).to(tl.float32)

        acc += tl.dot(L, R)
    # Store result
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

    L_ptr, R_ptr, O_ptr,

    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,

    BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute global row/col (M,N) offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) – fast Triton implementation
================================================

This implementation follows the reference PyTorch TriMul module but moves the
dominant 3‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

to a custom Triton kernel.  For each hidden channel `d` the contraction is a
standard matrix multiplication `L_d @ R_d.T` (L,R ∈ ℝ^{N×N}).  By flattening the
batch and hidden dimensions (`B·H`) we obtain a batched GEMM that is executed
with a highly‑tuned Triton kernel:

* 3‑D launch grid → (batch, tile_m, tile_n) eliminates expensive division
  in the original kernel.
* Block sizes (128×128×64) are chosen to keep the tile in registers while
  providing enough work per program to saturate the H100.
* FP16 operands with FP32 accumulation give the best TFLOP/s on the H100.
* The rest of the TriMul pipeline (LayerNorm, linear projections, gating,
  output LayerNorm & projection) is performed with PyTorch which is
  negligible compared with the O(N³·H) GEMM.

The kernel does **not** handle masking; masking is applied in the PyTorch
pre‑processing step, so the kernel can work on dense tensors.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (L @ Rᵀ) for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    L_ptr,                # half  * (BH, N, N)  left  tensor
    R_ptr,                # half  * (BH, N, N)  right tensor
    O_ptr,                # float * (BH, N, N)  output accumulator
    N,                    # sequence length (runtime)
    stride_bL, stride_iL, stride_kL,   # strides for L  (batch, row, col)
    stride_bR, stride_iR, stride_kR,   # strides for R
    stride_bO, stride_iO, stride_jO,   # strides for O
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for TriMul:
        out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
    The hidden dimension is already folded into the batch dimension (BH = B*H).
    """
    batch_idx = tl.program_id(0)        # which (B·hidden) slice we are processing
    pid_m    = tl.program_id(1)        # tile row
    pid_n    = tl.program_id(2)        # tile column

    # --------------------------------------------------------------
    # Compute the global row/col offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current batch slice
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load L tile: shape (BLOCK_M, BLOCK_K)
        L_ptrs = L_batch + \
                 offs_m[:, None] * stride_iL + \
                 offs_k[None, :] * stride_kL
        L = tl.load(
            L_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)           # up‑cast to fp32 for accumulation

        # Load R tile transposed (we need R[k, j]): shape (BLOCK_K, BLOCK_N)
        # Original layout of R is (row=j, col=k) → stride_iR = N, stride_kR = 1
        R_ptrs = R_batch + \
                 offs_k[:, None] * stride_kR + \
                 offs_n[None, :] * stride_iR
        R = tl.load(
            R_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(L, R)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Write the result back to global memory
    # --------------------------------------------------------------
    O_ptrs = O_batch + \
             offs_m[:, None] * stride_iO + \
             offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[bs, N, N]       (bool or 0/1)
        - weights_dict : mapping of all required weight tensors
        - config_dict  : contains "dim" and "hidden_dim" (int)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 – the heavy GEMM runs in half precision
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Reshape to (BH, N, N) where BH = B * hidden_dim
    # ------------------------------------------------------------------
    B, N, _, _ = left_h.shape      # left_h: [B, N, N, hidden_dim]
    # Bring hidden dimension forward so we can treat (B·hidden) as batch
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)                # [BH, N, N] (half)
    right_flat = right_perm.view(BH, N, N)              # [BH, N, N] (half)

    # ------------------------------------------------------------------
    # 7. Allocate output buffer for the GEMM (FP32 accumulation)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------
    # 8. Triton GEMM launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Number of tiles per matrix dimension
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (BH, num_tiles_m, num_tiles_n)   # (batch, tile_m, tile_n)

    _trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),  # L strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),  # R strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),      # O strides
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=8,            # H100 prefers 8 warps per program for fp16 GEMM
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to original layout: [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 10. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11. Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
