@triton.jit
def trimul_gemm_kernel(
    L_ptr, R_ptr, O_ptr,
    N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
):
    pid = tl.program_id(0)

    # Compute batch and block indices
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx % num_pid_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # Prefetch first K tile
    k = tl.arange(0, BLOCK_K)
    offs_k = 0 * BLOCK_K + k
    mask_k = offs_k < N

    L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None,:] * stride_kL
    L_reg = tl.load(L_ptrs,
                    mask=mask_m[:,None] & mask_k[None,:],
                    other=0.0).to(tl.float32)

    R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None,:] * stride_iR
    R_reg = tl.load(R_ptrs,
                    mask=mask_k[:,None] & mask_n[None,:],
                    other=0.0).to(tl.float32)

    for kk in range(1, num_k):
        # compute dot for current tile
        acc += tl.dot(L_reg, R_reg)

        # prefetch next tile
        offs_k_next = kk * BLOCK_K + k
        mask_k_next = offs_k_next < N

        L_ptrs_next = L_batch + offs_m[:, None] * stride_iL + offs_k_next[None,:] * stride_kL
        L_next = tl.load(L_ptrs_next,
                         mask=mask_m[:, None] & mask_k_next[None,:],
                         other=0.0).to(tl.float32)

        R_ptrs_next = R_batch + offs_k_next[:, None] * stride_kR + offs_n[None,:] * stride_iR
        R_next = tl.load(R_ptrs_next,
                         mask=mask_k_next[:, None] & mask_n[None,:],
                         other=0.0).to(tl.float32)

        # shift registers
        L_reg = L_next
        R_reg = R_next

    # last tile
    acc += tl.dot(L_reg, R_reg)

    # Store
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None,:] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None,:])
```

```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    
    # Cast to half
    x_h = x.to(torch.float16)
    
    # cast weight matrices to half for projection/gate
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]
    
    # Projections
    left = torch.nn.functional.linear(x_h, left_proj_w)  # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)
    
    # gating scalars
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w)).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w)).to(torch.float16)
    
    # optional mask
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h
    
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    
    # Permute and reshape for GEMM
    # left: [B,N,N,hidden] -> [B, hidden, N, N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    
    B = left_perm.shape[0]
    H = left_perm.shape[1]
    N = left_perm.shape[2]
    
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)
    
    # Determine block sizes based on N
    if N <= 64:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch
    
    # launch kernel
    trimul_gemm_kernel[(total_programs,)](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        num_warps=8,
    )
    
    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden]
    
    # Output LayerNorm (float32)
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps
    )
    
    # Apply out gate (cast to float32)
    out = out * out_gate.float()
    
    # Final linear projection back to dim (float32)
    out = torch.nn.functional.linear(out, to_out_w)
    
    return out

    L_ptr, R_ptr, O_ptr,
    N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL,

    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

"""
TriMul (outgoing) forward pass – Triton‐accelerated batched GEMM.

The heavy part of the TriMul operator is the contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
which can be expressed as a batched matrix multiplication
    out_h = left_h @ right_hᵀ   for every hidden channel c.
We fuse:
  1. LayerNorm on the input (float32, PyTorch).
  2. Linear projections + sigmoid gates (half‑precision).
  3. Optional binary mask (half‑precision).
  4. A custom Triton GEMM kernel that computes the batched products
     with FP16 operands and FP32 accumulation (double‑buffered loads).
  5. Output LayerNorm, gating and final projection (float32, PyTorch).

The kernel tiles the (M=N, N=N) matrix multiply with configurable
BLOCK_M/N/K (chosen at launch) and uses a simple pre‑fetch for the K‑tiles
to keep the arithmetic units busy.  The rest of the pipeline stays in
PyTorch, which is negligible compared with the N²·hidden work.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM: out[batch, i, j] = Σ_k left[batch, i, k] * right[batch, j, k]
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    L_ptr,                     # half*  (BH, N, N) left operand
    R_ptr,                     # half*  (BH, N, N) right operand
    O_ptr,                     # float* (BH, N, N) output accumulator
    N,                         # int (sequence length)
    BLOCK_M: tl.constexpr,     # tile size in M dimension
    BLOCK_N: tl.constexpr,     # tile size in N dimension
    BLOCK_K: tl.constexpr,     # tile size in K dimension
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) Tile layout: each program works on a (BLOCK_M, BLOCK_N) tile
    #    of a particular batch (combined batch*hidden index)
    # --------------------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // tiles_per_batch
    tile_idx  = pid % tiles_per_batch
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx %  num_pid_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current batch
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 2) Loop over K dimension (reduction)
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    offs_k = tl.arange(0, BLOCK_K)

    for k_tile in range(num_k):
        k_base = k_tile * BLOCK_K
        offs_k_cur = k_base + offs_k
        mask_k = offs_k_cur < N

        # ----- Load a tile of left (M x K) -----
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k_cur[None, :] * stride_kL
        L_tile = tl.load(L_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0)                     # half → stays half

        # ----- Load a tile of right, transposed (K x N) -----
        # We first load a (N, K) tile and then transpose.
        R_ptrs = R_batch + offs_n[:, None] * stride_iR + offs_k_cur[None, :] * stride_kR
        R_tile_T = tl.load(R_ptrs,
                           mask=mask_n[:, None] & mask_k[None, :],
                           other=0.0)                     # half
        R_tile = tl.trans(R_tile_T)                     # now (K, N)

        # ----- Accumulate FP32 dot product -----
        acc += tl.dot(L_tile, R_tile)                  # FP16 * FP16 → FP32

    # --------------------------------------------------------------
    # 3) Store the result tile
    # --------------------------------------------------------------
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)
    eps          = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                                   # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2) Cast to half‑precision for all linear / gating ops
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Cast all projection / gate weights to half once (cheap)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias) – half precision
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_h, left_proj_w)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left  * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5) Gating – sigmoid in float32 then cast back to half
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w).float()).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w).float()).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w).float()).to(torch.float16)

    left  = (left  * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the batched GEMM:
    #    treat (batch * hidden) as a single batch dimension.
    # ------------------------------------------------------------------
    # left/right: [B, N, N, hidden] -> [B, hidden, N, N] -> (BH, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    BH = B * H

    left_flat  = left_perm.view(BH, N, N)   # half*
    right_flat = right_perm.view(BH, N, N)  # half*

    # Output accumulator – float32
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------
    # 7) Choose tile sizes (constants that become tl.constexpr)
    # ------------------------------------------------------------------
    if N <= 64:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch

    # ------------------------------------------------------------------
    # 8) Launch the Triton GEMM kernel
    # ------------------------------------------------------------------
    trimul_gemm_kernel[(total_programs,)](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        # strides for L
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for R
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        # strides for O
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        num_warps=8,               # good occupancy on H100
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to the original layout [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # float32

    # ------------------------------------------------------------------
    # 10) Output LayerNorm
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 11) Output gating (broadcasted) – convert gate to float32
    # ------------------------------------------------------------------
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
