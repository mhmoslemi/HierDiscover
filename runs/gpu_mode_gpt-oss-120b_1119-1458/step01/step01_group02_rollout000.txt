```python
@triton.jit
def _trimul_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr,
):
    pid_z = tl.program_id(0)   # batch*hidden
    pid_m = tl.program_id(1)   # row tile
    pid_n = tl.program_id(2)   # col tile

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        # load A tile (M x K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=tl.float16(0.0))
        # load B tile (K x N) – note B is not transposed in memory; we need B^T so we treat as (K,N) reading with stride_bk, stride_bj.
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=tl.float16(0.0))

        # compute dot; cast to fp32
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # write out
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

    # data tuple: (input_tensor, mask_tensor, weights_dict, config_dict)
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm over dim (fp32)
    norm_weight_fp32 = weights["norm.weight"]
    norm_bias_fp32 = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight_fp32,
        bias=norm_bias_fp32,
    )  # shape [B,N,N,dim], fp32

    # Cast to fp16 for projection
    x_fp16 = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,hidden] fp16
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # Mask
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] in fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Contract using Triton
    out = _batched_matmul_transpose_fp16(left, right)  # [B,N,N,hidden] fp16

    # Second layer norm over hidden dim
    out = torch.nn.functional.layer_norm(

        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,

    # Gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_w)  # fp16

    # Cast back to fp32 for output
    out = out.to(torch.float32)

TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input over the feature dimension `dim` (FP32).
2. Cast the normalized tensor to FP16 and apply the four linear layers
   (left_proj, right_proj, left_gate, right_gate, out_gate) in FP16.
   The mask (if present) and the sigmoid gates are fused with the projections.
3. The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton kernel that computes many independent
   matrix‑multiplications `C = A @ Bᵀ` where `A`/`B` have shape
   `[B*H, N, N]` (H = hidden_dim).  The kernel works on FP16 inputs and
   accumulates in FP32, using 128×128 tiles and a 32‑wide K‑tile.
4. A second LayerNorm over `hidden_dim` (FP16) is applied.
5. The result is gated by `out_gate`, projected back to `dim` with a final
   linear layer (FP16), and finally cast back to FP32 for the output.

Only the O(N³) part is Triton‑implemented; all other ops use
high‑performance PyTorch kernels.
"""

from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (FP16 inputs -> FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # integer sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bj, stride_bk,   # strides for B (M,N,K) – note Bᵀ is accessed
    stride_cm, stride_ci, stride_cj,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr,
):
    """Compute a batch of matrix‑multiplications C = A @ Bᵀ.

    The leading dimension M corresponds to batch*hidden.  Each program
    processes a tile of size BLOCK_M × BLOCK_N.
    """
    pid_z = tl.program_id(0)   # index over batch*hidden (M)
    pid_m = tl.program_id(1)   # i‑tile
    pid_n = tl.program_id(2)   # j‑tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- Load A tile : shape (BLOCK_M, BLOCK_K) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=tl.float16(0.0)
        )                      # fp16

        # ---- Load B tile (as Bᵀ) : shape (BLOCK_K, BLOCK_N) ----
        # B is stored as (..., N, N, H); we view it as (M, N, K)
        # Access pattern yields Bᵀ by swapping the K/N axes in the address.
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=tl.float16(0.0)
        )                      # fp16

        # FP16 dot product, accumulate in FP32
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ---- Write the output tile ----
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.
    Input tensors must be FP16 and have shape [B, N, N, H].
    Output tensor has the same shape and dtype (FP16).
    """
    B, N, _, H = left.shape

    # Collapse batch and hidden dims → shape (M, N, K) where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate FP32 output buffer (better accumulation)
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # Strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    NUM_STAGES = 2

    grid = (
        B * H,                               # pid_z
        (N + BLOCK_M - 1) // BLOCK_M,        # pid_m (i‑tiles)
        (N + BLOCK_N - 1) // BLOCK_N,        # pid_n (j‑tiles)
    )

    _trimul_matmul_fp16_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_STAGES=NUM_STAGES,
        num_warps=8,
    )

    # Convert back to FP16 and reshape to original layout
    C = C.to(torch.float16)
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,           # input tensor [B, N, N, dim]
        torch.Tensor,           # mask tensor   [B, N, N]  (may be ignored)
        Dict[str, torch.Tensor],# weights dict
        Dict[str, Any]          # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).
    Returns a tensor of shape [B, N, N, dim] on the same device as ``input``.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (FP32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and apply linear projections + gates
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Cast weights once to FP16 for speed
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # Linear projections (no bias)
    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,hidden] fp16
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)    # fp16 back to dim

    # Cast back to FP32 for the final output
    out = out.to(torch.float32)
    return out
