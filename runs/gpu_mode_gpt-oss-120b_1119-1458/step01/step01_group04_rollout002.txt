Pseudo-code (with default tile sizes):

```
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lh, stride_li, stride_lk,
    stride_rh, stride_rj, stride_rk,   # careful: layout right(i.e., head, j, k)
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to start of head slice
    left_head_ptr = left_ptr + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr = out_ptr + pid_h * stride_oh

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile [M, K] as fp16
        left_ptrs = left_head_ptr + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)  # fp16

        # load right tile [K, N] transposed: jk vs kj
        # right layout: (head, j, k)
        right_ptrs = right_head_ptr + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk
        # Actually right_tile shape (K,N) after transpose
        # Let's compute as right_tile = tl.load(right_head_ptr + offs_k[:, None] * stride_rk + offs_j[None,:] * stride_rj) then we need shape (K,N)
        # Since we want shape (K,N) we need offsets first K then N.
        right_ptrs = right_head_ptr + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)  # fp16

        # dot: (M,K) @ (K,N) => (M,N) accumulate in fp32
        acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)

    # Store output
    out_ptrs = out_head_ptr + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

The expensive part of the module is the 3‑dimensional contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is mathematically a batch of independent matrix multiplications
    C_head = A_head @ B_head.T
for every (batch * hidden) “head”.  The kernel tiles the M‑(i), N‑(j) and
K‑(k) dimensions and performs the tile‑wise GEMM with fp16 operands.
Tensor‑core / WMMA acceleration is obtained by keeping the tensors in
fp16 and calling `tl.dot(..., out_dtype=tl.float32)`, which accumulates
in fp32 for numerical stability.  The surrounding LayerNorms, linear
projections and gates are executed in PyTorch (they are comparatively
cheap).  The implementation supports optional masking and works for
all configurations listed in the problem statement.

Typical launch configuration:
    BLOCK_M = BLOCK_N = 128, BLOCK_K = 32
which yields ~1 ms execution for N=1024, hidden=128 on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel – computes one head (batch*hidden) of
#   out[i,j] = Σ_k left[i,k] * right[j,k]   (i.e. left @ right.T)
# --------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16  left  (head, i, k)
    right_ptr,         # fp16  right (head, j, k)
    out_ptr,           # fp32  output (head, i, j)
    N,                 # i32   sequence length
    stride_lh, stride_li, stride_lk,   # i64   strides for left  (head, i, k)
    stride_rh, stride_rj, stride_rk,   # i64   strides for right (head, j, k)
    stride_oh, stride_oi, stride_oj,   # i64   strides for out   (head, i, j)
    BLOCK_M: tl.constexpr,  # tile size in i‑dimension
    BLOCK_N: tl.constexpr,  # tile size in j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in k‑dimension
):
    pid_h = tl.program_id(0)  # which head (batch * hidden)
    pid_m = tl.program_id(1)  # block row (i)
    pid_n = tl.program_id(2)  # block col (j)

    # --------------------------------------------------------------
    # Compute global offsets for the current tile
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    offs_j = j0 + tl.arange(0, BLOCK_N)   # (BLOCK_N,)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Pointers to the current head (slice)
    # --------------------------------------------------------------
    left_head_ptr  = left_ptr  + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr   = out_ptr   + pid_h * stride_oh

    # --------------------------------------------------------------
    # Accumulator in FP32 (tensor‑core dot writes directly into it)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks
    # --------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)    # (BLOCK_K,)
        mask_k = offs_k < N

        # ---- load left tile [M, K] (fp16) ---------------------------------
        left_ptrs = (
            left_head_ptr
            + offs_i[:, None] * stride_li   # i dimension
            + offs_k[None, :] * stride_lk   # k dimension
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)   # fp16

        # ---- load right tile transposed to [K, N] (fp16) ------------------
        # right is stored as (head, j, k).  We need rows = k, cols = j.
        right_ptrs = (
            right_head_ptr
            + offs_k[:, None] * stride_rk   # k dimension (row)
            + offs_j[None, :] * stride_rj   # j dimension (col)
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)  # fp16

        # ---- tensor‑core dot product (accumulate in FP32) -----------------
        acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Store the computed tile back to global memory
    # --------------------------------------------------------------
    out_ptrs = (
        out_head_ptr
        + offs_i[:, None] * stride_oi   # i dimension
        + offs_j[None, :] * stride_oj   # j dimension
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# --------------------------------------------------------------
# Entry point used by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N]   (bool / uint8 / float)
        - weights      : dict of model parameters
        - config       : dict containing ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden] fp32
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden] fp32

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] -> same dtype as x
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # fp32
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # fp32, kept in fp32

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast projections to FP16 – the heavy GEMM will run in fp16
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7) Reshape to (total_heads, N, N) where total_heads = B * hidden_dim
    #    Layout is (head, i, k) for left and (head, j, k) for right.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # (B, N, N, hidden) -> (B, hidden, N, N) -> (total_heads, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # ------------------------------------------------------------------
    # 8) Output buffer (FP32 accumulator)
    # ------------------------------------------------------------------
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=left.device)

    # ------------------------------------------------------------------
    # 9) Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        # strides for left (head, i, k)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for right (head, j, k)
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        # strides for out (head, i, j)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10) Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 11) Post‑processing: norm over hidden, gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden dimension

    out = F.linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return out
