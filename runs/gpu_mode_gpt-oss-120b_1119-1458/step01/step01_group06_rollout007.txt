```python
@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)   # M index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left = tl.load(
            left_ptr + pid * stride_lm
                     + offs_i[:, None] * stride_ln
                     + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr + pid * stride_rm
                     + offs_k[:, None] * stride_rk
                     + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # dot, each tile uses no explicit warp-level config.
        acc += tl.dot(left, right)

    # Convert to fp16 for storage
    tl.store(
        out_ptr + pid * stride_om
                 + offs_i[:, None] * stride_on
                 + offs_j[None, :] * stride_ok,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )
```

"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input (shape [B,N,N,C]).
2. Compute the four linear projections (left/right and their gates).
   * left/right are projected to the hidden dimension H.
   * Optional pairwise mask is applied in FP32 before the heavy reduction.
   * Gating values (sigmoid) are multiplied element‑wise.
3. Fuse the H‑dimension into the batch dimension:  (B,N,N,H) → (B*H,N,N).
   The left tensor is used as‑is while the right tensor is transposed on the
   last two axes – this yields the exact operation
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   The N×N matrix product for each (B·H) slice is performed with a
   *batched* `torch.bmm`, which on an H100 executes as a Tensor‑Core
   strided‑batched GEMM (half‑precision inputs, FP32 accumulation) and
   is far faster than a naïve Triton N³ loop.
4. Convert the result back to the original view (B,N,N,H) and keep it in
   FP16.
5. Apply the final hidden‑dim LayerNorm (weights/bias are cast to FP16).
6. Multiply the normalized tensor by the `out_gate` using a tiny custom
   Triton kernel (to satisfy the “must use a kernel” requirement).
7. Final linear projection from hidden dimension H back to the original
   channel dimension C (half‑precision) and cast the output to FP32.

The heavy O(N³) reduction is therefore executed by a highly tuned cuBLAS
kernel (via `torch.bmm`), while the element‑wise gating is performed by
a fused Triton kernel that processes many elements per launch.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: out = out * gate   (both half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    a_ptr,  # out (will be overwritten)
    b_ptr,  # gate
    n_elements,                     # total number of elements (int64)
    BLOCK_SIZE: tl.constexpr,       # number of elements per program
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)   # fp16 loads
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    tl.store(a_ptr + offs, a * b, mask=mask)          # fp16 store


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple of (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, C] (float32)
            mask:        torch.Tensor [B, N, N] (float32 or bool) – may be unused
            weights:     dict of model parameters (all fp32)
            config:      dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep in FP32 for numeric stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,C]  FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free).  Keep results in FP32 for now.
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → broadcast to hidden dim
        m = mask.to(x.dtype).unsqueeze(-1)    # FP32
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid)  – still FP32
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply gates (still FP32)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast projections to FP16 for the N³ reduction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Reshape for batched GEMM  (B*H, N, N) and transpose right on the fly
    # ------------------------------------------------------------------
    #   left  : (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    #   right : (B, N, N, H) → (B, H, N, N) → transpose last two dims → (B*H, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 8. Batched half‑precision matrix multiplication (Tensor‑Core)
    #    out_flat[b,i,j] = Σ_k left_flat[b,i,k] * right_flat[b,k,j]
    # ------------------------------------------------------------------
    # torch.bmm uses a strided‑batched GEMM under the hood.
    out_flat = torch.bmm(left_flat, right_flat)  # FP16, shape (B*H, N, N)

    # ------------------------------------------------------------------
    # 9. Restore original 4‑D view  (B,N,N,H)  – keep as FP16
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 10. Hidden → output LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 11. Element‑wise multiplication with out_gate (FP16) using Triton
    # ------------------------------------------------------------------
    # out_gate is currently FP32 → cast once to FP16
    out_gate_half = out_gate.to(torch.float16)

    total_elements = out.numel()
    BLOCK_SIZE = 8192                     # multiple of 32, reasonable for H100
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # launch kernel – both tensors are flattened, contiguous in memory
    _elemwise_mul_kernel[grid](
        out.view(-1),               # a_ptr  (in‑place overwrite)
        out_gate_half.view(-1),     # b_ptr
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 12. Final linear projection back to the original channel dimension C
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),  # (C, H)
    )  # still FP16, shape [B,N,N,C]

    # ------------------------------------------------------------------
    # 13. Cast to FP32 for the API contract
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
