import torch
import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    L_ptr, R_ptr, O_ptr,
    N,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
):
    """
    Batched GEMM for TriMul contraction.
    For each (batch, hidden) slice compute:
        O[b,i,j] = Σ_k L[b,i,k] * R[b,j,k]
    L,R are in fp16, O in fp32.
    """
    pid = tl.program_id(0)

    # number of tiles per batch dimension
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx % num_pid_n

    # coordinates inside tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch slice
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # L[i,k]
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # R[j,k] (need transposed view: we want R[k,j])
        R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(L, R)

    # store result
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[0] * stride_jO + tl.arange(0, BLOCK_N)[None, :] * stride_jO

    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple

    Returns
    -------
    torch.Tensor
        Output tensor of shape (bs, seq_len, seq_len, dim) (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # 1) LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # 2) Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3) Optional masking
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # 4) Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5) Cast to half for GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # left_h : [B, N, N, hidden_dim]

    # Rearrange to (B * hidden_dim, N, N) contiguous layout
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)    # (BH, N, N), half
    right_flat = right_perm.view(BH, N, N)  # (BH, N, N), half

    # Output buffer for GEMM (fp32 accumulation)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # 6) Triton GEMM launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N

    trimul_gemm_kernel[(total_programs,)](

        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        left_flat.stride(0),  # stride_bL
        left_flat.stride(1),  # stride_iL (i)
        left_flat.stride(2),  # stride_kL (k)
        right_flat.stride(0), # stride_bR
        right_flat.stride(1), # stride_iR (j)
        right_flat.stride(2), # stride_kR (k)
        out_flat.stride(0),   # stride_bO
        out_flat.stride(1),   # stride_iO (i)
        out_flat.stride(2),   # stride_jO (j)
        num_warps=8,
    )

    # 7) Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 8) Output LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(

        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 9) Output gating
    out = out * out_gate

    # 10) Final linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
Triton‑accelerated forward pass of the TriMul “outgoing” operator.

The heavy contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is reformulated as a batched GEMM:
    out_h = L_h @ R_h.T   for each hidden channel c.
A custom Triton kernel computes this GEMM in FP16 with FP32 accumulation.
All other steps (LayerNorm, linear projections, gating, output projection)
stay in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    L_ptr,                # const half*  (BH, N, N)   left
    R_ptr,                # const half*  (BH, N, N)   right
    O_ptr,                # float*      (BH, N, N)   output
    N,                    # seq_len (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
):
    """
    Batched GEMM for one TriMul batch slice.
    Compute   O = L @ Rᵀ   (L,R ∈ ℝ^{N×N}) with half inputs and fp32 accumulation.
    """
    pid = tl.program_id(0)

    # ----- tile decomposition -------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // tiles_per_batch          # index over (B * hidden_dim)
    tile_idx  = pid % tiles_per_batch
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑axis
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑axis

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch item
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over k ----------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # L[i,k]   (M × K)
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(
            L_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # R[j,k]   we need Rᵀ → load R[k,j]
        R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(
            R_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(L, R)      # (M×K)·(K×N) → (M×N)

    # ----- write back ---------------------------------------------------------
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dim)
    # -------------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -------------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Cast to half for GEMM
    # -------------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape         # left_h : (B, N, N, hidden_dim)

    # Move hidden_dim into the batch dimension so that each (B, hidden) pair
    # yields an independent GEMM of shape (N, N) × (N, N)ᵀ.
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # (BH, N, N), fp16
    right_flat = right_perm.view(BH, N, N) # (BH, N, N), fp16

    # -------------------------------------------------------------------------
    # 6. Allocate output buffer (fp32 accumulation)
    # -------------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # -------------------------------------------------------------------------
    # 7. Triton GEMM launch
    # -------------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch

    trimul_gemm_kernel[(total_programs,)](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        # strides for L (BH, N, N)
        left_flat.stride(0),   # stride_bL
        left_flat.stride(1),   # stride_iL (i)
        left_flat.stride(2),   # stride_kL (k)
        # strides for R (BH, N, N)
        right_flat.stride(0),  # stride_bR
        right_flat.stride(1),  # stride_iR (j)
        right_flat.stride(2),  # stride_kR (k)
        # strides for O (BH, N, N)
        out_flat.stride(0),    # stride_bO
        out_flat.stride(1),    # stride_iO (i)
        out_flat.stride(2),    # stride_jO (j)
        num_warps=8,
    )

    # -------------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim)
    # -------------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------------
    # 9. Output LayerNorm over hidden_dim
    # -------------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------------------------------
    # 10. Output gating
    # -------------------------------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------------------------------
    # 11. Final linear projection back to original dim
    # -------------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
