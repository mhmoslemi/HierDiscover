  input_tensor, mask_tensor, weights, config = data

  B,N,_,dim = input_tensor.shape
  hidden_dim = config["hidden_dim"]
  eps = config.get("eps", 1e-5)
  nomask = config.get("nomask", False)

  # Convert weights to half
  w = {}
  for key in weights:
      w[key] = weights[key].to(torch.float16)

  # Input to half
  x = input_tensor.to(torch.float16)

  # LayerNorm
  x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
  )

  # Projections
  left_proj = F.linear(x_norm, w["left_proj.weight"])
  right_proj = F.linear(x_norm, w["right_proj.weight"])

  # Gates (sigmoid)
  left_gate = torch.sigmoid(F.linear(x_norm, w["left_gate.weight"]))
  right_gate = torch.sigmoid(F.linear(x_norm, w["right_gate.weight"]))
  out_gate = torch.sigmoid(F.linear(x_norm, w["out_gate.weight"]))

  # Apply mask & gates
  if nomask:
      left = left_proj * left_gate
      right = right_proj * right_gate
  else:
      # mask as half, flatten
      mask_h = mask_tensor.to(torch.float16)
      left = fused_mul3_masked(left_proj, mask_h, left_gate)
      right = fused_mul3_masked(right_proj, mask_h, right_gate)

  # Contraction
  # Permute & reshape for batched GEMM
  # left: (B,N,N,H) -> (B,H,N,N)
  left_perm = left.permute(0, 3, 1, 2).contiguous()
  right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose k dimension

  # flatten batch
  left_mat = left_perm.view(B * hidden_dim, N, N)
  right_mat = right_perm.view(B * hidden_dim, N, N)

  out_mat = torch.bmm(left_mat, right_mat)

  # reshape back to (B,N,N,H)
  out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

  # Output layernorm over hidden dim
  out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
  )

  # Apply output gate
  out_norm = out_norm * out_gate

  # Final projection
  out_final = F.linear(out_norm, w["to_out.weight"])

  # Cast back to float32
  return out_final.to(torch.float32)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul3_masked_kernel(
    a_ptr, mask_ptr, c_ptr, out_ptr,
    total, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * mask * c
    a,c: [total] half, mask: [total//hidden_dim] half (broadcasted)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    c = tl.load(c_ptr + offs, mask=mask, other=tl.float16(0.0))

    idx_mask = offs // hidden_dim
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=tl.float16(0.0))

    out = a * m * c
    tl.store(out_ptr + offs, out, mask=mask)

def fused_mul3_masked(a: torch.Tensor, mask: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise multiplication of three tensors where mask has shape
    (B, N, N) and is broadcast over the hidden dimension.

    All tensors must be half‑precision and have matching shapes except the mask,
    which is missing the hidden dimension.
    """
    assert a.shape == c.shape
    # ensure contiguous storage
    a = a.contiguous()
    c = c.contiguous()
    mask = mask.contiguous().view(-1)  # flatten mask
    out = torch.empty_like(a)

    total = a.numel()
    hidden = a.shape[-1]

    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        a,
        mask,
        c,
        out,
        total,
        hidden,
        BLOCK_SIZE=BLOCK,
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator (AlphaFold3).

    The implementation operates largely in fp16 to exploit Tensor‑cores, using
    a small Triton kernel to fuse the optional mask with the left/right gates.
    The core O(N³) contraction is carried out with a batched GEMM (torch.bmm)
    on fp16 data. All LayerNorms and linear projections are performed in fp16
    as well; the final result is cast back to fp32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast everything to fp16 (half) once
    # ------------------------------------------------------------------
    # Input tensor
    x = inp.to(torch.float16)

    # Convert weights to half‑precision (no bias in any linear layer)
    w = {k: v.to(torch.float16) for k, v in weights.items()}

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 2. Linear projections to the hidden dimension (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, w["left_proj.weight"])    # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating tensors (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, w["out_gate.weight"]))  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Apply mask and gates – fused Triton kernel when a mask is present
    # ------------------------------------------------------------------
    if nomask:
        # No mask: simple element‑wise multiplication in PyTorch
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Fuse mask, projection and gate in a single Triton kernel.
        # mask is [B,N,N] (float or bool); we treat it as 0/1 fp16.
        mask_h = mask.to(torch.float16)
        left = fused_mul3_masked(left_proj, mask_h, left_gate)
        right = fused_mul3_masked(right_proj, mask_h, right_gate)

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batch of matrix multiplications in fp16.
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the batch axis for a single bmm call.
    # left_perm: (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_perm corresponds to right transposed on the summed dimension k:
    #   right[b,j,k,d] -> right_T[b,d,j,k]
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch+hidden into one dimension for torch.bmm
    left_mat = left_perm.view(B * hidden_dim, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N) # [B*H, N, N]

    # Batched matrix multiplication (fp16) → fp16 result
    out_mat = torch.bmm(left_mat, right_mat)         # [B*H, N, N]

    # Restore the original layout: (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 7. Apply the output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate  # element‑wise, fp16

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, w["to_out.weight"])  # [B,N,N,dim] (fp16)

    # Cast back to float32 for the downstream model
    return out_final.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise product a * mask * c
#   a,c : [B,N,N,hidden]  (flattened as 1‑D half tensors)
#   mask: [B,N,N]         (flattened as 1‑D half tensor, broadcasted over hidden)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,                 # left projection    (half*)
    mask_ptr,              # mask (broadcast)  (half*)
    c_ptr,                 # gate tensor        (half*)
    out_ptr,               # output tensor      (half*)
    total,                 # total number of elements = B*N*N*hidden
    hidden_dim,            # hidden dimension (int64)
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * mask * c  (mask is broadcasted over the hidden dimension)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total

    # Load the three operands
    a = tl.load(a_ptr + offs, mask=active, other=tl.float16(0.0))
    c = tl.load(c_ptr + offs, mask=active, other=tl.float16(0.0))

    # Broadcasted mask: one mask element per (b,i,k) → index = offs // hidden_dim
    mask_index = offs // hidden_dim
    m = tl.load(mask_ptr + mask_index, mask=active, other=tl.float16(0.0))

    out = a * m * c
    tl.store(out_ptr + offs, out, mask=active)


def fused_mul3_masked(a: torch.Tensor,
                      mask: torch.Tensor,
                      c: torch.Tensor) -> torch.Tensor:
    """
    Fuse ``a * mask[...,None] * c`` where ``mask`` lacks the hidden dimension.
    All tensors must be ``torch.float16``.
    """
    assert a.shape == c.shape, "projection and gate must share the same shape"
    # Ensure contiguous storage for the kernel
    a = a.contiguous()
    c = c.contiguous()
    # flatten mask to 1‑D half tensor
    mask = mask.contiguous().view(-1)

    out = torch.empty_like(a)
    total = a.numel()
    hidden = a.shape[-1]

    BLOCK = 1024  # can be tuned; 1024 works well on an H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        a,
        mask,
        c,
        out,
        total,
        hidden,
        BLOCK_SIZE=BLOCK,
    )
    return out

# Triton‑accelerated “outgoing” TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (AlphaFold3).

    The implementation is heavily fused in fp16:
        • All linear layers, LayerNorms and activations run in half‑precision.
        • The optional mask is fused with the left/right gates via a tiny Triton kernel.
        • The core O(N³) contraction is performed as a batched GEMM (torch.bmm) on
          fp16 tensors, which lets cuBLAS/Tensor‑cores run at peak throughput.
        • The final result is cast back to fp32.

    Expected input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 0. Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Cast everything to fp16 once
    # ------------------------------------------------------------------
    x = inp.to(torch.float16)                     # [B,N,N,dim] fp16
    w = {k: v.to(torch.float16) for k, v in weights.items()}  # half‑precision weights

    # ------------------------------------------------------------------
    # 2. Layer‑norm on the input (fp16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) → hidden dimension
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, w["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x_norm, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid of linear layers) – still fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, w["out_gate.weight"]))  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5. Apply mask (if any) and gates – fused Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask may be bool or float – cast to fp16
        mask_h = mask.to(torch.float16)
        left = fused_mul3_masked(left_proj, mask_h, left_gate)
        right = fused_mul3_masked(right_proj, mask_h, right_gate)

    # ------------------------------------------------------------------
    # 6. Core contraction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (fp16) over the hidden dimension.
    # ------------------------------------------------------------------
    # Bring hidden dimension to the batch axis for a single bmm call.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right must be transposed on the summed axis k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)     # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)   # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)           # [B*H, N, N] fp16

    # Restore original layout (B,N,N,hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output layer‑norm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 8. Apply output gate (fp16) and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise product
    out_final = F.linear(out_norm, w["to_out.weight"])  # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9. Cast back to float32 for downstream consumption
    # ------------------------------------------------------------------
    return out_final.to(torch.float32)
