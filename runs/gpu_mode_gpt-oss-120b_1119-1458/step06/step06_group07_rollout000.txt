```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    hidden_dim, size,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = offs < size
    # Load tensor and gate
    x = tl.load(x_ptr + offs, mask=mask_valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)

    # compute mask index: each hidden_dim group corresponds to one mask element
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask_valid)

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] using
    batched matrix multiplication on fp16 tensors.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right transpose across K dimension

"""
TriMul – Outgoing version (AlphaFold3)

Forward‑only implementation of the Triangle Multiplicative Update (TriMul)
operator for tensors of shape [B, N, N, C].  The kernel follows the
reference PyTorch implementation but aggressively fuses the mask·gate
multiplication with a small Triton kernel and performs the heavy
O(N³) contraction with a batched FP16 GEMM (cuBLAS).  The resulting
tensor is returned as float‑32.

Algorithm
---------
1. Layer‑norm the 4‑D input in fp32 (numerical stability) and cast to fp16.
2. Linear projections → left/right tensors (fp16).
3. Compute three gating tensors (fp16) with sigmoids.
4. If a mask exists, fuse “tensor * gate * mask” for left and right in‑place
   using a Triton kernel; otherwise apply gates only.
5. Contract over the third axis:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched matrix multiplication (left @ rightᵀ) on fp16.
6. Apply a second LayerNorm (fp16) on the hidden dimension.
7. Multiply by the out‑gate and project back to the original channel dimension.
8. Cast the final result to fp32.

Only the mask‑gate fusion is hand‑written in Triton – all other
operations use highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: fused mask·gate·tensor multiplication
# --------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated (flattened)
    gate_ptr,       # *fp16  gate tensor (same shape as x)
    mask_ptr,       # *fp16  mask tensor (flattened, shape [B*N*N])
    hidden_dim,     # i32    hidden dimension H
    size,           # i32    total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask_valid = offs < size                          # out‑of‑bounds guard

    # Load operand tensors (fp16)
    x = tl.load(x_ptr + offs, mask=mask_valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)

    # Compute mask index: each contiguous hidden‑dim block shares one mask entry
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)   # default: keep

    # Fuse: x = x * g * m
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask_valid)


# --------------------------------------------------------------
# Helper: batched GEMM (fp16) performing the O(N³) contraction
# out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H)  FP16
    Returns:
        out: (B, N, N, H) FP16
    """
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Compute left @ rightᵀ for each (b, h) pair → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
