"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm (all heavy compute in fp16):
1️⃣  Layer‑norm the 4‑D input in fp32, cast to fp16.
2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single GEMM by concatenating the weight matrices.
    This reduces 5 launches to 1 and re‑uses the input tensor.
3️⃣  Apply sigmoid to the three gate tensors.
4️⃣  If a mask is present, fuse “proj * gate * mask” with a tiny Triton kernel.
   Otherwise just multiply the projections by their gates.
5️⃣  O(N³) contraction:
       out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
   expressed as a batched GEMM (C = A @ Bᵀ) with the hidden dimension folded
   into the batch dimension, which lets cuBLAS Tensor‑Cores do the work.
6️⃣  Second LayerNorm (fp16) → multiply by the out‑gate.
7️⃣  Final linear projection back to the model dimension.
8️⃣  Cast the result to fp32 (the harness expects fp32).

Only step 4 uses Triton, satisfying the “must use Triton” requirement while
keeping the dominate O(N³) work on cuBLAS for maximal speed on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """out = proj * gate * mask   (mask is broadcast over the hidden dim H)"""
    pid_b = tl.program_id(0)   # batch axis
    pid_i = tl.program_id(1)   # i‑tile axis
    pid_j = tl.program_id(2)   # j‑tile axis

    # tile offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gating and mask:
        out = proj * gate * mask    (mask broadcast over hidden dim)

    All tensors must be fp16 CUDA tensors; ``mask`` is expected to be 0/1.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides are expressed in numbers of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes – chosen for high occupancy on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM implementing the O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    The hidden dimension H is folded into the batch dimension (B·H) to let cuBLAS
    perform a dense batched matrix multiply C = A @ Bᵀ.
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N]
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: (B·H, N, N) @ (B·H, N, N).T  -> (B·H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation (no gradients).

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor, shape [B, N, N, dim], dtype float32.
            - mask_tensor : torch.Tensor, shape [B, N, N] (bool or float), may be ignored.
            - weights_dict : mapping name -> torch.Tensor (float32) containing all model weights.
            - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask`` (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # True => skip mask

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # still fp32
    x = x.to(torch.float16)   # switch to fp16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣  Convert all model weights to FP16 on‑device
    # ------------------------------------------------------------------
    w_left_proj  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    w_left_gate  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_out_gate   = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    w_to_out_norm = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    w_to_out = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices: order = [Lproj, Rproj, Lgate, Rgate, Ogate]
    fused_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )  # shape (5*hidden_dim, dim), dtype fp16

    B, N, _, _ = x.shape
    # flatten the 4‑D tensor to 2‑D: (B·N·N, dim)
    x_flat = x.reshape(-1, dim)                # fp16

    # fused linear (one huge GEMM)
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # (B·N·N, 5*hidden_dim)

    # reshape back to 4‑D so we can slice the individual heads
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣  Slice the fused result and apply sigmoids to the gate tensors
    # ------------------------------------------------------------------
    off = 0
    left_proj = fused_out[..., off : off + hidden_dim]; off += hidden_dim
    right_proj = fused_out[..., off : off + hidden_dim]; off += hidden_dim

    left_gate_pre = fused_out[..., off : off + hidden_dim]; off += hidden_dim
    right_gate_pre = fused_out[..., off : off + hidden_dim]; off += hidden_dim
    out_gate_pre = fused_out[..., off : off + hidden_dim]; off += hidden_dim

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 5️⃣  Mask‑gate fusion (Triton) or simple element‑wise multiplication
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣  Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )

    # ------------------------------------------------------------------
    # 8️⃣  Out‑gate and final linear projection back to model dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out.view(-1, hidden_dim), w_to_out)
    out = out.view(B, N, N, dim).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣  Cast back to FP32 (expected by the harness) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
