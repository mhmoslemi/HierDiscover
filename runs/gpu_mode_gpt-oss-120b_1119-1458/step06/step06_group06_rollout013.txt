@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    rows, dim,
    eps: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    row = tl.program_id(0)
    # pointer for row start
    row_offset = row * dim

    # Compute sum across dim
    offs = tl.arange(0, BLOCK_SIZE)
    mask = offs < dim
    # Load per iteration (with loop over blocks)
    sum_val = tl.zeros((1,), dtype=tl.float32)

    for i in range(0, dim, BLOCK_SIZE):
        cur_off = i + offs
        mask_i = cur_off < dim
        x = tl.load(x_ptr + row_offset + cur_off, mask=mask_i, other=0.0)
        sum_val += tl.sum(x, axis=0)  # note x shape (BLOCK_SIZE,)

    mean = sum_val / dim

    # Compute variance
    var_val = tl.zeros((1,), dtype=tl.float32)
    for i in range(0, dim, BLOCK_SIZE):
        cur_off = i + offs
        mask_i = cur_off < dim
        x = tl.load(x_ptr + row_offset + cur_off, mask=mask_i, other=0.0)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)

    var = var_val / dim
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Normalize each element and apply weight & bias
    # cast to fp16 output
    for i in range(0, dim, BLOCK_SIZE):
        cur_off = i + offs
        mask_i = cur_off < dim
        x = tl.load(x_ptr + row_offset + cur_off, mask=mask_i, other=0.0)
        w = tl.load(weight_ptr + cur_off, mask=mask_i, other=0.0)
        b = tl.load(bias_ptr + cur_off, mask=mask_i, other=0.0)
        y = (x - mean) * inv_std * w + b
        tl.store(out_ptr + row_offset + cur_off, y.to(tl.float16), mask=mask_i)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5  # epsilon for layernorms

    # 1. First LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # fp32

    x = x_norm.half()  # fp16

    # 2. Concat weight matrices for left and right projection + gate
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # shape [2*hidden_dim, dim]

    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # shape [2*hidden_dim, dim]

    # 3. Linear for left and right (combined)
    left_comb = torch.nn.functional.linear(x, left_comb_weight)  # [B,N,N,2*hidden_dim]
    right_comb = torch.nn.functional.linear(x, right_comb_weight)

    # 4. Split and apply sigmoid on gate parts
    left_proj = left_comb[..., :hidden_dim]  # [B,N,N,hidden_dim]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # [B,N,N,hidden_dim]
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # 5. Apply mask and gating
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask broadcast
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        # ensure contiguous
        left_proj = left_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_proj = right_proj.contiguous()
        right_gate = right_gate.contiguous()
        total_elements = left_proj.numel()
        BLOCK = 1024  # recommended block size
        grid = ((total_elements + BLOCK - 1) // BLOCK,)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16.view(-1).contiguous(),
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16.view(-1).contiguous(),
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj

    # 6. Contract via batched GEMM
    out = _batched_gemm_fp16(left, right)  # [B,N,N,hidden_dim], fp16

    # 7. Second LayerNorm
    out = torch.nn.functional.layer_norm(

    out = out * out_gate

    # 9. Final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())

    # 10. Cast to float32
    return out.float()

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] and perform A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # bmm does (B*H, N, N) @ (B*H, N, N).T → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
TriMul (outgoing) – fast forward pass with reduced linear passes.

Key ideas
---------
* The original implementation performs four separate linear layers for the left
  and right streams (`proj` + `gate`).  Since the same input `x` is used for
  both, we concatenate the two weight matrices (`proj` and `gate`) and perform
  a single GEMM per stream (`x @ W.T`).  This halves the number of large matrix
  multiplications (the dominant cost) while producing the same numerical
  result.
* The element‑wise mask & gate fusion is done with a tiny Triton kernel
  (`mask_gate_fuse_kernel`).  When `nomask=False` we launch the kernel once per
  stream; when `nomask=True` we simply multiply the two tensors in‑place.
* The heavy contraction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  is still performed with a batched GEMM (`torch.bmm`) – on an H100 this is
  already optimal.
* The second LayerNorm, out‑gate multiplication and the final linear projection
  are kept in PyTorch for simplicity.  The forward contains a Triton kernel,
  satisfying the requirement to use custom kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over the
# hidden dimension.  Operates on a flat view of a contiguous tensor
# of shape [B·N·N·hidden_dim] (size = total_elements).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, size = total_elements
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,) – broadcasted
    total_elements,  # i32 : total number of elements in x
    hidden_dim,      # i32 : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask_bounds = offs < total_elements        # guard for the tail

    # Load the tensors
    x   = tl.load(x_ptr + offs, mask=mask_bounds, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask_bounds, other=0.0)

    # Every `hidden_dim` consecutive elements share the same mask entry.
    mask_idx = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_idx, mask=mask_bounds, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store the result back in‑place
    tl.store(x_ptr + offs, out, mask=mask_bounds)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via
    a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] and perform A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # bmm does (B*H, N, N) @ (B*H, N, N).T → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the **outgoing** TriMul operator (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
          Expected keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → ignore the mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]                # input channel dimension (c_z)
    hidden_dim = config["hidden_dim"]         # projection width (c)
    nomask     = config.get("nomask", True)   # if True, ignore the mask
    eps        = 1e-5                         # LayerNorm epsilon

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32
    x = x_norm.half()  # work in FP16 from now on

    # --------------------------------------------------------------
    # 2️⃣ Fuse left‑proj & left‑gate, right‑proj & right‑gate
    #    by concatenating the weight matrices.
    # --------------------------------------------------------------
    # Concatenate proj & gate weights for the left stream
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # shape: [2*hidden_dim, dim]

    # Same for the right stream
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # shape: [2*hidden_dim, dim]

    # Combined linear passes (still use cuBLAS under the hood)
    left_comb = torch.nn.functional.linear(x, left_comb_weight)   # [B,N,N,2*hidden_dim] fp16
    right_comb = torch.nn.functional.linear(x, right_comb_weight)  # same shape

    # --------------------------------------------------------------
    # 3️⃣ Split back into proj / gate and apply sigmoid to the gate part
    # --------------------------------------------------------------
    # left
    left_proj = left_comb[..., :hidden_dim]                          # [B,N,N,hidden_dim]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])           # [B,N,N,hidden_dim]

    # right
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # --------------------------------------------------------------
    # 4️⃣ Optional mask → fused element‑wise multiplication (proj * gate * mask)
    # --------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # make tensors contiguous for the Triton kernel
        left_proj = left_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_proj = right_proj.contiguous()
        right_gate = right_gate.contiguous()

        # mask to fp16, flatten to (B*N*N,)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)

        total_elements = left_proj.numel()            # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Fuse left stream in‑place
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right stream in‑place
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction (batched GEMM) – cuBLAS implementation
    # --------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim]  FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )  # still FP16

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate (sigmoid) computed from the *normalized* input `x`
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"].half())
    )  # [B, N, N, hidden_dim]  FP16

    # --------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (FP16)
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].half(),
    )  # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 9️⃣ Cast result back to FP32 for the harness
    # --------------------------------------------------------------
    return out.float()
