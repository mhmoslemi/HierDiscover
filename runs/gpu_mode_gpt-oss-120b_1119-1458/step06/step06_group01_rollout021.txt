@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # Program IDs
    pid_b = tl.program_id(0)  # batch index
    pid_i = tl.program_id(1)  # tile row
    pid_j = tl.program_id(2)  # tile col

    # Compute base indices
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Mask for i,j beyond N
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over hidden dimension tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Initialize accumulator
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # Loop over K dimension
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for k_tile in range(num_k_tiles):
            offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left tile (I,K)
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None] * stride_left_i
                + offs_k[None, :] * stride_left_j
                + offs_h[None, None, :] * stride_left_h
            )
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None] & mask_k[None, :] & mask_h[None, None, :],
                           other=0.0)
            # Load right tile transposed (K,J)
            # right_idx: // we need right[j,k,h], and we want shape (K,J) for dot
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_k[:, None] * stride_right_j  # using stride_right_j for k dimension
                + offs_j[None, :] * stride_right_i   # using stride_right_i for j dimension
                + offs_h[None, None, :] * stride_right_h
            )
            right = tl.load(right_ptrs,
                            mask=mask_k[:, None] & mask_j[None, :] & mask_h[None, None, :],
                            other=0.0)
            # Multiply accumulate
            # Use tl.dot: left shape (BLOCK_I, BLOCK_K) and right shape (BLOCK_K, BLOCK_J)
            # But we need cast to appropriate dtype for dot if possible (fp16->fp32)
            acc += tl.dot(left, right).to(tl.float32)

        # Write out tile after full K accumulation
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None] * stride_out_i
            + offs_j[None, :] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :])

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # hidden dimension loop
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator for each hidden channel
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # loop over K
        nz_k = tl.cdiv(N, BLOCK_K)
        for k_tile in range(nz_k):
            offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # load left tile shape (BLOCK_I, BLOCK_K, BLOCK_H)
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None] * stride_left_i
                + offs_k[None, :] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
            )
            # mask left
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k[None, :] & mask_h[None, None, :],
                other=0.0,
            )  # (BLOCK_I, BLOCK_K, BLOCK_H)

            # load right tile transposed: shape (BLOCK_K, BLOCK_J, BLOCK_H)
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_k[:, None] * stride_right_k
                + offs_j[None, :] * stride_right_i
                + offs_h[None, None, :] * stride_right_h
            )
            right_tile = tl.load(
                right_ptrs,
                mask=mask_k[:, None] & mask_j[None, :] & mask_h[None, None, :],
                other=0.0,
            )  # (BLOCK_K, BLOCK_J, BLOCK_H)

            # compute per hidden
            for hh in range(BLOCK_H):
                # apply mask for hidden channel
                mhh = mask_h[hh]
                if mhh:
                    a = left_tile[:, :, hh]   # shape (BLOCK_I, BLOCK_K)
                    b = right_tile[:, :, hh]   # shape (BLOCK_K, BLOCK_J)
                    acc[:, :, hh] += tl.dot(a, b)   # accumulate
        # store output for this hidden tile
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

"""
custom_kernel
--------------

Highly‑optimised forward pass for the “outgoing” TriMul operation used in
AlphaFold‑3. The implementation fuses the five linear projections into a
single GEMM, removes the unnecessary mask‑gate Triton kernels from the
reference, and performs the O(N³) contraction with a custom Triton kernel
that works directly on the original [B, N, N, H] layout (no permute +
contiguous copy). All heavy work is done in FP16 and accumulated in FP32;
the final result is returned as FP32.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) and cast to FP16.
2. One fused linear layer computes the three gating tensors and the two
   projection tensors in a single matrix‑multiply:
   combined = x_norm @ W_combined.T   where   W_combined =
   [W_left_proj; W_left_gate; W_right_proj; W_right_gate; W_out_gate].
3. Split the combined result, apply sigmoids to the three gate tensors,
   multiply projections by their gates and (optionally) apply the mask.
4. Contract left and right tensors:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   via a batched GEMM written in Triton (`tri_mul_outgoing_kernel`).
5. Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection.
6. Cast to FP32 for the harness.

The Triton kernel tiles the i, j and hidden dimensions while looping over
the reduction dimension k.  Each tile accumulates in FP32 and stores the
result in FP16.  Tile sizes were chosen (32×32×8) to keep shared‑memory
usage modest while providing high occupancy on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched O(N³) contraction (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # *fp16 pointers
    B, N, H,                               # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    The tensors are laid out as (B, N, N, H) with H being the
    innermost (fastest) dimension.  This kernel treats each (b, h)
    pair as an independent matrix‑multiply and tiles i, j, k, h.
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # ------------------------------------------------------------------
    # i / j tile coordinates
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator: shape (BLOCK_I, BLOCK_J, BLOCK_H) in FP32
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # ----------------------------------------------------------------
        # Reduction over the k dimension
        # ----------------------------------------------------------------
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for k_tile in range(num_k_tiles):
            offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # ------------------------------------------------------------
            # Load left tile  : shape (BLOCK_I, BLOCK_K, BLOCK_H)
            # ------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None] * stride_left_i
                + offs_k[None, :] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
            )
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k[None, :] & mask_h[None, None, :],
                other=0.0,
            )  # fp16

            # ------------------------------------------------------------
            # Load right tile transposed (K × J) : shape (BLOCK_K, BLOCK_J, BLOCK_H)
            # ------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_k[:, None] * stride_right_k
                + offs_j[None, :] * stride_right_i
                + offs_h[None, None, :] * stride_right_h
            )
            right_tile = tl.load(
                right_ptrs,
                mask=mask_k[:, None] & mask_j[None, :] & mask_h[None, None, :],
                other=0.0,
            )  # fp16

            # ------------------------------------------------------------
            # Compute dot‑products for each hidden channel in the tile.
            # ------------------------------------------------------------
            for hh in range(BLOCK_H):
                a = left_tile[:, :, hh]      # (BLOCK_I, BLOCK_K)
                b = right_tile[:, :, hh]     # (BLOCK_K, BLOCK_J)
                acc[:, :, hh] += tl.dot(a, b)   # FP32 accumulation

        # ------------------------------------------------------------
        # Store the result tile (cast back to fp16)
        # ------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: batched contraction using the Triton kernel above
# ----------------------------------------------------------------------
def _triton_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a
    custom Triton kernel (no copies, work directly on the [B,N,N,H] layout).
    Both inputs must be FP16 CUDA tensors.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides are expressed in element counts (not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # Tile sizes – chosen for good occupancy on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point – forward pass of the outgoing TriMul (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ---------------------------------
    # Unpack arguments
    # ---------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣  First LayerNorm (FP32) → cast to FP16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                         # FP32
    x = x.to(torch.float16)   # FP16 for the rest

    # -------------------------------------------------
    # 2️⃣  Fuse the five linear transforms into one GEMM
    # -------------------------------------------------
    # Cast all linear weights to FP16 (on‑device)
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Build the stacked weight matrix: rows = 5 * hidden_dim, cols = dim
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim), FP16

    # Single linear pass (X_norm [B,N,N,dim] @ combined_weight.T) → [B,N,N,5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_weight)  # FP16

    # -----------------------------------------------------------------
    # 3️⃣  Split and apply gates (sigmoid) + optional mask
    # -----------------------------------------------------------------
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )
    # Free the combined tensor early
    del combined

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate   # [B,N,N,hidden_dim]  FP16
    right = right_proj * right_gate

    if not nomask:
        # mask is broadcast over the hidden dimension
        mask_fp16 = mask_tensor.to(dtype=torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 4️⃣  O(N³) contraction via custom Triton kernel
    # -------------------------------------------------
    out = _triton_contraction(left, right)   # [B,N,N,hidden_dim] FP16

    # -------------------------------------------------
    # 5️⃣  Second LayerNorm (FP16)
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -------------------------------------------------
    # 6️⃣  Gating with out_gate and final projection back to dim
    # -------------------------------------------------
    out = out * out_gate                       # element‑wise gated, FP16
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 7️⃣  Cast back to FP32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
