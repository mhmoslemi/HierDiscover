"""
Optimized outgoing TriMul forward pass (AlphaFold‑3).

Key ideas
---------
1️⃣ Fuse all linear projections (left/right projections, the three gating
   projections and the output gating) into a **single** matrix multiplication.
   The weight matrix is built as a concatenation of the 5 weight tensors,
   reducing the number of passes over the input tensor from 5 → 1.

2️⃣ Fuse the element‑wise multiplications (projection • gate) **and** the
   optional mask in a Triton kernel.  The kernel reads the concatenated
   projection results, computes the three sigmoid gates, applies the mask
   (if present) and writes three compact tensors: ``left``, ``right`` and
   ``out_gate`` (all ``float16``).

3️⃣ Perform the O(N³) contraction with a single batched GEMM
   (`torch.bmm`) after reshaping the tensors to [B·H, N, N].  This
   leverages Tensor‑Core kernels on the H100.

4️⃣ The remaining inexpensive operations (LayerNorms, final gating,
   output projection) stay in PyTorch but use half‑precision where
   possible.  The final result is cast back to ``float32``.

The implementation satisfies the requirement of having at least one
custom Triton kernel while significantly reducing data movement and
overall runtime.
"""

    # scalars
    hidden_dim,          # i32   : H
    total_elements,      # i32   : B*N*N*H  (elements per left/right/out_gate)
    BLOCK: tl.constexpr, # compile‑time tile size
    has_mask: tl.constexpr,  # compile‑time flag (0/1)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear index for (b,i,j,h)
    mask = offs < total_elements                # guard

    # ------------------------------------------------------------------
    # Decode the 5‑group index.
    #   group_idx = (b,i,j)  flattened without the channel dimension.
    #   channel   = hidden index h.
    # ------------------------------------------------------------------
    group_idx = offs // hidden_dim
    channel   = offs % hidden_dim

    # Each spatial location has 5*H consecutive values in the concatenated
    # tensor (projL, gateL, projR, gateR, out_gate).
    stride = hidden_dim * 5
    base = group_idx * stride + channel

    # Load the 5 source scalars.
    projL      = tl.load(concat_ptr + base,                      mask=mask, other=0.0)
    gateL_log  = tl.load(concat_ptr + base + hidden_dim,        mask=mask, other=0.0)
    projR      = tl.load(concat_ptr + base + 2 * hidden_dim,    mask=mask, other=0.0)
    gateR_log  = tl.load(concat_ptr + base + 3 * hidden_dim,    mask=mask, other=0.0)
    out_gate_log = tl.load(concat_ptr + base + 4 * hidden_dim, mask=mask, other=0.0)

    # Sigmoid (gate = 1/(1+exp(-logits)) )
    gateL = 1.0 / (1.0 + tl.exp(-gateL_log))
    gateR = 1.0 / (1.0 + tl.exp(-gateR_log))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate_log))

    # Optional mask multiplication (mask is broadcast over H)
    if has_mask:
        # mask index same as group_idx (no channel dimension)
        m = tl.load(mask_ptr + group_idx, mask=mask, other=1.0)
        left_val  = projL * gateL * m
        right_val = projR * gateR * m
    else:
        left_val  = projL * gateL
        right_val = projR * gateR

    # Write results
    tl.store(left_ptr + offs,      left_val,    mask=mask)
    tl.store(right_ptr + offs,     right_val,   mask=mask)
    tl.store(out_gate_ptr + offs,  out_gate,    mask=mask)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Heavy O(N³) contraction via batched GEMM.
    left/right : [B, N, N, H] (fp16)
    Returns    : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel is a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (B*H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)         # (B*H, N, N)
    # Batched matrix multiplication (A @ Bᵀ)
    c = torch.bmm(a, b.transpose(1, 2))                                   # (B*H, N, N)
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) or None
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Build a single concatenated weight matrix for all five
    #    linear projections (left_proj, left_gate, right_proj, right_gate,
    #    out_gate).  All original Linear layers have ``bias=False``.
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output dimension: shape (5*H, dim)
    concat_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # (5*hidden_dim, dim)

    # -----------------------------------------------------------------
    # 4️⃣ Combined linear projection
    # -----------------------------------------------------------------
    # Result shape: [B, N, N, 5*H] (fp16)
    concat_proj = torch.nn.functional.linear(x, concat_weight)

    # -----------------------------------------------------------------
    # 5️⃣ Allocate output buffers for the fused kernel
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    left      = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)   # will hold the final gating tensor

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask handling (broadcast over hidden_dim)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and contiguous; we treat ``True`` as 1.0.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        # Dummy tensor – never read when has_mask=0
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)

    total_elements = left.numel()               # B * N * N * H
    BLOCK = 1024                                 # fits comfortably on H100
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    # -----------------------------------------------------------------
    # 7️⃣ Fuse gating, mask and split the concatenated projection.
    # -----------------------------------------------------------------
    has_mask_flag = 0 if nomask else 1
    fused_lr_gate_kernel[grid](
        concat_proj,
        left,
        right,
        out_gate,
        mask_fp16,
        hidden_dim,
        total_elements,
        BLOCK=BLOCK,
        num_warps=4,
        has_mask=has_mask_flag,
    )

    # -----------------------------------------------------------------
    # 8️⃣ O(N³) contraction via batched GEMM
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 9️⃣ Second LayerNorm (over hidden dimension)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 🔟 Gating after LayerNorm
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to ``dim``
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
