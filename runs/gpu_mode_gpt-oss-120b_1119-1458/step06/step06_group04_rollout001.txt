def _batched_gemm_fp16(left, right):
    B,N,_,H = left.shape
    left_ = left.permute(0,3,1,2).contiguous() # [B,H,N,N]
    right_ = right.permute(0,3,1,2).contiguous()
    left_mat = left_.reshape(B*H, N, N)
    right_mat = right_.reshape(B*H, N, N)
    # Compute left_mat @ right_mat.transpose(-2,-1)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

def mask_gate_fuse_kernel(
    x_ptr,
    gate_ptr,
    mask_ptr,
    hidden_dim,
    size,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index: each hidden block size hidden_dim.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fused op
    y = x * g * m

    tl.store(x_ptr + offs, y, mask=mask)
```

"""
TriMul outgoing forward pass (no autograd) – Triton‑accelerated mask‑gate fusion.

Algorithm
---------
1️⃣ First LayerNorm (float32) on the 4‑D input.
2️⃣ Cast to FP16 for all heavy arithmetic.
3️⃣ Linear projections:
   * left  = x @ W_left   (shape [B,N,N,hidden_dim])
   * right = x @ W_right
   * left_gate, right_gate, out_gate = σ(x @ W_gate)  (sigmoid)
4️⃣ Optional mask (shape [B,N,N]) is fused with the element‑wise
   product `x * gate` by a tiny Triton kernel (`mask_gate_fuse_kernel`).
   This updates `left` and `right` in‑place:
   left  ← left  * left_gate  * mask
   right ← right * right_gate * mask
5️⃣ Core O(N³) contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (fp16) using `torch.bmm`.
6️⃣ Second LayerNorm (fp16) on the hidden dimension.
7️⃣ Multiply by `out_gate` (element‑wise) and apply final
   linear projection back to the original channel dimension.
8️⃣ Cast the result back to float32 (the harness expects FP32).

The kernel only handles the mask‑gate fusion; the remaining
operations are performed with highly‑optimized PyTorch kernels
(e.g. cuBLAS for the GEMM).  All arithmetic, except the initial
LayerNorm, runs in FP16 to exploit Tensor‑Cores on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and tensor multiplication.
# Operates on flatted FP16 tensors.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gating tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → size of hidden dimension
    total_elements, # i32   → total number of elements in x (B*N*N*hidden_dim)
    BLOCK: tl.constexpr,  # compile‑time block size (≤1024)
):
    """
    In‑place computes:  x = x * gate * mask

    `mask` is broadcasted across `hidden_dim` dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                # [BLOCK]
    mask = offs < total_elements                             # [BLOCK] bool

    # Load the data for this block
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)          # fp16
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)       # fp16

    # Compute the mask index (each hidden_dim values share the same mask element)
    mask_idx = offs // hidden_dim                            # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # fp16

    # Fuse the three multiplications
    y = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, y, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# Works entirely in FP16 and leverages cuBLAS via torch.bmm.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        FP16 tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        FP16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # Bring hidden dimension to batch dimension: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)^T  →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Restore original shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input_tensor: [B, N, N, dim]  (float32)
        torch.Tensor,          # mask_tensor : [B, N, N]      (float32/bool or None)
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → the mask should be ignored
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for everything that follows
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights once to FP16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # -----------------------------------------------------------------
    # No bias in these layers → bias=None
    left = torch.nn.functional.linear(x, left_proj_w)      # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)    # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (FP16) – sigmoid
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Apply optional mask and fuse with gating via Triton.
    #    The fusion updates `left` and `right` in‑place.
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to FP16 and flatten to [B*N*N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                      # shape [B*N*N]

        # Ensure tensors are contiguous for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_elems = left.numel()                          # B * N * N * hidden_dim
        BLOCK = 1024                                        # ≤ max threads per block
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction via batched GEMM (FP16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim] (FP16)

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Gating with out_gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gate
    out = torch.nn.functional.linear(out, to_out_w)      # [B,N,N,dim] (FP16)

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
