"""
TriMul (outgoing) forward-pass implemented with heavy‑lifting in FP16.
---------------------------------------------------------------
Algorithm
---------

1️⃣  Layer‑norm the 4‑D input (float32) → FP16.
2️⃣  Fuse the “proj” and “gate” linear layers for the *left* and *right*
    branches by concatenating their weight matrices (size 2·H×D) and
    performing a single GEMM per branch.
    The gating tensors are obtained with a sigmoid.
3️⃣  Optional mask‑gating (mask ∈ {0,1}) is fused into one Triton kernel:
    `out = proj * gate * mask` (the mask is broadcast over the hidden dim).
    If no mask is used we simply do an element‑wise product.
4️⃣  Core O(N³) contraction:
    `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.
    This is a batched matrix‑multiply (A @ Bᵀ) with batch size B·H.
    We reshape left/right to (B·H, N, N) and call ``torch.bmm`` which
    runs on H100 Tensor‑Cores in FP16.
5️⃣  Second LayerNorm over the hidden dimension (FP16) → ``to_out_norm``.
6️⃣  Apply the output‑gate (element‑wise sigmoid) and the final linear
    projection back to the original channel dimension (FP16).
7️⃣  Cast to float32 for the external harness.

Only the mask‑gate fusion (step 3) is executed on a custom Triton kernel,
satisfying the “use Triton” requirement.  All other heavy arithmetic
runs in FP16 to fully exploit the H100 Tensor‑Cores.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask‑gate multiplication (in‑place).
#   x  – tensor to be updated   (flattened FP16)
#   g  – gate tensor (same shape as x, already sigmoid‑activated)
#   m  – mask tensor (flattened FP16, shape B·N·N, broadcast over hid)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  – tensor to be updated (flattened)
    g_ptr,          # *fp16  – gate tensor (flattened)
    mask_ptr,       # *fp16  – mask tensor (flattened), one value per (b,i,j)
    hidden_dim,     # i32    – hidden dimension H
    size,           # i32    – total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load tensor data
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(g_ptr + offs, mask=mask, other=0.0)

    # Compute the index into the mask (broadcast over H)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1

    # Fuse multiply
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using a batched GEMM.
    Input tensors are FP16 with shape [B, N, N, H] and the result has
    shape [B, N, N, H].

    The tensors are reshaped to (B·H, N, N) and a batched matrix‑multiply
    with transpose on the right operand implements:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    """
    B, N, _, H = left.shape
    # (B, H, N, N) – make hidden dim the batch dimension
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Compute A @ Bᵀ  (notice the transpose on the second operand)
    c = torch.bmm(a, b.transpose(1, 2))               # (B*H, N, N)
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, dim]  (float32)
        torch.Tensor,          # mask tensor  [B, N, N]       (float32 or bool) – may be None
        Dict[str, torch.Tensor],  # model weights (float32)
        Dict[str, Any],           # config dict: must contain "dim" and "hidden_dim"
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32, bool or None)
        - weights_dict : mapping from string → torch.Tensor (float32)
        - config_dict  : contains at least ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default=True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True ⇒ no mask (ignore mask_tensor)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32) – stable for the pre‑norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)

    # -----------------------------------------------------------------
    # 2️⃣  Cast to FP16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣  Cast model weights to FP16 once
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden_dim)

    # -----------------------------------------------------------------
    # 4️⃣  Fuse proj+gate linear layers (two calls instead of four)
    # -----------------------------------------------------------------
    # Left branch: concatenate proj and gate weights → 2·H × D
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, D)
    left_all = torch.nn.functional.linear(x, left_cat_w)       # (B,N,N,2*H)
    left = left_all[..., :hidden_dim]          # projection output
    left_gate = torch.sigmoid(left_all[..., hidden_dim:])   # gate output (sigmoid)

    # Right branch: same fusion
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, D)
    right_all = torch.nn.functional.linear(x, right_cat_w)        # (B,N,N,2*H)
    right = right_all[..., :hidden_dim]
    right_gate = torch.sigmoid(right_all[..., hidden_dim:])

    # -----------------------------------------------------------------
    # 5️⃣  Apply mask and fuse with gating via Triton (if a mask is present)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to FP16 and flatten for the kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)            # shape [B*N*N]

        total_elements = left.numel()            # B·N·N·H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        # In‑place fusion: left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Right side – same procedure
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just apply the gating
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣  Output‑gate (computed from the original normalized input)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7️⃣  Core O(N³) contraction via batched GEMM (FP16, Tensor‑Core)
    # -----------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = _batched_gemm_fp16(left, right)   # (B,N,N,H)  FP16

    # -----------------------------------------------------------------
    # 8️⃣  Second LayerNorm (FP16) on the hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # (B,N,N,H)  FP16

    # -----------------------------------------------------------------
    # 9️⃣  Gating with out_gate and final linear projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate                         # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)  FP16

    # -----------------------------------------------------------------
    # 🔟  Cast back to float32 for the external harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)

* Reducing the number of linear transforms from five to three by concatenating projection and gate weights.
* Performing all heavy arithmetic in FP16 to exploit H100 Tensor‑Cores.
* Using a custom Triton kernel to fuse mask‑application with gating, satisfying the requirement to run at least part of the computation on Triton.
* Keeping memory moves minimal (only the necessary reshapes for the batched GEMM) and performing the O(N³) contraction with a single cuBLAS‑backed ``torch.bmm`` call.
