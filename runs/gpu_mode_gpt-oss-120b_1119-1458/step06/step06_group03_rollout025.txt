    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,

):

    pid_j = tl.program_id(1)  # j tile
    pid_batch = tl.program_id(2) # fused batch index

    # compute starting offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute batch dims
    total_batch = B * H
    # compute b and h
    b = pid_batch // H
    h = pid_batch % H

    # base pointers for this batch
    left_batch_ptr = left_ptr + b * stride_left_b + h * stride_left_h
    right_batch_ptr = right_ptr + b * stride_right_b + h * stride_right_h
    out_batch_ptr = out_ptr + b * stride_out_b + h * stride_out_h

    # define range offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)  # shape (BLOCK_M,)
    offs_j = j_start + tl.arange(0, BLOCK_N)  # shape (BLOCK_N,)

    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_start * BLOCK_K

"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣ First LayerNorm (FP32) for numerical stability.
2️⃣ Cast to FP16 and fuse *all* linear projections (left/right projections,
   their gates, and the output gate) into a *single* matrix multiplication.
   This reduces memory traffic: the 4‑D input tensor is read only once.
3️⃣ Apply the element‑wise gate and (optional) mask using a tiny Triton kernel
   that fuses the three multiplicative factors.
4️⃣ Heavy O(N³) contraction:
        out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
   is implemented as a batched GEMM in Triton, avoiding the expensive
   `permute + contiguous` copy used by the PyTorch reference.
5️⃣ Second LayerNorm (still FP16) → multiply by the pre‑computed `out_gate`.
6️⃣ Final linear projection back to `dim` (FP16) and cast to FP32 for the
   harness output.

All heavy compute (steps 2‑4) run in FP16 on Tensor‑Cores, while the first
LayerNorm runs in FP32 for stability.  The implementation requires only the
forward pass – no autograd support.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Fuse (tensor * gate * mask) – very small kernel.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,               # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,            # *fp16 : gate tensor, same shape as x
    mask_ptr,            # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elts

    # mask is broadcast across the hidden dimension
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  [B,N,N,H]
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right [B,N,N,H]
    stride_rb, stride_ri, stride_rk, stride_rh,
    # strides for out   [B,N,N,H]
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)                     # tile over i (row)
    pid_j = tl.program_id(1)                     # tile over j (col)
    pid_batch = tl.program_id(2)                 # fused batch = b * H + h

    # starting coordinates of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # decode fused batch index into (b, h)
    b = pid_batch // H
    h = pid_batch % H

    # base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # ranges for the tile (with bounds check)
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (more stable, still Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_M, BLOCK_K)  – rows i, cols k
        a_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right: we need (BLOCK_K, BLOCK_N) i.e. rows k, cols j
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot‑product accumulation
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N)

    # write the tile back (cast to FP16)
    out_tile = acc.to(tl.float16)
    c_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(c_ptrs, out_tile, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data :
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : shape [B, N, N, dim] (float32)
        - mask_tensor  : shape [B, N, N] (float32 / bool) – may be unused
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – more stable than FP16 here
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)   # [B, N, N, dim] – FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections (left/right & their gates + out_gate)
    # ------------------------------------------------------------------
    # We concatenate the five weight matrices (each hidden_dim × dim) along the
    # out‑features axis and execute a *single* GEMM.
    comb_weights = torch.cat(
        [
            weights["left_proj.weight"],    # (H, dim)
            weights["left_gate.weight"],    # (H, dim)
            weights["right_proj.weight"],   # (H, dim)
            weights["right_gate.weight"],   # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, dim)

    # flatten the 4‑D tensor to (B·N·N, dim) for the linear
    x_flat = x.view(-1, dim)                         # (M, dim)
    proj_all = torch.nn.functional.linear(x_flat, comb_weights)  # (M, 5*H)

    # reshape back to 4‑D and split
    proj_all = proj_all.view(B, N, N, 5 * hidden_dim)   # (B,N,N,5H)
    cur = 0
    left_proj = proj_all[..., cur:cur + hidden_dim].contiguous()
    cur += hidden_dim
    left_gate = torch.sigmoid(
        proj_all[..., cur:cur + hidden_dim].contiguous()
    )
    cur += hidden_dim
    right_proj = proj_all[..., cur:cur + hidden_dim].contiguous()
    cur += hidden_dim
    right_gate = torch.sigmoid(
        proj_all[..., cur:cur + hidden_dim].contiguous()
    )
    cur += hidden_dim
    out_gate = torch.sigmoid(
        proj_all[..., cur:cur + hidden_dim].contiguous()
    )  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – fuse (proj * gate * mask) in‑place via Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()  # B * N * N * H
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via a custom Triton kernel
    # ------------------------------------------------------------------
    # Allocate output tensor
    out = torch.empty_like(left)

    # Strides for the 4‑D tensors (B, N, N, H)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes – tweaked for H100 Tensor‑Core performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid layout: (tiles over i, tiles over j, batch*hidden)
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_batch = B * hidden_dim
    grid = (grid_i, grid_j, grid_batch)

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → full utilization on H100
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16), gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Layer‑norm across the hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    # Apply the output gate computed earlier
    out = out * out_gate

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim), FP16

    # ------------------------------------------------------------------
    # 7️⃣ Return FP32 for the evaluation harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
