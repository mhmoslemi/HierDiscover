    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"])
    # 2. Cast to fp16
    x = x_norm.half()

    # 3. Prepare concatenated weights for left and right linear+gate
    # left_concat_weight: shape (2*hidden_dim, dim)
    left_concat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_concat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()

    left_concat = torch.nn.functional.linear(x, left_concat_weight)  # shape (b,n,n,2*hidden_dim)
    right_concat = torch.nn.functional.linear(x, right_concat_weight)

    # split
    left_proj = left_concat[..., :hidden_dim]
    left_gate_raw = left_concat[..., hidden_dim:]

        mask_flat = mask_fp16.view(-1) # B*N*N
        size = left_proj.numel()

        grid = (triton.cdiv(size, BLOCK),)

"""
Optimized outgoing TriMul (TriMul‑outgoing) forward pass.

Key optimizations
-----------------
1. **Fused Linear + Gate** – The two linear projections (proj & gate) for the
   “left” and “right” paths are combined into a single matmul by concatenating
   the weight matrices (shape (2·hidden_dim, dim)).  The same trick is used
   for the output gate (proj + gate + out‑gate) resulting in only **two**
   large linear layers instead of four.

2. **Mask‑gate fusion via Triton** – When a mask is present, a tiny Triton
   kernel fuses the element‑wise multiplication by the mask, the gate and
   the projection output (`x * gate * mask`) in‑place, avoiding an extra
   temporary and a second kernel launch.

3. **Batched GEMM for the O(N³) contraction** – The heavy contraction
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]` is performed with a
   standard batched GEMM (`torch.bmm`) on half‑precision data.  The tensors
   are merely permuted/re‑interpreted (no unnecessary copies) and the
   computation runs on tensor‑cores of the H100.

4. **Mixed precision** – All compute‑intensive operations run in `float16`,
   while the initial layer‑norm stays in `float32` for numerical stability.
   The final result is cast back to `float32` as required by the harness.

The implementation respects the required signature:
`custom_kernel(data) -> torch.Tensor` where `data` is a tuple
`(input_tensor, mask_tensor, weights, config)`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask * gate * tensor (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 input tensor (flattened)
    gate_ptr,       # *fp16 gate tensor (same shape as x)
    mask_ptr,       # *fp16 mask tensor (flattened, shape [B*N*N])
    hidden_dim,     # i32 hidden dimension H
    size,           # i32 total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Each mask entry corresponds to `hidden_dim` contiguous elements in x
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1

    # Fuse multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM (fp16) performing the O(N³) contraction.
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H) in half precision.
    Returns:
        out: (B, N, N, H) half precision.
    """
    B, N, _, H = left.shape
    # Move hidden dim to second axis (contiguous) and flatten batch*hidden
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Right is used transposed in the contraction
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched matmul: left_flat @ right_flatᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore original shape
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Arguments
    ---------
    data: (input_tensor, mask_tensor, weights, config)
        - input_tensor: shape [B, N, N, dim] (float32)
        - mask_tensor: shape [B, N, N] (bool / float)
        - weights: dict containing the model weights
        - config: contains "dim", "hidden_dim", and optional "nomask"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → keep FP32 for stability)
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim], dtype=float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy ops
    # -----------------------------------------------------------------
    x = x_norm.half()  # shape [B, N, N, dim], dtype=float16

    # -----------------------------------------------------------------
    # 3️⃣ Build fused linear weight matrices (fp16)
    # -----------------------------------------------------------------
    # left: proj + gate
    left_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # shape (2*hidden_dim, dim)

    # right: proj + gate
    right_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # shape (2*hidden_dim, dim)

    # out_gate is fused together with left side to avoid a separate matmul
    outgate_weight = weights["out_gate.weight"].half()  # shape (hidden_dim, dim)

    # Combine left_proj, left_gate, out_gate into a single weight matrix
    left_full_weight = torch.cat([left_weight, outgate_weight], dim=0)  # (3*H, dim)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections + gating (single matmul per side)
    # -----------------------------------------------------------------
    # left side produces [left_proj, left_gate, out_gate_raw]
    left_concat = torch.nn.functional.linear(x, left_full_weight)  # (B,N,N,3*H)
    left_proj = left_concat[..., :hidden_dim]                     # (B,N,N,H)
    left_gate_raw = left_concat[..., hidden_dim : 2 * hidden_dim]
    out_gate_raw = left_concat[..., 2 * hidden_dim :]            # (B,N,N,H)

    # right side produces [right_proj, right_gate]
    right_concat = torch.nn.functional.linear(x, right_weight)   # (B,N,N,2*H)
    right_proj = right_concat[..., :hidden_dim]                  # (B,N,N,H)
    right_gate_raw = right_concat[..., hidden_dim:]              # (B,N,N,H)

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_raw)  # (B,N,N,H)
    out_gate = torch.sigmoid(out_gate_raw)      # (B,N,N,H)

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fuse mask·gate·tensor multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Flatten mask to 1‑D fp16 for the kernel.  The mask is broadcast over H.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)               # shape [B*N*N]
        total_elems = left_proj.numel()               # B * N * N * H
        BLOCK = 1024                                   # tuning parameter (≤ 1024)
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left = left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just apply the gates (in‑place to save memory)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate
        # `*` creates a new tensor; we can optionally use `.mul_` to be in‑place:
        # left_proj.mul_(left_gate); right_proj.mul_(right_gate)

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – batched GEMM (fp16, tensor cores)
    # -----------------------------------------------------------------
    # left_proj and right_proj now hold the gated (and masked) tensors
    out = _batched_gemm_fp16(left_proj, right_proj)  # (B,N,N,H), dtype=half

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # -----------------------------------------------------------------
    # 8️⃣ Gating by out_gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (fp16)
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.float()
