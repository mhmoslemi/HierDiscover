def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    D = config["dim"]
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # 1 layernorm on float32
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x_norm.to(torch.float16)
    
    # fuse linear projections and gates into one linear
    # assemble fused weight matrix:
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape (5*H, D)
    
    # Linear transform: fused [B,N,N,5*H]
    fused_out = torch.nn.functional.linear(x, fused_weight)  # [B,N,N,5*H], fp16

    # split into components
    # slices indices:
    left_proj = fused_out[..., :H]
    left_gate = torch.sigmoid(fused_out[..., H:2*H])
    right_proj = fused_out[..., 2*H:3*H]
    right_gate = torch.sigmoid(fused_out[..., 3*H:4*H])
    out_gate = torch.sigmoid(fused_out[..., 4*H:5*H])
    
    # elementwise multiplication (apply gates)
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        # apply mask to left and right in-place using triton
        mask_flat = mask_tensor.to(torch.float16).contiguous().view(-1)  # shape [B*N*N]
        size = left.numel() # total elements of left and right
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)
        mask_mul_kernel[grid](
            left, mask_flat, H, size,
            BLOCK=BLOCK, num_warps=4,
        )
        mask_mul_kernel[grid](
            right, mask_flat, H, size,
            BLOCK=BLOCK, num_warps=4,
        )
    
    # heavy O(N^3) contraction: same as before
    # compute out = left @ right.T per head
    out = _batched_gemm_fp16(left, right)  # shape [B,N,N,H]
    
    # second layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    
    # apply out_gate gating
    out = out * out_gate
    
    # final projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))
    
    return out.to(torch.float32)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd), fused linear projections and mask handling.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # 1. LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x_norm.to(torch.float16)
    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    # 2. Fuse left projection and left gate
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0)
    left_fused = torch.nn.functional.linear(x, left_fused_w) # [B,N,N,2*H]
    left_proj = left_fused[..., :hidden_dim]
    left_gate = torch.sigmoid(left_fused[..., hidden_dim:2*hidden_dim])
    left = left_proj * left_gate
    # Free left_fused
    del left_fused
    # 3. Fuse right projection and right gate
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)
    right_fused = torch.nn.functional.linear(x, right_fused_w)
    right_proj = right_fused[..., :hidden_dim]
    right_gate = torch.sigmoid(right_fused[..., hidden_dim:2*hidden_dim])
    right = right_proj * right_gate
    del right_fused
    # 4. Compute out gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))
    # (Optional) mask multiplication via Triton kernel
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # shape [B*N*N]
        # left
        size = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)
        mask_mul_kernel[grid](
            left,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right
        size_r = right.numel()
        grid_r = (triton.cdiv(size_r, BLOCK),)
        mask_mul_kernel[grid_r](
            right,
            mask_flat,
            hidden_dim,
            size_r,
            BLOCK=BLOCK,
            num_warps=4,
        )
    # 5. O(N^3) contraction
    out = _batched_gemm_fp16(left, right)  # shape [B,N,N,H] fp16
    # 6. Second layernorm (fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # 7. Multiply by out_gate and final linear projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim]
    # 8. Cast back to fp32

"""
custom_kernel: Outgoing TriMul forward pass (no autograd).

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) – stable and cheap.
2. Cast to FP16 for the compute‑heavy part.
3. Fuse the four linear+gate projections (left‑proj/left‑gate and
   right‑proj/right‑gate) into two matrix‑multiplications, and compute the
   out‑gate in a third linear pass (only one read of the input per pass).
4. Apply the element‑wise mask (if present) with a tiny Triton kernel that
   broadcasts the mask over the hidden dimension.
5. Heavy O(N³) contraction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed as a batched GEMM (torch.bmm) on FP16 data.
6. Second LayerNorm (FP16) on the hidden dimension.
7. Multiply by the out‑gate and project back to the original channel dimension.
8. Cast the result back to float32 (the harness expects FP32).

The implementation reduces memory traffic by:
* reading the large 4‑D tensor only three times (instead of five separate
  linear layers),
* fusing each projection with its gate,
* optionally fusing mask·gate multiplication in a Triton kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: multiply a flattened [B,N,N,H] tensor by a broadcast mask
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    x_ptr,          # *fp16  tensor to be masked (flattened)
    mask_ptr,       # *fp16  mask tensor (flattened, shape B*N*N)
    hidden_dim,     # i32    size of hidden dimension H
    size,           # i32    total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load tensor element
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)

    # Broadcast mask over the hidden dimension.
    # Each mask entry corresponds to `hidden_dim` consecutive x entries.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default = 1

    # Fuse multiplication
    tl.store(x_ptr + offs, x * m, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM (fp16) performing the O(N³) contraction.
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H) in fp16
    Returns:
        out: (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and multiply A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Args:
        data: (input_tensor, mask_tensor, weights, config)
            - input_tensor: [B, N, N, D] (float32)
            - mask_tensor:  [B, N, N]   (bool or 0/1)
            - weights: dict with the model parameters
            - config: dict containing "dim" (D) and "hidden_dim" (H) and optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input channel dimension D
    hidden_dim = config["hidden_dim"] # hidden dimension H
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First layer‑norm (float32) → stable for large N
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for compute‑heavy part
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)  # [B, N, N, D]

    # -----------------------------------------------------------------
    # 3️⃣ Cast all model weights once to fp16
    # -----------------------------------------------------------------
    # Linear projection & gate weights (all have shape [H, D])
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Final LayerNorm + projection weights
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (D, H)

    # -----------------------------------------------------------------
    # 4️⃣ Fuse left projection + left gate into a single linear call
    # -----------------------------------------------------------------
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, D)
    left_fused = torch.nn.functional.linear(x, left_fused_w)     # [B,N,N,2*H]

    left_proj = left_fused[..., :hidden_dim]                     # (B,N,N,H)
    left_gate = torch.sigmoid(left_fused[..., hidden_dim:2*hidden_dim])
    left = left_proj * left_gate                                 # (B,N,N,H)
    del left_fused, left_proj, left_gate  # free large temporaries

    # -----------------------------------------------------------------
    # 5️⃣ Fuse right projection + right gate
    # -----------------------------------------------------------------
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, D)
    right_fused = torch.nn.functional.linear(x, right_fused_w)       # [B,N,N,2*H]

    right_proj = right_fused[..., :hidden_dim]
    right_gate = torch.sigmoid(right_fused[..., hidden_dim:2*hidden_dim])
    right = right_proj * right_gate                                  # (B,N,N,H)
    del right_fused, right_proj, right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Compute out‑gate (separate linear)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7️⃣ Optional mask – apply via Triton (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and flatten for the kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)               # shape [B*N*N]

        size = left.numel()                          # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * mask
        mask_mul_kernel[grid](
            left,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * mask
        mask_mul_kernel[grid](
            right,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    # -----------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction – batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)          # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 🔟 Gating + final projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                               # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)    # -> [B,N,N,D] fp16

    # -----------------------------------------------------------------
    # 🟢 Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
