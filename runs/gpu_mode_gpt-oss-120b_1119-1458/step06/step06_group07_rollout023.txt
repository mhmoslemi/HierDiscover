"""
Highly‑optimized Triton‑accelerated forward pass of the “outgoing” TriMul
operator (AlphaFold3). The implementation fuses several element‑wise
operations (mask, gating, layer‑norm) and reduces the number of huge
linear projections by concatenating weight matrices. Heavy O(N³) contraction
is still performed with a single batched GEMM (torch.bmm) which fully
leverages Tensor‑cores on H100. The only required Triton kernels are:

* `mask_gate_fuse_two_kernel` – fuses mask, left/right gating and
  multiplication in‑place.
* `norm_gate_fuse_kernel`   – fuses the second LayerNorm, out‑gate
  multiplication and writes the result back in‑place.

All other dense linear layers are executed with `torch.nn.functional.linear`
(which calls cuBLAS). The final output is cast back to float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel: fuse mask, left/right gating and multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_two_kernel(
    left_ptr,               # *fp16  left tensor (flattened)
    left_gate_ptr,          # *fp16  left gate (flattened)
    right_ptr,              # *fp16  right tensor (flattened)
    right_gate_ptr,         # *fp16  right gate (flattened)
    mask_ptr,               # *fp16  mask flattened [B*N*N]
    hidden_dim: tl.constexpr,   # i32 hidden dimension
    size: tl.constexpr,         # i64 total number of elements per tensor
    BLOCK: tl.constexpr,        # compile‑time tile size  (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load left/right tensors and their gates
    l  = tl.load(left_ptr + offs,        mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs,   mask=valid, other=0.0)
    r  = tl.load(right_ptr + offs,       mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs,  mask=valid, other=0.0)

    # mask is broadcast over the hidden dimension:
    #  each mask entry corresponds to `hidden_dim` contiguous values.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default = 1

    # Fuse the three multiplications
    l_out = l * lg * m
    r_out = r * rg * m

    tl.store(left_ptr  + offs, l_out, mask=valid)
    tl.store(right_ptr + offs, r_out, mask=valid)


# ----------------------------------------------------------------------
# Kernel: fuse second LayerNorm, out‑gate multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_fuse_kernel(
    x_ptr,                # *fp16  tensor to be normalized & gated (flattened, shape [P, H])
    gate_ptr,             # *fp16  out‑gate (flattened, same shape as x)
    weight_ptr,           # *fp16  layer‑norm weight (γ) – shape [H]
    bias_ptr,             # *fp16  layer‑norm bias   (β) – shape [H]
    hidden_dim: tl.constexpr,    # i32 hidden dimension (H)
    eps: tl.constexpr,           # fp32 epsilon for LayerNorm stability
    BLOCK: tl.constexpr,         # compile‑time tile size (≤ 1024)
):
    """Per‑position (batch*seq*seq) LayerNorm + gating."""
    pid = tl.program_id(0)                     # one program per position
    base = pid * hidden_dim                     # start offset of this position

    # -------- First pass: compute mean & variance (fp32) --------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for off in range(0, hidden_dim, BLOCK):
        cur = tl.minimum(BLOCK, hidden_dim - off)
        idx = base + off + tl.arange(0, BLOCK)
        mask = tl.arange(0, BLOCK) < cur
        xv = tl.load(x_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(xv)
        sum_sq  += tl.sum(xv * xv)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # -------- Second pass: normalize, apply γ/β and out‑gate --------
    for off in range(0, hidden_dim, BLOCK):
        cur = tl.minimum(BLOCK, hidden_dim - off)
        idx = base + off + tl.arange(0, BLOCK)
        mask = tl.arange(0, BLOCK) < cur

        x  = tl.load(x_ptr + idx,   mask=mask, other=0.0).to(tl.float32)
        g  = tl.load(gate_ptr + idx,mask=mask, other=0.0).to(tl.float32)
        w  = tl.load(weight_ptr + off + tl.arange(0, BLOCK), mask=mask, other=0.0).to(tl.float32)
        b  = tl.load(bias_ptr   + off + tl.arange(0, BLOCK), mask=mask, other=0.0).to(tl.float32)

        y = (x - mean) * inv_std          # standard LayerNorm
        y = y * w + b                     # γ/β
        y = y * g                         # apply out‑gate

        tl.store(x_ptr + idx, y.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point: forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Arguments
    ---------
    data : (input_tensor, mask_tensor, weights, config)
        input_tensor : [B, N, N, dim]   (float32)
        mask_tensor  : [B, N, N]       (float32/float16, optional)
        weights      : dict of model weights
        config       : dict containing "dim", "hidden_dim", "nomask"

    Returns
    -------
    output : [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True ⇒ ignore mask

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm – fp32 for numerical stability
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                 # [B, N, N, dim]  (float32)
    x = x_norm.to(torch.float16)      # cast once for the heavy part

    # -----------------------------------------------------------------
    # 2️⃣ Cast all weights to fp16 (once)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse left/right linear projections into a single matmul
    # -----------------------------------------------------------------
    proj_w_comb = torch.cat([left_proj_w, right_proj_w], dim=0)   # (2*H, dim)
    proj = torch.nn.functional.linear(x, proj_w_comb)            # (B,N,N,2*H)
    left  = proj[..., :hidden_dim]
    right = proj[..., hidden_dim:]

    # -----------------------------------------------------------------
    # 4️⃣ Fuse left/right gating projections similarly
    # -----------------------------------------------------------------
    gate_w_comb = torch.cat([left_gate_w, right_gate_w], dim=0)   # (2*H, dim)
    gate_pre = torch.nn.functional.linear(x, gate_w_comb)        # (B,N,N,2*H)
    left_gate  = torch.sigmoid(gate_pre[..., :hidden_dim])
    right_gate = torch.sigmoid(gate_pre[..., hidden_dim:])

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fuse mask·gate·tensor multiplication
    # -----------------------------------------------------------------
    if not nomask:
        # flatten mask to [B*N*N] (fp16)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                # shape [B*N*N]

        size = left.numel()                           # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # fuse left & right updates in a single kernel launch
        mask_gate_fuse_two_kernel[grid](
            left, left_gate,
            right, right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ O(N³) contraction: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) for a single batched GEMM
    left_reshape  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matmul (Tensor‑core FP16)
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7️⃣ Compute out‑gate (still needed after the contraction)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 8️⃣ Fuse second LayerNorm + out‑gate multiplication
    # -----------------------------------------------------------------
    # flatten for the kernel: shape [P, H] where P = B*N*N
    out_flat = out.view(-1, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(-1, hidden_dim).contiguous()

    total_positions = out_flat.shape[0]               # P = B*N*N
    BLOCK = 128                                        # threads per program (covers hidden_dim)
    grid = (total_positions,)

    norm_gate_fuse_kernel[grid](
        out_flat,
        out_gate_flat,
        to_out_norm_w,
        to_out_norm_b,
        hidden_dim,
        1e-5,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # out_flat now holds LayerNorm‑scaled & gated values (in‑place)

    # -----------------------------------------------------------------
    # 9️⃣ Final projection back to original channel dimension
    # -----------------------------------------------------------------
    out_proj = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness) and return
    # -----------------------------------------------------------------
    return out_proj.to(torch.float32)
