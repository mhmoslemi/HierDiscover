    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,

    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    # get program ids
    pid_m = tl.program_id(0) # block row
    pid_n = tl.program_id(1) # block col
    pid_b = tl.program_id(2) # batch index (head dimension)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers base
    a_batch_offset = pid_b * stride_batch_a
    b_batch_offset = pid_b * stride_batch_b
    c_batch_offset = pid_b * stride_batch_c

    # accumulate

    # loop over K dimension

        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # compute actual offsets for A and B blocks
        a_ptrs = a_ptr + a_batch_offset + (offs_m[:, None] * stride_m_a) + (offs_k[None, :] * stride_k_a)
        b_ptrs = b_ptr + b_batch_offset + (offs_k[:, None] * stride_k_b) + (offs_n[None, :] * stride_n_b)

        # mask bounds for A and B

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

The module computes
    out = LayerNorm(x)
    left = Linear(out, w_left)   * sigmoid(Linear(out, w_left_gate))   * mask
    right = Linear(out, w_right) * sigmoid(Linear(out, w_right_gate)) * mask
    out  = Σ_k left[..., i, k, :] * right[..., j, k, :]               (einsum)
    out  = LayerNorm(out) * sigmoid(Linear(out, w_out_gate))
    out  = Linear(out, w_to_out)

Key optimisations:
  * Fuse mask, left/right gates, and optional mask application in a Triton kernel.
  * Perform the N³ contraction with a batched GEMM (A @ Bᵀ) using a custom Triton
    kernel that processes every hidden slice as a separate batch.
  * All heavy‑weight work is done in FP16 (Tensor‑core friendly) while
    the final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ------------------------------------------------------------
# 1️⃣ Gate + (optional) mask fusion kernel
# ------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16 [B,N,N,H] (in‑place)
    right_ptr,         # *fp16 [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16 [B,N,N,H]
    right_gate_ptr,    # *fp16 [B,N,N,H]
    mask_ptr,          # *fp16 [B,N,N] (broadcasted over H)
    total_elements,    # i64 scalar: B·N·N·H
    H,                 # i64 hidden dimension (gate width)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # each hidden entry repeats the same mask element -> map flat index to mask index
    mask_offs = offs // H

    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


# ------------------------------------------------------------
# 2️⃣ Batched matrix multiplication kernel for the N³ contraction
#    Computes: C[b] = A[b] @ B[b].T   (all fp16, acc in fp32)
# ------------------------------------------------------------
@triton.jit
def _batched_matmul_outgoing_kernel(
    a_ptr,               # *fp16  [B_h, M, K]
    b_ptr,               # *fp16  [B_h, K, N]  (interpreted as transposed via stride swap)
    c_ptr,               # *fp16  [B_h, M, N]
    batch,               # i64 = number of batches (B_h = batch_size * hidden_dim)
    M, N, K,             # i64 problem dimensions
    stride_batch_a, stride_m_a, stride_k_a,   # int64 strides for A
    stride_batch_b, stride_k_b, stride_n_b,   # int64 strides for B (note: k and n swapped)
    stride_batch_c, stride_m_c, stride_n_c,   # int64 strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile column
    pid_b = tl.program_id(2)      # batch (head) index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current batch
    a_batch_off = pid_b * stride_batch_a
    b_batch_off = pid_b * stride_batch_b
    c_batch_off = pid_b * stride_batch_c

    # accumulator (fp32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k_offset = k_iter * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for this k‑tile
        mask_k = offs_k < K

        # A block pointer: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + a_batch_off + (offs_m[:, None] * stride_m_a) + (offs_k[None, :] * stride_k_a)
        a_mask = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # B block pointer (already transposed via stride swap):
        #   B is stored as [batch, K, N] but we set stride_k_b = col stride (original last dim)
        #   and stride_n_b = row stride (original middle dim) → loads B[n, k] = B_T[k, n].
        b_ptrs = b_ptr + b_batch_off + (offs_k[:, None] * stride_k_b) + (offs_n[None, :] * stride_n_b)
        b_mask = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # inner product (adds fp16 * fp16 into fp32 accumulator)
        acc += tl.dot(a, b)

    # write back result
    c_ptrs = c_ptr + c_batch_off + (offs_m[:, None] * stride_m_c) + (offs_n[None, :] * stride_n_c)
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ------------------------------------------------------------
# 3️⃣ Triton‑accelerated TriMul forward
# ------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        mask_tensor  : torch.Tensor [B, N, N]          (float32/bool, optional)
        weights_dict : mapping name → torch.Tensor (float32)
        config_dict  : must contain "dim" and "hidden_dim";
                       optional "nomask" (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the computation
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast all model weights to FP16 (same device/precision)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections → (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # FP16
    right = F.linear(x, right_proj_w)   # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Gating projections (sigmoid) → (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask (if any) and gates in‑place via Triton
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
    else:
        # Prepare tensors for the kernel
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()
        BLOCK = 1024  # ~1 K threads per block – works well for typical sizes
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporary tensors early
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction: batched GEMM (A @ Bᵀ) using Triton
    # ------------------------------------------------------------------
    B, N_seq, _, H = left.shape  # left.shape == (B, N, N, hidden_dim)
    # Rearrange to (B, H, N, N) and flatten the first two dims → (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(-1, N_seq, N_seq)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(-1, N_seq, N_seq)

    # Allocate output tensor
    out_reshaped = torch.empty_like(left_reshaped)  # FP16, shape (B*H, N, N)

    # Dimensions for the batched GEMM
    batch = left_reshaped.shape[0]   # = B * hidden_dim
    M = N_seq
    N = N_seq
    K = N_seq

    # Strides (int64) – all tensors are contiguous, so we can read strides directly
    stride_batch_a = left_reshaped.stride(0)
    stride_m_a      = left_reshaped.stride(1)
    stride_k_a      = left_reshaped.stride(2)

    stride_batch_b = right_reshaped.stride(0)
    # For the transpose we swap the last two strides:
    stride_k_b = right_reshaped.stride(2)   # original column stride (==1)
    stride_n_b = right_reshaped.stride(1)   # original row stride (==N)

    stride_batch_c = out_reshaped.stride(0)
    stride_m_c      = out_reshaped.stride(1)
    stride_n_c      = out_reshaped.stride(2)

    # Triton block sizes – tuned for fp16 Tensor‑cores on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    # Launch the kernel: grid = (num_blocks_m, num_blocks_n, batch)
    grid = (num_blocks_m, num_blocks_n, batch)
    _batched_matmul_outgoing_kernel[grid](
        a_ptr=left_reshaped,
        b_ptr=right_reshaped,
        c_ptr=out_reshaped,
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_batch_a=stride_batch_a,
        stride_m_a=stride_m_a,
        stride_k_a=stride_k_a,
        stride_batch_b=stride_batch_b,
        stride_k_b=stride_k_b,
        stride_n_b=stride_n_b,
        stride_batch_c=stride_batch_c,
        stride_m_c=stride_m_c,
        stride_n_c=stride_n_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # Reshape back to original layout (B, N, N, hidden_dim)
    out = out_reshaped.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply the final out‑gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise FP16 gate
    out = F.linear(out, to_out_w)            # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast to float32 for the caller (matches reference implementation)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
