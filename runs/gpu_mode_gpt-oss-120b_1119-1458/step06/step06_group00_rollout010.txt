"""
Optimized outgoing TriMul kernel (AlphaFold3 style) using Triton for mask‑gate fusion.
Key ideas:
* Fuse the five linear projections (left/right projections + their gates + out‑gate)
  into a single half‑precision matmul to reduce memory bandwidth.
* Apply sigmoid gates on the hidden‑dim tensors.
* Fuse mask multiplication and gate application for left/right tensors with a Triton kernel.
* Use a single batched GEMM (torch.bmm) to perform the O(N³) contraction.
* Use half‑precision for the heavy parts (projections, contraction, gates, LayerNorms,
  final projection) and cast back to float32 for the caller.
This implementation respects the required API and works for all listed test configs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with the per‑head gates.
#   left  <- left  * left_gate  * mask
#   right <- right * right_gate * mask
#   All tensors are contiguous FP16.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in‑place)
    right_ptr,         # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcasted)
    total_elements,    # i64   total number of FP16 elements in left/right
    H,                 # i32   hidden dimension (gate dim)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Corresponding mask index (broadcasted over H)
    mask_offs = offs // H

    # Load tensors (zero/1.0 for OOB)
    left      = tl.load(left_ptr + offs,    mask=valid, other=0.0)
    right     = tl.load(right_ptr + offs,   mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """Outgoing TriMul forward pass (no gradients).
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool / 0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)  # [B, N, N, dim] FP32

    # Convert to FP16 for the heavy path
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear transformations into a single matmul:
    #    left_proj, left_gate, right_proj, right_gate, out_gate
    # ------------------------------------------------------------------
    # Gather weight matrices (all shape [hidden_dim, dim] or [out_dim, hidden_dim])
    w_left_proj  = weights["left_proj.weight"]
    w_left_gate  = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate   = weights["out_gate.weight"]

    # Concatenate out_features dimension (5 * hidden_dim)
    # each weight has shape (hidden_dim, dim) → (out_features, in_features) for F.linear
    concat_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).to(device=device, dtype=fp16)  # shape (5*hidden_dim, dim)

    # Flatten x to (M, dim) where M = B * N * N
    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.reshape(M, dim)  # (M, dim)

    # One large half‑precision linear
    combined = F.linear(x_flat, concat_weight)  # (M, 5*hidden_dim)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Split the combined tensor into its components
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj   = combined[..., 0:h]           # (B,N,N,H)
    left_gate_p = combined[..., h:2*h]        # pre‑sigmoid
    right_proj  = combined[..., 2*h:3*h]
    right_gate_p= combined[..., 3*h:4*h]
    out_gate_p  = combined[..., 4*h:5*h]

    # Apply sigmoid to gate tensors
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and gates to left/right projections.
    #    This is fused in a Triton kernel for the masked case.
    # ------------------------------------------------------------------
    if nomask:
        # In‑place gated tensors when there is no mask.
        left = left_proj.mul_(left_gate)     # (B,N,N,H)
        right = right_proj.mul_(right_gate)
    else:
        # Prepare contiguous tensors for the kernel.
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(fp16).contiguous()

        total = left.numel()
        BLOCK = 1024  # number of elements per Triton block
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporary tensors
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction: batched GEMM (left @ rightᵀ)  (FP16)
    # ------------------------------------------------------------------
    # left/right are (B,N,N,H). Rearrange to (B*H, N, N) for torch.bmm.
    left_bhm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_bhm = right.permute(0, 3, 1, 2).contiguous()
    A = left_bhm.view(B * hidden_dim, N, N)            # (B*H, N, N)
    B_ = right_bhm.view(B * hidden_dim, N, N)          # (B*H, N, N)

    # Batched matrix multiplication with the transposed second operand.
    C = torch.bmm(A, B_.transpose(1, 2))               # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate (element‑wise) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate             # element‑wise gate, FP16

    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)  # (dim, hidden_dim)

    # Flatten and apply final linear (FP16)
    out_flat = out.view(-1, hidden_dim)          # (B*N*N, H)
    out = F.linear(out_flat, to_out_w)           # (B*N*N, dim)
    out = out.view(B, N, N, dim)                 # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
