"""
Optimised outgoing TriMul forward (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑Norm on the raw input (float32) – stabilises statistics.
2️⃣  Cast to fp16 for the bulk of the computation.
3️⃣  Fuse the two linear projections (proj & gate) for the left and right sides
    by concatenating the weight matrices.  This reduces the number of matmuls
    from four to two.
4️⃣  Apply sigmoid to the gate values *and* the optional binary mask in a
    single Triton kernel (`mask_gate_sigmoid_kernel`).  The kernel works on the
    flattened view and broadcasts the mask over the hidden dimension.
5️⃣  Heavy O(N³) contraction:  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h].
    Implemented as a batched GEMM via `torch.bmm` after reshaping the tensors
    to [B·H, N, N]; this leverages cuBLAS Tensor‑Core kernels on H100.
6️⃣  Second Layer‑Norm (fp16) over the hidden dimension.
7️⃣  Multiply by the output‑gate, then final linear projection back to the
    original dimension.
8️⃣  Cast the result back to float32.

Only the forward pass is required – no autograd is recorded.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def mask_gate_sigmoid_kernel(
    x_ptr,          # *fp16 : tensor to be updated (proj result)
    gate_raw_ptr,   # *fp16 : raw gate tensor (pre‑sigmoid)
    mask_ptr,       # *fp16 : binary mask, shape B·N·N (flattened)
    hidden_dim,     # i32   : hidden dimension H
    size,           # i32   : total number of elements in x/gate (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    """
    In‑place:  x = x * sigmoid(gate_raw) * mask
    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices for this block
    active = offs < size                              # guard

    # mask index = element index // hidden_dim  (mask has no hidden dim)
    mask_offs = offs // hidden_dim

    # Load data
    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g_raw = tl.load(gate_raw_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)  # default mask = 1

    # Sigmoid (1 / (1+exp(‑x))) – numerically stable for fp16
    g = 1.0 / (1.0 + tl.exp(-g_raw))

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the O(N³) contraction.
    left/right: [B, N, N, H] (fp16)
    Returns:    [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse proj + gate weights (half‑precision)
    # -----------------------------------------------------------------
    # left side
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # shape (2*H, dim)
    # right side
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                    # shape (2*H, dim)

    # other weights
    out_gate_weight    = weights["out_gate.weight"].to(torch.float16)       # (H, dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_weight      = weights["to_out.weight"].to(torch.float16)       # (dim, H)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (half‑precision) – fused proj+gate
    # -----------------------------------------------------------------
    left_comb = torch.nn.functional.linear(x, left_comb_weight)   # [B, N, N, 2*H]
    right_comb = torch.nn.functional.linear(x, right_comb_weight) # [B, N, N, 2*H]

    # Slice the concatenated tensors (make them contiguous for Triton)
    left_proj = left_comb[..., :hidden_dim].contiguous()           # [B, N, N, H]
    left_gate_raw = left_comb[..., hidden_dim:].contiguous()      # [B, N, N, H]

    right_proj = right_comb[..., :hidden_dim].contiguous()        # [B, N, N, H]
    right_gate_raw = right_comb[..., hidden_dim:].contiguous()   # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5️⃣ Out‑gate (sigmoid) – fp16
    # -----------------------------------------------------------------
    out_gate_raw = torch.nn.functional.linear(x, out_gate_weight)  # [B, N, N, H]
    out_gate = torch.sigmoid(out_gate_raw)                         # fp16

    # -----------------------------------------------------------------
    # 6️⃣ Optional masking + gate sigmoid (fused in Triton)
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – cast to fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()     # [B, N, N]
        total_elems = left_proj.numel()                           # B·N·N·H
        BLOCK = 1024                                               # block size (tunable)
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left side: left = left_proj * sigmoid(left_gate_raw) * mask
        mask_gate_sigmoid_kernel[grid](
            left_proj,
            left_gate_raw,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right side
        mask_gate_sigmoid_kernel[grid](
            right_proj,
            right_gate_raw,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel the tensors `left_proj` and `right_proj` already contain
        # the gated‑and‑masked values.
        left = left_proj
        right = right_proj
    else:
        # No mask – simple element‑wise sigmoid gating
        left = left_proj * torch.sigmoid(left_gate_raw)
        right = right_proj * torch.sigmoid(right_gate_raw)

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # -----------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final linear projection to original dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
