"""
Optimized outgoing TriMul forward pass (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Layer‑Norm on the raw FP32 input (stabilises statistics).
2️⃣  Cast to FP16 for the bulk of the work.
3️⃣  Fuse the five linear projections (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a **single** GEMM by
    concatenating their weight matrices.  This reduces the number
    of passes over the large [B, N, N, dim] tensor from five to one,
    cutting memory traffic dramatically.
4️⃣  Split the fused result, apply a sigmoid to the three gate
    slices and fuse *mask* and *gate* element‑wise with a tiny
    Triton kernel (`mask_gate_fuse_kernel`).  The kernel reads each
    hidden‑channel element only once.
5️⃣  Perform the O(N³) contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    as a batched GEMM using cuBLAS/Tensor‑Cores (torch.bmm) on the
    reshaped tensors of shape [B·H, N, N].
6️⃣  Layer‑Norm over the hidden dimension (FP16).
7️⃣  Apply the output‑gate and a final linear projection back to the
    original dimension.
8️⃣  Cast the result back to FP32 (the harness expects float32).

All heavy “tensor‑wide” work stays on the GPU; only the light
point‑wise fusions are performed with Triton kernels.  This
implementation is fully self‑contained and works for any batch size
(1‑2), sequence length (128‑1024) and hidden dimension (128, 384).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse masking and gating (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : hidden dimension H
    size,           # i32   : total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)    # linear offsets for this block
    mask = offs < size                           # guard for out‑of‑bounds

    # The mask index is the same for every hidden channel because H is the
    # innermost stride.  We compute it by integer division.
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=mask, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=mask, other=1.0)   # default mask=1

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

# ----------------------------------------------------------------------
# Helper: batched GEMM that implements the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H] (FP16)
    Returns:      out   : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # View as (B·H, N, N) for a batched matrix multiplication
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Perform (B·H, N, N) @ (B·H, N, N)ᵀ
    c = torch.bmm(a, b.transpose(1, 2))          # [B·H, N, N]
    # Restore shape [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32, already on device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – stability before casting down
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype=float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy lifting
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into one GEMM
    # -----------------------------------------------------------------
    # All linear layers have bias=False.
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension (rows)
    fused_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    ).to(torch.float16)            # shape (5*H, dim)

    # Single linear projection -> [B, N, N, 5*H]
    fused = torch.nn.functional.linear(x, fused_weight)

    # -----------------------------------------------------------------
    # 4️⃣ Slice and apply sigmoid to gate tensors
    # -----------------------------------------------------------------
    H = hidden_dim
    left_proj   = fused[..., 0 * H : 1 * H]
    right_proj  = fused[..., 1 * H : 2 * H]
    left_gate_p = fused[..., 2 * H : 3 * H]
    right_gate_p= fused[..., 3 * H : 4 * H]
    out_gate_p  = fused[..., 4 * H : 5 * H]

    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # -----------------------------------------------------------------
    # 5️⃣ Fuse mask & gate (element‑wise) with a tiny Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is FP16 contiguous (shape [B, N, N])
        mask_half = mask_tensor.to(torch.float16).contiguous()
        total = left_proj.numel()               # B·N·N·H
        BLOCK = 1024                            # <= max threads per block on H100
        grid = ((total + BLOCK - 1) // BLOCK,)
        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_half,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_half,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj   # now holds the gated & masked values
        right = right_proj
    else:
        # No mask – simple in‑place multiplication with the gates
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via a batched GEMM (Tensor‑Core)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H] (FP16)

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (FP16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # still FP16

    # -----------------------------------------------------------------
    # 8️⃣ Output gating + final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate          # element‑wise gating (FP16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (FP16)

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
