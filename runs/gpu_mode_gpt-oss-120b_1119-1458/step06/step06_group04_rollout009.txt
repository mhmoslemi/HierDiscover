```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Step 1: first layer norm (float32)
    # Use torch.nn.functional.layer_norm over last dimension dim.
    # Norm weight and bias present.

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # float32

    # Cast to fp16
    x = x_norm.to(torch.float16)

    # Step 2: combined projection for all gates and projections
    # Concatenate weight matrices
    cat_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Compute all linear transforms in one matmul
    # result shape [B,N,N,5*hidden_dim]
    combined = torch.nn.functional.linear(x, cat_weight)  # fp16

    H = hidden_dim
    left_proj = combined[..., 0:H]
    left_gate_raw = combined[..., H:2*H]
    right_proj = combined[..., 2*H:3*H]
    right_gate_raw = combined[..., 3*H:4*H]
    out_gate_raw = combined[..., 4*H:5*H]

    # Sigmoid for gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # Step 3: Mask gating fused with per-element gate
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # shape (B*N*N,)
        total = left_proj.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        # left_proj = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right_proj = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    left = left_proj
    right = right_proj

    # Step 4: Core contraction (batch GEMM)
    out = _batched_gemm_fp16(left, right)  # shape [B,N,N,hidden_dim]

    # Step 5: second layernorm (fp16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Step 6: out gate
    out = out * out_gate

    # Step 7: final linear projection
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim] fp16

    # Cast to float32
    return out.to(torch.float32)
```

```
def _batched_gemm_fp16(left, right):
    B, N, _, H = left.shape
    # reshape for B*H batch
    # left: [B,N,N,H] -> [B*H, N, N] (swap axes such that dim for matmul: left[i,k] = row i, col k)
    left_perm = left.permute(0, 3, 1, 2)  # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2)  # B, H, N, N
    # left has shape (B, H, N, N), right has shape (B, H, N, N)
    # We need left[b, h] @ right[b, h].T
    a = left_perm.reshape(B * H, N, N)  # B*H, N, N
    b = right_perm.reshape(B * H, N, N)  # B*H, N, N
    c = torch.bmm(a, b.transpose(1, 2))  # B*H, N, N
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

"""
Optimized Triton forward kernel for the outgoing TriMul module (AlphaFold3).

Algorithm:
1. Layer‑norm the 4‑D input (float32) → cast to fp16.
2. Fuse the five linear transforms (left/right projections and three gating
   signals) into a single matmul by concatenating the weight matrices.
3. Apply sigmoid to the three gates.
4. If a mask is provided, fuse mask, gate and projection with a tiny Triton
   kernel (`mask_gate_fuse_kernel`). Otherwise perform a plain element‑wise
   multiplication.
5. Core O(N³) contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   is performed as a batched GEMM (`C = A @ Bᵀ`) using fp16 cuBLAS
   (`torch.bmm`). The tensors are reshaped to (B*H, N, N) where H is the
   hidden dimension.
6. Second LayerNorm on the hidden dimension.
7. Multiply by the output gate.
8. Final linear projection back to the original channel dimension.
9. Cast the result back to float32 (the harness expects float32).

All heavy arithmetic runs in fp16 to exploit Tensor Cores on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask‑gate‑tensor multiplication.
# Input tensors are flattened fp16 arrays; mask is 1‑D (B·N·N).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask 1‑D tensor, broadcast over hidden_dim
    hidden_dim,     # i32   → hidden dimension size (H)
    size,           # i32   → total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,   # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Derive the mask index (broadcast over hidden_dim)
    idx = offs // hidden_dim
    m = tl.load(mask_ptr + idx, mask=mask, other=1.0)   # default mask = 1

    # Fuse multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the core contraction (fp16).
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM implementation for the TriMul contraction.
    Both inputs must be fp16 tensors of shape [B, N, N, H].
    Returns fp16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # Permute to bring the hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    a = left_perm.view(B * H, N, N)      # [B*H, N, N]
    b = right_perm.view(B * H, N, N)     # [B*H, N, N]

    # A @ Bᵀ
    c = torch.bmm(a, b.transpose(1, 2))  # [B*H, N, N]

    # Restore original ordering: [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

Optimized Triton forward kernel for the outgoing TriMul module (AlphaFold3).

Algorithm:
1️⃣  Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣  Concatenate the five linear weight matrices (left/right projections and three

    left_proj, left_gate, right_proj, right_gate, out_gate.
3️⃣  Apply sigmoid to the three gate tensors (still fp16).
4️⃣  If a mask is supplied, fuse mask, gate and projection with a tiny Triton
    kernel (`mask_gate_fuse_kernel`).  Otherwise perform a plain element‑wise
    multiplication.
5️⃣  Core O(N³) contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    dimension.
6️⃣  Second LayerNorm on the hidden dimension (fp16).
7️⃣  Multiply by the output gate.
8️⃣  Final linear projection back to the original channel dimension.
9️⃣  Cast the result back to float32 (the harness expects float32).

All heavy arithmetic runs in fp16 to exploit Tensor Cores on an H100.
"""

import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused mask‑gate‑tensor multiplication.
# Input tensors are flattened fp16 arrays; mask is 1‑D (B·N·N) broadcast over H.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask 1‑D tensor, broadcast over hidden_dim
    hidden_dim,     # i32   → hidden dimension size (H)
    size,           # i32   → total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,   # compile‑time tile size
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fuse the three multiplications
    out = x * g * m

    # Store result back in‑place
    tl.store(x_ptr + offs, out, mask=valid)

# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM implementation for the TriMul contraction.

    Parameters
    ----------
    left, right : torch.Tensor
        fp16 tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        fp16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape

    # Bring H into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    b = right_perm.view(B * H, N, N)     # [B*H, N, N]

    c = torch.bmm(a, b.transpose(1, 2))  # [B*H, N, N]

    # Restore original ordering => [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
