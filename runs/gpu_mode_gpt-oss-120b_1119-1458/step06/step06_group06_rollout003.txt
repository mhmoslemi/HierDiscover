"""
Outgoing TriMul (AlphaFold3) – fast forward pass.

Algorithm
---------
1. Layer‑norm the [B, N, N, C] input (FP32) → cast to FP16.
2. Fuse the left/right linear projections and their gating signals:
   * Concatenate the projection and gate weight matrices (shape (2·Cₕ, C)).
   * Perform a single `torch.nn.functional.linear` per side → tensor of shape
     [B, N, N, 2·Cₕ] (proj | gate_pre).
   * Split, apply sigmoid to the gate part.
3. If a mask is present, fuse (`proj * gate * mask`) for *both* sides in one
   Triton kernel (`mask_gate_fuse_two_streams_kernel`).  When there is no
   mask we simply multiply in‑place.
4. Heavy contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (A @ Bᵀ) on FP16 data using `torch.bmm`.
   This leverages cuBLAS Tensor‑Core kernels.
5. Second Layer‑norm on the hidden dimension (FP16) → out‑gate gating → final
   linear projection back to the original channel dimension.
6. Cast the result back to FP32 (the harness expects float32).

All heavy element‑wise fusions are done with custom Triton kernels; the
matrix‑multiplications rely on cuBLAS which already provides near‑optimal
performance on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) for the left and right streams
# in a *single* launch.  The mask is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_two_streams_kernel(
    left_ptr,          # *fp16 : left projection tensor, size = total_elements
    left_gate_ptr,     # *fp16 : left gate tensor, same shape as left_ptr
    right_ptr,         # *fp16 : right projection tensor, size = total_elements
    right_gate_ptr,    # *fp16 : right gate tensor, same shape as right_ptr
    mask_ptr,          # *fp16 : mask tensor of shape (B·N·N,)
    total_elements,    # i32   : total number of elements in each stream
    hidden_dim,        # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask_valid = offs < total_elements          # guard for the tail

    # Load streams
    left_val      = tl.load(left_ptr + offs,        mask=mask_valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs,   mask=mask_valid, other=0.0)
    right_val     = tl.load(right_ptr + offs,       mask=mask_valid, other=0.0)
    right_gate_val= tl.load(right_gate_ptr + offs,  mask=mask_valid, other=0.0)

    # Mask is shared for both streams – one mask element per hidden‑dim block
    mask_idx = offs // hidden_dim               # integer division
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    # Fused multiplication
    left_out  = left_val  * left_gate_val  * mask_val
    right_out = right_val * right_gate_val * mask_val

    # Store results in‑place
    tl.store(left_ptr + offs,  left_out,  mask=mask_valid)
    tl.store(right_ptr + offs, right_out, mask=mask_valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# Both inputs are FP16 tensors of shape [B, N, N, H] (H = hidden_dim).
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    using a batched matrix multiplication (A @ Bᵀ).
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] and compute A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)                 # (B*H,N,N)
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)                # (B*H,N,N)

    # (B*H,N,N) @ (B*H,N,N)ᵀ -> (B*H,N,N)
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32/bool) – may be ignored
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]               # original channel dimension (C_z)
    hidden_dim  = config["hidden_dim"]        # projection width (C_h)
    nomask      = config.get("nomask", True)  # True → no mask applied

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] in FP32
    x = x_norm.half()  # work in FP16 from now on

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate linear layers (two GEMMs instead of four)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights for the left side
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # shape (2*hidden_dim, dim)

    # Same for the right side
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # shape (2*hidden_dim, dim)

    # out‑gate weight (used later)
    out_gate_weight = weights["out_gate.weight"].half()

    # Compute combined linear transforms in a single pass per side
    left_comb = F.linear(x, left_comb_weight)   # [B, N, N, 2*hidden_dim] FP16
    right_comb = F.linear(x, right_comb_weight) # [B, N, N, 2*hidden_dim] FP16

    # Split back into projection and pre‑gate tensors
    left_proj = left_comb[..., :hidden_dim]                     # (proj)
    left_gate_pre = left_comb[..., hidden_dim:]                # (gate pre‑sigmoid)
    left_gate = torch.sigmoid(left_gate_pre)                   # (gate)

    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 3️⃣ Out‑gate (computed from the normalized input, used later)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and element‑wise gate multiplication.
    #    For the nomask case we simply multiply in‑place.
    # ------------------------------------------------------------------
    if nomask:
        # In‑place multiplication saves memory bandwidth
        left = left_proj.mul_(left_gate)
        right = right_proj.mul_(right_gate)
    else:
        # Ensure tensors are contiguous – required for the Triton kernel
        left_proj = left_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_proj = right_proj.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)

        total_elements = left_proj.numel()            # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # One launch processes both streams simultaneously
        mask_gate_fuse_two_streams_kernel[grid](
            left_proj,
            left_gate,
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel, left_proj/right_proj already hold the fused result
        left = left_proj
        right = right_proj

        # Gates are no longer needed – free memory
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction (batched GEMM) – O(N³) step
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)      # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) → out‑gate → final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias   = weights["to_out_norm.bias"].half()
    to_out_weight      = weights["to_out.weight"].half()

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )                                          # still FP16

    # Element‑wise out‑gate
    out = out * out_gate

    # Final projection back to the original channel dimension
    out = F.linear(out, to_out_weight)        # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast the result back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
