def custom_kernel(data):
   input_tensor, mask_tensor, weights, config = data

   device = input_tensor.device
   dim = config['dim']
   hidden_dim = config['hidden_dim']
   nomask = config.get('nomask', True)

   # layernorm 1
   norm_w = weights["norm.weight"]
   norm_b = weights["norm.bias"]
   x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
   # Cast to fp16
   x = x.to(torch.float16)

   # convert weights to fp16
   left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float16)
   right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)
   left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
   right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
   out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)

   # Compute projections and gates (fp16)
   left_proj = torch.nn.functional.linear(x, left_proj_w)         # shape [B,N,N,H]
   right_proj = torch.nn.functional.linear(x, right_proj_w)

   left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

   # Mask gating using Triton if needed, else simple multiplication
   if nomask:
       left = left_proj * left_gate
       right = right_proj * right_gate
   else:
       mask_fp16 = mask_tensor.to(torch.float16)
       left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
       right = _apply_mask_gate(right_proj, right_gate, mask_fp16)

   # Compute contracted out using custom batched GEMM kernel (no permute)
   out = _batched_matmul_internal(left, right) # custom

   # Second LayerNorm
   to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
   to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
   out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

   # out gate
   out = out * out_gate

   # final projection
   to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
   out = torch.nn.functional.linear(out, to_out_w)

   return out.to(torch.float32)

def _batched_matmul_internal(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute contraction using a custom Triton kernel.
    left and right shape: [B,N,N,H], dtype fp16.
    Returns output shape same: [B,N,N,H], dtype fp16.
    """

    )

def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, H, N,
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,
    stride_b_b, stride_b_i, stride_b_k, stride_b_h,
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # compute batch id
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b_idx = pid_batch // H
    h_idx = pid_batch % H

    # base pointers for each current batch+hidden
    a_batch_ptr = a_ptr + b_idx * stride_a_b + h_idx * stride_a_h
    b_batch_ptr = b_ptr + b_idx * stride_b_b + h_idx * stride_b_h
    c_batch_ptr = c_ptr + b_idx * stride_c_b + h_idx * stride_c_h

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_k_tiles):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile (BATCH: a_batch_ptr)
        a_ptrs = a_batch_ptr + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile (BATCH: b_batch_ptr) transposed: rows = k, cols = j

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch_ptr + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing TriMul (AlphaFold‑3) forward pass – Triton‑accelerated O(N³) contraction.
Algorithm
---------
1️⃣  Layer‑Norm (FP32) → cast to FP16.
2️⃣  Linear projections for left/right and their gates (FP16).
3️⃣  Fuse mask, projection and gate with a tiny Triton kernel (if a mask is
    provided).  Without mask we just multiply element‑wise.
4️⃣  Heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is a **batched GEMM** C = A @ Bᵀ where the batch dimension is
    (batch × hidden).  A custom Triton kernel operates directly on the
    original [B, N, N, H] layout, thus avoiding the costly
    permute‑contiguous copy performed by the reference implementation.
5️⃣  Second Layer‑Norm (FP16) → out‑gate (sigmoid) → final linear projection.
6️⃣  Cast result back to float32 (the harness expects FP32).

Only steps 3 and 4 are Triton kernels – the rest uses highly‑optimised torch
ops.  The kernel for the O(N³) step is tuned for H100 (BLOCK_M/N = 64,
BLOCK_K = 32, 4 warps) and accumulates in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: mask * projection * gate (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,      # *fp16  [B, N, N, H]
    gate_ptr,      # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,      # *fp16  [B, N, N]
    out_ptr,       # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask is broadcast over hidden dim H)
    """
    pid_b = tl.program_id(0)      # batch tile
    pid_i = tl.program_id(1)      # i‑tile
    pid_j = tl.program_id(2)      # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- load mask tile (BLOCK_I × BLOCK_J) -----
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + \
                offs_i[:, None] * stride_mask_i + \
                offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # ----- loop over hidden‑dim tiles -----
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj tile
        proj_ptrs = proj_ptr + pid_b * stride_proj_b + \
                    offs_i[:, None, None] * stride_proj_i + \
                    offs_j[None, :, None] * stride_proj_j + \
                    offs_h[None, None, :] * stride_proj_h
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # gate tile (already sigmoid‑ed)
        gate_ptrs = gate_ptr + pid_b * stride_gate_b + \
                    offs_i[:, None, None] * stride_gate_i + \
                    offs_j[None, :, None] * stride_gate_j + \
                    offs_h[None, None, :] * stride_gate_h
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # fused element‑wise product
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # store
        out_ptrs = out_ptr + pid_b * stride_out_b + \
                   offs_i[:, None, None] * stride_out_i + \
                   offs_j[None, :, None] * stride_out_j + \
                   offs_h[None, None, :] * stride_out_h
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor,
                     mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim)

    All inputs must be FP16 CUDA tensors.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides (in elements, not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A and B are [B,N,N,H])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, H, N,
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,
    stride_b_b, stride_b_i, stride_b_k, stride_b_h,
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for each hidden channel h (treated as part of the batch):
        out[b,i,j,h] = Σₖ a[b,i,k,h] * b[b,j,k,h]
    where a == left_gated, b == right_gated.
    """
    pid_batch = tl.program_id(0)   # flattened batch * hidden index
    pid_i = tl.program_id(1)       # i‑tile
    pid_j = tl.program_id(2)       # j‑tile

    # de‑compose flattened index
    b_idx = pid_batch // H
    h_idx = pid_batch % H

    # base pointers for this (b,h) slice
    a_base = a_ptr + b_idx * stride_a_b + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_b + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_b + h_idx * stride_c_h

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load A tile (i × k) ----
        a_ptrs = a_base + \
                 offs_i[:, None] * stride_a_i + \
                 offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)               # fp16

        # ---- load B tile (j × k) transposed as (k × j) ----
        # B is stored as (j,k) → we need rows=k, cols=j for Bᵀ
        b_ptrs = b_base + \
                 offs_k[:, None] * stride_b_k + \
                 offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)               # fp16

        # ---- dot‑product accumulation (FP32) ----
        acc += tl.dot(a, b)               # auto‑casts to FP32

    # ---- store result (cast back to fp16) ----
    c_ptrs = c_base + \
             offs_i[:, None] * stride_c_i + \
             offs_j[None, :] * stride_c_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def _batched_matmul_triton(left: torch.Tensor,
                           right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM kernel.
    Input tensors must be FP16 and have shape [B, N, N, H].
    Returns tensor of the same shape (FP16).
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # FP16 output

    # strides (elements, not bytes)
    s_a = left.stride()
    s_b = right.stride()
    s_c = out.stride()

    # tile sizes – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        s_a[0], s_a[1], s_a[2], s_a[3],
        s_b[0], s_b[1], s_b[2], s_b[3],
        s_c[0], s_c[1], s_c[2], s_c[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Custom forward kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : dictionary of model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ---------------------------------------------------------------
    # Unpack and utilities
    # ---------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ---------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16 for the rest
    # ---------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # FP32
    x = x.to(torch.float16)  # FP16 onward

    # ---------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (on‑device)
    # ---------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ---------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ---------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ---------------------------------------------------------------
    # 4️⃣ Gates (sigmoid)
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ---------------------------------------------------------------
    # 5️⃣ Mask‑gate fusion (Triton) or simple element‑wise multiplication
    # ---------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate      # element‑wise product, FP16
        right = right_proj * right_gate

    # ---------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – custom batched GEMM (no copies!)
    # ---------------------------------------------------------------
    out = _batched_matmul_triton(left, right)    # shape [B,N,N,hidden] FP16

    # ---------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ---------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------------------------------------------------------------
    # 8️⃣ Out‑gate and final projection back to model dim (FP16)
    # ---------------------------------------------------------------
    out = out * out_gate                       # gating
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # ---------------------------------------------------------------
    # 9️⃣ Cast back to FP32 – the harness expects float32
    # ---------------------------------------------------------------
    return out.to(torch.float32)
