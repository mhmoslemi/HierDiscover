import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _mask_gate_fusion_kernel(left_ptr, right_ptr,
                             left_gate_ptr, right_gate_ptr,
                             mask_ptr,
                             total_elements,
                             H,
                             BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    # Determine mask index
    mask_offs = offs // H
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0) # 1.0 for no mask
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).
    Implements a fused linear projection for left/right and gates,
    a Triton kernel that fuses mask application with gate multiplication,
    and a batched matrix multiplication for the O(N³) contraction.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # 1. Input LayerNorm (float32)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)  # [B,N,N,dim] float32

    # 2. Convert to fp16 for compute‑intensive ops
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3. Fuse the 5 linear transforms (left_proj, right_proj, left_gate,
    #    right_gate, out_gate) into a single GEMM.
    # ------------------------------------------------------------------
    # Build a single weight matrix of shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=fp16)

    # Flatten x to (B * N * N, dim)
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)          # contiguous

    # Single linear projection
    proj = F.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim), fp16
    proj = proj.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 4. Split projection into tensors and apply sigmoid gates
    # ------------------------------------------------------------------
    # Use torch.split for clarity
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5. Fuse mask and per‑head gates (optional)
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 (0/1) and make all tensors contiguous
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        # Ensure tensors are contiguous for the Triton kernel
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left_proj.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left_proj,
            right_ptr=right_proj,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporaries early
        del mask_fp16, left_gate, right_gate
    else:
        # No mask: just apply gate element‑wise
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. O(N³) contraction: per‑head matrix multiplication
    #    left[i,k,d] * right[j,k,d] summed over k.
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and reshape to (B*H, N, N)
    left = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM: left @ rightᵀ  (FP16)
    out = torch.bmm(left, right.transpose(1, 2))

    # Restore to (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)

    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 8. Apply out‑gate and final linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate  # element‑wise gating (FP16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)          # (B,N,N,dim) FP16

    # 9. Cast back to float32 for the caller
    return out.to(torch.float32)

"""
custom_kernel – Outgoing TriMul forward pass (no gradients).

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → stable statistics).
2. Cast to FP16 for all subsequent heavy work.
3. Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM:
     [B·N·N, dim] × [5·hidden_dim, dim]ᵀ → [B·N·N, 5·hidden_dim].
   The result is split into the five tensors; the three gate tensors
   are passed through a sigmoid.
4. Optional mask‑gate fusion: a Triton kernel multiplies the
   left‑ and right‑projection tensors element‑wise by their sigmoids
   and by the broadcast mask (the mask is 0/1, cast to FP16).  When
   `nomask=True` we simply multiply by the gate tensors in‑place.
5. O(N³) contraction: for each head `d` we compute
   out[...,i,j,d] = Σₖ left[...,i,k,d] * right[...,j,k,d].
   This is a batched matrix multiply `left @ rightᵀ` where the batch
   dimension is `B·hidden_dim`.  cuBLAS (torch.bmm) on FP16/Tensor‑core
   delivers the best performance on H100.
6. Layer‑norm over the hidden dimension, followed by the out‑gate
   (sigmoid) and a final linear projection back to `dim`.
7. Cast the result back to float32 for the caller.

All heavy arithmetic is performed in FP16; only the first LayerNorm and
the final cast use float32.  The mask‑gate fusion is the only hand‑written
Triton kernel required by the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with the per‑head gates.
#   left  ← left  * left_gate  * mask
#   right ← right * right_gate * mask
#   (mask is broadcast over the hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i32   B·N·N·H   (flat length of left/right)
    H,                 # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # flat index
    valid = offs < total_elements

    # index of the mask element that corresponds to this hidden‑dim entry
    mask_offs = offs // H

    # load tensors (zero for OOB)
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    # load mask (1.0 for OOB → no effect)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : tuple(input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → stable statistics)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)                         # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the compute‑intensive path
    # ------------------------------------------------------------------
    x = x.to(fp16)    # now [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (project & gates) into a single GEMM
    # ------------------------------------------------------------------
    # Combined weight: (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=fp16)

    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B·N·N, dim), contiguous

    # One fused linear pass
    proj = F.linear(x_flat, combined_weight)        # (B·N·N, 5*hidden_dim) FP16
    proj = proj.view(B, N, N, 5 * hidden_dim)       # [B,N,N,5*hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Split & apply sigmoid to the three gate tensors
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣ Mask‑gate fusion (optional)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: bool / float → FP16  (0.0 for masked, 1.0 otherwise)
        mask_fp16 = mask_tensor.to(fp16).contiguous()

        # Ensure tensors are contiguous for the Triton kernel
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left_proj.numel()
        BLOCK = 1024                                   # ~1 K threads per block
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left_proj,
            right_ptr=right_proj,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporary tensors early
        del mask_fp16, left_gate, right_gate
    else:
        # No mask → simple in‑place gating
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction: batched GEMM left @ rightᵀ (FP16)
    # ------------------------------------------------------------------
    # Rearrange to (B·H, N, N) for torch.bmm
    left = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (Tensor‑core accelerated)
    out = torch.bmm(left, right.transpose(1, 2))

    # Restore the original layout [B,N,N,hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)

    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate                               # element‑wise gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)                      # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
