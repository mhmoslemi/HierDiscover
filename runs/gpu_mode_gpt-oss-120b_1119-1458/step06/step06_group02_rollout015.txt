@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_bh = tl.program_id(2)  # batch*hidden block

    # compute valid ranges for i, j, b/h
    i_offs = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offs = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute batch and hidden: convert pid_bh to b and h
    batch_hidden = pid_bh
    b = batch_hidden // H
    h = batch_hidden % H

    # Compute base pointers for left and right for each i, j block: we need to accumulate over k
    # We will compute pointer offsets for left: left_ptr + b*stride_left_b + i*stride_left_i + k*stride_left_k + h*stride_left_h
    # For right: right_ptr + b*stride_right_b + j*stride_right_i + k*stride_right_k + h*stride_right_h

    # Build offsets for k dimension:
    k_offs = tl.arange(0, BLOCK_K)

    # Determine mask for out-of-bounds for i,j:
    mask_i = i_offs < N
    mask_j = j_offs < N

    # Initialize accumulator:
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks:
    for k_start in range(0, N, BLOCK_K):
        # k range, compute effective block size:
        k_curr = k_start + k_offs
        mask_k = k_curr < N

        # Compute offsets for left:
        # left offset = b*stride_left_b + i_offs[:,None]*stride_left_i + k_curr[None,:]*stride_left_k + h*stride_left_h
        # Actually left dimension uses hidden as innermost dimension; but we include it into offset.
        left_offs = (b * stride_left_b) + (i_offs[:, None] * stride_left_i) + (k_curr[None, :] * stride_left_k) + (h * stride_left_h)
        # left load
        a = tl.load(left_ptr + left_offs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute offsets for right (transposed):
        # right[ b, j, k, h ], but we need B_T(k,j) => shape (k,j) means offsets: j_offs[None,:] => index for j dimension as column, and k_curr[:, None] as row
        # Right base offset for element right[b, j, k, h] = b*stride_right_b + j*stride_right_i + k*stride_right_k + h*stride_right_h
        # But we need to load block with shape (BLOCK_K, BLOCK_N) as (k, j). We can construct offsets similarly:
        right_offs = (b * stride_right_b) + (j_offs[None, :] * stride_right_i) + (k_curr[:, None] * stride_right_k) + (h * stride_right_h)
        b_tile = tl.load(right_ptr + right_offs, mask=mask_j[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

        # Compute dot product across k dimension via dot or matmul? Since we have a shape (BLOCK_M, BLOCK_K) and b_tile shape (BLOCK_K, BLOCK_N). Use tl.dot.
        # However tl.dot accepts (M,K)*(K,N) shape? tf expects a dims (M,K) and b dims (K,N): returns (M,N)
        acc += tl.dot(a, b_tile)

        # Possibly next k block
    # End for

    # Write output: compute offset for out pointer:
    out_offs = (b * stride_out_b) + (i_offs[:, None] * stride_out_i) + (j_offs[None, :] * stride_out_j) + (h * stride_out_h)
    tl.store(out_ptr + out_offs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Highly‑optimized forward pass of the outgoing TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ First LayerNorm (float32) – stabilises statistics before casting to fp16.
2️⃣ Cast activations to fp16 for the heavy part of the computation.
3️⃣ Fuse the “proj” and “gate” linear layers for the left/right branches:
   * weight matrices `left_proj` & `left_gate` are concatenated and applied
     in a single matmul → shape (2·H, C).  The resulting tensor is split into
     proj and gate parts, the gate part is passed through a sigmoid.
   * same trick for the right branch.
4️⃣ Apply mask (if present) and the left/right gates in a fused Triton kernel
   (one launch per branch).  This reduces memory traffic compared to three
   separate elementwise passes.
5️⃣ O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is implemented as a custom batched GEMM kernel written in Triton.
   The hidden dimension H is folded into the batch dimension (B·H batches)
   and each kernel instance computes a tile of the (i,j) output matrix.
6️⃣ Second LayerNorm (fp16) over the hidden dimension.
7️⃣ Apply the output gate (sigmoid) and a final linear projection back to the
   original dimension.
8️⃣ Cast the result back to float32 (required by the harness).

Only the forward pass is implemented – no autograd is required.
"""

import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: fuse mask, gate and elementwise multiplication.
#   `x`  – tensor to be updated (left or right), shape (B, N, N, H)
#   `gate` – sigmoid‑gate tensor, same shape
#   `mask` – boolean/float mask, shape (B, N, N) broadcast over H
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,               # *fp16
    gate_ptr,            # *fp16
    mask_ptr,            # *fp16 (broadcasted)
    hidden_dim,          # i32
    size,                # i32 – total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr, # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices for this block
    mask_active = offs < size                         # guard for out‑of‑bounds

    # Each hidden element repeats the same mask; divide by hidden_dim
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=mask_active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=mask_active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=mask_active, other=1.0)  # mask=1 outside valid range

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask_active)


# ----------------------------------------------------------------------
# Kernel 2: Batched GEMM for the O(N³) contraction.
#   Computes, for each (b,h):
#       C[b,h,i,j] = Σₖ A[b,h,i,k] * B[b,h,j,k]   (i,j over N)
#   where A = left, B = right.
#   Note that B is accessed transposed (k ↔ j) inside the loop.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,   # *fp16  (B,N,N,H)
    right_ptr,  # *fp16  (B,N,N,H)
    out_ptr,    # *fp16  (B,N,N,H)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    # strides for left   (b, i, k, h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right  (b, j, k, h) – note the first N‑axis is “j”
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out    (b, i, j, h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile size along i
    BLOCK_N: tl.constexpr,  # tile size along j
    BLOCK_K: tl.constexpr,  # tile size along k
    num_warps: tl.constexpr,
):
    # ------------------------------------------------------------------
    #   Program IDs:
    #   pid_i → tile of rows (i dimension)
    #   pid_j → tile of cols (j dimension)
    #   pid_bh → combined batch * hidden index
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Compute i / j start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    i = i_start + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    j = j_start + tl.arange(0, BLOCK_N)   # (BLOCK_N,)

    # Guard for out-of‑bounds rows / cols
    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    #   De‑compose batch & hidden index
    # ------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    #   Accumulator (float32 for numerical safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #   Main K‑loop
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Left block: (i, k) with hidden h fixed
        # offset = b*stride_l_b + i*stride_l_i + k*stride_l_k + h*stride_l_h
        left_offs = (
            b * stride_l_b
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
            + h * stride_l_h
        )
        a = tl.load(left_ptr + left_offs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Right block (transposed): we need right[j,k,h] as Bᵀ(k,j)
        # offset = b*stride_r_b + j*stride_r_i + k*stride_r_k + h*stride_r_h
        right_offs = (
            b * stride_r_b
            + j[None, :] * stride_r_i
            + k[:, None] * stride_r_k
            + h * stride_r_h
        )
        b_tile = tl.load(right_ptr + right_offs,
                         mask=mask_j[None, :] & mask_k[:, None],
                         other=0.0).to(tl.float32)

        # Compute partial dot‑product
        acc += tl.dot(a, b_tile)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    #   Store result
    # ------------------------------------------------------------------
    out_offs = (
        b * stride_o_b
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptr + out_offs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _batched_gemm_triton(left: torch.Tensor,
                         right: torch.Tensor,
                         block_m: int = 32,
                         block_n: int = 32,
                         block_k: int = 32) -> torch.Tensor:
    """
    Compute the O(N³) contraction using the Triton GEMM kernel.
    Both `left` and `right` must be fp16 and have shape [B, N, N, H].
    Returns a tensor of identical shape (fp16).
    """
    B, N, _, H = left.shape

    # Retrieve strides (in elements, not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = left.stride()  # output stride same as left/right

    # Output buffer
    out = torch.empty_like(left)

    # Grid dimensions
    grid_m = (N + block_m - 1) // block_m
    grid_n = (N + block_n - 1) // block_n
    grid_hb = B * H
    grid = (grid_m, grid_n, grid_hb)

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,
    )
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input tensor   [B, N, N, C]  (float32)
        torch.Tensor,               # mask tensor    [B, N, N]    (float32/bool) or None
        Dict[str, torch.Tensor],    # model weights (float32)
        Dict[str, Any]              # config dict (must contain dim, hidden_dim, optional nomask)
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        inp, (dim,), weight=norm_w, bias=norm_b
    )  # [B,N,N,dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate linear layers for left/right branches
    # ------------------------------------------------------------------
    # Concatenate the projection and gate weight matrices (out_features stacked)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # shape (2*hidden, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # shape (2*hidden, dim)

    # Linear passes (produces proj|gate tensors)
    left_comb = torch.nn.functional.linear(x, left_comb_w)   # [B,N,N,2*H]
    right_comb = torch.nn.functional.linear(x, right_comb_w) # [B,N,N,2*H]

    # Split projection and gate components
    left_proj = left_comb[..., :hidden]                       # [B,N,N,H]
    left_gate_pre = left_comb[..., hidden:]                   # [B,N,N,H]
    right_proj = right_comb[..., :hidden]                     # [B,N,N,H]
    right_gate_pre = right_comb[..., hidden:]                 # [B,N,N,H]

    # Apply sigmoid to the gate pre‑activations
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Trition kernel)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is shape [B, N, N]; broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                # B·N·N·H
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden,
            total_elts,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden,
            total_elts,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernels, `left_proj` and `right_proj` already contain the
        # gated & masked values.
    else:
        # No mask – simple element‑wise product (in‑place to save memory)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # Ensure the tensors are contiguous before feeding them to the Triton GEMM.
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction via custom Triton GEMM (batched matmul)
    # ------------------------------------------------------------------
    # out has shape [B, N, N, hidden]  (fp16)
    out_mid = _batched_gemm_triton(left_proj, right_proj,
                                   block_m=32, block_n=32, block_k=32)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_mid = torch.nn.functional.layer_norm(
        out_mid,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate (sigmoid) and final linear projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x, weights["out_gate.weight"].to(torch.float16)
        )
    )
    out_mid = out_mid * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out_mid, to_out_w)   # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
