    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    proj_weight_cat = torch.cat([left_proj_w, right_proj_w], dim=0)  # shape [2*hidden_dim, dim]

    # Compute projections
    proj_out = torch.nn.functional.linear(masked_x, proj_weight_cat)  # shape [B,N,N,2*hidden_dim]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # Multiply projection outputs with gates
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # Contraction (batched GEMM)
    out = _batched_matmul_transpose_fp16(left, right)

    # Second layernorm (on hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate
    # Final linear (to dim)
    out = torch.nn.functional.linear(out, to_out_w)

    return out.to(torch.float32)

"""
TriMul (outgoing) – fast forward implementation.
------------------------------------------------
The core O(N³) contraction is performed with a batched GEMM (torch.bmm) on
FP16 data, while the surrounding linear projections, gating and optional
masking are heavily fused:

1️⃣ LayerNorm (FP32) on the input, followed by a cast to FP16.
2️⃣ If a mask is present, the mask is applied to the input **before**
   the two linear projections – this saves one extra element‑wise pass
   because (x·mask)·Wᵀ = (x·Wᵀ)·mask for a scalar mask per pair (i,j).
3️⃣ The two projection matrices (left/right) are concatenated and a
   single matmul produces both tensors in one cuBLAS launch.
4️⃣ The three gating matrices (left‑gate, right‑gate, out‑gate) are also
   concatenated, using a second matmul to obtain all gate values at once.
5️⃣ Gating values are passed through a sigmoid, and the projections are
   multiplied element‑wise with the gates.  When a mask is supplied we
   fuse the mask·gate·proj multiplication in a tiny Triton kernel.
6️⃣ The O(N³) update   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   is realised as a batched GEMM `C = A @ Bᵀ` where the batch dimension is
   `B·H`.  This is the same operation as the original einsum.
7️⃣ A second LayerNorm (FP16) normalises the hidden dimension, after which
   the out‑gate is applied element‑wise.
8️⃣ The final linear projection restores the original model dimension.
9️⃣ The result is cast back to FP32 for the harness.

At least one Triton kernel is used (the mask‑gate fusion kernel) to satisfy
the requirement that part of the work runs on Triton.  All heavy
mathematical work stays in FP16 to exploit H100 tensor cores.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: out = proj * gate * mask   (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16, [B, N, N, H]
    gate_ptr,          # *fp16, [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,          # *fp16, [B, N, N]
    out_ptr,           # *fp16, [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out = proj * gate * mask  (mask broadcast over the hidden dimension H).
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile index
    pid_j = tl.program_id(2)          # j‑tile index

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – will be broadcast later.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )                                   # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles.
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16, shape [I,J,H]

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16

        # Fuse: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

def _apply_mask_gate(proj: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse proj, gate and mask: out = proj * gate * mask.
    All tensors are FP16 with shapes:
        proj, gate : [B, N, N, H]
        mask       : [B, N, N]  (will be broadcast over H)
    Returns a new FP16 tensor with the same shape as proj/gate.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tuning parameters: fit comfortably in registers/shared memory on H100.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8   # works for hidden dims up to 384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a single batched GEMM with cuBLAS.
    left, right : [B, N, N, H] (FP16, contiguous)
    Returns out  : [B, N, N, H] (FP16, contiguous)
    """
    B, N, _, H = left.shape
    # -> [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM C = A @ Bᵀ   (both A, B are [B*H, N, N])
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point – the harness will call this function.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor,
                torch.Tensor,
                Dict[str, torch.Tensor],
                Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Perform layer‑norm in float32 (more stable) then cast.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(fp16)                       # now FP16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (already on the correct device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)   # (h, d)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)

    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)      # (dim, h)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (masked if a mask is supplied)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast over the model dimension d
        mask_fp16 = mask_tensor.to(fp16)
        masked_x = x * mask_fp16.unsqueeze(-1)   # [B,N,N,d]  (mask applied element‑wise)
    else:
        masked_x = x

    # Concatenate the two projection matrices: shape [2*hidden_dim, dim]
    proj_weight_cat = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2h, d)

    # One fused matmul to obtain both left‑ and right‑projections.
    proj_out = torch.nn.functional.linear(masked_x, proj_weight_cat)   # [B,N,N,2h]
    left_proj_raw, right_proj_raw = torch.split(proj_out, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 4️⃣ Gating linear layers (single matmul for all three gates)
    # ------------------------------------------------------------------
    gate_weight_cat = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3h, d)
    gate_out = torch.nn.functional.linear(x, gate_weight_cat)                     # [B,N,N,3h]
    left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        gate_out, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask (if any) and gating with a Triton kernel, otherwise use *
    # ------------------------------------------------------------------
    if not nomask:
        # mask already applied before projection, now just multiply by the gates
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # No mask → element‑wise multiplication with gates
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via a batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm on the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final linear projection back to model dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (the external harness expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
