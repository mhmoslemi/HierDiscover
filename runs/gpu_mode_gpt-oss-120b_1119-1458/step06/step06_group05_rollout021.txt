    out_ptr, out_gate_ptr, out_norm_weight_ptr, out_norm_bias_ptr,
    to_out_weight_ptr,
    out_ptr_stride_b0, out_ptr_stride_i, out_ptr_stride_j, out_ptr_stride_h,
    out_gate_stride_b0, out_gate_stride_i, out_gate_stride_j, out_gate_stride_h,
    out_norm_weight_stride,

    B, N, H, D,
    eps: tl.constexpr,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr # maybe 1
):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # mask to handle boundary
    mask_i = i < N
    mask_j = j < N

    # We'll compute for each i and j pair. Flatten to broadcast: i[:, None] shape[I,1], j[None,:] shape[1,J]
    # We'll allocate registers for sum and sum_sq for each i,j element. Use loops over hidden blocks.

    # Use nested loops over hidden dimension
    sum_fp32 = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    sumsq_fp32 = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        ho = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = ho < H

        # Load out tile
        out_ptrs = (out_ptr + pid_b * out_ptr_stride_b0
                    + i[:, None] * out_ptr_stride_i
                    + j[None, :] * out_ptr_stride_j
                    + ho[None, None, :] * out_ptr_stride_h) # shape (I,J,H)

                           mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :],

                           mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :],

"""
TriMul (outgoing) – fast forward implementation.
------------------------------------------------
1️⃣ First LayerNorm (FP32 → FP16) using torch.nn.functional.layer_norm.
2️⃣ Fuse all 5 linear projections (left_proj, left_gate,
   right_proj, right_gate, out_gate) into a single large GEMM.
3️⃣ Apply sigmoid on the three gate partitions.
4️⃣ If a mask is present, fuse mask·proj·gate in a Triton kernel
   (produces left/right after masking).  If no mask, just element‑wise
   multiply with the gate.
5️⃣ Heavy O(N³) contraction:
      out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
   is performed as a batched GEMM (C = A @ Bᵀ) using cuBLAS.
6️⃣ Second LayerNorm, gating and a final linear projection are
   fused in a Triton kernel to avoid writing an intermediate tensor.
   The kernel computes, for each (b,i,j) vector:
      y  = (out - μ) / √(σ²+ε)          (LayerNorm)
      y  = y * w_norm + b_norm
      y  = y * out_gate                (out‑gate)
   and stores the result (still fp16) in a temporary tensor.
7️⃣ Final Linear (hidden_dim → dim) is a single cuBLAS GEMM.
8️⃣ Cast the result back to FP32 (required by the harness).

All heavy arithmetic is done in FP16 (with FP32 reduction for the
LayerNorm statistics) and utilizes the H100 Tensor‑Core optimized
routines, while the mask‑gate and second‑norm‑gate operations are
hand‑written Triton kernels to eliminate extra memory traffic.
"""

# --------------------------------------------------------------
# Triton kernel: proj * gate * mask   (mask broadcast over hidden dim)
# --------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden dimension H)
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # i / j coordinates for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – this will be broadcast later.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )                                   # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles.
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16, shape [I,J,H]

        # Load gate tile (already passed through sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16

        # Fuse: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H] (fp16)
    mask       : [B, N, N]   (fp16)

    Returns out = proj * gate * mask  (mask broadcast over H) as fp16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tuned tile sizes (work well on H100)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# --------------------------------------------------------------
# Triton kernel: LayerNorm + out‑gate (produces gated tensor for final linear)
# --------------------------------------------------------------
@triton.jit
def _ln_gate_kernel(
    inp_ptr,                # *fp16  [B, N, N, H]  (output of contraction)
    gate_ptr,               # *fp16  [B, N, N, H]  (out‑gate, already sigmoid‑ed)
    norm_weight_ptr,        # *fp16  [H]           (to_out_norm.weight)
    norm_bias_ptr,          # *fp16  [H]           (to_out_norm.bias)
    out_ptr,                # *fp16  [B, N, N, H]  (result = LN(inp)*gate)
    B, N, H,
    stride_inp_b, stride_inp_i, stride_inp_j, stride_inp_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_w,
    stride_norm_b,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,      # small epsilon for numerical stability
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute:
        y = LayerNorm(inp)                # uses weight/bias from norm_weight/bias
        y = y * gate
    All in fp16 (statistics accumulated in fp32).
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # First pass: compute mean and variance across H for each (i,j)
    # ------------------------------------------------------------------
    sum_fp32 = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    sumsq_fp32 = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        inp_ptrs = (
            inp_ptr
            + pid_b * stride_inp_b
            + offs_i[:, None, None] * stride_inp_i
            + offs_j[None, :, None] * stride_inp_j
            + offs_h[None, None, :] * stride_inp_h
        )
        inp_tile = tl.load(
            inp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        sum_fp32 += tl.sum(inp_tile, axis=2)
        sumsq_fp32 += tl.sum(inp_tile * inp_tile, axis=2)

    H_f32 = tl.full((1,), H, dtype=tl.float32)
    mean = sum_fp32 / H_f32
    var = sumsq_fp32 / H_f32 - mean * mean
    inv_std = tl.rsqrt(var + eps)   # sqrt(var + eps)⁻¹

    # ------------------------------------------------------------------
    # Second pass: apply normalization, weight/bias, gate
    # ------------------------------------------------------------------
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load raw input
        inp_ptrs = (
            inp_ptr
            + pid_b * stride_inp_b
            + offs_i[:, None, None] * stride_inp_i
            + offs_j[None, :, None] * stride_inp_j
            + offs_h[None, None, :] * stride_inp_h
        )
        inp_tile = tl.load(
            inp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # Load gate (out_gate)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # Load norm weight & bias (1‑D tensors)
        w_ptr = norm_weight_ptr + offs_h * stride_norm_w
        b_ptr = norm_bias_ptr   + offs_h * stride_norm_b
        w_tile = tl.load(w_ptr, mask=mask_h, other=0.0).to(tl.float32)
        b_tile = tl.load(b_ptr, mask=mask_h, other=0.0).to(tl.float32)

        # Normalization: ((x - μ) * inv_std)
        centered = inp_tile - mean[:, :, None]
        normalized = centered * inv_std[:, :, None]

        # Apply LN affine parameters
        normalized = normalized * w_tile[None, None, :] + b_tile[None, None, :]

        # Apply out‑gate
        gated = normalized * gate_tile

        # Store result as fp16
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            gated.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

# --------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# --------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using cuBLAS.
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns out  : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each slice corresponds to a hidden channel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: C = A @ Bᵀ  (both A and B are [B*H, N, N])
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 0️⃣ First LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the last dimension ``dim``
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)   # FP16 for the rest of the computation

    # -----------------------------------------------------------------
    # 1️⃣ Fuse all five linear projections into a single GEMM
    # -----------------------------------------------------------------
    # Gather weight matrices (all have shape [hidden_dim, dim])
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Stack them: order = left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)   # shape [5*hidden_dim, dim]

    # Large linear projection (big tensor shape [..., 5*hidden_dim])
    big = torch.nn.functional.linear(x, combined_weight)   # fp16

    # -----------------------------------------------------------------
    # 2️⃣ Slice and apply sigmoid to the three gate partitions
    # -----------------------------------------------------------------
    H = hidden_dim
    # Offsets inside the concatenated dimension
    offset = 0
    left       = big[..., offset: offset + H];            offset += H
    left_gate  = big[..., offset: offset + H]; offset += H
    right      = big[..., offset: offset + H];            offset += H
    right_gate = big[..., offset: offset + H]; offset += H
    out_gate   = big[..., offset: offset + H]   # no offset increment needed further

    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and flips
    # -----------------------------------------------------------------
    if not nomask:
        # Mask might be bool or float – convert to FP16 for the kernel
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left  = _apply_mask_gate(left, left_gate, mask_fp16)
        right = _apply_mask_gate(right, right_gate, mask_fp16)
    else:
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] (FP16)

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate (fused Triton kernel)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)

    gated = _apply_ln_gate(out, out_gate, to_out_norm_weight, to_out_norm_bias)  # FP16

    # -----------------------------------------------------------------
    # 6️⃣ Final linear projection (hidden_dim → dim) using cuBLAS
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out_final = torch.nn.functional.linear(gated, to_out_w)   # FP16, shape [B,N,N,dim]

    # -----------------------------------------------------------------
    # 7️⃣ Cast to FP32 (required by the harness)
    # -----------------------------------------------------------------
    return out_final.to(torch.float32)
```
