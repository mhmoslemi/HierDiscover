"""
Outgoing TriMul forward pass (AlphaFold3 style) – highly‑optimized version.

Algorithm
---------
1.  LayerNorm on the input (FP32) → cast to FP16.
2.  **Fused linear projection**:
    All five linear maps (left_proj, left_gate, right_proj,
    right_gate, out_gate) are concatenated into a single weight
    matrix of shape (5*hidden_dim, dim) and applied in one GEMM.
    The result is split into the five tensors.
3.  Sigmoid on the raw gate tensors (still FP16).
4.  If a mask is present, fuse (proj * gate * mask) in a custom
    Triton kernel (`mask_gate_fuse_kernel`).  When `nomask==True`,
    the multiplication is done with a plain in‑place PyTorch op.
5.  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    performed as a batched GEMM (B*hidden_dim independent
    matrix multiplications) using torch.bmm, which calls the
    highly‑optimized cuBLAS Tensor‑Core kernels on H100.
6.  Second LayerNorm over the hidden dimension (FP16).
7.  Element‑wise out‑gate multiplication, final linear projection
    back to the original channel dimension, and cast back to FP32.

Only the forward pass is implemented; gradients are not required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) – mask is broadcast over the
# hidden dimension.  Operates on a flat contiguous view of a tensor
# of shape [B, N, N, hidden_dim] (total elements = B·N·N·hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # *fp16 : projection tensor (in‑place output)
    gate_ptr,          # *fp16 : gate tensor (already sigmoid-ed)
    mask_ptr,          # *fp16 : mask tensor of shape (B·N·N,)
    total_elements,    # i32 : total number of elements in proj/gate
    hidden_dim,        # i32 : hidden dimension size
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tiles
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (every hidden_dim elements share the same mask entry)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fused multiplication
    out = proj * gate * m

    # Store back in‑place
    tl.store(proj_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via a batched
    matrix multiplication (A @ Bᵀ) on FP16 data.  The batches are
    (B * H) independent (N×N) matrices.
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] and perform A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix‑matrix multiplication (cuBLAS Tensor‑Core)
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )

            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )

        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication & final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating (FP16)
    to_out_w = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast result back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```
