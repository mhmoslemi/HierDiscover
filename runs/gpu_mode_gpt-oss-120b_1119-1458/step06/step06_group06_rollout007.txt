"""
TriMul (outgoing) – fast fused forward pass

Algorithm
---------
1. FP32 LayerNorm on the 4‑D input → half‑precision.
2. Fuse the five linear maps (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single
   large GEMM:   [B·N·N, dim] @ [5·hidden_dim, dim].T.
   This reads the input tensor only once.
3. Split the fused result into the five streams and apply
   sigmoid to the gate streams.
4. If a mask is present, fuse element‑wise
   (proj * gate * mask) for the left and right streams with a
   tiny Triton kernel (`mask_gate_fuse_kernel`).
   Otherwise multiply proj * gate in‑place.
5. Heavy contraction
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (B·hidden_dim independent
   matrix‑multiplications) via torch.bmm (cuBLAS Tensor‑Core).
6. Second LayerNorm over the hidden dimension, out‑gate
   multiplication, and final linear projection back to the
   original channel dimension.
7. Cast the result back to float32.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place fuse (tensor * gate * mask).  Operates on a
# flat view of a contiguous tensor of shape [B, N, N, hidden_dim]
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, size = total_elements
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,) – broadcasted
    total_elements, # i32 : total number of elements in x
    hidden_dim,     # i32 : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = offs < total_elements

    # Load tensor elements
    x = tl.load(x_ptr + offs, mask=mask_valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)

    # Compute mask index (one mask value per hidden_dim entries)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask_valid)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, hidden_dim].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via
    a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each head is a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]       # projection width (c)
    nomask = config.get("nomask", True)     # True -> ignore mask

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16 for the rest
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # FP32
    x = x_norm.half()  # now FP16, shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear transformations into a single GEMM
    # ------------------------------------------------------------------
    # Stack weights along the output‑feature dimension
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0
    )  # shape (5*hidden_dim, dim), FP16

    # Flatten the 4‑D tensor to 2‑D for torch.nn.functional.linear
    B, N, _, _ = x.shape
    x_flat = x.reshape(B * N * N, dim)                # (B·N·N, dim)

    # One large matmul: (B·N·N, dim) @ (5*hidden_dim, dim).T → (B·N·N, 5*hidden_dim)
    fused_out_flat = torch.nn.functional.linear(x_flat, fused_weight)  # FP16
    fused_out = fused_out_flat.view(B, N, N, 5 * hidden_dim)  # [B,N,N,5*hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Slice the fused result and compute gates (sigmoid)
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj = fused_out[..., 0:h].contiguous()
    left_gate_raw = fused_out[..., h:2 * h].contiguous()
    right_proj = fused_out[..., 2 * h:3 * h].contiguous()
    right_gate_raw = fused_out[..., 3 * h:4 * h].contiguous()
    out_gate_raw = fused_out[..., 4 * h:5 * h].contiguous()

    left_gate = torch.sigmoid(left_gate_raw)   # FP16, shape [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask and gates
    # ------------------------------------------------------------------
    if nomask:
        # No mask → pure gated projection
        left = left_proj.mul_(left_gate)   # in‑place
        right = right_proj.mul_(right_gate)
    else:
        # Mask is present → fuse (proj * gate * mask) using Triton
        # Ensure contiguity for kernel launch
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B·N·N,)

        total_elements = left_proj.numel()               # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left stream
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right stream
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim], FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim], FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast result back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
