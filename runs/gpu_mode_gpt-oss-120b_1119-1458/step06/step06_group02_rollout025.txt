```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_mul_kernel(
    x_ptr,             # *fp16
    mask_ptr,          # *fp16, flattened mask
    hidden_dim: tl.constexpr,
    size: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)    # linear idx
    active = offs < size
    mask_offs = offs // hidden_dim  # each hidden entry repeats mask

    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)
    tl.store(x_ptr + offs, x * m, mask=active)

@triton.jit
def tri_mul_batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_ai, stride_ak,
    batch_stride_b, stride_bi, stride_bk,
    batch_stride_c, stride_ci, stride_cj,

    Compute: C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
    where:
      A (left) and B (right) have shape [B, N, N, H] (fp16),
      C the same shape.
    The hidden dimension `h` is fused into the batch dimension via
    program_id(0). The kernel works directly on the original layout
    (no permutes) and accumulates in fp32.
    """
    pid = tl.program_id(0)   # combines batch and hidden-id
    pid_i = tl.program_id(1) # tile over i dimension (row)
    pid_j = tl.program_id(2) # tile over j dimension (col)

    # unpack batch index and hidden channel
    b_id = pid // hidden_dim
    h_id = pid - b_id * hidden_dim

    # Compute base pointers for this (b,h)
    a_base = a_ptr + b_id * batch_stride_a + h_id
    b_base = b_ptr + b_id * batch_stride_b + h_id
    c_base = c_ptr + b_id * batch_stride_c + h_id

    # Tile offsets within the N×N plane
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis k
    for k in range(0, K, BLOCK_K):
        k_off = k
        # condition on k
        k_valid = (k_off + tl.arange(0, BLOCK_K)) < K

        # Load A tile (i,k)
        a_ptrs = a_base + (offs_i[:, None] * stride_ai) \
                       + ((k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & k_valid[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile transposed: we need B_T[k,j] = B[j,k]
        b_ptrs = b_base + ((k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk) \
                        + (offs_j[None, :] * stride_bi)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write result back in FP16
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(

    Highly‑optimized forward pass of the “outgoing” TriMul operator.
    The implementation fuses:
      * two linear projections (proj + gate) per side via weight concatenation,
      * optional mask multiplication via a small Triton kernel,
      * the N³ contraction via a custom batched GEMM Triton kernel that works
        directly on the original [B, N, N, H] layout (no permutes/copies),
      * a second layer‑norm, gating and final linear projection using
        fp16 arithmetic.
    All heavy compute is performed in fp16 with fp32 accumulation.
    The final tensor is returned as float32.

    # Unpack inputs
    # ------------------------------------------------------------------

    hidden_dim = config["hidden_dim"]   # hidden dim (c_z)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stabilises statistics
    # ------------------------------------------------------------------

    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the compute
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 3️⃣ Concatenate projection / gate weights (reduce #matmuls)
    # ------------------------------------------------------------------
    left_weight_cat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"]
    ], dim=0).to(torch.float16)   # (2*hidden_dim, dim)

    right_weight_cat = torch.cat([
        weights["right_proj.weight"],
        weights["right_gate.weight"]
    ], dim=0).to(torch.float16)   # (2*hidden_dim, dim)

    out_gate_weight = weights["out_gate.weight"].to(torch.float16)      # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight      = weights["to_out.weight"].to(torch.float16)    # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Project + gate in a single linear call per side
    # ------------------------------------------------------------------
    left_comb = torch.nn.functional.linear(x, left_weight_cat)   # [B,N,N,2*hidden_dim]
    right_comb = torch.nn.functional.linear(x, right_weight_cat)

    # split into “proj” and “gate” halves
    left_proj, left_gate_raw = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate_raw = torch.split(right_comb, hidden_dim, dim=-1)

    left_gate = torch.sigmoid(left_gate_raw)      # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(right_gate_raw)

    left = left_proj * left_gate   # apply gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Output gate (independent of the left/right gates)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask – fused element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N], expand to [B,N,N,hidden_dim] implicitly
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()                # B * N * N * hidden_dim
        BLOCK = 1024
        grid_mask = ((total_elems + BLOCK - 1) // BLOCK,)

        mask_mul_kernel[grid_mask](
            left,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        mask_mul_kernel[grid_mask](
            right,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via custom Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim, "Hidden dimension mismatch"

    # output placeholder
    out = torch.empty_like(left, dtype=torch.float16)

    # strides for the layout [B, N, N, H] (contiguous)
    stride_b = left.stride(0)   # B‑stride  = N*N*H
    stride_i = left.stride(1)   # i‑stride  = N*H
    stride_j = left.stride(2)   # k‑stride  = H
    stride_h = left.stride(3)   # = 1

    total_batches = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

        total_batches,
        tl.cdiv(N, BLOCK_M),
        tl.cdiv(N, BLOCK_N),

        out,
        (hidden_dim,),
        weight=to_out_norm_weight,

    # ------------------------------------------------------------------

    out = torch.nn.functional.linear(out, to_out_weight)

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
Highly‑optimized “outgoing” TriMul forward pass (AlphaFold‑3).

Key ideas
---------
1️⃣  First LayerNorm (FP32) – stabilises statistics before the heavy part.
2️⃣  Cast to fp16; all large‑scale ops use half‑precision with fp32 accumulation.
3️⃣  Fuse the projection and its gate per side by concatenating the two weight
    matrices and doing a single `torch.nn.functional.linear`.  This cuts the
    number of GEMMs from 6 → 4.
4️⃣  Optional mask multiplication is performed by a tiny Triton kernel that
    fuses `x = x * mask` for all hidden channels.
5️⃣  The O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is realised as a custom batched GEMM kernel (`tri_mul_batch_matmul_kernel`).
    The kernel works on the original layout `[B, N, N, H]` (no `permute` /
    `contiguous` copies) and accumulates in FP32.
6️⃣  Second LayerNorm (FP16) over the hidden dimension, followed by the output
    gate and a final linear to project back to the original dimension.
7️⃣  The result is cast back to FP32 for the downstream harness.

Only the forward pass is needed; no gradients are computed.
"""

import math
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: multiply a tensor (shape B·N·N·H) by a broadcast mask
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    x_ptr,               # *fp16  – tensor to be masked (flattened view)
    mask_ptr,            # *fp16  – mask of shape B·N·N (flattened)
    hidden_dim: tl.constexpr,
    size: tl.constexpr,  # total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # linear element indices
    active = offs < size                               # guard OOB
    # each contiguous H entries share the same mask element
    mask_offs = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)
    tl.store(x_ptr + offs, x * m, mask=active)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
#   C[b,i,j,h] = Σₖ A[b,i,k,h] * B[b,j,k,h]
# A, B, C are in layout [B, N, N, H] (row‑major, H innermost).
# The hidden dimension `h` is fused into the batch axis.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_ai, stride_ak,
    batch_stride_b, stride_bi, stride_bk,
    batch_stride_c, stride_ci, stride_cj,
    M, N, K,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids:
    pid = tl.program_id(0)        # combined batch & hidden index
    pid_i = tl.program_id(1)      # tile over i (rows)
    pid_j = tl.program_id(2)      # tile over j (cols)

    # ------------------------------------------------------------------
    # decode batch and hidden‑channel from pid
    # ------------------------------------------------------------------
    b_id = pid // hidden_dim
    h_id = pid - b_id * hidden_dim

    # base pointers for this (b,h) slice
    a_base = a_ptr + b_id * batch_stride_a + h_id
    b_base = b_ptr + b_id * batch_stride_b + h_id
    c_base = c_ptr + b_id * batch_stride_c + h_id

    # offsets for the i/j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        k_off = k
        k_valid = (k_off + tl.arange(0, BLOCK_K)) < K

        # ---- load A tile: shape (BLOCK_M, BLOCK_K) ----
        a_ptrs = a_base + (offs_i[:, None] * stride_ai) \
                       + ((k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & k_valid[None, :],
                    other=0.0).to(tl.float32)

        # ---- load B tile transposed (we need B_T[k,j] = B[j,k]) ----
        b_ptrs = b_base + ((k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk) \
                        + (offs_j[None, :] * stride_bi)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ---- accumulate ----
        acc += tl.dot(a, b)   # (BLOCK_M,K) @ (K,BLOCK_N) -> (BLOCK_M,BLOCK_N)

    # ------------------------------------------------------------------
    # store result back (FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])
