"""
Highly‑optimized forward pass for the “outgoing” TriMul operator (AlphaFold3).

Algorithm (FP16, minimal memory traffic)
-------------------------------------------------
1. Layer‑norm the input (FP32) → cast to FP16.
2. Fuse all linear projections (left/right projections, gating streams and the
   out‑gate) into a single matmul:
       combined = x @ W_combinedᵀ
   where W_combined = [left_proj; left_gate; right_proj; right_gate; out_gate].
   This reads the input only once and eliminates four separate matmul launches.
3. Split the combined tensor, apply the three sigmoids, and fuse the
   element‑wise product (proj * gate).  If a mask is present it is applied
   after the gating step (a tiny Triton kernel is used for the mask‑multiply).
4. Heavy O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed as a batched GEMM (A @ Bᵀ) using torch.bmm.
5. Layer‑norm over the hidden dimension, out‑gate multiplication,
   final linear projection back to `dim`.
6. Cast the result back to FP32 (the harness expects FP32).

All heavy‑compute kernels are run in FP16 (Tensor‑Core usage), while
layer‑norms accumulate in FP32 for numerical stability.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: mask * tensor (broadcast mask over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    tensor_ptr,      # *fp16 : tensor to be masked, shape = total_elements
    mask_ptr,        # *fp16 : mask tensor, shape = (num_positions,)
    total_elements,  # i32   : total number of elements in tensor (B*N*N*H)
    hidden_dim,      # i32   : hidden dimension (size of the broadcast axis)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensor element
    t = tl.load(tensor_ptr + offs, mask=valid, other=0.0)

    # Find the mask index (every `hidden_dim` entries share the same mask)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(tensor_ptr + offs, t * m, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]."""
    B, N, _, H = left.shape      # left/right are [B, N, N, H] (FP16)

    # reshape to [B*H, N, N] and perform A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # bmm does (B*H, N, N) @ (B*H, N, N).T → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True ⇒ mask ignored)

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, same device as the input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).half()                                   # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear layers into one matmul
    # ------------------------------------------------------------------
    # Order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                                   # (5*hidden_dim, dim)

    # combined shape -> [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_weight)  # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split combined tensor
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj = combined[..., :h]
    left_gate_raw = combined[..., h : 2 * h]
    right_proj = combined[..., 2 * h : 3 * h]
    right_gate_raw = combined[..., 3 * h : 4 * h]
    out_gate_raw = combined[..., 4 * h : 5 * h]

    # ------------------------------------------------------------------
    # 4️⃣ Activations (sigmoid) and element‑wise fusions
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)          # [B,N,N,h]
    right_gate = torch.sigmoid(right_gate_raw)        # [B,N,N,h]
    out_gate = torch.sigmoid(out_gate_raw)            # [B,N,N,h]

    # fuse projection * gate (in‑place)
    left = left_proj.mul_(left_gate)                  # left now holds final tensor
    right = right_proj.mul_(right_gate)               # right now holds final tensor

    if not nomask:
        # mask is broadcast over the hidden dimension → flatten for Triton kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        total_elements = left.numel()
        BLOCK = 8192
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left *= mask
        mask_mul_kernel[grid](
            left,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right *= mask
        mask_mul_kernel[grid](
            right,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    # If nomask=True the multiplication is omitted (everything stays in‑place)

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction (batched GEMM)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)            # [B, N, N, h] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                               # element‑wise gating (fp16)
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].half()
    )                                                   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
