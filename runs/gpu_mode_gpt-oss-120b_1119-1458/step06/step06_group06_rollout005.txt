def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr, total_elements, hidden_dim, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # compute mask index
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    result = x * gate * mask_val
    tl.store(x_ptr + offs, result, mask=mask)

"""
TriMul (outgoing) forward pass – H100‑optimized.

Algorithm
---------
1.  Input LayerNorm (FP32) → cast to FP16 for all subsequent ops.
2.  Fuse linear projections:
        - left/right projections are computed together by concatenating the
          two weight matrices and applying a single torch.nn.functional.linear.
        - left/right/out gates are also fused (3× hidden_dim) into one linear,
          followed by a sigmoid.
3.  If a mask is present, fuse (proj * gate * mask) for the left and right
    streams with a tiny Triton kernel (`mask_gate_fuse_kernel`).  The mask is
    broadcast over the hidden dimension, therefore only one launch per tensor.
    When there is no mask the gated multiplication is done in‑place.
4.  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM (A @ Bᵀ) using torch.bmm on FP16 data,
    which maps to cuBLAS Tensor‑Core kernels.
5.  Second LayerNorm (FP16) → element‑wise out‑gate → final Linear
    (hidden_dim → dim) back to the original channel size.
6.  Cast the result back to FP32 (the harness expects float32).

Only the forward pass is required; gradients are omitted.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over the
# hidden dimension.  Operates on a flat view of a contiguous tensor of shape
# [B, N, N, hidden_dim] (size = B·N·N·hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated in‑place, size = total_elements
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,) – broadcasted
    total_elements,  # i32 : total number of elements in x (B·N·N·hidden_dim)
    hidden_dim,      # i32 : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets handled by this block
    mask_active = offs < total_elements                # guard for tail elements

    # Load the two tensors (masked loads keep undefined values safe)
    x = tl.load(x_ptr + offs, mask=mask_active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask_active, other=0.0)

    # Each consecutive `hidden_dim` entries share the same mask entry.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask_active, other=1.0)

    # Fused multiplication: (proj * gate * mask)
    out = x * g * m

    # Store the result back in‑place
    tl.store(x_ptr + offs, out, mask=mask_active)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] via a batched
    matrix multiplication (A @ Bᵀ) on FP16 data.  The result is returned as FP16.
    """
    B, N, _, H = left.shape

    # From [B, N, N, H] → [B*H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]      (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                       # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]          # projection width (c)
    nomask = config.get("nomask", True)       # True → ignore the mask

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm in FP32 (more stable) → cast to FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32
    x = x_norm.half()  # switch to FP16 for the heavy ops

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 once (avoid per‑call casts)
    # ------------------------------------------------------------------
    # Normalisation weights
    norm_w_fp16 = weights["norm.weight"].half()  # not used after x_norm (kept for completeness)
    norm_b_fp16 = weights["norm.bias"].half()   # not used after x_norm

    # Projection matrices (left & right) – concatenated for a single matmul
    proj_weight_cat = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).half()  # shape (2*hidden_dim, dim)

    # Gating matrices (left_gate, right_gate, out_gate) – concatenated
    gate_weight_cat = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # shape (3*hidden_dim, dim)

    # Second‑stage normalisation and output projection
    to_out_norm_w_fp16 = weights["to_out_norm.weight"].half()
    to_out_norm_b_fp16 = weights["to_out_norm.bias"].half()
    to_out_w_fp16 = weights["to_out.weight"].half()  # shape (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (compressed into two matmuls for efficiency)
    # ------------------------------------------------------------------
    # Combined left/right projection: result ... [2*hidden_dim]
    proj = torch.nn.functional.linear(x, proj_weight_cat)  # [B, N, N, 2*hidden_dim]
    left = proj[..., :hidden_dim]   # left projection
    right = proj[..., hidden_dim:]  # right projection

    # Combined gating (left_gate, right_gate, out_gate): result ... [3*hidden_dim]
    gates = torch.nn.functional.linear(x, gate_weight_cat)  # [B, N, N, 3*hidden_dim]
    left_gate_raw = gates[..., :hidden_dim]
    right_gate_raw = gates[..., hidden_dim: 2 * hidden_dim]
    out_gate_raw = gates[..., 2 * hidden_dim:]

    # Apply sigmoid to gating signals (still FP16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)   # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Mask‑gate fusion (Triton) – only if a mask is present
    # ------------------------------------------------------------------
    if nomask:
        # Simple in‑place multiplication when no mask is required
        left = left.mul_(left_gate)
        right = right.mul_(right_gate)
    else:
        # Ensure tensors are contiguous for the Triton kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        # Convert mask → FP16 and flatten to 1‑D (B·N·N)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)

        total_elements = left.numel()            # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Left stream: proj * gate * mask in‑place
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Right stream: proj * gate * mask in‑place
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction (batched GEMM) – cuBLAS Tensor‑Core matmul
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w_fp16,
        bias=to_out_norm_b_fp16,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                 # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w_fp16)  # → [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast result back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```

This implementation follows the reference TriMul algorithm while:

* Reducing the number of separate linear layers (left/right projections and the three gates are each computed in a *single* matmul via weight concatenation).
* Performing the mask‑gate interaction completely on the GPU with a compact Triton kernel.
* Keeping the dominant O(N³) contraction as a batched GEMM, which on H100 maps to highly‑optimised cuBLAS Tensor‑Core kernels.
* Operating almost exclusively in FP16, only using FP32 for the initial LayerNorm (numerical stability).
