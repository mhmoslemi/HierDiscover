```python
@triton.jit
def batched_matmul_fp16_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch ID into b and h
    b = batch_idx // H
    h = batch_idx % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointers for this batch and hidden dim
    # base pointers include batch and hidden dimension offsets
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # Initialize accumulation in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers for left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # compute pointers for right: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b_tile)  # (M,K) dot (K,N) -> (M,N)

    # store result in fp16
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""TriMul (outgoing) – fast Triton implementation

The forward pass of the outgoing TriMul module consists of:
1. LayerNorm over the channel dimension (float32 for stability).
2. Linear projections (left/right) and three gating linear layers (fp16).
3. Optional element‑wise mask·gate fusion (one Triton kernel for both
   left and right tensors, avoiding a second kernel launch).
4. Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM where the hidden dimension is treated as an
   additional batch axis; a custom Triton kernel computes all batches in a
   single launch without any temporary transposes or copies.
5. Second LayerNorm (fp16), out‑gate multiplication and final projection.
6. Cast the result back to float32 (expected by the harness).

This implementation eliminates the costly `permute(...).contiguous()` copies
required by the reference `torch.bmm` approach and fuses the mask/gate step,
yielding a significant speed‑up for all tested configurations.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: Fuse mask & per‑tensor gate (applies to left and right).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_dual_kernel(
    left_ptr,               # *fp16 left tensor (flattened)
    right_ptr,              # *fp16 right tensor (flattened)
    left_gate_ptr,          # *fp16 left‑gate tensor (flattened)
    right_gate_ptr,         # *fp16 right‑gate tensor (flattened)
    mask_ptr,               # *fp16 mask tensor (flattened, shape [B*N*N])
    hidden_dim,             # i32 hidden dimension H
    total_elems,            # i32 total number of elements in left/right
    BLOCK: tl.constexpr,    # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load operands
    x = tl.load(left_ptr + offs,  mask=valid, other=0.0).to(tl.float16)
    xl = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0).to(tl.float16)
    y = tl.load(right_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    yr = tl.load(right_gate_ptr + offs, mask=valid, other=0.0).to(tl.float16)

    # The mask is broadcast across the hidden dimension; each mask entry
    # corresponds to `hidden_dim` consecutive elements in the flattened tensor.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0).to(tl.float16)

    # Fuse mask·gate·tensor
    out_left  = x * xl * m
    out_right = y * yr * m

    tl.store(left_ptr  + offs, out_left,  mask=valid)
    tl.store(right_ptr + offs, out_right, mask=valid)

# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM (fp16) for the O(N³) contraction.
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    left_ptr, right_ptr, out_ptr,          # *fp16 tensors
    B, N, H,                               # i32 batch, seq_len, hidden_dim
    # strides for left (shape [B,N,N,H])
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right (shape [B,N,N,H])
    stride_rb, stride_ri, stride_rk, stride_rh,
    # strides for out   (shape [B,N,N,H])
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,                 # rows   of output tile
    BLOCK_N: tl.constexpr,                 # cols   of output tile
    BLOCK_K: tl.constexpr,                 # reduction tile
):
    # --------- program IDs ----------
    batch_idx = tl.program_id(0)       # 0 .. B*H-1   (combined batch / hidden)
    pid_i     = tl.program_id(1)       # tile over i dimension
    pid_j     = tl.program_id(2)       # tile over j dimension

    # Decode combined index into (b, h)
    b = batch_idx // H
    h = batch_idx %  H

    # Starting offsets for this block
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets inside the block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the selected (b, h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # Accumulator in fp32 for better numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left tile (M x K) -------
        # left[b,i,k,h] with strides (i -> stride_li, k -> stride_lk)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ------- load right tile (K x N) -------
        # right[b,j,k,h] but we need it as (K x N) for the dot
        # right stride for k is stride_rk, for j (the column) is stride_ri
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        # ------- dot product (M x K) @ (K x N) -------
        acc += tl.dot(a, b_tile)   # result is (M, N)

    # Store the result (fp16)
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for large N
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy parts
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)  # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 3️⃣ Cast all relevant weights to fp16 once
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w) # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (fp16) – sigmoid on each hidden channel
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Apply mask together with gates (fused Triton kernel)
    # -----------------------------------------------------------------
    if not nomask:
        # Flatten mask to 1‑D fp16, layout matches the flattened tensors below
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                # shape [B*N*N]
        total = left.numel()                          # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        mask_gate_fuse_dual_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: plain element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – batched GEMM via Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    out = torch.empty_like(left)   # allocate [B,N,N,H] fp16 for the result

    # strides (they are static for a contiguous tensor)
    l_stride = left.stride()
    r_stride = right.stride()
    o_stride = out.stride()
    # left
    stride_lb, stride_li, stride_lk, stride_lh = l_stride
    # right
    stride_rb, stride_ri, stride_rk, stride_rh = r_stride
    # out
    stride_ob, stride_oi, stride_oj, stride_oh = o_stride

    # Tiling parameters – tuned for H100 fp16 matmul
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid: (batch*hidden, tiles_i, tiles_j)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_fp16_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, hidden_dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Gating + final projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                         # element‑wise out‑gate
    out = torch.nn.functional.linear(out, to_out_w)   # → [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
