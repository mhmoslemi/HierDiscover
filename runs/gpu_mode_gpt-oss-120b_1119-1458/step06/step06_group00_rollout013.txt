def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute block indices
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    total_batches = B * H
    
    batch_idx = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # base offset for the combined batch (b, h)
    b = batch_idx // H
    h = batch_idx % H

    j_valid = offs_j < N

            + offs_k[:, None] * stride_right_k \

def _tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # number of tiles per dimension
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    total_batches = B * H

    # Compute batch id and tile ids
    batch_idx = pid // (num_pid_m * num_pid_n)
    tile_rem = pid % (num_pid_m * num_pid_n)
    pid_m = tile_rem // num_pid_n
    pid_n = tile_rem % num_pid_n

    # Derive b and h
    b = batch_idx // H
    h = batch_idx % H

    # Compute base offset for this (b,h)
    offset_batch = b * stride_left_batch + h

    # Offsets for i and j (block rows and cols)
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k

            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j

        # Dot product accumulation
        acc += tl.dot(left, right)   # (M,N)

    # Convert accumulator to fp16

    # Store result
    out_ptrs = out_ptr + offset_batch \
        + offs_i[:, None] * stride_out_i \

"""
Outgoing TriMul kernel (AlphaFold‑3 style) implemented with Triton.

Algorithm
---------
1.  Layer‑Norm over the input (float32) → FP16.
2.  Linear projections:
        left  = x @ left_proj.w
        right = x @ right_proj.w
    Gates:
        left_gate  = sigmoid(x @ left_gate.w)
        right_gate = sigmoid(x @ right_gate.w)
        out_gate   = sigmoid(x @ out_gate.w)
3.  Fuse left/right gates and optional mask in‑place with a tiny Triton kernel.
4.  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched GEMM (B·H matrices of size N×N) performed by a custom
    Triton kernel that works directly on the original layout
    [B, N, N, H] → no extra permutes or copies.
5.  Second Layer‑Norm on the hidden result, element‑wise out‑gate, and a final
    linear projection back to `dim`.  These steps stay in FP16 and use
    PyTorch’s highly‑optimised kernels.
6.  Cast final tensor back to float32 for the caller.

All heavy arithmetic (steps 3‑4) runs on the GPU with Tensor‑cores and avoids
intermediate materialisation of large temporaries, giving a substantial speed‑up
versus a naïve implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1️⃣ Fuse mask and per‑head sigmoid gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in‑place)
    right_ptr,         # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcasted over H)
    total_elements,    # i64   = B·N·N·H
    H,                 # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)           # flat index
    valid = offs < total_elements

    # each H consecutive entries share the same mask value
    mask_offs = offs // H

    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)


# ----------------------------------------------------------------------
# 2️⃣ Batched GEMM for the TriMul contraction.
#      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_contract_kernel(
    left_ptr,          # *fp16  [B,N,N,H]
    right_ptr,         # *fp16  [B,N,N,H]
    out_ptr,           # *fp16  [B,N,N,H]   (output)
    B,                 # i64   batch size
    N,                 # i64   sequence length
    H,                 # i64   hidden dimension
    stride_left_batch, # i64   stride of left over batch dimension
    stride_left_i,     # i64   stride of left over first N dimension (i)
    stride_left_k,     # i64   stride of left over second N dimension (k)
    stride_right_batch,# i64   stride of right over batch dimension
    stride_right_j,    # i64   stride of right over first N dimension (j)
    stride_right_k,    # i64   stride of right over second N dimension (k)
    stride_out_batch,  # i64   stride of out over batch dimension (unused)
    stride_out_i,      # i64   stride of out over first N dimension (i)
    stride_out_j,      # i64   stride of out over second N dimension (j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Grid layout
    # --------------------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)            # tiles over i
    num_n = tl.cdiv(N, BLOCK_N)            # tiles over j
    total_batches = B * H                  # combined (batch, hidden) axis

    batch_tile = pid // (num_m * num_n)    # which (b,h) we are processing
    tile_rem  = pid % (num_m * num_n)      # linear tile id inside that batch
    tile_i    = tile_rem // num_n          # tile row   (i‑direction)
    tile_j    = tile_rem %  num_n          # tile col   (j‑direction)

    # derive concrete batch index b and hidden index h
    b_idx = batch_tile // H
    h_idx = batch_tile % H

    # starting offset for this (b,h) pair
    base_off = b_idx * stride_left_batch + h_idx   # same for left/right/out

    # --------------------------------------------------------------
    # Tile offsets in i / j dimensions
    # --------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (better precision for large reductions)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k‑tiles
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block = A[i,k]  (shape M×K)
        a_ptrs = left_ptr + base_off \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right block = Bᵀ[k,j]   we read right[b,j,k,…] and treat it as (K,N)
        b_ptrs = right_ptr + base_off \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate product
        acc += tl.dot(a, b)      # (M,N) ← (M,K)·(K,N)

    # --------------------------------------------------------------
    # Write the result back (convert to FP16)
    # --------------------------------------------------------------
    out_tile = acc.to(tl.float16)

    out_ptrs = out_ptr + base_off \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             out_tile,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
        mask_tensor  : torch.Tensor  [B, N, N]      (bool / 0‑1 float, optional)
        weights_dict : dict[str, torch.Tensor]       (float32)
        config_dict  : must contain ``dim`` and ``hidden_dim``; may contain
                       ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → stable statistics)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (no bias terms in the Linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device, dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections → (B,N,N,hidden_dim)  (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)     # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Gate projections (sigmoid) → (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask (+ optional) and gates in‑place with Triton
    # ------------------------------------------------------------------
    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:
        # mask can be bool or 0/1 float – cast to fp16 for the kernel
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gate tensors are now fused into left/right – we can free them
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction (batched GEMM) – Triton implementation
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    out_hidden = torch.empty_like(left)          # [B,N,N,hidden_dim] fp16

    # Strides (in elements) of the three tensors – they are identical layout
    stride_batch = left.stride(0)   # B‑stride = N·N·H
    stride_i     = left.stride(1)   # stride across the first N   = N·H
    stride_k     = left.stride(2)   # stride across the second N  = H

    # Output strides (same layout)
    out_stride_i = out_hidden.stride(1)
    out_stride_j = out_hidden.stride(2)

    # Blocking parameters – tuned for H100/Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = B * hidden_dim

    grid = (total_batches * num_m * num_n,)

    _tri_mul_contract_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_batch=stride_batch,
        stride_left_i=stride_i,
        stride_left_k=stride_k,
        stride_right_batch=stride_batch,
        stride_right_j=stride_i,          # right uses the same stride for its “j” dim
        stride_right_k=stride_k,
        stride_out_batch=stride_batch,    # not used in the kernel, kept for interface consistency
        stride_out_i=out_stride_i,
        stride_out_j=out_stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply the final out‑gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    out_hidden = out_hidden * out_gate                     # element‑wise gate
    out = F.linear(out_hidden, to_out_w)                  # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
