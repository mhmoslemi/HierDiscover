def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # Norm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # cast to fp16
    x_fp16 = x.to(torch.float16)

    # Combine projection and gating weights
    left_proj_w = weights['left_proj.weight']
    left_gate_w = weights['left_gate.weight']
    right_proj_w = weights['right_proj.weight']
    right_gate_w = weights['right_gate.weight']
    out_gate_w = weights['out_gate.weight']

    # concatenate
    combined_w = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0).to(torch.float16)  # shape [5*hidden_dim, dim]

    # Linear compute
    combined = torch.nn.functional.linear(x_fp16, combined_w)  # [B,N,N,5*H]

    H = hidden_dim
    left_proj = combined[..., :H]
    left_gate_pre = combined[..., H:2*H]
    right_proj = combined[..., 2*H:3*H]
    right_gate_pre = combined[..., 3*H:4*H]

    left_gate = torch.sigmoid(left_gate_pre)  # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Free combined to release memory if needed
    del combined
    # optional garbage.

    # mask gating
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()

        BLOCK = 1024
        grid = ((total_left + BLOCK - 1) // BLOCK,)

            hidden_dim,

        left = left_proj * left_gate

```python
import torch
import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,  # *fp16
    gate_ptr,  # *fp16
    mask_ptr,  # *fp16
    hidden_dim,  # int32
    size,   # int32
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < size
    # mask index: each hidden_dim entries share same mask entry
    mask_offs = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)
    tl.store(x_ptr + offs, x * g * m, mask=active)
```

"""
TriMul (outgoing) forward pass – highly‑optimized Triton‑augmented implementation.

Algorithm
---------
1️⃣  Layer‑Norm on the raw input (float32) → stabilises statistics.
2️⃣  Cast the normalized tensor to fp16 (the heavy part of the pipeline).
3️⃣  Fuse *all* linear projections (left/right projections, their gates and the
    output gate) into one large mat‑mul:
        combined = x @ W_comb.T   with   W_comb ∈ ℝ^(5·H×D)
    – the first H rows are left_proj,
      the next H rows are left_gate,
      the next H rows are right_proj,
      the next H rows are right_gate,
      the last H rows are out_gate.
4️⃣  Split the combined tensor, apply sigmoid to the three gate slices.
5️⃣  Fuse element‑wise mask, left‑gate and right‑gate multiplication via a tiny
    Triton kernel (only executed when a mask is present).  This yields the
    final “left” and “right” tensors of shape [B, N, N, H].
6️⃣  Compute the O(N³) contraction using a batched GEMM:
        out[i,j] = Σₖ left[...,k,:] · right[...,k,:]ᵀ
    performed as a single torch.bmm over the reshaped view
    [B·H, N, N] → [B·H, N, N].
7️⃣  Second Layer‑Norm (fp16) over the hidden dimension.
8️⃣  Multiply by the pre‑computed output gate (broadcasted over the hidden
    dimension) and project back to the original dimension with a final linear
    layer.
9️⃣  Cast the result back to float32 (the harness expects float32).

Only the forward path is implemented; gradients are not required.
A Triton kernel is used to fuse mask & gating, reducing memory traffic.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 – tensor to be updated (left_proj or right_proj)
    gate_ptr,       # *fp16 – gate tensor (same shape as x_ptr)
    mask_ptr,       # *fp16 – mask tensor (shape B·N·N)
    hidden_dim,     # i32   – size of hidden dimension H
    size,           # i32   – total number of elements in x_ptr
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets for this block
    active = offs < size                               # guard out‑of‑bounds

    # each hidden_dim contiguous entries share the same mask entry
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)  # mask=1 where out‑of‑range

    tl.store(x_ptr + offs, x * g * m, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the O(N³) contraction.
    left/right: [B, N, N, H] (fp16)
    Returns: [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each hidden channel becomes a batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))
    # restore shape
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be unused
        - weights_dict : mapping name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the computation
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse all linear projections (+ gating) into ONE matmul
    # -----------------------------------------------------------------
    # Gather weight matrices (float32 → fp16)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension: shape [5*H, D]
    combined_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)

    # Linear projection: result shape [B, N, N, 5*H]
    combined = torch.nn.functional.linear(x_fp16, combined_weight)

    # -----------------------------------------------------------------
    # 4️⃣ Slice combined output & apply sigmoids to the gate slices
    # -----------------------------------------------------------------
    H = hidden_dim
    left_proj    = combined[..., :H]               # [B, N, N, H]
    left_gate_pre = combined[..., H:2*H]
    right_proj   = combined[..., 2*H:3*H]
    right_gate_pre = combined[..., 3*H:4*H]
    out_gate_pre = combined[..., 4*H:]            # [B, N, N, H]

    left_gate  = torch.sigmoid(left_gate_pre)      # [B, N, N, H]
    right_gate = torch.sigmoid(right_gate_pre)    # [B, N, N, H]
    out_gate   = torch.sigmoid(out_gate_pre)      # [B, N, N, H]

    # Release the combined tensor early
    del combined, left_gate_pre, right_gate_pre, out_gate_pre

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fuse mask, projection and gate via Triton
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        # left = left_proj * left_gate * mask
        total = left_proj.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        total_r = right_proj.numel()
        grid_r = ((total_r + BLOCK - 1) // BLOCK,)
        mask_gate_fuse_kernel[grid_r](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_r,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel left_proj/right_proj hold the fused result
        left = left_proj
        right = right_proj
    else:
        # No mask – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Free intermediate tensors that are no longer needed
    del left_proj, left_gate, right_proj, right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑Core)
    # -----------------------------------------------------------------
    out_hidden = _batched_matmul_torch(left, right)   # [B, N, N, H] fp16

    # Cleanup
    del left, right

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) over the hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out_hidden = out_hidden * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out_hidden, to_out_w)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
