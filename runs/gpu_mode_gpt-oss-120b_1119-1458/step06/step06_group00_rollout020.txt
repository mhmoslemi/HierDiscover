@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_n_b, stride_k_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute pointers for A and B blocks
    a_ptrs = a_ptr + pid_batch * stride_batch_a + offs_m[:, None] * stride_m_a + (0) * stride_k_a
    b_ptrs = b_ptr + pid_batch * stride_batch_b + (0) * stride_k_b + offs_n[None, :] * stride_n_b

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # offsets for this tile
        k_offset = k * BLOCK_K
        # Compute pointers for next K tile
        a_tile_ptrs = a_ptrs + k_offset * stride_k_a
        b_tile_ptrs = b_ptrs + k_offset * stride_k_b

        # Load a tile and b tile
        a = tl.load(a_tile_ptrs, mask=offs_m[:, None] < M and k_offset + tl.arange(0, BLOCK_K)[None, :] < K,
                    other=0.0)   # shape (BLOCK_M, BLOCK_K)
        b = tl.load(b_tile_ptrs, mask=offs_n[None, :] < N and k_offset + tl.arange(0, BLOCK_K)[None, :] < K,
                    other=0.0)   # shape (BLOCK_K, BLOCK_N)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + pid_batch * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_size, M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, num_warps: tl.constexpr
):
    pid_batch = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block col index

    # Compute offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for a tile
    a_offset = pid_batch * stride_batch_a + offs_m[:, None] * stride_m_a + 0 * stride_k_a
    b_offset = pid_batch * stride_batch_b + 0 * stride_k_b + offs_n[None, :] * stride_n_b

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # offsets for subtree
        a_ptrs = a_ptr + a_offset + k_offset * stride_k_a
        b_ptrs = b_ptr + b_offset + k_offset * stride_k_b

        # Load tiles
        mask_a = (offs_m < M)[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K)[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        mask_b = (offs_n < N)[None, :] & (k_offset + tl.arange(0, BLOCK_K) < K)[:, None]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # Write result
    c_offset = pid_batch * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_offset, acc.to(tl.float16), mask=mask_c)
```

"""
Optimized TriMul (outgoing) forward pass using Triton.

The kernel performs three stages:
1. Fuse mask (if any) and the per‑head sigmoid gates in‑place on the left/right
   projections (FP16).
2. Compute the O(N³) pairwise contraction using a custom batched GEMM kernel.
   The contraction implements `einsum('...i k d, ...j k d -> ...i j d')` as a
   batch of matrix multiplications A @ Bᵀ where `A = left[b,h]` and
   `B = right[b,h]`.  Hidden‑dimension (`hidden_dim`) is folded into the batch
   dimension, so the kernel runs a single GEMM for each (batch, head) slice.
   The kernel is written in Triton with configurable tile sizes, Tensor‑core
   dot products (FP16 inputs, FP32 accumulation) and boundary masks.
3. The remaining cheap operations (LayerNorms, final gating, output linear)
   are performed with PyTorch.

All heavy computation is done in FP16 and accumulated in FP32 for numerical
stability.  The final result is cast back to float32 to match the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1) Mask + gate fusion kernel (in‑place on left/right)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i64   (B·N·N·H)
    H,                 # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat index
    valid = offs < total_elements

    mask_offs = offs // H  # each mask element corresponds to H consecutive hidden entries

    left_val      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right_val     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate_val= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    mask_val      = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left_val  * left_gate_val  * mask_val
    right_out = right_val * right_gate_val * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel for the O(N³) contraction.
#    Computes C[b] = A[b] @ B[b]ᵀ   for each batch slice.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,               # *fp16  [batch, M, K]
    b_ptr,               # *fp16  [batch, K, N]   (already transposed)
    c_ptr,               # *fp16  [batch, M, N]   output
    batch,               # i64   = B * hidden_dim
    M, N, K,             # i64
    stride_batch_a, stride_m_a, stride_k_a,   # strides for A
    stride_batch_b, stride_k_b, stride_n_b,   # strides for B (transposed)
    stride_batch_c, stride_m_c, stride_n_c,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)    # which (batch * hidden_dim) slice
    pid_m = tl.program_id(1)        # tile row index
    pid_n = tl.program_id(2)        # tile column index

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointer for this batch slice
    a_batch_ptr = a_ptr + pid_batch * stride_batch_a
    b_batch_ptr = b_ptr + pid_batch * stride_batch_b
    c_batch_ptr = c_ptr + pid_batch * stride_batch_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Pointers to A and B tiles
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_m_a) + (offs_k[None, :] * stride_k_a)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_k_b) + (offs_n[None, :] * stride_n_b)

        # Load with bounds checking
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_m_c) + (offs_n[None, :] * stride_n_c)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : Tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  float32
        - mask_tensor  : [B, N, N]  bool / 0‑1 float (may be all‑ones when nomask=True)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as ``input_tensor``.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B,N,N,dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy computation
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (same device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Gating projections (sigmoid) (FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Fuse optional mask and gates (in‑place) with Triton
    # ------------------------------------------------------------------
    if nomask:
        # No mask – just apply the gates element‑wise.
        left.mul_(left_gate)
        right.mul_(right_gate)
    else:
        # Convert mask to FP16 (0.0 / 1.0) and make tensors contiguous.
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporaries early
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via a batched GEMM kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    batch_size = B * H        # folds hidden dimension into the batch axis

    # Permute to bring hidden dim into the leading batch dimension,
    # then view as a 3‑D tensor: (batch, M=N, K=N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch_size, N, N)
    # Right must be transposed on its two N dimensions before flattening
    right_flat_T = right.permute(0, 3, 2, 1).contiguous().view(batch_size, N, N)

    # Allocate output tensor (FP16) that will receive the GEMM result.
    out_flat = torch.empty_like(left_flat)

    # Strides for the three tensors (batch, M, K) / (batch, K, N) / (batch, M, N)
    stride_batch_a, stride_m_a, stride_k_a = left_flat.stride()
    stride_batch_b, stride_k_b, stride_n_b = right_flat_T.stride()
    stride_batch_c, stride_m_c, stride_n_c = out_flat.stride()

    # Tunable tile sizes – good defaults for the tested N.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_size,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat_T,
        c_ptr=out_flat,
        batch=batch_size,
        M=N,
        N=N,
        K=N,
        stride_batch_a=stride_batch_a,
        stride_m_a=stride_m_a,
        stride_k_a=stride_k_a,
        stride_batch_b=stride_batch_b,
        stride_k_b=stride_k_b,
        stride_n_b=stride_n_b,
        stride_batch_c=stride_batch_c,
        stride_m_c=stride_m_c,
        stride_n_c=stride_n_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # a good choice on H100
        num_stages=3,
    )

    # Restore the original layout: (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # -> [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
