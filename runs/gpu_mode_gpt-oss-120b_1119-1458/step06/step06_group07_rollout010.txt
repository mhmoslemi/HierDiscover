import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Mask gating kernel for both left and right
@triton.jit
def mask_gate_fuse_two_kernel(
    left_ptr,   # fp16 * left
    right_ptr,  # fp16 * right
    left_gate_ptr,  # fp16 * left_gate
    right_gate_ptr, # fp16 * right_gate
    mask_ptr,     # fp16 * mask flat
    hidden_dim: tl.constexpr,
    size: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load left and right (fp16)
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)

    # Load gates
    lgate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Compute mask index
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default 1

    # Fuse multiplication
    left_out = left * lgate * m

"""
TriMul (outgoing) – fast Triton‑accelerated implementation
---------------------------------------------------------

Algorithm (all ops are performed without autograd):
1. Layer‑norm the input (float32) → cast to float16.
2. Fuse the five linear projections (left, right, left‑gate, right‑gate,
   out‑gate) into a single GEMM:   X_flat @ W_all.T
   (X_flat is [B·N·N, dim],   W_all is [5·hidden_dim, dim]).
3. Slice the concatenated result and apply sigmoid to the three gate
   tensors.
4. If a mask is present, fuse “tensor * gate * mask” for *both* left
   and right tensors in one Triton kernel (reads mask only once).
   Otherwise do a simple element‑wise multiplication.
5. Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realised with a single call to ``torch.einsum`` which internally
   uses a strided‑batched GEMM and avoids costly permutes/copies.
6. Second LayerNorm (fp16), out‑gate multiplication and final linear
   projection back to the original channel dimension.
7. Cast the result back to float32.

The implementation removes the five separate linear passes over the
large tensor (saving ~5× memory traffic on the input) and eliminates
the explicit permute/contiguous copies required by the naïve batched
GEMM.  One Triton kernel fuses two gated mask‑applications, further
reducing kernel launch overhead.  The heavy contraction remains
cuBLAS‑accelerated via ``einsum``, which is the fastest path on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask·gate·tensor for left & right simultaneously
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_two_kernel(
    left_ptr,            # *fp16  left tensor   (flattened)
    right_ptr,           # *fp16  right tensor  (flattened)
    left_gate_ptr,       # *fp16  left gate tensor  (flattened)
    right_gate_ptr,      # *fp16  right gate tensor (flattened)
    mask_ptr,            # *fp16  mask tensor (flattened, one per (b,i,k))
    hidden_dim: tl.constexpr,   # i32 hidden dimension H
    size: tl.constexpr,         # i32 total number of elements in left/right
    BLOCK: tl.constexpr,        # i32 compile‑time SIMD width (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load tensors (fp16)
    l = tl.load(left_ptr + offs, mask=valid, other=0.0)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # One mask value per hidden slice
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default = 1.0

    # Fuse multiplication
    l_out = l * lg * m
    r_out = r * rg * m

    # In‑place store
    tl.store(left_ptr + offs, l_out, mask=valid)
    tl.store(right_ptr + offs, r_out, mask=valid)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward (no autograd)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass.

    Arguments
    ---------
    data : Tuple containing
        - input_tensor  : torch.Tensor of shape [B, N, N, dim] (float16/float32)
        - mask_tensor   : torch.Tensor of shape [B, N, N] (bool/float)
        - weights        : dict of model weights (float32 tensors)
        - config         : dict with keys:
            * dim         : int  – input channel dimension
            * hidden_dim  : int  – hidden dimension for the multiplicative update
            * nomask      : bool – True if mask should be ignored
    Returns
    -------
    output : torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)
    x = x_norm.to(torch.float16)   # cast once for the heavy part

    B, N, _, _ = x.shape
    M = B * N * N                     # number of (b,i,k) positions

    # -----------------------------------------------------------------
    # 2️⃣ Concatenated linear projection for left, right and gates
    # -----------------------------------------------------------------
    # Convert weight tensors to fp16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate on the output‑feature dimension
    all_proj_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape [5*hidden_dim, dim]

    # Flatten the input to 2‑D and perform a single GEMM
    x_flat = x.reshape(M, dim)                     # [M, dim]
    proj_flat = torch.nn.functional.linear(x_flat, all_proj_w)   # [M,5*hidden_dim]

    # Reshape back to 4‑D and slice the five tensors
    proj = proj_flat.view(B, N, N, 5 * hidden_dim)  # contiguous
    left      = proj[..., 0:hidden_dim]                          # [B,N,N,H]
    right     = proj[..., hidden_dim : 2 * hidden_dim]           # [B,N,N,H]
    left_gate = torch.sigmoid(proj[..., 2 * hidden_dim : 3 * hidden_dim])
    right_gate = torch.sigmoid(proj[..., 3 * hidden_dim : 4 * hidden_dim])
    out_gate   = torch.sigmoid(proj[..., 4 * hidden_dim : 5 * hidden_dim])

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask + gate fusion (Triton)
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)          # shape [B*N*N]
        size = left.numel()                     # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # In‑place fusion for left and right simultaneously
        mask_gate_fuse_two_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ O(N³) contraction via einsum (batched GEMM internally)
    # -----------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum("bikd,bjkd->bijd", left, right)   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Out‑gate multiplication and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate                                   # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 (expected by harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
