def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # 1. First LN
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_norm.half()  # FP16

    # 2. Concatenate projection + gating weights
    # Convert to FP16
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    # shape (hidden_dim, dim); concatenate along output dim (rows)
    weight_concat = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).half()  # shape (5*hidden_dim, dim)

    # 3. Linear projection for all streams in a single GEMM
    # x shape [B,N,N,dim] -> out_concat shape [B,N,N,5*hidden_dim]
    y = torch.nn.functional.linear(x, weight_concat)  # FP16

    # 4. Slice the concatenated tensor
    hid = hidden_dim
    left_proj = y[..., 0:hid]
    left_gate_raw = y[..., hid:2*hid]
    right_proj = y[..., 2*hid:3*hid]
    right_gate_raw = y[..., 3*hid:4*hid]
    out_gate_raw = y[..., 4*hid:5*hid]

    # 5. Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # 6. Multiply with gates and optional mask using Triton kernel
    # Ensure contiguous for kernel
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()
    # Prepare mask: shape (B,N,N) broadcast to hidden dim
    if mask_tensor is None or nomask:
        mask_fp16 = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
                               dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    mask_flat = mask_fp16.view(-1)  # (B*N*N)

    # Launch kernel for left
    total_elements = left_proj.numel()  # B*N*N*hid
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)
    mask_gate_fuse_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )

    mask_gate_fuse_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # left_proj now holds left = left_proj * left_gate * mask
    # right_proj now holds right = right_proj * right_gate * mask
    left = left_proj
    right = right_proj
    del left_gate, right_gate, left_proj, right_proj, weight_concat

    # 7. Heavy contraction via batched GEMM (FP16)
    out = _batched_gemm_fp16(left, right)  # [B,N,N,hidden_dim], fp16

"""
TriMul (outgoing) – optimized forward pass.

The operator implements the following steps:

1️⃣  Layer‑norm the input tensor (FP32) and cast to FP16.
2️⃣  Fuse **five** linear maps (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single GEMM by
    concatenating the weight matrices.  This reduces the
    number of passes over the large input from 5 → 1.
3️⃣  Apply the sigmoid non‑linearity to the three gate
    results.
4️⃣  Fuse element‑wise projection × gate × mask with a tiny
    Triton kernel (`mask_gate_fuse_kernel`).  The mask is
    broadcast over the hidden dimension, so a single kernel
    launch per tensor suffices.
5️⃣  Heavy contraction
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    is performed as a batched GEMM (FP16) using cuBLAS.
6️⃣  Second layer‑norm (FP16) → element‑wise out‑gate
    multiplication → final linear projection back to the
    original channel dimension.
7️⃣  Cast the result back to FP32 (the harness expects FP32).

All heavy tensor operations stay on the GPU.  The Triton
kernel is used for the mask‑gate‑fusion step, which is
memory‑bound and benefits from the GPU’s tensor‑core
throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) – mask is broadcast over the
# hidden dimension.  Operates on a flat view of a contiguous tensor
# of shape [B·N·N·hid] (size = B·N·N·hid).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (proj)
    gate_ptr,       # *fp16 : gate tensor (sigmoid already applied)
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,) – broadcasted
    total_elements,  # i32 : total number of elements in x
    hidden_dim,      # i32 : hidden dimension size (hid)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask_off = offs < total_elements           # guard for tail

    # Load tensors
    x   = tl.load(x_ptr + offs,    mask=mask_off, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask_off, other=0.0)

    # Each `hidden_dim` consecutive elements share the same mask entry.
    mask_idx = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_idx, mask=mask_off, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask_off)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via
    a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and perform A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # cuBLAS will accumulate in FP32 internally for FP16 inputs
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        right_proj,
        right_gate,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # ``left_proj`` now holds the final ``left`` tensor,
    # ``right_proj`` holds the final ``right`` tensor.
    left  = left_proj
    right = right_proj

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (FP16)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating (FP16)
    to_out_w = weights["to_out.weight"].half()
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```
