```
@triton.jit
def gated_mask_kernel(
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr,
    mask_ptr,          # optional; shape M
    hidden_dim: tl.constexpr,
    total_elements: tl.constexpr, # total number of elements per side? Actually total_elements = M*hidden_dim.
    mask_enabled: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # compute mask index as before
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
    if mask_enabled:
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
    else:
        mask_val = 1.0

    left_out = left_proj * left_gate_sig * mask_val
    right_out = right_proj * right_gate_sig * mask_val

    tl.store(left_proj_ptr + offs, left_out, mask=active)
    tl.store(right_proj_ptr + offs, right_out, mask=active)
```

The heavy part of the TriMul forward pass (the four linear projections & three gates) is fused into **one** half‑precision matrix‑multiply followed by a **custom Triton kernel** that:

* applies the sigmoid gate,
* optionally multiplies by the pairwise mask,
* writes the gated results back in‑place for `left` and `right`.

All remaining operations (the O(N³) contraction, the two LayerNorms and the final linear) are kept in PyTorch because cuBLAS/Tensor‑Core kernels are already optimal for those workloads on an H100.

```python
"""
Optimised outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑Norm on the raw input (float32) → stabilises statistics.
2️⃣  Cast to fp16 for the compute‑intensive part.
3️⃣  Fuse the five linear projections (left_proj, left_gate,
    right_proj, right_gate, out_gate) into a single matmul:
       Y = X @ W_concatᵀ   where   W_concat = [
           left_proj; left_gate; right_proj; right_gate; out_gate
       ] ∈ ℝ⁽⁵·H⁾ˣC .
   The matmul is performed once with torch.nn.functional.linear (cuBLAS).
4️⃣  Split Y into the five tensors and reshape to (B,N,N,H) (half‑precision).
5️⃣  Triton kernel `gate_mask_fuse_kernel`:
        left  = left_proj  * sigmoid(left_gate ) * mask
        right = right_proj * sigmoid(right_gate) * mask
    The kernel works on the flattened view, broadcasting the mask
    across the hidden dimension.  If `nomask=True` the mask is a tensor
    of ones and the kernel skips the load.
6️⃣  O(N³) contraction is realised as a batched GEMM:
        out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
    This is a `torch.bmm` on a view of shape (B·H, N, N) – the fastest
    path on H100.
7️⃣  Second Layer‑Norm (fp16) over the hidden dimension.
8️⃣  Multiply by the pre‑computed `out_gate` and apply the final linear
    projection back to the original dimension.
9️⃣  Cast the result back to float32 (required by the harness) and return.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse sigmoid gating, optional mask and in‑place write.
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_proj_ptr,   # *fp16 : left projection tensor   [M·H]
    left_gate_ptr,   # *fp16 : left gate pre‑sigmoid   [M·H]
    right_proj_ptr,  # *fp16 : right projection tensor  [M·H]
    right_gate_ptr,  # *fp16 : right gate pre‑sigmoid  [M·H]
    mask_ptr,        # *fp16 : mask tensor (M) – broadcast over H
    hidden_dim: tl.constexpr,   # int32  – size H
    total_elems: tl.constexpr,  # int32  – M·H
    MASK_ENABLED: tl.constexpr, # int32  – 0 = no mask, 1 = apply mask
    BLOCK: tl.constexpr,        # compile‑time tile size (<=1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_elems

    # ------------------------------------------------------------------
    # Load operands
    # ------------------------------------------------------------------
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # sigmoid gate (1/(1+exp(-x))) – executed in FP16 arithmetic
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if MASK_ENABLED:
        # each hidden element belongs to position pos = idx // H
        pos = offs // hidden_dim
        m = tl.load(mask_ptr + pos, mask=mask, other=1.0)   # mask defaults to 1
    else:
        m = 1.0

    # ------------------------------------------------------------------
    # Write gated *masked results back in‑place
    # ------------------------------------------------------------------
    tl.store(left_proj_ptr + offs,  lp * lg_sig * m, mask=mask)
    tl.store(right_proj_ptr + offs, rp * rg_sig * m, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched matrix multiply for the O(N³) Tensor‑Core contraction
# ----------------------------------------------------------------------
def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a batched GEMM: each hidden channel becomes an independent batch.
    """
    B, N, _, H = left.shape
    # [B, H, N, N] → (B·H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N                     # number of (b,i,j) positions
    total_linear_output = 5 * hidden_dim

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections (left/right proj & gates, out_gate)
    # ------------------------------------------------------------------
    # Concatenate the weight matrices (shape: (5·H, dim))
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    # Convert once to fp16
    concat_w = torch.cat(
        [
            left_proj_w,      # [H, dim]
            left_gate_w,      # [H, dim]
            right_proj_w,     # [H, dim]
            right_gate_w,     # [H, dim]
            out_gate_w,       # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # (5·H, dim)

    # Flatten X to (M, dim) and apply a single linear
    x_flat = x.view(M, dim)                  # (M, dim)
    y = torch.nn.functional.linear(x_flat, concat_w)   # (M, 5·H)

    # ------------------------------------------------------------------
    # 4️⃣ Split and reshape the projected tensors (still half‑precision)
    # ------------------------------------------------------------------
    # Using torch.split gives views – no extra copies.
    split_tensors = torch.split(y, hidden_dim, dim=1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = split_tensors

    # Reshape to the original 4‑D layout
    left_proj_raw  = left_proj_raw.view(B, N, N, hidden_dim)
    left_gate_raw  = left_gate_raw.view(B, N, N, hidden_dim)
    right_proj_raw = right_proj_raw.view(B, N, N, hidden_dim)
    right_gate_raw = right_gate_raw.view(B, N, N, hidden_dim)
    out_gate_raw   = out_gate_raw.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid for out_gate (used later) – cheap element‑wise op
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Fuse gating + optional mask for left/right using Triton
    # ------------------------------------------------------------------
    # Flatten tensors to 1‑D for the kernel
    left_proj_flat  = left_proj_raw.view(-1)   # size M·H
    left_gate_flat  = left_gate_raw.view(-1)
    right_proj_flat = right_proj_raw.view(-1)
    right_gate_flat = right_gate_raw.view(-1)

    if nomask or mask_tensor is None:
        # Broadcast‑mask of ones (no runtime cost in the kernel)
        mask_fp16 = torch.ones((M,), dtype=torch.float16, device=device)
        mask_enabled = 0
    else:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)   # (M,)
        mask_enabled = 1

    # One kernel launch processes all hidden elements in a single sweep.
    BLOCK = 1024
    total_elems = left_proj_flat.numel()   # = M * hidden_dim
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    gate_mask_fuse_kernel[grid](
        left_proj_flat,
        left_gate_flat,
        right_proj_flat,
        right_gate_flat,
        mask_fp16,
        hidden_dim,
        total_elems,
        MASK_ENABLED=mask_enabled,
        BLOCK=BLOCK,
        num_warps=8,
    )
    # After the kernel, left_proj_flat and right_proj_flat already contain
    # the gated (and masked) values.  We can reuse them as the final tensors.
    left  = left_proj_raw   # shape [B,N,N,H]
    right = right_proj_raw  # shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul(left, right)      # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate                               # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the test harness) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
