@triton.jit
def tri_mul_kernel(out_ptr, a_ptr, b_ptr, 
                   B, N, H,
                   stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
                   stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
                   stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)  # combined batch and hidden channel index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden indices
    b_idx = pid_b // H
    h_idx = pid_b % H

    # Base offsets for this batch and hidden channel
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = out_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # Indices within M,N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute K loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # Load block of A (left)
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + (k_start + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=(offs_m < N) & (k_start + tl.arange(0, BLOCK_K) < N), other=0.0).to(tl.float16)

        # Load block of B (right) - note B is used as transposed, shape (K,N)
        b_ptrs = b_base + (k_start + tl.arange(0, BLOCK_K))[:, None] * stride_b_k + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=(k_start + tl.arange(0,BLOCK_K) < N) & (offs_n < N), other=0.0).to(tl.float16)

        # dot
        acc += tl.dot(a, b)  # shapes (blk_M,blk_K) dot (blk_K,blk_N) -> (blk_M,blk_N)

    # Store results
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None,:] * stride_c_j
    mask = (offs_m < N) & (offs_n < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

@triton.jit
def tri_mul_outgoing_kernel(
    out_ptr, left_ptr, right_ptr,
    batch, N, hidden_dim,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)  # combined batch * hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute batch and hidden channel indices
    b = pid_bh // hidden_dim
    h = pid_bh % hidden_dim

    # Base pointer offsets for this batch & hidden
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # Tile indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # prepare k offsets
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        # left stride dims: i -> stride_left_i, k -> stride_left_k
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: need shape (BLOCK_K, BLOCK_N) such that B[j,k] (as right[j,k]) => we need tile with rows k, cols j.
        # Compute pointer: (k offset) * stride_right_k + (j offset) * stride_right_i
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product accumulate (FP16 MatMul) => accumulate in FP32
        acc += tl.dot(a, b)

    # Store result back to output
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Highly optimized TriMul “outgoing” forward pass for AlphaFold3‑style modules.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣  Linear projections (left/right) and gating‐projections (left_gate,
    right_gate, out_gate) in fp16.
3️⃣  If a mask is present, fuse “tensor * gate * mask” in‑place with a tiny
    Triton kernel (`mask_gate_fuse_kernel`).  This removes extra loads/stores.
4️⃣  Core O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is implemented as a custom Triton batched GEMM (`tri_mul_outgoing_kernel`).
    The hidden dimension `h` is folded into the batch axis, so each (b,h) pair
    runs a standard mat‑mul (N×N) using Tensor‑Core‑compatible fp16 loads and
    fp32 accumulation.
5️⃣  Second Layer‑norm (still fp16) → element‑wise multiplication with the
    out‑gate.
6️⃣  Final linear projection back to the original channel dimension.
7️⃣  Cast the result back to float32 (the harness expects float32).

All heavyweight arithmetic runs in fp16, harness‑compatible float32 tensors are
only used where necessary (first LayerNorm).  The approach avoids the costly
permute‑and‑contiguous copies required by the naïve batched‑GEMM implementation
and fuses the mask‑gate step, yielding a noticeable speed‑up on H100 GPUs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate, and tensor (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → hidden dimension size (D_h)
    size,           # i32   → total number of elements in x (B*N*N*D_h)
    BLOCK: tl.constexpr,   # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # load original tensor and gate
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask=1

    # fused multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# The hidden dimension is folded into the batch axis (B*H total batches).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    out_ptr,               # *fp16  → output tensor [B,N,N,H] (flattened)
    left_ptr,              # *fp16  → left tensor  [B,N,N,H]
    right_ptr,             # *fp16  → right tensor [B,N,N,H]
    batch,                 # i32    → batch size B
    N,                     # i32    → sequence length
    hidden_dim,            # i32    → hidden dimension H
    # strides for out
    stride_out_batch,      # i64
    stride_out_i,          # i64
    stride_out_j,          # i64
    stride_out_h,          # i64
    # strides for left
    stride_left_batch,     # i64
    stride_left_i,         # i64
    stride_left_k,         # i64
    stride_left_h,         # i64
    # strides for right
    stride_right_batch,    # i64
    stride_right_i,        # i64   (this is the “j” dimension)
    stride_right_k,        # i64
    stride_right_h,        # i64
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, # rows per tile (i‑dimension)
    BLOCK_N: tl.constexpr, # cols per tile (j‑dimension)
    BLOCK_K: tl.constexpr, # reduction tile (k‑dimension)
):
    pid_batch_h = tl.program_id(0)   # combined batch * hidden index
    pid_i = tl.program_id(1)         # tile index on the i‑axis
    pid_j = tl.program_id(2)         # tile index on the j‑axis

    # decompose combined index
    b = pid_batch_h // hidden_dim
    h = pid_batch_h % hidden_dim

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # thread‑local offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile: shape (BLOCK_M, BLOCK_K) --------------------
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)               # fp16 load
        # ----- right tile (already transposed): shape (BLOCK_K, BLOCK_N) -----
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)               # fp16 load

        # ----- inner dot product (Tensor Core friendly) -----------------
        # tl.dot performs fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Forward entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor  [B, N, N, dim] (float32)
        torch.Tensor,          # mask tensor   [B, N, N]   (float32) or None
        Dict[str, torch.Tensor],  # model weights (float32)
        Dict[str, Any]            # config dict containing “dim”, “hidden_dim”, “nomask”
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Returns a tensor of shape [B, N, N, dim] (float32), placed on the same device
    as the input.  All heavy arithmetic runs in fp16 on Tensor‑Core‑enabled GPUs.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability) and cast to fp16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] (float32)

    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to fp16 once (avoid repeated casts)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Gating signals (fp16) — sigmoid   (still separate tensors)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Fuse optional mask with gates (Triton kernel) OR simple multiply
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Flatten mask and ensure contiguous fp16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                     # [B*N*N]

        total_elems = left.numel()                         # B*N*N*hidden_dim
        BLOCK = 1024                                        # 1‑K tiles (tunable)
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³) contraction via custom Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    out = torch.empty_like(left)   # allocate output tensor (fp16)

    # Compute strides (each stride is an int)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()

    # Choose tile sizes that work well on H100 (empirically good)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions:
    #   axis 0 → batch*hidden (folded)
    #   axis 1 → i‑tiles
    #   axis 2 → j‑tiles
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_bh = B * hidden_dim
    grid = (grid_bh, grid_m, grid_n)

    tri_mul_outgoing_kernel[grid](
        out,
        left,
        right,
        B,
        N,
        hidden_dim,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (element‑wise) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to original dim

    # ------------------------------------------------------------------
    # 9️⃣ Cast to float32 for the harness output
    # ------------------------------------------------------------------
    return out.to(torch.float32)
