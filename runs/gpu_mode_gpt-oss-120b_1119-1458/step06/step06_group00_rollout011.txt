"""TriMul outgoing kernel with fused linear projections (combined weight matrix) and mask-gate fusion.
...
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fusion_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
                             total_elements: tl.int64, H: tl.int64,
                             BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Index of mask element for this location (mask is broadcast across H)
    mask_offs = offs // H

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """Outgoing TriMul forward pass (no gradients).

    Combines all linear projections (left/right projections and gate
    projections) into a single large GEMM, applies sigmoid gates,
    fuses the optional mask with the gates using a Triton kernel,
    performs the O(N³) contraction via batched GEMM, and finally
    applies the second LayerNorm, out‑gate and final linear projection.

    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
        - input_tensor : Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : Tensor of shape [B, N, N] (bool or 0/1 float) – optional
        - weights      : Mapping of weight name → Tensor (float32)
        - config       : Dict with keys ``dim`` and ``hidden_dim`` (int);
                         Optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – keep numerics stable
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)  # shape [B, N, N, dim], FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Build a combined weight matrix for the 5 linear layers
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    #    Shape: (5 * hidden_dim, dim)
    # ------------------------------------------------------------------
    # Note: nn.Linear stores weight as (out_features, in_features)
    combined_weight = torch.cat([
        weights["left_proj.weight"],   # (hidden_dim, dim)
        weights["left_gate.weight"],   # (hidden_dim, dim)
        weights["right_proj.weight"],  # (hidden_dim, dim)
        weights["right_gate.weight"],  # (hidden_dim, dim)
        weights["out_gate.weight"],    # (hidden_dim, dim)
    ], dim=0).to(device=device, dtype=torch.float16)  # (5*hidden_dim, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Single GEMM x @ combined_weight → (B, N, N, 5*hidden_dim)
    # ------------------------------------------------------------------
    batch_size, seq_len, _, _ = x.shape
    flat_x = x.reshape(batch_size * seq_len * seq_len, dim)  # (M, dim)
    combined_out = F.linear(flat_x, combined_weight)  # (M, 5*hidden_dim), FP16
    combined_out = combined_out.view(batch_size, seq_len, seq_len, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Slice the combined output and apply sigmoids to gate tensors
    # ------------------------------------------------------------------
    # ordering: [left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre]
    hidden = hidden_dim
    left_proj = combined_out[..., 0:hidden]                         # (B,N,N,hidden)
    left_gate_pre = combined_out[..., hidden:2*hidden]              # (B,N,N,hidden)
    right_proj = combined_out[..., 2*hidden:3*hidden]                # (B,N,N,hidden)
    right_gate_pre = combined_out[..., 3*hidden:4*hidden]           # (B,N,N,hidden)
    out_gate_pre = combined_out[..., 4*hidden:]                      # (B,N,N,hidden)

    # Sigmoid activation for gates (creates new tensors)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse optional mask & gates – in‑place gating of left/right.
    #     If there is no mask we just multiply.
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate          # (B,N,N,hidden)
        right = right_proj * right_gate
    else:
        # Convert mask to fp16 and ensure continuity
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        # Ensure left/right and gate tensors are contiguous
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        # Triton kernel: left_out = left * left_gate * mask, etc.
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # After kernel, left/right hold gated‑and‑masked tensors.
        # Gates no longer needed.
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction: batched GEMM A @ Bᵀ  (B*hidden_dim independent mats)
    # ------------------------------------------------------------------
    # Re‑order to (B, hidden, N, N) for easy flattening.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_t.shape   # B batch, H == hidden_dim, N sequence length
    A = left_t.view(B * H, N, N)     # (B*H, N, N)
    B_mat = right_t.view(B * H, N, N)

    # Batched matrix multiply: each (i,j) pair is left[i] @ right[j]^T
    C = torch.bmm(A, B_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout: (B, N, N, hidden)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 9️⃣ Apply the final out‑gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
TriMul outgoing kernel (forward only) – highly optimized version.

Algorithm
---------
1️⃣  Layer‑Norm on the input (FP32, stable statistics).
2️⃣  Cast to FP16 for the heavy computation.
3️⃣  Fuse the five linear projections (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single GEMM by concatenating their
    weight matrices into a `(5·hidden_dim, dim)` matrix.  This reduces the
    number of passes over the large `[B,N,N,dim]` tensor from five to one.
4️⃣  Slice the GEMM result into the five tensors and apply the sigmoid
    activation to the three gate tensors.
5️⃣  If a mask is present, fuse mask application and the two gate
    multiplications into an in‑place Triton kernel (`_mask_gate_fusion_kernel`);
    otherwise, multiply by the gates directly.
6️⃣  Perform the O(N³) contraction: for each batch and hidden channel
    compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.  This is
    implemented as a batched GEMM (`torch.bmm`) on tensors reshaped to
    `(B·hidden_dim, N, N)`.
7️⃣  Apply the second LayerNorm over the hidden dimension (FP16).
8️⃣  Multiply by the out‑gate, then a final linear projection back to `dim`.
9️⃣  Cast the result back to float32 for the caller.

The only Triton‐written part is the mask‑gate fusion kernel, which
eliminates an extra memory pass when masks are used.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]   (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]   (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]     (broadcasted)
    total_elements,    # int   (B·N·N·H)
    H,                 # int   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    """
    Apply mask (if any) and the sigmoid gates to the left/right projections
    in a single pass.  The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # mask index (mask is 1‑D flattened, each entry corresponds to H hidden values)
    mask_offs = offs // H

    left       = tl.load(left_ptr       + offs, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + offs, mask=valid, other=0.0)
    left_gate  = tl.load(left_gate_ptr  + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val   = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


def custom_kernel(
    data: tuple[torch.Tensor, torch.Tensor, dict[str, torch.Tensor], dict[str, any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask_tensor  : torch.Tensor [B, N, N]           (bool / 0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Build a combined weight matrix for the 5 linear layers
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # ------------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden_dim, dim)
            weights["left_gate.weight"],   # (hidden_dim, dim)
            weights["right_proj.weight"],  # (hidden_dim, dim)
            weights["right_gate.weight"],  # (hidden_dim, dim)
            weights["out_gate.weight"],    # (hidden_dim, dim)
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)  # (5*hidden_dim, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Single GEMM: x @ combined_weight → (B, N, N, 5*hidden_dim)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    M = B * N * N
    flat_x = x.view(M, dim)                     # (M, dim)
    combined_out = F.linear(flat_x, combined_weight)  # (M, 5*hidden_dim)
    combined_out = combined_out.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Slice the fused output and apply sigmoids to the gate tensors
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj      = combined_out[..., 0:H]                 # (B,N,N,H)
    left_gate_pre  = combined_out[..., H:2*H]               # (B,N,N,H)
    right_proj     = combined_out[..., 2*H:3*H]              # (B,N,N,H)
    right_gate_pre = combined_out[..., 3*H:4*H]              # (B,N,N,H)
    out_gate_pre   = combined_out[..., 4*H:]                # (B,N,N,H)

    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_pre)  # (B,N,N,H)
    out_gate   = torch.sigmoid(out_gate_pre)    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse optional mask with gates (in‑place)
    # ------------------------------------------------------------------
    if nomask:
        left  = left_proj * left_gate      # (B,N,N,H)
        right = right_proj * right_gate
    else:
        # Prepare tensors for the Triton kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        left  = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates are no longer needed after the kernel
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction: batched GEMM A @ Bᵀ  (B·H independent mats)
    # ------------------------------------------------------------------
    # Bring hidden dim to the second axis for easy flattening
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_t.shape
    A = left_t.view(B * H, N, N)          # (B*H, N, N)
    B_mat = right_t.view(B * H, N, N)     # (B*H, N, N)

    C = torch.bmm(A, B_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final linear projection to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
