@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0) # row tile idx w.r.t N dimension of left
    pid_j = tl.program_id(1) # col tile idx w.r.t N dimension of right
    pid_hb = tl.program_id(2) # combined head and batch index

    # compute batch and head indices
    # total heads = H, total batch = B
    b = pid_hb // H
    h = pid_hb % H

    # compute tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # bound checks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers for left tile (M,K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h

        # compute pointers for right tile (K,N) (transposed)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_j + \
            h * stride_right_h

        # load tiles
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)  # shape (BLOCK_K, BLOCK_N)

        # perform dot product (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile)

        # increment pointers not needed as we recomputed each iteration

    # Write back the result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h

    # store with casting to half
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – optimized forward pass.

Algorithm
---------
1. **LayerNorm (FP32)** – normalize the 4‑D input and cast to FP16.
2. **Fused linear projections** – the five independent linear layers
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`) are
   concatenated into a single weight matrix and executed once with a
   single half‑precision GEMM (`torch.nn.functional.linear`).  The result is
   split back into the five tensors and the three gating streams are passed
   through a sigmoid.
3. **Mask + gate fusion** – when a mask is present a tiny Triton kernel
   fuses `proj * gate * mask` (mask is broadcast over the hidden
   dimension).  No‑mask case uses an in‑place element‑wise multiplication.
4. **Heavy O(N³·C) contraction** – compute
   `out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]`.  This is a batch of
   GEMMs: for each head `c` we perform `C = A @ Bᵀ`.  The contraction is
   implemented as a custom Triton kernel that tiles the i, j, k dimensions
   and uses Tensor‑Core `tl.dot` (FP16×FP16→FP32 accumulation).  The kernel
   works on the flattened batch‑head dimension `b*h`.
5. **Second LayerNorm + out‑gate** – apply a half‑precision LayerNorm over
   the hidden dimension, multiply by the `out_gate` (sigmoid), then a final
   half‑precision linear projection back to the original channel dimension.
6. **Cast back to FP32** – the harness expects a float32 tensor.

Only the forward pass is implemented; gradients are omitted.
"""

import math
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1: fuse (tensor * gate * mask) – mask is broadcast over the hidden dim.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16: tensor to be updated, total_elements long
    gate_ptr,       # *fp16: gate tensor, same shape as x
    mask_ptr,       # *fp16: mask tensor of shape (B * N * N,)
    total_elements,  # i32: total number of elements in x
    hidden_dim,      # i32: width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load the values
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Compute mask index (same mask for every `hidden_dim` consecutive elements)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for TriMul contraction.
#   out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,    # *fp16: lhs tensor [B,N,N,C]
    right_ptr,   # *fp16: rhs tensor [B,N,N,C]
    out_ptr,     # *fp16: output tensor [B,N,N,C]

    B, N, C,    # i32: batch, sequence length, hidden dim

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_c,
    stride_r_b, stride_r_i, stride_r_k, stride_r_c,
    stride_o_b, stride_o_i, stride_o_j, stride_o_c,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # Program IDs:
    #   pid_i  -> tile index along the row   (i dimension)
    #   pid_j  -> tile index along the column(j) dimension
    #   pid_hb -> combined batch & head index (b*C + c)
    # ---------------------------------------------------------------
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    # decompose combined head index
    b = pid_hb // C
    c = pid_hb % C

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # LHS tile: shape (BLOCK_M, BLOCK_K) -> left[b,i,k,c]
        lhs_ptrs = left_ptr + \
            b * stride_l_b + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k + \
            c * stride_l_c

        lhs = tl.load(lhs_ptrs,
                      mask=mask_i[:, None] & mask_k[None, :],
                      other=0.0)

        # RHS tile (transposed): we need shape (BLOCK_K, BLOCK_N)
        # right[b,j,k,c]   → we load as (k, j) to match (K,N)
        rhs_ptrs = right_ptr + \
            b * stride_r_b + \
            offs_k[:, None] * stride_r_k + \
            offs_j[None, :] * stride_r_i + \
            c * stride_r_c

        rhs = tl.load(rhs_ptrs,
                      mask=mask_k[:, None] & mask_j[None, :],
                      other=0.0)

        # dot: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(lhs, rhs)   # FP16 inputs → FP32 accumulation

    # Store the result (cast back to FP16)
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j + \
        c * stride_o_c

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _trimul_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the TriMul contraction using the Triton kernel defined above.
    left/right: [B, N, N, C] (FP16)
    Returns:
        out: [B, N, N, C] (FP16)
    """
    B, N, _, C = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    s_l_b, s_l_i, s_l_k, s_l_c = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_c = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_c = out.stride()

    # tuning parameters – empirically good for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * C,
    )

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        C,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_c,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_c,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps to keep the tensor cores busy
        #num_stages=3,        # optional pipelining; Triton defaults are fine
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim]      (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]           (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → ignore mask)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # original channel dim (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32
    x = x.half()  # work in FP16 from now on

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single GEMM
    # ------------------------------------------------------------------
    # Stack the five weight matrices (shape: (5*hidden_dim, dim))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # FP16

    # Single linear projection – result shape: [B, N, N, 5*hidden_dim]
    proj = torch.nn.functional.linear(x, combined_weight)  # no bias

    # ------------------------------------------------------------------
    # 3️⃣ Slice the fused projection & apply sigmoids to the gate streams
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj = proj[..., :h]
    left_gate = torch.sigmoid(proj[..., h : 2 * h])
    right_proj = proj[..., 2 * h : 3 * h]
    right_gate = torch.sigmoid(proj[..., 3 * h : 4 * h])
    out_gate = torch.sigmoid(proj[..., 4 * h :])   # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ (Optional) mask‑gate fusion – in‑place multiplication
    # ------------------------------------------------------------------
    if nomask:
        # simple element‑wise product when no mask is needed
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # ensure tensors are contiguous for the Triton kernel
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        total_elements = left.numel()
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Fuse proj * gate * mask for the left stream (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse proj * gate * mask for the right stream (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction (TriMul) – custom Triton GEMM
    # ------------------------------------------------------------------
    # left / right are now the masked‑gated projections with shape [B, N, N, hidden_dim]
    out = _trimul_gemm_triton(left, right)   # FP16, shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final linear back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].half(),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Cast result back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
