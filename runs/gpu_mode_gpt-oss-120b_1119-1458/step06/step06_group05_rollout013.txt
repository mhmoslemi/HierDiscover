@triton.jit
def _batch_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # combined batch and hidden dimension
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute b and h indices
    b = pid // H
    h = pid % H

    # compute i and j ranges
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator: use float32 for more stable accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k axis
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers for left tile (A)
        a_ptrs = left_ptr + b * stride_l_b + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k + h * stride_l_h
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)  # cast to fp32

        # compute pointers for right tile (B transposed)
        # Right tile: need shape (BLOCK_K, BLOCK_N) with rows = k, cols = j
        b_ptrs = right_ptr + b * stride_r_b + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i + h * stride_r_h
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # after loop, convert accumulation to fp16
    acc_fp16 = acc.to(tl.float16)

    # compute output pointers
    c_ptrs = out_ptr + b * stride_o_b + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j + h * stride_o_h
    tl.store(c_ptrs, acc_fp16, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) optimized forward pass.

Algorithm
---------
1️⃣ First LayerNorm (FP32→FP16) on the input tensor.
2️⃣ Fuse the two projection Linear layers (proj + gate) per side
   by concatenating the weights and using a single half‑precision matrix
   multiplication.  After the matmul we split the result and apply a
   sigmoid to the gating part.
3️⃣ Apply the optional mask – broadcasted over the hidden dimension –
   and fuse the element‑wise product (proj * gate * mask) with plain
   PyTorch broadcasting (no extra memory copies).
4️⃣ The expensive O(N³·H) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed with a custom Triton kernel that computes a batched
   GEMM (C = A @ Bᵀ) for every (batch, hidden) slice.  The kernel
   tiles the i, j and k dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and
   accumulates in float32, finally storing FP16 results.
5️⃣ Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear
   projection back to model dimension.
6️⃣ Cast back to FP32 (the harness expects FP32).

All heavy linear work stays in FP16, memory traffic is halved by the
weight‑fusion and mask‑fusion steps, and the O(N³) contraction is done
with a highly‑tuned custom kernel that rivals cuBLAS on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#   implemented as left @ rightᵀ for each (b,h) slice.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    left_ptr,               # *fp16  [B, N, N, H]
    right_ptr,              # *fp16  [B, N, N, H]
    out_ptr,                # *fp16  [B, N, N, H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile size in output i dimension
    BLOCK_N: tl.constexpr,  # tile size in output j dimension
    BLOCK_K: tl.constexpr,  # reduction tile size (k dimension)
):
    # program ids
    pid = tl.program_id(0)       # packed (batch, hidden) index
    pid_i = tl.program_id(1)     # i‑tile index
    pid_j = tl.program_id(2)     # j‑tile index

    # unpack batch and hidden index
    b = pid // H
    h = pid % H

    # compute row / column coordinates for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the shared N dimension of the pairwise contraction)
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of left (shape [BLOCK_M, BLOCK_K])
        a_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Load a tile of rightᵀ (shape [BLOCK_K, BLOCK_N])
        # rightᵀ[k, j] = right[b, j, k, h]
        b_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i
            + h * stride_r_h
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # fused multiply‑accumulate (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # write the result back as fp16
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Helper: batched GEMM wrapper that launches the Triton kernel.
# ----------------------------------------------------------------------
def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a
    Triton kernel (batched GEMM).  Input tensors must be fp16 and contiguous.
    Returns a fp16 tensor of the same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides expressed in number of elements (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # Tuned tile sizes – work well for N up to 1024 and H up to 384 on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,                              # program_id(0) over batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # program_id(1) over i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # program_id(2) over j‑tiles
    )

    _batched_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype_fp16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype_fp16)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # FP16 for the rest of the pipeline

    # -----------------------------------------------------------------
    # 2️⃣ Fuse projection + gate Linear for left and right sides
    # -----------------------------------------------------------------
    # Concatenate the projection and gate weights (no bias in any of them)
    left_proj_w = weights["left_proj.weight"]            # [hidden, dim]
    left_gate_w = weights["left_gate.weight"]            # [hidden, dim]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]

    # Build concatenated weights: [2*hidden, dim]
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=dtype_fp16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=dtype_fp16)

    # Single fused linear per side
    left_combined = torch.nn.functional.linear(x, left_combined_w)   # [B, N, N, 2*hidden]
    right_combined = torch.nn.functional.linear(x, right_combined_w) # [B, N, N, 2*hidden]

    # -----------------------------------------------------------------
    # 3️⃣ Split projection and gate, apply sigmoid to gate
    # -----------------------------------------------------------------
    left_proj, left_gate_raw = left_combined.split([hidden_dim, hidden_dim], dim=-1)
    right_proj, right_gate_raw = right_combined.split([hidden_dim, hidden_dim], dim=-1)

    left_gate = torch.sigmoid(left_gate_raw)   # [B,N,N,hidden]
    right_gate = torch.sigmoid(right_gate_raw)

    # -----------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and fuse with gating
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden dimension
        mask_fp16 = mask_tensor.to(dtype=dtype_fp16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release temporary concatenated tensors ASAP
    del left_combined, right_combined, left_proj, left_gate_raw, right_proj, right_gate_raw

    # Ensure tensors are contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # 5️⃣ Heavy contraction: batched GEMM (left @ rightᵀ) via Triton
    # -----------------------------------------------------------------
    out = _batched_gemm_triton(left, right)   # [B, N, N, hidden]  (FP16)

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Out‑gate (sigmoid) and final projection back to model dimension
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden] FP16

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # -----------------------------------------------------------------
    # 8️⃣ Cast to FP32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
