@triton.jit
def _gate_reorder_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,  # *fp16 [B,N,N]; if mask_present==0, pointer can be any
    left_out_ptr, right_out_ptr,   # *fp16 [B*H,N,N] (contiguous)
    B, N, H, total_elements,
    mask_present: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # decode indices
    off = offs
    # Compute hidden index
    h = off % H
    off = off // H
    j = off % N
    off = off // N
    i = off % N
    b = off // N

    # read values (if not valid, handle others)
    left_val = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_val = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # mask gating
    if mask_present:
        mask_off = b * N * N + i * N + j
        mask_val = tl.load(mask_ptr + mask_off, mask=valid, other=1.0)
    else:
        mask_val = 1.0

    # apply sigmoid gates
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate_val))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate_val))

    # combine
    val_left = left_val * left_gate_sig * mask_val
    val_right = right_val * right_gate_sig * mask_val

    # compute flat output indices
    batch_idx = b * H + h   # batch index for output (B*H)
    out_off_left = (batch_idx * N + i) * N + j
    out_off_right_transposed = (batch_idx * N + j) * N + i

    tl.store(left_out_ptr + out_off_left, val_left, mask=valid)
    tl.store(right_out_ptr + out_off_right_transposed, val_right, mask=valid)

"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

Algorithm
---------
1. Layer‑norm the input (float32 → stable statistics).
2. Cast to fp16 for the bulk of the computation.
3. Fuse the five linear projections (left‑proj, right‑proj,
   left‑gate, right‑gate, out‑gate) into a *single* matmul to
   avoid repeatedly loading the huge (B×N×N) tensor.
4. A Triton kernel simultaneously
   • applies the sigmoid gates,
   • applies the optional mask (broadcast over the hidden dim),
   • reorders the gated tensors into the layout required by batched GEMM:
        left_out   : (B·H, N, N)   – row‑major matrix L[i,k]
        right_out_T: (B·H, N, N)   – row‑major transposed matrix Rᵀ[k,j]
   This eliminates the costly permute + .contiguous() copies.
5. Perform the O(N³) contraction with a single batched
   `torch.bmm(left_out, right_out_T)` (Tensor‑core fp16 GEMM).
6. Reshape back to (B, N, N, H) and apply a second LayerNorm.
7. Multiply by the *out‑gate* (sigmoid of its linear projection).
8. Final linear projection back to `dim`.
9. Cast the result to float32 for the caller.

The kernels are written for H100 (Triton 3.3.1) and make heavy use of
fp16 arithmetic to hit the target runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse sigmoid gates, optional mask and reorder the
#                tensors to a (B·H,N,N) layout that can be used directly
#                by torch.bmm.  The right tensor is written transposed.
# ----------------------------------------------------------------------
@triton.jit
def _gate_reorder_kernel(
    left_ptr,                # *fp16  [B,N,N,H]  (raw left projection)
    right_ptr,               # *fp16  [B,N,N,H]  (raw right projection)
    left_gate_ptr,           # *fp16  [B,N,N,H]  (pre‑sigmoid left gate)
    right_gate_ptr,          # *fp16  [B,N,N,H]  (pre‑sigmoid right gate)
    mask_ptr,                # *fp16  [B,N,N]    (broadcasted mask, ignored if MASK=0)
    left_out_ptr,            # *fp16  [(B·H),N,N] (contiguous, not transposed)
    right_out_ptr,           # *fp16  [(B·H),N,N] (contiguous, *already* transposed)
    B, N, H,                 # int: dimensions
    total_elements,          # int: B·N·N·H
    MASK: tl.constexpr,      # 0 → no mask, 1 → mask present
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)      # flat index
    mask = offs < total_elements

    # ---- decode flat index into (b, i, k, h) ----
    # flat layout of the input tensors: ((b*N + i)*N + k)*H + h
    idx = offs
    h = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ---- load raw values -------------------------------------------------
    left_val      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right_val     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate_raw= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # ---- sigmoid gates (half‑precision) ----------------------------------
    left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # ---- optional mask ----------------------------------------------------
    if MASK:
        # mask stored as (B,N,N) → flatten offset = b·N·N + i·N + k
        mask_off = b * N * N + i * N + k
        mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)
    else:
        mask_val = 1.0

    # ---- apply gates and mask --------------------------------------------
    left_processed  = left_val  * left_gate  * mask_val
    right_processed = right_val * right_gate * mask_val

    # ---- compute batch index for the (B·H) dimension -----------------------
    batch_idx = b * H + h                       # 0 ≤ batch_idx < B·H

    # ---- write to contiguous output buffers (row‑major) -------------------
    # left_out  : rows = i , cols = k   → offset = ((batch_idx·N) + i)·N + k
    # right_out_T : rows = k , cols = i → offset = ((batch_idx·N) + k)·N + i
    out_off_left  = (batch_idx * N + i) * N + k
    out_off_right = (batch_idx * N + k) * N + i

    tl.store(left_out_ptr  + out_off_left,  left_processed,  mask=mask)
    tl.store(right_out_ptr + out_off_right, right_processed, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool / 0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 → stable statistics)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )                              # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into one matmul
    # ------------------------------------------------------------------
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Stack them along the output‑feature dimension
    combined_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(device=device, dtype=fp16)        # (5·H, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Single linear pass → (B,N,N,5·H)
    # ------------------------------------------------------------------
    combined = F.linear(x, combined_weight)   # fp16

    B, N, _, _ = combined.shape
    H = hidden_dim

    # ------------------------------------------------------------------
    # 5️⃣ Slice the combined tensor
    # ------------------------------------------------------------------
    left_proj_raw   = combined[..., 0:H]                 # (B,N,N,H)
    right_proj_raw  = combined[..., H:2 * H]             # (B,N,N,H)
    left_gate_raw   = combined[..., 2 * H:3 * H]         # (B,N,N,H)
    right_gate_raw  = combined[..., 3 * H:4 * H]         # (B,N,N,H)
    out_gate_raw    = combined[..., 4 * H:5 * H]         # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate (sigmoid) – required after the second LayerNorm
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)               # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Allocate buffers for the gated tensors in the layout needed
    #     by batched GEMM (no extra permute / copy later)
    # ------------------------------------------------------------------
    left_out      = torch.empty((B * H, N, N), dtype=fp16, device=device)
    right_out_T   = torch.empty((B * H, N, N), dtype=fp16, device=device)  # already transposed

    total_elements = B * N * N * H

    # ------------------------------------------------------------------
    # 8️⃣ Optional mask handling (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if nomask:
        # dummy mask – the kernel will ignore it when MASK=0
        dummy_mask = torch.empty(0, dtype=fp16, device=device)
        mask_tensor_fp16 = dummy_mask
        mask_flag = 0
    else:
        mask_tensor_fp16 = mask_tensor.to(fp16).contiguous()
        mask_flag = 1

    # ------------------------------------------------------------------
    # 9️⃣ Triton kernel: apply sigmoid gates, optional mask and reorder
    # ------------------------------------------------------------------
    BLOCK = 1024                         # max threads per block on H100
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    _gate_reorder_kernel[grid](
        left_ptr=left_proj_raw,
        right_ptr=right_proj_raw,
        left_gate_ptr=left_gate_raw,
        right_gate_ptr=right_gate_raw,
        mask_ptr=mask_tensor_fp16,
        left_out_ptr=left_out,
        right_out_ptr=right_out_T,
        B=B,
        N=N,
        H=H,
        total_elements=total_elements,
        MASK=mask_flag,                 # compile‑time constant (0 or 1)
        BLOCK_SIZE=BLOCK,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 🔟 Batched matrix multiplication (B·H, N, N) @ (B·H, N, N).T
    # ------------------------------------------------------------------
    # right_out_T already contains the transposed right matrix,
    # therefore a plain bmm yields the desired contraction.
    contracted = torch.bmm(left_out, right_out_T)     # (B·H, N, N) fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Restore (B, N, N, H) layout
    # ------------------------------------------------------------------
    out = contracted.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 1️⃣3️⃣ Apply out‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating (fp16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)             # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 1️⃣4️⃣ Cast to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
