def trilu_outgoing_contract_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B,
    N,
    H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)  # combined batch * hidden
    pid_m = tl.program_id(1)  # row tile
    pid_n = tl.program_id(2)  # col tile

    # decode batch and hidden dimensions
    batch_idx = pid_batch // H
    hidden_idx = pid_batch % H

    # compute row and column offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # shape (BLOCK_M,)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # shape (BLOCK_N,)
    offs_k = tl.arange(0, BLOCK_K)  # 0:blockK

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # TOT: offsets for accumulation
    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k indices
        cur_k = k_iter * BLOCK_K + offs_k

        # compute pointer offsets for left: left[batch_idx, i, cur_k, hidden_idx]
        # left pointer stride array:
        # pos = batch_idx * stride_left_b + offs_i[:,None] * stride_left_i + cur_k[None,:] * stride_left_k + hidden_idx * stride_left_h
        # left_ptrs shape (BLOCK_M,BLOCK_K)
        left_ptrs = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # compute pointer offsets for right_T: we need right[batch_idx, cur_j?,?]. Actually we need B_T such that its shape is (BLOCK_K,BLOCK_N), where rows correspond to k and columns to j.
        # So we load right as:
        # pos = batch_idx * stride_right_b + hidden_idx * stride_right_h + cur_k[:,None] * stride_right_k + offs_j[None,:] * stride_right_j
        right_ptrs_T = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b_T = tl.load(right_ptrs_T, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

def trilu_outgoing_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)  # combined batch*hidden index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and hidden idx
    batch_idx = pid_batch // H
    hidden_idx = pid_batch % H

    # Compute block offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Setup base pointers
    base_left = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h
    base_right = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K + offs_k
        mask_cur_k = cur_k < N

        # left tile with shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_left + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k

        b_ptrs_T = base_right + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

"""
TriMul outgoing (AlphaFold3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Layer‑Norm (float32) → cast to fp16.
2️⃣  Fuse the two linear projections (proj + gate) for the *left* and *right*
    branches by concatenating their weight matrices (shape 2·H×C).  The
    resulting tensor is split into projection and gate tensors.  The gate
    tensor is passed through sigmoid.
3️⃣  Fuse element‑wise multiplication of projection, gate and optional mask
    with a tiny Triton kernel (`mask_gate_fuse_kernel`).  This writes the
    gated tensors back in‑place, eliminating temporary copies.
4️⃣  Perform the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    with a custom Triton batched‑GEMM that works directly on the
    original [B,N,N,H] layout (no permute/contiguous copies).  The kernel
    accumulates in FP32 and writes the result in FP16.
5️⃣  Second Layer‑Norm (fp16) → out‑gate gating (sigmoid already computed)
    → final linear projection back to the input channel dimension.
6️⃣  Cast the result to float32 (the harness expects float32).

All heavy arithmetic runs in fp16 to exploit H100 TensorCores, while the
mask‑gate kernel supplies the required Triton component.
"""

import math
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection‑gate tensor with optional mask.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask tensor, shape [B*N*N] (flattened)
    hidden_dim,     # i32   → hidden dimension
    size,           # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands
    x   = tl.load(x_ptr   + offs, mask=valid, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=valid, other=0.0)

    # Index of the mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)    # default mask = 1

    # Fuse multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core N³ contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# Input tensors must be fp16, strides are supplied explicitly.
# ----------------------------------------------------------------------
@triton.jit
def trilu_outgoing_contract_kernel(
    left_ptr, right_ptr, out_ptr,            # *fp16 pointers
    B, H,                                     # runtime ints
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    N: tl.constexpr,                         # sequence length (compile‑time)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for TriMul (outgoing) contraction.
    The layout is [B, N, N, H] (batch, i, k/j, hidden).  We compute a
    per‑hidden‑channel GEMM A @ Bᵀ for each (b, h) pair.
    """
    # 0‑dim : combined batch & hidden index
    pid_batch = tl.program_id(0)   # 0 ≤ pid_batch < B*H
    pid_m     = tl.program_id(1)   # block row (i)
    pid_n     = tl.program_id(2)   # block col (j)

    # decode actual batch and hidden indices
    batch_idx = pid_batch // H
    hidden_idx = pid_batch % H

    # offsets within N
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this (batch, hidden) slice
    base_left  = left_ptr  + batch_idx * stride_left_b  + hidden_idx * stride_left_h
    base_right = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h

    # accumulator (FP32 for TensorCore accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along K
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_left + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # Right element layout: (j, k) → offset = j*stride_right_j + k*stride_right_k
        # For transposed view we index as (k, j)
        b_ptrs = base_right + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b_T = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # GEMM accumulation
        acc += tl.dot(a, b_T)   # (BLOCK_M, BLOCK_N)

    # Write result out[b,i,j,h] in FP16
    out_base = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper for the contraction using the Triton kernel.
# ----------------------------------------------------------------------
def _batched_triton_gemm_fp16(
    left: torch.Tensor, right: torch.Tensor, hidden_dim: int
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a
    custom Triton kernel that works on the native [B,N,N,H] layout.
    """
    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden dimension mismatch"

    # Output buffer
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # Strides in element units
    slb, sli, slk, slh = left.stride()
    srb, srj, srk, srh = right.stride()
    sob, soi, soj, soh = out.stride()

    # Block sizes – tuned for H100 TensorCores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden, n_blocks_M, n_blocks_N)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trilu_outgoing_contract_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        H=hidden_dim,
        stride_left_b=slb,
        stride_left_i=sli,
        stride_left_k=slk,
        stride_left_h=slh,
        stride_right_b=srb,
        stride_right_j=srj,
        stride_right_k=srk,
        stride_right_h=srh,
        stride_out_b=sob,
        stride_out_i=soi,
        stride_out_j=soj,
        stride_out_h=soh,
        N=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input  [B, N, N, C]  (float32)
        torch.Tensor,          # mask   [B, N, N]    (float32 or bool) – may be None
        Dict[str, torch.Tensor],  # weights dict (float32)
        Dict[str, Any],       # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Returns a float32 tensor of shape [B, N, N, dim].
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input/ output channel dimension  (C)
    hidden_dim = config["hidden_dim"] # internal hidden dimension
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)  # fp16

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once
    # -----------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # [hidden_dim, dim]
    left_gate_weight  = weights["left_gate.weight"]   # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]  # [hidden_dim, dim]
    right_gate_weight = weights["right_gate.weight"]  # [hidden_dim, dim]

    # Concatenate proj & gate weights to reduce the number of GEMMs
    left_combined_w  = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16)   # [2*H, dim]
    right_combined_w = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16) # [2*H, dim]

    out_gate_w   = weights["out_gate.weight"].to(torch.float16)   # [hidden_dim, dim]
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]

    # -----------------------------------------------------------------
    # 4️⃣ Compute left & left‑gate in a single linear (and right analogously)
    # -----------------------------------------------------------------
    # left_concat : [B, N, N, 2*H]   (first H = proj, second H = gate pre‑sigmoid)
    left_concat = torch.nn.functional.linear(x, left_combined_w)     # fp16
    right_concat = torch.nn.functional.linear(x, right_combined_w)   # fp16

    # Split projection and gate tensors
    left_proj = left_concat[..., :hidden_dim]          # [B,N,N,H]
    left_gate_pre = left_concat[..., hidden_dim:]      # [B,N,N,H]
    left_gate = torch.sigmoid(left_gate_pre)           # fp16

    right_proj = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)         # fp16

    # -----------------------------------------------------------------
    # 5️⃣ Compute out_gate (separate linear) – sigmoid
    # -----------------------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N,H] fp16
    out_gate = torch.sigmoid(out_gate_pre)                     # fp16

    # -----------------------------------------------------------------
    # 6️⃣ Fuse element‑wise multiplication (proj * gate) and optional mask
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask → fp16, flatten for the Triton kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)   # shape [B*N*N]
        total_elems = left_proj.numel()   # B*N*N*H
        BLOCK = 65536                    # reasonable tile for the mask kernel
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_flat, hidden_dim, total_elems,
            BLOCK=BLOCK, num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_flat, hidden_dim, total_elems,
            BLOCK=BLOCK, num_warps=4,
        )
        # At this point left_proj and right_proj already contain the gated values
        left = left_proj
        right = right_proj
    else:
        # No mask – plain element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction via our Triton batched GEMM
    # -----------------------------------------------------------------
    out = _batched_triton_gemm_fp16(left, right, hidden_dim)   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Apply out_gate and final linear projection (both fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                          # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
