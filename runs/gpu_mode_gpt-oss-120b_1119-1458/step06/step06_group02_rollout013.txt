def mul_gate_mask_kernel(x_ptr, gate_ptr, mask_ptr, hidden_dim, total_elems, 
                         mask_enabled: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)

    active = offs < total_elems

    x = tl.load(x_ptr + offs, mask=active, other=0.0)

        out = x * g * m
    else:
        out = x * g
    tl.store(x_ptr + offs, out, mask=active)
```

    (left_proj, left_gate, right_proj, right_gate, out_gate) and
    perform a single matmul (`torch.nn.functional.linear`).  This reduces
    the four separate linear layers to one, halving the number of reads of
    the large input tensor.
4️⃣  Apply `sigmoid` in‑place to the three gate tensors.
5️⃣  Fuse element‑wise multiplication (projection × gate) and optional masking
    with a custom Triton kernel (`mask_gate_fuse_kernel`).  This avoids two
    passes over the data when a mask is present.
6️⃣  Heavy O(N³) contraction
    `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`
    is realized as a batched GEMM (`torch.bmm`) after reshaping the tensors
    to [B·H, N, N].  This leverages cuBLAS/Tensor‑Core kernels on the H100.
7️⃣  Second Layer‑Norm over the hidden dimension (fp16).
8️⃣  Multiply by the output gate (computed in step 3) and a final linear
    projection back to `dim`.
9️⃣  Cast the final result back to float32 (required by the harness).

Only the forward pass is implemented – no gradients are needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel to fuse element‑wise multiplication with gating and mask.
# `x`  : tensor to be updated (left or right projection)
# `g`  : gate tensor (sigmoid already applied)
# `m`  : mask tensor (shape B·N·N), broadcast over the hidden dim.
# When a mask is not required we call the lighter `gate_mul_kernel`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    g_ptr,          # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : hidden dimension H
    total_elems,    # i64   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets for this block
    active = offs < total_elems                       # out‑of‑bounds guard

    # mask index = floor(offset / H) because H is the innermost stride
    mask_offs = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(g_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)  # mask=1.0 outside range

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


@triton.jit
def gate_mul_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    g_ptr,          # *fp16 : gate tensor (same shape as x)
    total_elems,    # i64   : total number of elements in x
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elems
    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(g_ptr + offs, mask=active, other=0.0)
    tl.store(x_ptr + offs, x * g, mask=active)


def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the heavy O(N³) contraction using a batched GEMM.
    Both `left` and `right` have shape [B, N, N, H] (fp16).
    Returns a tensor of the same shape.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))       # [B*H, N, N]

    # Restore original layout [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """Forward pass of the outgoing TriMul module (no autograd)."""
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Concatenate all five projection/gate weight matrices and perform
    #    a single fused matmul (no bias).  Result shape: [B, N, N, 5*H].
    # -----------------------------------------------------------------
    #   Order: left_proj, left_gate, right_proj, right_gate, out_gate
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Single linear layer – equivalent to 5 independent linear layers
    combined = torch.nn.functional.linear(x, w_concat)  # [B, N, N, 5*H]

    # -----------------------------------------------------------------
    # 4️⃣ Split the concatenated tensor into its components
    # -----------------------------------------------------------------
    H = hidden_dim
    left_proj = combined[..., :H]                     # [B, N, N, H]
    left_gate = combined[..., H:2 * H]                # logits for left_gate
    right_proj = combined[..., 2 * H:3 * H]           # [B, N, N, H]
    right_gate = combined[..., 3 * H:4 * H]           # logits for right_gate
    out_gate = combined[..., 4 * H:5 * H]             # logits for out_gate

    # Apply sigmoid in‑place to the gate logits (no extra memory)
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()

    # -----------------------------------------------------------------
    # 5️⃣ Fuse gating (and optional masking) with a Triton kernel.
    # -----------------------------------------------------------------
    total_elems = left_proj.numel()  # = B·N·N·H
    BLOCK = 1024  # ≤ max threads per block on H100
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    if not nomask:
        # Convert mask to fp16 and make it contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just multiply by the gates
        gate_mul_kernel[grid](
            left_proj,
            left_gate,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        gate_mul_kernel[grid](
            right_proj,
            right_gate,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )

    # At this point `left_proj` and `right_proj` hold the gated tensors.
    left = left_proj
    right = right_proj

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM
    # -----------------------------------------------------------------
    out = _batched_matmul(left, right)   # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (over hidden dim)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # fp16

    # -----------------------------------------------------------------
    # 8️⃣ Multiply by the output gate and final linear projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate  # element‑wise gating (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)       # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
