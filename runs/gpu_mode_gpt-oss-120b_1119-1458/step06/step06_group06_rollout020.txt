def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B,N,_,H = left.shape
    # Rearrange to (B*H,N,N) shape
    a = left.permute(0,3,1,2).reshape(B*H, N, N)
    b = right.permute(0,3,1,2).reshape(B*H, N, N)

    out = c.view(B, H, N, N).permute(0,2,3,1).contiguous()

"""
Optimized outgoing TriMul kernel (AlphaFold3 style).

Algorithm
---------
1. First LayerNorm on the 4‑D input (FP32) → cast to FP16.
2. Stack the five weight matrices (left/right projections + their gates + out‑gate)
   into a single matrix of shape (5·hidden_dim, dim) and perform ONE fused linear
   projection.  This yields the tensors:
   * left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre.
3. Apply sigmoid to the three *‑gate tensors.
4. If a mask is supplied, fuse element‑wise multiplication
   (proj * gate * mask) for *both* left and right streams with a tiny Triton
   kernel; otherwise multiply proj * gate directly.
5. Contract left and right via the heavy O(N³) operation:
   out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   implemented as a batched GEMM (Tensor‑Core) on FP16 data.
6. Second LayerNorm (FP16) → multiply by the out‑gate → final linear projection
   back to the original channel dimension.
7. Cast the result back to FP32 for the harness.

At least part of the computation (mask‑gate fusion) is performed with a
Triton kernel, while the large matrix multiplications leverage cuBLAS
Tensor‑Core kernels for maximal throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask).  Mask is broadcast over the
# hidden dimension.  Operates on a flat view of a contiguous tensor of
# shape [B·N·N·hidden_dim] (size = total_elements).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, size = total_elements
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,) – broadcasted
    total_elements,  # i32 : total number of elements in x
    hidden_dim,      # i32 : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load the data, gate and mask (the same mask entry for each hidden_dim)
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fused multiplication
    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS (Tensor‑Core).
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] via batched matmul.
    Input tensors must be half‑precision.
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] and perform (A @ Bᵀ) for each head
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))          # (B·H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Release temporary gate tensors
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction (batched GEMM) – cuBLAS implementation
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) → out‑gate → final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )
    out = out * out_gate                         # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())

    # ------------------------------------------------------------------
    # 7️⃣ Cast result back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```
