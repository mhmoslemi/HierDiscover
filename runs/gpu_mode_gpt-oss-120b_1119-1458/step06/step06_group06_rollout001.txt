@triton.jit
def proj_gate_mask_fuse_kernel(proj_ptr, gate_pre_ptr, mask_ptr, total_elements, hidden_dim, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate_pre = tl.load(gate_pre_ptr + offs, mask=mask, other=0.0)

    # compute mask index
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # compute sigmoid(gate_pre)
    # Use torch's approximated sigmoid: 0.5 * (1 + tanh(0.5*x))? Or use exponent.
    # Use tl.math.exp: NEG = -gate_pre, then exp.

    # fused multiply

    # Store back

import torch

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    ).half()

    # Concatenated projection weights
    # left_proj.weight shape (hidden_dim, dim)
    # left_gate.weight shape (hidden_dim, dim)
    # right_proj, right_gate similarly.

    # Construct concatenated weight matrix of shape (4*hidden_dim, dim)
    # Ensure we use .half(), store on device.
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).half()

    # Compute linear combine
    proj_concat = torch.nn.functional.linear(x_norm, concat_weight)  # [B,N,N,4*H]

    # Split
    # shape: [B,N,N, hidden_dim] each.
    B, N, _, _ = proj_concat.shape
    # Use view to unify last dimension
    # Actually we can reshape last dimension: split.

    proj_concat = proj_concat.view(B, N, N, 4, hidden_dim)  # [B,N,N,4,H]
    left_proj = proj_concat[...,0,:]  # shape [B,N,N,H]
    left_gate_pre = proj_concat[...,1,:]
    right_proj = proj_concat[...,2,:]
    right_gate_pre = proj_concat[...,3,:]

    # Compute out_gate (sigmoid)
    out_gate_pre = torch.nn.functional.linear(x_norm, weights["out_gate.weight"].half())
    out_gate = torch.sigmoid(out_gate_pre)  # [B,N,N,H]

    # For mask handling: produce mask FP16 as view(-1). shape [B*N*N].
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elements = left_proj.numel()
        BLOCK = 1024  # adjust
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Fuse left: left_proj and gate_pre -> left_proj = left_proj * sigmoid(gate_pre) * mask
        proj_gate_mask_fuse_kernel[grid](
            left_proj,
            left_gate_pre,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4
        )
        # Fuse right similarly:
        proj_gate_mask_fuse_kernel[grid](
            right_proj,
            right_gate_pre,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4
        )
    else:
        # No mask: just multiply with sigmoid
        # left = left_proj * sigmoid(left_gate_pre)
        # Same for right
        left_gate = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # After no mask case, we need to discard gate_pre to free memory.
    # Ensure left_proj and right_proj used.

    # Batched GEMM
    out = _batched_gemm_fp16(left_proj, right_proj)  # shape [B,N,N,H]

    # Second layer_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )
    # Multiply out_gate
    out = out * out_gate

    # Final linear to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())

    return out.float()

def proj_gate_mask_fuse_kernel(proj_ptr, gate_pre_ptr, mask_ptr,
                              total_elements, hidden_dim,

    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)

    out = proj * g * mask_val
    tl.store(proj_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) – fast forward pass, H100‑optimized.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) → cast to FP16.
2. Fuse the four linear projections (left_proj, left_gate, right_proj, right_gate)
   into a single mat‑mul by concatenating their weight matrices.  This reduces
   four kernel launches / four reads of the input tensor to one.
3. If a mask is present, fuse the element‑wise ops
       proj * sigmoid(gate) * mask
   for the left and right streams with a tiny Triton kernel
   (`proj_gate_mask_fuse_kernel`).  The mask is broadcast over the hidden
   dimension, so only one launch per stream is required.
   If no mask is given we simply multiply by the sigmoid’ed gates in PyTorch.
4. The heavy contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (A @ Bᵀ) using torch.bmm, which maps to a
   cuBLAS Tensor‑Core kernel (FP16 × FP16 → FP32 accumulation).
5. Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection back
   to the original channel dimension.
6. Cast the result back to FP32 for the harness.

Only the forward pass is required, and the implementation works for all
configurations listed in the problem statement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: proj * sigmoid(gate) * mask   (mask broadcast over hidden_dim)
# Operates on a flat view of a contiguous tensor of shape
#   [B, N, N, hidden_dim]   (size = B·N·N·hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_mask_fuse_kernel(
    proj_ptr,          # *fp16 : projection tensor (in‑place updated)
    gate_pre_ptr,      # *fp16 : pre‑sigmoid gate tensor
    mask_ptr,          # *fp16 : mask tensor of shape (B·N·N,)
    total_elements,    # i32   : total number of elements in proj_ptr
    hidden_dim,        # i32   : size of hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear offsets for this block
    mask = offs < total_elements                        # guard for tail

    # Load projection and gate_pre
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate_pre = tl.load(gate_pre_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask across hidden_dim:
    # Each `hidden_dim` consecutive elements share the same mask entry.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Sigmoid on the gate (in‑place to avoid extra memory)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate_pre))

    # Fused multiplication
    out = proj * gate_sig * m

    # Store back (in‑place)
    tl.store(proj_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via
    a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] and perform A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [B*H, N, N]
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)         # [B*H, N, N]

    # bmm does (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(

        torch.Tensor,          # input_tensor  [B, N, N, dim] (float32)
        torch.Tensor,          # mask_tensor   [B, N, N]  (float32/bool) – may be unused

        Dict[str, Any]            # config dict

    # ------------------------------------------------------------------

        input_tensor,

        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],

    proj_concat = torch.nn.functional.linear(x, weight_concat)  # [B,N,N,4*hidden_dim]

    right_gate_pre = proj_concat[..., 3, :]

    # ------------------------------------------------------------------
    # 3️⃣ Out‑gate (sigmoid) – simple linear + sigmoid
    # ------------------------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"].half())
    out_gate = torch.sigmoid(out_gate_pre)   # [B,N,N,hidden_dim]

    # 4️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if nomask:
        # No mask: just multiply by sigmoid(gate)
        left = left_proj * torch.sigmoid(left_gate_pre)
        right = right_proj * torch.sigmoid(right_gate_pre)
    else:
        # Convert mask to FP16 and flatten – shape (B·N·N,)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)

        total_elements = left_proj.numel()            # B * N * N * hidden_dim
        BLOCK = 1024                                   # 1 KB per block – tunable
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Fuse left stream
        proj_gate_mask_fuse_kernel[grid](
            left_proj,
            left_gate_pre,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right stream
        proj_gate_mask_fuse_kernel[grid](
            right_proj,
            right_gate_pre,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel the proj tensors already contain the gated+masked values
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction (batched GEMM) – cuBLAS implementation
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())  # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast result back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```
