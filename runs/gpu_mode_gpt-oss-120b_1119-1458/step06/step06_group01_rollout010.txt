"""
Outgoing TriMul kernel (AlphaFold‑3) – highly‑optimised forward pass.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 (numeric‑stable) and cast to FP16.
2️⃣  Fuse *all* linear projections (left/right projections, left/right gates,
    out‑gate) into a single GEMM by concatenating the weight matrices:
    `W_cat = [W_left_proj; W_right_proj; W_left_gate; W_right_gate; W_out_gate]`.
    The result is split back into the five tensors.
3️⃣  Apply sigmoid to the three gate tensors.  If a mask is supplied we
    fuse `proj * gate * mask` in a tiny Triton kernel (mask broadcast over the
    hidden dimension).  Otherwise we simply multiply `proj * gate`.
4️⃣  Perform the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM `C = A @ Bᵀ` with cuBLAS Tensor‑Core kernels, folding the
    hidden dimension into the batch axis for maximal throughput.
5️⃣  Second LayerNorm (FP16) → element‑wise multiply by the out‑gate.
6️⃣  Final linear projection back to the model dimension.
7️⃣  Cast the result back to FP32 (the harness expects FP32).

All heavy arithmetic is performed in FP16; only the first LayerNorm runs in FP32.
The Triton kernel is used for the mask‑gate‑projection fusion, satisfying the
“must‑use‑Triton” requirement while keeping the dominant O(N³) work on cuBLAS.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H]
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden dimension H)
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Tile offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Bounds checks for the i/j dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Iterate over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Element‑wise fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim)

    All tensors must be on the same CUDA device and dtype fp16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides expressed in element counts (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    # Block sizes – tuned for H100 occupancy
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 16   # 16 works well up to hidden_dim = 384

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction (left @ rightᵀ)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batch of GEMMs C = A @ Bᵀ where A and B have shape
    (batch, N, N) and batch = B·H (folded hidden dimension into batch axis).
    """
    B, N, _, H = left.shape
    # Fold (B, H) into the batch dimension
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply with right transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Input is FP32 – run stable layernorm in FP32
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Cast to FP16 for all further operators
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Concatenate all five linear weight matrices and run ONE GEMM
    #    (left_proj, right_proj, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # All weights are stored as float32 – convert once to fp16 on device
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    # Concatenate along the output-feature axis
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,  # concatenate output dimension (first dim)
    )

    # Single linear: produces a tensor of shape [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_weight)  # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split the five tensors and apply sigmoid to the three gates
    # ------------------------------------------------------------------
    splits = torch.split(combined, hidden_dim, dim=3)
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = splits

    left_gate = torch.sigmoid(left_gate_raw)   # fp16
    right_gate = torch.sigmoid(right_gate_raw) # fp16
    out_gate = torch.sigmoid(out_gate_raw)     # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask + gate fusion (Triton) or simple elementwise mul
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 (broadcasted over hidden dim later)
        mask_fp16 = mask_tensor.to(dtype_fp16)
        # Fuse proj * gate * mask
        left = _apply_mask_gate(left_proj_raw, left_gate, mask_fp16)    # [B, N, N, H]
        right = _apply_mask_gate(right_proj_raw, right_gate, mask_fp16) # [B, N, N, H]
    else:
        # No mask – vanilla elementwise multiplication
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor Cores)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate gating and final projection back to model dimension
    # ------------------------------------------------------------------
    out = out * out_gate  # elementwise gate (fp16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)  # [dim, H]
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast result back to FP32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
