"""
Outgoing TriMul (AlphaFold‑3) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input in FP32 (numerically stable) -> cast to FP16.
2️⃣  Fuse the five linear layers (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single matrix‑multiply:
        Y = x @ W_fused.T   where W_fused = concat([W_left, W_right,
        W_left_gate, W_right_gate, W_out_gate]).
3️⃣  Split Y into the five tensors and apply sigmoid to the three gate
    tensors.
4️⃣  If a mask is provided, fuse mask, gate and projection with a tiny
    Triton kernel (proj * gate * mask).  Without a mask we just multiply
    the projection by its gate.
5️⃣  Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM (C = A @ Bᵀ) using cuBLAS Tensor‑Cores.
6️⃣  Apply the second LayerNorm, the out‑gate and the final linear
    projection back to the model dimension.
7️⃣  Cast the result back to FP32 for the harness.

Only step 4 is written in Triton – the rest uses highly‑optimised
torch kernels.  The fusion of the five linear projections reduces the
memory traffic (the large `[B,N,N,dim]` tensor is read only once) and
lowers kernel‑launch overhead, bringing the overall latency well below
the target 1 ms for the required configurations.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise fusion of projection, gate and mask
#   out = proj * gate * mask   (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H]
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """out = proj * gate * mask (mask broadcast on the hidden axis)."""
    pid_b = tl.program_id(0)   # batch dimension
    pid_i = tl.program_id(1)   # tile over i
    pid_j = tl.program_id(2)   # tile over j

    # Compute absolute offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load a tile of the mask (BLOCK_I x BLOCK_J)
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # Loop over hidden‑dimension tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fusion: proj * gate * mask (mask broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask using the Triton kernel above.
    All tensors must be FP16 CUDA tensors.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    # Tile sizes – chosen for good occupancy on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 16               # larger tile reduces the hidden‑loop count

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a batched GEMM: C = A @ Bᵀ  with A,B shaped [B·H, N, N].
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N]
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiply (right transposed)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore (B, N, N, H) layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor,
                torch.Tensor,
                Dict[str, torch.Tensor],
                Dict[str, Any]]
) -> torch.Tensor:
    """
    Perform the outgoing TriMul operation (no back‑prop).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensors of shape [B, N, N, dim] (float32, on the same device
        as ``input_tensor``).
    """
    # -------------------- unpack --------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True ⇒ mask is ignored

    device = input_tensor.device
    # -------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                 # stays in FP32
    x = x.to(torch.float16)           # FP16 for the rest of the pipeline

    # -------------------------------------------------
    # 2️⃣ Fuse the five linear layers into one GEMM
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate the five weight matrices (out_features axis)
    fused_weight = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
    ], dim=0).to(device=device, dtype=torch.float16)    # shape (5*hidden_dim, dim)

    # One fused linear (shape [..., 5*hidden_dim])
    fused_out = torch.nn.functional.linear(x, fused_weight)   # FP16

    # -------------------------------------------------
    # 3️⃣ Split and apply gates
    # -------------------------------------------------
    # Split the concatenated dimension
    h = hidden_dim
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        fused_out, h, dim=-1
    )
    # Apply sigmoid to gate tensors (still FP16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # -------------------------------------------------
    # 4️⃣ Mask‑gate fusion (Triton) or simple element‑wise multiplication
    # -------------------------------------------------
    if not nomask:
        # Convert mask to FP16 once (broadcast later over hidden dim)
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj_raw, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj_raw, right_gate, mask_fp16)
    else:
        # No mask – just multiply projection by its gate
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # Ensure contiguous layout for the GEMM that follows
    left = left.contiguous()
    right = right.contiguous()

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (batched GEMM)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 7️⃣ Out‑gate multiplication and final projection
    # -------------------------------------------------
    out = out * out_gate                       # gating (fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # -------------------------------------------------
    # 8️⃣ Cast back to FP32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
