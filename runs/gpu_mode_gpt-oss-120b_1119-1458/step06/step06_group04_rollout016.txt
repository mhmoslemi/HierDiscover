@triton.jit
def batch_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)  # B*H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden index
    h = pid_bh % H
    b = pid_bh // H

    # Compute offsets for i and j
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Generate range indices within tile
    offset_i = i0 + tl.arange(0, BLOCK_M)
    offset_j = j0 + tl.arange(0, BLOCK_N)
    # Valid masks for boundaries
    mask_i = offset_i < N
    mask_j = offset_j < N

    # Initialize accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointer base for left and right for this batch/h
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        offset_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offset_k < N

        # Load A: left[i,k] tile
        # Compute pointer offsets: left_base + offset_i[:, None] * stride_left_i + offset_k[None, :] * stride_left_k
        a_ptrs = left_base \
                + offset_i[:, None] * stride_left_i \
                + offset_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B: right_T[k,j] tile = right[j,k] original
        # Compute pointer offsets: right_base + offset_k[:, None] * stride_right_k + offset_j[None, :] * stride_right_i
        b_ptrs = right_base \
                + offset_k[:, None] * stride_right_k \
                + offset_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product and accumulate
        acc += tl.dot(a, b)

    # Write results
    # Compute output pointer offsets: out_base + offset_i[:, None] * stride_out_i + offset_j[None, :] * stride_out_j
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    out_ptrs = out_base \
            + offset_i[:, None] * stride_out_i \
            + offset_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – optimized forward pass.

Algorithm
---------
1️⃣  First LayerNorm on the 4‑D input (float32) → cast to fp16.
2️⃣  Fuse *all* linear projections (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a **single** GEMM
    by stacking the five weight matrices (shape (5·hidden_dim, dim)).
    The result is split into the five tensors, three of which are
    passed through a sigmoid.
3️⃣  Apply the optional pair‑wise mask and multiply by the gate tensors.
4️⃣  Core O(N³) contraction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom Triton kernel that computes a batched
    GEMM (A @ Bᵀ) for every hidden channel without any extra
    permutation / copy.  The kernel works in fp16 with fp32 accumulation.
5️⃣  Second LayerNorm (fp16) on the contracted tensor.
6️⃣  Multiply by the out‑gate (element‑wise) and apply the final
    linear projection back to `dim` (fp16).
7️⃣  Cast the result to float32 (the harness expects float32).

All heavy arithmetic runs in fp16 to exploit Tensor‑cores on an H100.
The only kernels written in Triton are the mask‑free batched GEMM
(batched matrix multiplication) and the optional mask‑gate fuse
kernel (kept for completeness).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton batched GEMM for the O(N³) contraction.
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _triton_batch_gemm_kernel(
    left_ptr,          # *fp16  left  [B,N,N,H]
    right_ptr,         # *fp16  right [B,N,N,H]
    out_ptr,           # *fp16  output [B,N,N,H]
    B, N, H,           # i32   tensor dimensions
    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(0)   # combined batch × hidden channel
    pid_i  = tl.program_id(1)   # block row   (i‑dimension)
    pid_j  = tl.program_id(2)   # block col   (j‑dimension)

    # derive batch index b and hidden index h
    h = pid_bh % H
    b = pid_bh // H

    # tile start coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b  + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b  + h * stride_o_h

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (i,k) ----
        # left[i, k] = left[b, i, k, h]
        a_ptrs = left_base \
                 + offs_i[:, None] * stride_l_i \
                 + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)   # shape (BLOCK_M, BLOCK_K)

        # ---- load right tile (k,j) i.e. right[j, k] (transpose on‑the‑fly) ----
        # right_T[k, j] = right[b, j, k, h]
        b_ptrs = right_base \
                 + offs_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)   # shape (BLOCK_K, BLOCK_N)

        # ---- dot product and accumulate ----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) → FP32

    # store result (convert back to FP16)
    out_ptrs = out_base \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: fused linear that computes the five projections in one GEMM.
# ----------------------------------------------------------------------
def _fused_initial_proj(x: torch.Tensor,
                        left_proj_w: torch.Tensor,
                        right_proj_w: torch.Tensor,
                        left_gate_w: torch.Tensor,
                        right_gate_w: torch.Tensor,
                        out_gate_w: torch.Tensor) -> Tuple[torch.Tensor,
                                                            torch.Tensor,
                                                            torch.Tensor,
                                                            torch.Tensor,
                                                            torch.Tensor]:
    """
    Given input tensor x (fp16) of shape [B,N,N,dim], compute the five
    linear transformations in a single GEMM, split the result and apply
    sigmoid to the three gate tensors.
    """
    # stack weights: (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w,
         right_proj_w,
         left_gate_w,
         right_gate_w,
         out_gate_w],
        dim=0).contiguous()

    y = F.linear(x, combined_weight)     # [B,N,N,5*hidden_dim] (fp16)

    h = left_proj_w.shape[0]             # hidden_dim
    left      = y[..., 0       :h      ]  # proj
    right     = y[..., h       :2*h    ]  # proj
    left_gate = torch.sigmoid(y[..., 2*h   :3*h])   # gate
    right_gate= torch.sigmoid(y[..., 3*h   :4*h])   # gate
    out_gate  = torch.sigmoid(y[..., 4*h   :5*h])   # gate

    return left, right, left_gate, right_gate, out_gate


# ----------------------------------------------------------------------
# Triton‑based batched GEMM wrapper (fp16 inputs, fp16 output).
# ----------------------------------------------------------------------
def _triton_batched_gemm_fp16(left: torch.Tensor,
                              right: torch.Tensor,
                              BLOCK_M: int = 64,
                              BLOCK_N: int = 64,
                              BLOCK_K: int = 32) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a
    custom Triton kernel.  All tensors must be fp16 and on the same
    CUDA device.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element counts (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # launch a 3‑D grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _triton_batch_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,               # good default on H100
    )
    return out


# ----------------------------------------------------------------------
# Entrypoint – the forward pass of the outgoing TriMul module.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,             # input tensor [B, N, N, dim] (float32)
        torch.Tensor,             # mask tensor    [B, N, N]   (float32 / bool) or None
        Dict[str, torch.Tensor],  # model weights (float32)
        Dict[str, Any],           # config {dim, hidden_dim, nomask?}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Returns a float32 tensor of shape [B, N, N, dim] on the same device as the
    input.
    """
    # --------------------------- unpack ------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5  # LayerNorm epsilon – same default as torch.nn.LayerNorm

    # -------------------- first layer‑norm (float32) -----------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B,N,N,dim] float32

    # cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ---------------------- load weights in fp16 ------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------- fused linear + gates ---------------------------
    left, right, left_gate, right_gate, out_gate = _fused_initial_proj(
        x, left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w
    )   # all fp16, shape [B,N,N,hidden_dim]

    # -------------------- optional mask & gating -------------------------
    if (not nomask) and mask_tensor is not None:
        # mask is broadcasted over the hidden dimension
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left  = left * left_gate
        right = right * right_gate

    # -------------------- core contraction (O(N³)) -----------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = _triton_batched_gemm_fp16(left, right,
                                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # -------------------- second layer‑norm (fp16) ------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)  # [B,N,N,hidden_dim] fp16

    # -------------------- out‑gate gating --------------------------------
    out = out * out_gate

    # -------------------- final projection back to dim --------------------
    out = F.linear(out, to_out_w)          # [B,N,N,dim] fp16

    # -------------------- cast back to float32 ---------------------------
    return out.to(torch.float32)
