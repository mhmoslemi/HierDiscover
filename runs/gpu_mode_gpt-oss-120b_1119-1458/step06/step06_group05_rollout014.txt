def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # combined index for (batch, hidden)
    pid_i = tl.program_id(1)  # i tile
    pid_j = tl.program_id(2)  # j tile

    # decode batch and hidden
    total_h = H
    total_b = B
    # Let's decode: busize = total_b * total_h
    # Using integer division:
    bh = pid
    b = bh // total_h
    h = bh % total_h
    
    # coordinates for i and j
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # mask out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to output:
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + offs_i[:, None]*stride_out_i + offs_j[None,:]*stride_out_j

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32) # compute in fp32

    # Loop over K blocks
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile shape (I,K)
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_i[:, None]*stride_left_i + offs_k[None, :]*stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None,:], other=0.0).to(tl.float32)

        # Load right tile transposed shape (K,J):
        # Right original shape: [b, j, k, h] b, N_j(i.e., j), N_k(k), H
        # To get right_transposed tile shape [k, j], we need offsets: K dimension from k offset, J dimension from j offset.
        # So pointer: right_ptr + b*stride_right_b + h*stride_right_h + offs_k[:, None]*stride_right_k + offs_j[None,:]*stride_right_j

        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None,:], other=0.0).to(tl.float32)

        # Compute partial product: left_tile (I,K) dot right_tile.T? Actually right_tile is (K,J) if we loaded as described.
        # left_tile: shape (I,K). right_tile: shape (K,J). Then we can compute acc += tl.dot(left_tile, right_tile).
        acc += tl.dot(left_tile, right_tile)

    # After loop, cast to half and store 
    # Convert to fp16
    out_tile = acc.to(tl.float16)
    tl.store(out_ptrs, out_tile, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – optimized implementation.

Algorithm
---------
1️⃣  LayerNorm (FP32 → FP16) on the input.
2️⃣  Linear projections (left / right) and gate projections (sigmoid) in FP16.
3️⃣  If a mask is present, fuse mask, projection and gate in a single Triton
    kernel (`_apply_mask_gate`) to avoid extra memory traffic.
4️⃣  Perform the O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a single `torch.einsum` call – this reads the original strided
    layout directly and avoids costly permute‑and‑contiguous copies.
5️⃣  Second LayerNorm (FP16) → output‑gate (sigmoid) → final linear projection.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

At least part of the work (mask‑gate fusion) is done in Triton, while the
heavy O(N³) contraction is performed by the highly‑optimized cuBLAS path
invoked through `torch.einsum`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # fp16*  [B, N, N, H]
    gate_ptr,          # fp16*  [B, N, N, H] (sigmoid‑ed)
    mask_ptr,          # fp16*  [B, N, N]
    out_ptr,           # fp16*  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden dimension H)
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # i / j coordinates for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )                                   # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles.
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16, shape [BLOCK_I, BLOCK_J, BLOCK_H]

        # Load gate tile (already passed through sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16

        # Fuse: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: fuse projection, gating and mask using Triton.
# ----------------------------------------------------------------------
def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H] (fp16)
    mask       : [B, N, N]   (fp16)

    Returns out = proj * gate * mask  (mask broadcast over H) as fp16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tune tile sizes (empirically good on H100)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)

    # -----------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 on device
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # -----------------------------------------------------------------
    # 3️⃣ Projections (FP16)
    # -----------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 4️⃣ Gates (sigmoid, FP16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 5️⃣ Mask & gate fusion (if needed)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy contraction via a single einsum (avoids costly permutes)
    # -----------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Out‑gate + final projection back to model dimension
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 9️⃣ Cast to FP32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
