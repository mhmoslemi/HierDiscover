Implementation of Triton kernel:

```
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_raw_gate_ptr, right_raw_gate_ptr,
    mask_ptr,
    total_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Compute pointer offsets
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    left_gate_raw = tl.load(left_raw_gate_ptr + offs, mask=mask, other=0.0)
    right_gate_raw = tl.load(right_raw_gate_ptr + offs, mask=mask, other=0.0)

    # mask offset is vector index // hidden_dim
    mask_offs = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Compute sigmoid of gates
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements,
    hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    # Load values
    left_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask broadcast offset (each hidden element maps to same mask)
    mask_offs = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Sigmoid in FP16
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # Apply gates and mask
    left_out = left_val * left_gate * mask_val
    right_out = right_val * right_gate * mask_val

    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Efficient forward pass of the “outgoing” TriMul module using a fused
    projection layer, a Triton kernel for mask‑gate fusion and a single
    batched GEMM for the N³ contraction.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # 1) First LayerNorm (float32 for numerical stability)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B,N,N,dim] fp32

    # 2) Cast to FP16 for the rest of the pipeline
    x = x.to(fp16)

    # 3) Build a fused projection weight (5 * hidden_dim, dim)
    # Each linear in the original implementation has no bias.
    w_left = weights["left_proj.weight"].to(device=device, dtype=fp16)      # (h, d)
    w_right = weights["right_proj.weight"].to(device=device, dtype=fp16)
    w_left_gate = weights["left_gate.weight"].to(device=device, dtype=fp16)
    w_right_gate = weights["right_gate.weight"].to(device=device, dtype=fp16)
    w_out_gate = weights["out_gate.weight"].to(device=device, dtype=fp16)

    fused_w = torch.cat([w_left, w_left_gate, w_right, w_right_gate, w_out_gate], dim=0)
    # fused_w shape: (5*hidden_dim, dim)

    # 4) Linear projection for all five tensors at once.
    B, N, _, _ = x.shape
    total_positions = B * N * N
    x_flat = x.reshape(total_positions, dim)  # (B·N·N, dim)
    proj = F.linear(x_flat, fused_w)          # (B·N·N, 5*hidden_dim)
    proj = proj.view(B, N, N, 5 * hidden_dim)  # (B,N,N,5*hidden)

    # 5) Split projections
    # Slice order corresponds to the concatenation order used above.
    h = hidden_dim
    left_proj = proj[..., 0:h]          # (B,N,N,h)
    left_gate_raw = proj[..., h:2*h]
    right_proj = proj[..., 2*h:3*h]
    right_gate_raw = proj[..., 3*h:4*h]
    out_gate_raw = proj[..., 4*h:5*h]

    # 6) Compute out‑gate (sigmoid) – needed later after the second LN
    out_gate = torch.sigmoid(out_gate_raw)  # (B,N,N,h)

    # 7) Apply mask and the two per‑head gates.
    if nomask:
        # No mask ⇒ just sigmoid the gate tensors and multiply.
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Combine mask, sigmoid(gate) and the element‑wise multiplication in a
        # Triton kernel to keep the memory traffic to a minimum.
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate_raw.contiguous()
        right_gate = right_gate_raw.contiguous()

        total_el = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_el, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total_el,
            hidden_dim=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # left and right now hold the gated & masked values.
        # Release temporaries early.
        del left_gate, right_gate, mask_fp16

    # 8) N³ contraction: batched GEMM over the hidden dimension.
    # We reorganise tensors to (B·h, N, N) and use torch.bmm.
    # NOTE: .contiguous() forces a copy but the layout required by cuBLAS.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Perform A @ Bᵀ for each hidden slice.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Restore shape (B, N, N, h)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 9) Second LayerNorm (still FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B,N,N,h) fp16

    # 10) Apply out‑gate and final linear projection back to dim.
    out = out * out_gate  # element‑wise gating
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)  # (B,N,N,dim) fp16

    # 11) Cast back to float32 for the caller.
    return out.to(torch.float32)

"""
Optimized outgoing TriMul forward pass (AlphaFold3).

Algorithm
---------
1️⃣ LayerNorm on the input (float‑32 → stable statistics).
2️⃣ Cast to FP16 for the heavy part of the computation.
3️⃣ Fuse the five linear projections (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) into a single GEMM: weight shape (5·C, C_in).
4️⃣ Split the fused output into the five tensors.
5️⃣ Compute `out_gate = sigmoid(out_gate_raw)`.
6️⃣ If a mask exists, fuse mask application, sigmoid‑gating and the element‑wise
   multiplication of left/right projections in a Triton kernel.
   If no mask, apply the gates with a cheap element‑wise multiplication.
7️⃣ Compute the N³ contraction as a batched matrix multiplication:
   `out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]`.
   This is performed as a single `torch.bmm` on tensors of shape
   (B·C, N, N).
8️⃣ Second LayerNorm on the hidden dimension.
9️⃣ Multiply by `out_gate` (element‑wise) and apply the final linear
   projection back to `dim`.
🔟 Cast the result back to float‑32 for the caller.

Only the mask‑gate fusion is written in Triton; all GEMMs use the highly
optimized cuBLAS kernels, dramatically reducing kernel‑launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # *fp16   left projection (in‑place)
    right_ptr,              # *fp16   right projection (in‑place)
    left_gate_ptr,          # *fp16   left‑gate (pre‑sigmoid)
    right_gate_ptr,         # *fp16   right‑gate (pre‑sigmoid)
    mask_ptr,               # *fp16   mask (broadcast over hidden dim)
    total_elements,         # i32    total number of elements in left/right
    hidden_dim,             # i32    hidden dimension (C)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat index
    valid = offs < total_elements

    # Load tensors (OOB entries get a neutral value)
    left_val      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right_val     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate_raw= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    # The mask is broadcast over the hidden dimension.
    # For an index (b,i,j,c) the corresponding mask index is (b,i,j).
    mask_offs = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # Sigmoid (exact, FP16 friendly)
    left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # Apply gates and mask
    left_out  = left_val  * left_gate  * mask_val
    right_out = right_val * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask_tensor  : torch.Tensor [B, N, N]       (bool / 0‑1 float, optional)
        weights_dict : mapping name → torch.Tensor   (float32)
        config_dict  : must contain ``dim`` and ``hidden_dim``;
                       optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ----------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → stable statistics)
    # ----------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]   float32

    # ----------------------------------------------------------------------
    # 2️⃣ Convert to FP16 for the remaining heavy work
    # ----------------------------------------------------------------------
    x = x.to(fp16)                     # [B, N, N, dim]   fp16

    # ----------------------------------------------------------------------
    # 3️⃣ Fuse five linear layers into one GEMM
    # ----------------------------------------------------------------------
    # Cast all weights to FP16
    w_left      = weights["left_proj.weight"].to(device=device, dtype=fp16)   # (h, d)
    w_right     = weights["right_proj.weight"].to(device=device, dtype=fp16)
    w_left_gate = weights["left_gate.weight"].to(device=device, dtype=fp16)
    w_right_gate= weights["right_gate.weight"].to(device=device, dtype=fp16)
    w_out_gate  = weights["out_gate.weight"].to(device=device, dtype=fp16)

    # Concatenate along the output‑feature dimension → (5·h, d)
    fused_w = torch.cat(
        [w_left, w_left_gate, w_right, w_right_gate, w_out_gate],
        dim=0,
    )  # (5*hidden_dim, dim)

    B, N, _, _ = x.shape
    positions = B * N * N
    x_flat = x.reshape(positions, dim)                # (B·N·N, dim)

    # Single linear projection → (B·N·N, 5·hidden_dim)
    proj = F.linear(x_flat, fused_w)                  # fp16
    proj = proj.view(B, N, N, 5 * hidden_dim)         # (B,N,N,5·h)

    # ----------------------------------------------------------------------
    # 4️⃣ Split the fused projection
    # ----------------------------------------------------------------------
    h = hidden_dim
    left_proj      = proj[..., 0:h]
    left_gate_raw  = proj[..., h:2 * h]
    right_proj     = proj[..., 2 * h:3 * h]
    right_gate_raw = proj[..., 3 * h:4 * h]
    out_gate_raw   = proj[..., 4 * h:5 * h]

    # ----------------------------------------------------------------------
    # 5️⃣ Out‑gate (sigmoid) – needed after the second LayerNorm
    # ----------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)            # (B,N,N,h) fp16

    # ----------------------------------------------------------------------
    # 6️⃣ Apply mask + per‑head gates
    # ----------------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise gating
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Mask present → fuse mask, sigmoid(gate) and multiplication in Triton
        mask_fp16 = mask_tensor.to(fp16).contiguous()      # (B,N,N)

        # Ensure contiguous layout for the kernel
        left  = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate_raw.contiguous()
        right_gate = right_gate_raw.contiguous()

        total_el = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_el, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total_el,
            hidden_dim=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # After the kernel `left` and `right` hold the gated + masked values.
        del left_gate, right_gate, mask_fp16

    # ----------------------------------------------------------------------
    # 7️⃣ N³ contraction via a batched GEMM
    #    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    # ----------------------------------------------------------------------
    # Rearrange to (B·h, N, N) layout required by cuBLAS.
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiply (right is transposed on the inner dimension)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))    # (B·h, N, N)

    # Restore (B, N, N, h) layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still FP16)
    # ----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B,N,N,h) fp16

    # ----------------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final linear projection back to `dim`
    # ----------------------------------------------------------------------
    out = out * out_gate                    # element‑wise gating (fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)  # (dim, h)
    out = F.linear(out, to_out_w)           # (B,N,N,dim) fp16

    # ----------------------------------------------------------------------
    # 10️⃣ Cast back to float32 for the caller
    # ----------------------------------------------------------------------
    return out.to(torch.float32)
```
