1️⃣ Layer‑norm the input (FP32 → FP16 for speed).
2️⃣ Fuse the two linear projections (proj + gate) per side:
   * left_combined = Linear(x, [left_proj; left_gate]) → [B,N,N,2·H]
   * right_combined = Linear(x, [right_proj; right_gate])
   Split the concatenated dimension, apply sigmoid to the gate half.
3️⃣ If a mask is present, fuse `proj * gate * mask` in a Triton kernel
   (`_mask_gate_kernel`).  Otherwise just multiply the projection by its gate.
4️⃣ Heavy O(N³) contraction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is realized as a batched GEMM (C = A @ Bᵀ) using cuBLAS/Tensor‑cores.
5️⃣ Second LayerNorm (still FP16).
6️⃣ Apply the out‑gate (sigmoid of a linear on the original normalized input)
   and the final linear projection back to the model dimension.
7️⃣ Cast the result back to FP32 as required by the harness.

Only the mask‑gate fusion (step 3) is performed with a custom Triton kernel;
all other heavy work uses highly‑optimized cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H] (already after sigmoid)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out = proj * gate * mask.
    mask has shape (B,N,N) and is broadcast over the hidden dimension H.
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # i/j offsets for this tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask out‑of‑bounds for edge tiles
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J)
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # [BLOCK_I, BLOCK_J, BLOCK_H]

        # Load gate tile (already after sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fuse: proj * gate * mask   (mask broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: fuse mask, gate and projection using Triton.
# ----------------------------------------------------------------------
def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H] (fp16)
    mask       : [B, N, N]   (fp16)

    Returns out = proj * gate * mask (broadcast mask over H) as fp16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides in number of elements (torch's .stride already gives that)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes – chosen to fit into registers/shared memory on H100.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8   # works for hidden_dim up to 384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using cuBLAS.
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns out  : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] for cuBLAS batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point – the forward pass of TriMul (outgoing version).
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) – may be ignored if config["nomask"]=True
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    half = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=half)
    norm_bias = weights["norm.bias"].to(device=device, dtype=half)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    x = x.to(half)          # FP16 for all subsequent ops

    # -----------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=half)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=half)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=half)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=half)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=half)

    # Fuse left_proj + left_gate into a single weight matrix of shape (2*H, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # [2*H, dim]
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections + gated outputs (FP16)
    # -----------------------------------------------------------------
    # left side
    left_combined = F.linear(x, left_combined_w)                  # [B,N,N,2*H]
    left_proj = left_combined[..., :hidden_dim]                  # [B,N,N,H]
    left_gate_raw = left_combined[..., hidden_dim:]               # [B,N,N,H]
    left_gate = torch.sigmoid(left_gate_raw)                     # FP16

    # right side
    right_combined = F.linear(x, right_combined_w)
    right_proj = right_combined[..., :hidden_dim]
    right_gate_raw = right_combined[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_raw)

    # -----------------------------------------------------------------
    # 4️⃣ out‑gate (computed on the *original* normalized input)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))            # [B,N,N,H] (FP16)

    # -----------------------------------------------------------------
    # 5️⃣ Apply mask (if any) & fuse projection with gate
    # -----------------------------------------------------------------
    if not nomask:
        # convert mask to half‑precision (broadcast over H)
        mask_fp16 = mask_tensor.to(dtype=half)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple element‑wise multiplication with the gate
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy contraction (O(N³) across the k dimension)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] FP16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (on the hidden dimension)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=half)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=half)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final linear projection back to model dim
    # -----------------------------------------------------------------
    out = out * out_gate                # element‑wise gating, still FP16
    to_out_w = weights["to_out.weight"].to(device=device, dtype=half)  # [dim, H]
    out = F.linear(out, to_out_w)      # [B,N,N,dim] in FP16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to FP32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
