def mask_gate_sigmoid_fuse(in_proj_ptr, in_gate_ptr, mask_ptr, out_ptr, hidden_dim, size, BLOCK, nomask):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    proj = tl.load(in_proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(in_gate_ptr + offs, mask=valid, other=0.0)

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate))

    if not NOMASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        proj = proj * mask_val
        gate = gate * mask_val

    out = proj * gate
    tl.store(out_ptr + offs, out, mask=valid)

def final_proj_gate_kernel(
   out_norm_ptr,  # *fp16
   out_gate_ptr,  # *fp16
   weight_ptr,    # *fp16 weight shape (dim, hidden_dim) maybe flattened
   out_ptr,       # *fp32 final output pointer

   size: tl.constexpr,   # total output elements = B*N*N*dim
   BLOCK: tl.constexpr,
):

   # We may compute accumulations across hidden_dim.

   pid = tl.program_id(0)
   # Determine location index
   offs = pid * BLOCK + tl.arange(0, BLOCK)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Kernel: applies elementwise gate (sigmoid) and mask to projected features.
# The inputs proj and gate are separate tensors, each of shape [size] where
# size = batch * N * N * hidden_dim. For each element we compute:
# out = proj * sigmoid(gate) * mask
# mask has shape [batch*N*N] and is broadcast across hidden_dim.
@triton.jit
def proj_gate_fuse_kernel(
    proj_ptr,   # *fp16 projection tensor
    gate_ptr,   # *fp16 gate pre‑activation tensor
    mask_ptr,   # *fp16 mask tensor, shape [B*N*N] (broadcast)
    out_ptr,    # *fp16 output (projected * gated * mask)
    hidden_dim: tl.constexpr,
    size: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load projection and gate
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # sigmoid activation on gate
    gate = 1.0 / (1.0 + tl.exp(-gate))

    # Broadcast mask across hidden_dim
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

# Kernel: computes final projection after second LN and out‑gate.
# For each output element (location d of dim), we accumulate:
# out = Σ_h (norm * out_gate * weight[d, h])
@triton.jit
def final_proj_gate_kernel(
    norm_ptr,      # *fp16 normalized tensor (B,N,N,hidden_dim)
    gate_ptr,      # *fp16 out‑gate tensor (same shape)
    weight_ptr,    # *fp16 to_out weight (dim, hidden_dim)
    out_ptr,       # *fp32 output (B,N,N,dim)
    dim: tl.constexpr,
    hidden_dim: tl.constexpr,
    size: tl.constexpr,   # total number of output elements = B*N*N*dim
    BLOCK: tl.constexpr,
    HSTEP: tl.constexpr   # step size for iterating hidden_dim (e.g. 32)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_o = offs < size

    # Determine output channel index d and location idx
    d = offs % dim                # output channel in [0, dim)
    loc = offs // dim             # flattened location index in [0, B*N*N)

    # Base offsets for the hidden‑dim vectors
    hidden_off = loc * hidden_dim

    # Accumulator in FP32
    acc = tl.zeros([1], dtype=tl.float32)

    # Iterate over hidden_dim in chunks of HSTEP
    for k in range(0, hidden_dim, HSTEP):
        # hidden indices inside the chunk
        hid = k + tl.arange(0, HSTEP)
        valid_h = hid < hidden_dim

        # Load normalized and gate values
        norm_val = tl.load(norm_ptr + hidden_off + hid, mask=valid_h, other=0.0)    # fp16
        gate_val = tl.load(gate_ptr + hidden_off + hid, mask=valid_h, other=0.0)    # fp16

        # Element‑wise product (fp16 -> promote)
        prod = norm_val * gate_val

        # Load weight slice for current output channel
        w = tl.load(weight_ptr + d * hidden_dim + hid, mask=valid_h, other=0.0)    # fp16

        # Convert to FP32 and accumulate the dot product
        acc += tl.cast(prod, tl.float32) * tl.cast(w, tl.float32)

    # Store result as FP32
    tl.store(out_ptr + offs, acc, mask=mask_o)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the core O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single batched GEMM on the hidden‑dim axis.
    """
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) and multiply left @ rightᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))  # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Optimized outgoing TriMul forward pass (no autograd).

    Steps (high‑level):
    1️⃣ Layer‑norm the input (float32) → half.
    2️⃣ Fuse the two linear projections (proj & gate) for *left* and *right*
       by concatenating the weight matrices and performing a single matmul.
    3️⃣ Apply sigmoid + mask in‑place with a Triton kernel.
    4️⃣ Perform the core O(N³) contraction with a batched GEMM (torch.bmm).
    5️⃣ Second Layer‑norm on the hidden‑dim axis.
    6️⃣ Compute the out‑gate (linear + sigmoid) on the normalized input.
    7️⃣ Fuse the out‑gate multiplication and final linear projection with a Triton kernel.
    8️⃣ Cast the result back to float32.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → half
    # -----------------------------------------------------------------
    # The built‑in LayerNorm works on float32 safely, then we cast.
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # Cast all model weights once to fp16
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse left/right projection + gate (single matmul each)
    # -----------------------------------------------------------------
    # Concatenate proj and gate weights ( [proj; gate] )
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2*H, dim)

    # Linear over the flattened leading dimensions: (B*N*N, dim) @ (dim, 2*H).T
    left_comb = torch.nn.functional.linear(x, left_comb_w)      # (B,N,N,2*H)
    right_comb = torch.nn.functional.linear(x, right_comb_w)    # (B,N,N,2*H)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse sigmoid, mask, and element‑wise multiplication
    # -----------------------------------------------------------------
    B, N, _, _ = left_comb.shape
    # Prepare mask – either the user mask or a tensor of ones
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=x.device)
    mask_flat = mask_fp16.view(-1)          # shape [B*N*N]

    # Helper to compute gated tensor from combined projection+gate
    def _apply_proj_gate(comb_tensor: torch.Tensor) -> torch.Tensor:
        # Split projection and gate halves
        proj = comb_tensor[..., :hidden_dim].contiguous()
        gate_pre = comb_tensor[..., hidden_dim:].contiguous()
        # Allocate output
        out = torch.empty_like(proj)
        size = proj.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)
        proj_ptr = proj.data_ptr()
        gate_ptr = gate_pre.data_ptr()
        mask_ptr = mask_flat.data_ptr()
        out_ptr = out.data_ptr()
        proj_gate_fuse_kernel[grid](
            proj_ptr,
            gate_ptr,
            mask_ptr,
            out_ptr,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        return out

    left = _apply_proj_gate(left_comb)
    right = _apply_proj_gate(right_comb)

    # Release now‑unneeded tensors to free memory
    del left_comb, right_comb, proj, gate_pre, mask_fp16, mask_flat

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (tensor cores)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)        # (B,N,N,hidden_dim) fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden_dim
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Compute out‑gate (linear + sigmoid) on the *original* normalized input
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 7️⃣ Fuse out_gate multiplication + final projection (hidden→dim)
    # -----------------------------------------------------------------
    # Flatten tensors for kernel launch
    B, N, _, _ = out_norm.shape
    total_out_elems = B * N * N * dim
    BLOCK_OUT = 1024
    grid_out = (triton.cdiv(total_out_elems, BLOCK_OUT),)
    # Prepare pointers
    norm_ptr = out_norm.contiguous().view(-1).data_ptr()
    gate_ptr = out_gate.contiguous().view(-1).data_ptr()
    weight_ptr = to_out_w.contiguous().view(-1).data_ptr()
    # Allocate final output (float32)
    out_final = torch.empty((B, N, N, dim), dtype=torch.float32, device=x.device)
    out_final_ptr = out_final.view(-1).data_ptr()
    # Choose a step size for hidden_dim reduction (e.g. 32)
    HSTEP = 32
    final_proj_gate_kernel[grid_out](
        norm_ptr,
        gate_ptr,
        weight_ptr,
        out_final_ptr,
        dim,
        hidden_dim,
        total_out_elems,
        BLOCK=BLOCK_OUT,
        HSTEP=HSTEP,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # Return the result (float32)
    # -----------------------------------------------------------------
    return out_final

"""
Optimised outgoing TriMul forward (no autograd).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣  Fuse the two linear projections (proj + gate) for the *left* and *right*
    sides by concatenating the weight matrices and performing a single matmul.
3️⃣  Apply the sigmoid activation, optional mask and the element‑wise proj·gate
    product with a Triton kernel (`proj_gate_fuse_kernel`).
4️⃣  Perform the heavy O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a batched GEMM (`torch.bmm` on fp16 data – exploits Tensor‑cores).
5️⃣  Second Layer‑norm over the hidden dimension.
6️⃣  Compute the out‑gate (linear + sigmoid) on the normalised input.
7️⃣  Fuse the out‑gate multiplication and the final projection (hidden→dim)
    with a Triton kernel (`final_proj_gate_kernel`), accumulating in FP32.
8️⃣  Return the float32 result.

The implementation reduces memory traffic by:
* reading the normalised input only three times (instead of five);
* concatenating projection+gate weights (single matmul for each side);
* fusing sigmoid + mask + element‑wise multiplication in Triton;
* fusing the final gating + projection in Triton.

All heavy arithmetic is performed in float16 (Tensor‑core friendly) and the
final output is accumulated in float32 to preserve numerical quality.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: apply sigmoid, optional mask and element‑wise multiplication.
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fuse_kernel(
    proj_ptr,   # *fp16 projection tensor (size = total elements)
    gate_ptr,   # *fp16 gate pre‑activation tensor (same size)
    mask_ptr,   # *fp16 mask tensor, shape [B*N*N] – broadcast across hidden_dim
    out_ptr,    # *fp16 output tensor (projected * sigmoid(gate) * mask)
    hidden_dim: tl.constexpr,   # hidden dimension H
    size: tl.constexpr,         # total number of elements = B*N*N*H
    BLOCK: tl.constexpr,        # thread‑block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load projection and gate (both contiguous in memory)
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # sigmoid activation on the gate
    gate = 1.0 / (1.0 + tl.exp(-gate))

    # Broadcast mask across hidden_dim (each mask entry corresponds to H elements)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fuse everything
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: final hidden→dim projection after second LN and out‑gate.
# ----------------------------------------------------------------------
@triton.jit
def final_proj_gate_kernel(
    norm_ptr,   # *fp16 normalized tensor (B,N,N,hidden_dim)
    gate_ptr,   # *fp16 out‑gate tensor (same shape)
    weight_ptr, # *fp16 weight matrix (dim, hidden_dim) – row‑major
    out_ptr,    # *fp32 output tensor (B,N,N,dim)
    dim: tl.constexpr,          # output channel count
    hidden_dim: tl.constexpr,   # hidden dimension H
    size: tl.constexpr,         # total #output elements = B*N*N*dim
    BLOCK: tl.constexpr,        # thread‑block size
    HSTEP: tl.constexpr,        # chunk size for hidden‑dim reduction (e.g. 32)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Decode output channel d and flattened location index loc
    d = offs % dim                 # [0, dim)
    loc = offs // dim               # [0, B*N*N)

    # Base offset for the hidden‑dim slice of this location
    hidden_off = loc * hidden_dim

    # Accumulator (FP32)
    acc = tl.zeros([1], dtype=tl.float32)

    # Loop over hidden_dim in chunks of HSTEP
    for k in range(0, hidden_dim, HSTEP):
        # Hidden‑dim indices inside the current chunk
        hid = k + tl.arange(0, HSTEP)
        hvalid = hid < hidden_dim

        # Load normalized and gate values (fp16)
        n = tl.load(norm_ptr + hidden_off + hid, mask=hvalid, other=0.0)
        g = tl.load(gate_ptr + hidden_off + hid, mask=hvalid, other=0.0)

        # Element‑wise product (fp16 → fp32 later)
        prod = n * g

        # Load weight slice for output channel d
        w = tl.load(weight_ptr + d * hidden_dim + hid, mask=hvalid, other=0.0)

        # Accumulate dot product in FP32
        acc += tl.cast(prod, tl.float32) * tl.cast(w, tl.float32)

    # Write the FP32 result
    tl.store(out_ptr + offs, acc, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM (fp16) performing the O(N³) contraction.
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H) in fp16
    Returns:
        out: (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and multiply A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Entrypoint used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward (no autograd).  All heavy work is performed in
    fp16, final output is cast to float32.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → fp16
    # -----------------------------------------------------------------
    # torch.layer_norm is stable in float32; we cast the result to half.
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # Cast model weights once to fp16
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse proj & gate for left / right via concatenated linear
    # -----------------------------------------------------------------
    # concatenate weights: [proj; gate]  → shape (2*H, dim)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2*H, dim)

    # Single linear per side (input shape [..., dim] → [..., 2*H])
    left_comb = torch.nn.functional.linear(x, left_comb_w)      # (B,N,N,2*H)
    right_comb = torch.nn.functional.linear(x, right_comb_w)    # (B,N,N,2*H)

    # -----------------------------------------------------------------
    # 3️⃣ Apply sigmoid, mask and element‑wise multiplication (Triton)
    # -----------------------------------------------------------------
    B, N, _, _ = left_comb.shape
    # Prepare mask – either the provided mask or a tensor of ones
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=x.device)
    mask_flat = mask_fp16.view(-1)          # shape [B*N*N]

    def _apply_proj_gate(comb: torch.Tensor) -> torch.Tensor:
        """
        Split a combined tensor ([..., 2*H]) into projection and gate halves,
        then compute proj * sigmoid(gate) * mask via Triton.
        """
        proj = comb[..., :hidden_dim].contiguous().view(-1)       # (B*N*N*H)
        gate = comb[..., hidden_dim:].contiguous().view(-1)      # (B*N*N*H)
        out = torch.empty_like(proj)                             # allocate result

        total = proj.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        proj_ptr = proj.data_ptr()
        gate_ptr = gate.data_ptr()
        mask_ptr = mask_flat.data_ptr()
        out_ptr = out.data_ptr()

        proj_gate_fuse_kernel[grid](
            proj_ptr,
            gate_ptr,
            mask_ptr,
            out_ptr,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # reshape back to [B, N, N, hidden_dim]
        return out.view(B, N, N, hidden_dim)

    left = _apply_proj_gate(left_comb)
    right = _apply_proj_gate(right_comb)

    # Release intermediate tensors early
    del left_comb, right_comb, mask_fp16, mask_flat

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑cores)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # (B,N,N,hidden_dim), fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden_dim (still fp16)
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Compute out‑gate (linear + sigmoid) on the *original* normed input
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 7️⃣ Fuse out‑gate multiplication and final projection (hidden→dim)
    # -----------------------------------------------------------------
    total_out = B * N * N * dim
    BLOCK_OUT = 1024
    grid_out = (triton.cdiv(total_out, BLOCK_OUT),)

    # Prepare flat pointers; ensure contiguous layout
    norm_flat = out_norm.contiguous().view(-1)       # fp16
    gate_flat = out_gate.contiguous().view(-1)       # fp16
    weight_flat = to_out_w.contiguous().view(-1)    # fp16

    out_fp32 = torch.empty((B, N, N, dim), dtype=torch.float32, device=x.device)
    out_fp32_flat_ptr = out_fp32.view(-1).data_ptr()

    final_proj_gate_kernel[grid_out](
        norm_flat.data_ptr(),
        gate_flat.data_ptr(),
        weight_flat.data_ptr(),
        out_fp32_flat_ptr,
        dim,
        hidden_dim,
        total_out,
        BLOCK=BLOCK_OUT,
        HSTEP=32,                     # hidden‑dim chunk size (tune if needed)
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # Return the final tensor (float32)
    # -----------------------------------------------------------------
    return out_fp32
