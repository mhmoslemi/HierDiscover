def _batched_matmul(left, right):
    B, N, _, H = left.shape
    # Permute to B,H,N,N
    left_perm = left.permute(0, 3, 1, 2).contiguous() # shape B,H,N,N
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # Right transposed
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mul_proj_gate_mask_kernel(proj_ptr, gate_raw_ptr, mask_ptr,
                              total_elts: tl.int32,
                              hidden_dim: tl.constexpr,
                              BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    off = pid * BLOCK + tl.arange(0, BLOCK)
    mask = off < total_elts

    # map to mask index: each hidden_dim consecutive entries share same mask element.
    mask_index = off // hidden_dim

    # load
    proj = tl.load(proj_ptr + off, mask=mask, other=0.0)
    gate_raw = tl.load(gate_raw_ptr + off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_index, mask=mask, other=1.0)   # default 1 -> no mask

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    # fused multiply
    res = proj * gate * m

    tl.store(proj_ptr + off, res, mask=mask)
```

"""
TriMul (outgoing) – high‑performance forward pass.

Key ideas
---------
*   Fuse the five cheap linear layers (proj + gate × 2 + out_gate) into a
    single matrix multiply `x @ Wᵀ`.  This eliminates five separate launches
    and reads the input tensor only once.
*   Fuse the *mask*, *projection* and *sigmoid(gate)* for the left and right
    branches in a tiny Triton kernel (`mul_proj_gate_mask_kernel`).  The
    kernel works in FP16 and therefore keeps the heavy part of the model
    in the fast Tensor‑Core domain.
*   Perform the O(N³) contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] *
    right[b,j,k,h]` with a single batched GEMM (`torch.bmm`).  The two
    permutes are kept but the tensors are made contiguous only once.
*   The remaining two small operations (second LayerNorm and the final
    projection) stay in PyTorch – they are memory‑bound and negligible
    compared to the N³ term.

The implementation respects the original forward semantics while
significantly reducing kernel launch overhead.  It runs on H100
(Tr​iton 3.3.1) and returns a float‑32 tensor as required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: left/right projection * sigmoid(gate) * mask (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mul_proj_gate_mask_kernel(
    proj_ptr,            # *fp16 : projected tensor (left_proj or right_proj)
    gate_raw_ptr,        # *fp16 : raw gate tensor (pre‑sigmoid)
    mask_ptr,            # *fp16 : mask tensor (broadcast over hidden_dim)
    total_elts: tl.int32,   # total number of elements = B·N·N·hidden_dim
    hidden_dim: tl.constexpr,  # hidden dimension (size of last axis)
    BLOCK: tl.constexpr,   # tile size (≤ 1024)
):
    """
    Writes `proj = proj * sigmoid(gate_raw) * mask` back into ``proj_ptr``.
    ``mask`` is a [B, N, N] tensor broadcast across the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear element ids
    mask = offs < total_elts                                 # out‑of‑bounds guard

    # mask index = element // hidden_dim  (because mask is repeated hidden_dim times)
    mask_idx = offs // hidden_dim

    proj      = tl.load(proj_ptr      + offs, mask=mask, other=0.0)
    gate_raw  = tl.load(gate_raw_ptr  + offs, mask=mask, other=0.0)
    m         = tl.load(mask_ptr      + mask_idx, mask=mask, other=1.0)   # default 1 → no mask

    # sigmoid(x) = 1/(1+exp(-x))  – work in FP16 (torch uses the same definition)
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    out = proj * gate * m
    tl.store(proj_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    All tensors are FP16 and have shape [B,N,N,hidden_dim].

    Returns:
        out : [B, N, N, hidden_dim] (FP16)
    """
    B, N, _, H = left.shape
    # (B, H, N, N) – make contiguous for cuBLAS
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to batch dimension B*H
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # restore original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask_tensor  : [B, N, N]       (float32 / bool) – may be all‑ones
        - weights_dict : model parameters (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``, optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm in FP32 (stable) → cast to FP16 for the rest
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32, shape [B,N,N,dim]

    x = x_norm.to(torch.float16)   # FP16

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the five cheap linear layers into one matrix multiply
    # ------------------------------------------------------------------
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # concatenate along the output‑feature dimension → (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # FP16

    B, N, _, _ = input_tensor.shape
    # flatten the spatial dimensions – the last axis (dim) is kept for matmul
    x_flat = x.view(-1, dim)                       # shape (B·N·N, dim)

    # fused linear pass (FP16)
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B·N·N, 5·hidden_dim)

    # reshape to separate the five streams
    combined_out = combined_out.view(B, N, N, 5, hidden_dim)   # (B,N,N,5,hidden_dim)

    left_proj   = combined_out[..., 0, :]   # (B,N,N,hidden_dim)
    right_proj  = combined_out[..., 1, :]
    left_gate_r = combined_out[..., 2, :]   # raw (pre‑sigmoid) gates
    right_gate_r = combined_out[..., 3, :]
    out_gate_r  = combined_out[..., 4, :]

    # ------------------------------------------------------------------
    # 3️⃣  Apply sigmoid(gate) and (optional) mask in a fused Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast across the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * sigmoid(left_gate_r) * mask
        mul_proj_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate_r.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * sigmoid(right_gate_r) * mask
        mul_proj_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate_r.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # no mask – elementwise multiply only
        left_gate  = torch.sigmoid(left_gate_r)
        right_gate = torch.sigmoid(right_gate_r)
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣  Heavy O(N³) contraction via a single batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul(left, right)   # (B,N,N,hidden_dim), FP16

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm (still in FP16 for speed)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # FP16

    # ------------------------------------------------------------------
    # 6️⃣  Output gate (computed earlier) and final projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_r)           # (B,N,N,hidden_dim), FP16
    out = out * out_gate                           # gated output

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)         # (B,N,N,dim), FP16

    # ------------------------------------------------------------------
    # 7️⃣  Return to FP32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
