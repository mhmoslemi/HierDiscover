@triton.jit
def batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BH, M, N, K,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute start indices
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Offsets arrays
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

    # mask for valid i, j
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_tile in range(tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a tile
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # b tile transposed: we need B_T so that B_T[k,j] = B[j,k]
        # Use strides stride_b_i for original i dimension (row index, originally j after transposition), and stride_b_j for original j dimension (col index, originally k)
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_j \
                 + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product (a*b) accumulate; a shape (M_block,K_block), b shape (K_block,N_block)
        acc += tl.dot(a, b)

        # Increment offsets? Not needed; we recompute each iteration.

    # Convert to FP16
    c = acc.to(tl.float16)

    # Store
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
            + offs_i[:, None] * stride_c_i \
            + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])
```

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------------------
# Triton kernel: mask * projection * gate (mask broadcast over hidden dim)
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                      B, N, H,
                      stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                      stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                      stride_mask_b, stride_mask_i, stride_mask_j,
                      stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                      BLOCK_I: tl.constexpr,
                      BLOCK_J: tl.constexpr,
                      BLOCK_H: tl.constexpr):
    # program ids for batch, i tile, j tile
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute global offsets for i,j
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # load mask tile (broadcast later over h)
    mask_ptrs = mask_ptr + pid_b * stride_mask_b \
                + offs_i[:, None] * stride_mask_i \
                + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs,
                        mask=mask_i[:, None] & mask_j[None, :],
                        other=0.0)  # fp16

    # iterate over h tiles (hidden)
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load proj tile
        proj_ptrs = proj_ptr + pid_b * stride_proj_b \
                    + offs_i[:, None, None] * stride_proj_i \
                    + offs_j[None, :, None] * stride_proj_j \
                    + offs_h[None, None, :] * stride_proj_h
        proj_tile = tl.load(proj_ptrs,
                            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                            other=0.0)

        # load gate tile
        gate_ptrs = gate_ptr + pid_b * stride_gate_b \
                    + offs_i[:, None, None] * stride_gate_i \
                    + offs_j[None, :, None] * stride_gate_j \
                    + offs_h[None, None, :] * stride_gate_h
        gate_tile = tl.load(gate_ptrs,
                            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                            other=0.0)

        # multiply
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # store
        out_ptrs = out_ptr + pid_b * stride_out_b \
                   + offs_i[:, None, None] * stride_out_i \
                   + offs_j[None, :, None] * stride_out_j \
                   + offs_h[None, None, :] * stride_out_h
        tl.store(out_ptrs,
                 out_tile,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1)//BLOCK_I, (N + BLOCK_J - 1)//BLOCK_J)
    _mask_gate_kernel[grid](
        proj, gate, mask, out,
        B, N, H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B^T  (A, B, C of shape [BH, N, N])
# -------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BH, M, N, K,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)  # batch*head id
    pid_i = tl.program_id(1)   # i tile
    pid_j = tl.program_id(2)   # j tile

    # Global offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = a_ptr + pid_bh * stride_a_batch \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # load B tile transposed (K x N) -> B[j,k] in memory
        # B_T tile element at (k, j) is B[j, k] in original layout
        b_ptrs = b_ptr + pid_bh * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # dot accumulation
        acc += tl.dot(a, b)

    # cast to fp16 and store
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_bh * stride_c_batch \
             + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs,
             c,
             mask=mask_i[:, None] & mask_j[None, :])

def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Input tensors must be FP16 and have shape [B, N, N, H].
    Returns a FP16 tensor of same shape.
    """
    B, N, _, H = left.shape
    # Permute to [B, H, N, N] so that the hidden dimension becomes batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    # Reshape to [BH, N, N]
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Strides in elements
    s_a = left_flat.stride()
    s_b = right_flat.stride()
    s_c = out_flat.stride()

    # Tile sizes: 64x64x64 usually works well on H100 for fp16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the downstream code
    # -----------------------------------------------------------------

"""
Outgoing TriMul (AlphaFold‑3) forward pass – highly‑optimized mixed‑precision
implementation.

Key ideas
---------
1️⃣  Layer‑norm the 4‑D input in FP32 and cast to FP16.
2️⃣  Fuse each pair of linear projection + gate into a *single* linear call
    by concatenating the weights (e.g. left_proj ⊕ left_gate).  This halves
    the number of passes over the giant [B,N,N,dim] tensor and reduces
    memory traffic.
3️⃣  If a mask is present, fuse `proj * gate * mask` in a tiny Triton kernel
    (`_mask_gate_kernel`).  Without a mask we simply multiply by the gates.
4️⃣  The O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is expressed as a batched GEMM C = A @ Bᵀ where the batch dimension is
    B·H (batch×hidden).  The GEMM is implemented in Triton (`_batched_gemm_kernel`)
    and runs directly on the H100 Tensor‑Cores.
5️⃣  Second LayerNorm (FP16) → gating by the out‑gate → final linear projection
    back to the model dimension.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

Both Triton kernels are deliberately lightweight (no shared‑memory staging)
so they launch with a small number of warps and keep occupancy high.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over hidden)
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H] (sigmoid‑ed)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask is broadcast over the hidden dimension H)
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # -----------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # -----------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # -----------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim)

    All inputs must be FP16 CUDA tensors.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A, B, C of shape [BH, N, N])
# -------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BH, M, N, K,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)   # (batch × hidden) index
    pid_i = tl.program_id(1)    # i‑tile
    pid_j = tl.program_id(2)    # j‑tile

    # Global offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile:   [BLOCK_M, BLOCK_K]
        a_ptrs = (
            a_ptr
            + pid_bh * stride_a_batch
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Bᵀ tile (requires transposition):
        # B_T[k, j] = B[j, k]  →  index with original row‑stride for j and col‑stride for k
        b_ptrs = (
            b_ptr
            + pid_bh * stride_b_batch
            + offs_k[:, None] * stride_b_k      # k goes over the original column stride
            + offs_j[None, :] * stride_b_i
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) accumulation in FP32

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + pid_bh * stride_c_batch
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (FP16, contiguous).  Hidden dimension H is folded into the
        batch dimension for the kernel.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # Fold batch and hidden → [BH, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides in element units
    s_a = left_flat.stride()
    s_b = right_flat.stride()
    s_c = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        N,
        N,
        s_a[0],
        s_a[1],
        s_a[2],
        s_b[0],
        s_b[1],
        s_b[2],
        s_c[0],
        s_c[1],
        s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # Restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left & right streams (single linear each)
    # -----------------------------------------------------------------
    # left_proj + left_gate
    left_comb_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)   # [2*H, dim]
    left_comb = torch.nn.functional.linear(x, left_comb_weight)                # [B,N,N,2*H]
    left = left_comb[..., :hidden_dim]          # projection
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # gate

    # right_proj + right_gate
    right_comb_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)
    right_comb = torch.nn.functional.linear(x, right_comb_weight)
    right = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # out‑gate (separate linear)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask‑gate fusion (Triton) or simple element‑wise multiplication
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left, left_gate, mask_fp16)
        right = _apply_mask_gate(right, right_gate, mask_fp16)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via a fused batched GEMM Triton kernel
    # -----------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # [B, N, N, hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Gating and final projection back to model dim
    # -----------------------------------------------------------------
    out = out * out_gate                         # out‑gate (element‑wise)
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast result back to FP32 for the caller
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
