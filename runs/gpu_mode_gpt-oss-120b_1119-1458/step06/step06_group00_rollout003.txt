def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    device = input_tensor.device
    fp16 = torch.float16
    # 1. LayerNorm input
    norm_w = weights["norm.weight"]; norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # 2. Convert to FP16
    x = x.to(fp16)
    # 3. Cast weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # concat for fused linear
    combined_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0).to(device=device, dtype=fp16)
    # 4. Fused linear projection
    combined = F.linear(x, combined_weight) # shape (B,N,N,5*hidden_dim)
    # split
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = combined.split(hidden_dim, dim=-1)

    # 5. Gates
    left_gate = torch.sigmoid(left_gate_raw) # FP16
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # 6. Apply gating and mask via triton
    if nomask:
        # no mask: just multiply left_proj_raw * left_gate, etc.
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # allocate left and right mutated in place
        left = left_proj_raw.contiguous()
        right = right_proj_raw.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        total = left.numel()
        BLOCK = 1024 # or maybe 2048
        grid = ((total + BLOCK - 1)//BLOCK,)
        _gate_mask_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # After kernel, left and right contain left_proj_raw * left_gate * mask
        # out_gate remains separate.

    # 7. Perform contraction: left and right are (B,N,N,H)
    B, N, _, H = left.shape
    # reshape to (B*H,N,N) for matmul
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    # bmm
    out = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back to (B,N,N,H)
    out = out.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # 8. Second LayerNorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # 9. Multiply out_gate (already FP16) and final linear projection
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)

    # 10. cast back 32bit
    return out.to(torch.float32)
```

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. LayerNorm on the input (float32 → FP32 statistics).
2. Cast to FP16 for the heavy inner‑loop.
3. Fuse the five linear projections (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single matmul by
   concatenating their weight matrices into a (5*H, dim) tensor.
   This reduces the number of cuBLAS launches from 5 → 1.
4. Split the fused tensor into the five raw tensors.
5. Compute the output gate (sigmoid) once.
6. If a mask is present, fuse mask, sigmoid‑gate and element‑wise
   multiplication for *left* and *right* in a Triton kernel
   (one pass, avoiding extra loads/stores).  Without a mask we
   simply multiply the left/right projections by the sigmoid gates
   in‑place.
7. Perform the O(N³) contraction:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   by reshaping to (B*H, N, N) and calling a single batched GEMM
   (`torch.bmm`), which uses Tensor‑cores on H100.
8. Second LayerNorm on the hidden dimension (FP16).
9. Apply the output gate and a final linear projection back to
   the original feature dimension.
10. Cast the result back to float32 and return.

The only Triton kernel needed is the mask‑gate fusion kernel,
which also computes the sigmoid for each gate.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: mask + sigmoid gate + element‑wise multiplication.
#   left  = left_proj  * sigmoid(left_gate)  * mask
#   right = right_proj * sigmoid(right_gate) * mask
# All tensors are contiguous FP16.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]  (raw gate pre‑sigmoid)
    right_gate_ptr,    # *fp16  [B,N,N,H]  (raw gate pre‑sigmoid)
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # int   (B·N·N·H)
    H,                 # int   hidden dimension (constexpr)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat index
    valid = offs < total_elements

    # Load raw tensors
    left_proj = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # sigmoid (1/(1+exp(-x))) – TL supports fp16 exp directly
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # Load mask (broadcast over H).  OOB entries get neutral 1.0.
    mask_off = offs // H
    mask_val = tl.load(mask_ptr + mask_off, mask=valid, other=1.0)

    # Apply mask and gates
    left_out = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val

    # Store results back in‑place
    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float, ignored if config["nomask"] is True)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain "dim" and "hidden_dim".
                         Optional "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )                               # -> [B,N,N,dim] (FP32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Prepare fused weight matrix (5 * hidden_dim x dim)
    # ------------------------------------------------------------------
    # All projection/gate weights have shape (hidden_dim, dim) and no bias.
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    # Concatenate along the output‑feature axis
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device=device, dtype=fp16)     # (5*H, dim)

    # ------------------------------------------------------------------
    # 4️⃣ One fused linear projection -> shape [B,N,N,5*H]
    # ------------------------------------------------------------------
    fused = F.linear(x, fused_weight)      # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Split the fused tensor into the five components
    # ------------------------------------------------------------------
    # Resulting views are contiguous in the hidden dimension.
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = fused.split(
        hidden_dim, dim=-1
    )  # each: [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Compute the output gate (used after the second LN)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply mask and sigmoid gates to left/right projections
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simply apply sigmoid gates (in‑place multiplication)
        left_gate = torch.sigmoid(left_gate_raw)   # FP16
        right_gate = torch.sigmoid(right_gate_raw) # FP16
        left = left_proj_raw.mul_(left_gate)      # in‑place, final left
        right = right_proj_raw.mul_(right_gate)   # final right
    else:
        # Convert mask to FP16 (0/1) and make all tensors contiguous.
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        # Ensure contiguity for the kernel (required for flat indexing)
        left = left_proj_raw.contiguous()
        right = right_proj_raw.contiguous()
        left_gate_raw = left_gate_raw.contiguous()
        right_gate_raw = right_gate_raw.contiguous()

        total = left.numel()
        BLOCK = 1024                      # threads per block (multiple of 32)
        grid = ((total + BLOCK - 1) // BLOCK,)

        # Fuse mask, sigmoid(gate) and multiplication in a single Triton launch
        _gate_mask_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate_raw,
            right_gate_ptr=right_gate_raw,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,          # 8 warps → 256 threads per CTA, good for fp16 math
            num_stages=1,
        )
        # Release temporaries (helps memory pressure)
        del left_gate_raw, right_gate_raw, mask_fp16

    # ------------------------------------------------------------------
    # 8️⃣ O(N³) contraction: batched GEMM A @ Bᵀ   (FP16)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape      # H == hidden_dim
    # Collapse batch and hidden dimension → (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (torch.bmm uses Tensor‑core acceleration)
    out_hidden = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: (B, N, N, H)
    out = out_hidden.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    # ------------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gate (FP16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)  # (dim, H)
    out = F.linear(out, to_out_w)            # FP16 → [B,N,N,dim]

    # ------------------------------------------------------------------
    # 11️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
