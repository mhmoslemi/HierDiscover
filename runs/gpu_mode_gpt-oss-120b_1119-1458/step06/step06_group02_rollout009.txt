"""
TriMul – outgoing version (AlphaFold‑3) forward pass.
--------------------------------------------------

The kernel follows the exact computation of the reference PyTorch
implementation but fuses the element‑wise mask‑gate multiplication
into a single Triton kernel to cut memory traffic.

Steps
=====
1️⃣  Layer‑Norm over the last dimension (float32 for numeric stability).
2️⃣  Cast the normalized tensor to fp16 – the heavy part of the
    computation runs in half‑precision.
3️⃣  Linear projections (`left_proj`, `right_proj`) and the three
    gating projections (`*_gate`) are performed with fp16 weights
    via `torch.nn.functional.linear` (efficient cuBLAS GEMM).
4️⃣  When a mask is present we fuse

        tensor = tensor * gate * mask

    with a Triton kernel (`mask_gate_fuse_kernel`).  This removes
    two separate element‑wise passes.
5️⃣  The tri‑multiplicative contraction

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    is realized as a batched GEMM: the tensors are reshaped to
    `[B·H, N, N]` (H = hidden_dim) and multiplied with `torch.bmm`,
    which on H100 maps to Tensor‑Core kernels.
6️⃣  A second Layer‑Norm (fp16) is applied over the hidden dimension.
7️⃣  The output gate is multiplied element‑wise and a final linear
    projection maps the hidden dimension back to the original `dim`.
8️⃣  The result is cast back to float32 (expected by the caller).

Only the forward pass is required – no autograd is recorded.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: mask * gate * tensor   (tensor and gate have shape
# [B, N, N, H] – flattened view), mask has shape [B, N, N].
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : size of hidden dimension H
    total_elems,    # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,                     # compile‑time block size
):
    """Fuse `x = x * gate * mask` in‑place."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # linear indices
    active = offs < total_elems                         # bounds check

    # each hidden_dim values belong to the same mask entry
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`
    using a batched GEMM: for each hidden channel `d` we compute a
    matrix product of size [N, N] with its mate transposed.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel becomes a batch entry
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  ->  (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # restore to original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry‑point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask_tensor  : torch.Tensor [B, N, N]          (float32/bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on same device as input)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                                 # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy lifting
    # -----------------------------------------------------------------
    x = x.to(torch.float16)           # now fp16

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 (single copy)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 5️⃣ Gating values (sigmoid) – also fp16
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Fuse mask (if any) and gating into the projected tensors
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and contiguous (shape B, N, N)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()                     # B·N·N·H
        BLOCK = 1024                                    # ≤ max threads per block
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just fuse the gating multiplication (in‑place)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Output gating then final linear projection back to original dim
    # -----------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim]  fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (as required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
