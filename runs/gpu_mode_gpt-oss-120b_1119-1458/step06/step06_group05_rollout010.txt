    gate_ptr,    // *fp16 [B,N,N,2H] (pre-sigmoid)

    left_out_ptr, // *fp16 [B,N,N,H]
    right_out_ptr, // *fp16 [B,N,N,H]
    B,N,H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h,
    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):

    pid_i = tl.program_id(1)

@triton.jit
def _mask_proj_gate_sigmoid_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = offs_i < N
    j_mask = offs_j < N

    # Load mask tile (shape I,J)
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=i_mask[:, None] & j_mask[None, :], other=1.0)  # default 1

    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h in range(num_h_tiles):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        h_mask = offs_h < H

        # proj tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )

        # gate pre tile
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_pre_tile = tl.load(
            gate_ptrs,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )

        # sigmoid
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_pre_tile))

        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
        )
```

def _apply_mask_gate_sigmoid(proj, gate_pre, mask):
    B,N,_,H = proj.shape
    out = torch.empty_like(proj) # same dtype (fp16)
    s_proj = proj.stride()
    s_gate = gate_pre.stride()
    s_mask = mask.stride()
    s_out = out.stride()
    # tile sizes
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8
    grid = (B, (N + BLOCK_I -1)//BLOCK_I, (N + BLOCK_J -1)//BLOCK_J)
    _mask_proj_gate_sigmoid_kernel[grid](
        proj,
        gate_pre,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out

"""
TriMul (outgoing) – fast forward implementation.
------------------------------------------------
1️⃣  LayerNorm (FP32 → FP16) on the input.
2️⃣  Fuse the two projection Linear layers (left & right) into a single
    MatMul (weight shape (2·H, D)).  Likewise fuse the two gate Linear layers.
3️⃣  If a mask exists we fuse mask‑multiplication, sigmoid gating and the
    projection → final left/right tensors in a Triton kernel (one kernel
    per side).  No‑mask case uses a cheap element‑wise multiplication.
4️⃣  The O(N³) contraction  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    is performed as a batched GEMM (C = A @ Bᵀ) using cuBLAS Tensor‑Core
    kernels – we simply reshape to [B·H, N, N] and call torch.bmm.
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → element‑wise multiplier.
6️⃣  Final Linear (hidden → dim) and cast back to FP32.

Only the mask‑gate‑sigmoid fusion (step 3) is implemented in Triton; all
large matrix multiplications use highly‑optimized cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: out = proj * sigmoid(gate) * mask  (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_sigmoid_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H]  (pre‑sigmoid)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute  out = proj * sigmoid(gate) * mask
    mask is 2‑D (B,N,N) and is broadcast over the hidden dimension H.
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # i / j coordinates for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J). Broadcast later.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=1.0,                     # treat out‑of‑bounds as 1 (no effect)
    )                                   # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles.
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                                    # fp16, shape [I,J,H]

        # Load gate pre‑sigmoid tile
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_pre_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                                    # fp16, shape [I,J,H]

        # Sigmoid (stable, no extra temporaries)
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_pre_tile))

        # Fuse projection, gate and mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: apply mask, gate (sigmoid) and projection with a single kernel.
# ----------------------------------------------------------------------
def _apply_proj_gate_mask_sigmoid(
    proj: torch.Tensor,       # [B, N, N, H]  fp16, projection (already split)
    gate_pre: torch.Tensor,   # [B, N, N, H]  fp16, pre‑sigmoid gate
    mask: torch.Tensor,       # [B, N, N]     fp16 (0/1)
) -> torch.Tensor:
    """
    Returns out = proj * sigmoid(gate_pre) * mask   (fp16).
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides expressed in element counts (not bytes)
    s_proj = proj.stride()
    s_gate = gate_pre.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8  # works well up to hidden = 384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _proj_gate_mask_sigmoid_kernel[grid](
        proj,
        gate_pre,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  using cuBLAS.
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns       out : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # Collapse batch and hidden dimensions for a single bmm call
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul (no gradients).
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Parameters
    ----------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        weights       : dict of model weights (float32)
        config        : must contain ``dim`` and ``hidden_dim``;
                       optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # 0️⃣  Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)                     # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 2️⃣  Cast model weights to fp16 (on‑device)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w   = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # -----------------------------------------------------------------
    # 3️⃣  Fuse left/right projections into a single MatMul
    # -----------------------------------------------------------------
    proj_weight_concat = torch.cat([left_proj_w, right_proj_w], dim=0)   # (2·H, D)
    proj_concat = torch.nn.functional.linear(x, proj_weight_concat)    # [B, N, N, 2·H]

    # -----------------------------------------------------------------
    # 4️⃣  Fuse left/right gates (pre‑sigmoid) into a single MatMul
    # -----------------------------------------------------------------
    gate_weight_concat = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2·H, D)
    gate_concat_pre    = torch.nn.functional.linear(x, gate_weight_concat)  # [B,N,N,2·H]

    # -----------------------------------------------------------------
    # 5️⃣  Out‑gate (independent linear + sigmoid)
    # -----------------------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)           # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)                             # fp16

    # -----------------------------------------------------------------
    # 6️⃣  Split concatenated tensors into left/right halves
    # -----------------------------------------------------------------
    left_proj   = proj_concat[..., :hidden_dim]      # [B,N,N,H]
    right_proj  = proj_concat[..., hidden_dim:]      # [B,N,N,H]
    left_gate_pre  = gate_concat_pre[..., :hidden_dim]
    right_gate_pre = gate_concat_pre[..., hidden_dim:]

    # -----------------------------------------------------------------
    # 7️⃣  Apply mask + sigmoid gating
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 (0/1) and keep original shape [B,N,N]
        mask_fp16 = mask_tensor.to(dtype=dtype_fp16)
        left  = _apply_proj_gate_mask_sigmoid(left_proj,  left_gate_pre,  mask_fp16)
        right = _apply_proj_gate_mask_sigmoid(right_proj, right_gate_pre, mask_fp16)
    else:
        # No mask → simple element‑wise multiply with sigmoid'ed gates
        left_gate  = torch.sigmoid(left_gate_pre)   # fp16
        right_gate = torch.sigmoid(right_gate_pre)  # fp16
        left  = left_proj  * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 8️⃣  Heavy O(N³) contraction via batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 9️⃣  Second LayerNorm (fp16) → out‑gate multiplication
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                                  # element‑wise (fp16)

    # -----------------------------------------------------------------
    # 🔟  Final projection back to model dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
