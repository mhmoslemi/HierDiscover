@triton.jit
def fuse_gates_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr, hidden_dim, total_elems,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr  # compile-time flag
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems
    # load left, left_gate, right, right_gate
    l = tl.load(left_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    # compute mask if needed
    if HAS_MASK:
         mask_idx = offs // hidden_dim
         m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
    else:
         m = 1.0
    # compute fused outputs
    l_f = l * lg * m
    r_f = r * rg * m
    # store back in place
    tl.store(left_ptr + offs, l_f, mask=valid)
    tl.store(right_ptr + offs, r_f, mask=valid)

import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def fused_gate_mask_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim, total_elems,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (broadcast across hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    l_out = l * lg * m
    r_out = r * rg * m

    tl.store(left_ptr + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and multiply
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Triton kernel: fuse mask, left‑gate and right‑gate in‑place.

@triton.jit
def fused_gate_mask_fuse_kernel(
    left_ptr,            # *fp16  – left projection tensor (flattened)
    left_gate_ptr,       # *fp16  – left gate tensor (flattened)
    right_ptr,           # *fp16  – right projection tensor (flattened)
    right_gate_ptr,      # *fp16  – right gate tensor (flattened)
    mask_ptr,            # *fp16  – mask tensor flattened, shape [B*N*N]
    hidden_dim,          # i32   – size of the hidden dimension (H)
    total_elems,         # i32   – total number of elements in left/right (B*N*N*H)
    BLOCK: tl.constexpr, # compile‑time tile size (≤1024)
):
    """
    For each element `e` of the tensors:
        left[e]  = left[e]  * left_gate[e]  * mask[ e // H ]
        right[e] = right[e] * right_gate[e] * mask[ e // H ]

    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    l_val   = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    lg_val  = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r_val   = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    rg_val  = tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    mask_idx = offs // hidden_dim
    m_val   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default mask = 1

    l_out = l_val * lg_val * m_val
    r_out = r_val * rg_val * m_val

    # Store back in‑place
    tl.store(left_ptr      + offs, l_out, mask=valid)
    tl.store(right_ptr     + offs, r_out, mask=valid)

    torch.Tensor (fp16) of shape [B, N, N, H]
