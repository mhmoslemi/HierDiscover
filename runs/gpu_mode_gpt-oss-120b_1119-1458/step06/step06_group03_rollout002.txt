@triton.jit
def fused_proj_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elts: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elts
    # compute mask index:
    mask_idx = offs // hidden_dim
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # mask fp16 broadcast: should be mask shape (B,N,N), broadcast over hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)
```

"""
TriMul (outgoing) – forward only, mixed‑precision implementation.

Algorithm
---------
1️⃣  First LayerNorm in FP32 (keeps numerical stability), then cast to fp16.
2️⃣  Linear projections (left/right) and three gate‑linear layers are done in fp16.
3️⃣  If a mask is provided we fuse the three element‑wise factors
    (proj * gate * mask) for *both* left and right tensors in a single Triton kernel.
    This halves the kernel launch overhead compared to two separate kernels.
4️⃣  The O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM using torch.bmm.
    By reshaping to (B·H, N, N) we let cuBLAS/Tensor‑Cores handle the heavy compute.
5️⃣  Second LayerNorm (fp16) on the hidden dimension, element‑wise gating by out_gate,
    and a final linear projection back to `dim`.
6️⃣  The result is returned in fp32 (as required by the harness).

Only the forward pass is needed; all kernels are pure inference.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) for *both* left and right tensors.
# The mask has shape (B, N, N) and is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_left_right_kernel(
    left_ptr,          # *fp16 : left tensor       [total_elts]
    right_ptr,         # *fp16 : right tensor      [total_elts]
    left_gate_ptr,     # *fp16 : left gate tensor  [total_elts]
    right_gate_ptr,    # *fp16 : right gate tensor [total_elts]
    mask_ptr,          # *fp16 : mask tensor (B,N,N) [total_masks]
    total_elts: tl.int32,   # total number of elements in left/right
    hidden_dim: tl.int32,   # size of hidden dimension
    BLOCK: tl.constexpr,    # tile size (≤1024)
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask_valid = offs < total_elts

    # Index of the associated mask entry (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    # Load the required values (masked loads avoid out‑of‑bounds reads)
    left_val      = tl.load(left_ptr + offs,      mask=mask_valid, other=0.0)
    right_val     = tl.load(right_ptr + offs,     mask=mask_valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=mask_valid, other=0.0)
    right_gate_val= tl.load(right_gate_ptr + offs,mask=mask_valid, other=0.0)
    mask_val      = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)  # default 1 ⇒ no mask

    # Fuse: proj * gate * mask
    left_out  = left_val  * left_gate_val  * mask_val
    right_out = right_val * right_gate_val * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=mask_valid)
    tl.store(right_ptr + offs, right_out, mask=mask_valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM via torch.bmm (Tensor‑Core).
# ----------------------------------------------------------------------
def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    for fp16 tensors `left` and `right` of shape [B, N, N, H].

    Returns:
        out : tensor of shape [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # Reshape to treat each hidden channel as a separate batch element.
    # left_perm: [B, H, N, N]; right_perm: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Merge batch dimensions.
    left_mat = left_perm.view(B * H, N, N)                     # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)                   # (B*H, N, N)
    # GEMM on each hidden slice: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)
    # Restore original layout.
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    # 🔟 Apply out_gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # --------------------------------------------------------------
    # 𝟣️⃣ Return FP32 (required by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
