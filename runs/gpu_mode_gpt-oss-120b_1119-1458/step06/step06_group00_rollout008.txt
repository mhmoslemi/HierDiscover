"""
TriMul (outgoing) Triton‐accelerated forward pass.

Algorithm
---------
1.  LayerNorm on the input (float32 → fp16 for later ops).
2.  Fuse the four linear projections (left_proj, right_proj, left_gate, right_gate)
    into a single FP16 GEMM:
        X @ W_concat.T   (W_concat = [left_proj; right_proj; left_gate; right_gate])
    and split the result into the four tensors.
3.  Apply sigmoid to the two gate tensors.
4.  Fuse mask (if present) and the sigmoid gates onto the projected tensors
    in‑place with a small Triton kernel.
5.  Batched GEMM for the O(N³) contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    performed as a batched matmul after flattening the (batch, hidden) axes.
6.  LayerNorm on the contracted tensor (hidden_dim), multiply by the out‑gate,
    and project back to the original dimension.
7.  Return the result in float32.

The implementation reduces kernel launch overhead by merging the four linear
projections into a single GEMM and by fusing mask + gate multiplication with a
Triton kernel.  All heavy arithmetic is performed in FP16, while the two
LayerNorm steps stay numerically stable by using the provided FP32 weights.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) with the sigmoid gates.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]  (broadcasted over H)
    total_elements,    # i64    (B·N·N·H)
    H,                 # i64    hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elements

    # mask index (broadcast over hidden dim)
    mask_offs = offs // H

    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    left_out  = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=mask)
    tl.store(right_ptr + offs, right_out,  mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : Tuple[input, mask, weights, config]
        - input : torch.Tensor [B, N, N, dim]  (float32)
        - mask  : torch.Tensor [B, N, N]      (bool / 0‑1 float, may be all‑ones)
        - weights : dict of weight tensors (float32)
        - config  : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 → stable statistics)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)               # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Convert to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast all model weights to fp16
    # ------------------------------------------------------------------
    # projection + gate weights (concatenated)
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)  # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)   # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)  # (H, dim)

    # out‑gate weight (separate)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)    # (H, dim)

    # final Linear and final LayerNorm weights
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)       # (dim, H)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projections + gate linear layers (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate weight matrices: [left_proj; right_proj; left_gate; right_gate]
    # shape after concat: (4*H, dim)
    weight_concat = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )
    # Linear: X (B,N,N,dim) @ weight_concat.T → (B,N,N,4*H)
    proj = F.linear(x, weight_concat)  # fp16, shape [B,N,N,4*H]

    # Split projections
    H = hidden_dim
    left_raw      = proj[..., 0:H]          # (B,N,N,H)
    right_raw     = proj[..., H:2*H]        # (B,N,N,H)
    left_gate_raw = proj[..., 2*H:3*H]      # (B,N,N,H)
    right_gate_raw= proj[..., 3*H:4*H]      # (B,N,N,H)

    # --------------------------------------------------------------
    # 5️⃣ Apply sigmoid to gate tensors
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # fp16
    right_gate = torch.sigmoid(right_gate_raw) # fp16

    # --------------------------------------------------------------
    # 6️⃣ Fuse mask (+ optional) and sigmoid gates in‑place
    # --------------------------------------------------------------
    if nomask:
        # Simple in‑place gating – no mask to read.
        left_raw.mul_(left_gate)   # left = left_raw * left_gate
        right_raw.mul_(right_gate) # right = right_raw * right_gate
    else:
        # Convert mask to fp16 (0/1) and make tensors contiguous.
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        left_raw = left_raw.contiguous()
        right_raw = right_raw.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left_raw.numel()
        BLOCK = 1024  # threads per program (max 1024)
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left_raw,
            right_ptr=right_raw,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=H,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporaries early
        del left_gate, right_gate, mask_fp16
    # After this block, left_raw & right_raw hold the gated (and masked) tensors.
    left = left_raw
    right = right_raw
    del left_raw, right_raw

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction: batched GEMM A @ Bᵀ   (FP16)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Reshape to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul: left @ rightᵀ  over the 2nd axis (k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 9️⃣ Compute out‑gate (sigmoid) from the original X
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 🔟 Apply out‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate               # element‑wise gating
    out = F.linear(out, to_out_w)      # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 11️⃣ Cast back to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
