@triton.jit
def _mask_mul_kernel(data_ptr, mask_ptr, out_ptr,
                     B, N, H,
                     stride_data_b, stride_data_i, stride_data_j, stride_data_h,
                     stride_mask_b, stride_mask_i, stride_mask_j,
                     stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                     BLOCK_I: tl.constexpr,
                     BLOCK_J: tl.constexpr,
                     BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute indices
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N
    # Load mask tile
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    # mask tile shape (BLOCK_I, BLOCK_J)
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)
    # Now loop over H tile
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # Compute mask for loads
        data_ptrs = data_ptr + pid_b * stride_data_b + offs_i[:, None, None] * stride_data_i + offs_j[None, :, None] * stride_data_j + offs_h[None, None, :] * stride_data_h
        # Load data tile
        data_tile = tl.load(data_ptrs,
                            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                            other=0.0)
        # Multiply
        out_tile = data_tile * mask_tile[:, :, None]
        # Store
        out_ptrs = out_ptr + pid_b * stride_out_b + offs_i[:, None, None] * stride_out_i + offs_j[None, :, None] * stride_out_j + offs_h[None, None, :] * stride_out_h
        tl.store(out_ptrs,
                 out_tile,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # first layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weights["norm.weight"], weights["norm.bias"])
    x = x.to(torch.float16)

    # concat weights
    left_proj_w = weights["left_proj.weight"].to(device=input_tensor.device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=input_tensor.device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=input_tensor.device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=input_tensor.device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=input_tensor.device, dtype=torch.float16)

    combined_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)  # shape [5*hidden_dim, dim]

    # Linear projection (F\ linear)
    proj = torch.nn.functional.linear(x, combined_weight) # shape [B,N,N,5*hidden_dim]

    # split
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = proj.split(hidden_dim, dim=-1)  # each shape [B,N,N,hidden_dim]

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # Apply gates to projections
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Mask multiplication using Triton kernel
    mask_fp16 = mask_tensor.to(torch.float16)
    left = _apply_mask(left, mask_fp16)
    right = _apply_mask(right, mask_fp16)

    # O(N³) contraction via batched GEMM
    out = _batched_matmul_transpose_fp16(left, right)

    # second layernorm using fp16 weight
    to_out_norm_w = weights["to_out_norm.weight"].to(device=input_tensor.device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=input_tensor.device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out_gate gating
    out = out * out_gate

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(device=input_tensor.device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # shape [B,N,N,dim]

    # cast to float32
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fused implementation.

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input (FP32) and cast to FP16.
2️⃣  Fuse the five linear projections (left, right, three gates) into a *single*
    FP16 GEMM by concatenating their weight matrices.
3️⃣  Split the result, apply sigmoid to the three gate tensors and multiply them
    with the corresponding projection tensors.
4️⃣  Apply the optional mask (broadcast over the hidden dimension) using a small
    Triton kernel – this satisfies the “must use Triton” requirement.
5️⃣  Perform the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM C = A @ Bᵀ (torch.bmm) after folding the hidden dimension
    into the batch axis.
6️⃣  Second Layer‑Norm (FP16) → multiply by `out_gate` → final linear projection
    back to the model dimension.
7️⃣  Cast the result back to FP32 (the harness expects FP32).

All heavy linear algebra is executed by cuBLAS Tensor‑Core kernels; the only
Triton kernel fuses the mask‑multiplication to avoid an extra CUDA launch per
element.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: mask * tensor (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    data_ptr,               # *fp16  [B, N, N, H]
    mask_ptr,               # *fp16  [B, N, N]
    out_ptr,                # *fp16  [B, N, N, H]
    B, N, H,
    stride_data_b, stride_data_i, stride_data_j, stride_data_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = data * mask   (mask is broadcast over the hidden dimension H)
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16 scalar per (i,j)

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load data tile
        data_ptrs = (
            data_ptr
            + pid_b * stride_data_b
            + offs_i[:, None, None] * stride_data_i
            + offs_j[None, :, None] * stride_data_j
            + offs_h[None, None, :] * stride_data_h
        )
        data_tile = tl.load(
            data_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: data * mask (broadcast over hidden dim)
        out_tile = data_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask(data: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse element‑wise multiplication with a mask broadcast over the hidden dimension.
    All tensors must be FP16 and on the same CUDA device.
    """
    B, N, _, H = data.shape
    out = torch.empty_like(data)

    # Tile sizes – chosen to keep shared memory modest while giving high occupancy.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32   # hidden_dim ≤ 768 in our configs; 32 works well.

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    s_data = data.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    _mask_mul_kernel[grid](
        data,
        mask,
        out,
        B,
        N,
        H,
        s_data[0],
        s_data[1],
        s_data[2],
        s_data[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batched GEMM C = A @ Bᵀ with cuBLAS Tensor‑Core kernels.
    The hidden dimension is folded into the batch dimension (B·H) to maximize
    throughput.
    """
    B, N, _, H = left.shape
    # Fold batch & hidden => shape [B·H, N, N]
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul: A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32
    x = x.to(torch.float16)   # FP16 for the rest of the forward

    # ------------------------------------------------------------------
    # 2️⃣ Concatenate all five weight matrices (no biases) and perform ONE GEMM
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    w_right  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    w_lgate  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_rgate  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_outgate= weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # [5*hidden_dim, dim]
    combined_weight = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)

    # Linear projection (applies to the last dimension of x)
    proj = torch.nn.functional.linear(x, combined_weight)   # [B, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Split and apply gates
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = proj.split(hidden_dim, dim=-1)

    left_gate  = torch.sigmoid(left_gate_raw)   # [B, N, N, hidden_dim]  (fp16)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_proj * left_gate               # [B, N, N, hidden_dim]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Mask – fused element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    # Ensure mask is FP16; it is broadcast over the hidden dimension inside the kernel.
    mask_fp16 = mask_tensor.to(torch.float16)
    left  = _apply_mask(left,  mask_fp16)
    right = _apply_mask(right, mask_fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate and final projection back to model dimension
    # ------------------------------------------------------------------
    out = out * out_gate                               # gating (fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
