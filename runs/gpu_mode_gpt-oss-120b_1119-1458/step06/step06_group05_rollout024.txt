@triton.jit
def fused_norm_linear_kernel(
    x_ptr,   # FP32 input shape (M, dim)
    y_ptr,   # FP16 output shape (M, 5 * hidden_dim) combined
    B, N, dim, hidden_dim,
    stride_x_m, stride_x_k,
    stride_y_m, stride_y_oh,
    weight_ptr,  # FP16 weight shape (5*hidden_dim, dim)
    stride_w_oh, stride_w_k,
    epsilon: tl.constexpr,
    BLOCK_M: tl.constexpr,   # number of rows per program
    BLOCK_K: tl.constexpr,   # tile size of dim
    BLOCK_O: tl.constexpr,   # #output features per thread (maybe 1, then we can loop over hidden dims).
):
   pid = tl.program_id(0)
   block_row = pid * BLOCK_M + tl.arange(0, BLOCK_M)
   # We'll compute row_mask = block_row < M
   # Stage 1: Compute mean and var
   sum = tl.zeros((BLOCK_M,), dtype=tl.float32)
   sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)
   for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim
        x = tl.load(x_ptr + block_row[:, None] * stride_x_m + offs_k[None, :] * stride_x_k,
                    mask=row_mask[:, None] & mask_k[None, :], other=0.0, eviction_policy='evict_last')
        x_f32 = x.to(tl.float32)
        sum += tl.sum(x_f32, axis=1)  # We need to sum across K dimension (like reduce)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)
   mean = sum / dim
   var = sum_sq / dim - mean*mean
   inv_std = 1.0 / tl.sqrt(var + epsilon)
   # Stage 2: Compute output linear via weight
   # Compute normalized x = (x - mean) * inv_std for each element 
   # Then compute dot product with weight: for each output feature (output hidden dim) multiply normalized_x * weight.
   # We can compute by iterating over hidden dim in tail loops.
   for oh in range(0, tl.cdiv(5*hidden_dim, BLOCK_O)):
        outs = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float16)
        for k in range(0, tl.cdiv(dim, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < dim
            x = tl.load(x_ptr + block_row[:, None]*stride_x_m + offs_k[None,:] * stride_x_k,
                         mask=row_mask[:, None] & mask_k[None,:], other=0.0).to(tl.float16)

            # Load weight tile
            off_w_k = offs_k[:, None]  # (BLOCK_K,1)
            off_w_o = oh * BLOCK_O + tl.arange(0, BLOCK_O)
            mask_w_o = off_w_o < 5*hidden_dim

            # Multiply add: accumulate
            outs += tl.dot(x_norm, w)  # shape (BLOCK_M, BLOCK_O)
        # store outs into y_ptr

    stride_y_O,
    stride_w_out,
    stride_w_k,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_O: tl.constexpr,
    EPS: tl.constexpr
):
    # calculate row offsets
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)
    row_mask = rows < (B*N*N)
    # sum and sum_sq
    sum_x = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_x2 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    # do K loop to compute sum and sum_sq and accumulate partial dot for each O tile
    # We will have outer loop over O tiles
    # For O_tile in range(0, ceil(total_O / BLOCK_O)):
    o_tile_start = 0
    while o_tile_start < total_O:
        # accumulators for dot products (BLOCK_M, BLOCK_O)
        acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float32)
        # iterate over K
        for k in range(0, tl.cdiv(dim, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < dim
            # load x tile
            x = tl.load(x_ptr + rows[:, None] * stride_x_dim + offs_k[None, :],
                        mask=row_mask[:, None] & mask_k[None, :],
                        other=0.0) # shape (BLOCK_M, BLOCK_K)
            sum_x += tl.sum(x, axis=1)
            sum_x2 += tl.sum(x * x, axis=1)
            # load weight tile for this O tile
            offs_o = o_tile_start + tl.arange(0, BLOCK_O)
            mask_o = offs_o < total_O
            # weight tile: shape (BLOCK_O, BLOCK_K)??? Actually we need weight for each output channel (row) and K dimension (col). Since weight is organized as rows=total_O, cols=dim.

            w = tl.load(weight_ptr + offs_o[:, None] * stride_w_out + offs_k[None, :] * stride_w_k,
                        mask=mask_o[:, None] & mask_k[None, :],
                        other=0.0) # shape (BLOCK_O, BLOCK_K)
            # Compute dot product for each row: x (BLOCK_M, BLOCK_K) dot w^T (BLOCK_K, BLOCK_O)
            # We need to compute acc += x @ w.T (since w is shape (BLOCK_O, BLOCK_K) we want w^T shape (BLOCK_K, BLOCK_O)

            acc += tl.dot(x, w) # tl.dot expects shape (M,K) @ (K,N) = (M,N). That's fine with (BLOCK_M,BLOCK_K) and (BLOCK_K,BLOCK_O)
        # after K loops, compute mean/inv_std per row using sum_x, sum_x2
        mean = sum_x / dim
        var = sum_x2 / dim - mean * mean
        inv_std = 1.0 / tl.sqrt(var + EPS)

        # compute bias term and weight_sum per output channel
        # load weight_sum for current O tile
        weight_sum_tile = tl.load(weight_sum_ptr + offs_o,
                                  mask=mask_o,
                                  other=0.0)  # shape (BLOCK_O)
        bias_tile = tl.load(bias_ptr + offs_o,
                            mask=mask_o,
                            other=0.0)  # shape (BLOCK_O)

        # Since mean shape (BLOCK_M,), weight_sum_tile shape (BLOCK_O,)

        # store to y
        y_ptrs = y_ptr + rows[:, None] * stride_y_O + offs_o[None, :] * 1 # Actually stride_y_O is stride w.r.t output dim = total_O? We will compute pointer across output dimension.
        tl.store(y_ptrs,
                 acc.to(tl.float16),
                 mask=row_mask[:, None] & mask_o[None, :])

        o_tile_start += BLOCK_O

@triton.jit
def mask_gate_apply_kernel(
    proj_ptr,          # *fp16 [M, H] maybe shape flattened?
    gate_ptr,          # *fp16 same shape
    mask_ptr,          # *fp16 [M] flattened mask? Actually mask shape is (M,) after flatten.
    out_ptr,
    M, H,
    stride_proj_m, stride_proj_h,
    stride_gate_m, stride_gate_h,
    stride_mask_m,
    stride_out_m, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)
    row_mask = rows < M

    # Load mask tile (scalar per row)
    mask_tile = tl.load(mask_ptr + rows * stride_mask_m, mask=row_mask, other=0.0)  # shape (BLOCK_M,)

    # Loop over hidden dimension tiles
    off_h = 0
    while off_h < H:
        offs_h = off_h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # load proj, gate
        proj = tl.load(proj_ptr + rows[:, None] * stride_proj_m + offs_h[None, :] * stride_proj_h,
                       mask=row_mask[:, None] & mask_h[None, :], other=0.0)
        gate = tl.load(gate_ptr + rows[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
                       mask=row_mask[:, None] & mask_h[None, :], other=0.0)
        # compute sigmoid on gate
        gate_sig = 1.0 / (1.0 + tl.exp(-gate))
        # multiply
        out = proj * gate_sig * mask_tile[:, None]

        # store out
        tl.store(out_ptr + rows[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
                out,
                mask=row_mask[:, None] & mask_h[None, :])
        off_h += BLOCK_H

"""
TriMul (outgoing) – highly fused forward implementation.

Key ideas
---------
1️⃣ Fuse the first LayerNorm with the five independent Linear layers
   (left_proj, right_proj, left_gate, right_gate, out_gate) into a *single*
   linear projection.  The LayerNorm scaling (γ) is merged into the weights,
   and the LayerNorm bias (β) is merged into a bias term for each output
   channel.  This reduces the number of full‑tensor passes from 6 (norm + five
   matmuls) to 2 (norm + one matmul).

2️⃣ Apply the mask and the sigmoid gates together in a tiny Triton kernel.
   The kernel loads `proj`, `gate_raw` and `mask`, computes `gate = sigmoid(gate_raw)`
   and writes `proj * gate * mask` (or `proj * gate` when there is no mask).

3️⃣ The expensive O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (`torch.bmm`) after reshaping
   `[B,N,N,H] → [B·H,N,N]`.  This lets cuBLAS tensor‑core kernels handle the
   heavy work efficiently.

4️⃣ The second LayerNorm, the out‑gate multiplication and the final
   Linear projection are left to PyTorch – they are cheap compared to the
   contraction and already use highly‑optimised kernels.

The function `custom_kernel` follows the exactly same API as the reference
module and returns a `float32` tensor.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: mask * sigmoid(gate) * proj
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [M, H]
    gate_ptr,          # *fp16  [M, H]  (raw, before sigmoid)
    mask_ptr,          # *fp16  [M]     (broadcasted over H)
    out_ptr,           # *fp16  [M, H]
    M, H,
    stride_proj_m, stride_proj_h,
    stride_gate_m, stride_gate_h,
    stride_mask_m,
    stride_out_m, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute out = proj * sigmoid(gate) * mask (mask broadcast over H)."""
    pid = tl.program_id(0)

    # tile of rows
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_mask = offs_m < M

    # load mask (scalar per row) and broadcast
    mask = tl.load(mask_ptr + offs_m * stride_mask_m,
                   mask=row_mask,
                   other=0.0)                     # fp16, shape [BLOCK_M]

    # iterate over hidden dimension tiles
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        col_mask = offs_h < H

        # load proj & gate
        proj = tl.load(
            proj_ptr
            + offs_m[:, None] * stride_proj_m
            + offs_h[None, :] * stride_proj_h,
            mask=row_mask[:, None] & col_mask[None, :],
            other=0.0,
        )
        gate = tl.load(
            gate_ptr
            + offs_m[:, None] * stride_gate_m
            + offs_h[None, :] * stride_gate_h,
            mask=row_mask[:, None] & col_mask[None, :],
            other=0.0,
        )
        # sigmoid(gate)
        sig_gate = 1.0 / (1.0 + tl.exp(-gate))
        out = proj * sig_gate * mask[:, None]

        # store
        tl.store(
            out_ptr
            + offs_m[:, None] * stride_out_m
            + offs_h[None, :] * stride_out_h,
            out,
            mask=row_mask[:, None] & col_mask[None, :],
        )


def _apply_mask_gate(proj: torch.Tensor,
                     gate_raw: torch.Tensor,
                     mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse mask and sigmoid gate:
        out = proj * sigmoid(gate_raw) * mask
    All tensors must be on the same device and have dtype torch.float16.
    """
    M, H = proj.shape
    out = torch.empty_like(proj)

    # strides expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate_raw.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes – chosen to fit registers / shared memory on H100
    BLOCK_M = 64
    BLOCK_H = 8

    grid = ( (M + BLOCK_M - 1) // BLOCK_M, )
    _mask_gate_kernel[grid](
        proj,
        gate_raw,
        mask,
        out,
        M,
        H,
        s_proj[0],
        s_proj[1],
        s_gate[0],
        s_gate[1],
        s_mask[0],
        s_out[0],
        s_out[1],
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using cuBLAS.
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns out  : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul C = A @ Bᵀ ; we need B transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # keep the normalized tensor in fp16

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the five Linear layers (proj / gate) into ONE matmul.
    #    Weight matrices are pre‑scaled by the LayerNorm γ and the bias
    #    contributed by the LayerNorm β is added as a bias term.
    # -----------------------------------------------------------------
    # Gather the original (float32) weight matrices
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Stack them → shape [5*H, dim]
    combined_w_original = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # (5*hidden_dim, dim)  float32

    # ----- incorporate the first LayerNorm (γ, β) -----
    # γ (norm_weight) scales every column of the weight matrix
    weight_adj = combined_w_original * norm_weight.unsqueeze(0)          # (5H, dim)
    # β (norm_bias) contributes a bias per output channel
    bias_adj = (norm_bias.unsqueeze(0) * combined_w_original).sum(dim=1)  # (5H,)

    # Cast to fp16 for the matmul
    weight_adj_fp16 = weight_adj.to(dtype=dtype_fp16, device=device)
    bias_adj_fp16 = bias_adj.to(dtype=dtype_fp16, device=device)

    # Flatten the normalized tensor to (M, dim) where M = B·N·N
    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.reshape(M, dim).contiguous()

    # Single fused linear pass → shape (M, 5*H)
    combined_out_flat = torch.nn.functional.linear(
        x_flat, weight_adj_fp16, bias=bias_adj_fp16
    )  # fp16

    # Reshape back to [B, N, N, 5*H]
    combined_out = combined_out_flat.view(B, N, N, 5 * hidden_dim)

    # -----------------------------------------------------------------
    # 3️⃣ Split into the five tensors we need
    # -----------------------------------------------------------------
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply mask and sigmoid‑gates (left/right) – Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        # mask is broadcast over H; make it fp16 for the kernel
        mask_fp16 = mask_tensor.to(dtype=dtype_fp16)
        # fuse: proj * sigmoid(gate_raw) * mask
        left = _apply_mask_gate(left_proj, left_gate_raw, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate_raw, mask_fp16)
    else:
        # no mask → just multiply by sigmoid(gate_raw)
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # out_gate used later (after the contraction)
    out_gate = torch.sigmoid(out_gate_raw)  # fp16, shape [B,N,N,H]

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS tensor cores)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) → out‑gate multiplication → final Linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # fp16

    # Apply the out‑gate (elementwise)
    out = out * out_gate

    # Final projection back to the model dimension
    out = torch.nn.functional.linear(
        out, to_out_weight.to(device=device, dtype=dtype_fp16)
    )  # fp16, shape [B,N,N,dim]

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
