@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,  # left
    b_ptr,  # right
    c_ptr,  # output
    BATCH, M, N, K,
    stride_aba, stride_abm, stride_abk,
    stride_bba, stride_bbm, stride_bbk,
    stride_cca, stride_ccm, stride_ccn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)  # batch dimension
    pid_m = tl.program_id(1)  # block row index
    pid_n = tl.program_id(2)  # block column index

    # offsets within current batch
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in float32 for accuracy

    # pointer offsets for a and b
    # a shape: [batch, M, K]
    # b shape: [batch, N, K] (we treat b as original right shape; we use its transpose in multiplication)
    # We will load b elements as [k, n] because we need B_T.

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute block offsets for a and b
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)  # shape [BLOCK_K]

        # compute mask for k dimension
        mask_k = offs_k < K

        # load A block: shape [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + batch_idx * stride_aba + offs_m[:, None] * stride_abm + offs_k[None, :] * stride_abk
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B block transposed: shape [BLOCK_K, BLOCK_N]
        # B has shape [batch, N, K]; we need B_T which has shape [batch, K, N]
        # So offset: batch offset + offs_k[:, None]*stride_bbk + offs_n[None, :]*stride_bbn

        b_ptrs = b_ptr + batch_idx * stride_bba + offs_k[:, None] * stride_bbk + offs_n[None, :] * stride_bbn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate acc += a @ b
        acc += tl.dot(a, b)

    # store result: C shape [batch, M, N]
    c_ptrs = c_ptr + batch_idx * stride_cca + offs_m[:, None] * stride_ccm + offs_n[None, :] * stride_ccn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)

    # boundaries
    mask_m = off_m < M
    mask_n = off_n < N
    mask_k = off_k < K

    # compute block start pointers
    a_batch_offset = batch_idx * stride_a_batch
    b_batch_offset = batch_idx * stride_b_batch
    c_batch_offset = batch_idx * stride_c_batch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k range within block
        k_off = k * BLOCK_K
        off_k = k_off + tl.arange(0, BLOCK_K)
        cur_mask_k = off_k < K

        # load A [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + a_batch_offset + off_m[:, None] * stride_a_m + off_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)

        # load B_T [BLOCK_K, BLOCK_N] (transposed)
        # B has layout [batch,N,K] but we need B_T's (K,N). So we load using k dimension index for rows and n dimension for columns.
        b_ptrs = b_ptr + b_batch_offset + off_k[:, None] * stride_b_k + off_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=cur_mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # store output
    c_ptrs = c_ptr + c_batch_offset + off_m[:, None] * stride_c_m + off_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def batched_gemm_triton(left, right):
    B, N, _, H = left.shape
    # Permute to [B,H,N,N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    batch_size = B * H
    a = left_perm.view(batch_size, N, N)
    b = right_perm.view(batch_size, N, N)
    c = torch.empty_like(a)
    # get strides
    stride_ab, stride_am, stride_ak = a.stride()
    stride_bb, stride_bn, stride_bk = b.stride()
    stride_cb, stride_cm, stride_cn = c.stride()
    # cell constants
    BLOCK_M = 128; BLOCK_N = 128; BLOCK_K = 32
    grid_x = batch_size
    grid_y = (N + BLOCK_M - 1) // BLOCK_M
    grid_z = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_x, grid_y, grid_z)
    tri_mul_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        batch=batch_size,
        M=N,
        N_=N,
        K=N,
        stride_a_batch=stride_ab,
        stride_a_m=stride_am,
        stride_a_k=stride_ak,
        stride_b_batch=stride_bb,
        stride_b_n=stride_bn,
        stride_b_k=stride_bk,
        stride_c_batch=stride_cb,
        stride_c_m=stride_cm,
        stride_c_n=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # reshape back
    out = c.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

"""
Optimized outgoing TriMul forward (AlphaFold‑3) using fused linear projections
and Triton kernels.

Algorithm
---------
1️⃣  Apply the first LayerNorm (FP32) to the input.
2️⃣  Cast to FP16 for the heavy portion.
3️⃣  Fuse the five linear projections (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single GEMM (weights are concatenated).
4️⃣  Split the fused result, apply sigmoid to the three gate tensors.
5️⃣  Fuse element‑wise gating and optional masking in a Triton kernel
    (`mask_gate_fuse_kernel`).  This avoids multiple memory passes.
6️⃣  Compute the O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM using a custom Triton kernel
    (`tri_mul_gemm_kernel`).  The kernel works on the flattened batch
    dimension B·H and handles the transposition of the right matrix.
7️⃣  Second LayerNorm (FP16) over the hidden dimension.
8️⃣  Multiply by the output gate and apply the final linear projection
    back to the original dimension.
9️⃣  Cast to FP32 (required by the harness) and return.

Only the forward pass is implemented – no autograd support is needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse element‑wise multiplication with gating and mask.
# `x` and `gate` have shape (B·N·N·H), `mask` has shape (B·N·N).
# The hidden dimension `H` is the innermost stride of `x` & `gate`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (flattened B·N·N)
    hidden_dim,     # i32   : size of hidden dimension H
    size,           # i32   : total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear indices for this block
    active = offs < size                                 # guard for OOB

    # map element index to mask index (mask is broadcast over H)
    mask_offs = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)  # outside range => 1.0 (no masking)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul contraction.
# Computes C = A @ Bᵀ  where A,B are (batch, N, N) fp16 tensors.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # *fp16
    batch, M, N, K,                           # i32
    stride_a_batch, stride_a_m, stride_a_k,   # i32
    stride_b_batch, stride_b_n, stride_b_k,   # i32   (B is treated as [batch, N, K])
    stride_c_batch, stride_c_m, stride_c_n,   # i32
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute starting pointers for this batch
    a_batch_offset = batch_id * stride_a_batch
    b_batch_offset = batch_id * stride_b_batch
    c_batch_offset = batch_id * stride_c_batch

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the reduction axis)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # ---- load A block (M × K) ----
        a_ptrs = a_ptr + a_batch_offset \
                 + offs_m[:, None] * stride_a_m \
                 + k_off[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # ---- load Bᵀ block (K × N) ----
        # Original B layout is [batch, N, K]; we load its transpose.
        b_ptrs = b_ptr + b_batch_offset \
                 + k_off[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # store result C = A @ Bᵀ  (FP16)
    c_ptrs = c_ptr + c_batch_offset \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul C = left @ rightᵀ using Triton.
    Input tensors: [B, N, N, H] (fp16).
    Output: same shape.
    """
    B, N, _, H = left.shape
    # Move hidden dim into the batch dimension and make the rows/cols contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch = B * H
    a = left_perm.view(batch, N, N)   # [batch, M, K]
    b = right_perm.view(batch, N, N)  # [batch, N, K]   (we will use its transpose)

    c = torch.empty_like(a)           # output buffer, fp16

    # strides (in element units, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_n, stride_b_k = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Block sizes – chosen to give high occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        batch=batch,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_n=stride_b_n,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into one GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices (out_features, in_features)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]

    # Single linear projection
    fused_out = F.linear(x, fused_weight)   # [B, N, N, 5*H] (fp16)

    # Split into the individual tensors
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused_out, hidden_dim, dim=-1
    )

    # Gates need a sigmoid
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – fuse gating + mask with a Triton kernel
    # ------------------------------------------------------------------
    # Prepare mask (broadcast to hidden dimension).  If no mask, use all‑ones.
    if nomask or mask_tensor is None:
        mask_fp16 = torch.ones_like(input_tensor[..., 0], dtype=torch.float16, device=device)
    else:
        # Ensure mask has shape [B, N, N] and dtype fp16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()

    # Fuse left = left_proj * left_gate * mask
    total_elems = left_proj.numel()
    BLOCK = 1024                     # ≤ max threads per block
    grid_mask = ((total_elems + BLOCK - 1) // BLOCK,)

    mask_gate_fuse_kernel[grid_mask](
        left_proj,
        left_gate,
        mask_fp16,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # Fuse right = right_proj * right_gate * mask
    mask_gate_fuse_kernel[grid_mask](
        right_proj,
        right_gate,
        mask_fp16,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # After the kernel, `left_proj` and `right_proj` now contain the gated & masked tensors
    left = left_proj
    right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction using a custom Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_gemm_triton(left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16) over the hidden dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final gating and projection back to original dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    out = F.linear(out, to_out_w)                         # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 (required by the harness) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
