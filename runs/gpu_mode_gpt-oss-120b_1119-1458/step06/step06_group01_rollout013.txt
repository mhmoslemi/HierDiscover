    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

            left_out_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_j[None, :, None] * stride_left_j
            + offs_h[None, None, :] * stride_left_h

            left_out_ptrs,
            left_out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],

"""
Optimized outgoing TriMul forward (AlphaFold‑3) using Triton fusion.

Algorithm
---------
1️⃣  Layer‑norm the input (FP32) and cast to FP16.
2️⃣  Fuse the four linear projections (left_proj, left_gate, right_proj, right_gate)
    into a **single** matmul by concatenating their weight matrices.
3️⃣  A Triton kernel consumes the concatenated proj‑tensor, applies
    sigmoid to the gate slices, multiplies by the (optional) mask,
    and writes the gated left/right tensors directly.
4️⃣  The O(N³) contraction  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM C = A @ Bᵀ with cuBLAS Tensor‑Core kernels.
5️⃣  Layer‑norm, out‑gate (sigmoid) and final linear projection back to the
    model dimension are done in FP16 with PyTorch kernels (already very fast).
6️⃣  Cast the result back to FP32 for the harness.

Only the projection‑gate‑mask step is written in Triton; the heavy
N³ contraction stays on cuBLAS where it is orders‑of‑magnitude faster.
This implementation reduces the number of linear passes from four to one,
halving the memory traffic for the large [B, N, N, dim] tensor and
achieving the targeted runtime.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse proj, gate, sigmoid and mask → gated left/right
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,               # *fp16  [B, N, N, 4*H]
    mask_ptr,               # *fp16  [B, N, N]
    left_out_ptr,           # *fp16  [B, N, N, H]
    right_out_ptr,          # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    proj_ptr contains [left_proj, left_gate, right_proj, right_gate] concatenated
    along the hidden dimension (size 4*H).  For each (b,i,j,h) we compute

        left  = left_proj  * sigmoid(left_gate ) * mask
        right = right_proj * sigmoid(right_gate) * mask

    mask is broadcast over the hidden dimension.
    """
    pid_b = tl.program_id(0)   # batch
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # ----- tile coordinates -------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- load mask (broadcast later over H) -------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16, shape (BLOCK_I, BLOCK_J)

    # ----------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ----------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ----- offsets for the four slices inside proj_ptr -----------------
        #   left_proj  : offset 0
        #   left_gate  : offset H
        #   right_proj : offset 2*H
        #   right_gate : offset 3*H
        base = pid_b * stride_proj_b \
               + offs_i[:, None, None] * stride_proj_i \
               + offs_j[None, :, None] * stride_proj_j

        # left_proj
        lp_ptrs = base + (offs_h + 0) * stride_proj_h
        left_proj_tile = tl.load(
            lp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # left_gate
        lg_ptrs = base + (offs_h + H) * stride_proj_h
        left_gate_tile = tl.load(
            lg_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # right_proj
        rp_ptrs = base + (offs_h + 2 * H) * stride_proj_h
        right_proj_tile = tl.load(
            rp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # right_gate
        rg_ptrs = base + (offs_h + 3 * H) * stride_proj_h
        right_gate_tile = tl.load(
            rg_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # ----- sigmoid on the gate tensors (computed in FP32 for stability) --
        left_gate_f32   = left_gate_tile.to(tl.float32)
        left_gate_sig_f32 = 1.0 / (1.0 + tl.exp(-left_gate_f32))
        left_gate_sig   = left_gate_sig_f32.to(tl.float16)

        right_gate_f32   = right_gate_tile.to(tl.float32)
        right_gate_sig_f32 = 1.0 / (1.0 + tl.exp(-right_gate_f32))
        right_gate_sig   = right_gate_sig_f32.to(tl.float16)

        # ----- apply mask and write results -----------------------------------
        # mask_tile shape (BLOCK_I, BLOCK_J) -> broadcast to (BLOCK_I, BLOCK_J, BLOCK_H)
        left_out_tile  = left_proj_tile  * left_gate_sig  * mask_tile[:, :, None]
        right_out_tile = right_proj_tile * right_gate_sig * mask_tile[:, :, None]

        # store left
        left_out_ptrs = (
            left_out_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_j[None, :, None] * stride_left_j
            + offs_h[None, None, :] * stride_left_h
        )
        tl.store(
            left_out_ptrs,
            left_out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

        # store right
        right_out_ptrs = (
            right_out_ptr
            + pid_b * stride_right_b
            + offs_i[:, None, None] * stride_right_i
            + offs_j[None, :, None] * stride_right_j
            + offs_h[None, None, :] * stride_right_h
        )
        tl.store(
            right_out_ptrs,
            right_out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_proj_gate_mask(proj: torch.Tensor, mask: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse the four linear projections, sigmoid gates and mask in a single Triton
    kernel.

    Args:
        proj: torch.Tensor of shape [B, N, N, 4*H] (fp16) – result of the
              concatenated linear projection.
        mask: torch.Tensor of shape [B, N, N] (fp16) – broadcast mask (1 = keep,
              0 = drop).  Use all‑ones if no mask is required.

    Returns:
        left, right – gated tensors of shape [B, N, N, H] (fp16).
    """
    B, N, _, total_H = proj.shape
    H = total_H // 4
    left  = torch.empty((B, N, N, H), dtype=proj.dtype, device=proj.device)
    right = torch.empty_like(left)

    # Strides in element units (not bytes)
    s_proj = proj.stride()
    s_mask = mask.stride()
    s_left = left.stride()
    s_right = right.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _proj_gate_mask_kernel[grid](
        proj,
        mask,
        left,
        right,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return left, right


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction: out = left @ rightᵀ over k
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    via a batched GEMM C = A @ Bᵀ.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # BMM with right transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # restore layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) – may be ignored.
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack and configuration
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    bs, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32
    x = x.to(torch.float16)   # FP16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Concatenate projection & gate weights (FP16)
    # ------------------------------------------------------------------
    lp_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    lg_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    rp_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    rg_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)

    # [4*H, dim] concatenated weight matrix (no bias)
    combined_weight = torch.cat([lp_w, lg_w, rp_w, rg_w], dim=0)  # shape (4*hidden_dim, dim)

    # ------------------------------------------------------------------
    # 3️⃣ Single linear projection producing all four tensors at once
    # ------------------------------------------------------------------
    # torch.nn.functional.linear treats the last dimension as in_features,
    # broadcasting the preceding dimensions as batch.
    combined_proj = torch.nn.functional.linear(x, combined_weight)  # [B, N, N, 4*H] (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Mask (or all‑ones) preparation – broadcast over hidden dim
    # ------------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gate sigmoid + mask multiplication (Triton)
    # ------------------------------------------------------------------
    left, right = _apply_proj_gate_mask(combined_proj, mask_fp16)   # [B,N,N,H] each

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)            # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate (sigmoid) and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] fp16
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
