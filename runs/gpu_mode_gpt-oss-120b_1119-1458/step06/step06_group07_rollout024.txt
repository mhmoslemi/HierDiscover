@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch = pid_batch // H
    h = pid_batch - batch * H  # or pid_batch % H

    # Offsets within matrix
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    valid_i = offs_i < N
    valid_j = offs_j < N

    # base pointers
    left_base = left_ptr + batch * stride_l_batch + h * stride_l_h
    right_base = right_ptr + batch * stride_r_batch + h * stride_r_h
    out_base = out_ptr + batch * stride_o_batch + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        valid_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=valid_i[:, None] & valid_k[None,:], other=0.0).to(tl.float32)

        # right tile transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs, mask=valid_k[:, None] & valid_j[None,:], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # store output
    c_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=valid_i[:, None] & valid_j[None, :])

"""
TriMul (outgoing) forward pass – fused and accelerated with Triton.

Algorithm
---------
1. LayerNorm the 4‑D input (float32) → cast to fp16.
2. Fuse the two linear projections (proj + gate) for “left” and “right”
   by stacking the weight matrices → one matmul each.
3. Optional mask is fused with projection + gate by a tiny Triton kernel.
4. Heavy O(N³) contraction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a custom batched GEMM kernel that works directly on the
   original [B,N,N,H] layout – no permutation / copy needed.
5. Second LayerNorm (fp16) over the hidden dimension, element‑wise gating
   with `out_gate`, and final linear projection back to `dim` using
   torch.nn.functional.linear (cuBLAS).
6. Cast the result to float32 (expected by the harness).

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1 – fuse mask, gate and tensor multiplication (used when a mask
# is present). Works on a flattened view of the tensor [B*N*N*H].
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,                # *fp16  tensor to be updated (flattened)
    gate_ptr,             # *fp16  gate tensor (same shape as x)
    mask_ptr,             # *fp16  mask tensor (flattened, shape [B*N*N])
    hidden_dim,           # i32   hidden dimension (H)
    size,                 # i32   total number of elements in `x`
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the three operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Each mask entry corresponds to `hidden_dim` contiguous elements in x
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fuse multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2 – batched GEMM for the O(N³) contraction.
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# The tensors are stored in the original layout [B,N,N,H]; no copy needed.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,    # *fp16  left tensor,   shape [B,N,N,H]
    right_ptr,   # *fp16  right tensor,  shape [B,N,N,H]
    out_ptr,     # *fp16  output tensor, shape [B,N,N,H]
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_hb = tl.program_id(0)          # combined batch * hidden index
    pid_i  = tl.program_id(1)          # tile over i dimension
    pid_j  = tl.program_id(2)          # tile over j dimension

    # Decode batch and hidden‑dim index
    batch = pid_hb // H
    hid   = pid_hb - batch * H          # pid_hb % H

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    valid_i = offs_i < N
    valid_j = offs_j < N

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_l_batch  + hid * stride_l_h
    right_base = right_ptr + batch * stride_r_batch + hid * stride_r_h
    out_base   = out_ptr   + batch * stride_o_batch + hid * stride_o_h

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        valid_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=valid_i[:, None] & valid_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        # right[j,k] → we need rows = k, cols = j
        b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=valid_k[:, None] & valid_j[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write the result back
    c_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=valid_i[:, None] & valid_j[None, :])


# ----------------------------------------------------------------------
# Entry point – performs the complete outgoing TriMul forward pass.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask_tensor  : [B, N, N]        (float32 or bool) – may be None
        - weights      : dict of model weight tensors (float32)
        - config       : dict containing "dim", "hidden_dim", "nomask"

    Returns
    -------
    output : torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16 for compute‑heavy part
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] (float32)

    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Convert all required weights to fp16 once
    # -----------------------------------------------------------------
    # Stacked linear weights for left and right (proj + gate)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    left_stack_w  = torch.cat([left_proj_w, left_gate_w], dim=0)    # (2*H, dim)
    right_stack_w = torch.cat([right_proj_w, right_gate_w], dim=0) # (2*H, dim)

    # Output gate (scalar per hidden unit)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Second LayerNorm parameters (hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # Final projection (hidden_dim → dim)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections + gating (two fused matmuls)
    # -----------------------------------------------------------------
    # left
    left_concat = torch.nn.functional.linear(x, left_stack_w)            # [B,N,N,2*H]
    left_proj = left_concat[..., :hidden_dim]                            # [B,N,N,H]
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])            # [B,N,N,H]

    # right
    right_concat = torch.nn.functional.linear(x, right_stack_w)          # [B,N,N,2*H]
    right_proj = right_concat[..., :hidden_dim]                          # [B,N,N,H]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])          # [B,N,N,H]

    # out gate (used later)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w)) # [B,N,N,H]

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                       # shape [B*N*N]

        # left = left_proj * left_gate * mask
        size = left_proj.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        size_r = right_proj.numel()
        grid_r = (triton.cdiv(size_r, BLOCK),)
        mask_gate_fuse_kernel[grid_r](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            size_r,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction using a custom Triton GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left_proj.shape
    # allocate output tensor for the contraction
    out = torch.empty_like(left_proj)

    # Stride information (in elements, not bytes)
    l_s_batch, l_s_i, l_s_k, l_s_h = left_proj.stride()
    r_s_batch, r_s_i, r_s_k, r_s_h = right_proj.stride()
    o_s_batch, o_s_i, o_s_j, o_s_h = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                           # batch * hidden‑dim dimension
        (N + BLOCK_M - 1) // BLOCK_M,             # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,             # tiles over j
    )
    tri_mul_gemm_kernel[grid](
        left_proj,
        right_proj,
        out,
        l_s_batch, l_s_i, l_s_k, l_s_h,
        r_s_batch, r_s_i, r_s_k, r_s_h,
        o_s_batch, o_s_i, o_s_j, o_s_h,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16) over the hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Gating with out_gate and final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] (fp16)

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
