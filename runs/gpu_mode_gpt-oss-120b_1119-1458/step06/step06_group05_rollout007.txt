@triton.jit
def _gate_mask_combined_mask_kernel(comb_ptr, mask_ptr, out_ptr,
                                   B, N, H,
                                   stride_comb_b, stride_comb_i, stride_comb_j, stride_comb_h,
                                   stride_mask_b, stride_mask_i, stride_mask_j,
                                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                                   BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    # compute for tile
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # Load mask tile
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)  # fp16
    
    # loop over hidden tiles
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        
        # compute pointer offsets for proj part
        proj_ptrs = (comb_ptr + pid_b * stride_comb_b +
                     offs_i[:, None, None] * stride_comb_i +
                     offs_j[None, :, None] * stride_comb_j +
                     offs_h[None, None, :] * stride_comb_h)
        # proj_tile and gate_raw_tile are contiguous in combined dimension where projection part is first H entries.
        # Since combined dimension size is 2*H. We'll load as two separate loads offset with stride h.
        proj_tile = tl.load(proj_ptrs,
                             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None,None,:],
                             other=0.0)
        # Gate raw tile at offset H
        gate_ptrs = proj_ptrs + H * stride_comb_h  # offset for gate raw start
        gate_raw_tile = tl.load(gate_ptrs,
                                mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None,None,:],
                                other=0.0)
        # Apply sigmoid to gate raw
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_raw_tile))
        # Compute out tile: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]
        
        # Store out
        out_ptrs = (out_ptr + pid_b * stride_out_b +
                    offs_i[:, None, None] * stride_out_i +
                    offs_j[None, :, None] * stride_out_j +
                    offs_h[None, None, :] * stride_out_h)
        tl.store(out_ptrs, out_tile,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None,None,:])

"""
TriMul (outgoing) – fast Triton‑plus‑PyTorch forward.

Algorithm
---------
1️⃣ First LayerNorm (FP32) → cast to fp16.
2️⃣ Fuse projection+gate linear layers for *left* and *right*:
   - concatenate (proj, gate) weights → a single matmul produces a tensor
     `comb` of shape [B, N, N, 2·H] (proj || gate_raw).
3️⃣ Triton kernel fuses:
   - sigmoid on raw‑gate,
   - element‑wise mask (if provided) and
   - projection‑gate multiplication → final tensors `left` and `right`
     of shape [B, N, N, H] (fp16).  This replaces three separate
     element‑wise kernels.
4️⃣ Heavy O(N³) contraction realized as a batched GEMM:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   We flatten batch‑and‑hidden dimensions and call `torch.bmm`.
5️⃣ Second LayerNorm (fp16) → out‑gate (sigmoid) → final linear projection.
6️⃣ Cast result back to float32 (the harness expects FP32).

Only the mask‑gate‑sigmoid fusion is written in Triton; all matmul‑heavy
operations use cuBLAS tensor‑core kernels.  The implementation works for
all test configurations (with/without mask, various sequence lengths,
and both normal / Cauchy input distributions).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, sigmoid‑gate and optional mask.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_combined_mask_kernel(
    comb_ptr,               # *fp16  [B, N, N, 2*H]
    mask_ptr,               # *fp16  [B, N, N]
    out_ptr,                # *fp16  [B, N, N, H]
    B, N, H,
    stride_comb_b, stride_comb_i, stride_comb_j, stride_comb_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # coordinates inside the tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (broadcast later over the hidden dimension)
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Pointers to the projection part (first H entries)
        proj_ptrs = (
            comb_ptr
            + pid_b * stride_comb_b
            + offs_i[:, None, None] * stride_comb_i
            + offs_j[None, :, None] * stride_comb_j
            + offs_h[None, None, :] * stride_comb_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16, shape [I,J,H]

        # Pointers to the raw‑gate part (second H entries)
        gate_raw_ptrs = proj_ptrs + H * stride_comb_h
        gate_raw_tile = tl.load(
            gate_raw_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # sigmoid
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_raw_tile))

        # Fuse: proj * sigmoid(gate) * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


@triton.jit
def _gate_mask_combined_nomask_kernel(
    comb_ptr,               # *fp16  [B, N, N, 2*H]
    out_ptr,                # *fp16  [B, N, N, H]
    B, N, H,
    stride_comb_b, stride_comb_i, stride_comb_j, stride_comb_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        proj_ptrs = (
            comb_ptr
            + pid_b * stride_comb_b
            + offs_i[:, None, None] * stride_comb_i
            + offs_j[None, :, None] * stride_comb_j
            + offs_h[None, None, :] * stride_comb_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        gate_raw_ptrs = proj_ptrs + H * stride_comb_h
        gate_raw_tile = tl.load(
            gate_raw_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_raw_tile))
        out_tile = proj_tile * gate_tile

        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: apply gate+mask to the concatenated projection tensor.
# ----------------------------------------------------------------------
def _apply_gate_and_mask(comb: torch.Tensor, mask: torch.Tensor | None) -> torch.Tensor:
    """
    comb : [B, N, N, 2*H]  (fp16)
    mask : [B, N, N] or None  (fp16 if provided)
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, twoh = comb.shape
    H = twoh // 2
    out = torch.empty(B, N, N, H, device=comb.device, dtype=comb.dtype)

    s_comb = comb.stride()          # (b,i,j,2H)
    s_out = out.stride()            # (b,i,j,H)

    # --- tile sizes (tuned for H100, fp16) ---
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 32   # works well up to H=384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16)
        s_mask = mask_fp16.stride()
        _gate_mask_combined_mask_kernel[grid](
            comb,
            mask_fp16,
            out,
            B,
            N,
            H,
            s_comb[0],
            s_comb[1],
            s_comb[2],
            s_comb[3],
            s_mask[0],
            s_mask[1],
            s_mask[2],
            s_out[0],
            s_out[1],
            s_out[2],
            s_out[3],
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_H=BLOCK_H,
            num_warps=8,
            num_stages=2,
        )
    else:
        _gate_mask_combined_nomask_kernel[grid](
            comb,
            out,
            B,
            N,
            H,
            s_comb[0],
            s_comb[1],
            s_comb[2],
            s_comb[3],
            s_out[0],
            s_out[1],
            s_out[2],
            s_out[3],
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_H=BLOCK_H,
            num_warps=8,
            num_stages=2,
        )

    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (fp16, contiguous)
    Returns out  : [B, N, N, H]   (fp16)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM: C = A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float32) – ignored if config["nomask"] is True
        - weights_dict : string → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ---------------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ---------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16
    # ---------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # now FP16 for everything that follows

    # ---------------------------------------------------------------
    # 2️⃣ Prepare concatenated projection+gate weights (FP16)
    # ---------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]      # (H, dim)
    left_gate_w   = weights["left_gate.weight"]      # (H, dim)
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # concat projection and gate weights (2*H, dim)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=dtype_fp16)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=dtype_fp16)

    # remaining linear weights (already FP16)
    out_gate_w = out_gate_w.to(device=device, dtype=dtype_fp16)
    to_out_norm_w = to_out_norm_w.to(device=device, dtype=dtype_fp16)
    to_out_norm_b = to_out_norm_b.to(device=device, dtype=dtype_fp16)
    to_out_w = to_out_w.to(device=device, dtype=dtype_fp16)

    # ---------------------------------------------------------------
    # 3️⃣ Combined projection+gate (single matmul each side)
    # ---------------------------------------------------------------
    # left_comb: [B,N,N,2*H]
    left_comb = torch.nn.functional.linear(x, left_comb_w)   # bf16 matmul uses cuBLAS tensor cores
    right_comb = torch.nn.functional.linear(x, right_comb_w)

    # ---------------------------------------------------------------
    # 4️⃣ Fuse sigmoid‑gate, mask (if any) → final left/right tensors
    # ---------------------------------------------------------------
    if not nomask:
        left = _apply_gate_and_mask(left_comb, mask_tensor)
        right = _apply_gate_and_mask(right_comb, mask_tensor)
    else:
        left = _apply_gate_and_mask(left_comb, None)
        right = _apply_gate_and_mask(right_comb, None)

    # ---------------------------------------------------------------
    # 5️⃣ Compute out‑gate (sigmoid) – still FP16
    # ---------------------------------------------------------------
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)  # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(out_gate_raw)                    # fp16

    # ---------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (batched GEMM)
    # ---------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)       # [B,N,N,hidden_dim] fp16

    # ---------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ---------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final linear projection back to `dim`
    # ---------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)          # [B,N,N,dim] fp16

    # ---------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (expected by the harness)
    # ---------------------------------------------------------------
    return out.to(torch.float32)
