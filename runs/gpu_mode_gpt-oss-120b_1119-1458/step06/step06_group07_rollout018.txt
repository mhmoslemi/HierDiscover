    left_ptr, right_ptr, out_ptr,
    batch_h, M, N, K,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
   pid_b = tl.program_id(2)
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)

   # mapping batch index into original b and h
   hidden_dim = tl.constexpr  # passed via runtime? Or we compute via stride?
   # We'll pass hidden_dim as a kernel argument.
   hidden_dim = H_ARG (kernel constant)
   batch_id = pid_b # combined id
   b = batch_id // hidden_dim
   h = batch_id % hidden_dim

   # compute base offsets
   base_left = left_ptr + b * stride_left_b + h * stride_left_h
   base_right = right_ptr + b * stride_right_b + h * stride_right_h
   base_out = out_ptr + b * stride_out_b + h * stride_out_h

   # compute start indices
   i0 = pid_i * BLOCK_M
   j0 = pid_j * BLOCK_N

   offs_j = j0 + tl.arange(0, BLOCK_N)

   mask_i = offs_i < M
   mask_j = offs_j < N

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation

Algorithm
---------
1️⃣ First LayerNorm on the 4‑D input (float32) – keep the full precision for the
   statistics, then cast to fp16 for the heavy part.
2️⃣ Linear projections to the hidden dimension (fp16): left_proj and right_proj.
3️⃣ Compute three gating tensors (fp16) with a sigmoid.
4️⃣ If a mask is provided fuse ``tensor * gate * mask`` in‑place with a
   Triton kernel (no temporaries, perfect memory coalescing).
   If there is no mask, simply multiply the gates.
5️⃣ Heavy O(N³) contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched fp16 GEMM (A @ Bᵀ) via a custom Triton kernel.
6️⃣ Second LayerNorm (fp16) over the hidden dimension.
7️⃣ Multiply by out_gate (element‑wise) and project back to the original
   channel dimension with a final linear layer.
8️⃣ Cast the result back to float32 (the harness expects float32).

The implementation follows exactly the semantics of the reference PyTorch
module while keeping the numerics stable (LayerNorm in float32, accumulation
in float32 inside the GEMM).  It runs entirely on the GPU, uses only a few
kernel launches and is tuned for the H100 (block sizes are multiples of 16
to map to tensor‑core tiles).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask * gate * tensor (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated (flattened)
    gate_ptr,       # *fp16  gate tensor (same shape as x)
    mask_ptr,       # *fp16  mask tensor (flattened, shape [B*N*N])
    hidden_dim,     # i32   hidden dimension (H)
    size,           # i32   total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # tile size (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load tensors (fp16)
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # One mask value per hidden_dim contiguous entries of x/g.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fuse the three multiplications
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    left_ptr,    # *fp16  left tensor   (B, N, N, H)
    right_ptr,   # *fp16  right tensor  (B, N, N, H)
    out_ptr,     # *fp16  output tensor (B, N, N, H)
    M, N, K,     # i32   dimensions (seq_len)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    hidden_dim,  # i32   hidden dimension (H)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    # 3‑D grid: (block_i, block_j, batch_h) where batch_h = B * H
    pid_i = tl.program_id(0)          # block row index   (i dimension)
    pid_j = tl.program_id(1)          # block col index   (j dimension)
    pid_bh = tl.program_id(2)         # combined batch & hidden index

    # Decode batch and hidden channel from the combined id
    b = pid_bh // hidden_dim          # original batch index 0 .. B-1
    h = pid_bh % hidden_dim           # hidden channel 0 .. H-1

    # Compute the start coordinates of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Ranges for this tile (will be masked at borders)
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Offsets for the batch / channel base
    base_left  = left_ptr  + b * stride_left_b  + h * stride_left_h
    base_right = right_ptr + b * stride_right_b + h * stride_right_h
    base_out   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Masks for the i/j dimensions (broadcastable)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator in float32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ------------------------------------------------------------------
        # Load a tile of A = left[b, i, k, h]
        # A shape in kernel: (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        a_ptrs = base_left + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ------------------------------------------------------------------
        # Load a tile of B = right[b, j, k, h] but interpreted as (k, j)
        # B shape in kernel: (BLOCK_K, BLOCK_N)
        # ------------------------------------------------------------------
        b_ptrs = base_right + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Multiply‑accumulate (fp32 accumulator)
        acc += tl.dot(a, b)

    # ----------------------------------------------------------------------
    # Write the result (convert back to fp16)
    # ----------------------------------------------------------------------
    out_tile = acc.to(tl.float16)
    c_ptrs = base_out + \
             offs_i[:, None] * stride_out_i + \
             offs_j[None, :] * stride_out_j
    tl.store(c_ptrs,
             out_tile,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Args:
        data: tuple containing
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor : [B, N, N]   (float32 or int) – may be unused
            - weights      : dict of model parameters
            - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for large N
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # shape (B, N, N, dim), dtype float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent heavy compute
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)   # (B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (fp16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and contiguous for the Triton kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                    # shape [B*N*N]
        size = left.numel()                               # B * N * N * H

        BLOCK = 1024                                        # fits 1024 threads per block
        grid = (triton.cdiv(size, BLOCK),)

        # Fuse left = left * left_gate * mask   (in place)
        mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_flat.data_ptr(),
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right = right * right_gate * mask (in place)
        mask_gate_fuse_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_flat.data_ptr(),
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just apply the gates (element‑wise, in place)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – custom batched GEMM in Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B, N, N, H), fp16, uninitialized

    # Strides in element units (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tiling parameters – chosen to be multiples of 16 for Tensor‑Core efficiency
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),  # block rows (i)
        triton.cdiv(N, BLOCK_N),  # block cols (j)
        B * hidden_dim,           # batch * hidden channel
    )

    batched_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N, N, N,                            # M, N, K all equal to seq_len
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                        # 8 warps give good occupancy on H100
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                 # element‑wise out‑gate
    out = F.linear(out, to_out_w)        # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
