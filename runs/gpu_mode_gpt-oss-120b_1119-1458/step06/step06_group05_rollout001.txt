@triton.jit
def _mask_gate_fused_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    n_h_tiles = tl.cdiv(H, BLOCK_H)
    # Preload mask tile for each i,j tile.
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)  # shape [I,J]

    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load left projection
        lp_ptrs = (left_proj_ptr + pid_b * stride_lp_b + offs_i[:, None, None] * stride_lp_i + offs_j[None, :, None] * stride_lp_j + offs_h[None, None, :] * stride_lp_h)
        lp_tile = tl.load(lp_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # Load left gate
        lg_ptrs = (left_gate_ptr + pid_b * stride_lg_b + offs_i[:, None, None] * stride_lg_i + offs_j[None, :, None] * stride_lg_j + offs_h[None, None, :] * stride_lg_h)
        lg_tile = tl.load(lg_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # Load right projection
        rp_ptrs = (right_proj_ptr + pid_b * stride_rp_b + offs_i[:, None, None] * stride_rp_i + offs_j[None, :, None] * stride_rp_j + offs_h[None, None, :] * stride_rp_h)
        rp_tile = tl.load(rp_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # Load right gate
        rg_ptrs = (right_gate_ptr + pid_b * stride_rg_b + offs_i[:, None, None] * stride_rg_i + offs_j[None, :, None] * stride_rg_j + offs_h[None, None, :] * stride_rg_h)
        rg_tile = tl.load(rg_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # Compute fused output
        left_tile = lp_tile * lg_tile * mask_tile[:, :, None]
        right_tile = rp_tile * rg_tile * mask_tile[:, :, None]

        # Store left output
        lo_ptrs = (left_out_ptr + pid_b * stride_lo_b + offs_i[:, None, None] * stride_lo_i + offs_j[None, :, None] * stride_lo_j + offs_h[None, None, :] * stride_lo_h)
        tl.store(lo_ptrs, left_tile, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

        # Store right output

   multiplication and the broadcast of the mask over the hidden channel
   dimension in one Triton kernel (`_mask_gate_fused_kernel`).  This kernel
   produces both `left` and `right` tensors in a single memory pass,
   halving the bandwidth required for the mask.
   When there is no mask a simple element‑wise multiplication is used.
4️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (`left @ rightᵀ`) using cuBLAS Tensor‑Core
   kernels (via a single `torch.bmm` on a view of shape `[B*H, N, N]`).
5️⃣ Second LayerNorm (FP16) on the result.
6️⃣ Apply the out‑gate (sigmoid) and a final linear projection back to the
   model dimension.
7️⃣ Cast the final tensor to `float32` (the harness expects FP32).

Only the mask‑gate fusion is written in Triton; the remaining dense
operations rely on highly‑optimized cuBLAS/torch kernels.  The code works
for all provided test configurations (different batch sizes, sequence
lengths, dimensions and optional masks).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuses mask, left‑gate, right‑gate and the two projections.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fused_kernel(
    # input tensors
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    # output tensors
    left_out_ptr, right_out_ptr,
    # sizes
    B, N, H,
    # strides (in number of elements, not bytes)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    # compile‑time block sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """\
    Compute
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
    with mask broadcast over the hidden dimension H.
    The launch grid is (B, ceil(N/BLOCK_I), ceil(N/BLOCK_J)).
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # i / j coordinates for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load the mask tile once (shape [BLOCK_I, BLOCK_J])
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load left projection
        lp_ptrs = (
            left_proj_ptr
            + pid_b * stride_lp_b
            + offs_i[:, None, None] * stride_lp_i
            + offs_j[None, :, None] * stride_lp_j
            + offs_h[None, None, :] * stride_lp_h
        )
        lp_tile = tl.load(
            lp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Load left gate (already sigmoid‑ed)
        lg_ptrs = (
            left_gate_ptr
            + pid_b * stride_lg_b
            + offs_i[:, None, None] * stride_lg_i
            + offs_j[None, :, None] * stride_lg_j
            + offs_h[None, None, :] * stride_lg_h
        )
        lg_tile = tl.load(
            lg_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Load right projection
        rp_ptrs = (
            right_proj_ptr
            + pid_b * stride_rp_b
            + offs_i[:, None, None] * stride_rp_i
            + offs_j[None, :, None] * stride_rp_j
            + offs_h[None, None, :] * stride_rp_h
        )
        rp_tile = tl.load(
            rp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Load right gate (already sigmoid‑ed)
        rg_ptrs = (
            right_gate_ptr
            + pid_b * stride_rg_b
            + offs_i[:, None, None] * stride_rg_i
            + offs_j[None, :, None] * stride_rg_j
            + offs_h[None, None, :] * stride_rg_h
        )
        rg_tile = tl.load(
            rg_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Broadcast mask over H and fuse the three multiplications
        mask_broadcast = mask_tile[:, :, None]  # [I,J,1]
        left_tile   = lp_tile * lg_tile * mask_broadcast
        right_tile  = rp_tile * rg_tile * mask_broadcast

        # Store left output
        lo_ptrs = (
            left_out_ptr
            + pid_b * stride_lo_b
            + offs_i[:, None, None] * stride_lo_i
            + offs_j[None, :, None] * stride_lo_j
            + offs_h[None, None, :] * stride_lo_h
        )
        tl.store(
            lo_ptrs,
            left_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

        # Store right output
        ro_ptrs = (
            right_out_ptr
            + pid_b * stride_ro_b
            + offs_i[:, None, None] * stride_ro_i
            + offs_j[None, :, None] * stride_ro_j
            + offs_h[None, None, :] * stride_ro_h
        )
        tl.store(
            ro_ptrs,
            right_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper that launches the mask‑gate‑fusion kernel.
# ----------------------------------------------------------------------
def _apply_mask_gate_fused(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse `proj * gate * mask` for both the left and right branches.

    All tensors are expected to be fp16 on the same device.
    Returns left, right tensors in fp16.
    """
    B, N, _, H = left_proj.shape
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # strides expressed in elements (not bytes)
    s_lp = left_proj.stride()
    s_lg = left_gate.stride()
    s_rp = right_proj.stride()
    s_rg = right_gate.stride()
    s_m = mask.stride()
    s_lo = left_out.stride()
    s_ro = right_out.stride()

    # Tuned tile sizes – work well for H100 FP16 kernels
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_fused_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        left_out,
        right_out,
        B,
        N,
        H,
        s_lp[0],
        s_lp[1],
        s_lp[2],
        s_lp[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_rp[0],
        s_rp[1],
        s_rp[2],
        s_rp[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        s_m[0],
        s_m[1],
        s_m[2],
        s_lo[0],
        s_lo[1],
        s_lo[2],
        s_lo[3],
        s_ro[0],
        s_ro[1],
        s_ro[2],
        s_ro[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Batched GEMM (left @ rightᵀ) using cuBLAS tensor‑cores.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] efficiently.

    left, right : [B, N, N, H] (fp16, contiguous)
    Returns out  : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # Reshape to [B*H, N, N] – one batch per hidden channel per batch element
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_flat = left_flat @ right_flatᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore the original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # switch to FP16 for the rest of the pipeline

    # -----------------------------------------------------------------
    # 2️⃣ Fuse all linear projections & gates into a *single* GEMM.
    # -----------------------------------------------------------------
    # Gather weights (shape [hidden_dim, dim] each)
    left_proj_w = weights["left_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]  # also hidden_dim × dim

    # Concatenate -> shape [5*hidden_dim, dim]
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)

    # Single linear pass (produces all pre‑activations at once)
    combined = torch.nn.functional.linear(x, combined_weight)  # [B, N, N, 5*H]

    # Split the combined tensor back into its components
    H = hidden_dim
    left_proj        = combined[..., 0 : H]
    left_gate_logits = combined[..., H : 2 * H]
    right_proj       = combined[..., 2 * H : 3 * H]
    right_gate_logits = combined[..., 3 * H : 4 * H]
    out_gate_logits  = combined[..., 4 * H : 5 * H]

    # Apply sigmoid to gate logits
    left_gate  = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)
    out_gate   = torch.sigmoid(out_gate_logits)

    # -----------------------------------------------------------------
    # 3️⃣ Mask & gate fusion (if present)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and broadcastable
        mask_fp16 = mask_tensor.to(dtype=dtype_fp16)
        # Fuse mask, projection and gate for both sides in a single kernel
        left, right = _apply_mask_gate_fused(
            left_proj, left_gate, right_proj, right_gate, mask_fp16
        )
    else:
        # No mask → simple element‑wise multiplication with the gates
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Cores)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Out‑gate gating + final projection back to model dimension
    # -----------------------------------------------------------------
    out = out * out_gate  # element‑wise gating

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
