"""
TriMul (outgoing) – fast mixed‑precision forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input in FP32 (numerically stable) and cast to FP16.
2️⃣ Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single GEMM:
          x @ W_combined.T   →   shape [B, N, N, 5·H]
   where H = hidden_dim.  The result is reshaped and split to obtain the
   individual tensors.  Sigmoid is applied to the three gating tensors.
3️⃣ If a mask is supplied, fuse the three multiplicative factors
   (proj * gate * mask) with a tiny Triton kernel (`fused_gate_mask_kernel`).
   Otherwise a simple element‑wise multiplication is used.
4️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM using `torch.bmm`, which on H100 maps to
   a Tensor‑Core kernel (batch = B·H, matmul = N×N).
5️⃣ Second Layer‑norm (FP16), gating by `out_gate`, final linear projection,
   and cast back to FP32.

Only the forward pass is required.  The kernel for step 3 is written in
Triton; all other heavy work uses highly‑optimised CUDA kernels via PyTorch.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def fused_gate_mask_kernel(
    x_ptr,        # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,     # *fp16 : gate tensor, same shape as x
    mask_ptr,     # *fp16 : mask tensor (broadcast over hidden dim)
    total_elts: tl.int32,   # total number of elements in x/gate (B·N·N·H)
    hidden_dim: tl.int32,   # hidden dimension H
    BLOCK: tl.constexpr,    # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)        # linear indices
    mask = offs < total_elts

    # Compute mask index (broadcast across hidden dimension)
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # mask defaults to 1

    tl.store(x_ptr + offs, x * g * m, mask=mask)


def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] (FP16).

    Parameters
    ----------
    left, right : torch.Tensor   [B, N, N, H] (contiguous, FP16)

    Returns
    -------
    torch.Tensor   [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # bring the hidden dim into the batch dimension: [B, H, N, N]
    left_batch = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right needs a transpose on the last two axes before the batch flatten
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ  →  (B*H, N, N)
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # restore original layout
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) – may be all‑ones
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x_fp32, mask_fp32, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x_fp32.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – more stable
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x_fp32, (dim,), weight=norm_w, bias=norm_b
    )                     # [B, N, N, dim]   FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)   # now FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five Linear layers into one GEMM
    # ------------------------------------------------------------------
    # Build a single weight matrix of shape (5*hidden_dim, dim)
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Concatenate in the order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_w = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Convert to FP16 once
    combined_w = combined_w.to(torch.float16)

    # Linear: y = x @ combined_w.T   → shape [B, N, N, 5*hidden_dim]
    y = torch.nn.functional.linear(x, combined_w)  # FP16

    # Reshape to separate the five tensors
    B, N, _, _ = y.shape
    y = y.view(B, N, N, 5, hidden_dim)  # [B, N, N, 5, H]

    left_proj   = y[..., 0, :]                     # [B, N, N, H]
    left_gate   = torch.sigmoid(y[..., 1, :])      # → [B, N, N, H]
    right_proj  = y[..., 2, :]                     # [B, N, N, H]
    right_gate  = torch.sigmoid(y[..., 3, :])      # → [B, N, N, H]
    out_gate    = torch.sigmoid(y[..., 4, :])      # → [B, N, N, H]

    # Release the fused tensor early
    del y

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and multiply by gates – fused Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_fp32.to(torch.float16).contiguous()
        total_elts = left_proj.numel()  # B * N * N * H
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # After the in‑place masking we no longer need the gate tensors
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul(left_proj, right_proj)   # [B, N, N, H]  FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Gating + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 8️⃣ Return in FP32 (as expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
