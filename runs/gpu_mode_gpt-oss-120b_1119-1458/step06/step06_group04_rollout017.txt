"""
TriMul (outgoing) custom Triton kernel.

Algorithm
---------
1.  Layer‑norm the input tensor (float32) → convert to fp16.
2.  Fuse *all* linear projections (left, left‑gate, right, right‑gate,
    out‑gate) into a single GEMM:  x @ Wᵀ  where
    W = [left_proj; left_gate; right_proj; right_gate; out_gate].
    The result has shape [B, N, N, 5·hidden_dim] and is split into
    five views that share the same storage.
3.  If a mask is provided, launch one Triton kernel that
    * (a) computes the sigmoid of the gate tensors,
    * (b) multiplies the gate‑activated projections by the mask,
    * (c) writes the results back in‑place to the left/right tensors.
    (Both left and right are processed in the same kernel, halving launch
    overhead.)  If no mask, perform the same operation with a couple of
    cheap torch ops.
4.  Compute the heavy O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM on fp16 using torch.bmm (re‑interpret the tensors
    as [B·hidden_dim, N, N]).
5.  Apply the second LayerNorm (fp16) on the hidden dimension,
    multiply by the out‑gate (sigmoid of the fifth slice), and
    project back to the original `dim` via a final linear layer.
6.  Cast the result back to float32 for the harness.

All heavyweight arithmetic runs in fp16 (TensorCore‑friendly) and the
only Triton kernel is the fused mask‑gate‑sigmoid kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel: fused mask*gate*sigmoid for both left and right tensors.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_sigmoid_fuse_two_kernel(
    left_ptr,          # *fp16  → left projection (in‑place)
    gate_left_ptr,     # *fp16  → left‑gate (pre‑sigmoid)
    right_ptr,         # *fp16  → right projection (in‑place)
    gate_right_ptr,    # *fp16  → right‑gate (pre‑sigmoid)
    mask_ptr,          # *fp16  → mask flattened, shape [B*N*N]
    hidden_dim,        # i32   → size of the hidden dimension (H)
    size,              # i32   → total number of elements in each proj (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load data
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    gl   = tl.load(gate_left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    gr   = tl.load(gate_right_ptr + offs, mask=valid, other=0.0)

    # Broadcast mask over hidden_dim
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Sigmoid (fp16) – use the stable 1/(1+exp(-x)) formulation
    sigl = 1.0 / (1.0 + tl.exp(-gl))
    sigr = 1.0 / (1.0 + tl.exp(-gr))

    # Apply gate, mask and write back in‑place
    tl.store(left_ptr + offs,  left * sigl * m, mask=valid)
    tl.store(right_ptr + offs, right * sigr * m, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the core O(N³) contraction.
# Operates in fp16 and relies on cuBLAS via torch.bmm.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    Parameters
    ----------
    left, right : torch.Tensor
        fp16 tensors of shape [B, N, N, H] (contiguous).

    Returns
    -------
    torch.Tensor
        fp16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                     # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
