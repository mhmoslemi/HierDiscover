---------
1️⃣  LayerNorm over the channel dimension (FP32 → FP16) – numerical stability.
2️⃣  Fuse *all* linear projections that share the same input:
        - left_proj + left_gate + right_proj + right_gate are produced by a
          single `torch.nn.functional.linear` with a concatenated weight
          matrix.  The gate tensors are passed through a sigmoid.
3️⃣  If a mask is present, fuse mask, projection and gate using a Triton
    kernel (`_mask_gate_kernel`).  Otherwise a plain element‑wise multiply
    is performed.
4️⃣  The expensive O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is implemented as a batched GEMM (`torch.bmm`) over the hidden axis.
5️⃣  Second LayerNorm (FP16) → element‑wise multiplication with `out_gate`
    (previously obtained from the fused projection) → final linear
    projection back to the model dimension.
6️⃣  Cast the result to FP32 (the harness expects float32).

Only the mask‑gate fusion is performed in Triton; all other heavy work
leverages highly‑optimized cuBLAS kernels.  This reduces the number of
memory passes and linear calls, giving a substantial speed‑up over the
reference implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask   (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H]  (already sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute  out = proj * gate * mask   (mask broadcast over the hidden
    dimension H).  All tensors are stored in row‑major order.
    """
    pid_b = tl.program_id(0)  # batch index
    pid_i = tl.program_id(1)  # i‑tile index
    pid_j = tl.program_id(2)  # j‑tile index

    # coordinates within the tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # bounds checks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles.
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load proj tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16, shape [I,J,H]

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Fuse: proj * gate * mask   (broadcast mask over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: fuse projection, gating and optional mask via Triton.
# ----------------------------------------------------------------------
def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H] (float16)
    mask       : [B, N, N]   (float16, values 0 or 1)

    Returns out = proj * gate * mask  (mask broadcast over H) as float16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes – empirically tuned for H100.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8  # works well up to H=384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] (fp16).

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (contiguous, dtype=torch.float16).

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (contiguous, dtype=torch.float16).
    """
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) to feed torch.bmm (batched GEMM).
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Perform C = A @ Bᵀ    (both are [B*H, N, N])
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype=torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] in float32.
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # work in fp16 from now on

    # --------------------------------------------------------------
    # 2️⃣ Pack projection + gate weights (single linear call)
    # --------------------------------------------------------------
    # Cast weight matrices to fp16 once.
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)

    # Concatenate the four matrices → shape (4*hidden_dim, dim)
    weight_proj_gate = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*H, dim)

    # --------------------------------------------------------------
    # 3️⃣ Linear: produce left/right projections & gates in one pass
    # --------------------------------------------------------------
    combined = torch.nn.functional.linear(x, weight_proj_gate)  # (B,N,N,4*H)

    # Slice the combined tensor.
    hidden = hidden_dim
    left_proj = combined[..., 0:hidden]                     # (B,N,N,H)
    left_gate_pre = combined[..., hidden:2 * hidden]       # (B,N,N,H)
    right_proj = combined[..., 2 * hidden:3 * hidden]      # (B,N,N,H)
    right_gate_pre = combined[..., 3 * hidden:4 * hidden]  # (B,N,N,H)

    # Apply sigmoid to gate tensors.
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # --------------------------------------------------------------
    # 4️⃣ Out‑gate (separate linear) – still cheap.
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5️⃣ Apply mask (if any) and element‑wise gating.
    # --------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 once.
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (fp16 tensor cores)
    # --------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # (B,N,N,H), fp16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) → out‑gate multiplication
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dimension
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast to FP32 (required by the harness) and return.
    # --------------------------------------------------------------
    return out.to(torch.float32)
