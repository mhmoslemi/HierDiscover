@triton.jit
def masked_gate_fuse_kernel(
    left_ptr,
    right_ptr,
    gates_ptr,
    mask_ptr,
    out_gate_ptr,
    hidden_dim,
    size,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < size

    # compute idx (position for mask) and d (hidden index)
    idx = offs // hidden_dim
    d = offs % hidden_dim

    # Load raw left and right
    left_val = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right_val = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))

    # gate base offset
    base = idx * (3 * hidden_dim)

    # gate values
    left_gate_raw = tl.load(gates_ptr + base + d, mask=active, other=tl.float16(0.0))
    right_gate_raw = tl.load(gates_ptr + base + hidden_dim + d, mask=active, other=tl.float16(0.0))
    out_gate_raw = tl.load(gates_ptr + base + 2 * hidden_dim + d, mask=active, other=tl.float16(0.0))

    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

    # mask term
    mask_val = tl.load(mask_ptr + idx, mask=active, other=tl.float16(1.0))

    # Apply gating and mask
    left_out = left_val * left_gate * mask_val
    right_out = right_val * right_gate * mask_val

    # Store
    tl.store(left_ptr + offs, left_out, mask=active)
    tl.store(right_ptr + offs, right_out, mask=active)

    tl.store(out_gate_ptr + offs, out_gate, mask=active)

"""
Optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑Norm on the input (float32) – stabilises statistics.
2️⃣  Cast to fp16 for the heavy part of the computation.
3️⃣  Fuse the two projections (`left_proj`, `right_proj`) into a single linear
    layer (weight shape 2·hidden_dim × dim).  Likewise fuse the three gating
    projections (`left_gate`, `right_gate`, `out_gate`) into a single linear
    layer (weight shape 3·hidden_dim × dim).  This removes two extra matmul
    passes.
4️⃣  A single Triton kernel `gated_mask_fuse_kernel` reads the raw
    `left`/`right` projections, the fused gating tensor and the mask (if any)
    and computes in‑place:
        left  = left  * sigmoid(left_gate)  * mask
        right = right * sigmoid(right_gate) * mask
        out_gate = sigmoid(out_gate)   (stored separately, no mask)
    This collapses three element‑wise passes (gate sigmoid, mask, store) into
    one memory‑efficient pass.
5️⃣  The O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM using `torch.bmm` after reshaping to
    [B·H, N, N] – cuBLAS/Tensor‑Core handles this efficiently.
6️⃣  Second Layer‑Norm over the hidden dimension (fp16).
7️⃣  Multiply by the output gate produced in step 4.
8️⃣  Final linear projection back to the original dimensionality.
9️⃣  Cast the result back to float32 (required by downstream code).

All heavy tensor arithmetic is performed in fp16; only the first LayerNorm
(and the final cast) uses float32.  The design reduces the number of
linear layers from five to three and replaces three element‑wise kernels
with a single Triton kernel, yielding a substantial speed‑up on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: combine gating (sigmoid), optional mask, and store out_gate
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_fuse_kernel(
    left_ptr,       # *fp16 : left projection (raw, before gating)
    right_ptr,      # *fp16 : right projection (raw)
    gates_ptr,      # *fp16 : fused gate tensor (3*hidden_dim per position)
    mask_ptr,       # *fp16 : mask tensor (B·N·N), broadcast over hidden dim
    out_gate_ptr,   # *fp16 : output gate (hidden_dim per position)
    hidden_dim: tl.constexpr,   # int : size of hidden dimension (H)
    size: tl.int32,             # total number of elements in left/right tensors
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear indices
    active = offs < size

    # Position inside the (B,N,N) grid and offset inside the hidden dim
    idx = offs // hidden_dim           # 0 … B·N·N‑1  (mask index)
    d   = offs % hidden_dim            # 0 … hidden_dim‑1 (channel index)

    # Load raw projections
    left_val  = tl.load(left_ptr  + offs, mask=active, other=tl.float16(0.0))
    right_val = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))

    # Offsets into the fused gate tensor: [left_gate | right_gate | out_gate]
    base = idx * (3 * hidden_dim)                # start of the 3*H slice
    left_gate_raw  = tl.load(gates_ptr + base + d,                     mask=active, other=tl.float16(0.0))
    right_gate_raw = tl.load(gates_ptr + base + hidden_dim + d,        mask=active, other=tl.float16(0.0))
    out_gate_raw   = tl.load(gates_ptr + base + 2 * hidden_dim + d,    mask=active, other=tl.float16(0.0))

    # Sigmoid (stable in fp16)
    left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
    out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

    # Load (or dummy) mask value; mask is 1.0 everywhere when not provided
    mask_val = tl.load(mask_ptr + idx, mask=active, other=tl.float16(1.0))

    # Apply gating and mask
    left_out  = left_val  * left_gate  * mask_val
    right_out = right_val * right_gate * mask_val

    # Store results
    tl.store(left_ptr  + offs, left_out,  mask=active)
    tl.store(right_ptr + offs, right_out, mask=active)
    tl.store(out_gate_ptr + offs, out_gate, mask=active)

# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    via a batched matrix multiplication.
    left/right shape: [B, N, N, H] (fp16)
    Returns tensor of the same shape (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each hidden channel becomes a batch item
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Forward pass entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float32) – ignored if ``nomask`` is True
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x.to(torch.float16)                     # cast for heavy ops

    # -----------------------------------------------------------------
    # 2️⃣ Fuse projection weights (left + right) and gating weights
    # -----------------------------------------------------------------
    # left_proj.weight / right_proj.weight : (hidden_dim, dim)
    proj_weight_fused = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                         # shape (2*hidden_dim, dim)

    # left_gate / right_gate / out_gate weights : (hidden_dim, dim)
    gate_weight_fused = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # shape (3*hidden_dim, dim)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (batched) – fp16
    # -----------------------------------------------------------------
    proj = torch.nn.functional.linear(x, proj_weight_fused)   # [B, N, N, 2*hidden_dim]
    left_raw  = proj[..., :hidden_dim]   # [B, N, N, H]
    right_raw = proj[..., hidden_dim:]   # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4️⃣ Gating (fused) – compute sigmoid and mask in a single Triton kernel
    # -----------------------------------------------------------------
    gates = torch.nn.functional.linear(x, gate_weight_fused)  # [B, N, N, 3*hidden_dim]

    B, N, _, _ = left_raw.shape

    # Prepare mask: broadcasted over hidden_dim; use all‑ones when no mask
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=x.device)

    # Allocate output‑gate tensor
    out_gate = torch.empty_like(left_raw)  # same shape & dtype (fp16)

    # Launch Triton kernel – one pass for left, right and out_gate
    total_elems = left_raw.numel()  # B * N * N * hidden_dim
    BLOCK = 1024
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    gated_mask_fuse_kernel[grid](
        left_raw,
        right_raw,
        gates,
        mask_fp16,
        out_gate,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left_raw, right_raw)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate                               # element‑wise (fp16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
