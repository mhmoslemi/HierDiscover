```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for each tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Determine mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_offs_m = offs_m[:, None] * stride_a_m
    b_offs_n = offs_n[None, :] * stride_b_n

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A tile
        a_ptrs = a_ptr + batch * stride_a_batch + a_offs_m + offs_k[None, :] * stride_a_k
        # Load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Pointers for B tile (transposed)
        b_ptrs = b_ptr + batch * stride_b_batch + offs_k[:, None] * stride_b_k + b_offs_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store results
    c_ptrs = c_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – fast Triton implementation
---------------------------------------------------------

The module implements the forward pass of the “outgoing” TriMul operator.
The algorithm is:

1️⃣  Layer‑Norm on the raw 4‑D tensor (float32) – stabilises statistics.
2️⃣  Cast to FP16 for the bulk of the work.
3️⃣  Fuse all five linear projections (left‑proj, left‑gate, right‑proj,
    right‑gate, out‑gate) into a single FP16 GEMM.
4️⃣  Apply sigmoid gates, then an optional binary mask (in‑place).
5️⃣  Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched matrix‑multiply (A @ Bᵀ) using a custom Triton kernel.
    The hidden dimension is folded into the batch dimension so the kernel
    launches one GEMM per (batch, hidden) pair.
6️⃣  Layer‑Norm over the hidden dimension, multiply by the output gate,
    and a final linear projection back to the original channel size.
7️⃣  Cast back to float32 (the harness expects float32 output).

Only the forward pass is required – no autograd support.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where A and B are [batch, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (M=N, K=N)
    stride_a_batch, stride_a_m, stride_a_k,  # A strides
    stride_b_batch, stride_b_k, stride_b_n,  # Bᵀ strides (k,row; n,col)
    stride_c_batch, stride_c_m, stride_c_n,  # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute C = A @ Bᵀ for a batch of matrices.
    A – shape (batch, M, K)  – row‑major (stride_m = K, stride_k = 1)
    B – shape (batch, N, K)  – row‑major (stride_n = K, stride_k = 1)
    We treat Bᵀ by loading with swapped indices (k‑row, n‑col).
    """
    batch_id = tl.program_id(0)           # batch index
    pid_m = tl.program_id(1)              # tile‑row index
    pid_n = tl.program_id(2)              # tile‑col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Offsets for the row dimension of A and the column dimension of C
    a_batch_offset = batch_id * stride_a_batch
    c_batch_offset = batch_id * stride_c_batch

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- Load A tile -------------------------------------------------
        # A[batch, i, k] -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + a_batch_offset \
                      + offs_m[:, None] * stride_a_m \
                      + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- Load Bᵀ tile ------------------------------------------------
        # Bᵀ[batch, k, j] corresponds to B[batch, j, k] in memory.
        # B row stride (j) = stride_b_n, col stride (k) = stride_b_k (=1)
        b_ptrs = b_ptr + batch_id * stride_b_batch \
                      + offs_k[:, None] * stride_b_k \
                      + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate (FP16 dot → FP32 acc)
        acc += tl.dot(a, b)

    # ----- Store C tile ----------------------------------------------------
    c_ptrs = c_ptr + c_batch_offset \
                  + offs_m[:, None] * stride_c_m \
                  + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the batched GEMM kernel
# ----------------------------------------------------------------------
def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform out‑going TriMul contraction via the custom Triton GEMM.
    left/right: (B, N, N, H) fp16 tensors.
    Returns: (B, N, N, H) fp16.
    """
    B, N, _, H = left.shape
    # Permute to (B, H, N, N) and flatten batch & hidden dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # New batch dimension = B * H
    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # Strides (all tensors are contiguous after the view)
    stride_a_batch = left_flat.stride(0)  # N * N
    stride_a_m = left_flat.stride(1)      # N
    stride_a_k = left_flat.stride(2)      # 1

    stride_b_batch = right_flat.stride(0)  # N * N
    stride_b_k = right_flat.stride(2)      # 1   (original column stride)
    stride_b_n = right_flat.stride(1)      # N   (original row stride)

    stride_c_batch = out_flat.stride(0)    # N * N
    stride_c_m = out_flat.stride(1)        # N
    stride_c_n = out_flat.stride(2)        # 1

    # Tile sizes – empirically good for H100 FP16
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid = (batch, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M  (rows of A)
        N,               # N (cols of Bᵀ)
        N,               # K (shared dimension)
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 likes many warps for large tiles
    )

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]      (bool / float or None)
        - weights_dict : model weights (float32)
        - config_dict  : contains ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, same device as input)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32) – stabilises the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # -> [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣  Cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣  Fuse all five linear projections (left/right proj & gates + out_gate)
    # -----------------------------------------------------------------
    # Gather the weight tensors (originally float32) and cast once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)      # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)   # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)    # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)   # (H, dim)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)     # (H, dim)

    # Concatenate along the output‑feature axis → shape (5*H, dim)
    combined_weight = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)                                            # (5*H, dim)

    # One big linear projection – reads `x` only once
    proj = torch.nn.functional.linear(x, combined_weight)   # [B, N, N, 5*H]

    # Split back into the five components
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )

    # -----------------------------------------------------------------
    # 4️⃣  Apply sigmoid gates
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)      # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5️⃣  Element‑wise gating (proj * gate) – mask applied later
    # -----------------------------------------------------------------
    left = left_proj_raw * left_gate            # [B, N, N, H]
    right = right_proj_raw * right_gate         # [B, N, N, H]

    # -----------------------------------------------------------------
    # 6️⃣  Optional binary mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        # Broadcast mask to hidden dim: shape (B,N,N,1) → (B,N,N,H)
        left = left * mask_fp16.unsqueeze(-1)
        right = right * mask_fp16.unsqueeze(-1)

    # -----------------------------------------------------------------
    # 7️⃣  Heavy O(N³) contraction via the custom Triton GEMM
    # -----------------------------------------------------------------
    # left/right: (B, N, N, H) fp16 → out: same shape
    out = _batched_matmul_triton(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 8️⃣  Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -----------------------------------------------------------------
    # 9️⃣  Apply output gate and final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)      # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 🔟  Cast back to float32 for the harness and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
