def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Step 1: First LayerNorm
    norm_weight, norm_bias = weights["norm.weight"], weights["norm.bias"]
    # input_tensor is float32; layer_norm in float32, produce float32
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias) # shape [B,N,N,dim], dtype float32

    # Cast to fp16 for heavy compute
    x_fp16 = x_norm.to(torch.float16)

    # Step 2: Combined linear for left side
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    # flatten input for matmul (B*N*N, dim)
    B, N, _, _ = x_fp16.shape
    x_flat = x_fp16.view(-1, dim) # shape (B*N*N, dim)
    left_concat_out = torch.nn.functional.linear(x_flat, left_concat_w) # shape (B*N*N, 2*hidden_dim)
    left_proj_raw = left_concat_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate_pre = left_concat_out[:, hidden_dim:].view(B, N, N, hidden_dim)

    # left_gate = sigmoid(left_gate_pre)
    left_gate = torch.sigmoid(left_gate_pre)

    # Step 3: Combined linear for right side
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    right_concat_out = torch.nn.functional.linear(x_flat, right_concat_w)
    right_proj_raw = right_concat_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate_pre = right_concat_out[:, hidden_dim:].view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate_pre)

    # Step 4: Apply mask and gating via Triton kernel if mask else elementwise
    if not nomask and mask_tensor is not None:
        # mask shape [B,N,N], broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1) # [B*N*N]
        total_elems = left_proj_raw.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)
        # left
        mask_gate_fuse_kernel[grid](
            left_proj_raw, left_gate, mask_flat, hidden_dim, total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right
        mask_gate_fuse_kernel[grid](
            right_proj_raw, right_gate, mask_flat, hidden_dim, total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj_raw
        right = right_proj_raw

        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # Step 5: Batched GEMM contraction.
    # But left and right now are of shape [B,N,N,hidden_dim]. Already ensure contiguous.
    out = _batched_gemm_fp16(left, right)  # shape [B,N,N,hidden_dim]

    # Step 6: Second LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16))

    # Step 7: Compute out_gate: out_gate = linear(x_fp16, out_gate_weight).sigmoid()

    # Step 8: Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32
    return out.to(torch.float32)

```python
"""
TriMul outgoing forward pass (optimized).

Algorithm:
1. Layer‑norm the 4‑D input (float32) – stable and performed with torch.
2. Cast to fp16 for all subsequent heavy work.
3. Fuse the left‑ and left‑gate projections into a single GEMM:
   - weight matrix is [2*hidden_dim, dim]; the first half gives left_proj,
     the second half yields pre‑sigmoid left_gate.
   - Same for the right side.
4. Apply mask (if present) and gating in a fused Triton kernel
   `mask_gate_fuse_kernel` that multiplies left/right projections by
   their gates and by the binary mask, broadcasting the mask over the
   hidden dimension.
5. Perform the O(N³) contraction using a batched GEMM: for every hidden
   channel we compute A @ Bᵀ, where A = left[..., :, h] and B = right[..., :, h].
6. Apply a second LayerNorm (fp16) on the contracted tensor.
7. Compute the output gate from the original normalized input, multiply
   element‑wise with the tensor and apply the final linear projection.
8. Cast the result back to float32.

All heavy linear layers and the mask‑gate fusion run in fp16. The
contraction exploits cuBLAS batched GEMM which on H100 harnesses
Tensor‑Cores, yielding the best possible throughput for the O(N³)
step. The Triton kernel is required for the mask‑gate fusion to
satisfy the problem constraints.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel to fuse mask, gate and tensor multiplication.
# Input tensors are flattened fp16 arrays; mask has shape [B*N*N] and
# broadcasted over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    tensor_ptr,    # *fp16 → tensor to be updated (flattened)
    gate_ptr,      # *fp16 → gate tensor (same flattening as tensor)
    mask_ptr,      # *fp16 → mask flattened, shape [B*N*N]
    hidden_dim,    # i32   → size of hidden dim (H)
    size,          # i32   → total number of elements in tensor (B*N*N*H)
    BLOCK: tl.constexpr,  # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load tensor, gate and mask (broadcast over hidden dim)
    x = tl.load(tensor_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index: each mask element corresponds to hidden_dim
    # consecutive tensor elements.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default mask = 1

    # Fusion: x * g * mask
    out = x * g * m
    tl.store(tensor_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Parameters
    ----------
    left, right : torch.Tensor
        fp16 tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        fp16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) and perform bmm with the transpose of right.
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                 # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,
        torch.Tensor,
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32, bool, or None)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; may
                         contain ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – keep numeric stability.
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], dtype float32

    # -----------------------------------------------------------------
    # 2️⃣ Switch to fp16 for the compute‑heavy part.
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Prepare combined weights for left/right + their gates.
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden_dim, dim)

    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden_dim, dim)

    # -----------------------------------------------------------------
    # 4️⃣ Flatten input for GEMM (B*N*N, dim).
    # -----------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)            # (B*N*N, dim)

    # -----------------------------------------------------------------
    # 5️⃣ Linear combo for the left side → proj and gate in one GEMM.
    # -----------------------------------------------------------------
    left_combined_out = torch.nn.functional.linear(x_flat, left_combined_w)  # (B*N*N, 2*hidden_dim)
    # Split into projection and pre‑gate part.
    left_proj_raw = left_combined_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate_pre = left_combined_out[:, hidden_dim:].view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_gate_pre)  # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 6️⃣ Linear combo for the right side.
    # -----------------------------------------------------------------
    right_combined_out = torch.nn.functional.linear(x_flat, right_combined_w)
    right_proj_raw = right_combined_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate_pre = right_combined_out[:, hidden_dim:].view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate_pre)

    # -----------------------------------------------------------------
    # 7️⃣ Optional mask + gate fusion (Triton).  If mask is absent we perform
    #     a simple element‑wise multiplication.
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16, flatten to [B*N*N].
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                     # (B*N*N)

        total_elems = left_proj_raw.numel()                # B*N*N*hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # Fuse projection * gate * mask for the left side.
        mask_gate_fuse_kernel[grid](
            left_proj_raw,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse the same for the right side.
        mask_gate_fuse_kernel[grid](
            right_proj_raw,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj_raw
        right = right_proj_raw
    else:
        # No mask -> just multiply by the gate.
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # -----------------------------------------------------------------
    # 8️⃣ Core O(N³) contraction via batched GEMM (Tensor‑Core accelerated).
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # shape [B,N,N,hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 9️⃣ Second LayerNorm (fp16) over the hidden dimension.
    # -----------------------------------------------------------------
    out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=out_norm_weight,
        bias=out_norm_bias,
    )

    # -----------------------------------------------------------------
    # 🔟 Compute out‑gate from the *original* normalized tensor and apply it.
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)   # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(out_gate_pre)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 1️⃣1️⃣ Final projection back to the original channel dimension.
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 for the harness.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```

"""
TriMul outgoing forward pass (optimized).

Algorithm
---------
1️⃣  First LayerNorm (float32) – stable normalization of the 4‑D input.
2️⃣  Cast to fp16; all heavy compute runs in half‑precision.
3️⃣  Fuse left‑projection + left‑gate (and analogously right) into a
   single GEMM:
   - Weight matrix is stacked as [proj; gate] → shape (2·hidden_dim, dim).
   - The first half of the output yields the projection,
     the second half supplies the pre‑sigmoid gate.
4️⃣  If a mask is present, fuse mask, gate and projection with a tiny
   Triton kernel (`mask_gate_fuse_kernel`).  This satisfies the requirement
   to use Triton while keeping launch overhead negligible.
5️⃣  Core O(N³) contraction: for every hidden channel we compute a
   batched GEMM `C = A @ Bᵀ` where `A = left[..., :, h]`,
   `B = right[..., :, h]`.  cuBLAS (Tensor‑Cores) handles this efficiently.
6️⃣  Second LayerNorm (fp16) over the hidden dimension.
7️⃣  Compute the output‑gate from the original normalized tensor,
   multiply element‑wise.
8️⃣  Final linear projection back to the original `dim`.
9️⃣  Cast the result back to float32 for the evaluation harness.

All linear layers and mask‑gate fusion run in fp16; the large contraction
leverages cuBLAS batched GEMM which on an H100 provides Tensor‑Core
throughput.  The Triton kernel is required for the mask‑gate fusion part.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask‑gate‑tensor multiplication.
# Operates on flattened fp16 tensors; the mask is broadcast over the
# hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    tensor_ptr,   # *fp16 – tensor to be updated (flattened)
    gate_ptr,     # *fp16 – gate tensor (same flattening as tensor)
    mask_ptr,     # *fp16 – mask flattened, shape [B*N*N]
    hidden_dim,   # i32   – hidden dimension size (H)
    size,         # i32   – total number of tensor elements (B*N*N*H)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load tensor, gate and mask (mask broadcast over hidden_dim)
    x = tl.load(tensor_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Derive mask index (each `hidden_dim` consecutive elements share the same mask)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1

    # Fuse multiplication
    out = x * g * m
    tl.store(tensor_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right via:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    Parameters
    ----------
    left, right : torch.Tensor (fp16)
        Shape [B, N, N, H].

    Returns
    -------
    torch.Tensor (fp16)
        Shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # Fold the batch and hidden dimensions together to obtain (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched matrix multiplication: A @ Bᵀ
    c = torch.bmm(a, b.transpose(1, 2))                     # (B*H, N, N)
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,
        torch.Tensor,
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float or None)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; may contain
                         ``nomask`` (bool, defaults to True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ---------------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32)
    # ---------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], dtype float32

    # ---------------------------------------------------------------
    # 2️⃣  Cast to fp16 for the heavy computations
    # ---------------------------------------------------------------
    x = x_norm.to(torch.float16)          # [B, N, N, dim], dtype fp16

    # ---------------------------------------------------------------
    # 3️⃣  Stack projection + gate weights (fp16) for left and right sides
    # ---------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)    # (2*hidden_dim, dim)

    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden_dim, dim)

    # ---------------------------------------------------------------
    # 4️⃣  Flatten the input for a single GEMM per side
    # ---------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # (B * N * N, dim)

    # ---------------------------------------------------------------
    # 5️⃣  Left side: projection + gate in a single linear call
    # ---------------------------------------------------------------
    left_combined_out = torch.nn.functional.linear(x_flat, left_combined_w)  # (B*N*N, 2*hidden_dim)
    left_proj_raw = left_combined_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate_pre = left_combined_out[:, hidden_dim:].view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_gate_pre)                     # (B, N, N, hidden_dim)

    # ---------------------------------------------------------------
    # 6️⃣  Right side: projection + gate in a single linear call
    # ---------------------------------------------------------------
    right_combined_out = torch.nn.functional.linear(x_flat, right_combined_w)
    right_proj_raw = right_combined_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate_pre = right_combined_out[:, hidden_dim:].view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate_pre)

    # ---------------------------------------------------------------
    # 7️⃣  Optional mask + gate fusion (Triton)
    # ---------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16 and flatten to [B * N * N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                     # (B*N*N)

        total_elems = left_proj_raw.numel()                # B*N*N*hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # Fuse (projection * gate * mask) for the left side
        mask_gate_fuse_kernel[grid](
            left_proj_raw,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse the same for the right side
        mask_gate_fuse_kernel[grid](
            right_proj_raw,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj_raw
        right = right_proj_raw
    else:
        # No mask – simple element‑wise multiplication
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # ---------------------------------------------------------------
    # 8️⃣  Core contraction: batched GEMM (Tensor‑Core accelerated)
    # ---------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)    # (B, N, N, hidden_dim), fp16

    # ---------------------------------------------------------------
    # 9️⃣  Second LayerNorm over the hidden dimension (fp16)
    # ---------------------------------------------------------------
    out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=out_norm_weight,
        bias=out_norm_bias,
    )

    # ---------------------------------------------------------------
    # 🔟  Output gate (computed from the original normalized tensor)
    # ---------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)   # (B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate_pre)

    out = out * out_gate

    # ---------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to `dim`
    # ---------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim), fp16

    # ---------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 for the harness
    # ---------------------------------------------------------------
    return out.to(torch.float32)
