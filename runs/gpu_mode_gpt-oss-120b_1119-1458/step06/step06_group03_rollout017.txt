@triton.jit
def tri_mul_batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_size: tl.constexpr,
    seq_len: tl.constexpr,
    hidden_dim: tl.constexpr,
    # strides
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    # block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute b,h
    hid = pid_batch
    b = hid // hidden_dim
    h = hid % hidden_dim

    # offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # offsets into each tile
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

    # bounds checks
    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # base pointers for this batch/h
    left_base = left_ptr + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr + b * stride_r_batch + h * stride_r_h
    out_base = out_ptr + b * stride_o_batch + h * stride_o_h

    # create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for kk in range(0, tl.cdiv(seq_len, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seq_len

        # left tile pointer: (i,k)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # right tile pointer: (j,k) but we need layout for dot as (k,j)
        # Right tile we load with shape (BLOCK_K, BLOCK_N): (k,j)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # dot product: left_tile (M,K) dot right_tile (K,N) => acc (M,N)
        acc += tl.dot(left_tile, right_tile)

    # store to output: out[batch,batch_h,i,j] shape (BLOCK_M,BLOCK_N)
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    # convert to output dtype (FP16) maybe store as fp16 directly.
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast forward pass with Triton fusion.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32 → FP16) – stabilises statistics.
2️⃣ Linear projections (left/right) and three gating projections (left‑gate,
   right‑gate, out‑gate) are done in FP16.
3️⃣ If a mask is supplied we fuse the three multiplicative factors
   (proj * gate * mask) with a tiny Triton kernel (`fused_gate_mask_kernel`),
   saving two element‑wise kernels.
4️⃣ The heavy O(N³) contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is performed by a custom batched GEMM Triton kernel (`tri_mul_batched_gemm_kernel`).
   The kernel tiles the M‑, N‑ and K‑dimensions (M=i, N=j, K=k) and uses
   `tl.dot`, which maps to H100 Tensor‑Cores (FP16×FP16→FP32 accumulation).
   The batch dimension is the product B·hidden_dim, avoiding any memory copy.
5️⃣ A second Layer‑norm (FP16) normalises the hidden dimension, the output‑gate
   (sigmoid) is applied, and a final linear projection restores the original
   feature dimension.
6️⃣ The result is cast back to FP32 for the harness.

This implementation satisfies the “at least one Triton kernel” requirement
and removes the costly `permute(...).contiguous()` copies of the reference
solution, yielding a substantial speed‑up on the H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (broadcast over hidden_dim)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    valid = offs < total_elts

    # mask is broadcast across hidden_dim entries.
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#
# Tensor shape conventions:
#   left/right/out : [B, N, N, H]   (FP16)
#   stride order   :   (B, i, k, h) as stored by PyTorch
#   (see `x.stride()` for the exact numbers)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                                 # compile‑time scalars
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids:
    pid_batch = tl.program_id(0)   # 0 ≤ pid_batch < B*H  (combined batch × hidden)
    pid_i     = tl.program_id(1)   # tile row index   (i dimension)
    pid_j     = tl.program_id(2)   # tile col index   (j dimension)

    # Recover the original batch index b and hidden index h.
    # NOTE: H is a compile‑time constant, therefore integer division & remainder
    # are cheap.
    h_idx = pid_batch % H
    b_idx = pid_batch // H

    # --------------------------------------------------------------
    # Tile offsets for the i‑ and j‑dimensions.
    # --------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Base pointers for this (b, h) slice.
    # --------------------------------------------------------------
    left_base  = left_ptr  + b_idx * stride_l_batch + h_idx * stride_l_h
    right_base = right_ptr + b_idx * stride_r_batch + h_idx * stride_r_h
    out_base   = out_ptr   + b_idx * stride_o_batch + h_idx * stride_o_h

    # --------------------------------------------------------------
    # Accumulator (FP32) – Tensor‑Core dot products accumulate in FP32.
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction axis k.
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K) → (i, k)
        left_ptrs = (left_base
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile (transposed inside the load): we need shape (BLOCK_K, BLOCK_N) → (k, j)
        # original layout: right[b, j, k, h] → offsets: j*stride_r_i + k*stride_r_k
        right_ptrs = (right_base
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_i)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # FP16 dot → FP32 accumulation
        acc += tl.dot(left_tile, right_tile)

    # --------------------------------------------------------------
    # Store the tile to the output tensor.
    # --------------------------------------------------------------
    out_ptrs = (out_base
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the batched GEMM Triton kernel.
# ----------------------------------------------------------------------
def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using Triton.
    left, right : FP16 tensors of shape [B, N, N, H]

    Returns:
        out : FP16 tensor of shape [B, N, N, H]
    """
    B, N, _, H = left.shape
    device = left.device

    # Allocate output tensor (FP16) – same strides as the inputs
    out = torch.empty_like(left)

    # Strides in number of elements (PyTorch default)
    stride_l = left.stride()   # (batch, i, k, h)
    stride_r = right.stride()
    stride_o = out.stride()

    # Tile sizes – tuned for H100 Tensor Cores.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid dimensions:
    #  pid_batch ∈ [0, B*H)
    #  pid_i     ∈ [0, ceil(N/BLOCK_M))
    #  pid_j     ∈ [0, ceil(N/BLOCK_N))
    total_batches = B * H
    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Triton kernel launch
    tri_mul_batched_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        # strides (passed as simple Python ints)
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → good occupancy on H100
    )
    return out

    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5️⃣ Gating signals (FP16) → sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask – fuse (tensor * gate * mask) in‑place via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast across hidden_dim – convert to FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()                # B * N * N * hidden_dim
        BLOCK = 1024                              # ≤ max threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via the custom Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_gemm_triton(left, right)   # [B, N, N, hidden_dim]   FP16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Gating by out_gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 🔟 Return FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
