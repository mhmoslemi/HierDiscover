"""
Optimized outgoing TriMul forward pass (no autograd).

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32) and cast to fp16.
2.  Flatten the tensor to [B·N·N, dim] and perform three linear projections
    (combined into two matmuls):
       * left‑proj & right‑proj  → shape [M, 2·H]
       * left‑gate & right‑gate  → shape [M, 2·H]
       * out‑gate                → shape [M, H]
    where  M = B·N·N  and  H = hidden_dim.
3.  Apply sigmoid to the gate tensors.
4.  Fuse element‑wise multiplication with the optional mask using a tiny
    Triton kernel (proj * gate * mask) – in‑place, no extra temporaries.
5.  Reshape the left/right tensors back to [B, N, N, H].
6.  Heavy O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    implemented as a batched GEMM (torch.bmm) after a cheap permute+view.
7.  Second LayerNorm over the hidden dimension.
8.  Multiply by the out‑gate, then a final linear projection to the original
    channel dimension.
9.  Cast the result back to float32 (the harness expects float32).

Key speed‑ups
------------
* 5 matmul‑style linear layers are collapsed to **3** large matmuls.
* Mask‑gate‑multiply is performed by a custom Triton kernel (≈1 µs launch).
* The O(N³) contraction uses cuBLAS batched GEMM (highly tuned) with
  minimal data copies.
* All heavy arithmetic is carried out in fp16; only the two LayerNorms stay
  in fp32 for numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) in‑place.
# Used only when a mask is supplied.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,                 # *fp16 tensor to be updated (flattened, shape [M*H])
    gate_ptr,              # *fp16 gate tensor (same shape as x)
    mask_ptr,              # *fp16 mask tensor (flattened, shape [M])
    hidden_dim,            # i32 hidden dimension (H)
    size,                  # i32 total number of elements in x (M*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the three operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # The mask is broadcast over the hidden dimension.
    # Each mask entry corresponds to `hidden_dim` consecutive elements in x.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default = 1

    # Fuse multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM (fp16) performing the O(N³) contraction.
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H) in fp16
    Returns:
        out: (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    #   1) bring H into the batch dimension
    #   2) perform B*H independent N×N matrix multiplications
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : [B, N, N, dim]    (float32)
        - mask_tensor  : [B, N, N]        (float32 or bool)
        - weights      : dict of model weights (fp32)
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16 for compute‑heavy part
    # --------------------------------------------------------------
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]   (float32)

    x = x_norm_fp32.to(torch.float16)                     # fp16

    B, N, _, _ = input_tensor.shape
    M = B * N * N                                         # flattened sequence length

    # --------------------------------------------------------------
    # 2️⃣ Cast model weights once to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, dim)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, dim)

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)

    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (dim, H)

    # --------------------------------------------------------------
    # 3️⃣ Flatten the input for massive matrix multiplies
    # --------------------------------------------------------------
    x_flat = x.view(M, dim)   # (M, dim), fp16

    # --------------------------------------------------------------
    # 4️⃣ Compute left/right projections & their gates in just 3 matmuls
    # --------------------------------------------------------------
    # combine left/right projections
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)   # (2H, dim)
    proj = torch.nn.functional.linear(x_flat, proj_weight)       # (M, 2H)

    # combine left/right gates
    gate_weight = torch.cat([left_gate_w, right_gate_w], dim=0)   # (2H, dim)
    gate = torch.nn.functional.linear(x_flat, gate_weight)       # (M, 2H)

    # out‑gate (used later)
    out_gate_raw = torch.nn.functional.linear(x_flat, out_gate_w)   # (M, H)

    # split the concatenated results
    H = hidden_dim
    left_proj = proj[:, :H]                # (M, H)
    right_proj = proj[:, H:]               # (M, H)

    left_gate_raw = gate[:, :H]            # (M, H)
    right_gate_raw = gate[:, H:]           # (M, H)

    # apply sigmoid to the gate tensors
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)          # (M, H)

    # --------------------------------------------------------------
    # 5️⃣ Apply mask (if any) and combine proj·gate in‑place with a Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        # mask is shape [B, N, N]; broadcast over hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                # (M,)

        # left = left_proj * left_gate * mask
        size = left_proj.numel()                      # M * H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # fuse multiplication (proj * gate * mask) in‑place on left_proj
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # fuse multiplication on right_proj
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # after the kernels the tensors contain the gated‑and‑masked values
        left = left_proj
        right = right_proj
    else:
        # No mask – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Reshape to 4‑D tensors [B, N, N, H]
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)        # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                            # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 9️⃣ Final gating with out_gate and projection back to dim
    # --------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)          # (B, N, N, H) fp16
    out = out * out_gate                          # element‑wise gating

    # final linear projection to the original dimension
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim) fp16

    # --------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
