@triton.jit
def second_norm_gate_kernel(
    out_ptr, out_gate_ptr, norm_weight_ptr, norm_bias_ptr,
    H, M,
    BLOCK_H: tl.constexpr, EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    # check out-of-range positions
    if pid >= M:
        return
    # compute offset base for this position
    base = pid * H

    # Initialize accumulators for mean and variance
    sum_val = tl.float32(0)
    sum_sq = tl.float32(0)

    # Loop over hidden dimension chunks
    for off in range(0, H, BLOCK_H):
        # calculate hidden indices for this chunk
        h = off + tl.arange(0, BLOCK_H)
        # mask for within hidden_dim bounds
        mask = h < H
        # compute linear offset into out and load values
        offs = base + h
        x = tl.load(out_ptr + offs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=0)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)

    # compute mean and variance
    inv_h = 1.0 / H
    mean = sum_val * inv_h
    var = sum_sq * inv_h - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Second pass: apply layernorm, scaling, bias, gating, and write back
    for off in range(0, H, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask = h < H
        offs = base + h
        x = tl.load(out_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        g = tl.load(out_gate_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(norm_weight_ptr + h, mask=mask, other=0.0).to(tl.float32)

"""
TriMul (Outgoing) – fast Triton‑accelerated version
===================================================

Algorithm
---------
1. **First LayerNorm** (float32) on the input tensor (B,N,N,dim).
2. Cast the normalized tensor to fp16 for the compute‑heavy part.
3. **Fused projection & gating**: concatenate the five linear weight
   matrices (left_proj, right_proj, left_gate, right_gate, out_gate)
   into one big matrix of shape (5*hidden_dim, dim) and perform a single
   GEMM (`torch.nn.functional.linear`).  The result is split into the
   five tensors and the three gating tensors are passed through a sigmoid.
4. **Mask & gate fusion** – when a mask is present we launch a
   Triton kernel that computes ``tensor = tensor * gate * mask`` in‑place
   for both the left and the right branches.
5. **Heavy O(N³) contraction** – compute
   ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``
   using a batched GEMM (cuBLAS) on fp16 data.
6. **Second LayerNorm + out‑gate** – a custom Triton kernel fuses the
   per‑position mean/variance computation, LayerNorm (γ/β), multiplication
   by the pre‑computed out‑gate and writes the result back in‑place.
7. **Final projection** – a standard linear layer maps the hidden_dim
   vector back to the original `dim`.
8. Cast the final tensor back to float32 (the harness expects float32).

The implementation reduces the number of separate GEMM calls (5→2) and
fuses two element‑wise passes (LayerNorm + gating) into a single
kernel, cutting memory traffic and kernel launch overhead while keeping
the heavy O(N³) matrix multiplication on cuBLAS (the most optimal
routine on H100).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: mask * gate * x   (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated (flattened)
    gate_ptr,       # *fp16  gate tensor (same shape as x)
    mask_ptr,       # *fp16  mask tensor (flattened, shape [B*N*N])
    hidden_dim,     # i32   hidden dimension (H)
    size,           # i32   total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the three operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Broadcast mask across the hidden dimension
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default = 1

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (over hidden_dim) + out_gate
# ----------------------------------------------------------------------
@triton.jit
def second_norm_gate_kernel(
    out_ptr,          # *fp16   tensor [M, H] – will be overwritten
    out_gate_ptr,     # *fp16   gating tensor (same shape)
    norm_w_ptr,       # *fp16   LayerNorm weight (γ) size H
    norm_b_ptr,       # *fp16   LayerNorm bias  (β) size H
    M,                # i32     number of positions (B*N*N)
    H: tl.constexpr,  # i32     hidden dimension (known at compile time)
    BLOCK_H: tl.constexpr,   # compile‑time tile size for H (≤1024)
    EPS: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)
    if pid >= M:
        return

    base = pid * H                         # start offset for this position

    # ---- pass 1: compute mean and variance ----
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    # iterate over hidden_dim in chunks of BLOCK_H
    for off in range(0, H, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)               # lane index within the chunk
        mask = h < H                                   # guard out‑of‑range lanes
        offs = base + h
        x = tl.load(out_ptr + offs, mask=mask, other=0.0)
        xf = x.to(tl.float32)
        sum_val += tl.sum(xf, axis=0)                  # reduce across threads in block
        sum_sq  += tl.sum(xf * xf, axis=0)

    inv_h = 1.0 / tl.float32(H)
    mean = sum_val * inv_h
    var  = sum_sq * inv_h - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ---- pass 2: apply LayerNorm, scaling, bias, out_gate ----
    for off in range(0, H, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask = h < H
        offs = base + h
        x = tl.load(out_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        g = tl.load(out_gate_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(norm_w_ptr + h, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(norm_b_ptr + h, mask=mask, other=0.0).to(tl.float32)

        # LayerNorm
        norm = (x - mean) * inv_std
        # γ / β
        norm = norm * w + b
        # Out‑gate
        out_val = norm * g
        tl.store(out_ptr + offs, out_val.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM (fp16) performing the O(N³) contraction.
# out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H) in fp16
    Returns:
        out: (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # bring hidden dimension to the batch axis to expose many independent GEMMs
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd) with Triton fusions.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, values 0/1) or None
        - weights      : dict of model parameters (fp32)
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32, stable) → fp16 for the rest
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)    # cast once for the heavy part

    # -----------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections (left_proj, right_proj,
    #    left_gate, right_gate, out_gate) into a single GEMM.
    # -----------------------------------------------------------------
    B, N, _, _ = x.shape
    M = B * N * N                   # flattened number of pairwise positions

    # Concatenate weight matrices: shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)            # now (5*H, dim)

    # Linear projection on the flattened tensor
    x_flat = x.view(M, dim)        # [M, dim]
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # [M, 5*H]

    H = hidden_dim
    # Split the concatenated output
    left      = combined_out[:, 0:H]
    right     = combined_out[:, H:2*H]
    left_gate = torch.sigmoid(combined_out[:, 2*H:3*H])
    right_gate = torch.sigmoid(combined_out[:, 3*H:4*H])
    out_gate  = torch.sigmoid(combined_out[:, 4*H:5*H])

    # Restore original 4‑D shape
    left      = left.view(B, N, N, H)
    right     = right.view(B, N, N, H)
    left_gate = left_gate.view(B, N, N, H)
    right_gate = right_gate.view(B, N, N, H)
    out_gate  = out_gate.view(B, N, N, H)

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)          # [B*N*N]
        size = left.numel()                     # total elements in left/right
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → just apply the gates
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction – batched GEMM (cuBLAS, fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + out_gate
    # -----------------------------------------------------------------
    out_flat = out.view(M, H)                # [M, H]
    out_gate_flat = out_gate.view(M, H)

    # Load to_out_norm parameters
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    BLOCK_H = 32                               # 32‑thread block works for any H ≤ 384
    grid2 = (M,)
    second_norm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        to_out_norm_w,
        to_out_norm_b,
        M,                # runtime value
        H=hidden_dim,     # compile‑time constant for reduction loops
        BLOCK_H=BLOCK_H,
        EPS=1e-5,
        num_warps=4,
    )
    # Reshape back to 4‑D tensor
    out = out_flat.view(B, N, N, H)

    # -----------------------------------------------------------------
    # 6️⃣ Final projection back to the original channel dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast to float32 (expected by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```

This implementation fuses the linear projections (five separate `nn.Linear` calls) into a single GEMM, performs mask‑gate multiplication in one Triton kernel, and fuses the second LayerNorm with the out‑gate in another kernel, thereby cutting memory traffic and kernel launch overhead while keeping the compute‑intensive O(N³) step on cuBLAS for maximum throughput on the H100.
