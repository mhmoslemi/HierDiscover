@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # grid dimension: tile_idx (i_tile * num_tiles_n + j_tile)
    batch_id = tl.program_id(1) # batch (BATCH)
    
    # compute tile coordinates
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_tiles_n
    pid_n = pid % num_tiles_n
    # offsets
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # keep track of offsets
    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # pointers to left A tile
        a_ptrs = left_ptr + (batch_id * stride_left_b
                             + i_offset * stride_left_i
                             + k_offset * stride_left_k
                             + tl.arange(0, BLOCK_M)[:, None] * stride_left_i
                             + tl.arange(0, BLOCK_K)[None, :] * stride_left_k)
        # pointers to right B_T tile
        # B_T tile uses stride_right_k to move along k axis and stride_right_i for j axis.
        b_ptrs = right_ptr + (batch_id * stride_right_b
                              + k_offset * stride_right_k
                              + j_offset * stride_right_i
                              + tl.arange(0, BLOCK_K)[:, None] * stride_right_k
                              + tl.arange(0, BLOCK_N)[None, :] * stride_right_i)
        # Load
        a = tl.load(a_ptrs, mask=(i_offset+tl.arange(0,BLOCK_M) < N) & (k_offset+tl.arange(0,BLOCK_K) < N), other=0.0)
        b = tl.load(b_ptrs, mask=(k_offset+tl.arange(0,BLOCK_K) < N) & (j_offset+tl.arange(0,BLOCK_N) < N), other=0.0)
        # dot product
        acc += tl.dot(a, b)
    # compute final indices for output store
    c_ptrs = out_ptr + (batch_id * stride_out_b
                        + i_offset * stride_out_i
                        + j_offset * stride_out_j
                        + tl.arange(0,BLOCK_M)[:, None] * stride_out_i
                        + tl.arange(0,BLOCK_N)[None, :] * stride_out_j)

    # mask store
    mask_i = i_offset + tl.arange(0, BLOCK_M) < N
    mask_j = j_offset + tl.arange(0, BLOCK_N) < N
    mask = mask_i[:, None] & mask_j[None, :]

def _trimul_bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_dim: tl.constexpr,
    N: tl.constexpr,
    hidden_dim: tl.constexpr,
    stride_batch: tl.constexpr,
    stride_i: tl.constexpr,
    stride_k: tl.constexpr,
    # output strides
    out_stride_batch: tl.constexpr,
    out_stride_i: tl.constexpr,
    out_stride_j: tl.constexpr,
    # compile-time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # each program ID:
    pid = tl.program_id(0)
    batch_head_id = tl.program_id(1)
    # compute batch and head
    batch_idx = batch_head_id // hidden_dim
    head_idx = batch_head_id % hidden_dim

    # compute tile coordinates
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    i_tile = (pid // num_tiles_j)
    j_tile = (pid % num_tiles_j)
    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N

    # compute base pointer offsets
    batch_offset = batch_idx * stride_batch
    # head index -> offset in hidden dimension
    head_offset = head_idx  # since stride_h = 1

    # compute block loops
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # iterate over k-blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_offset = kb * BLOCK_K
        # compute the ranges
        i_range = tl.arange(0, BLOCK_M)
        j_range = tl.arange(0, BLOCK_N)
        k_range = tl.arange(0, BLOCK_K)
        # mask for a tile
        mask_i = i_offset + i_range < N
        mask_j = j_offset + j_range < N
        mask_k = k_offset + k_range < N

        # load A tile
        a_ptrs = left_ptr + batch_offset + head_offset \
                 + (i_offset + i_range)[:, None] * stride_i \
                 + (k_offset + k_range)[None, :] * stride_k

                 + (j_offset + j_range)[None, :] * stride_i

@triton.jit
def _trimul_bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_stride, i_stride, k_stride,
    out_i_stride, out_j_stride,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_K_ITER: tl.constexpr,
    NUM_TILES_J: tl.constexpr
):
    pid = tl.program_id(0)
    batch_head = tl.program_id(1)

    batch_idx = batch_head // H
    head_idx = batch_head % H

    # compute tile indices
    i_tile = pid // NUM_TILES_J
    j_tile = pid % NUM_TILES_J

    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N

    # compute indices
    i_idx = i_offset + tl.arange(0, BLOCK_M)
    j_idx = j_offset + tl.arange(0, BLOCK_N)
    # base offset for batch and head
    batch_off = batch_idx * batch_stride
    head_off = head_idx  # stride_h = 1

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k tiles
    for kk in range(NUM_K_ITER):
        k_offset = kk * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)
        # masks
        mask_i = i_idx < N
        mask_k = k_idx < N
        mask_j = j_idx < N

        # Load A tile (i, k)
        a_ptrs = left_ptr + batch_off + head_off + i_idx[:,None] * i_stride + k_idx[None,:] * k_stride
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None,:],
                    other=0.0)

        # Load B_T tile (k, j)
        b_ptrs = right_ptr + batch_off + head_off + k_idx[:,None] * k_stride + j_idx[None,:] * i_stride
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None,:],
                    other=0.0)

        acc += tl.dot(a, b)

    # store results
    c_ptrs = out_ptr + batch_off + head_off + i_idx[:, None] * out_i_stride + j_idx[None, :] * out_j_stride
    mask_store = (i_idx < N)[:, None] & (j_idx < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)

"""
TriMul “outgoing’’ forward pass (no gradients) – Triton‑accelerated.

The implementation follows the reference PyTorch code but moves the
O(N³) contraction (the einsum over the third axis) to a custom Triton
kernel that computes a batched matrix multiplication:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

All heavy work (the projection, gating and the contraction) is done in
FP16 while the surrounding LayerNorms stay in FP32 for numerical
stability.  An optional mask (+ sigmoid gates) is fused in a tiny
Triton kernel.  The final LayerNorm, out‑gate and linear projection are
performed with PyTorch for simplicity.

The kernel processes the (B·hidden) independent GEMMs in a single launch
by folding the batch and head dimensions into the Triton program ID.
Each program computes a tile of size BLOCK_M×BLOCK_N (default 64×64)
using Tensor‑Core‑friendly `tl.dot` accumulation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# 1️⃣ Mask & gate fusion (optional)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,           # *fp16  [B,N,N,H] (in‑place)
    right_ptr,          # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,      # *fp16  [B,N,N,H]
    right_gate_ptr,     # *fp16  [B,N,N,H]
    mask_ptr,           # *fp16  [B,N,N] (broadcasted over H)
    total_elements,     # i64   (B·N·N·H)
    H,                  # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # each hidden entry maps to one mask entry
    mask_offs = offs // H

    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)

# ----------------------------------------------------------------------
# 2️⃣ Batched GEMM kernel for the O(N³) contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_bmm_kernel(
    left_ptr,               # *fp16  [B,N,N,H]
    right_ptr,              # *fp16  [B,N,N,H]
    out_ptr,                # *fp16  [B,N,N,H]  (result)
    stride_batch,           # i64  stride on dim 0  (B)
    stride_i,               # i64  stride on dim 1  (i axis)
    stride_k,               # i64  stride on dim 2  (k axis)
    out_stride_i,           # i64  stride for i in output (same as stride_i)
    out_stride_j,           # i64  stride for j in output (same as stride_k)
    N: tl.constexpr,        # i32  sequence length
    H: tl.constexpr,        # i32  hidden dimension
    BLOCK_M: tl.constexpr,  # tile size in i direction
    BLOCK_N: tl.constexpr,  # tile size in j direction
    BLOCK_K: tl.constexpr,  # tile size in reduction (k) direction
):
    # ----- program IDs -------------------------------------------------
    pid = tl.program_id(0)                    # tile (i,j)
    batch_head = tl.program_id(1)            # combined batch & head index

    # split batch and head
    batch_idx = batch_head // H
    head_idx  = batch_head % H

    # number of tiles along each axis
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    # i‑ and j‑tile coordinates
    i_tile = pid // num_tiles_j
    j_tile = pid % num_tiles_j

    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N

    # linear indices for the current tile (used for masking)
    i_idx = i_offset + tl.arange(0, BLOCK_M)
    j_idx = j_offset + tl.arange(0, BLOCK_N)

    # base pointer offset for this (batch, head) pair
    batch_off = batch_idx * stride_batch
    head_off  = head_idx                      # stride_h == 1 for fp16 layout

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over k (summation axis) ---------------------------
    # loop count is static (covers the max sequence length 1024)
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_offset = kk * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)

        # mask for valid i/k coordinates
        mask_i = i_idx < N
        mask_k = k_idx < N
        mask_j = j_idx < N

        # left tile: rows i, columns k
        a_ptrs = left_ptr + batch_off + head_off \
                 + i_idx[:, None] * stride_i \
                 + k_idx[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right^T tile: rows k, columns j (notice the stride swap)
        b_ptrs = right_ptr + batch_off + head_off \
                 + k_idx[:, None] * stride_k \
                 + j_idx[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate with Tensor‑Core dot
        acc += tl.dot(a, b)

    # ----- write back -------------------------------------------------
    c_ptrs = out_ptr + batch_off + head_off \
             + i_idx[:, None] * out_stride_i \
             + j_idx[None, :] * out_stride_j
    store_mask = (i_idx < N)[:, None] & (j_idx < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)

        weights_dict : model parameters (float32)

    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=ft16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=ft16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=ft16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=ft16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=ft16)

    # --------------------- 4️⃣ Linear projections (→ [B,N,N,H]) ------------------
    left  = F.linear(x, left_proj_w)    # FP16
    right = F.linear(x, right_proj_w)   # FP16

    # --------------------- 5️⃣ Gate projections (sigmoid) ----------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------- 6️⃣ Fuse mask & gates (if needed) --------------------
    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:
        # mask is broadcast over hidden dim; everything must be contiguous
        mask_fp16 = mask_tensor.to(ft16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # release temporaries
        del left_gate, right_gate, mask_fp16

    B, N, _, H = left.shape
    assert H == hidden_dim, "Hidden dimension mismatch."

    # --------------------- 7️⃣ O(N³) contraction via Triton ----------------------
    out = torch.empty_like(left)        # [B,N,N,H] FP16

    # Strides for the (B,N,N,H) layout (contiguous in last dim)
    stride_batch = left.stride(0)   # N*N*H
    stride_i      = left.stride(1)   # N*H
    stride_k      = left.stride(2)   # H
    # Output strides (i -> stride_i, j -> stride_k)
    out_stride_i = stride_i
    out_stride_j = stride_k

    # Tiling parameters – can be tuned for the target hardware
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_x = num_tiles_i * num_tiles_j          # tiles over i/j
    grid_y = B * hidden_dim                     # (batch, head)

    _trimul_bmm_kernel[(grid_x, grid_y)](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_batch=stride_batch,
        stride_i=stride_i,
        stride_k=stride_k,
        out_stride_i=out_stride_i,
        out_stride_j=out_stride_j,
        N=N,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # 8 warps ≈ 256 threads per block – good for H100
        num_stages=2,
    )

    # --------------------- 8️⃣ Second LayerNorm (still FP16) --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------- 9️⃣ Apply out‑gate and final linear -------------------
    out = out * out_gate                     # element‑wise gating (FP16)
    out = F.linear(out, to_out_w)            # → [B,N,N,dim] (FP16)

    # --------------------- 10️⃣ Cast back to FP32 for the caller -----------------
    return out.to(torch.float32)
```
