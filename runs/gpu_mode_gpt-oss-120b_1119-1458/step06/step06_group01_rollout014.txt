"""
TriMul (outgoing) – fast forward‐only implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 → cast to FP16.
2️⃣  Fuse the five linear layers (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single matmul `x·W_comb`.  The
    first four slices are split and the gate slices are passed through
    `sigmoid`.  This reduces the number of reads of the large input
    tensor `x` from four to one.
3️⃣  If a mask is present, fuse `proj * gate * mask` with a tiny Triton
    kernel (`_mask_gate_kernel`).  When `nomask=True` we simply
    multiply the projected tensors by their gates element‑wise.
4️⃣  The O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM (`out = A @ Bᵀ`) where the batch
    dimension is `B·H`.  This re‑uses cuBLAS Tensor‑Core kernels.
5️⃣  Apply a second LayerNorm on the hidden dimension, multiply by the
    out‑gate, and project back to the model dimension.
6️⃣  Cast the result back to FP32 (expected by the harness).

Only the mask‑gate fusion is written in Triton; all heavy matmul work
leverages cuBLAS, which on an H100 reaches >100 TFLOPS for the
required FP16 tensor‑core operations.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """ out = proj * gate * mask   (mask is broadcast over hidden dim H) """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Iterate over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj[i, k, h]
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # gate[i, k, h]  (already passed through sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask   (mask broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim)

    All tensors must be contiguous fp16 cuda tensors.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes – chosen for good occupancy while keeping shared memory low
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Batched matrix multiplication for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as batched GEMM (C = A @ Bᵀ) where hidden dimension is
    folded into the batch dimension for maximal Tensor‑Core throughput.
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul with right transposed -> (N × N) @ (N × N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    w_right_gate = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Concatenate along the output‑feature axis.
    weight_combined = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )  # shape [5*hidden, dim] in FP16 and on‑device

    # Single linear pass – shape [B, N, N, 5*hidden]
    combined = torch.nn.functional.linear(x_fp16, weight_combined)

    # Split the combined tensor into its constituent pieces.
    H = hidden_dim
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = torch.split(
        combined, [H, H, H, H, H], dim=-1
    )
    # Free the large intermediate early
    del combined

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask & gate (Triton fusion) or simple element‑wise mul
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and broadcastable.
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
        # mask_fp16 can be released now
        del mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) on the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply out‑gate and final projection back to model dim
    # ------------------------------------------------------------------
    out = out * out_gate                 # element‑wise gating
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
