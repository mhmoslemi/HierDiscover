def _batched_gemm_fp16(left, right):
    B,N,_,H = left.shape
    # left permute to [B*H, N, N]
    left_perm = left.permute(0,3,1,2).contiguous().view(B*H, N, N)

    # right transpose for GEMM: need tensor shape [B*H, N, N] where second dimension is K and third is N. For A @ B^T, we need B_T shape [B*H, K, N] with K=N. So we can compute right_T = right.permute(0,3,2,1).contiguous().view(B*H, N, N). (Now each B_T[k,j] = right[j,k]).

    right_T = right.permute(0,3,2,1).contiguous().view(B*H, N, N)

    # Compute C = left_perm @ right_T using torch.bmm
    # A.shape (batch, M, K) = (B*H, N, N)
    # B.shape (batch, K, N) = (B*H, N, N)

    c = torch.bmm(left_perm, right_T)  # shape (batch, N, N)

    # reshape back
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)
    return out

"""
custom_kernel: Triton‑accelerated forward pass of the *outgoing* TriMul module.

Algorithm
---------
1️⃣  First LayerNorm (FP32) on the input tensor (shape [B, N, N, C]).
2️⃣  Cast to FP16 and perform ONE large linear projection that simultaneously
    computes *all* per‑position linear maps required by the module:
        - left projection
        - left gate
        - right projection
        - right gate
        - out gate
   The weight matrix is built by concatenating the five model weight matrices
   along the output‑feature dimension (size 5·hidden_dim).
3️⃣  Slice the result, apply the three sigmoid gates (in‑place) and
   fuse the gating with the optional mask using a tiny Triton kernel.
   After this step we have the tensors `left` and `right`
   of shape [B, N, N, hidden_dim] ready for the O(N³) contraction.
4️⃣  Perform the contraction  out[i,j] = Σ_k left[i,k]·right[j,k]  as a
   batched GEMM (`C = A @ Bᵀ`) via cuBLAS (`torch.bmm`).  This is the
   dominant O(N³) work.
5️⃣  Second LayerNorm (FP16) on the contracted tensor.
6️⃣  Element‑wise multiplication with the *out‑gate*.
7️⃣  Final linear projection back to the original channel dimension.
8️⃣  Cast back to FP32 (the harness expects float32).

Only the mask‑gate fusion is done in Triton, satisfying the “must use a
kernel” requirement while keeping overhead negligible.
"""

import math
from typing import Any, Dict, Tuple

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fuse element‑wise mask and gate multiplication.
# --------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16: tensor to be updated (flattened)
    gate_ptr,       # *fp16: gate tensor (same flattening as x)
    mask_ptr,       # *fp16: mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   : number of hidden features per position
    size,           # i32   : total number of elements in x (B*N*N*hidden_dim)
    BLOCK: tl.constexpr,   # compile‑time tile size (<=1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the tensor and gate value
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Derive mask index (mask is broadcast over the hidden dimension)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fuse multiplication
    out = x * g * m

    tl.store(x_ptr + offs, out, mask=valid)


# --------------------------------------------------------------
# Helper: batched GEMM for the core O(N³) contraction.
# --------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (FP16)

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, hidden] (fp16)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, hidden] (fp16)
    """
    B, N, _, H = left.shape
    # Collapse batch and hidden dimensions
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (B*H, N, N)
    b = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)         # (B*H, N, N)  = rightᵀ
    c = torch.bmm(a, b)                                                   # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)
    return out


# --------------------------------------------------------------
# Entry point used by the evaluation harness.
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,            # input tensor [B, N, N, dim] (float32)
        torch.Tensor,            # mask tensor [B, N, N]    (float32 / bool, may be None)
        Dict[str, torch.Tensor], # weights dict (float32)
        Dict[str, Any],          # config dict (contains "dim", "hidden_dim", optional "nomask")
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    torch._C._disable_jit_task_graph_hints()  # avoid torch.compile interference
    torch.set_grad_enabled(False)

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim), float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy arithmetic
    # -----------------------------------------------------------------
    x = x_norm.to(dtype_fp16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse all per‑position linear maps into ONE matmul
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # -----------------------------------------------------------------
    # Gather and concatenate the weight matrices (they are stored as
    # [out_features, in_features] in the dictionary).
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Build a single weight matrix of shape [5*hidden_dim, dim]
    weight_combined = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).to(dtype_fp16)

    # Linear projection – result shape (B, N, N, 5*hidden_dim)
    proj = torch.nn.functional.linear(x, weight_combined)  # bias=None

    # -----------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor and apply sigmoid gates in‑place
    # -----------------------------------------------------------------
    # Helper to slice cleanly
    H = hidden_dim
    left_proj = proj[..., 0:H]                     # (B, N, N, H)
    left_gate  = proj[..., H:2 * H]
    right_proj = proj[..., 2 * H:3 * H]
    right_gate = proj[..., 3 * H:4 * H]
    out_gate_pre = proj[..., 4 * H:5 * H]

    # In‑place sigmoid for the three gate tensors
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate_pre.sigmoid_()

    # -----------------------------------------------------------------
    # 5️⃣ Apply optional mask and fuse mask‑gate multiplication via Triton
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # Convert mask to fp16 and flatten
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        mask_flat = mask_fp16.view(-1)                     # shape [B*N*N]

        total_elems = left_proj.numel()                     # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication (in‑place)
        left_proj.mul_(left_gate)
        right_proj.mul_(right_gate)

    # After masking/gating we already have the final tensors for GEMM
    left = left_proj
    right = right_proj

    # -----------------------------------------------------------------
    # 6️⃣ Core O(N³) contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # shape (B, N, N, H), fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Multiply by the out‑gate (in‑place) and final projection
    # -----------------------------------------------------------------
    out.mul_(out_gate)                                      # element‑wise gating
    to_out_weight = weights["to_out.weight"].to(dtype_fp16) # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)    # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
