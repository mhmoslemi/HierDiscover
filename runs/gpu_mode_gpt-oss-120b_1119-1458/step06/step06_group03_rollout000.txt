The kernel definition:

def fused_gate_mask_kernel(
    x_ptr,
    gate_ptr,
    mask_ptr,
    total_elts: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elts

    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=valid)
```

"""
TriMul (outgoing) – fast mixed‑precision forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input tensor in FP32 (numerically stable) and cast to FP16.
2️⃣ Linear projections (`left_proj`, `right_proj`) and three gating linear layers are
   performed in FP16.  The three element‑wise factors (projection, gate, mask)
   are fused for the left/right branches with a tiny Triton kernel
   (`fused_gate_mask_kernel`).  When no mask is provided we simply multiply by
   the gates.
3️⃣ The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is reduced to a batched GEMM: for each hidden channel `h` we compute
   `left_{b,h} @ right_{b,h}.T`.  This is performed by `torch.bmm`, which on an
   H100 invokes a Tensor‑Core kernel (batch = B·H, matmul shape = N×N).
4️⃣ A second Layer‑Norm normalises the hidden dimension; the output‑gate
   (sigmoid of a linear) is applied and a final linear projection maps the hidden
   dimension back to `dim`.
5️⃣ Result is cast back to FP32 for the harness.

Only the forward pass (no autograd) is required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (B,N,N) – no hidden dim
    total_elts: tl.int32, # total number of elements in x/gate = B*N*N*H
    hidden_dim: tl.int32, # hidden dimension (= size of last axis of x)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    """x ← x * gate * mask (mask broadcast over hidden_dim)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    valid = offs < total_elts

    # Locate the corresponding mask element (mask has no hidden dimension)
    mask_idx = offs // hidden_dim                     # integer division

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=valid)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a batched GEMM.

    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H] in FP16.

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] in FP16.
    """
    B, N, _, H = left.shape
    # Collapse batch & hidden dimensions: (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Matrix multiply with the second operand transposed
    c = torch.bmm(a, b.transpose(1, 2))
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) – ignored when ``nomask`` is True
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack inputs & config
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True ⇒ no mask or mask all‑ones

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (FP32) – keep high precision for mean/var
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to FP16 for the heavy part of the network
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)   # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 4️⃣ Cast all model weights once to FP16 (biases are absent)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 6️⃣ Gating signals (FP16) and apply optional mask via Triton kernel
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    if not nomask and mask_tensor is not None:
        # Broadcast mask across hidden_dim; store as FP16 for the kernel.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()

        total_elts = left.numel()          # B * N * N * hidden_dim
        BLOCK = 1024                       # ≤ max threads per block (1024)
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # Fuse (proj * gate * mask) for left branch
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse (proj * gate * mask) for right branch
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 🔟 Return as float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
