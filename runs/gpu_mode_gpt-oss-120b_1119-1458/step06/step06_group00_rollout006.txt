import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: fuse mask, left_gate and right_gate with left_proj and right_proj.
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,            # *fp16 [B,N,N,H] (in-place)
    right_ptr,           # *fp16 [B,N,N,H] (in-place)
    left_gate_ptr,       # *fp16 [B,N,N,H]
    right_gate_ptr,      # *fp16 [B,N,N,H]
    mask_ptr,            # *fp16 [B,N,N] (broadcasted)
    total_elements,      # i64 total number of elements in left/right (B*N*N*H)
    H,                   # i32 hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    """Apply per-pair mask (if present) and elementwise sigmoid gates to left/right
    projections. Each hidden channel has the same mask element, so the mask index
    is `idx // H`."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # mask index = element index divided by hidden dimension
    mask_offs = offs // H

    # load values
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # mask broadcasted over hidden dimension (1.0 for OOB)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # fuse mask and gates
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Steps:
    1. LayerNorm on the input (float32).
    2. Cast to fp16 and perform a *single* large Linear that computes
       (left_proj, right_proj, left_gate, right_gate, out_gate) together.
    3. Sigmoid gates for the three gating vectors.
    4. Optional mask‑and‑gate fusion (Triton kernel) that multiplies
       left_proj * left_gate * mask and right_proj * right_gate * mask.
    5. Batched matmul across the sequence dimension (the core N³ update) using
       torch.bmm on a flattened (batch * hidden_dim) batch dimension.
    6. Second LayerNorm on the hidden representation.
    7. Apply out_gate and final Linear projection back to dim.
    8. Return result in float32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # True → skip mask handling

    device = input_tensor.device
    fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # layer_norm returns same dtype as input (float32)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent heavy work
    # --------------------------------------------------------------
    x_fp16 = x.to(fp16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single matmul
    # --------------------------------------------------------------
    # Gather weights and cast once
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # Stack vertically: (5*hidden_dim, dim)
    w_combined = torch.cat([
        w_left_proj,
        w_right_proj,
        w_left_gate,
        w_right_gate,
        w_out_gate,
    ], dim=0).to(device=device, dtype=fp16)

    # Single large linear: shape (..., 5*hidden_dim)
    y = F.linear(x_fp16, w_combined)  # fp16

    # --------------------------------------------------------------
    # 4️⃣ Split into the individual tensors
    # --------------------------------------------------------------
    # y shape: [B, N, N, 5*hidden_dim]
    splits = torch.split(y, hidden_dim, dim=-1)  # list of length 5
    left_proj = splits[0]   # [B,N,N,hidden_dim]
    right_proj = splits[1]  # [B,N,N,hidden_dim]
    left_gate_raw = splits[2]  # [B,N,N,hidden_dim]
    right_gate_raw = splits[3]  # [B,N,N,hidden_dim]
    out_gate_raw = splits[4]  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 5️⃣ Compute the sigmoid gates (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)  # will be used after second LN

    # --------------------------------------------------------------
    # 6️⃣ Fuse mask (if needed) and element‑wise gate multiplication
    # --------------------------------------------------------------
    if nomask:
        # Simple elementwise multiplication without mask
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure tensors are contiguous for Triton
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(fp16).contiguous()  # shape [B,N,N]
        total = left.numel()
        BLOCK = 1024  # tuneable, ~1K threads per block
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # After kernel, left/right contain the gated & masked values
        # No need to keep the gate tensors further
        del left_gate, right_gate, mask_fp16

    # --------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction (batched GEMM)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Bring hidden dimension into batch to reuse torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: each head computes A @ B^T over the second axis
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Restore original 4‑D shape: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still fp16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # --------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate  # element‑wise gating
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)  # shape [B, N, N, dim], fp16

    # --------------------------------------------------------------
    # 10️⃣ Cast back to float32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – high‑performance implementation.

Algorithm
---------
1️⃣  LayerNorm on the raw input (float32 → stable statistics).
2️⃣  Cast to fp16 and perform a *single* large Linear that simultaneously computes
    the five required projections:
        left_proj, right_proj, left_gate, right_gate, out_gate.
3️⃣  Apply sigmoid to the three gating tensors.
4️⃣  Fuse optional mask, left‑gate, and right‑gate in‑place with a Triton kernel:
        left = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
    (If `nomask=True` the mask step is omitted and only the element‑wise gate is applied.)
5️⃣  Core O(N³) update – batched matrix multiplication:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    Implemented as a batched GEMM on a flattened batch dimension.
6️⃣  Second LayerNorm on the hidden representation (fp16).
7️⃣  Apply the out‑gate (element‑wise) and a final Linear projection back to `dim`.
8️⃣  Cast the result to float32 for the caller.

All heavy arithmetic is carried out in fp16 and the only Triton kernel used
is the mask‑gate fusion kernel, which eliminates an extra memory pass when a mask
is present.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: mask + gate fusion (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,            # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,           # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,       # *fp16  [B,N,N,H]
    right_gate_ptr,      # *fp16  [B,N,N,H]
    mask_ptr,            # *fp16  [B,N,N]    (broadcasted over H)
    total_elements,      # i64   total number of fp16 elements in left/right
    H,                   # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    """
    For each hidden element:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    The mask is broadcast over the hidden dimension; each mask entry
    corresponds to H consecutive FP16 entries.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # flat index
    valid = offs < total_elements

    # derive mask index (same mask for H hidden entries)
    mask_offs = offs // H

    # load values (OOB → zero)
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    # load mask (OOB → neutral 1.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # fused operation
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward (no autograd).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim]          (float32)
        mask_tensor  : torch.Tensor [B, N, N] (bool / 0‑1)  (float32 or bool)
        weights_dict : mapping name → torch.Tensor (float32)
        config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # True → mask is ignored

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 – stable statistics)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)                     # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(fp16)                                   # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Collect and cast to fp16 in one go
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    w_combined = torch.cat([
        w_left_proj,
        w_right_proj,
        w_left_gate,
        w_right_gate,
        w_out_gate,
    ], dim=0).to(device=device, dtype=fp16)          # (5*H, dim)

    # Single large linear: produces all five tensors in one pass
    y = F.linear(x, w_combined)                     # [B,N,N,5*H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Split the combined output
    # ------------------------------------------------------------------
    splits = torch.split(y, hidden_dim, dim=-1)
    left_proj   = splits[0]   # [B,N,N,H]
    right_proj  = splits[1]   # [B,N,N,H]
    left_gate_raw   = splits[2]   # [B,N,N,H]
    right_gate_raw  = splits[3]   # [B,N,N,H]
    out_gate_raw    = splits[4]   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)          # used after the 2nd LN

    # ------------------------------------------------------------------
    # 6️⃣ Mask + gate fusion (Triton) or simple element‑wise multiply
    # ------------------------------------------------------------------
    if nomask:
        # No mask – just element‑wise gate
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure tensors are contiguous for the Triton kernel
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(fp16).contiguous()   # [B,N,N]

        total = left.numel()
        BLOCK = 1024                                     # ~1 K threads per block
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporary tensors early
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction: batched GEMM across the hidden dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Move hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left_perm.view(B * H, N, N)            # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)           # (B*H, N, N)

    # Batched matrix multiplication: left @ rightᵀ (per hidden head)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Restore original 4‑D layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise gate (fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)                # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
