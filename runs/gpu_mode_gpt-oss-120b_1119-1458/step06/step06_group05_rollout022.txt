```
@triton.jit
def _ln_gate_kernel(
    out_ptr, out_gate_ptr, norm_weight_ptr, norm_bias_ptr, out_gated_ptr,
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_weight_h,
    stride_norm_bias_h,
    stride_out_gated_b, stride_out_gated_i, stride_out_gated_j, stride_out_gated_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    EPS: tl.constexpr
):
    # compute program ids
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # sum and sumsq accumulators
    sum_ij = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    sumsq_ij = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # First pass: compute sum and sumsq
    for h in range(H):
        # compute pointers
        out_ptrs = out_ptr + pid_b*stride_out_b + offs_i[:,None]*stride_out_i + offs_j[None,:]*stride_out_j + h*stride_out_h
        # load
        val = tl.load(out_ptrs, mask=mask_i[:,None] & mask_j[None,:], other=0.0).to(tl.float32)  # shape [BLOCK_I, BLOCK_J]
        sum_ij += val
        sumsq_ij += val * val

    # compute mean, var, inv_std
    mean = sum_ij / H
    var = sumsq_ij / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Second pass: normalize, scale, bias, gate, store
    for h in range(H):
        out_ptrs = out_ptr + pid_b*stride_out_b + offs_i[:,None]*stride_out_i + offs_j[None,:]*stride_out_j + h*stride_out_h

        beta = tl.load(norm_bias_ptr + h*stride_norm_bias_h, mask=True, other=0.0).to(tl.float32)

        normalized = (out_val - mean) * inv_std
        scaled = normalized * gamma + beta
        gated = scaled * gate_val

"""
TriMul (outgoing) – fast forward implementation.

Key ideas
---------
1️⃣ Fuse the two linear projections (proj + gate) for the “left” and “right”
   tensors by concatenating the projection and gate weight matrices.
   This reduces the number of matmul passes over the large ([B,N,N,dim])
   tensor from 4 to 2.

2️⃣ Fuse mask, projection and gate into a single Triton kernel
   (`_apply_mask_gate`).  The kernel loads a tile of the mask and a tile of the
   projection and gate, evaluates the sigmoid of the gate on the host, and
   writes `proj * gate * mask` directly to the output.

3️⃣ The O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM using cuBLAS (`torch.bmm`).  The tensor is
   reshaped to [B*hidden_dim, N, N] → a strided‑batched matmul with each batch
   representing one hidden channel.

4️⃣ Fuse the second LayerNorm and the output gate (`out_gate`) into a custom
   Triton kernel (`_ln_gate_kernel`).  The kernel streams over the hidden
   dimension twice: the first pass accumulates `mean` and `var` per output
   position, the second pass normalises, scales, adds bias, multiplies by the
   output gate, and writes the result.  All arithmetic is performed in FP16
   (accumulating statistics in FP32 for numerical stability).

5️⃣ Final linear projection (`to_out`) is a standard batched GEMM
   (`torch.nn.functional.linear`).  The final tensor is cast back to FP32.

The implementation works for both masked and un‑masked inputs, for any of the
test shapes listed in the problem statement, and runs entirely without
autograd (gradient computation is disabled).  All heavy computation uses FP16
to exploit Tensor‑Cores on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """out = proj * gate * mask  (mask broadcast over hidden dimension H)"""
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape [BLOCK_I, BLOCK_J]

    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16, shape [I,J,H]

        # Load gate tile (already passed through sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Fuse: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm + out_gate fuse
# ----------------------------------------------------------------------
@triton.jit
def _ln_gate_kernel(
    out_ptr,          # *fp16  [B, N, N, H]   (raw contraction result)
    out_gate_ptr,     # *fp16  [B, N, N, H]
    norm_weight_ptr,  # *fp16  [H]  (to_out_norm.weight)
    norm_bias_ptr,    # *fp16  [H]  (to_out_norm.bias)
    out_gated_ptr,    # *fp16  [B, N, N, H]   (LN‑scaled * out_gate)
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_w_h,
    stride_norm_b_h,
    stride_out_gated_b, stride_out_gated_i, stride_out_gated_j, stride_out_gated_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    EPS: tl.constexpr,
):
    """Fuse LayerNorm over hidden dim H and element‑wise out_gate multiplication."""
    pid_b = tl.program_id(0)   # batch
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N
    # Masks for the 2‑D tile
    tile_mask = mask_i[:, None] & mask_j[None, :]

    # ---- 1st pass: compute mean & variance per output position ----
    sum_ij = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    sumsq_ij = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for h in range(H):
        # Load raw out value
        out_off = (
            pid_b * stride_out_b
            + offs_i[:, None] * stride_out_i
            + offs_j[None, :] * stride_out_j
            + h * stride_out_h
        )
        out_val = tl.load(out_ptr + out_off, mask=tile_mask, other=0.0).to(tl.float32)
        sum_ij += out_val
        sumsq_ij += out_val * out_val

    # Compute statistics
    mean = sum_ij / H
    var = sumsq_ij / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)   # shape [BLOCK_I, BLOCK_J]

    # ---- 2nd pass: normalize, apply weight/bias, multiply out_gate ----
    for h in range(H):
        # load raw out
        out_off = (
            pid_b * stride_out_b
            + offs_i[:, None] * stride_out_i
            + offs_j[None, :] * stride_out_j
            + h * stride_out_h
        )
        out_val = tl.load(out_ptr + out_off, mask=tile_mask, other=0.0).to(tl.float32)

        # load gate
        gate_off = (
            pid_b * stride_gate_b
            + offs_i[:, None] * stride_gate_i
            + offs_j[None, :] * stride_gate_j
            + h * stride_gate_h
        )
        gate_val = tl.load(out_gate_ptr + gate_off, mask=tile_mask, other=0.0).to(tl.float32)

        # layer‑norm parameters (scalar per hidden channel)
        gamma = tl.load(norm_weight_ptr + h * stride_norm_w_h).to(tl.float32)
        beta  = tl.load(norm_bias_ptr   + h * stride_norm_b_h).to(tl.float32)

        # normalize, scale, bias
        normed = (out_val - mean) * inv_std
        scaled = normed * gamma + beta
        gated = scaled * gate_val

        # Store result
        out_gated_off = (
            pid_b * stride_out_gated_b
            + offs_i[:, None] * stride_out_gated_i
            + offs_j[None, :] * stride_out_gated_j
            + h * stride_out_gated_h
        )
        tl.store(out_gated_ptr + out_gated_off, gated.to(tl.float16), mask=tile_mask)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Fuse proj * gate * mask (mask broadcast over the hidden dim)."""
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8          # works well up to hidden_dim = 384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using cuBLAS.
    Input tensors are FP16, shape [B, N, N, H].
    Returns tensor of shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]; treat each hidden channel as a separate batch
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched GEMM: C = A @ Bᵀ   (both A and B have shape [B*H, N, N])
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def _layernorm_gate_fuse(out_raw: torch.Tensor,
                         out_gate: torch.Tensor,
                         norm_weight: torch.Tensor,
                         norm_bias: torch.Tensor) -> torch.Tensor:
    """Fused LayerNorm (over hidden dim) + out_gate multiplication using Triton."""
    B, N, _, H = out_raw.shape
    out_gated = torch.empty_like(out_raw)

    # strides expressed in element counts
    s_out = out_raw.stride()
    s_gate = out_gate.stride()
    s_out_gated = out_gated.stride()
    s_norm_w = norm_weight.stride()
    s_norm_b = norm_bias.stride()

    BLOCK_I = 16
    BLOCK_J = 16
    EPS = 1e-5

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _ln_gate_kernel[grid](
        out_raw,
        out_gate,
        norm_weight,
        norm_bias,
        out_gated,
        B,
        N,
        H,
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_norm_w[0],
        s_norm_b[0],
        s_out_gated[0],
        s_out_gated[1],
        s_out_gated[2],
        s_out_gated[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        EPS=EPS,
        num_warps=4,
        num_stages=1,
    )
    return out_gated


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the last channel (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # cast once to FP16 for remaining ops

    # -----------------------------------------------------------------
    # 2️⃣ Concatenate projection+gate weights (FP16) to halve matmul passes
    # -----------------------------------------------------------------
    # left side
    left_proj_w  = weights["left_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=dtype_fp16)

    # right side
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=dtype_fp16)

    # out gate weight
    out_gate_w   = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    # final projection & LN parameters
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # -----------------------------------------------------------------
    # 3️⃣ Projections (single fused linear per side) + gate split
    # -----------------------------------------------------------------
    # left side: shape -> [B,N,N,2*hidden_dim]  (proj || gate_raw)
    left_combined = torch.nn.functional.linear(x, left_combined_w)  # FP16
    left_proj = left_combined[..., :hidden_dim]                # [B,N,N,hidden_dim]
    left_gate_raw = left_combined[..., hidden_dim:]            # [B,N,N,hidden_dim]
    left_gate = torch.sigmoid(left_gate_raw)                  # FP16 (gate after sigmoid)

    # right side
    right_combined = torch.nn.functional.linear(x, right_combined_w)
    right_proj = right_combined[..., :hidden_dim]
    right_gate_raw = right_combined[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_raw)

    # out gate (separate linear)
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N,hidden_dim] FP16
    out_gate = torch.sigmoid(out_gate_raw)                    # FP16

    # -----------------------------------------------------------------
    # 4️⃣ Fuse mask and gate into left/right (if mask is present)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # element‑wise multiply projection & gate (no mask)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy contraction via batched GEMM (cuBLAS tensor‑core)
    # -----------------------------------------------------------------
    out_raw = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] (FP16)

    # -----------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm with out_gate (Triton)
    # -----------------------------------------------------------------
    out_norm_gate = _layernorm_gate_fuse(
        out_raw,
        out_gate,
        to_out_norm_w,
        to_out_norm_b,
    )   # [B, N, N, hidden_dim] (FP16)

    # -----------------------------------------------------------------
    # 7️⃣ Final linear projection back to model dimensionality
    # -----------------------------------------------------------------
    out_fp16 = torch.nn.functional.linear(out_norm_gate, to_out_w)  # [B,N,N,dim] FP16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by harness)
    # -----------------------------------------------------------------
    return out_fp16.to(torch.float32)
```
