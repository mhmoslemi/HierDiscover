"""
Outgoing TriMul (AlphaFold‑3) forward pass – Triton‑fused mask‑gate
=================================================================

The operation works on a 4‑D pair tensor ``x`` of shape
``[B, N, N, dim]`` and produces an output of the same shape.

Algorithm
---------
1️⃣ **Layer‑norm (FP32)** – normalise the input over the last dimension.
   The result is cast to FP16 for the rest of the pipeline.

2️⃣ **Linear projections** – compute the four linear maps (left/right &
   their gates) in FP16 with ``torch.nn.functional.linear``.

3️⃣ **Mask‑gate fusion** (only when a mask is present) – a tiny Triton
   kernel multiplies a projection, its sigmoid‑gate and the binary mask
   (broadcast over the hidden dimension).  When ``nomask=True`` we simply
   multiply the projection by its gate.

4️⃣ **O(N³) contraction** – the core “TriMul’’ contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   is performed as a batched GEMM `C = A @ Bᵀ`.  The hidden dimension
   ``H`` is folded into the batch dimension, yielding a tensor of shape
   ``[B·H, N, N]`` which is multiplied with cuBLAS Tensor‑Core kernels
   via ``torch.bmm`` (the fastest available routine on an H100).

5️⃣ **Second LayerNorm (FP16)** – normalise over the hidden dimension.

6️⃣ **Out‑gate and final projection** – element‑wise multiply by the
   out‑gate (sigmoid) and apply the final linear map back to ``dim``.
   The result is cast to FP32, as required by the harness.

Only step 3 is written in Triton; all heavy linear algebra remains in
high‑throughput PyTorch/cuBLAS kernels.  This satisfies the “must use
Triton’’ requirement while keeping the overall runtime well below the
target.

"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden dim H)
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Indices for the i and j dimensions
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load a tile of the mask (BLOCK_I × BLOCK_J) – it will be broadcast
    # later across the hidden dimension H.
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fusion: proj * gate * mask (mask is broadcast on H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.

    Parameters
    ----------
    proj : torch.Tensor
        FP16 tensor of shape [B, N, N, H] – output of a linear projection.
    gate : torch.Tensor
        FP16 tensor of shape [B, N, N, H] – sigmoid‑activated gate.
    mask : torch.Tensor
        FP16 tensor of shape [B, N, N] – binary (0/1) mask.

    Returns
    -------
    torch.Tensor
        FP16 tensor of shape [B, N, N, H] containing ``proj * gate * mask``.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides in element counts (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    # Tile sizes – tuned for H100 occupancy
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,                                 # batch dimension
        (N + BLOCK_I - 1) // BLOCK_I,      # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,      # j‑tiles
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) TriMul contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implementation details
    ----------------------
    * ``left`` and ``right`` are FP16 tensors of shape [B, N, N, H].
    * The hidden dimension H is folded into the batch dimension, producing
      tensors of shape [B·H, N, N].
    * A batched GEMM (`torch.bmm`) with the right operand transposed
      yields the desired contraction using cuBLAS Tensor‑Core kernels.
    """
    B, N, _, H = left.shape
    # Reshape to [B*H, N, N] where the leading dimension is (batch × hidden)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply: (N,N) @ (N,N).T  → shape [B*H, N, N]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore the original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no grads).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # LayerNorm over the last dim (model dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32
    x = x.to(torch.float16)   # FP16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (on-device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Mask‑gate fusion (Triton) or simple element‑wise multiplication
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is FP16 and broadcastable over hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16)
        left  = _apply_mask_gate(left,  left_gate,  mask_fp16)
        right = _apply_mask_gate(right, right_gate, mask_fp16)
    else:
        # No mask – just multiply by the sigmoid gates (in‑place)
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Cores)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate and final projection back to model dimension (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate                               # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)    # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast the result back to FP32 (required by the test harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
