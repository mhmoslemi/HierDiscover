def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,

    stride_cm, stride_cn,

```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aba, stride_a_row, stride_a_col,  # a_batch, a_row, a_col
    stride_bba, stride_b_row, stride_b_col,  # b_batch, b_row, b_col
    stride_cca, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_idx * stride_aba + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # b: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch_idx * stride_bba + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + batch_idx * stride_cca + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A
        a_ptrs = a_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B
        b_ptrs = b_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – fast mixed‑precision forward pass.

Algorithm
---------
1️⃣  LayerNorm on the 4‑D input (FP32 → FP16 for speed).
2️⃣  Linear projections (left/right) and three gating Linear layers (all FP16).
3️⃣  Fuse element‑wise multiplication by the corresponding gates and optional mask
    in a Triton kernel (`fused_gate_mask_kernel`).  This avoids a separate
    element‑wise kernel and a memory pass.
4️⃣  Heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a *single* batched FP16 GEMM with a custom Triton
    kernel (`batched_matmul_kernel`).  The hidden dimension `h` is flattened
    into the batch dimension, so the kernel launches one 3‑D grid:
    (batch = B·H, tiles_i, tiles_j).  Accumulation is done in FP32 and
    the result is written back in FP16.
5️⃣  Second LayerNorm (FP16) on the hidden dimension, then gating by the
    pre‑computed `out_gate`.
6️⃣  Final Linear projection back to the original channel dimension.
7️⃣  Cast the result back to FP32 for the harness.

All heavy compute (mask‑gate fusion + batched GEMM) is performed in Triton,
while lightweight linear / norm ops use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # hidden dimension (= size of last axis of x)
    BLOCK: tl.constexpr,  # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elts

    # broadcast mask over hidden_dim
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default 1 → no mask

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM (C = A @ B^T) where each hidden channel is
#          flattened into the batch dimension.
#          Shapes:
#             A : [batch, M, K]   (fp16)
#             B : [batch, K, N]   (fp16)
#             C : [batch, M, N]   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B^T for many independent batch matrices.
    """
    batch_idx = tl.program_id(0)          # 0 … batch‑1
    pid_m = tl.program_id(1)              # tile along M (i‑dimension)
    pid_n = tl.program_id(2)              # tile along N (j‑dimension)

    # ---- compute tile offsets ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the reduction axis K ------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch_idx, offs_m, offs_k]  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = (a_ptr
                  + batch_idx * stride_a_batch
                  + offs_m[:, None] * stride_a_row
                  + offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B[batch_idx, offs_k, offs_n]  -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = (b_ptr
                  + batch_idx * stride_b_batch
                  + offs_k[:, None] * stride_b_row
                  + offs_n[None, :] * stride_b_col)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot accumulation
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) FP32

    # ---- write result -------------------------------------------------------
    c_ptrs = (c_ptr
              + batch_idx * stride_c_batch
              + offs_m[:, None] * stride_c_row
              + offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (FP32)

    # --------------------------------------------------------------
    # 2️⃣ cast to FP16 for the heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ cast model weights to FP16 once
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ linear projections
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim] (fp16)
    right = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5️⃣ gating signals (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6️⃣ fuse (proj * gate * mask) in‑place via Triton (if mask exists)
    # --------------------------------------------------------------
    if not nomask:
        # mask broadcast over hidden_dim → FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()
        # block size ≤ max threads per block (1024)
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # no mask → simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ O(N³) contraction via a single batched GEMM Triton kernel
    # --------------------------------------------------------------
    B, N, _, H = left.shape      # H == hidden_dim
    batch = B * H                # flattened batch dimension

    # permute to (B, H, N, N) and make contiguous
    # left: (i,k) ; right: (k,j) after `right.permute(0,3,2,1)`
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # flatten batch & hidden into a single batch dimension
    left_flat  = left_perm.view(batch, N, N)    # [batch, M, K] where M=N, K=N
    right_flat = right_perm.view(batch, N, N)   # [batch, K, N]

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)      # [batch, M, N] fp16

    # strides (in elements) for the flattened tensors
    stride_a_batch = left_flat.stride(0)
    stride_a_row   = left_flat.stride(1)   # stride over M (i)
    stride_a_col   = left_flat.stride(2)   # stride over K (k)

    stride_b_batch = right_flat.stride(0)
    stride_b_row   = right_flat.stride(1)   # stride over K (k)
    stride_b_col   = right_flat.stride(2)   # stride over N (j)

    stride_c_batch = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)    # stride over M (i)
    stride_c_col   = out_flat.stride(2)    # stride over N (j)

    # launch parameters (tuned for H100 Tensor‑Cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        batch,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,    # H100 works best with 8 warps for 128×128 tiles
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ gating by out_gate and final projection
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # --------------------------------------------------------------
    # 🔟 cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
