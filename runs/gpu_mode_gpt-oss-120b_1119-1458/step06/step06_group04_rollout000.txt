"""
TriMul outgoing forward pass (no autograd) – Triton‑fused mask+gate.

The implementation follows the reference PyTorch module:

1️⃣ First LayerNorm (float32) → cast to fp16.
2️⃣ Linear projections (left/right) and three gating MLPs (left/right/out) in fp16.
3️⃣ Optional mask is fused with the element‑wise product
   (tensor * gate * mask) via a tiny Triton kernel (`mask_gate_fuse_kernel`).
   This satisfies the “use Triton” requirement while keeping launch overhead tiny.
4️⃣ Core O(N³) contraction
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (B·H independent matrix multiplies) using
   fp16 tensor cores (`torch.bmm`).  This is essentially the same as the
   `einsum('... i k d, ... j k d -> ... i j d')` from the reference.
5️⃣ Second LayerNorm (fp16) → element‑wise multiplication with `out_gate`.
6️⃣ Final linear projection back to the original channel dimension.
7️⃣ Cast the result back to float32 (the harness expects float32).

All heavy arithmetic runs in fp16 to exploit Tensor‑Cores on an H100.
The mask‑gate fusion is the only part written in Triton.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask‑gate‑tensor multiplication.
# Input tensors are flattened fp16 arrays; mask has no hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16   → tensor to be updated (flattened)
    gate_ptr,       # *fp16   → gate tensor (same flattening as x)
    mask_ptr,       # *fp16   → mask tensor flattened (shape [B*N*N])
    hidden_dim,     # i32    → size of the hidden dimension (H)
    size,           # i32    → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    """
    x = x * gate * mask   (in‑place)
    mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Compute the index for the mask (mask does not have hidden dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default: 1.0

    # Fuse the three multiplications
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=valid)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Both inputs are fp16 tensors of shape [B, N, N, H].

    The implementation reshapes the tensors so that each (b, h) pair
    becomes a separate matrix multiply A @ Bᵀ where A and B are
    (N × N) matrices.  This can be performed efficiently by a single
    batched `torch.bmm` call which uses cuBLAS Tensor‑Cores.
    """
    B, N, _, H = left.shape
    # (B, N, N, H) → (B*H, N, N)   (permute to bring H next to batch)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Compute A @ Bᵀ for each batch element
    c = torch.bmm(a, b.transpose(1, 2))          # [B*H, N, N]

    # Reshape back to (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor   [B, N, N, dim] (float32)
        torch.Tensor,          # mask tensor    [B, N, N] (float32/bool) or None
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights_dict : mapping of weight names → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → ignore mask even if supplied

    device = input_tensor.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast static model weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(dtype_fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear implements x @ w.T + bias (bias is disabled)
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Gating signals (fp16) – sigmoid activations
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Apply optional mask and fuse with gating using Triton
    # ------------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # Ensure mask is float16 with shape [B, N, N]
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        mask_flat = mask_fp16.view(-1)                       # [B*N*N]
        total_elems = left.numel()                           # B*N*N*hidden_dim
        BLOCK = 1024                                          # tile size (≤1024)
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – plain element‑wise multiplication with gates
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction via batched GEMM (fp16)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final out‑gate multiplication and linear projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate                           # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness.
    # ------------------------------------------------------------------
    return out.to(dtype_fp32)
