"""
TriMul (outgoing) – Triton‑fused forward pass.

Algorithm
---------
1️⃣ First LayerNorm (FP32) → cast to FP16 for the heavy part.
2️⃣ Fuse the two projection Linear layers (proj & gate) for *left* and *right*
   into two concatenated matmuls.  After a single `torch.nn.functional.linear`
   we split the result, apply a sigmoid to the gate part and keep the
   projection part.
3️⃣ If a mask is present we fuse the three element‑wise multiplications
   (proj * gate * mask) for *both* left and right in a single Triton kernel.
   This saves one extra kernel launch and one load of the mask.
4️⃣ Heavy O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication `left @ rightᵀ`.  We reshape the tensors
   to (B·H, N, N) (H = hidden_dim) and use `torch.bmm`, which maps to an
   H100 Tensor‑Core kernel.
5️⃣ Second LayerNorm (FP16), out‑gate (sigmoid) and final Linear projection
   back to `dim`.
6️⃣ Return FP32 for the harness.

Only the forward pass is implemented – no autograd is required.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) for both left and right tensors.
# The mask is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_double_kernel(
    left_ptr, left_gate_ptr,        # *fp16: left tensor & its gate
    right_ptr, right_gate_ptr,      # *fp16: right tensor & its gate
    mask_ptr,                       # *fp16: mask (no hidden dim)
    total_elts: tl.int32,          # total number of elements per tensor (B·N·N·H)
    hidden_dim: tl.int32,          # hidden dimension H
    BLOCK: tl.constexpr,           # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                 # linear element ids
    valid = offs < total_elts

    # mask is indexed without the hidden dimension
    mask_idx = offs // hidden_dim

    # load operands
    l_val   = tl.load(left_ptr + offs,    mask=valid, other=0.0)
    lg_val  = tl.load(left_gate_ptr + offs,mask=valid, other=0.0)
    r_val   = tl.load(right_ptr + offs,   mask=valid, other=0.0)
    rg_val  = tl.load(right_gate_ptr + offs,mask=valid, other=0.0)
    m_val   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default 1 → no mask

    # fused update
    tl.store(left_ptr  + offs, l_val * lg_val * m_val, mask=valid)
    tl.store(right_ptr + offs, r_val * rg_val * m_val, mask=valid)


# ----------------------------------------------------------------------
# Batched GEMM (left @ rightᵀ) – uses torch.bmm (highly‑optimized cuBLAS)
# ----------------------------------------------------------------------
def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the heavy contraction out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (FP16).  H = hidden_dim.

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) – each hidden channel becomes a separate batch entry
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)        # (B·H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)       # (B·H, N, N)
    # batched matrix multiply with transposed right matrix
    c = torch.bmm(a, b.transpose(1, 2))                               # (B·H, N, N)
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()        # (B, N, N, H)
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data :
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool); ignored if ``nomask`` is True
        - weights_dict  : mapping from weight name → torch.Tensor (float32)
        - config_dict   : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, on the same device as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – more stable, then cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  – FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the performance‑critical part
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)   # [B, N, N, dim]  – FP16

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections & gating – fuse proj+gate via concatenated weights
    # ------------------------------------------------------------------
    # Cast all linear weights once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # Fuse left projection + left gate (2·hidden_dim output)
    left_weight_cat = torch.cat([left_proj_w, left_gate_w], dim=0)          # (2·H, dim)
    left_cat = torch.nn.functional.linear(x, left_weight_cat)              # (B,N,N,2·H)
    left_proj, left_gate_pre = left_cat.split([hidden_dim, hidden_dim], dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)                                # (B,N,N,H)

    # Fuse right projection + right gate
    right_weight_cat = torch.cat([right_proj_w, right_gate_w], dim=0)       # (2·H, dim)
    right_cat = torch.nn.functional.linear(x, right_weight_cat)
    right_proj, right_gate_pre = right_cat.split([hidden_dim, hidden_dim], dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)                              # (B,N,N,H)

    # Out‑gate (separate)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – fuse (proj * gate * mask) in‑place via a Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dimension → convert once to fp16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()               # B·N·N·H
        BLOCK = 1024                                 # ≤ max threads per block on H100
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        fused_gate_mask_double_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel the tensors already contain the masked & gated values
        left = left_proj
        right = right_proj
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core on H100)
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)      # (B, N, N, H)  – FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) + out‑gate + final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                               # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)    # (B, N, N, dim) – FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
