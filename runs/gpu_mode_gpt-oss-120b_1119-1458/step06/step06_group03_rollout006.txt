"""
TriMul (outgoing) – fused‑linear, fused‑gate‑mask and batched GEMM implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32 → FP16).
2️⃣  Fuse the five linear layers (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single matrix‑multiply with a weight of shape
    (5·hidden_dim, dim).  Slice the result and apply sigmoid to the three gate
    tensors.
3️⃣  Fuse element‑wise mask (if present) with the left/right projections and
    the corresponding gates via a tiny Triton kernel.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM using torch.bmm (Tensor‑Core friendly):
        a = left.permute(0,3,1,2).contiguous().view(B·H, N, N)
        b = right.permute(0,3,1,2).contiguous().view(B·H, N, N)
        out = bmm(a, b.transpose(1,2)).
5️⃣  Layer‑norm over the hidden dimension (FP16), apply the out‑gate, and a
    final linear projection back to `dim`.
6️⃣  Cast the result to FP32 for the harness.

Only the forward pass (no autograd) is implemented.
All heavy matrix‑multiplications are dispatched to cuBLAS/Tensor‑Core, while
the element‑wise mask‑gate fusion is performed with a custom Triton kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # hidden dimension (= size of last axis of x)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_elts

    # Collapse the mask dimension: each (b,i,j) is repeated `hidden_dim` times.
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1 → “no mask”

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM (heavy O(N³) step) – uses torch.bmm which maps to
# Tensor‑Core kernels on an H100.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Shapes:
        left , right : [B, N, N, H]  (FP16)
    Returns:
        out : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] – each hidden channel becomes a separate batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First Layer‑norm (FP32 -> FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   FP32
    x = x.to(torch.float16)  # FP16 for the heavy part

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single GEMM
    #    The weight matrix has shape (5·H, dim)
    # ------------------------------------------------------------------
    # Grab the five weight tensors (all have shape [hidden_dim, dim])
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Stack them vertically → shape (5·hidden_dim, dim)
    combined_w_fp16 = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)

    # --------------------------------------------------------------
    # Linear projection (FP16) → shape [B, N, N, 5·hidden_dim]
    # --------------------------------------------------------------
    proj = torch.nn.functional.linear(x, combined_w_fp16)  # fused linear
    # Slice the packed tensor into its logical components
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )
    # Delete the large packed tensor ASAP to free memory
    del proj

    # --------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) – still in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask – fuse (proj * gate * mask) in‑place via Triton
    # --------------------------------------------------------------
    if not nomask:
        # mask is broadcast across the hidden dimension => expand to FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()
        BLOCK = 1024                                 # ≤ max threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # At this point:
    #   left_proj  : [B, N, N, H]  (masked & gated)
    #   right_proj : [B, N, N, H]  (masked & gated)

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # [B, N, N, H]   FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second Layer‑norm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Gating by out_gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 8️⃣ Return FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)

            self.to_out = torch.nn.Linear(hidden_dim, dim, bias=False)

            left = self.left_proj(x)
            right = self.right_proj(x)

            left = left * mask
            right = right * mask

            out_gate = self.out_gate(x).sigmoid()

            right = right * right_gate

            out = self.to_out_norm(out)

            out = self.to_out(out)
