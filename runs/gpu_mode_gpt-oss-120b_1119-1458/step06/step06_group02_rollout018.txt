def custom_kernel(data):
    """TriMul outgoing forward pass.

    This implementation fuses the five linear projections into a single
    matmul, performs the per‑channel gating and (optional) masking with a
    custom Triton kernel, and computes the O(N³) contraction using a
    strided‑batched GEMM.
    The whole forward pass is executed in fp16 except for the first
    LayerNorm which stays in fp32 for numerical stability.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. First LayerNorm (float32) then cast to fp16
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # 2. Combine the five linear weight matrices into one
    # weights are float32, convert to half once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    combined_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0) # shape (5*hidden_dim, dim)
    
    # 3. Single linear projection: [B, N, N, dim] -> [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_w)  # fp16, shape [B, N, N, 5*hidden_dim]
    # split
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = combined.split([hidden_dim]*5, dim=-1)

    # 4. Compute out_gate after sigmoid (no mask needed yet)
    out_gate = torch.sigmoid(out_gate_pre)   # fp16

    # 5. Apply per‑channel gating and optional mask using Triton kernel
    # mask_fp16: shape (B, N, N) always
    if nomask:
        mask_fp16 = torch.ones_like(mask_tensor, dtype=torch.float16) if mask_tensor is not None else torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=torch.float16, device=input_tensor.device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    # ensure mask is contiguous
    mask_fp16 = mask_fp16.contiguous()
    
    total_elems = left_proj.numel()
    BLOCK = 1024
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    # left
    proj_gate_mask_kernel[grid](
        left_proj,
        left_gate_pre,
        mask_fp16,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right
    proj_gate_mask_kernel[grid](
        right_proj,
        right_gate_pre,
        mask_fp16,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # we can delete gate_pre tensors after use to free memory
    del left_gate_pre, right_gate_pre

    # 6. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = _batched_matmul_torch(left_proj, right_proj)  # fp16

    # 7. Second LayerNorm over hidden_dim
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # 8. Apply out gate (in place)
    out.mul_(out_gate)

    # 9. Final linear to original dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # fp16
    
    # 10. Cast back to float32
    return out.to(torch.float32)

"""
Optimised outgoing TriMul forward (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Layer‑Norm on the raw input (float32) – stabilises statistics.
2️⃣  Cast the normalised tensor to fp16 for the compute‑heavy part.
3️⃣  Fuse the five linear projections (`left_proj`, `right_proj`,
    `left_gate`, `right_gate`, `out_gate`) into a single matmul by
    concatenating their weight matrices.
4️⃣  Split the fused result and compute the per‑channel gating together
    with the optional mask in a custom Triton kernel
    (`proj_gate_mask_kernel`).  This kernel evaluates the sigmoid on‑the‑fly,
    multiplies by the mask (all‑ones when `nomask=True`) and writes the
    gated tensor back in‑place, eliminating intermediate tensors and
    extra memory passes.
5️⃣  Compute the O(N³) contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    as a batched GEMM (strided batched `torch.bmm`) after reshaping to
    [B·H, N, N] (H = hidden_dim).  This leverages cuBLAS Tensor‑Cores.
6️⃣  Second Layer‑Norm over the hidden dimension (fp16).
7️⃣  Apply the output gate (`out_gate`) and project back to the original
    dimension with a final linear layer.
8️⃣  Cast the result back to float32 (required by the harness).

Only the forward pass is implemented – gradients are omitted.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel to fuse projection, sigmoid‑gate and optional mask.
# --------------------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    proj_ptr,          # *fp16 : tensor to be updated (left or right projection)
    gate_pre_ptr,      # *fp16 : pre‑sigmoid gate tensor (same shape as proj)
    mask_ptr,          # *fp16 : mask tensor (shape B·N·N, broadcasted over H)
    hidden_dim,        # i32   : size H of the hidden dimension
    size,              # i32   : total number of elements in proj (B·N·N·H)
    BLOCK: tl.constexpr,   # compile‑time block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets for this block
    active = offs < size                               # guard for OOB

    # mask index = floor(offs / hidden_dim)  (since hidden dim is the innermost stride)
    mask_offs = offs // hidden_dim

    # Load data
    proj   = tl.load(proj_ptr     + offs, mask=active, other=0.0)
    gate   = tl.load(gate_pre_ptr + offs, mask=active, other=0.0)
    mask   = tl.load(mask_ptr     + mask_offs, mask=active, other=1.0)  # default 1.0

    # Sigmoid on‑the‑fly (all fp16)
    sig = 1.0 / (1.0 + tl.exp(-gate))

    # In‑place update: proj ← proj * sigmoid(gate) * mask
    out = proj * sig * mask
    tl.store(proj_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Heavy O(N³) contraction implemented as a batched GEMM.

    Both `left` and `right` have shape [B, N, N, H] (fp16). The result
    also has shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] – each hidden channel becomes its own batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  ->  (B·H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    batch_size, seq_len, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → fp16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x.to(torch.float16)   # now fp16 for the bulk computation

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into one GEMM
    # --------------------------------------------------------------
    # Cast all model weights to fp16 once
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Concatenate along the output‑feature dimension: (5·H, dim)
    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Single linear layer: [B, N, N, dim] → [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_w)  # fp16

    # Split back into the five parts
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = combined.split(
        [hidden_dim] * 5, dim=-1
    )  # each: [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Output gate (no mask involvement) → sigmoid
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_pre)  # fp16, shape [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 4️⃣ Fuse per‑channel gating (sigmoid) and optional mask
    # --------------------------------------------------------------
    # Prepare mask: either the real mask or an all‑ones tensor.
    if nomask:
        mask_fp16 = torch.ones(
            (batch_size, seq_len, seq_len), dtype=torch.float16, device=device
        )
    else:
        # mask may be bool / float; ensure fp16 and contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()

    # Triton kernel launch parameters
    total_elems = left_proj.numel()          # B·N·N·H
    BLOCK = 1024
    grid = ( (total_elems + BLOCK - 1) // BLOCK, )

    # Left branch
    proj_gate_mask_kernel[grid](
        left_proj,
        left_gate_pre,
        mask_fp16,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # Right branch
    proj_gate_mask_kernel[grid](
        right_proj,
        right_gate_pre,
        mask_fp16,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # Release gate pre‑sigmoid tensors early (they are no longer needed)
    del left_gate_pre, right_gate_pre

    # --------------------------------------------------------------
    # 5️⃣ Heavy N³ contraction via batched GEMM
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # fp16, shape [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # --------------------------------------------------------------
    out.mul_(out_gate)                                  # in‑place fp16 multiplication
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)    # fp16 → [B, N, N, dim]

    # --------------------------------------------------------------
    # 8️⃣ Return float32 result
    # --------------------------------------------------------------
    return out.to(torch.float32)
