import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr, total_elements, hidden_dim, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load x and gate
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # mask index: each mask entry repeats hidden_dim times
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m

    tl.store(x_ptr + offs, out, mask=mask)

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] for batched GEMM
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # compute A @ B^T using bmm
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Fast forward pass of the outgoing TriMul operator.
    Performs:
      1) LayerNorm (FP32) -> FP16.
      2) Fused projection (left/right) and gating (left/right/out) linear layers.
      3) Optional mask+gate fusion via Triton.
      4) Batched GEMM contraction left @ right^T for each hidden head.
      5) Second LayerNorm, out‑gate, and final projection.
    All heavy compute runs in FP16 with Tensor‑Core friendly kernels.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------- 1. First LayerNorm (FP32) ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to FP16 for the rest of the computation
    x = x.half()

    # --------- 2. Prepare flattened view ----------
    B, N, _, _ = x.shape
    total_pairs = B * N * N
    x_flat = x.view(total_pairs, dim)  # [B*N*N, dim]  FP16

    # --------- 3. Fuse projection Linear layers ----------
    # combine left and right proj weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    proj_weight = torch.cat([left_proj_weight, right_proj_weight], dim=0).half()  # [(2*hidden_dim), dim]

    proj = torch.nn.functional.linear(x_flat, proj_weight)  # [total_pairs, 2*hidden_dim]
    # split
    left_proj_flat = proj[:, :hidden_dim]
    right_proj_flat = proj[:, hidden_dim:]

    # free proj to save memory
    del proj

    # --------- 4. Fuse gating Linear layers ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    gate_weight = torch.cat([left_gate_weight, right_gate_weight, out_gate_weight], dim=0).half()  # [(3*hidden_dim), dim]

    gates = torch.nn.functional.linear(x_flat, gate_weight)  # [total_pairs, 3*hidden_dim]
    # split and apply sigmoid
    left_gate_flat = torch.sigmoid(gates[:, :hidden_dim])
    right_gate_flat = torch.sigmoid(gates[:, hidden_dim:2 * hidden_dim])
    out_gate_flat = torch.sigmoid(gates[:, 2 * hidden_dim:])

    # free gates and x_flat (no longer needed)
    del gates
    del x_flat

    # --------- 5. Mask + gate fusion (in‑place) ----------
    if nomask:
        # No mask: simply multiply projections with gates
        left_proj_flat = left_proj_flat * left_gate_flat
        right_proj_flat = right_proj_flat * right_gate_flat
    else:
        # Flatten mask to 1‑D and convert to FP16
        mask_flat = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        total_elements = left_proj_flat.numel()  # B*N*N*hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left stream
        mask_gate_fuse_kernel[grid](
            left_proj_flat,
            left_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right stream
        mask_gate_fuse_kernel[grid](
            right_proj_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # gates are no longer needed
        del left_gate_flat, right_gate_flat

    # --------- 6. Reshape projected tensors ----------
    left = left_proj_flat.view(B, N, N, hidden_dim)   # [B,N,N,hidden_dim] FP16
    right = right_proj_flat.view(B, N, N, hidden_dim)

    # optional cleanup
    del left_proj_flat, right_proj_flat

    # --------- 7. Heavy contraction via batched GEMM ----------
    out = _batched_gemm_fp16(left, right)  # [B,N,N,hidden_dim] FP16

    # --------- 8. Second LayerNorm (FP16) ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # --------- 9. Apply out‑gate ----------
    out_gate = out_gate_flat.view(B, N, N, hidden_dim)  # [B,N,N,hidden_dim]
    out = out * out_gate

    # ---------10. Final projection ----------
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())  # [B,N,N,dim] FP16

    # Return FP32 as expected by the harness
    return out.float()

   operations:
   * One projection that computes *left* and *right* simultaneously
     (`weight = [left_proj; right_proj]`).
   * One gating that computes the three sigmoid‑gates simultaneously
     (`weight = [left_gate; right_gate; out_gate]`).
3️⃣ If a mask is present, fuse the three element‑wise ops
   `proj * gate * mask` for *left* and *right* with a tiny Triton kernel.
   When there is no mask we simply multiply the projections with their gates.
4️⃣ Reshape the gated tensors to `[B,N,N,hidden]` and run the O(N³) contraction
   as a batched GEMM:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is implemented with `torch.bmm` on FP16 (Tensor‑Core friendly).
5️⃣ Second LayerNorm on the hidden dimension (still FP16), apply the *out‑gate*,
   and a final linear projection back to the original channel dimension.
6️⃣ Cast the result back to FP32 (the harness expects FP32).

All heavy compute runs in FP16, while the only Triton kernel is used for the
mask‑gate fusion, satisfying the requirement that “part of the operations” be
implemented in Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) – mask is broadcast over the
# hidden dimension.  Operates on a flat view of a contiguous tensor
# of shape [total_elements] where total_elements = B·N·N·hidden.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated, size = total_elements
    gate_ptr,       # *fp16  gate tensor, same shape as x
    mask_ptr,       # *fp16  mask tensor of shape (B·N·N,) – broadcasted
    total_elements,  # i32    total number of elements in x
    hidden_dim,      # i32    width of the hidden dimension
    BLOCK: tl.constexpr,  # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                # linear offsets
    mask = offs < total_elements                             # guard against tail

    # Load tensors (zero for OOB elements)
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Every `hidden_dim` consecutive elements share the same mask entry.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store the result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via
    a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] for batched GEMM
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → ignore the mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                   # original channel dim (c_z)
    hidden_dim = config["hidden_dim"]      # projection width (c)
    nomask = config.get("nomask", True)   # True → mask should be ignored

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32
    x = x.half()  # work in FP16 from now on

    B, N, _, _ = x.shape
    total_pairs = B * N * N                # B·N·N

    # --------------------------------------------------------------
    # 2️⃣ Flatten for efficient linear layers
    # --------------------------------------------------------------
    x_flat = x.view(total_pairs, dim)       # [B·N·N, dim]  FP16

    # --------------------------------------------------------------
    # 3️⃣ Combined projection (left + right)
    # --------------------------------------------------------------
    # Concatenate the two projection weight matrices
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).half()                                 # [(2·hidden_dim), dim]

    proj_out = torch.nn.functional.linear(x_flat, proj_weight)   # [pairs, 2·hidden]
    left_proj_flat = proj_out[:, :hidden_dim]                    # [pairs, hidden]
    right_proj_flat = proj_out[:, hidden_dim:]                   # [pairs, hidden]
    del proj_out  # free intermediate

    # --------------------------------------------------------------
    # 4️⃣ Combined gating (left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                                 # [(3·hidden), dim]

    gate_out = torch.nn.functional.linear(x_flat, gate_weight)   # [pairs, 3·hidden]
    left_gate_flat = torch.sigmoid(gate_out[:, :hidden_dim])
    right_gate_flat = torch.sigmoid(gate_out[:, hidden_dim:2 * hidden_dim])
    out_gate_flat = torch.sigmoid(gate_out[:, 2 * hidden_dim :])
    del gate_out, x_flat  # no longer needed

    # --------------------------------------------------------------
    # 5️⃣ Mask & gate fusion (in‑place)
    # --------------------------------------------------------------
    if nomask:
        # No mask → simple elementwise multiplication
        left_proj_flat = left_proj_flat * left_gate_flat
        right_proj_flat = right_proj_flat * right_gate_flat
    else:
        # Prepare mask as a flat FP16 vector (broadcast over hidden dim)
        mask_flat = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B·N·N,)
        total_elements = left_proj_flat.numel()                         # B·N·N·hidden
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Left stream
        mask_gate_fuse_kernel[grid](
            left_proj_flat,
            left_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Right stream
        mask_gate_fuse_kernel[grid](
            right_proj_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Gates are fused, can be released
        del left_gate_flat, right_gate_flat

    # --------------------------------------------------------------
    # 6️⃣ Reshape back to 4‑D tensors
    # --------------------------------------------------------------
    left = left_proj_flat.view(B, N, N, hidden_dim)   # [B,N,N,hidden]  FP16
    right = right_proj_flat.view(B, N, N, hidden_dim)
    del left_proj_flat, right_proj_flat

    # --------------------------------------------------------------
    # 7️⃣ Heavy contraction (batched GEMM) – cuBLAS Tensor‑Core path
    # --------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)           # [B,N,N,hidden] FP16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # --------------------------------------------------------------
    # 9️⃣ Out‑gate multiplication
    # --------------------------------------------------------------
    out_gate = out_gate_flat.view(B, N, N, hidden_dim)   # [B,N,N,hidden] FP16
    out = out * out_gate                                 # element‑wise gate
    del out_gate

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].half()
    )   # [B,N,N,dim]  FP16

    # --------------------------------------------------------------
    # Return FP32 as expected by the harness
    # --------------------------------------------------------------
    return out.float()
```
