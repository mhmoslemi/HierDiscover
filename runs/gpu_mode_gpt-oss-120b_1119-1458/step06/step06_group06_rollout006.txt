```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16
    gate_ptr,       # *fp16
    mask_ptr,       # *fp16
    total_elements,  # i32
    hidden_dim,      # i32
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # compute mask index
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left/right: [B, N, N, H] half
    B, N, _, H = left.shape
    # view as batch of (B*H) matrix multiply A @ B.T
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass.

    Performs:
    - LayerNorm + FP16 cast
    - Fused projection (left/right) and gating (left/right) linear layers
    - Optional mask+gate fusion via Triton kernel
    - Batched GEMM contraction left @ rightᵀ
    - Second LayerNorm and out‑gate gating
    - Final linear projection back to original channel dimension
    Returns FP32 tensor [B, N, N, dim].
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

"""
TriMul (outgoing) – optimized forward pass.

Key ideas
---------
*   The four‑dimensional tensor is layer‑normed in FP32 and cast once to FP16.
*   Linear projections and gating are **fused**:
        - Left and right projections share one GEMM (2×hidden_dim rows).
        - Left and right gates share another GEMM (2×hidden_dim rows).
        - The out‑gate is a third GEMM.
  This reduces the number of passes over the large input tensor.
*   When a mask is present we fuse ``proj * gate * mask`` with a tiny
    Triton kernel (`mask_gate_fuse_kernel`).  The kernel works on a
    flat view of the tensor and multiplies each element by its gate and
    by the broadcast mask entry (one mask value per `(b,i,k)`).
*   The heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
  is performed as a **batched GEMM** (A @ Bᵀ) using cuBLAS / Tensor‑Core
  kernels via ``torch.bmm``.  The tensor is reshaped to a batch dimension
  of ``B*hidden_dim``.
*   A second LayerNorm, out‑gate multiplication and the final linear
  projection are carried out in FP16.  The final result is cast back to
  FP32 as required by the harness.

Only the forward pass is needed; gradients are not tracked.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) – mask is broadcast over the
# hidden dimension. Operates on a flat view of a contiguous tensor of
# shape [B, N, N, hidden_dim] (total_elements = B·N·N·hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, size = total_elements
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,) – broadcasted
    total_elements,  # i32 : total number of elements in x
    hidden_dim,      # i32 : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask = offs < total_elements                # guard for the tail of the tensor

    # Load tensors
    x   = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Each hidden_dim consecutive elements share the same mask entry.
    mask_idx = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store the result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] via
    a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] and perform A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a.contiguous(), b.contiguous().transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
