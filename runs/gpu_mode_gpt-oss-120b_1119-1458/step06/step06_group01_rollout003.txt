Pseudo:

```
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch = pid // (num_pid_i * num_pid_j)
    # Actually better to map 3D grid: program_id(0) for batch; program_id(1) for i tile; program_id(2) for j tile.
    # Use multi-dim grid.

    pid_b = tl.program_id(0) # batch dimension
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H) # we will broadcast across H in tile.

    # mask test:
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # Initialize accumulator per tile: shape (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)  # accumulate in fp32 for precision.

    # Loop over k dimension
    for k_tile in range(tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + \
          pid_b * stride_left_b + \
          offs_i[:, None, None] * stride_left_i + \
          offs_k[None, :, None] * stride_left_k + \
          offs_h[None, None, :] * stride_left_h
        left_tile = tl.load(left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ) # fp16
        # Convert to fp32:
        left_tile_f = left_tile.to(tl.float32)

        # Load right tile: (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + \
          pid_b * stride_right_b + \
          offs_j[:, None, None] * stride_right_j + \
          offs_k[None, :, None] * stride_right_k + \
          offs_h[None, None, :] * stride_right_h
        right_tile = tl.load(right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        right_tile_f = right_tile.to(tl.float32)

        # Compute outer product: left_tile[i,k,h] * right_tile[j,k,h] sum over k:
        # We can use broadcasting: left_tile_f shape (I,K,H) and right_tile_f shape (J,K,H).
        # We need to multiply left_tile_f[..., h] and right_tile_f[..., h] and sum over k.
        # In Triton we can do:
        # broadcast left onto J dimension or right onto I as needed.
        # For each hidden index h, we want sum_{k} left[i,k,h] * right[j,k,h].
        # This is essentially a matmul per h: left_h(i,k) @ right_h(j,k).T?
        # Actually right has shape (J,K,H); we want right[j,k,h] but we need right[j,k,h] as we sum k across left[i,k,h] * right[j,k,h].
        # So at each iteration we have left[i,k,h] and right[j,k,h]. Multiplying and summing yields contribution to out[i,j,h].
        # So we can iterate over k tiles and accumulate:
        # For each value of k tile index, we can compute elementwise multiplication and sum across k dimension (since we loop over k tiles).
        # However we can accelerate using dot product across k dimension per tile: we can compute for each (i,j,h) sum_{k_in_tile} left[i,k,h] * right[j,k,h].
        # In Triton we can do something like:
        #   # expand left tile: shape (I, K, H) -> (I, K, 1, H)
        #   # expand right tile: shape (J, K, H) -> (1, K, J, H)
        #   # we need to multiply left[i,k,h] * right[j,k,h] => shape (I,J,K,H).
        #   # then sum over k.

Outgoing TriMul (AlphaFold‑3) – Fast mixed‑precision implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣  Fuse the two linear projections (proj + gate) for the left and right

5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear
    projection back to the model dimension.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask is broadcast over hidden dimension H)
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask  (mask broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Fuse projection, gate and mask:
       out = proj * gate * mask   (mask broadcast over hidden dim).

       All tensors must be FP16 CUDA tensors.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides are expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    # Tile sizes – selected for good occupancy on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batched GEMM C = A @ Bᵀ with cuBLAS Tensor‑Core kernels.
    The hidden dimension H is folded into the batch dimension (B·H, N, N).
    """
    B, N, _, H = left.shape
    # Rearrange to [B, H, N, N] without an extra copy (the view is non‑contiguous,
    # but torch.bmm can handle it efficiently when we .contiguous() the flattening).
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply with right transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, on the same device as the input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True => skip mask

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # still FP32
    x = x.to(torch.float16)   # remaining ops in FP16

    # --------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (they already live on the device)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)

    # Fuse proj + gate for each side (2 * hidden_dim → join into one linear)
    left_combined_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # shape [2*H, dim]
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0) # shape [2*H, dim]

    out_gate_w     = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w  = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b  = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w       = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (proj + gate) in a single matmul per side
    # --------------------------------------------------------------
    # Output shape: [B, N, N, 2*hidden_dim]
    left_combined = torch.nn.functional.linear(x, left_combined_w)   # proj + gate logits
    right_combined = torch.nn.functional.linear(x, right_combined_w)

    # Split into proj and gate parts
    left_proj = left_combined[..., :hidden_dim]                  # [B,N,N,hidden_dim]
    left_gate_logits = left_combined[..., hidden_dim:]           # [B,N,N,hidden_dim]
    right_proj = right_combined[..., :hidden_dim]
    right_gate_logits = right_combined[..., hidden_dim:]

    # Apply sigmoid to gate logits (still FP16)
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if present) – fused with gate via Triton,
    #    otherwise just multiply proj * gate.
    # --------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 (broadcast later over hidden dim)
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)   # [B,N,N,hidden_dim]
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # --------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden_dim] fp16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate (sigmoid) → final linear projection back to `dim`
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,hidden_dim]
    out = out * out_gate                         # gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] (FP16)

    # --------------------------------------------------------------
    # 8️⃣ Cast result back to FP32 (expected by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
