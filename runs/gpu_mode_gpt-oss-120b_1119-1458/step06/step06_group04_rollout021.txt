def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,

    num_warps: tl.constexpr):
    pid_n = tl.program_id(0)  # block in N dimension (columns)
    pid_m = tl.program_id(1)  # block in M dimension (rows)
    pid_b = tl.program_id(2)  # batch id

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask
    offs_m_valid = offs_m < M
    offs_n_valid = offs_n < N

    # base pointer offset for batch dimension
    a_batch_ptr = a_ptr + pid_b * stride_ab
    b_batch_ptr = b_ptr + pid_b * stride_bb
    c_batch_ptr = c_ptr + pid_b * stride_cb

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K

@triton.jit
def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr, hidden_dim: tl.constexpr, total_elems: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # each hidden_dim group share mask index
    # compute index for mask: floor division integer; we need to divide by hidden_dim
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default mask 1

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)
```

```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr):
    pid_n = tl.program_id(0)  # column block index (j)
    pid_m = tl.program_id(1)  # row block index (i)
    pid_b = tl.program_id(2)  # batch index

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + pid_b * stride_ab
    b_batch_ptr = b_ptr + pid_b * stride_bb
    c_batch_ptr = c_ptr + pid_b * stride_cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

"""
custom_kernel: Optimized forward pass of the outgoing TriMul module (no autograd).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣ Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single matmul (weight shape 5·hidden_dim × dim).
3️⃣ Slice the fused result and apply sigmoid to the three gating tensors.
4️⃣ Apply the optional mask and fuse the element‑wise multiplication
   left = left_proj·left_gate·mask , right = right_proj·right_gate·mask.
   (If no mask – just left_proj·left_gate and right_proj·right_gate.)
5️⃣ Compute the heavy O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h]·right[b,j,k,h]
   via a custom Triton batched GEMM kernel (C = A @ Bᵀ) where A and B are
   reshaped to (B·hidden_dim, N, N).  The kernel uses fp16 inputs,
   fp32 accumulation and Tensor‑Core dot products.
6️⃣ Second LayerNorm (fp16) on the result.
7️⃣ Multiply by out_gate (broadcasted) and apply the final linear projection
   back to the original channel dimension.
8️⃣ Cast to float32 (the harness expects float32).

All heavy arithmetic runs in fp16 to exploit TensorCores on an H100.
Both a Triton kernel for mask‑gate fusion (optional) and a Triton kernel for
the batched GEMM are used, satisfying the “must use Triton” requirement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse element‑wise mask, projection and gate multiplication.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 → tensor to be updated (flattened)
    gate_ptr,       # *fp16 → gate tensor (same flattening as x)
    mask_ptr,       # *fp16 → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → size of the hidden dimension (H)
    total_elems,    # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # load tensors
    x   = tl.load(x_ptr   + offs, mask=valid, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=valid, other=0.0)

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: batched row‑wise GEMM C = A @ Bᵀ.
# A and B are (batch, M, K) and (batch, N, K) respectively.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: row, col, batch stride
    stride_bk, stride_bn, stride_bb,   # B: row(K), col(N), batch stride
    stride_cm, stride_cn, stride_cb,   # C: row, col, batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    # program IDs
    pid_n = tl.program_id(0)   # column block
    pid_m = tl.program_id(1)   # row block
    pid_b = tl.program_id(2)   # batch index (B * hidden_dim)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers offset to the current batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ tile (K x N) – note that we load B as (K,N) i.e. rows are K
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write C (M x N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,
        torch.Tensor,
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool/None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack and parse
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent heavy ops
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (left/right proj & gates + out_gate)
    # --------------------------------------------------------------
    # Load all five weight matrices and concatenate them (no bias)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16) # (hidden, dim)
    right_gate_w = weights["right_gate.weight"].to(torch.float16) # (hidden, dim)
    out_gate_w  = weights["out_gate.weight"].to(torch.float16)   # (hidden, dim)

    # Combined weight: (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Single fused matmul
    combined = torch.nn.functional.linear(x, combined_weight)   # [B, N, N, 5*H] (fp16)

    # --------------------------------------------------------------
    # 4️⃣ Slice the fused tensor into its components
    # --------------------------------------------------------------
    H = hidden_dim
    left_proj   = combined[..., 0:H]          # (B, N, N, H)
    left_gate   = torch.sigmoid(combined[..., H:2*H])   # (B, N, N, H)
    right_proj  = combined[..., 2*H:3*H]      # (B, N, N, H)
    right_gate  = torch.sigmoid(combined[..., 3*H:4*H]) # (B, N, N, H)
    out_gate    = torch.sigmoid(combined[..., 4*H:5*H]) # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask + gate fusion (Triton) or plain element‑wise
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B,N,N] → fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                     # shape [B*N*N]
        total_elems = left_proj.numel()                    # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # No mask – just element‑wise gating
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core O(N³) contraction via a custom batched GEMM kernel
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Permute to (B, H, N, N) and flatten batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch = B * H

    left_flat = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    right_flat = right_perm.view(batch, N, N) # (batch, K=N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)    # fp16

    # Tile sizes – chosen to hit TensorCores efficiently
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_N - 1) // BLOCK_N,   # program_id(0) – column blocks
        (N + BLOCK_M - 1) // BLOCK_M,   # program_id(1) – row blocks
        batch,                          # program_id(2) – per (B·H) batch
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(1),   # stride_am (row stride = N)
        left_flat.stride(2),   # stride_ak (col stride = 1)
        left_flat.stride(0),   # stride_ab (batch stride = N*N)
        right_flat.stride(2),  # stride_bk (rows = K ; stride = 1)
        right_flat.stride(1),  # stride_bn (cols = N ; stride = N)
        right_flat.stride(0),  # stride_bb (batch stride)
        out_flat.stride(1),    # stride_cm
        out_flat.stride(2),    # stride_cn
        out_flat.stride(0),    # stride_cb
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8️⃣ Gating with out_gate and final projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness.
    # --------------------------------------------------------------
    return out.to(torch.float32)
