"""
TriMul (outgoing) – fast forward pass.

The implementation follows the reference TriMul module but drastically reduces
memory traffic and kernel launch overhead:

1️⃣  First LayerNorm (FP32 → FP16).
2️⃣  All five linear layers (left/right projections, three gates) are fused
    into a single GEMM: `x_norm ∈ ℝ^{B·N²×C}` × `W ∈ ℝ^{5·H×C}`.
    The result is reshaped to `[B,N,N,5·H]` and split into the five branches.
3️⃣  `sigmoid` for the three gate tensors is performed in‑place.
4️⃣  When a mask is present we fuse **projection × gate × mask** with a tiny
    Triton kernel that also computes the sigmoid, saving two extra kernel
    launches.
5️⃣  The O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    is expressed as a strided batched GEMM (`left @ rightᵀ`) using cuBLAS
    tensor‑core kernels.
6️⃣  Second LayerNorm, out‑gate multiplication and the final linear projection
    back to the model dimension are performed with native PyTorch ops.
7️⃣  The output is cast back to FP32 (the harness expects float32).

Only the mask‑gate‑fusion kernel is written in Triton; the rest relies on
high‑performance cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: proj * sigmoid(gate_pre) * mask   (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]   (projected tensor)
    gate_pre_ptr,      # *fp16  [B, N, N, H]   (gate pre‑sigmoid)
    mask_ptr,          # *fp16  [B, N, N]      (binary mask)
    out_ptr,           # *fp16  [B, N, N, H]   (result)
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute  out = proj * sigmoid(gate_pre) * mask   (mask broadcast over H)
    All tensors are stored as row‑major (contiguous) tensors.
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # --- tile indices -------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --- load mask tile (BLOCK_I × BLOCK_J) ---------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )                                   # fp16, shape [BLOCK_I, BLOCK_J]

    # --- loop over hidden dimension ------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16, shape [I,J,H]

        # gate_pre tile
        gate_ptrs = (
            gate_pre_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_pre_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16

        # sigmoid(gate_pre) = 1 / (1 + exp(-x))
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_pre_tile))

        # fused multiply: proj * sigmoid(gate_pre) * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: fused mask‑gate‑sigmoid (proj, gate_pre, mask) → out
# ----------------------------------------------------------------------
def _apply_proj_gate_mask(
    proj: torch.Tensor, gate_pre: torch.Tensor, mask: torch.Tensor
) -> torch.Tensor:
    """
    proj, gate_pre : [B, N, N, H] (fp16)
    mask           : [B, N, N]    (fp16)

    Returns out = proj * sigmoid(gate_pre) * mask  (mask broadcast over H)
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides in element units (not bytes)
    s_proj = proj.stride()
    s_gate = gate_pre.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # tile sizes – chosen to keep registers & shared memory modest on H100
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8   # works for hidden_dim up to at least 384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _proj_gate_mask_kernel[grid](
        proj,
        gate_pre,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using cuBLAS.
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns out  : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # Merge batch and channel dimensions for a single strided batched gemm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out = left_flat @ right_flatᵀ   (both [B*H, N, N])
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,
        torch.Tensor,
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32 → FP16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)   # FP16 for the rest of the pipeline

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers into a single GEMM
    # -----------------------------------------------------------------
    # Collect weights (already bias‑free)
    w_left    = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    w_right   = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    w_left_g  = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    w_right_g = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    w_out_g   = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    # Concatenate along the output‑feature dimension (5×hidden_dim)
    # Final shape: [5*hidden_dim, dim]
    all_weight = torch.cat(
        [w_left, w_right, w_left_g, w_right_g, w_out_g],
        dim=0,
    )   # (5*H, dim)

    B, N, _, _ = x.shape

    # Flatten leading dimensions for the GEMM: [B·N·N, dim]
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    # Single fused linear → shape (B*N*N, 5*hidden_dim)
    fused = torch.nn.functional.linear(x_flat, all_weight)  # FP16

    # Reshape back to 4‑D tensor
    fused = fused.view(B, N, N, 5 * hidden_dim)      # (B,N,N,5*H)

    # -----------------------------------------------------------------
    # 4️⃣ Split fused tensor into the five branches
    # -----------------------------------------------------------------
    split = torch.split(fused, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = split

    # Out‑gate (no mask) – apply sigmoid in‑place
    out_gate = torch.sigmoid(out_gate_pre)

    # -----------------------------------------------------------------
    # 5️⃣ Apply mask + gate (gate includes sigmoid)
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 once (broadcasted over hidden dim later)
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left  = _apply_proj_gate_mask(left_proj, left_gate_pre, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate_pre, mask_fp16)
    else:
        # No mask – elementwise multiplication with sigmoid(gate)
        left_gate  = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)

        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # (B,N,N,hidden_dim), FP16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (over hidden_dim)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication and final projection back to model dim
    # -----------------------------------------------------------------
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
