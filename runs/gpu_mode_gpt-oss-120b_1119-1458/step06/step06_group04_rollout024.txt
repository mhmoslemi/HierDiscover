```
@triton.jit
def left_right_gate_fuse_kernel(
    left_ptr,
    left_gate_ptr,
    right_ptr,
    right_gate_ptr,
    mask_ptr,
    hidden_dim,
    total_elems,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # mask broadcast across hidden_dim
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default mask=1

    left_new = left * left_gate * mask_val
    right_new = right * right_gate * mask_val

    tl.store(left_ptr + offs, left_new, mask=valid)
    tl.store(right_ptr + offs, right_new, mask=valid)
```

def layernorm_gate_kernel(
    out_ptr,            # *fp16
    out_gate_ptr,       # *fp16
    ln_weight_ptr,      # *fp16
    ln_bias_ptr,        # *fp16
    hidden_dim,
    epsilon: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    row = tl.program_id(0)  # row index

    # Compute base pointer for this row
    row_ptr = out_ptr + row * hidden_dim
    row_gate_ptr = out_gate_ptr + row * hidden_dim

    # First pass: compute mean and variance
    sum_val = tl.float32(0)
    sum_sq_val = tl.float32(0)
    # Compute number of blocks needed for hidden_dim
    # We will use a loop over dim with stride BLOCK_H
    for offset in range(0, hidden_dim, BLOCK_H):
        # offsets for this block
        offs = offset + tl.arange(0, BLOCK_H)
        valid = offs < hidden_dim
        # Load the element (float16 -> convert to float32)
        x = tl.load(row_ptr + offs, mask=valid, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)  # sum across all threads? Actually each thread loads one element; tl.sum reduces across block. But we have multiple threads (BLOCK_H). Using tl.sum inside the loop will sum across all threads in the block each iteration and produce same sum across all threads. We need to accumulate across loops.
        sum_sq_val += tl.sum(x * x)

    mean = sum_val / hidden_dim
    var = sum_sq_val / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # Second pass: compute normalized, apply LN weight+bias, then out_gate
    # For each block again
    for offset in range(0, hidden_dim, BLOCK_H):
        offs = offset + tl.arange(0, BLOCK_H)
        valid = offs < hidden_dim
        x = tl.load(row_ptr + offs, mask=valid, other=0.0).to(tl.float32)
        gate = tl.load(row_gate_ptr + offs, mask=valid, other=0.0).to(tl.float32)
        # LN weight & bias per hidden feature
        w = tl.load(ln_weight_ptr + offs, mask=valid, other=0.0).to(tl.float32)
        b = tl.load(ln_bias_ptr + offs, mask=valid, other=0.0).to(tl.float32)

        # Normalize
        x_norm = (x - mean) * inv_std
        # Apply LN weight & bias
        x_norm = x_norm * w + b
        # Apply out gate
        out_val = x_norm * gate
        # Cast back to fp16
        out_fp16 = out_val.to(tl.float16)
        tl.store(row_ptr + offs, out_fp16, mask=valid)

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1️⃣ Layer‑norm the input tensor (float32) → `x_norm`.
2️⃣ Cast to fp16 for all heavy arithmetic.
3️⃣ Linear projections:
   * `left  = x_norm @ left_proj_weight`
   * `right = x_norm @ right_proj_weight`
4️⃣ Compute gating signals (sigmoid of a linear projection):
   * `left_gate  = sigmoid(x_norm @ left_gate_weight)`
   * `right_gate = sigmoid(x_norm @ right_gate_weight)`
   * `out_gate   = sigmoid(x_norm @ out_gate_weight)`
5️⃣ Fuse left/right gating together with optional mask in a single Triton kernel.
6️⃣ Core O(N³) contraction:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   – performed as a batched GEMM (`C = A @ Bᵀ`) using cuBLAS (torch.bmm) on fp16.
7️⃣ Fuse the second LayerNorm and the out‑gate multiplication with a Triton kernel.
8️⃣ Final linear projection back to the original channel dimension (`dim`).
9️⃣ Cast the result back to float32 for the harness.

All matrix‑multiplications and the bulk of arithmetic run in fp16 to exploit
Tensor‑Cores on the H100.  Two Triton kernels are used:
* `gate_fuse_kernel` – fuses `left = left * left_gate * mask` and
  `right = right * right_gate * mask`.
* `layernorm_gate_kernel` – computes per‑pair (`b,i,j`) LayerNorm over the
  hidden dimension, applies the learned `to_out_norm` weights/biases, and
  multiplies by `out_gate` in‑place.

The implementation is fully self‑contained and respects the required
function signature `custom_kernel(data)`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel 1: fuse left/right linear gating with optional mask.
# ----------------------------------------------------------------------
@triton.jit
def gate_fuse_kernel(
    left_ptr,          # *fp16  → left tensor (flattened)
    left_gate_ptr,     # *fp16  → left gate (flattened)
    right_ptr,         # *fp16  → right tensor (flattened)
    right_gate_ptr,    # *fp16  → right gate (flattened)
    mask_ptr,          # *fp16  → mask (flattened, shape [B*N*N])
    hidden_dim,        # i32   → hidden dimension = H
    size,              # i64   → total number of elements (=B*N*N*H)
    BLOCK: tl.constexpr,   # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)        # flat offsets
    mask = offs < size                               # guard for last tile

    # Load operands (fp16)
    left        = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate   = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right       = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask over the hidden dimension.
    # Each group of `hidden_dim` consecutive elements shares a mask entry.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1

    # Fuse the three multiplications.
    left_new  = left * left_gate * m
    right_new = right * right_gate * m

    tl.store(left_ptr + offs,  left_new,  mask=mask)
    tl.store(right_ptr + offs, right_new, mask=mask)

# ----------------------------------------------------------------------
# Triton kernel 2: fused second LayerNorm + out‑gate multiplication.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    out_ptr,          # *fp16 → tensor to be normalized & gated (flattened [B*N*N, H])
    out_gate_ptr,     # *fp16 → out‑gate tensor (flattened, same shape)
    ln_weight_ptr,    # *fp16 → LayerNorm weight (H,)
    ln_bias_ptr,      # *fp16 → LayerNorm bias   (H,)
    hidden_dim: tl.constexpr,          # int – hidden dimension H
    epsilon: tl.constexpr,             # float – LayerNorm epsilon
    BLOCK_H: tl.constexpr,             # block size used for the reduction (≤128)
):
    row = tl.program_id(0)                      # one program per (b,i,j) pair
    row_base = row * hidden_dim                 # start offset of this row
    row_end  = row_base + hidden_dim

    # ---------- 1️⃣ Compute mean & variance (float32) ----------
    sum_val  = tl.float32(0.0)
    sum_sq   = tl.float32(0.0)

    # Loop over the hidden dimension in tiles of size BLOCK_H
    for start in range(0, hidden_dim, BLOCK_H):
        offs = row_base + start + tl.arange(0, BLOCK_H)
        valid = offs < row_end
        x = tl.load(out_ptr + offs, mask=valid, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)                # sum across the BLOCK_H lanes
        sum_sq  += tl.sum(x * x)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # ---------- 2️⃣ Apply LN weight/bias, out‑gate, store back ----------
    for start in range(0, hidden_dim, BLOCK_H):
        hid_off = start + tl.arange(0, BLOCK_H)          # hidden‑dim index inside the row
        offs    = row_base + hid_off
        valid   = offs < row_end

        # Load values
        x      = tl.load(out_ptr + offs, mask=valid, other=0.0).to(tl.float32)
        gate   = tl.load(out_gate_ptr + offs, mask=valid, other=0.0).to(tl.float32)
        w      = tl.load(ln_weight_ptr + hid_off, mask=valid, other=0.0).to(tl.float32)
        b      = tl.load(ln_bias_ptr   + hid_off, mask=valid, other=0.0).to(tl.float32)

        # LayerNorm: (x - mean) / sqrt(var+ε) * w + b
        x_norm = (x - mean) * inv_std
        x_norm = x_norm * w + b

        # Multiply by out‑gate
        out_val = x_norm * gate

        # Write back as fp16
        tl.store(out_ptr + offs, out_val.to(tl.float16), mask=valid)

# ----------------------------------------------------------------------
# Helper for the O(N³) contraction (batched GEMM in fp16).
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] for all batches.
    The operation is expressed as a batched matrix multiplication:
        A = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
        B = right.permute(0,3,2,1).contiguous().view(B*H, N, N)   # already transposed
        C = torch.bmm(A, B) → shape [B*H, N, N]
    Returns:
        out tensor of shape [B, N, N, H] (fp16, contiguous).
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose the inner N×N matrix for the right operand
    b = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b)                         # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    return out

        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=device)

    # All tensors are contiguous in memory; the kernel works on the flattened view.
    gate_fuse_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        mask_flat,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # After the kernel, `left`, `right` already contain the gated (and masked) values.

    # ------------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction via a batched GEMM (fp16, TensorCores)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Fuse the second LayerNorm and the out‑gate multiplication
    # ------------------------------------------------------------------
    B, N, _, H = out.shape
    total_rows = B * N * N                               # number of (b,i,j) entries
    # Choose a reasonable tile size for the reduction (power of two, ≤128)
    BLOCK_H = 64 if hidden_dim > 64 else hidden_dim

    norm_grid = (total_rows,)
    layernorm_gate_kernel[norm_grid](
        out,
        out_gate,
        to_out_norm_w,
        to_out_norm_b,
        hidden_dim,
        epsilon=1e-5,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim` (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness.
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
