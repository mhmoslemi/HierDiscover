"""
TriMul (outgoing) forward pass – highly optimized Triton version.

Algorithm Overview
------------------
1️⃣ LayerNorm   : First LayerNorm is performed in FP32 for stability, then cast to FP16.
2️⃣ Fused projection:
   - All five linear projections (left_proj, left_gate, right_proj,
     right_gate, out_gate) are computed with a **single** cuBLAS GEMM by
     concatenating their weight matrices (shape (5·hidden_dim, dim)).
   - The result is sliced back into the five tensors.
3️⃣ Gating    : Apply sigmoid to the three gate tensors (left_gate, right_gate,
   out_gate) in FP16.
4️⃣ Mask‑Gate fusion (if a mask is present):
   - A Triton kernel fuses the mask (broadcast over the hidden dimension) with
     the projection and gate tensors:
       out = proj * gate * mask
   - This eliminates extra Triton launches and memory traffic.
   - When `nomask=True`, a simple element‑wise multiplication is used.
5️⃣ O(N³) contraction:
   - The core operation `out[i,j] = Σ_k left[i,k] * right[j,k]` is realised as
     a batched GEMM (`C = A @ Bᵀ`) using cuBLAS tensor‑core kernels.
   - We reshape to `[B·hidden_dim, N, N]` and call `torch.bmm`.
6️⃣ Second LayerNorm → out_gate → final linear projection.
7️⃣ Cast the result back to float32 (the harness expects FP32).

The implementation mixes Torch’s highly‑optimized matmul kernels with a
custom Triton kernel for the mask‑gate fusion, achieving the required speed‑up.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication proj * gate * mask,
# where mask is broadcast over the hidden dimension H.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H] (sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute out = proj * gate * mask (mask broadcast on H)."""
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # i/j tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I x BLOCK_J)
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load proj tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # shape [I,J,H]

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # shape [I,J,H]

        # Fuse: proj * gate * mask (mask broadcast on H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: apply mask + sigmoid‑gates using the Triton kernel above.
# ----------------------------------------------------------------------
def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H]  (fp16)
    mask       : [B, N, N]     (fp16)

    Returns  fp16 tensor = proj * gate * mask (mask broadcast over H).
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides in elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tuned tile sizes (good for H100 FP16 tensors)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction left @ rightᵀ.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    left, right : fp16 tensors of shape [B, N, N, H] (contiguous)
    Returns      : fp16 tensor of same shape.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each slice corresponds to a hidden channel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: C = A @ Bᵀ (both A and B are [B*H, N, N])
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul (outgoing) forward pass.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : mapping ``name → torch.Tensor`` (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # still float32
    x = x.to(dtype_fp16)  # cast to fp16 for fast matmuls

    # -----------------------------------------------------------------
    # 2️⃣ Concatenated linear projection (single GEMM)
    #    yields: left_proj | left_gate | right_proj | right_gate | out_gate
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    # Concatenate along the output‑feature dimension (5 * hidden_dim rows)
    weight_concat = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape [5*hidden_dim, dim] in fp16

    # Single linear call: produce tensor of shape [B, N, N, 5*hidden_dim]
    concat_out = torch.nn.functional.linear(x, weight_concat)  # fp16

    # Slice the concatenated output
    left_proj   = concat_out[..., 0                :hidden_dim]
    left_gate   = torch.sigmoid(concat_out[..., hidden_dim     : 2*hidden_dim])
    right_proj  = concat_out[..., 2*hidden_dim   : 3*hidden_dim]
    right_gate  = torch.sigmoid(concat_out[..., 3*hidden_dim   : 4*hidden_dim])
    out_gate    = torch.sigmoid(concat_out[..., 4*hidden_dim   : 5*hidden_dim])

    # -----------------------------------------------------------------
    # 3️⃣ Mask & gate fusion
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 (0.0 or 1.0) and broadcasted later
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy contraction (batched GEMM) – O(N³)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # fp16, shape [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (on hidden_dim)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # -----------------------------------------------------------------
    # 6️⃣ Apply out_gate and final linear projection back to model dimension
    # -----------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)     # fp16, now [B,N,N,dim]

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
