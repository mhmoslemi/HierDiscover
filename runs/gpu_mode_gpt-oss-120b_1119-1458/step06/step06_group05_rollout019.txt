"""
TriMul (outgoing) – fully fused forward pass.
~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
Algorithm Overview
------------------
1️⃣  Layer‑Norm (float32) → cast to fp16.
2️⃣  One Triton kernel processes the *whole* 4‑D tensor:
     * Computes the three linear transforms required for the left side,
       the corresponding three for the right side,
       and the single transform for the output‑gate.
     * Each linear transform is a GEMM between the flattened pairwise
       features (M = B·N·N rows, K = dim columns) and a weight matrix
       (K × H).  The kernel computes five GEMMs simultaneously, re‑using
       the input tile for all weight matrices.
     * The gate tensors are passed through a sigmoid, the left/right
       tensors are multiplied by their gates and by the optional mask
       (broadcast over the hidden dimension).  The output‑gate is just
       the sigmoid of its linear projection.
3️⃣  The expensive contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is realised as a batched GEMM (left @ rightᵀ) using cuBLAS/Tensor‑Core
    via a single ``torch.bmm`` call on a flattened view (B·H batches).
4️⃣  Second Layer‑Norm (fp16) → element‑wise multiplication with the
    output‑gate → final linear projection back to the model dimension.
5️⃣  Cast the result back to float32 (the harness expects FP32).

All heavy memory traffic (the three linear projections plus the gating
operations) is fused into a single Triton kernel, cutting the number of
times the 4‑D input tensor is read from memory by more than a factor
of three.  The O(N³) contraction still dominates the runtime, but using
batched tensor‑core GEMM yields the best performance on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused linear projections + gating + optional mask.
# Computes left, right and out_gate tensors in one pass.
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_all_kernel(
    # Input tensor (flattened) and mask
    x_ptr,                 # *fp16  [M, K]
    mask_ptr,              # *fp16  [M]   (used only if USE_MASK)
    # Weight matrices (already transposed: K × H)
    lp_ptr,                # *fp16  [K, H] left_proj
    lg_ptr,                # *fp16  [K, H] left_gate
    rp_ptr,                # *fp16  [K, H] right_proj
    rg_ptr,                # *fp16  [K, H] right_gate
    og_ptr,                # *fp16  [K, H] out_gate
    # Output buffers
    left_out_ptr,          # *fp16  [M, H]
    right_out_ptr,         # *fp16  [M, H]
    out_gate_out_ptr,      # *fp16  [M, H]
    # Sizes
    M, K, H,
    # Strides (in elements)
    stride_x_m, stride_x_k,
    stride_mask_m,
    stride_lp_k, stride_lp_h,
    stride_lg_k, stride_lg_h,
    stride_rp_k, stride_rp_h,
    stride_rg_k, stride_rg_h,
    stride_og_k, stride_og_h,
    stride_left_m, stride_left_h,
    stride_right_m, stride_right_h,
    stride_outgate_m, stride_outgate_h,
    # Tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,   # hidden dimension tile
    BLOCK_K: tl.constexpr,   # reduction (K) tile
    USE_MASK: tl.constexpr,
):
    """
    Computes:
        left  = (x @ lp) * sigmoid(x @ lg) * mask
        right = (x @ rp) * sigmoid(x @ rg) * mask
        out_gate = sigmoid(x @ og)
    All tensors are flattened (M = B·N·N rows).
    """
    pid_m = tl.program_id(0)          # row tile index
    pid_n = tl.program_id(1)          # hidden (column) tile index

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_h = offs_h < H

    # Load optional mask (scalar per row, broadcast over hidden)
    if USE_MASK:
        mask_tile = tl.load(mask_ptr + offs_m * stride_mask_m,
                            mask=mask_m,
                            other=0.0)               # fp16, shape [BLOCK_M]
        mask_tile = mask_tile[:, None]                 # [BLOCK_M,1]
    else:
        mask_tile = tl.full((BLOCK_M, 1), 1.0, dtype=tl.float16)

    # Accumulators in FP32 for numerical stability
    acc_lp = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of the input matrix X
        x_tile = tl.load(
            x_ptr + (offs_m[:, None] * stride_x_m + offs_k[None, :] * stride_x_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        x_tile = x_tile.to(tl.float32)   # fp16 → fp32

        # Load weight tiles
        lp_tile = tl.load(
            lp_ptr + (offs_k[:, None] * stride_lp_k + offs_h[None, :] * stride_lp_h),
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        lg_tile = tl.load(
            lg_ptr + (offs_k[:, None] * stride_lg_k + offs_h[None, :] * stride_lg_h),
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        rp_tile = tl.load(
            rp_ptr + (offs_k[:, None] * stride_rp_k + offs_h[None, :] * stride_rp_h),
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        rg_tile = tl.load(
            rg_ptr + (offs_k[:, None] * stride_rg_k + offs_h[None, :] * stride_rg_h),
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        og_tile = tl.load(
            og_ptr + (offs_k[:, None] * stride_og_k + offs_h[None, :] * stride_og_h),
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # GEMM updates
        acc_lp += tl.dot(x_tile, lp_tile)   # (M,H)
        acc_lg += tl.dot(x_tile, lg_tile)
        acc_rp += tl.dot(x_tile, rp_tile)
        acc_rg += tl.dot(x_tile, rg_tile)
        acc_og += tl.dot(x_tile, og_tile)

    # Convert to fp16 for the remaining element‑wise ops
    proj_left  = acc_lp.to(tl.float16)
    gate_left  = acc_lg.to(tl.float16)
    proj_right = acc_rp.to(tl.float16)
    gate_right = acc_rg.to(tl.float16)
    out_gate   = acc_og.to(tl.float16)

    # Sigmoid for the gate tensors
    gate_left  = 1.0 / (1.0 + tl.exp(-gate_left))
    gate_right = 1.0 / (1.0 + tl.exp(-gate_right))
    out_gate   = 1.0 / (1.0 + tl.exp(-out_gate))

    # Apply mask (mask already broadcast)
    left  = proj_left * gate_left * mask_tile
    right = proj_right * gate_right * mask_tile

    # Store results (guard against out‑of‑bounds in the trailing tiles)
    store_mask = mask_m[:, None] & mask_h[None, :]

    tl.store(left_out_ptr + (offs_m[:, None] * stride_left_m + offs_h[None, :] * stride_left_h),
             left,
             mask=store_mask)

    tl.store(right_out_ptr + (offs_m[:, None] * stride_right_m + offs_h[None, :] * stride_right_h),
             right,
             mask=store_mask)

    tl.store(out_gate_out_ptr + (offs_m[:, None] * stride_outgate_m + offs_h[None, :] * stride_outgate_h),
             out_gate,
             mask=store_mask)


# ----------------------------------------------------------------------
# Helper to launch the fused projection/gate kernel.
# ----------------------------------------------------------------------
def _fused_proj_gate(
    x: torch.Tensor,               # [B,N,N,dim] fp16 (already layer‑norm’d)
    mask: torch.Tensor,            # [B,N,N] fp16   (may be all ones)
    lp_w: torch.Tensor,            # [dim, hidden] fp16 (transposed weight)
    lg_w: torch.Tensor,
    rp_w: torch.Tensor,
    rg_w: torch.Tensor,
    og_w: torch.Tensor,
    hidden_dim: int,
    use_mask: bool,
) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Returns (left, right, out_gate) each shaped [B,N,N,hidden] fp16.
    """
    B, N, _, dim = x.shape
    M = B * N * N
    device = x.device

    # flatten the pairwise dimension
    x_flat = x.view(M, dim).contiguous()
    mask_flat = mask.view(M).contiguous()       # will be ignored if use_mask=False

    # Allocate output buffers
    left_out = torch.empty((M, hidden_dim), device=device, dtype=torch.float16)
    right_out = torch.empty_like(left_out)
    out_gate_out = torch.empty_like(left_out)

    # Strides (in elements, not bytes)
    s_x_m, s_x_k = x_flat.stride()
    s_mask_m = mask_flat.stride(0)

    # Weight strides (transposed weight: K × H)
    s_lp_k, s_lp_h = lp_w.stride()
    s_lg_k, s_lg_h = lg_w.stride()
    s_rp_k, s_rp_h = rp_w.stride()
    s_rg_k, s_rg_h = rg_w.stride()
    s_og_k, s_og_h = og_w.stride()

    s_left_m, s_left_h = left_out.stride()
    s_right_m, s_right_h = right_out.stride()
    s_outgate_m, s_outgate_h = out_gate_out.stride()

    # Tile sizes – chosen to fit H100 registers and shared memory.
    BLOCK_M = 64
    BLOCK_N = 64          # hidden‑dim tile
    BLOCK_K = 32          # input‑dim reduction tile

    grid = (tl.cdiv(M, BLOCK_M), tl.cdiv(hidden_dim, BLOCK_N))

    _proj_gate_all_kernel[grid](
        x_flat,
        mask_flat,
        lp_w,
        lg_w,
        rp_w,
        rg_w,
        og_w,
        left_out,
        right_out,
        out_gate_out,
        M,
        dim,
        hidden_dim,
        s_x_m,
        s_x_k,
        s_mask_m,
        s_lp_k,
        s_lp_h,
        s_lg_k,
        s_lg_h,
        s_rp_k,
        s_rp_h,
        s_rg_k,
        s_rg_h,
        s_og_k,
        s_og_h,
        s_left_m,
        s_left_h,
        s_right_m,
        s_right_h,
        s_outgate_m,
        s_outgate_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        USE_MASK=use_mask,
        num_warps=8,
        num_stages=3,
    )
    # Reshape to the original 4‑D layout
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)
    out_gate = out_gate_out.view(B, N, N, hidden_dim)
    return left, right, out_gate


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using cuBLAS.
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns out  : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # Flatten batch and hidden dimensions => [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM C = A @ Bᵀ ; both A and B are fp16 tensors
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Entry point for the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    x = x.to(dtype_fp16)  # now fp16 for the rest of the pipeline

    # -----------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 and transpose for GEMM
    # -----------------------------------------------------------------
    # projector & gating weights
    lp_w = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16).t().contiguous()   # [dim, hidden]
    lg_w = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16).t().contiguous()
    rp_w = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16).t().contiguous()
    rg_w = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16).t().contiguous()
    og_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16).t().contiguous()
    # output projection & its layer‑norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)   # shape [dim, hidden]

    # -----------------------------------------------------------------
    # 3️⃣ Fuse all linear transforms + gating + mask (if any) in Triton
    # -----------------------------------------------------------------
    # Ensure mask is fp16 (if not present we still pass a tensor – it's ignored when nomask=True)
    mask_fp16 = mask_tensor.to(dtype=dtype_fp16) if mask_tensor is not None else torch.ones_like(input_tensor[..., 0], dtype=dtype_fp16)

    left, right, out_gate = _fused_proj_gate(
        x,
        mask_fp16,
        lp_w,
        lg_w,
        rp_w,
        rg_w,
        og_w,
        hidden_dim,
        use_mask=not nomask,
    )
    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction using batched tensor‑core GEMM
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) and output‑gate multiplication
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 6️⃣ Final linear projection back to model dimension (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
