@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # out-of-bound masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute pointer for C (output)
    # set accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # using FP32 accumulation
    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = offs_k < K

        # Load A tile
        A_ptrs = A_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        A = tl.load(A_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (transposed view)
        B_ptrs = B_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        B = tl.load(B_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Dot product (A * B)
        acc += tl.dot(A, B)  # shape (BLOCK_M, BLOCK_N)
    
    # store
    C_ptrs = C_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(C_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

```python
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # track base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate k tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # Load B tile (transposed view)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Convert to fp32 for accumulation
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)
        acc += tl.dot(a_f32, b_f32)

    # Write back
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def _batched_matmul_kernel(
   a_ptr, b_ptr, c_ptr,
   BATCH, M, N, K,
   stride_a_batch, stride_a_m, stride_a_k,
   stride_b_batch, stride_b_k, stride_b_n,
   stride_c_batch, stride_c_m, stride_c_n,
   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pidb = tl.program_id(0)
    pidm = tl.program_id(1)
    pidn = tl.program_id(2)
    offs_m = pidm * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pidn * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N
    # Pointers for batch
    a_ptr_batch = a_ptr + pidb * stride_a_batch
    b_ptr_batch = b_ptr + pidb * stride_b_batch
    c_ptr_batch = c_ptr + pidb * stride_c_batch
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ---------------------------------------------------
# Triton kernel: fuse projection, gate and mask
# ---------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,      # *fp16 [B, N, N, H]
    gate_ptr,      # *fp16 [B, N, N, H] (sigmoid already)
    mask_ptr,      # *fp16 [B, N, N]
    out_ptr,       # *fp16 [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load mask tile (BLOCK_I, BLOCK_J)
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16

    # Loop over hidden tiles
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # Load proj tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Load gate tile
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # out_tile = proj * gate * mask (mask broadcast on hidden)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Multiply projection, gate and mask (mask broadcast over hidden dim)."""
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)
    # Strides in elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()
    # Tuned block sizes
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8
    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ---------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ)
# ---------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,  # *fp16 [B*, M, K]   A = left
    b_ptr,  # *fp16 [B*, N, K]   B = right (will be read as K×N)
    c_ptr,  # *fp16 [B*, M, N]   C = output
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch index   (0 .. BATCH-1)
    pid_m = tl.program_id(1)   # row tile
    pid_n = tl.program_id(2)   # col tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tile of A  (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load tile of Bᵀ  (K x N)
        # b is stored [batch, N, K]; we interpret it as K×N by swapping the strides
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_triton_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using Triton.
    left/right: [B, N, N, H] (fp16)
    Returns out: [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # Rearrange to [B, H, N, N] then flatten batch*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_flat = B * H
    left_flat = left_perm.view(batch_flat, N, N)   # [B*H, M, K]
    right_flat = right_perm.view(batch_flat, N, N) # [B*H, N, K]

    # Output buffer
    out_flat = torch.empty_like(left_flat)  # fp16

    # Strides in elements
    s_a = left_flat.stride()   # (M*K, K, 1)
    s_b = right_flat.stride()  # (N*K, K, 1)
    s_c = out_flat.stride()

    # Tuning parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_flat,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_flat,
        N,
        N,
        N,
        s_a[0],
        s_a[1],
        s_a[2],
        s_b[0],
        s_b[2],  # stride for K in B matrix (col dimension)
        s_b[1],  # stride for N (row dimension)
        s_c[0],
        s_c[1],
        s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # Reshape back to original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Main entry point
# ---------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    This implementation fuses:
      • Mask‑gate multiplication (proj * gate * mask) via a Triton kernel.
      • The O(N³) contraction left @ rightᵀ using a batched GEMM Triton kernel.
      • All linear projections that share the same input are collapsed into a single
        matrix‑multiply (5 × hidden_dim outputs) to avoid repeated reads of the
        large [B,N,N,dim] tensor.
    The algorithm stays in FP16 throughout the heavy work and only materialises
    the final result in FP32.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Cast to half precision for all subsequent ops
    x = x.to(dtype_fp16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse all linear transforms that read ``x`` into one matmul.
    #    The concatenated weight matrix has shape (5*hidden_dim, dim).
    # -----------------------------------------------------------------
    # Concatenate weights (biases are all False in the reference model)
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]
    # Stack on dim=0 (out_features)
    w_concat = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0
    ).to(device=device, dtype=dtype_fp16)

    B, N, _, _ = input_tensor.shape
    # Flatten x to (B·N·N, dim) for the matmul
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    # Single fused linear
    y_flat = torch.nn.functional.linear(x_flat, w_concat)  # (B*N*N, 5*hidden_dim)

    # Reshape back to [B, N, N, 5*hidden_dim] and split
    y = y_flat.view(B, N, N, 5 * hidden_dim)
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        y, hidden_dim, dim=-1
    )
    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 3️⃣ Apply mask & gate (mask‑gate fusion in Triton if needed)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure arbitrary mask dtype is handled (bool or float)
        mask_fp16 = mask_tensor.to(dtype=dtype_fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # Simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction with a batched GEMM kernel
    # -----------------------------------------------------------------
    out = _batched_matmul_triton_fp16(left, right)  # [B, N, N, hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (hidden_dim) and out‑gate
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 6️⃣ Final linear projection back to model dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```

"""
Highly‑optimized forward pass for the outgoing TriMul operator.

Key ideas
---------
1️⃣ First LayerNorm (float32 → FP16) – cheap compared to the O(N³) part.
2️⃣ All linear transforms that read the same input `x` are fused into a single
   matrix‑multiply: we concatenate the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) → shape
   (5·hidden_dim, dim). A single `torch.nn.functional.linear` therefore reads
   the huge input tensor only once.
3️⃣ Mask‑gate multiplication (proj * gate * mask) is fused with a Triton kernel
   (`_mask_gate_kernel`).
4️⃣ The cubic contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM (`_batched_gemm_kernel`) that computes
   `left @ rightᵀ` for each hidden slice, exploiting Tensor‑Cores directly.
5️⃣ Second LayerNorm, out‑gate sigmoid and final linear projection finish the
   forward pass.
All heavy work is performed in FP16; the result is cast back to FP32 for the
caller.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask   (mask broadcast on hidden dim H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # i / j coordinates for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – this will be broadcast later.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles.
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16, shape [I,J,H]

        # Load gate tile (already passed through sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Fuse: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """proj, gate : [B,N,N,H] (fp16) ; mask : [B,N,N] (fp16) → out = proj*gate*mask."""
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for out = left @ rightᵀ   (O(N³) contraction)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                # *fp16  [B_flat, M, K]  (left)
    b_ptr,                # *fp16  [B_flat, N, K]  (right, will be read as K×N)
    c_ptr,                # *fp16  [B_flat, M, N]  (output)
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch index (0 … BATCH‑1)
    pid_m = tl.program_id(1)          # row‑tile
    pid_n = tl.program_id(2)          # col‑tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator (use FP32 for safety, Tensor‑Core dot will still be fast)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tile of A   (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load tile of Bᵀ   (K × N) – note that right is stored as [batch,N,K]
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_triton_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a custom Triton batched GEMM.
    left/right : [B, N, N, H] (fp16)
    Returns     : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # bring hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_flat = B * H
    left_flat = left_perm.view(batch_flat, N, N)       # [B*H, M, K]
    right_flat = right_perm.view(batch_flat, N, N)     # [B*H, N, K]

    out_flat = torch.empty_like(left_flat)  # fp16

    s_a = left_flat.stride()   # (M*K, K, 1)
    s_b = right_flat.stride()  # (N*K, K, 1)
    s_c = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_flat,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_flat,
        N,
        N,
        N,
        s_a[0],
        s_a[1],
        s_a[2],
        s_b[0],
        s_b[2],         # stride over K in B (col dimension)
        s_b[1],         # stride over N in B (row dimension)
        s_c[0],
        s_c[1],
        s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    This implementation fuses mask‑gate multiplication and the cubic
    contraction using Triton kernels, while re‑using a single large linear
    layer to read the input tensor only once.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # everything after this runs in fp16

    # -----------------------------------------------------------------
    # 2️⃣ Fuse all linear layers that read `x` (5 matrices) into one matmul
    # -----------------------------------------------------------------
    # Concatenate the five weight matrices (order matters for the split)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*hidden_dim, dim)
    ).to(device=device, dtype=dtype_fp16)

    B, N, _, _ = input_tensor.shape
    x_flat = x.view(-1, dim)                     # (B·N·N, dim)
    y_flat = torch.nn.functional.linear(x_flat, w_concat)  # (B·N·N, 5·hidden_dim)

    # Split back into the five branches
    y = y_flat.view(B, N, N, 5 * hidden_dim)
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        y, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 3️⃣ Apply mask + gate (mask‑gate kernel if a mask is present)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (Triton)
    # -----------------------------------------------------------------
    out = _batched_matmul_triton_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (hidden_dim) + out‑gate
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 6️⃣ Final linear projection back to model dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 7️⃣ Cast to FP32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
