@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,  # *fp16
    gate_ptr,  # *fp16
    mask_ptr,  # *fp16 mask flattened
    hidden_dim: tl.constexpr,
    size: tl.constexpr,

    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a

Outgoing TriMul (AlphaFold‑3) forward pass – optimized for H100.

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32) → cast to fp16.
2.  Stack the five weight matrices (left_proj, left_gate, right_proj,
    right_gate, out_gate) into one big matrix and perform a single
    half‑precision GEMM.  This reduces memory traffic by reading the
    input only once.

4.  If a mask is present, fuse ``proj * gate * mask`` in‑place with a

    element‑wise multiply.
5.  Heavy contraction

6.  Second LayerNorm (fp16), multiply by the out‑gate, and a final
    linear projection back to the original channel dimension.
7.  Cast the result to float32 for downstream code.

    gate_ptr,          # *fp16 gate tensor (same shape)
    mask_ptr,          # *fp16 mask tensor (flattened, shape [B*N*N])
    hidden_dim: tl.constexpr,   # hidden dimension size (H)

            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*hidden_dim, dim)

    B, N, _, _ = input_tensor.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    # One fused linear layer
    fused = torch.nn.functional.linear(x_flat, stacked_weight)  # (B*N*N, 5*H)

    # Reshape back to 4‑D
    fused = fused.view(B, N, N, 5 * hidden_dim)

    # -----------------------------------------------------------------
    # 3️⃣ Slice results and compute sigmoid gates
    # -----------------------------------------------------------------
    h = hidden_dim
    left_proj   = fused[..., 0:h]               # (B, N, N, H)
    left_gate_r = fused[..., h:2 * h]
    right_proj  = fused[..., 2 * h:3 * h]
    right_gate_r = fused[..., 3 * h:4 * h]
    out_gate_r  = fused[..., 4 * h:5 * h]

    left_gate   = torch.sigmoid(left_gate_r)
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)

    # -----------------------------------------------------------------
    # 4️⃣ Apply mask + gate (in‑place) – Triton fusion if mask present
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                # (B*N*N)

        size = left_proj.numel()                      # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left_proj = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right_proj = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM
    # -----------------------------------------------------------------
    contract_out = _batched_gemm_fp16(left_proj, right_proj)   # (B, N, N, H) fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    contract_out = torch.nn.functional.layer_norm(
        contract_out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # -----------------------------------------------------------------
    # 7️⃣ Final gating + projection
    # -----------------------------------------------------------------
    contract_out = contract_out * out_gate                     # apply out‑gate
    to_out_w = weights["to_out.weight"].to(torch.float16)    # (dim, hidden_dim)
    out = torch.nn.functional.linear(contract_out, to_out_w)  # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
