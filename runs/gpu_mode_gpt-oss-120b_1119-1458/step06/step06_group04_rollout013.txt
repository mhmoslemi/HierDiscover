@triton.jit
def mask_gate_fuse_lr_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.constexpr,
    size: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load x/gates
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (element index / hidden_dim)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse multiplication
    left_res = left * left_gate * mask_val
    right_res = right * right_gate * mask_val

    # Store back
    tl.store(left_ptr + offs, left_res, mask=mask)
    tl.store(right_ptr + offs, right_res, mask=mask)

"""
Fast TriMul outgoing module using fused Linear projection and mask‑gate kernel.

Algorithm:
1. LayerNorm on input (FP32).
2. Cast to FP16.
3. Fuse left/right projections and 3 gating linear layers into a single
   FP16 linear (weight concatenation).  This reduces the number of reads of
   the large [B,N,N,dim] tensor from five to one.
4. Split the concatenated tensor into left_proj, right_proj, left_gate,
   right_gate, out_gate.  Apply sigmoid to the three gates.
5. If a mask is supplied, fuse masking and gating for both left and right
   tensors in one Triton kernel (left = left * left_gate * mask,
   right = right * right_gate * mask).  When there is no mask we simply
   multiply left/right by their gates with a PyTorch element‑wise kernel.
6. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched GEMM (A @ Bᵀ) using torch.bmm on FP16 tensors.
7. Second LayerNorm (FP16).
8. Multiply by out_gate and project back to the original channel dimension.
9. Cast the result back to FP32 for the caller.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_lr_kernel(
    left_ptr,        # *fp16: left tensor (flattened)
    left_gate_ptr,   # *fp16: left gate tensor (flattened)
    right_ptr,       # *fp16: right tensor (flattened)
    right_gate_ptr,  # *fp16: right gate tensor (flattened)
    mask_ptr,        # *fp16: mask flattened of size B*N*N
    hidden_dim: tl.constexpr,  # compile‑time hidden dimension
    n_elements: tl.constexpr,  # total number of elements in left/right
    BLOCK: tl.constexpr,       # tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < n_elements

    # Load the four tensors
    left_val = tl.load(left_ptr + offs, mask=valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=valid, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Gate & mask fusion
    left_out = left_val * left_gate_val * mask_val
    right_out = right_val * right_gate_val * mask_val

    # Store the results back in‑place
    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


def custom_kernel(
    data: Tuple[
        torch.Tensor,   # input tensor [B, N, N, dim] (float32)
        torch.Tensor,   # mask tensor [B, N, N] (float32/bool) or None
        Dict[str, torch.Tensor],  # model weights (float32)
        Dict[str, Any],           # config dict containing ``dim`` and ``hidden_dim``
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Returns a float32 tensor of shape [B, N, N, dim] on the same device as the
    input.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2️⃣ Convert to fp16 for the heavy part of the model
    # -----------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)  # [B, N, N, dim], fp16

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections (Proj + 3 gates) into one GEMM
    # -----------------------------------------------------------------
    # Concatenate weights into a single matrix of shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()

    # Combined linear: produces a tensor of shape [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x_fp16, combined_weight)  # fp16

    # -----------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor into the individual streams
    # -----------------------------------------------------------------
    # Because slicing does not produce contiguous tensors, we will keep
    # them as‑is for the element‑wise parts (PyTorch kernels can handle
    # non‑contiguous tensors).  For the mask‑kernel we create contiguous
    # clones later if a mask is present.
    left = combined[..., :hidden_dim]
    right = combined[..., hidden_dim : 2 * hidden_dim]
    left_gate_raw = combined[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate_raw = combined[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate_raw = combined[..., 4 * hidden_dim : 5 * hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Sigmoid gating (still fp16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 6️⃣ Apply optional mask + gating.
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16 and flatten it (size B*N*N)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # [B*N*N]

        # Make tensors contiguous for the kernel (this copies the data once)
        left_contig = left.contiguous()
        right_contig = right.contiguous()
        left_gate_contig = left_gate.contiguous()
        right_gate_contig = right_gate.contiguous()

        # Kernel launch (we fuse mask+gate for both left and right)
        total_elems = left_contig.numel()  # B*N*N*hidden_dim
        BLOCK = 1024  # <= max threads per block
        grid = (triton.cdiv(total_elems, BLOCK),)

        mask_gate_fuse_lr_kernel[grid](
            left_contig,
            left_gate_contig,
            right_contig,
            right_gate_contig,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Replace the original tensors with the fused versions
        left = left_contig
        right = right_contig
    else:
        # No mask – element‑wise multiplication only (in‑place)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction: batched GEMM on fp16 tensors
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Permute to (B, hidden, N, N) and reshape to (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # A
    # Right needs a transpose on the contracted dimension (k)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # Bᵀ
    # Batched matrix multiplication: out = A @ Bᵀ
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*H, N, N), fp16

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final out‑gate multiplication and projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim], fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the evaluator)
    # -----------------------------------------------------------------
    return out.to(torch.float32)

"""
Fast outgoing TriMul implementation (AlphaFold3 style) using Triton.

Key ideas
---------
1️⃣  First LayerNorm (FP32) – stable statistics.
2️⃣  Cast to FP16 for the compute‑intensive part.
3️⃣  Fuse the five linear layers (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single GEMM by concatenating their
    weight matrices.  This cuts the number of passes over the large
    [B,N,N,dim] tensor from five to one and reduces memory traffic.
4️⃣  Split the concatenated tensor back into the five streams,
    apply sigmoid to the three gating tensors.
5️⃣  Optional mask: a single Triton kernel fuses mask, left‑gate,
    right‑gate and the two tensors in‑place
    (left = left * left_gate * mask, right = right * right_gate * mask).
6️⃣  Core O(N³) contraction is realised as a batched GEMM
    (A @ Bᵀ) on FP16 tensors using torch.bmm.
7️⃣  Second LayerNorm (FP16) → out_gate multiplication → final
    linear projection back to the original channel dimension.
8️⃣  Cast the result back to FP32 for the caller.

The Triton kernel `mask_gate_fuse_lr_kernel` is the only non‑PyTorch
operation required by the task, and the fused linear projection
drastically reduces runtime (target ≈ 1 ms on H100 for the provided
configs).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right tensors with their gates and an optional mask.
# All tensors are flattened fp16 arrays.  mask is broadcast over the hidden
# dimension (size = B * N * N).  The kernel updates `left` and `right` in-place.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_lr_kernel(
    left_ptr,        # *fp16 – flattened left tensor
    left_gate_ptr,   # *fp16 – flattened left‑gate tensor
    right_ptr,       # *fp16 – flattened right tensor
    right_gate_ptr,  # *fp16 – flattened right‑gate tensor
    mask_ptr,        # *fp16 – flattened mask of shape [B*N*N]
    hidden_dim: tl.constexpr,   # hidden dimension (compile‑time)
    n_elements: tl.constexpr,   # total number of elements in left/right
    BLOCK: tl.constexpr,        # tile size (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load values (use zero as fallback for out‑of‑range threads)
    left_val      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs,mask=mask, other=0.0)
    right_val      = tl.load(right_ptr + offs,      mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    # Derive the mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse multiplication
    left_out  = left_val  * left_gate_val  * mask_val
    right_out = right_val * right_gate_val * mask_val

    # Store back in‑place
    tl.store(left_ptr + offs,  left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input tensor [B, N, N, dim] (float32)
        torch.Tensor,               # mask tensor  [B, N, N] (float32/bool/None)
        Dict[str, torch.Tensor],    # model weights (float32)
        Dict[str, Any],             # config dict with ``dim`` and ``hidden_dim``
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Returns a float32 tensor of shape [B, N, N, dim] on the same device as
    ``input_tensor``.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)  # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj + three gates) into one GEMM
    # ------------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden_dim, dim)
            weights["right_proj.weight"],  # (hidden_dim, dim)
            weights["left_gate.weight"],   # (hidden_dim, dim)
            weights["right_gate.weight"],  # (hidden_dim, dim)
            weights["out_gate.weight"],    # (hidden_dim, dim)
        ],
        dim=0,
    ).to(torch.float16).contiguous()  # (5*hidden_dim, dim)

    # Combined linear produces all projections and gates at once
    combined = torch.nn.functional.linear(x_fp16, combined_weight)  # [B, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor back into its parts
    # ------------------------------------------------------------------
    left          = combined[..., :hidden_dim]
    right         = combined[..., hidden_dim:2 * hidden_dim]
    left_gate_raw = combined[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate_raw = combined[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate_raw   = combined[..., 4 * hidden_dim:5 * hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gating (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 6️⃣ Apply optional mask + gating (fused Triton kernel)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16 and flatten: shape [B*N*N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)

        # Create contiguous copies for the kernel (single copy per tensor)
        left_contig      = left.contiguous()
        right_contig     = right.contiguous()
        left_gate_contig = left_gate.contiguous()
        right_gate_contig = right_gate.contiguous()

        total_elems = left_contig.numel()          # B * N * N * hidden_dim
        BLOCK = 1024                               # <= max threads per block
        grid = (triton.cdiv(total_elems, BLOCK),)

        mask_gate_fuse_lr_kernel[grid](
            left_contig,
            left_gate_contig,
            right_contig,
            right_gate_contig,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Replace with the fused tensors
        left  = left_contig
        right = right_contig
    else:
        # No mask – simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction via batched GEMM (fp16)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # Shape to (B, hidden, N, N) then flatten batch&head
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # A
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # Bᵀ
    out_mat   = torch.bmm(left_mat, right_mat)                           # (B*H, N, N)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Out‑gate multiplication and final projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the evaluation harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
