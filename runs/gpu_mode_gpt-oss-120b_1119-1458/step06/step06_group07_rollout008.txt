"""
TriMul (outgoing) forward pass – highly‑fused Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → fp16).
2. Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single matrix multiply.
3. Apply sigmoid to the three gate tensors.
4. Optionally fuse mask·gate·tensor multiplication for `left` and `right`
   using a tiny Triton kernel.
5. Heavy O(N³) contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed as a batched GEMM with Tensor‑Core fp16.
6. Second Layer‑Norm over the hidden dimension (fp16).
7. Element‑wise gating with `out_gate` and a final linear projection back to
   `dim`.
8. Cast the result to float32 (the harness expects float32).

The kernel reduces the original 7 matrix multiplications to 3 (two
batched GEMMs + a single fused linear), and uses Triton only for the mask‑gate
fusion (the only operation not covered efficiently by cuBLAS).  This yields a
significant speed‑up while keeping the exact same numerical result (within
floating‑point tolerances).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask·gate·tensor multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated (flattened)
    gate_ptr,       # *fp16  gate tensor (same shape as x)
    mask_ptr,       # *fp16  mask tensor (flattened, one element per vector)
    hidden_dim,     # i32   hidden dimension H
    size,           # i32   total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load tensors
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Broadcast mask across hidden dimension.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM (fp16) performing the O(N³) contraction.
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H) in fp16
    Returns:
        out: (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and multiply A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------

) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Args:
        data: (input, mask, weights, config)
            - input:  Tensor of shape [B, N, N, dim] (float32)
            - mask:   Tensor of shape [B, N, N] (float32 or bool) – may be ignored.
            - weights: dict containing the LayerNorm and Linear weights (float32).
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool).

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 0️⃣ Disable autograd – we only need a forward pass
    # -----------------------------------------------------------------
    torch.set_grad_enabled(False)

    # -----------------------------------------------------------------
    # 1️⃣ First layer‑norm (float32 → fp16)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # PyTorch's layer_norm is already highly optimized; keep it in float32
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    # Cast to fp16 for the remaining compute‑heavy stages
    x_norm = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single matmul
    # -----------------------------------------------------------------
    # Gather the five weight matrices (no bias in the original implementation)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension: shape (5*H, dim)
    weight_all = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0
    ).to(torch.float16)   # fp16 for Tensor‑Core matmul

    # Single fused linear: result shape [B, N, N, 5*hidden_dim]
    fused = torch.nn.functional.linear(x_norm, weight_all)

    # Split the fused tensor into the five logical tensors
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        fused, hidden_dim, dim=-1
    )

    # -----------------------------------------------------------------
    # 3️⃣ Compute sigmoid gates (fp16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 4️⃣ Apply mask + gate to left/right (fused Triton kernel)
    # -----------------------------------------------------------------
    if not nomask:
        # mask is broadcasted over the hidden dimension; flatten it.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                     # [B*N*N]
        size = left_proj.numel()                           # B*N*N*hidden_dim
        BLOCK = 1024                                        # ≤ 1024 threads per block
        grid = (triton.cdiv(size, BLOCK),)

        # left := left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right := right_proj * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – batched GEMM (Tensor‑Core, fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left_proj, right_proj)   # [B, N, N, hidden_dim]  fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Gating with out_gate and final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)      # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 (expected by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```

This implementation fuses the five per‑position linear transforms, uses a tiny Triton kernel for the mask·gate multiplication, and keeps the heavy O(N³) contraction as an optimized batched GEMM. The overall memory traffic is reduced and the number of kernel launches is minimized, delivering a substantial speed‑up compared with the original reference implementation.
