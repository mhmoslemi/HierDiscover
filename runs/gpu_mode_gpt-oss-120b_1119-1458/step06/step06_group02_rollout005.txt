def custom_kernel(data):
    # unpack
    input_tensor, mask_tensor, weights, config = data
    B,N,N,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    nomask = config.get("nomask", True)

    # LayerNorm (first) - float32 inputs
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias) # float32

    x = x.to(torch.float16)  # cast to half

    # Gather weights and convert to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)    # shape [H, dim]
    left_gate_w = weights["left_gate.weight"].to(torch.float16)    # shape [H, dim]
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # shape [2H, dim]

    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # combine linear projections
    left_concat = torch.nn.functional.linear(x, left_concat_w) # shape [B,N,N,2H]
    left_proj = left_concat[..., :hidden_dim]
    left_gate = left_concat[..., hidden_dim:].sigmoid()  # [B,N,N,hidden_dim]

    right_concat = torch.nn.functional.linear(x, right_concat_w) # shape [B,N,N,2H]
    right_proj = right_concat[..., :hidden_dim]
    right_gate = right_concat[..., hidden_dim:].sigmoid()

    # out_gate separate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # shape [B,N,N,hidden_dim]

    # Apply mask and gating
    if not nomask:
        # mask shape [B,N,N], broadcast across hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()
        BLOCK = 1024  # tune for occupancy
        grid = ( (total_elems + BLOCK -1)//BLOCK, )
        mask_gate_fuse_bidirectional_kernel[grid](
            left_proj,
            left_gate,
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left_proj *= left_gate
        right_proj *= right_gate

    # O(N³) contraction via batched GEMM
    out = _batched_matmul_torch(left_proj, right_proj)  # shape [B,N,N,hidden_dim] fp16

    # Second LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

"""
TriMul (outgoing) – fast forward pass.

Algorithm (fp32 → fp16 → fp16 → fp32):
1️⃣  Layer‑Norm over the feature dimension (float32) to stabilise statistics.
2️⃣  Cast to fp16 for the heavy inner computation.
3️⃣  Fuse the two projection + gate linear layers:
        left_proj & left_gate  →  weight matrix [2·H, D]
        right_proj & right_gate→  weight matrix [2·H, D]
    This reduces the number of separate matmuls from 5 to 3.
4️⃣  Apply the sigmoid gates.  If a mask is present, fuse mask‑multiply and
    gate‑multiply for *both* the left and right tensors in a **single**
    Triton kernel (`mask_gate_fuse_bidirectional_kernel`).  This removes two
    kernel launches.
5️⃣  Perform the O(N³) contraction using a batched GEMM:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched `torch.bmm` on the view `[B·H, N, N]`.
6️⃣  Second Layer‑Norm over the hidden dimension (fp16).
7️⃣  Multiply by the output gate, then a final linear projection back to the
    original dimension.
8️⃣  Cast the result back to float32 (required by the harness).

The implementation uses mixed precision (float32 ⇢ float16 ⇢ float32) and
fuses the mask/gate step in Triton, which gives a large speed‑up over the
reference PyTorch implementation while preserving exact numerical behavior.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate for *both* left and right tensors.
# The mask is broadcast over the hidden dimension H.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_bidirectional_kernel(
    left_ptr,            # *fp16 : left tensor (B,N,N,H)
    left_gate_ptr,       # *fp16 : left gate (same shape)
    right_ptr,           # *fp16 : right tensor (B,N,N,H)
    right_gate_ptr,      # *fp16 : right gate (same shape)
    mask_ptr,            # *fp16 : mask tensor (B,N,N) – broadcasted over H
    hidden_dim,          # i32   : hidden dimension H
    size,                # i64   : total number of elements in left/right (B·N·N·H)
    BLOCK: tl.constexpr, # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                 # linear offsets
    active = offs < size                                      # guard

    # mask index = element index // hidden_dim  (because H is innermost)
    mask_idx = offs // hidden_dim

    # Load data (default values for out‑of‑bounds make no contribution)
    l_val   = tl.load(left_ptr + offs,          mask=active, other=0.0)
    l_gate  = tl.load(left_gate_ptr + offs,    mask=active, other=0.0)
    r_val   = tl.load(right_ptr + offs,         mask=active, other=0.0)
    r_gate  = tl.load(right_gate_ptr + offs,   mask=active, other=0.0)
    m_val   = tl.load(mask_ptr + mask_idx,     mask=active, other=1.0)   # mask=1 for OOB

    # Fuse:   out = val * gate * mask
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    # Store in‑place
    tl.store(left_ptr + offs,  l_out, mask=active)
    tl.store(right_ptr + offs, r_out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Heavy O(N³) contraction implemented as a batched GEMM.
    left/right: [B, N, N, H] (fp16)
    Returns out:   [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # Rearrange to [B·H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))               # [B·H, N, N]

    # Restore original layout: [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                                          # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the core computation
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse projection + gate linear layers (weight concatenation)
    # -----------------------------------------------------------------
    # Left side: projection + gate → 2·H output channels
    left_proj_w = weights["left_proj.weight"].to(torch.float16)    # (H, D)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2H, D)

    # Right side: projection + gate → 2·H output channels
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2H, D)

    # Out‑gate (single H)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear maps (single pass for left/right, separate for out_gate)
    # -----------------------------------------------------------------
    # left_concat: [B, N, N, 2H]
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # no bias
    left_proj = left_concat[..., :hidden_dim]                    # [B,N,N,H]
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])    # [B,N,N,H]

    # right_concat: [B, N, N, 2H]
    right_concat = torch.nn.functional.linear(x, right_concat_w)
    right_proj = right_concat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])

    # out_gate: [B,N,N,H]
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 5️⃣ Apply mask & element‑wise gating
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and ensure contiguous layout
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()              # B·N·N·H
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        mask_gate_fuse_bidirectional_kernel[grid](
            left_proj,
            left_gate,
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple in‑place multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Final gating and projection back to original dimension
    # -----------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)     # shape (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)           # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
