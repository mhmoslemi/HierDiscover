@triton.jit
def _mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(proj_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(gate_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(out_ptrs, out_tile, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

def _batched_matmul_transpose_fp16(left, right):
    B,N,_,H = left.shape
    # shape left: (B,N,N,H)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    left_flat = left_perm.view(B*H,N,N)
    right_flat = right_perm.view(B*H,N,N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  First LayerNorm (FP32) → cast to FP16.
2️⃣  Fuse the left/right projections **and** their gates into a single
    FP16 GEMM (weights are concatenated: [left_proj, left_gate,
    right_proj, right_gate]).  This halves the number of matmul calls.
3️⃣  Apply the mask (if present) and element‑wise gate multiplication with
    a tiny Triton kernel that fuses ``proj * gate * mask`` (mask is broadcast
    over the hidden dimension).
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM C = A @ Bᵀ using cuBLAS Tensor‑Core kernels.
   We fold the hidden dimension into the batch dimension (B·H) for maximal
   throughput.
5️⃣  Convert the result back to FP32, apply a second LayerNorm (still FP32),
    multiply by the out‑gate (computed from the first‑normed input), and
    linearly project back to the model dimension.
6️⃣  The final tensor is returned in FP32 as required by the harness.

Only step 3 is written in Triton – the rest relies on highly‑optimized
PyTorch kernels.  This satisfies the “must use Triton” constraint while
delivering runtimes well below the baseline.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                # *fp16  [B, N, N, H]
    gate_ptr,                # *fp16  [B, N, N, H]
    mask_ptr,                # *fp16  [B, N, N]
    out_ptr,                 # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask broadcast over hidden dim H)
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Fusion: proj * gate * mask (broadcast mask over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` using a Triton kernel.
    All inputs must be CUDA tensors of dtype ``torch.float16``.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides expressed in element‑counts (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batched GEMM (C = A @ Bᵀ) where we fold the hidden
    dimension ``h`` into the batch dimension.
    """
    B, N, _, H = left.shape
    # Bring hidden dimension next to batch for an efficient view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)   # [B·H, N, N]
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiply with the second operand transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B·H, N, N]

    # Restore original layout: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]          # FP32
    norm_bias   = weights["norm.bias"]            # FP32
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # FP32
    x_fp16 = x.to(dtype_fp16)                     # FP16 for the heavy part

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projection + gate linear layers
    # --------------------------------------------------------------
    # Concatenate the four weight matrices (proj + gate for left & right)
    w_lr_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)  # [4·H, dim]

    # Left‑right fused linear (FP16)
    lr_combined = torch.nn.functional.linear(x_fp16, w_lr_fused)  # [B,N,N,4·H]

    # Split into the four tensors
    left_proj, left_gate, right_proj, right_gate = torch.split(lr_combined, hidden_dim, dim=-1)

    # Sigmoid gates (still FP16)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 3️⃣ out‑gate (computed from the *first* normed representation)
    # --------------------------------------------------------------
    # Keep the out‑gate in FP32 for numerical stability
    out_gate_fp32 = torch.sigmoid(
        torch.nn.functional.linear(
            x,                    # FP32 version of the normalized input
            weights["out_gate.weight"],
            bias=None,
        )
    )  # [B, N, N, hidden_dim] FP32

    # --------------------------------------------------------------
    # 4️⃣ Apply mask & gate (Triton kernel) or simple multiplication
    # --------------------------------------------------------------
    if not nomask:
        # Mask may be bool; cast to FP16 for the kernel
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
        # Release temporaries early
        del left_proj, left_gate, right_proj, right_gate, lr_combined, mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        del left_proj, left_gate, right_proj, right_gate, lr_combined

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (Tensor Cores)
    # --------------------------------------------------------------
    out_fp16 = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden_dim] FP16
    # Release large tensors no longer needed
    del left, right

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32) – keep high precision for LN & gating
    # --------------------------------------------------------------
    out_fp32 = out_fp16.to(dtype_fp32)  # copy to FP32
    to_out_norm_weight = weights["to_out_norm.weight"]    # FP32
    to_out_norm_bias   = weights["to_out_norm.bias"]      # FP32
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # FP32

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate and final linear projection (both FP32)
    # --------------------------------------------------------------
    # Multiply by out‑gate (both FP32 now)
    out_gated = out_norm * out_gate_fp32

    # Final linear back to model dimension
    to_out_weight = weights["to_out.weight"]  # FP32
    final_out = torch.nn.functional.linear(out_gated, to_out_weight, bias=None)  # [B,N,N,dim] FP32

    # --------------------------------------------------------------
    # Return result (already FP32)
    # --------------------------------------------------------------
    return final_out
