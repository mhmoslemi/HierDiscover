@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch & hidden
    b = pid_bh // H
    h = pid_bh % H

    # compute base offsets for left/right/out
    # For left and right:
    # offset = b * stride_left_batch + h * 1
    offs_batch = b * stride_left_batch + h

    # compute offsets for i and j indices within tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile
        a_ptrs = left_ptr + offs_batch \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (note using same stride for j dimension)
        b_ptrs = right_ptr + offs_batch \
                 + offs_j[:, None] * stride_right_i \
                 + offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # accumulate dot product
        acc = acc + tl.dot(a, b)

    # Store result
    out_ptrs = out_ptr + offs_batch \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass
================================
Implementation of the “outgoing” TriMul operation used in AlphaFold‑3 and related
models.  The algorithm follows the reference PyTorch version but is heavily
optimised for GPU execution:

1.  Layer‑norm the 4‑D input (float32 → float16 for the heavy part).
2.  Two linear projections (left/right) and three gating projections (left,
    right, out) are performed in half‑precision.
3.  If a mask is present we fuse the element‑wise mask & gate multiplication
    (left * left_gate * mask, right * right_gate * mask) with a tiny Triton
    kernel that works on the flattened view.
4.  The core O(N³) contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    is realised as a batched GEMM in a custom Triton kernel that works directly
    on the original [B,N,N,H] layout – thus avoiding the costly
    `permute(...).contiguous()` copy required by the naïve batched‑bmm approach.
    The kernel iterates over the reduction dimension K in tiles and accumulates
    in float32 before storing back to fp16.
5.  A second Layer‑Norm over the hidden dimension, output gating, and a final
    linear projection back to the original dimension are performed with ordinary
    PyTorch ops.
6.  The result is cast back to float32 (required by the harness).

Only the forward pass is needed and the implementation works for all
configurations listed in the problem statement, with an optional mask.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel to fuse mask & gate (mask is broadcast over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : hidden dimension size (H)
    size,           # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices for this block
    active = offs < size                               # OOB guard
    # mask index = linear index // hidden_dim (mask is not expanded over H)
    mask_offs = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)  # default mask = 1

    tl.store(x_ptr + offs, x * g * m, mask=active)


# ----------------------------------------------------------------------
# Triton kernel for the O(N³) contraction (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,        # *fp16 : left tensor  [B,N,N,H]
    right_ptr,       # *fp16 : right tensor [B,N,N,H]
    out_ptr,         # *fp16 : output tensor [B,N,N,H]
    B,               # i32   : batch size
    N,               # i32   : sequence length
    H,               # i32   : hidden dimension
    stride_left_batch,  # i32 : stride over batch    (B‑dim)
    stride_left_i,      # i32 : stride over first N (i‑dim)
    stride_left_k,      # i32 : stride over second N (k‑dim)
    stride_right_batch, # i32 : same as left batch stride
    stride_right_i,     # i32 : same as left i stride
    stride_right_k,     # i32 : same as left k stride
    stride_out_batch,   # i32 : stride over batch for output
    stride_out_i,       # i32 : stride over first N for output
    stride_out_j,       # i32 : stride over second N for output (j‑dim)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] for all (b,i,j,h).
    The hidden dimension H is folded into the batch dimension (B*H) so that each
    (b,h) pair is treated as an independent GEMM.
    """

    pid_i = tl.program_id(0)      # tile index along the i (row) axis
    pid_j = tl.program_id(1)      # tile index along the j (col) axis
    pid_bh = tl.program_id(2)     # flattened (batch, hidden) index

    # ------------------------------------------------------------------
    # 1️⃣  Resolve batch and hidden indices
    # ------------------------------------------------------------------
    b = pid_bh // H               # original batch index
    h = pid_bh % H                # hidden channel index

    # base offset for this (b, h) pair in the flattened view
    base_offset = b * stride_left_batch + h          # stride_h == 1 for fp16 tensors

    # ------------------------------------------------------------------
    # 2️⃣  Compute tile start coordinates
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3️⃣  Loop over the reduction dimension K (the second N dimension)
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load a tile from left (shape BLOCK_M × BLOCK_K) ----------
        a_ptrs = left_ptr + base_offset \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ------- load a tile from right (shape BLOCK_N × BLOCK_K) ----------
        b_ptrs = right_ptr + base_offset \
                 + offs_j[:, None] * stride_right_i \
                 + offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)

        # ------- accumulate dot product across K ---------------------------
        acc = acc + tl.dot(a, b)   # a: (M,K), b: (N,K) → (M,N) summed over K

    # ------------------------------------------------------------------
    # 4️⃣  Store the result back to the output tensor
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + base_offset \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton GEMM kernel
# ----------------------------------------------------------------------
def _tri_mul_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction using Triton.
    left/right: tensors shaped [B, N, N, H] (fp16, contiguous)
    Returns: tensor of the same shape and dtype (fp16)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16)

    # stride values (int64 → Python int)
    stride_left_batch = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)

    stride_right_batch = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)

    # Tile sizes – chosen to fit comfortably in H100 with 8 warps per block
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M ),      # i‑tiles
        ( (N + BLOCK_N - 1) // BLOCK_N ),      # j‑tiles
        B * H,                                 # (batch, hidden) dimension
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_right_batch,
        stride_right_i,
        stride_right_k,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps → 256 threads per block (fits H100)
    )
    return out


# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool) or None
        - weights_dict : dict with all model weights (float32)
        - config_dict  : must contain 'dim' and 'hidden_dim'; optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on same device as input)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        inp,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                         # [B, N, N, dim]  float32
    x = x.to(torch.float16)   # cast once for the heavy part

    # -----------------------------------------------------------------
    # 2️⃣ Cast model weights to fp16 (once)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)      # [B, N, N, H]
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 4️⃣ Gating tensors (sigmoid) – also fp16
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 5️⃣ Fuse mask & gate (if a mask is present)
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool, float, or a torch.Tensor on the same device
        mask_fp16 = mask.to(torch.float16).contiguous()
        total_elems = left.numel()                # B·N·N·H
        BLOCK = 1024                               # ≤ 2**10 elements per Triton launch
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just multiply by the gate (in‑place to keep memory low)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – custom Triton GEMM
    # -----------------------------------------------------------------
    out = _tri_mul_gemm_triton(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Output gating and final projection back to original dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)           # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
