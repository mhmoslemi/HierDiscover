@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bi, stride_Bk,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bound M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for C output tile base
    c_ptrs = C_ptr + batch_idx * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M,K)
        a_ptrs = A_ptr + batch_idx * stride_Ab + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (K,N), note we need B transposed (i.e., B[j,k] as B_T[k,j])
        # B original shape: (batch, N, K) with stride_Bb (N*N), stride_Bi (N), stride_Bk (1)
        # To compute B_T shape (K,N) we need pointers:
        #   row offset for k: offs_k (0..K-1) * stride_Bk (1)
        #   col offset for j: offs_n (0..N-1) * stride_Bi (N)
        b_ptrs = B_ptr + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bi
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # matrix multiply accumulate
        acc += tl.dot(a, b)

    # cast to FP16 and store output

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']
    device = input_tensor.device
    dp_type = torch.float16

    # Step 1: LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights['norm.weight'],
                     bias=weights['norm.bias'])
    x = x.to(dp_type)

    # Step 2: Linear projections
    left_proj = F.linear(x, weights["left_proj.weight"].to(device=device, dtype=dp_type))
    right_proj = F.linear(x, weights["right_proj.weight"].to(device=device, dtype=dp_type))

    # Step 3: Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device=device, dtype=dp_type)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device=device, dtype=dp_type)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device=device, dtype=dp_type)))

    # Step 4: Apply mask and gates
    nomask = config.get("nomask", True)
    if nomask:
        left_proj.mul_(left_gate)
        right_proj.mul_(right_gate)
    else:
        # ensure contiguous and cast mask to fp16
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(dp_type).contiguous()

        total = left_proj.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)
        _mask_gate_fusion_kernel[grid](
            left_ptr=left_proj,
            right_ptr=right_proj,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        del left_gate, right_gate, mask_fp16

    # Step 5: Contraction (TriMul)
    # Allocate out tensor
    out = torch.empty_like(left_proj)
    # Compute strides
    stride_batch = left_proj.stride(0)   # N*N*H
    stride_i = left_proj.stride(1)       # N*H
    stride_k = left_proj.stride(2)       # H
    # Size arguments
    B = input_tensor.shape[0]
    N = input_tensor.shape[1]  # seq_len
    H = hidden_dim

    # Choose block sizes (tuned)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # maybe 32 or 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)
    _tri_mul_gemm_kernel[grid](
        a_ptr=left_proj,
        b_ptr=right_proj,
        c_ptr=out,
        B=B,
        H=H,
        N=N,
        stride_a_batch=stride_batch,
        stride_a_i=stride_i,
        stride_a_k=stride_k,
        stride_b_batch=stride_batch,
        stride_b_i=stride_i,
        stride_b_k=stride_k,
        stride_c_batch=stride_batch,
        stride_c_i=stride_i,
        stride_c_j=stride_k,  # because j dimension uses stride_k (i.e., third axis)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Step 6: Second layernorm, gating, projection
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(device=device, dtype=dp_type),
                       bias=weights["to_out_norm.bias"].to(device=device, dtype=dp_type))

    out = out * out_gate   # broadcasted shape matches

    out = F.linear(out, weights["to_out.weight"].to(device=device, dtype=dp_type))

    # Cast to float32 for caller
    return out.to(torch.float32)
```

@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr, 
    B, H, N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # compute pid
    pid = tl.program_id(0) # head index (batch*H + head)
    pid_m = tl.program_id(1) # row tile index
    pid_n = tl.program_id(2) # col tile index

    # Compute batch and head indices
    batch_idx = pid // H
    head_idx = pid % H

    # compute base offset
    base_offset = batch_idx * stride_a_batch + head_idx  # because stride_h = 1

    # compute tile start indices
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute offsets for K loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load a tile
        a_ptrs = a_ptr + base_offset + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load b tile as transposed
        b_ptrs = b_ptr + base_offset + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write output
    c_ptrs = c_ptr + base_offset + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

```python
"""
Optimized outgoing TriMul implementation for AlphaFold3 using Triton.
- First performs LayerNorm and linear projections in fp32/fp16.
- Fuses mask application and gating into a Triton kernel.
- Executes the O(N³) pairwise contraction with a custom batched GEMM kernel
  that treats the hidden dimension as an extra batch dimension, avoiding
  costly permutes and extra memory copies.
- Finishes with a second LayerNorm, gating and output projection.
The kernel works for the provided configuration ranges and returns a
float32 tensor of shape [B, N, N, dim].
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel to fuse mask, left/right gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i64   (B·N·N·H)
    H,                 # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Each element belongs to a mask entry that spans H hidden slots
    mask_offs = offs // H

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel for the O(N³) contraction (outgoing TriMul)
# Computes for each hidden “head”:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr,               # *fp16  [B,N,N,H]
    b_ptr,               # *fp16  [B,N,N,H]
    c_ptr,               # *fp16  [B,N,N,H] (output)
    B,                   # i64  batch size
    H,                   # i64  hidden dimension
    N,                   # i64  sequence length
    stride_a_batch,      # i64
    stride_a_i,
    stride_a_k,
    stride_b_batch,
    stride_b_i,
    stride_b_k,
    stride_c_batch,
    stride_c_i,
    stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (batch*hidden, tile_m, tile_n)
    pid = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)        # tile row index
    pid_n = tl.program_id(2)        # tile col index

    # De‑compose combined index into (batch, head)
    batch_idx = pid // H
    head_idx = pid - batch_idx * H

    # Base offset for this (batch,head) slice.
    # In the original layout stride_h == 1, so we add head_idx directly.
    base = batch_idx * stride_a_batch + head_idx

    # Compute tile offsets inside the N×N plane
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load A tile (shape BLOCK_M × BLOCK_K) ----
        a_ptrs = a_ptr + base \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- Load B tile transposed (shape BLOCK_K × BLOCK_N) ----
        # B is stored as [batch, i, k, h]; we need B[j,k,h] = B[row=j, col=k]
        b_ptrs = b_ptr + base \
                 + offs_k[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ---- Matrix‑multiply‑accumulate ----
        acc += tl.dot(a, b)

    # ---- Write result back to C (shape BLOCK_M × BLOCK_N) ----
    c_ptrs = c_ptr + base \
             + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool / 0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 → stable statistics)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B,N,N,dim]  fp32

    # --------------------------------------------------------------
    # 2️⃣ Convert to fp16 for the heavy part
    # --------------------------------------------------------------
    x = x.to(fp16)

    # --------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 on‑device
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections → (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # fp16
    right = F.linear(x, right_proj_w)    # fp16

    # --------------------------------------------------------------
    # 5️⃣ Gating projections (sigmoid) → (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6️⃣ Fuse optional mask with the per‑head gates (in‑place)
    # --------------------------------------------------------------
    if nomask:
        left.mul_(left_gate)
        right.mul_(right_gate)
    else:
        # Ensure tensors are contiguous for Triton
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(fp16).contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporary tensors early
        del left_gate, right_gate, mask_fp16

    # --------------------------------------------------------------
    # 7️⃣ O(N³) contraction: batched GEMM A @ Bᵀ   (fp16, pure Triton)
    # --------------------------------------------------------------
    Bsz, N, _, H = left.shape  # H == hidden_dim

    # Allocate output tensor
    out = torch.empty_like(left)

    # Strides (all tensors share the same layout)
    stride_batch = left.stride(0)   # N * N * H
    stride_i     = left.stride(1)   # N * H
    stride_k     = left.stride(2)   # H
    # For the output we need stride for the j‑dimension (the 3rd axis)
    stride_j_out = left.stride(2)   # also H

    # Tiling parameters – tuned for H100 fp16 tensor‑core performance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 32‑wide reduction yields good occupancy

    grid = (
        Bsz * H,                                   # combined batch×head dimension
        (N + BLOCK_M - 1) // BLOCK_M,              # number of row tiles
        (N + BLOCK_N - 1) // BLOCK_N,              # number of column tiles
    )

    _tri_mul_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        B=Bsz,
        H=H,
        N=N,
        stride_a_batch=stride_batch,
        stride_a_i=stride_i,
        stride_a_k=stride_k,
        stride_b_batch=stride_batch,
        stride_b_i=stride_i,
        stride_b_k=stride_k,
        stride_c_batch=stride_batch,
        stride_c_i=stride_i,
        stride_c_j=stride_j_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Apply the final out‑gate and linear projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate                # element‑wise gate (fp16)
    out = F.linear(out, to_out_w)       # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # 10️⃣ Cast back to float32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
```

"""
Optimized outgoing TriMul (AlphaFold3) forward pass using Triton.

Steps
-----
1️⃣  LayerNorm on the input (FP32 → FP16).  
2️⃣  Linear projections (left/right) and gate projections (left/right/out) in FP16.  
3️⃣  Fuse optional mask with left/right gates using a Triton kernel (in‑place).  
4️⃣  Perform the O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    via a custom batched GEMM kernel that treats the hidden dimension as
    an additional batch axis, thus avoiding the costly permute + view.
5️⃣  Second LayerNorm, out‑gate multiplication and final linear projection back
    to the original `dim`.  The result is returned as float32.

The implementation works for the full range of test configurations
(B ∈ {1,2}, N ∈ {128,256,512,768,1024}, dim ∈ {128,384}, hidden_dim = 128)
and runs entirely on the GPU (H100).  No gradients are computed.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and per‑head gates (in‑place on left/right)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i64   (B·N·N·H)
    H,                 # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # each mask entry corresponds to H consecutive hidden entries
    mask_offs = offs // H

    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    tl.store(left_ptr  + offs, left  * left_gate  * mask_val, mask=valid)
    tl.store(right_ptr + offs, right * right_gate * mask_val, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# Computes, for each (batch, head):
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr,               # *fp16  [B,N,N,H]
    b_ptr,               # *fp16  [B,N,N,H]
    c_ptr,               # *fp16  [B,N,N,H] (output)
    B,                   # i64   batch size
    H,                   # i64   hidden dimension
    N,                   # i64   sequence length
    stride_a_batch,      # i64   stride over the batch dimension of A
    stride_a_i,          # i64   stride over the i‑axis of A
    stride_a_k,          # i64   stride over the k‑axis of A
    stride_b_batch,      # i64   stride over the batch dimension of B
    stride_b_i,          # i64   stride over the i‑axis of B (used as j)
    stride_b_k,          # i64   stride over the k‑axis of B
    stride_c_batch,      # i64   stride over the batch dimension of C
    stride_c_i,          # i64   stride over the i‑axis of C
    stride_c_j,          # i64   stride over the j‑axis of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (batch*head, tile_m, tile_n)
    pid = tl.program_id(0)          # combined batch*head index
    pid_m = tl.program_id(1)        # tile row index
    pid_n = tl.program_id(2)        # tile column index

    # De‑compose combined index
    batch_idx = pid // H
    head_idx  = pid - batch_idx * H   # equivalent to pid % H

    # Base address of the (batch,head) slice.
    # Original layout: stride_h == 1, so we can just add head_idx.
    base = batch_idx * stride_a_batch + head_idx

    # Tile start indices inside the N×N plane
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load A tile (M × K) ----
        a_ptrs = a_ptr + base \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- Load B tile transposed: (K × N) ----
        # B is stored as [batch, i, k, h]; we need B[j,k,h] for each j.
        b_ptrs = b_ptr + base \
                 + offs_k[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ---- Write tile back to C ----
    c_ptrs = c_ptr + base \
             + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool / 0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the raw input (FP32 → more stable stats)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]  FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy parts
    # --------------------------------------------------------------
    x = x.to(fp16)

    # --------------------------------------------------------------
    # 3️⃣ Cast all model weights to FP16 (same device)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections → (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # FP16
    right = F.linear(x, right_proj_w)  # FP16

    # --------------------------------------------------------------
    # 5️⃣ Gate projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6️⃣ Fuse mask (if any) and gating in‑place with Triton
    # --------------------------------------------------------------
    if nomask:
        # Simple in‑place gating – no mask to read.
        left.mul_(left_gate)
        right.mul_(right_gate)
    else:
        # Ensure tensors are contiguous for Triton
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_tensor.to(fp16).contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporaries early
        del left_gate, right_gate, mask_fp16

    # --------------------------------------------------------------
    # 7️⃣ O(N³) contraction using a custom Triton GEMM kernel
    # --------------------------------------------------------------
    Bsz, N, _, H = left.shape                # H == hidden_dim
    out = torch.empty_like(left)               # allocate output (FP16)

    # Strides in the original layout [B,N,N,H] (contiguous)
    stride_batch = left.stride(0)   # N * N * H
    stride_i     = left.stride(1)   # N * H
    stride_k     = left.stride(2)   # H
    stride_j_out = left.stride(2)   # same as stride_k for the j‑axis

    # Tile sizes – empirically good for H100 FP16 tensor‑core GEMM
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        Bsz * H,                                   # combined batch × head
        (N + BLOCK_M - 1) // BLOCK_M,              # rows tiles
        (N + BLOCK_N - 1) // BLOCK_N,              # cols tiles
    )

    _tri_mul_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        B=Bsz,
        H=H,
        N=N,
        stride_a_batch=stride_batch,
        stride_a_i=stride_i,
        stride_a_k=stride_k,
        stride_b_batch=stride_batch,
        stride_b_i=stride_i,
        stride_b_k=stride_k,
        stride_c_batch=stride_batch,
        stride_c_i=stride_i,
        stride_c_j=stride_j_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Apply final out‑gate and projected linear layer back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate                           # element‑wise gate
    out = F.linear(out, to_out_w)                  # (B,N,N,dim) FP16

    # --------------------------------------------------------------
    # 10️⃣ Cast to float32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
