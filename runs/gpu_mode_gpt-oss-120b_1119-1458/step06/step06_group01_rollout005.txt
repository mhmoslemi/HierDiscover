"""
Outgoing TriMul (AlphaFold‑3) forward pass – mixed‑precision Triton‑accelerated version.

Algorithm
---------
1️⃣  LayerNorm the 4‑D input in FP32 for numerical stability, then cast to FP16.
2️⃣  Five linear projections (two for the pairwise terms, three for sigmoid gates)
    are performed with FP16 weights (torch.nn.functional.linear).
3️⃣  If a mask is present we fuse:  proj * gate * mask  using a tiny Triton kernel.
   Without a mask we simply multiply the projections by their gates in‑place.
4️⃣  The O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is expressed as a batched GEMM (C = A @ Bᵀ) with cuBLAS Tensor‑Core kernels.
   We fold the batch and hidden dimensions (B·H) to obtain shape [B·H, N, N].
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection back
   to the model dimension (FP16).
6️⃣  Cast the final result back to FP32 (expected by the harness).

Only step 3 is implemented in Triton – the rest relies on highly‑optimized
PyTorch kernels.  This satisfies the “must use Triton” requirement while
keeping the intensive O(N³) work on cuBLAS for maximal throughput on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b,  stride_out_i,  stride_out_j,  stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask is broadcast over hidden dimension H)
    """
    pid_b = tl.program_id(0)   # batch dimension
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Tile coordinates (may go out‑of‑bounds, will be masked)
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask  (mask broadcasted on the last dim)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim)

    All inputs must be FP16 CUDA tensors. Returns a new FP16 tensor with
    the same shape as ``proj``.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    # Tile sizes – chosen to give good occupancy while staying within shared memory limits
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Helper: O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core kernels)
# ----------------------------------------------------------------------
def _batched_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H], dtype FP16, contiguous.

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H], dtype FP16.
    """
    B, N, _, H = left.shape

    # Fold batch and hidden dims → [B·H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM with the right‑hand side transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout → [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point – the custom kernel required by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : Tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32, bool, or None)
        - weights_dict : map from parameter name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim], dtype float32, same device as input.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True => no mask applied

    device = input_tensor.device
    eps = 1e-5  # LayerNorm epsilon – matches PyTorch's default

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerical stability) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # torch.nn.functional.layer_norm works with any dtype; we keep FP32 here.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    x = x.to(torch.float16)   # switch to FP16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2️⃣ Cast all model weights to FP16 once (device‑side)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim] (fp16)
    left  = torch.nn.functional.linear(x, left_proj_w)   # left_proj
    right = torch.nn.functional.linear(x, right_proj_w)  # right_proj

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – in‑place to avoid extra allocations
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w)
    left_gate.sigmoid_()
    right_gate = torch.nn.functional.linear(x, right_gate_w)
    right_gate.sigmoid_()
    out_gate   = torch.nn.functional.linear(x, out_gate_w)
    out_gate.sigmoid_()

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask (if present) and fuse projection with gating
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to FP16; mask is broadcast over hidden_dim inside the kernel
        mask_fp16 = mask_tensor.to(dtype=torch.float16)
        # Fuse: proj * gate * mask  using Triton
        left  = _apply_mask_gate(left,  left_gate,  mask_fp16)
        right = _apply_mask_gate(right, right_gate, mask_fp16)
    else:
        # No mask – just multiply by the gates in‑place
        left.mul_(left_gate)   # left = left * left_gate
        right.mul_(right_gate) # right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM (Tensor‑Core cuBLAS)
    # ------------------------------------------------------------------
    # Result: [B, N, N, hidden_dim] (fp16)
    out = _batched_contraction(left, right)

    # Free memory for left/right early (optional, helps memory reuse)
    del left, right

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to model dim
    # ------------------------------------------------------------------
    out.mul_(out_gate)                        # out = out * out_gate (in‑place)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast to FP32 as required by the evaluation harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
