```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16: tensor to write (in-place)
    gate_ptr,       # *fp16: gate tensor same shape as x
    mask_ptr,       # *fp16: mask flattened, length B*N*N
    total_elements, # i32: total # elements in x (== B*N*N*H)
    hidden_dim,     # i32: hidden dimension (gate channel)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear indices
    mask = offs < total_elements                # guard
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # every `hidden_dim` consecutive elements share the same mask element
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Fast forward pass of the outgoing TriMul operator.
    Strategy:
      1) Layer‑norm the input (FP32) then cast to FP16.
      2) Fuse the five separate Linear layers (left_proj, right_proj,
         left_gate, right_gate, out_gate) into a single matmul using a
         concatenated weight matrix – input is read once.
      3) Split the result, apply sigmoid to the gating tensors.
      4) Fuse (proj * gate * mask) with a tiny Triton kernel.
      5) Batched matrix‑multiplication for the O(N³) contraction
         using torch.bmm on a view (B*H, N, N). No extra permute/copy
         of the contraction operands.
      6) Layer‑norm on the hidden dimension, element‑wise out‑gate,
         final linear projection back to the original channel dimension.
      7) Return FP32 tensor.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # if True, ignore the mask

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) First LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    # weight and bias are float32 tensors
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse all five linear transformations into one matmul
    # ------------------------------------------------------------------
    # Convert model weights to half‑precision once
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # Concatenate the five projection weight matrices (shape 5*hidden_dim × dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*hidden_dim, dim)

    # Linear projection – result shape [B, N, N, 5*hidden_dim] (FP16)
    proj = torch.nn.functional.linear(x, combined_weight)  # fused projection

    # ------------------------------------------------------------------
    # 3) Split and compute gates
    # ------------------------------------------------------------------
    # Split along the last dimension
    split_tensors = torch.split(proj, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = split_tensors

    # Compute sigmoid gates (still half‑precision)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4) Apply mask (if present) and fuse projection * gate
    # ------------------------------------------------------------------
    if nomask:
        # Simple in‑place multiplication when there is no mask
        left = left_proj.mul_(left_gate)
        right = right_proj.mul_(right_gate)
    else:
        # Ensure inputs are contiguous for the Triton kernel
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)

        total_elements = left_proj.numel()  # B*N*N*hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        # Fuse proj * gate * mask for left stream (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse proj * gate * mask for right stream (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After fusion the gate tensors are no longer needed
        left = left_proj
        right = right_proj
        del left_gate, right_gate, left_proj, right_proj

    # ------------------------------------------------------------------
    # 5) Heavy contraction – batched GEMM across the hidden dimension
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    We use torch.bmm after reshaping to (B*H, N, N) without
    #    extra copies of the data.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # view as (B*H, N, N) – this is a view, no copy
    left_mat = left.permute(0, 3, 1, 2).flatten(0, 1)   # shape (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).flatten(0, 1) # shape (B*H, N, N)

    # right needs to be transposed on the last two dimensions for the dot‑product
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm on hidden dimension (still FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7) Out‑gate multiplication and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 8) Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithmic tricks used
-----------------------
1️⃣  First LayerNorm (FP32) → cast to FP16.
2️⃣  Fuse the five separate Linear layers (left/right projections + three gating
    streams) into a *single* matmul by concatenating the weight matrices.
    This reads the 4‑D input only once (big memory‑bandwidth win).
3️⃣  Split the fused projection, apply sigmoids to the gate tensors.
4️⃣  Fuse the element‑wise product (proj * gate * mask) with a tiny Triton
    kernel (`mask_gate_fuse_kernel`).  The mask is broadcast over the hidden
    dimension, hence only one kernel launch per stream.
5️⃣  Heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed with a plain ``torch.einsum`` which operates directly on the
    original layout – no extra permute / copy of the tensors.
6️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final projection back to
    the original channel size.
7️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the forward pass is needed; gradients are not computed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over the
# hidden dimension.  Operates on a flat view of a contiguous tensor of
# shape [B·N·N·hidden_dim] (size = total_elements).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, size = total_elements
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,) – broadcasted
    total_elements,  # i32 : total number of elements in x
    hidden_dim,     # i32 : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask = offs < total_elements                # guard for the tail of the tensor

    # Load tensor and gate
    x   = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Every `hidden_dim` consecutive elements share the same mask entry.
    mask_idx = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → ignore the mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"] # projection width (c)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32
    x = x_norm.half()  # work in FP16 from here on

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five Linear layers into a single matmul
    # ------------------------------------------------------------------
    # Convert all weights to FP16 once
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # Concatenate the five weight matrices (output dim = 5*hidden_dim)
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Linear projection (single GEMM) → [B, N, N, 5*hidden_dim] (FP16)
    proj = torch.nn.functional.linear(x, fused_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Split and compute gates (sigmoid)
    # ------------------------------------------------------------------
    split = torch.split(proj, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = split

    left_gate  = torch.sigmoid(left_gate_raw)   # FP16
    right_gate = torch.sigmoid(right_gate_raw) # FP16
    out_gate   = torch.sigmoid(out_gate_raw)   # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Mask + gating fusion (Triton kernel) – in‑place on the projections
    # ------------------------------------------------------------------
    if nomask:
        # No mask – simple element‑wise multiplication
        left  = left_proj.mul_(left_gate)      # in‑place
        right = right_proj.mul_(right_gate)
    else:
        # Ensure contiguous layout for the kernel
        left_proj  = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B·N·N,)

        total_el = left_proj.numel()               # B·N·N·hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_el, BLOCK),)

        # Fuse proj * gate * mask for the left stream (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_el,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse proj * gate * mask for the right stream (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_el,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left, right = left_proj, right_proj
        del left_gate, right_gate, left_proj, right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with einsum which operates on the original layout
    #    (no extra permutes or copies).
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden_dim) – still in FP16
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate gating + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise (FP16)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
