def mask_gate_fuse_kernel(
    x_ptr,
    gate_ptr,
    mask_ptr,
    hidden_dim,
    size,
    BLOCK: tl.constexpr,

"""
TriMul (outgoing) forward pass – high‑performance Triton implementation.

Algorithm
---------
1. First LayerNorm (FP32) on the 4‑D pairwise tensor.
2. Cast to FP16 and apply two linear projections (left/right) → shape [B,N,N,H].
3. Compute three sigmoid gates (left_gate, right_gate, out_gate).
4. If a mask is present fuse `tensor * gate * mask` in‑place with a tiny Triton
   kernel (`mask_gate_fuse_kernel`).  Otherwise just multiply by the gates.
5. Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed by a custom Triton kernel (`tri_mul_contract_kernel`) that works
   directly on the original layout [B,N,N,H] – avoiding costly
   permute/contiguous copies and the batched cuBLAS call.
6. Second LayerNorm (FP16) over the hidden dimension.
7. Multiply by `out_gate` and project back to the original channel size.
8. Return FP32 result (expected by the harness).

The heavy contraction dominates runtime, so the custom kernel removes the
expensive data layout transformations and uses Tensor‑Core‑friendly tiling.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# 1️⃣ Mask‑gate fusion kernel (used only when a mask is supplied)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated (flattened)
    gate_ptr,       # *fp16  gating tensor (same shape as x)
    mask_ptr,       # *fp16  mask tensor (flattened, shape [B*N*N])
    hidden_dim,     # i32   hidden dimension H
    size,           # i32   total number of fp16 elements (B*N*N*H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Broadcast mask across the hidden dimension
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fuse multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# 2️⃣ O(N³) contraction kernel (left @ rightᵀ per batch‑head)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_contract_kernel(
    left_ptr,          # *fp16  left  tensor  [B,N,N,H]
    right_ptr,         # *fp16  right tensor  [B,N,N,H]
    out_ptr,           # *fp16  output tensor [B,N,N,H]
    B: tl.constexpr,   # batch size
    N: tl.constexpr,   # sequence length
    H: tl.constexpr,   # hidden dimension
    stride_batch,      # stride of the batch dimension (B‑stride)
    stride_i,          # stride of the first N dimension (i‑stride = N*H)
    stride_k,          # stride of the second N dimension (k‑stride = H)
    out_stride_i,      # stride of output i dimension (same as stride_i)
    out_stride_j,      # stride of output j dimension (same as stride_k)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[...,i,k,h] * right[...,j,k,h]"""

    # ------------------------------------------------------------------
    # Program IDs: (i_tile, j_tile, batch*head)
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)               # combined batch & head index

    b = pid_hb // H                         # batch index
    h = pid_hb % H                          # head index

    # ------------------------------------------------------------------
    # Offsets for the tile (i,j) in the output matrix
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)[:, None]   # (M,1)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)[None, :]   # (1,N)

    # ------------------------------------------------------------------
    # Base pointers for this (b,h) slice
    # ------------------------------------------------------------------
    base_left  = left_ptr  + b * stride_batch + h
    base_right = right_ptr + b * stride_batch + h
    base_out   = out_ptr   + b * stride_batch + h

    # ------------------------------------------------------------------
    # Accumulator (float32 for numeric stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k_start = kk * BLOCK_K

        # ---- left tile (M x K) ---------------------------------------
        k_off_a = k_start + tl.arange(0, BLOCK_K)[None, :]            # (1,K)
        a_ptr = base_left + offs_i * stride_i + k_off_a * stride_k
        mask_a = (offs_i < N) & (k_off_a < N)
        a = tl.load(a_ptr, mask=mask_a, other=0.0)

        # ---- right tile transposed (K x N) ----------------------------
        k_off_b = k_start + tl.arange(0, BLOCK_K)[:, None]            # (K,1)
        b_ptr = base_right + offs_j * stride_i + k_off_b * stride_k   # rightᵀ
        mask_b = (k_off_b < N) & (offs_j < N)
        b = tl.load(b_ptr, mask=mask_b, other=0.0)

        # ---- Accumulate dot product -----------------------------------
        acc += tl.dot(a, b)   # (M,N) += (M,K) @ (K,N)

    # ------------------------------------------------------------------
    # Write output tile (cast back to FP16)
    # ------------------------------------------------------------------
    c_ptr = base_out + offs_i * out_stride_i + offs_j * out_stride_j
    mask_c = (offs_i < N) & (offs_j < N)
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_c)

# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]

    """
    Outgoing TriMul forward pass (no autograd).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights, config)

    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First layer‑norm (FP32) → stable for large N
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast all linear weights to FP16 once
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)

    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------

    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )

            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )

        left = left * left_gate
        right = right * right_gate

    B, N, _, H = left.shape
    out = torch.empty_like(left)   # allocate fp16 output tensor

    stride_batch = left.stride(0)   # B‑stride = N*N*H
    stride_i     = left.stride(1)   # i‑stride = N*H
    stride_k     = left.stride(2)   # k‑stride = H

    # output strides (identical layout)
    out_stride_i = out.stride(1)    # i‑stride
    out_stride_j = out.stride(2)    # j‑stride (= k‑stride)

    # Tiling parameters (tuned for H100 FP16 Tensor‑Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (

        triton.cdiv(N, BLOCK_N),      # j‑tiles

    )
    tri_mul_contract_kernel[grid](
        left, right, out,
        B, N, H,
        stride_batch,
        stride_i,
        stride_k,
        out_stride_i,
        out_stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # 8️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(

        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection (FP16)
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] (fp16)

    # 🔟 Cast back to FP32 (expected by the harness) and return
    # -----------------------------------------------------------------
