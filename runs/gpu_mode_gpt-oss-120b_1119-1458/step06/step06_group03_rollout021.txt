def apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr,
    total_elts: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = offs < total_elts

    # Determine mask index: each hidden_dim entries correspond to the same mask position
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate_raw = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)

    hidden_dim: tl.int32,

):

"""
TriMul (outgoing) – fused projection + gating + mask + batched GEMM.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) → cast to float16.
2️⃣ Fuse the five linear transforms (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single matmul using a concatenated weight
   matrix of shape (5·hidden_dim, dim).  This reduces memory traffic by
   loading the input tensor only once.
3️⃣ If a mask is present we fuse (proj * sigmoid(gate) * mask) for the left
   and right branches inside one Triton kernel (`compute_left_right_kernel`).
   The kernel reads the concatenated projection tensor, applies a sigmoid
   to the gate values, multiplies with the projection values and the mask,
   and writes the gated tensors `left` and `right`.
   When no mask is supplied, the gating reduces to a simple
   element‑wise product using torch.
4️⃣ Perform the O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM with `torch.bmm`.  This maps to Tensor‑Core kernels
   on an H100.
5️⃣ Second Layer‑norm (hidden_dim) in fp16.
6️⃣ Apply the already‑computed `out_gate` (sigmoid) and a final linear
   projection (hidden_dim → dim).
7️⃣ Cast the result back to float32 for the harness.

Only the forward pass is implemented; no autograd is required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: compute left/right tensors = proj * sigmoid(gate) * mask
# The concatenated projection tensor has layout (B, N, N, 5*hidden_dim)
# and is stored contiguously in memory.
# ----------------------------------------------------------------------
@triton.jit
def compute_left_right_kernel(
    proj_ptr,            # *fp16 : concatenated tensor of shape (B,N,N,5*H)
    left_ptr,            # *fp16 : output left tensor (B,N,N,H)
    right_ptr,           # *fp16 : output right tensor (B,N,N,H)
    mask_ptr,            # *fp16 : flattened mask of shape (B*N*N), optional (may be all‑ones)
    total_elts: tl.int32,# total number of elements in each of left/right (B*N*N*H)
    hidden_dim: tl.int32,# hidden dimension H
    BLOCK: tl.constexpr, # compile‑time tile size (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices into left/right
    valid = offs < total_elts

    # Position index (b,i,j) and hidden index
    pos = offs // hidden_dim                # integer division → which (b,i,j) entry
    hid = offs % hidden_dim                 # hidden feature index

    stride5h = hidden_dim * 5                # distance (in fp16 elements) between successive (b,i,j) blocks

    base = pos * stride5h                    # start of the 5‑H block for this (b,i,j)

    # Load the four relevant raw values from the concatenated projection tensor
    left_proj = tl.load(proj_ptr + base + hid,                mask=valid, other=0.0)
    left_gate_raw = tl.load(proj_ptr + base + hidden_dim + hid, mask=valid, other=0.0)
    right_proj = tl.load(proj_ptr + base + 2*hidden_dim + hid, mask=valid, other=0.0)
    right_gate_raw = tl.load(proj_ptr + base + 3*hidden_dim + hid, mask=valid, other=0.0)

    # Sigmoid on the gate values
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # Load mask (broadcasted over hidden_dim)
    mask_val = tl.load(mask_ptr + pos, mask=valid, other=1.0)   # mask is 0/1 → default 1 (no mask)

    # Apply gate and mask
    left_val = left_proj * left_gate * mask_val
    right_val = right_proj * right_gate * mask_val

    # Store results
    tl.store(left_ptr + offs, left_val, mask=valid)
    tl.store(right_ptr + offs, right_val, mask=valid)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using
    a batched GEMM (Tensor‑Core).
    """
    B, N, _, H = left.shape
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)       # (B*H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)      # (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                               # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()        # (B, N, N, H)
    return out

    """

    else:

        left_proj = proj[..., :hidden_dim]                     # (B,N,N,H)

        right_proj = proj[..., 2 * hidden_dim:3 * hidden_dim]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 7️⃣ Return FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
