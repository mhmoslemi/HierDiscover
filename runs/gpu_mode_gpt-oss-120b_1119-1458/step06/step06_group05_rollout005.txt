"""
TriMul outgoing forward kernel.

Algorithm:
1. LayerNorm over the last channel dimension (FP32 → FP16).
2. Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single FP16 GEMM, reducing memory traffic.
3. Sigmoid activation for the gate tensors.
4. Fuse mask multiplication (if provided) with gate scaling using a small Triton kernel.
5. Perform the O(N³) contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   as a batched matrix multiplication (left @ rightᵀ) via cuBLAS.
6. Apply second LayerNorm, out‑gate scaling, and final projection.
7. Return FP32 tensor.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------
# Triton kernel: proj * gate * mask (mask broadcast)
# -------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,         # *fp16 [B, N, N, H]
    gate_ptr,         # *fp16 [B, N, N, H]
    mask_ptr,         # *fp16 [B, N, N]
    out_ptr,          # *fp16 [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out = proj * gate * mask, broadcasting mask over H.
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # Compute offsets for I, J dimensions.
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load mask tile.
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)   # [BLOCK_I, BLOCK_J]

    # Loop over H tiles.
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load proj tile.
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Load gate tile.
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fuse multiplication.
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store.
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(out_ptrs, out_tile,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse proj, gate and mask (broadcast over hidden dimension) using Triton.
    proj, gate: [B, N, N, H]   (fp16)
    mask:       [B, N, N]      (fp16)
    Returns:    [B, N, N, H]   (fp16)
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides in element count (not bytes).
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes. Tuned for H100, can be overridden.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8  # works for hidden_dim up to 384

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right with a summed inner dimension:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batched GEMM (A @ Bᵀ) on FP16 tensors using cuBLAS.
    left/right: [B, N, N, H] (fp16, contiguous)
    returns:    [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul primitive (no gradients).

    Args:
        data: tuple containing (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor:  [B, N, N] (float32/bool). Ignored if config['nomask'] is True.
            - weights_dict: mapping name → torch.Tensor (float32)
            - config_dict: must contain 'dim' and 'hidden_dim'; may contain 'nomask' (bool).

    Returns:
        Tensor of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # -------------------------------------------------
    # Unpack.
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.nn.functional.layer_norm works in the dtype of the input, we give FP32 input.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)

    # -------------------------------------------------
    # 2️⃣ Fuse all five linear projections (left, right, left_gate, right_gate, out_gate)
    # -------------------------------------------------
    # Gather weights (biases are all False for these layers).
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Stack along output dimension: shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate], dim=0
    ).to(device=device, dtype=dtype_fp16)

    # Combined GEMM: shape (B, N, N, 5*hidden_dim)
    combined_out = torch.nn.functional.linear(x, combined_weight)  # bias omitted

    # Split back.
    H = hidden_dim
    left_proj = combined_out[..., 0:H]                # [B,N,N,H]
    right_proj = combined_out[..., H:2*H]             # [B,N,N,H]
    left_gate_logit = combined_out[..., 2*H:3*H]      # [B,N,N,H]
    right_gate_logit = combined_out[..., 3*H:4*H]     # [B,N,N,H]
    out_gate_logit = combined_out[..., 4*H:5*H]       # [B,N,N,H]

    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # -------------------------------------------------
    # 3️⃣ Mask + gate fusion (if mask present)
    # -------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and broadcastable.
        mask_fp16 = mask_tensor.to(dtype=dtype_fp16)
        # Use Triton kernel to fuse proj * gate * mask.
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # Simple elementwise multiplication (proj * gate).
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM.
    # -------------------------------------------------
    # left/right already FP16.
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) + out‑gate.
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------------------------------------
    # 6️⃣ Final projection back to model dimension.
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)  # bias omitted

"""
TriMul (outgoing) forward kernel – highly‑optimized implementation.

Algorithm
---------
1️⃣ First LayerNorm (FP32 → FP16) on the input.
2️⃣ Fuse the five linear transformations (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a *single* FP16 GEMM.
3️⃣ Apply sigmoid to the three gate tensors.
4️⃣ If a mask is present, fuse mask, projection and gate with a small
   Triton elementwise kernel (`_mask_gate_kernel`).  Otherwise just
   multiply projection × gate.
5️⃣ Perform the expensive O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM (A @ Bᵀ) using cuBLAS tensor‑core kernels.
6️⃣ Second LayerNorm (FP16) → multiply by the out‑gate → final linear
   projection back to the model dimension.
7️⃣ Cast the result back to float32 (the harness expects FP32).

Only the mask‑gate fusion is performed in Triton; all matmul‑heavy work
leverages cuBLAS which is optimal on an H100.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask   (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H]   (already sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out = proj * gate * mask   (mask broadcast over H).
    """
    pid_b = tl.program_id(0)    # batch index
    pid_i = tl.program_id(1)    # i‑tile
    pid_j = tl.program_id(2)    # j‑tile

    # I/J coordinates for the tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask (BLOCK_I × BLOCK_J)
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16, shape [I,J,H]

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Fuse multiplication: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse proj, gate and mask (broadcast over hidden dim) using Triton.

    Parameters
    ----------
    proj : torch.Tensor
        [B, N, N, H]   (fp16)
    gate : torch.Tensor
        [B, N, N, H]   (fp16) – already passed through sigmoid
    mask : torch.Tensor
        [B, N, N]      (fp16)

    Returns
    -------
    torch.Tensor
        [B, N, N, H]   (fp16) = proj * gate * mask
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides expressed in element counts (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tuned tile sizes (fit well on H100)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8   # works for H up to 384

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batched GEMM (A @ Bᵀ) on FP16 tensors using cuBLAS.
    left, right : [B, N, N, H] (fp16, contiguous)
    returns      : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # C = A @ Bᵀ ; both A and B are (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # restore layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 -> FP16)
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(fp16)

    # -------------------------------------------------
    # 2️⃣ Fuse the five linear maps
    # -------------------------------------------------
    # Weights (all bias=False in the reference implementation)
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # Stack → shape (5*hidden_dim, dim)
    combined_w = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate], dim=0
    ).to(device=device, dtype=fp16)

    # One GEMM: (B,N,N,dim) @ (5*hidden_dim, dim).T  → (B,N,N,5*hidden_dim)
    combined = torch.nn.functional.linear(x, combined_w)  # no bias

    H = hidden_dim
    # Split back into the original heads
    left_proj   = combined[..., 0:H]
    right_proj  = combined[..., H:2*H]
    left_gate_l = combined[..., 2*H:3*H]
    right_gate_l= combined[..., 3*H:4*H]
    out_gate_l  = combined[..., 4*H:5*H]

    # Apply sigmoid to the gate logits
    left_gate  = torch.sigmoid(left_gate_l)
    right_gate = torch.sigmoid(right_gate_l)
    out_gate   = torch.sigmoid(out_gate_l)

    # -------------------------------------------------
    # 3️⃣ Mask & gate fusion (if required)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype=fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 4️⃣ Heavy contraction via batched GEMM
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) + out‑gate
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------------------------------------
    # 6️⃣ Final linear projection back to model dim
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = torch.nn.functional.linear(out, to_out_w)  # bias=False

    # -------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
