def batched_matmul_fp16(left, right):
    B,N,_,H = left.shape
    LEFT = left.permute(0,3,1,2).reshape(B*H, N, N)
    RIGHT = right.permute(0,3,1,2).reshape(B*H,N,N)
    out = torch.bmm(LEFT, RIGHT.transpose(1,2))
    out = out.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

"""
Optimized outgoing TriMul forward pass (AlphaFold‑3) using half‑precision and
fused Triton kernels.

Algorithm
---------
1️⃣  Layer‑Norm on the raw input (float32) → stabilises statistics.
2️⃣  Cast to fp16 for the compute‑intensive part.
3️⃣  Fuse **all** per‑location linear projections (left/right projections + their
    gates + the output gate) into a **single** matrix‑multiply:
        X @ W_combined.T   where W_combined stacks the five weight matrices.
4️⃣  Split the combined result and apply sigmoid to the three gate tensors.
5️⃣  If a mask is present, fuse mask, gate and projection multiplication via a
    Triton kernel `mask_gate_fuse_kernel` (in‑place on `left` and `right`).
6️⃣  Heavy O(N³) contraction is realised as a batched GEMM (torch.bmm) after
    reshaping to [B·H, N, N].
7️⃣  Second Layer‑Norm over the hidden dimension (fp16).
8️⃣  Multiply by the output gate.
9️⃣  Final linear projection back to the original dimension.
🔟  Cast the result back to float32 (required by the harness).

Only the forward pass is implemented – no autograd.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuses mask, gating and projection multiplication.
# The mask is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : flattened mask of shape (B·N·N)
    hidden_dim,     # i32   : size of hidden dimension H
    total_size,     # i32   : total number of elements in x
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices for this block
    mask = offs < total_size                           # bounds check

    # map linear offset to mask offset (mask does not have hidden dim)
    mask_offs = offs // hidden_dim

    # load tensors (out‑of‑bounds loads return a safe default)
    x_val   = tl.load(x_ptr   + offs, mask=mask, other=0.0)
    g_val   = tl.load(gate_ptr+ offs, mask=mask, other=0.0)
    m_val   = tl.load(mask_ptr+ mask_offs, mask=mask, other=1.0)   # default mask = 1

    # fused operation: x = x * gate * mask
    tl.store(x_ptr + offs, x_val * g_val * m_val, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the O(N³) contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    Both ``left`` and ``right`` have shape [B, N, N, H] in fp16.
    Returns a tensor of the same shape.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))  # shape [B*H, N, N]

    # restore original layout [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    left_gate_w  = left_gate_w.to(device=device, dtype=torch.float16)
    right_gate_w = right_gate_w.to(device=device, dtype=torch.float16)
    out_gate_w   = out_gate_w.to(device=device, dtype=torch.float16)

    # Stack the five matrices: order is important for later slicing
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape [(5*H), dim]

    # Single linear projection to obtain all intermediate tensors
    # Result shape: [B, N, N, 5*H]
    combined = torch.nn.functional.linear(x, combined_weight)  # fp16

    # -----------------------------------------------------------------
    # 4️⃣ Split and apply sigmoids to gate tensors
    # -----------------------------------------------------------------
    H = hidden_dim
    left_proj   = combined[..., 0:H].contiguous()
    left_gate_r = combined[..., H:2 * H].contiguous()
    right_proj  = combined[..., 2 * H:3 * H].contiguous()
    right_gate_r= combined[..., 3 * H:4 * H].contiguous()
    out_gate_r  = combined[..., 4 * H:5 * H].contiguous()

    left_gate   = torch.sigmoid(left_gate_r)   # [B, N, N, H] fp16
    right_gate  = torch.sigmoid(right_gate_r) # [B, N, N, H] fp16
    out_gate    = torch.sigmoid(out_gate_r)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fused multiplication (left/right * gate * mask)
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and flatten for the kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # shape (B·N·N,)

        total_elems = left_proj.numel()            # B·N·N·H
        BLOCK = 1024                                # ≤ max threads per block
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # Fuse left side
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right side
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj   # now contains left_proj * left_gate * mask
        right = right_proj # now contains right_proj * right_gate * mask
    else:
        # No mask – simple element‑wise multiplication (in‑place to keep memory low)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑Core)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # -----------------------------------------------------------------
    # 8️⃣ Apply the output gate
    # -----------------------------------------------------------------
    out = out * out_gate   # elementwise fp16 multiplication

    # -----------------------------------------------------------------
    # 9️⃣ Final projection back to the original dimension (fp16)
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
