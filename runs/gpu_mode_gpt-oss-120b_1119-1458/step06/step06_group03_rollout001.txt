"""
TriMul (outgoing) – fast fused forward pass.

Algorithm
---------
1️⃣  First LayerNorm (FP32) → cast to FP16.
2️⃣  Fuse the five linear layers (left_proj, left_gate, right_proj, right_gate,
    out_gate) into one large GEMM.  Split the result and apply sigmoid to the
    three gate slices.
3️⃣  If a mask is present, fuse the three multiplicative factors
    (proj * gate * mask) in‑place with a tiny Triton kernel.
    (When `nomask=True` we simply multiply proj and gate.)
4️⃣  Heavy O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is performed as a batched GEMM using `torch.bmm`.  We reshape the tensors
    to shape (B·H, N, N) and multiply L @ Lᵀ for each hidden channel H.
5️⃣  Second LayerNorm (FP16) on the contracted tensor.
6️⃣  Multiply by the pre‑computed `out_gate` (sigmoid) and apply the final
    linear projection back to `dim`.
7️⃣  Cast the result to FP32 (as required by the harness).

All heavy linear work is kept in FP16 to exploit Tensor Cores, while the mask
fusion is done in a custom Triton kernel to avoid extra memory passes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def fused_gate_mask_kernel(
    x_ptr,        # *fp16 : tensor to be updated (proj)
    gate_ptr,     # *fp16 : gate tensor (sigmoid output)
    mask_ptr,     # *fp16 : mask tensor (broadcast over hidden_dim)
    total_elts: tl.int32,  # total number of elements in x (B·N·N·H)
    hidden_dim: tl.int32,  # hidden dimension H
    BLOCK: tl.constexpr,
):
    """In‑place fusion:  x = x * gate * mask   (mask broadcast on hidden dim)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)    # linear offsets
    mask = offs < total_elts

    # mask index: each block of `hidden_dim` elements corresponds to a single
    # (b,i,k) entry of the original 4‑D tensor.
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default 1 → no mask

    tl.store(x_ptr + offs, x * g * m, mask=mask)


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] via batched GEMM.
    left/right are [B, N, N, H] float16 tensors.
    Returns [B, N, N, H] float16.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where each hidden channel is a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # matrix multiplication L @ Rᵀ
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – more stable
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy computation
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)  # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj & gates) into a single GEMM.
    #    We stack the corresponding weight matrices along the output
    #    dimension (rows).  All linear layers in the reference model have
    #    bias=False, so we can simply use torch.nn.functional.linear.
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------

    right_proj = fused[..., 2 * hs:3 * hs]
    right_gate = torch.sigmoid(fused[..., 3 * hs:4 * hs])
    out_gate = torch.sigmoid(fused[..., 4 * hs:5 * hs])

    # free the large fused tensor early
    del fused

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask (if present) and fuse proj * gate; otherwise simple mul.
    #    Using a tiny Triton kernel for the masked case reduces the memory
    #    traffic of three separate element‑wise passes.
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dimension → convert once to FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()          # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel calls `left_proj`/`right_proj` already contain the
        # final values, we can reuse the tensors directly.
        left = left_proj
        right = right_proj
    else:
        # No mask – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Clean up intermediate tensors that are no longer needed.
    del left_proj, left_gate, right_proj, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # [B, N, N, hidden_dim] FP16

    # Release memory that is not needed any longer.
    del left, right

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by out_gate (computed earlier) and final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate  # element‑wise gating, FP16
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    final = torch.nn.functional.linear(gated, to_out_w)    # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return final.to(torch.float32)
