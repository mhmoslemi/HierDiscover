"""

TriMul kernel (outgoing).

- Performs first LayerNorm in FP32 then cast to FP16.
- Fuses five linear transforms (two projections, three gating) into one FP16 GEMM.
- Applies sigmoid to gating outputs.
- If a mask is present, fuses mask, projection and gate multiplication with a Triton kernel.
- Computes the core O(N^3) contraction as a strided‑batched GEMM (left @ rightᵀ) in FP16.
- Second LayerNorm on the hidden dimension, multiplies by the previously computed out‑gate,
  and a final linear projection back to the original channel dimension.
- Returns FP32 result.

All heavy linear algebra uses FP16 and Tensor‑Core kernels; mask‑fusion uses a Triton kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel that multiplies a tensor by a gate and a (broadcasted) mask.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,        # *fp16 : tensor to be updated (size = total_elts)
    gate_ptr,     # *fp16 : gate tensor (same size as x)
    mask_ptr,     # *fp16 : mask tensor (size = total_elts // hidden_dim)
    total_elts: tl.int32,   # total number of elements in x/gate
    hidden_dim: tl.int32,   # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,    # tile size, must be ≤ 1024
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # mask is broadcast across the hidden dimension.
    mask_idx = offs // hidden_dim

    x_val   = tl.load(x_ptr + offs,   mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out_val = x_val * gate_val * mask_val
    tl.store(x_ptr + offs, out_val, mask=mask)


# ----------------------------------------------------------------------
# Strided‑batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction using a strided‑batched bmm.
    Expected shapes: left, right -> [B, N, N, H] (FP16)
    Returns: out -> [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel becomes one batch entry
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Public entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]      (float32/bool) – optional
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerics) and cast to FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # LayerNorm over the last dimension (dim)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32

    x = x_norm.to(torch.float16)               # [B, N, N, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear transforms (proj + 3 gates) into a single GEMM
    # --------------------------------------------------------------
    # Gather the five weight matrices (all have shape [hidden_dim, dim])
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature dimension → shape (5*hidden_dim, dim)
    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # FP16

    # Single linear projection producing all five intermediate tensors at once
    combined = torch.nn.functional.linear(x, combined_w)   # [B,N,N,5*hidden_dim] (FP16)

    # Split into the five components
    splits = torch.split(combined, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = splits

    # Apply sigmoid to gating tensors
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)   # shape [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and element‑wise gating.
    #    Fuse (proj * gate * mask) using a Triton kernel to avoid temporaries.
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()
        BLOCK = 1024
        grid = ( (total_elts + BLOCK - 1) // BLOCK, )
        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel the tensors contain the gated, masked values.
        left = left_proj
        right = right_proj
    else:
        # No mask → straightforward element‑wise multiply
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Core O(N³) contraction via a strided‑batched GEMM
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (hidden_dim) – stays in FP16
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 6️⃣ Multiply by out‑gate (computed earlier) and final linear projection
    # --------------------------------------------------------------
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim] FP16

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast fused forward pass.

Algorithm
---------
1️⃣ First LayerNorm (FP32 → FP16) stabilises the input.
2️⃣ Five linear transformations (left/right projections + three gates) are
   performed with FP16 matrix‑multiplications.  The three gate tensors are
   passed through sigmoid.
3️⃣ If a mask is supplied we fuse the element‑wise triple product
   (proj * gate * mask) with a tiny Triton kernel, saving two separate
   element‑wise kernels.
4️⃣ The heavy O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is executed as a strided‑batched GEMM (torch.bmm) which maps to a
   Tensor‑Core kernel on H100.
5️⃣ A second LayerNorm (still FP16) normalises the hidden dimension,
   the previously computed out‑gate (sigmoid) is applied, and a final
   linear projection maps back to the original feature dimension.
6️⃣ The result is cast back to FP32 for the harness.

All large GEMMs run in half‑precision; only the final output is FP32.
The mask‑fusion kernel guarantees that at least part of the workload
is executed in Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuses (tensor * gate * mask) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,          # *fp16 : tensor to be updated (size = total_elts)
    gate_ptr,       # *fp16 : gate tensor (same size as x)
    mask_ptr,       # *fp16 : mask tensor (size = total_elts // hidden_dim)
    total_elts: tl.int32,   # total number of elements in x/gate
    hidden_dim: tl.int32,   # hidden dimension (size of the last axis)
    BLOCK: tl.constexpr,    # tile size (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # Mask is broadcast across the hidden dimension.
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,   mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # 1 → no mask

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    Shapes:
        left , right : [B, N, N, H] (FP16)
    Returns:
        out : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # Turn each hidden slice into a separate batch entry → (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                     # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : map of weight name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (True ⇢ no mask).

    Returns
    -------
    torch.Tensor[B, N, N, dim]  (float32, same device as ``input_tensor``)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)                         # [B, N, N, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Five separate linear transforms (proj + gates)
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate_pre = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16))

    left_gate = torch.sigmoid(left_gate_pre)    # [B,N,N,hidden_dim] FP16
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)      # used after the second LN

    # --------------------------------------------------------------
    # 3️⃣ Mask‑fusion (if required)
    # --------------------------------------------------------------
    if not nomask:
        # broadcast mask to hidden_dim, keep it in FP16 for the kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Core O(N³) contraction via a strided‑batched GEMM
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] (FP16)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (hidden_dim) – stays in FP16
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply out‑gate and final projection back to ``dim``
    # --------------------------------------------------------------
    out = out * out_gate                         # element‑wise gating
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),  # shape [dim, hidden_dim]
    )                                            # [B,N,N,dim] FP16

    # --------------------------------------------------------------
    # 7️⃣ Return FP32 result (as required by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
