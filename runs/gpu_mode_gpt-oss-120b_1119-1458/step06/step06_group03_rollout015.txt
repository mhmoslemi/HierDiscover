```
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def fused_proj_gate_mask_kernel(
    y_ptr,            # *fp16: pointer to concatenated Y tensor
    out_ptr,          # *fp16: output pointer for left or right projection
    mask_ptr,         # *fp16: mask pointer (B,N,N) flattened
    total_elts: tl.int32,
    hidden_dim: tl.int32,
    total_dim_per_pos: tl.int32,  # =5*hidden_dim
    proj_start: tl.int32,         # offset within per-position block for proj raw
    gate_start: tl.int32,         # offset within per-position block for gate raw
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # Compute position and hidden indices
    pos = offs // hidden_dim      # integer division: index of (b,i,k)
    h = offs % hidden_dim
    # Compute absolute address offsets
    proj_off = proj_start + pos * total_dim_per_pos + h
    gate_off = gate_start + pos * total_dim_per_pos + h
    mask_off = pos  # mask flattened index

    proj = tl.load(y_ptr + proj_off, mask=mask, other=0.0)
    gate_raw = tl.load(y_ptr + gate_off, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)

    gate = 1.0 / (1.0 + tl.exp(-gate_raw))
    out = proj * gate * mask_val

    tl.store(out_ptr + offs, out, mask=mask)
```

```
@triton.jit
def fused_proj_gate_mask_kernel(
    y_ptr,            # *fp16 pointer to concat Y
    out_ptr,          # *fp16 output pointer (left or right)
    mask_ptr,         # *fp16 mask pointer (flat)
    total_elts: tl.int32,
    hidden_dim: tl.int32,
    total_dim_per_pos: tl.int32,
    proj_start: tl.int32,
    gate_start: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # compute positions and channel
    pos = offs // hidden_dim
    ch = offs - pos * hidden_dim   # or offs % hidden_dim using remainder - more efficient?

    # compute absolute offsets
    proj_offset = proj_start + pos * total_dim_per_pos + ch
    gate_offset = gate_start + pos * total_dim_per_pos + ch
    mask_offset = pos

    proj_val = tl.load(y_ptr + proj_offset, mask=mask, other=tl.float16(0.0))
    gate_raw_val = tl.load(y_ptr + gate_offset, mask=mask, other=tl.float16(0.0))
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=tl.float16(1.0))

    gate_val = 1.0 / (1.0 + tl.exp(-gate_raw_val))
    out_val = proj_val * gate_val * mask_val

    tl.store(out_ptr + offs, out_val, mask=mask)
```

"""
TriMul (outgoing) – fast fused‑gate + mask kernel + single‑matmul concat.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 (stable) and cast to FP16.
2️⃣  Concatenate the five weight matrices (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single `(5*H, D)` matrix
    and perform one large FP16 linear projection:
        Y = Linear(x, W_concat)  →  shape [B,N,N,5*H].
3️⃣  Slice Y into the raw projections, gate‑pre‑sigmoid tensors and the
    output‑gate tensor.
4️⃣  If a mask is present we fuse the element‑wise operations
    `proj * sigmoid(gate) * mask` for *left* and *right* in a Triton
    kernel (the sigmoid is computed on‑device).  If there is no mask we
    simply use PyTorch element‑wise ops.
5️⃣  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM with `torch.bmm` (each hidden channel becomes a
    separate batch element).
6️⃣  Apply the second LayerNorm (FP16), multiply by the pre‑computed
    `out_gate` and a final FP16 linear projection back to `dim`.
7️⃣  Cast the result back to FP32 for the harness.

Only the forward pass is required; no gradients are stored.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: proj * sigmoid(gate) * mask (in‑place)
#   - `y_ptr` points to the concatenated Y = [proj, proj, gate, gate, gate]
#   - `out_ptr` is the destination for the gated projection (left or right)
#   - mask is a flat FP16 tensor of length B*N*N, broadcast over H
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_mask_kernel(
    y_ptr,               # *fp16 : concatenated tensor (B,N,N,5*H)
    out_ptr,             # *fp16 : output tensor (B,N,N,H)
    mask_ptr,            # *fp16 : mask tensor flattened (B*N*N)
    total_elts: tl.int32,    # total elements in the output = B*N*N*H
    hidden_dim: tl.int32,    # H
    total_dim_per_pos: tl.int32,  # 5*H
    proj_start: tl.int32,    # offset of projection slice within a block of 5*H
    gate_start: tl.int32,    # offset of gate slice within a block of 5*H
    BLOCK: tl.constexpr,     # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_elts

    # Position index (flattened over B,N,N) and inner hidden index
    pos = offs // hidden_dim
    h_idx = offs - pos * hidden_dim   # equivalent to offs % hidden_dim

    # Offsets into the concatenated Y tensor
    pos_stride = total_dim_per_pos
    proj_off = proj_start + pos * pos_stride + h_idx
    gate_off = gate_start + pos * pos_stride + h_idx

    # Load raw projection, raw gate and mask
    proj_val = tl.load(y_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    gate_raw = tl.load(y_ptr + gate_off, mask=mask, other=tl.float16(0.0))
    mask_val = tl.load(mask_ptr + pos, mask=mask, other=tl.float16(1.0))

    # Sigmoid in fp16
    gate_val = 1.0 / (1.0 + tl.exp(-gate_raw))

    # Final gated value
    out_val = proj_val * gate_val * mask_val
    tl.store(out_ptr + offs, out_val, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) TriMul contraction
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H] (fp16)
    Returns:
        out : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)ᵀ
    c = torch.bmm(a, b.transpose(1, 2))
    # Restore shape (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_pos = B * N * N
    total_dim_per_pos = 5 * hidden_dim                     # 5 tensors are concatenated

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the computation
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Concatenate all linear weights (no bias) and cast to FP16
    # ------------------------------------------------------------------
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*H, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Single fused Linear → Y shape: [B,N,N,5*H]
    # ------------------------------------------------------------------
    y = torch.nn.functional.linear(x, concat_weight)   # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Split and apply gates / mask
    # ------------------------------------------------------------------
    if not nomask:
        # ----- mask present -----
        # Flatten mask to 1‑D and force FP16 (0.0 / 1.0)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # length B*N*N

        # Allocate destination tensors for the gated projections
        left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
        right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

        # Compute total number of output elements for each of left/right
        total_elts = total_pos * hidden_dim
        BLOCK = 1024
        grid = ( (total_elts + BLOCK - 1) // BLOCK, )

        y_ptr = y.data_ptr()
        mask_ptr = mask_flat.data_ptr()

        # Left: proj slice offset 0, gate slice offset 2*H
        fused_proj_gate_mask_kernel[grid](
            y_ptr,
            left.data_ptr(),
            mask_ptr,
            total_elts,
            hidden_dim,
            total_dim_per_pos,
            0,                         # proj_start
            2 * hidden_dim,            # gate_start
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Right: proj slice offset H, gate slice offset 3*H
        fused_proj_gate_mask_kernel[grid](
            y_ptr,
            right.data_ptr(),
            mask_ptr,
            total_elts,
            hidden_dim,
            total_dim_per_pos,
            hidden_dim,                # proj_start
            3 * hidden_dim,            # gate_start
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Output‑gate (no mask, only sigmoid)
        out_gate_raw = y[..., 4 * hidden_dim : 5 * hidden_dim]
        out_gate = torch.sigmoid(out_gate_raw)  # FP16
    else:
        # ----- no mask case -----
        # Simple slicing + sigmoid + element‑wise product
        left_raw = y[..., :hidden_dim]
        right_raw = y[..., hidden_dim : 2 * hidden_dim]
        left_gate_raw = y[..., 2 * hidden_dim : 3 * hidden_dim]
        right_gate_raw = y[..., 3 * hidden_dim : 4 * hidden_dim]
        out_gate_raw = y[..., 4 * hidden_dim : 5 * hidden_dim]

        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        out_gate = torch.sigmoid(out_gate_raw)

        left = left_raw * left_gate
        right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    out = torch.nn.functional.linear(out, to_out_weight)       # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Return FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
