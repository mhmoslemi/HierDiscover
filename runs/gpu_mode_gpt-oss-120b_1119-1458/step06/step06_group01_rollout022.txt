    B, N, H,

    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr

   pid_i = tl.program_id(1)
   pid_j = tl.program_id(2)

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    B, N, _, _ = input_tensor.shape

    # 1. Layer norm

    x = x.to(torch.float16)

    right_proj_weight = weights["right_proj.weight"]

```
@triton.jit
def _fused_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, C,
    stride_proj_b, stride_proj_c, stride_proj_h, stride_proj_w,
    stride_gate_b, stride_gate_c, stride_gate_h, stride_gate_w,
    stride_mask_b, stride_mask_h, stride_mask_w,
    stride_out_b, stride_out_c, stride_out_h, stride_out_w,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load mask tile (BLOCK_I x BLOCK_J)
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_h + offs_j[None, :] * stride_mask_w
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

    num_c_tiles = tl.cdiv(C, BLOCK_H)
    for c_tile in range(num_c_tiles):
        offs_c = c_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_c = offs_c < C
        # load proj
        proj_ptrs = (proj_ptr
                     + pid_b * stride_proj_b
                     + offs_c[None, None, :] * stride_proj_c
                     + offs_i[:, None, None] * stride_proj_h
                     + offs_j[None, :, None] * stride_proj_w)
        proj_tile = tl.load(proj_ptrs,
                            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :],
                            other=0.0)
        # load gate raw
        gate_ptrs = (gate_ptr + pid_b * stride_gate_b
                     + offs_c[None, None, :] * stride_gate_c
                     + offs_i[:, None, None] * stride_gate_h
                     + offs_j[None, :, None] * stride_gate_w)
        gate_raw_tile = tl.load(gate_ptrs,
                                mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :],
                                other=0.0)
        # sigmoid
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_raw_tile))

        # broadcast mask over hidden
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # store out
        out_ptrs = (out_ptr + pid_b * stride_out_b
                    + offs_c[None, None, :] * stride_out_c
                    + offs_i[:, None, None] * stride_out_h
                    + offs_j[None, :, None] * stride_out_w)
        tl.store(out_ptrs,
                 out_tile,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :])
```

But we need to be careful about ordering of dimensions: In the pointer arithmetic for proj, gate, out we treat dimension ordering: (batch, channel, height, width). Our strides are `stride_proj_b`, `stride_proj_c`, `stride_proj_h`, `stride_proj_w`. That matches the shape dimension order.

Now we need to pass these strides in function arguments in proper order.

Now correct indexing for offsets: Offsets for height dimension (i) are multiplied by stride_proj_h, offset for width (j) multiplied by stride_proj_w. Offsets for channel (c) multiplied by stride_proj_c.

def custom_kernel(data):

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    right_proj_w = weights["right_proj.weight"]

    out_gate_w = weights["out_gate.weight"]

    combined_w = torch.cat([left_proj_w, right_proj_w,
                     left_gate_w, right_gate_w,
                     out_gate_w], dim=0) # shape (5*hidden_dim, dim)

    s_proj = proj.stride()

"""
TriMul (outgoing) – optimized forward pass.

Algorithm
---------
1️⃣ First LayerNorm (FP32) → cast to FP16.
2️⃣ Project the 4‑D tensor and compute all three gates in a single 1×1
   convolution (conv2d) by concatenating the five weight matrices.
   This reads the input only once (no repeated linear layers).
3️⃣ If a mask is present, fuse the gate sigmoid and mask multiplication
   in a tiny Triton kernel (proj * sigmoid(gate) * mask).  Without a mask
   we perform the simple element‑wise product in PyTorch.
4️⃣ The O(N³) contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched GEMM (C = A @ Bᵀ) with cuBLAS/Tensor‑Core
   kernels – we flatten B and hidden dimensions to obtain a batch of
   B·hidden_dim matrices of size N×N.
5️⃣ Second LayerNorm (FP16) → out‑gate sigmoid → gating.
6️⃣ Final 1×1 convolution maps back to the model dimension.
7️⃣ Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate‑sigmoid fusion is written in Triton; the heavy
contraction stays on cuBLAS which on an H100 is far faster than a pure
Triton implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, sigmoid gate and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _fuse_gate_mask_kernel(
    proj_ptr,            # *fp16  [B, C, N, N]   (projection result)
    gate_ptr,            # *fp16  [B, C, N, N]   (raw gate pre‑sigmoid)
    mask_ptr,            # *fp16  [B, N, N]      (mask broadcast over channels)
    out_ptr,             # *fp16  [B, C, N, N]   (output after gate+mask)
    B, N, C,
    stride_proj_b, stride_proj_c, stride_proj_h, stride_proj_w,
    stride_gate_b, stride_gate_c, stride_gate_h, stride_gate_w,
    stride_mask_b, stride_mask_h, stride_mask_w,
    stride_out_b, stride_out_c, stride_out_h, stride_out_w,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * sigmoid(gate) * mask        (mask is broadcast over C)
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile (height)
    pid_j = tl.program_id(2)   # j‑tile (width)

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over C
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_h
        + offs_j[None, :] * stride_mask_w
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Loop over channel tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(C, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_c = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_c = offs_c < C

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_c[None, None, :] * stride_proj_c
            + offs_i[:, None, None] * stride_proj_h
            + offs_j[None, :, None] * stride_proj_w
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate‑raw tile
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_c[None, None, :] * stride_gate_c
            + offs_i[:, None, None] * stride_gate_h
            + offs_j[None, :, None] * stride_gate_w
        )
        gate_raw_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :],
            other=0.0,
        )   # fp16

        # Sigmoid (gate = 1/(1+exp(-x)))
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_raw_tile))

        # Fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_c[None, None, :] * stride_out_c
            + offs_i[:, None, None] * stride_out_h
            + offs_j[None, :, None] * stride_out_w
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :],
        )


def _apply_fused_gate_mask(
    proj: torch.Tensor, gate_raw: torch.Tensor, mask: torch.Tensor
) -> torch.Tensor:
    """
    Fuse projection, sigmoid‑gate and mask (mask broadcast over channels).

    Arguments
    ---------
    proj      – [B, C, N, N]  FP16 projection result.
    gate_raw  – [B, C, N, N]  pre‑sigmoid gate.
    mask      – [B, N, N]    FP16 mask (0/1) – broadcast over C.

    Returns
    -------
    out – [B, C, N, N] FP16, equal to proj * sigmoid(gate_raw) * mask.
    """
    B, C, N, _ = proj.shape
    out = torch.empty_like(proj)

    # strides expressed in element counts
    s_proj = proj.stride()
    s_gate = gate_raw.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes – tuned for H100 occupancy
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _fuse_gate_mask_kernel[grid](
        proj,
        gate_raw,
        mask,
        out,
        B,
        N,
        C,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Heavy O(N³) contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    Implemented as a batched GEMM (C = A @ Bᵀ) with cuBLAS Tensor‑Core kernels.
    """
    B, C, N, _ = left.shape          # left/right are [B, C, N, N] (FP16)
    # Fold batch and channel dimensions → [B·C, N, N]
    left_flat = left.view(B * C, N, N)
    right_flat = right.view(B * C, N, N)
    # Batched matmul with right transposed on the last two dims
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore layout → [B, C, N, N]
    out = out_flat.view(B, C, N, N)
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack and config
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True ⇒ no mask applied

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32
    x = x.to(torch.float16)   # FP16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Rearrange to NCHW for 1×1 conv2d (no data copy)
    # ------------------------------------------------------------------
    x_nchw = x.permute(0, 3, 1, 2)   # [B, dim, N, N]  (view only)

    # ------------------------------------------------------------------
    # 3️⃣ Concatenate projection & gate weights, then apply a single conv2d
    # ------------------------------------------------------------------
    # Gather the five weight matrices (shape: hidden_dim × dim)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate -> (5*hidden_dim, dim)
    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device=device, dtype=torch.float16)      # FP16
    combined_w = combined_w.view(5 * hidden_dim, dim, 1, 1)   # 1×1 conv kernel

    # One‑shot 1×1 convolution
    proj = torch.nn.functional.conv2d(x_nchw, combined_w)   # [B, 5*hidden_dim, N, N]

    # ------------------------------------------------------------------
    # 4️⃣ Split into raw tensors
    # ------------------------------------------------------------------
    left_raw      = proj[:, 0:hidden_dim, :, :]                # [B, H, N, N]
    right_raw     = proj[:, hidden_dim:2*hidden_dim, :, :]     # [B, H, N, N]
    left_gate_raw = proj[:, 2*hidden_dim:3*hidden_dim, :, :]   # [B, H, N, N]
    right_gate_raw= proj[:, 3*hidden_dim:4*hidden_dim, :, :]   # [B, H, N, N]
    out_gate_raw  = proj[:, 4*hidden_dim:, :, :]              # [B, H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates (and mask if present)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)   # [B, N, N]
        # fused gate + mask for left and right
        left  = _apply_fused_gate_mask(left_raw,  left_gate_raw,  mask_fp16)
        right = _apply_fused_gate_mask(right_raw, right_gate_raw, mask_fp16)
    else:
        # No mask → simple sigmoid + element‑wise multiply (in‑place)
        left  = left_raw  * torch.sigmoid(left_gate_raw)
        right = right_raw * torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # left/right are [B, H, N, N] FP16 and already contiguous.
    out = _batched_matmul_transpose_fp16(left, right)   # [B, H, N, N] FP16

    # Rearrange to [B, N, N, H] for LayerNorm/out‑gate
    out = out.permute(0, 2, 3, 1)                       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate (sigmoid) and gating
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw).permute(0, 2, 3, 1)   # [B, N, N, H]
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to model dimension via 1×1 conv2d
    # ------------------------------------------------------------------
    out_nchw = out.permute(0, 3, 1, 2)   # [B, H, N, N]
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16).view(
        dim, hidden_dim, 1, 1
    )
    out_final = torch.nn.functional.conv2d(out_nchw, to_out_w)   # [B, dim, N, N]
    out_final = out_final.permute(0, 2, 3, 1)  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to FP32 (expected by the harness)
    # ------------------------------------------------------------------
    return out_final.to(torch.float32)
