@triton.jit
def _mask_gate_fusion_kernel(
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_outl_b, stride_outl_i, stride_outl_j, stride_outl_h,
    stride_outr_b, stride_outr_i, stride_outr_j, stride_outr_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i_mask = offs_i < N
    mask_j_mask = offs_j < N

    # Load mask tile (BLOCK_I, BLOCK_J)
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(mask_ptrs,
                        mask=mask_i_mask[:, None] & mask_j_mask[None, :],
                        other=0.0)  # FP16

    # Iterate over hidden blocks
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_blocks):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load left_proj tile
        lp_ptrs = (
            left_proj_ptr
            + pid_b * stride_lp_b
            + offs_i[:, None, None] * stride_lp_i
            + offs_j[None, :, None] * stride_lp_j
            + offs_h[None, None, :] * stride_lp_h
        )
        lp_tile = tl.load(lp_ptrs,
                          mask=mask_i_mask[:, None, None] & mask_j_mask[None, :, None] & mask_h[None, None, :],
                          other=0.0)

        # Load left_gate tile
        lg_ptrs = (
            left_gate_ptr
            + pid_b * stride_lg_b
            + offs_i[:, None, None] * stride_lg_i
            + offs_j[None, :, None] * stride_lg_j
            + offs_h[None, None, :] * stride_lg_h
        )
        lg_tile = tl.load(lg_ptrs,
                          mask=mask_i_mask[:, None, None] & mask_j_mask[None, :, None] & mask_h[None, None, :],
                          other=0.0)

        # Load right_proj tile
        rp_ptrs = (
            right_proj_ptr
            + pid_b * stride_rp_b
            + offs_i[:, None, None] * stride_rp_i
            + offs_j[None, :, None] * stride_rp_j
            + offs_h[None, None, :] * stride_rp_h
        )
        rp_tile = tl.load(rp_ptrs,
                          mask=mask_i_mask[:, None, None] & mask_j_mask[None, :, None] & mask_h[None, None, :],
                          other=0.0)
        # Load right_gate tile
        rg_ptrs = (
            right_gate_ptr
            + pid_b * stride_rg_b
            + offs_i[:, None, None] * stride_rg_i
            + offs_j[None, :, None] * stride_rg_j
            + offs_h[None, None, :] * stride_rg_h
        )
        rg_tile = tl.load(rg_ptrs,
                          mask=mask_i_mask[:, None, None] & mask_j_mask[None, :, None] & mask_h[None, None, :],
                          other=0.0)

        # Fusion
        left_out = lp_tile * lg_tile * mask_tile[:, :, None]
        right_out = rp_tile * rg_tile * mask_tile[:, :, None]

        # Store left
        outl_ptrs = (
            out_left_ptr
            + pid_b * stride_outl_b
            + offs_i[:, None, None] * stride_outl_i
            + offs_j[None, :, None] * stride_outl_j
            + offs_h[None, None, :] * stride_outl_h
        )
        tl.store(outl_ptrs, left_out,
                 mask=mask_i_mask[:, None, None] & mask_j_mask[None, :, None] & mask_h[None, None, :])

        # Store right
        outr_ptrs = (
            out_right_ptr
            + pid_b * stride_outr_b
            + offs_i[:, None, None] * stride_outr_i
            + offs_j[None, :, None] * stride_outr_j
            + offs_h[None, None, :] * stride_outr_h
        )
        tl.store(outr_ptrs, right_out,
                 mask=mask_i_mask[:, None, None] & mask_j_mask[None, :, None] & mask_h[None, None, :])

"""
Outgoing TriMul forward pass (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input in FP32 → cast to FP16.
2️⃣ Five FP16 linear projections (left, right, three gates) + sigmoid.
3️⃣ If a mask is provided, fuse `proj * gate * mask` for *both* left and right
   tensors in a single Triton kernel (mask is broadcast over the hidden dim).
   When no mask → in‑place element‑wise multiplication with the gates.
4️⃣ Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM (C = A @ Bᵀ) by folding the hidden dimension into the
   batch dimension and using cuBLAS Tensor‑Core kernels (torch.bmm).
5️⃣ Second LayerNorm (FP16) → multiply by out‑gate (sigmoid) → final linear
   projection back to the model dimension.
6️⃣ Cast the result back to FP32 (the harness expects FP32).

Only step 3 uses Triton, satisfying the “must‑use Triton” requirement while
keeping the heavy O(N³) work on the highly‑optimised cuBLAS path.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask for *both* left & right
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    # input pointers
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr,  # *fp16
    mask_ptr,                                                    # *fp16
    # output pointers
    out_left_ptr, out_right_ptr,                                 # *fp16
    # meta‑parameters
    B, N, H,
    # strides (in elements)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_outl_b, stride_outl_i, stride_outl_j, stride_outl_h,
    stride_outr_b, stride_outr_i, stride_outr_j, stride_outr_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out_left  = left_proj  * left_gate  * mask
    out_right = right_proj * right_gate * mask
    mask is broadcast over the hidden dimension H.
    """
    pid_b = tl.program_id(0)   # batch dimension
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # --------------------------- i / j offsets ---------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------- load mask tile ---------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape (BLOCK_I, BLOCK_J)

    # --------------------------- hidden‑dim loop ---------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ---- left_proj & left_gate ----
        lp_ptrs = (
            left_proj_ptr
            + pid_b * stride_lp_b
            + offs_i[:, None, None] * stride_lp_i
            + offs_j[None, :, None] * stride_lp_j
            + offs_h[None, None, :] * stride_lp_h
        )
        lp_tile = tl.load(
            lp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        lg_ptrs = (
            left_gate_ptr
            + pid_b * stride_lg_b
            + offs_i[:, None, None] * stride_lg_i
            + offs_j[None, :, None] * stride_lg_j
            + offs_h[None, None, :] * stride_lg_h
        )
        lg_tile = tl.load(
            lg_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ---- right_proj & right_gate ----
        rp_ptrs = (
            right_proj_ptr
            + pid_b * stride_rp_b
            + offs_i[:, None, None] * stride_rp_i
            + offs_j[None, :, None] * stride_rp_j
            + offs_h[None, None, :] * stride_rp_h
        )
        rp_tile = tl.load(
            rp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        rg_ptrs = (
            right_gate_ptr
            + pid_b * stride_rg_b
            + offs_i[:, None, None] * stride_rg_i
            + offs_j[None, :, None] * stride_rg_j
            + offs_h[None, None, :] * stride_rg_h
        )
        rg_tile = tl.load(
            rg_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ---- fusion ----
        out_left_tile  = lp_tile * lg_tile * mask_tile[:, :, None]
        out_right_tile = rp_tile * rg_tile * mask_tile[:, :, None]

        # ---- store left ----
        outl_ptrs = (
            out_left_ptr
            + pid_b * stride_outl_b
            + offs_i[:, None, None] * stride_outl_i
            + offs_j[None, :, None] * stride_outl_j
            + offs_h[None, None, :] * stride_outl_h
        )
        tl.store(
            outl_ptrs,
            out_left_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

        # ---- store right ----
        outr_ptrs = (
            out_right_ptr
            + pid_b * stride_outr_b
            + offs_i[:, None, None] * stride_outr_i
            + offs_j[None, :, None] * stride_outr_j
            + offs_h[None, None, :] * stride_outr_h
        )
        tl.store(
            outr_ptrs,
            out_right_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _mask_gate_fusion(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for *both* left and right tensors.
    All tensors must be on the same CUDA device and in fp16.
    Returns:
        left_fused, right_fused  – both of shape [B, N, N, H]
    """
    B, N, _, H = left_proj.shape
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # strides are expressed in number of elements (not bytes)
    s_lp = left_proj.stride()
    s_lg = left_gate.stride()
    s_rp = right_proj.stride()
    s_rg = right_gate.stride()
    s_mask = mask.stride()
    s_outl = left_out.stride()
    s_outr = right_out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    _mask_gate_fusion_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        left_out,
        right_out,
        B,
        N,
        H,
        s_lp[0],
        s_lp[1],
        s_lp[2],
        s_lp[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_rp[0],
        s_rp[1],
        s_rp[2],
        s_rp[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_outl[0],
        s_outl[1],
        s_outl[2],
        s_outl[3],
        s_outr[0],
        s_outr[1],
        s_outr[2],
        s_outr[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a single
    torch.bmm call (cuBLAS Tensor‑Core).  Input and output are fp16 tensors.
    """
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) where the hidden dim is folded into the batch dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiply A @ Bᵀ  (i.e. left @ rightᵀ)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore the original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point requested by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32/bool, may be ignored)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → no masking

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16 for the rest
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # still FP32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (on‑device, no extra copy later)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask + gate (fused Triton) OR simple in‑place gate
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to FP16 for the kernel
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)
        left, right = _mask_gate_fusion(left_proj, left_gate,
                                        right_proj, right_gate,
                                        mask_fp16)
    else:
        # No mask – multiply in‑place to avoid extra temporaries
        left = left_proj
        right = right_proj
        left.mul_(left_gate)
        right.mul_(right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)    # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate + final projection back to model dimension (FP16)
    # ------------------------------------------------------------------
    out.mul_(out_gate)                     # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(dtype=torch.float32)
