def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)
    # Determine batch and head index
    batch = pid_hb // H
    head = pid_hb % H
    # offsets:
    offset_batch = batch * stride_left_b
    offset_head = head * stride_left_h
    # compute offsets for left and right (both same base offset)
    base_ptr_offset = offset_batch + offset_head

    # start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for within bounds
    valid_i = offs_i < N
    valid_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        valid_k = offs_k < N

        # compute pointer offsets
        # left: (i, k)
        a_ptrs = left_ptr + base_ptr_offset + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # right transposed: shape (k, j)
        b_ptrs = right_ptr + base_ptr_offset + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

        # Load
        a = tl.load(a_ptrs, mask=valid_i[:, None] & valid_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=valid_k[:, None] & valid_j[None, :], other=0.0)

        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N) accumulation
    # After loop, store result
    # Compute out_ptr offsets
    out_offset = batch * stride_out_b + head * stride_out_h
    c_ptrs = out_ptr + out_offset + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    mask = valid_i[:, None] & valid_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
TriMul “outgoing’’ forward pass – highly optimized.

The implementation follows the reference PyTorch module but:
* The first LayerNorm is kept in PyTorch (FP32, numerically stable).
* The four linear projections that share the same input tensor are fused:
  - left_proj + left_gate → one matmul with weight (2·hidden, dim)
  - right_proj + right_gate → one matmul with weight (2·hidden, dim)
  - out_gate is a separate matmul.
* Masking and sigmoid gating are applied with plain PyTorch element‑wise ops
  (they are already fused into a single kernel by the library).
* The costly O(N³) contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
  is performed by a custom Triton kernel that works directly on the
  (B,N,N,hidden) layout, avoiding the expensive permute‑and‑contiguous
  copies required for a torch‑bmm implementation.
* All heavy computations are carried out in FP16 with FP32 accumulation
  inside the Triton kernel.  The final result is cast back to FP32.

The kernel launch grid is (⌈N/BM⌉, ⌈N/BN⌉, B·hidden) where each block
computes a BM×BN tile of the output for a particular batch‑head pair.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: O(N³) contraction out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # *fp16  [B,N,N,H]   (input)
    right_ptr,         # *fp16  [B,N,N,H]   (input)
    out_ptr,           # *fp16  [B,N,N,H]   (output)
    B, N, H,                               # int   shapes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,   # strides for left
    stride_right_b, stride_right_j, stride_right_k, stride_right_h, # strides for right
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,       # strides for out
    BLOCK_M: tl.constexpr,    # tile size along the i‑dimension
    BLOCK_N: tl.constexpr,    # tile size along the j‑dimension
    BLOCK_K: tl.constexpr,    # tile size along the reduction (k) dimension
):
    # ------------------------------------------------------------------
    # Program identifiers
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)          # tile index along i
    pid_j = tl.program_id(1)          # tile index along j
    pid_hb = tl.program_id(2)         # combined batch‑head index

    # ------------------------------------------------------------------
    # De‑compose the combined batch‑head index
    # ------------------------------------------------------------------
    batch_id = pid_hb // H
    head_id  = pid_hb % H

    # ------------------------------------------------------------------
    # Starting offsets for this block
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # Offsets within each tile
    # ------------------------------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # Masks for out‑of‑bounds indices (N may not be a multiple of BLOCK_*)
    # ------------------------------------------------------------------
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Compute the constant offset for the current (batch, head)
    # ------------------------------------------------------------------
    # left/right/out have the same inner‑most layout, so the head offset
    # is simply `head_id` (stride along the hidden dimension is 1).
    off_batch = batch_id * stride_left_b           # stride over the batch dimension
    off_head  = head_id * stride_left_h            # = head_id
    base_off  = off_batch + off_head               # add once for left/right/out

    # ------------------------------------------------------------------
    # Accumulator in FP32 (high‑precision sum)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load a tile of the left matrix A[i, k] (shape = BLOCK_M × BLOCK_K)
        # ------------------------------------------------------------------
        a_ptrs = (
            left_ptr
            + base_off
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Load a tile of the *transposed* right matrix Bᵀ[k, j]
        #   right is stored as [b, j, k, h]; we need Bᵀ[k, j] = right[b, j, k, h]
        # ------------------------------------------------------------------
        b_ptrs = (
            right_ptr
            + base_off
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Accumulate the partial products
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_N)  FP32 accumulation

    # ------------------------------------------------------------------
    # Write the tile back to the output tensor
    # ------------------------------------------------------------------
    c_ptrs = (
        out_ptr
        + base_off
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradient computation).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool / 0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, defaults to True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → stable statistics)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float32)

    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the computation
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights used in matmuls to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)   # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)        # (dim, H)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse left projection & left gate into a single matmul
    #    Same for the right side – saves two kernel launches.
    # ------------------------------------------------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2·H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2·H, dim)

    # Combined linear maps (shape → [B,N,N,2·H])
    left_comb = F.linear(x, left_comb_w)        # FP16
    right_comb = F.linear(x, right_comb_w)      # FP16

    # Split back into projection & gate pre‑activations
    left_proj = left_comb[..., :hidden_dim]                     # (B,N,N,H)
    left_gate_pre = left_comb[..., hidden_dim:2*hidden_dim]    # (B,N,N,H)
    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:2*hidden_dim]

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # 5️⃣ out‑gate (used after the second LayerNorm)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Apply mask (if any) and element‑wise gates.
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Broadcast mask over the hidden dimension
        mask_fp16 = mask_tensor.to(fp16).unsqueeze(-1)   # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via a custom Triton kernel.
    #    The inputs are already in the required layout [B,N,N,H]
    # ------------------------------------------------------------------
    B_val, N_val, _, H_val = left.shape  # sanity check
    out = torch.empty_like(left, dtype=fp16)   # (B,N,N,H)

    # Strides for the three tensors (the layout is contiguous,
    # i.e. (B,N,N,H) with H as the fastest‑varying dimension).
    stride_left = left.stride()   # (stride_b, stride_i, stride_k, stride_h)
    stride_right = right.stride()
    stride_out = out.stride()

    # Tile sizes – tuned for H100 (128‑bit registers, Tensor‑cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N_val + BLOCK_M - 1) // BLOCK_M,
        (N_val + BLOCK_N - 1) // BLOCK_N,
        B_val * H_val,
    )

    _trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B_val,
        N=N_val,
        H=H_val,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still in FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply the final gate and project back to the original dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)   FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
