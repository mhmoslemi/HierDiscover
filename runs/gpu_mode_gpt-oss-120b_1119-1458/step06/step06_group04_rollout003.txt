@triton.jit
def mask_gate_fuse_both_kernel( left_ptr, left_gate_ptr, right_ptr, right_gate_ptr, mask_ptr, hidden_dim, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    left_val = tl.load(left_ptr + offs, mask=mask)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=mask)
    right_val = tl.load(right_ptr + offs, mask=mask)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=mask)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    new_left = left_val * left_gate_val * mask_val
    new_right = right_val * right_gate_val * mask_val

    tl.store(left_ptr + offs, new_left, mask=mask)
    tl.store(right_ptr + offs, new_right, mask=mask)

"""
TriMul outgoing forward pass (no autograd).

Algorithm
---------
1️⃣ First LayerNorm (float32) – keeps numerical stability.
2️⃣ Cast to FP16 for the heavy parts.
3️⃣ Fuse the linear projection and its gate for the left and right sides:
   * Concatenate the projection‑and‑gate weight matrices.
   * One matmul per side produces both the projected features and the raw gate
     pre‑activations.
4️⃣ Apply sigmoid to the gate tensors.
5️⃣ Optional mask + gate fusion using a tiny Triton kernel (in‑place on the
   projected tensors).  This saves two memory loads per element.
6️⃣ Core O(N³) contraction:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented as a batched GEMM (B·H, N, N) × (B·H, N, N)^T via torch.bmm,
   which exploits TensorCores on H100.
7️⃣ Second LayerNorm (FP16) + out‑gate multiplication.
8️⃣ Final linear projection back to the original channel dimension.
9️⃣ Cast result back to float32 for the harness.

All heavy arithmetic runs in FP16; the only Triton code is the mask‑gate‑fusion
kernel.  The implementation works for both masked and un‑masked inputs and
supports the dimensions required by the AlphaFold3 “outgoing’’ TriMul.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate multiplications.
# In‑place updates `left` and `right`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_both_kernel(
    left_ptr,            # *fp16  → left tensor (flattened)
    left_gate_ptr,       # *fp16  → left‑gate tensor (flattened)
    right_ptr,           # *fp16  → right tensor (flattened)
    right_gate_ptr,      # *fp16  → right‑gate tensor (flattened)
    mask_ptr,            # *fp16  → mask tensor [B·N·N], broadcast over hidden_dim
    hidden_dim,          # i32   → size of the hidden dimension (H)
    total_elems,         # i64   → total number of elements in left/right (B·N·N·H)
    BLOCK: tl.constexpr, # compile‑time block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_g = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_g = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (broadcast across hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1

    # Fuse multiplications
    left_out = left * left_g * m
    right_out = right * right_g * m

    # Store in‑place
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM that implements the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.

    Parameters
    ----------
    left, right : torch.Tensor
        FP16 tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        FP16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) so we can use torch.bmm.
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B*H, N, N]
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [B*H, N, N]
    # Perform A @ Bᵀ for each head‑batch.
    c = torch.bmm(a, b.transpose(1, 2))                           # [B*H, N, N]
    # Reshape back to [B, N, N, H].
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
