def tri_mul_contraction_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute tile offsets for i and j
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

# -*- coding: utf-8 -*-
"""
Optimized outgoing TriMul kernel.

Algorithm:
1. Apply LayerNorm to input (float32) → fp16.
2. Linear projections for left/right (fp16) and gate projections (sigmoid, fp16).
3. If a mask is present, fuse mask‐gate‐projection with a Triton kernel.
   Otherwise, fuse maskless projection with elementwise multiplication.
4. Compute the O(N³) contraction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that jointly loads all hidden channels
   for a (i,k) or (j,k) tile, avoiding the costly transpositions and
   copies required for cuBLAS batched GEMM.
5. Apply second LayerNorm, multiply by out_gate and final linear projection.
6. Cast result back to float32.

All heavy compute is performed in FP16 (with FP32 accumulation) to
exploit Tensor Cores on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: mask * gate * proj (broadcast mask over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,           # *fp16 [B,N,N,H]
    gate_ptr,           # *fp16 [B,N,N,H] (sigmoid already)
    mask_ptr,           # *fp16 [B,N,N]
    out_ptr,            # *fp16 [B,N,N,H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load mask tile
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape [BLOCK_I, BLOCK_J]

    # Hidden‐dim tiles
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # projection
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # gate
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fuse
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

# ----------------------------------------------------------------------
# Triton kernel: O(N³) contraction (left @ rightᵀ) without materialising
# contiguous copies per hidden channel.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_contraction_kernel(
    left_ptr,      # *fp16 [B,N,N,H]
    right_ptr,     # *fp16 [B,N,N,H]
    out_ptr,       # *fp16 [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled matmul with hidden‑dim fusion.
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over hidden‑dim tiles
    for h_tile in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator in FP32
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # Loop over K‑tiles
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left tile: shape [I, K, H]
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
            )
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )  # fp16

            # Load right tile transposed: shape [K, J, H]
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_k[:, None, None] * stride_right_k
                + offs_j[None, :, None] * stride_right_j
                + offs_h[None, None, :] * stride_right_h
            )
            right_tile = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )  # fp16

            # Compute per‑hidden‑dim dot product
            for h in range(0, BLOCK_H):
                # left_slice: [I, K], right_slice: [K, J]
                left_slice = left_tile[:, :, h]
                right_slice = right_tile[:, :, h]
                # dot product over K
                acc[:, :, h] += tl.dot(left_slice, right_slice)

        # Write out the tile for this hidden block
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Fuse proj * gate * mask via Triton."""
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    stride_proj = proj.stride()
    stride_gate = gate.stride()
    stride_mask = mask.stride()
    stride_out = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8  # works well up to hidden_dim=384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        stride_proj[0],
        stride_proj[1],
        stride_proj[2],
        stride_proj[3],
        stride_gate[0],
        stride_gate[1],
        stride_gate[2],
        stride_gate[3],
        stride_mask[0],
        stride_mask[1],
        stride_mask[2],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


def _contract_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:

# -*- coding: utf-8 -*-
"""
Optimized outgoing TriMul kernel (forward‑only).

Algorithm
---------
1. LayerNorm on the input (float32) → fp16.
2. Linear projections for “left” and “right” and their sigmoid gates
   (all fp16).
3. If a mask is present, fuse mask, projection and gate with a Triton
   kernel:
       out = proj * gate * mask   (mask broadcast over the hidden dim).
   Otherwise we simply multiply the projection by the gate.
4. Compute the O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton kernel that loads all hidden channels of a tile
   at once, avoiding the costly transpose+contiguous copy required by
   cuBLAS batched GEMM.  The kernel accumulates in FP32 and writes back
   fp16.
5. Second LayerNorm (fp16) → out‑gate (sigmoid) → final linear
   projection back to the model dimension.
6. Cast the result back to float32 (the harness expects fp32).

All heavy arithmetic is performed in FP16 (accumulating in FP32) to
exploit Tensor‑Core performance on H100.  At least part of the work –
mask‑gate fusion and the N³ contraction – is implemented in Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask   (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H]  (sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """out = proj * gate * mask   (mask broadcast over H)"""
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # load mask tile (BLOCK_I × BLOCK_J)
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16

    # hidden‑dim tiles
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # gate
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Triton kernel: O(N³) contraction (left @ rightᵀ) without per‑hidden copies
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_contraction_kernel(
    left_ptr,      # *fp16 [B, N, N, H]
    right_ptr,     # *fp16 [B, N, N, H]
    out_ptr,       # *fp16 [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using hidden‑dim fusion.
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # iterate over hidden tiles
    for h_tile in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # iterate over K tiles
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left tile  [I, K, H]
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
            )
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )  # fp16

            # right tile transposed  [K, J, H]
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_k[:, None, None] * stride_right_k
                + offs_j[None, :, None] * stride_right_j
                + offs_h[None, None, :] * stride_right_h
            )
            right_tile = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )  # fp16

            # per‑hidden‑dim matmul
            for h in range(0, BLOCK_H):
                left_slice = left_tile[:, :, h]      # (I, K)
                right_slice = right_tile[:, :, h]    # (K, J)
                acc[:, :, h] += tl.dot(left_slice, right_slice)

        # write the block for this hidden tile
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )
