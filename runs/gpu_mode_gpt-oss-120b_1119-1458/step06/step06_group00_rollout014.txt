import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_fusion_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements, H,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load raw projection values
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Sigmoid: 1 / (1 + exp(-x))
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # Mask handling
    if HAS_MASK:
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)
    else:
        mask_val = 1.0

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Implements the AlphaFold3 TriMul operation using a fused
    linear projection + sigmoid + optional mask kernel and a batched
    GEMM expressed as a torch.bmm over the hidden dimension.
    The implementation is heavily fused to reduce memory traffic and
    exploits FP16 tensor‑core kernels on H100.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16
    dtype_fp32 = torch.float32

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim] FP32

    # --------------------------------------------------------------
    # 2. Cast to FP16 for heavy ops
    # --------------------------------------------------------------
    x = x.to(dtype_fp16)

    # --------------------------------------------------------------
    # 3. Fuse all linear projections into a single matmul
    # --------------------------------------------------------------
    # Concatenate the five projection weight matrices (each [hidden_dim, dim])
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)  # [5*hidden_dim, dim]

    # Combined linear: shape [B, N, N, 5*hidden_dim]
    combined = F.linear(x, combined_weight)  # FP16

    # --------------------------------------------------------------
    # 4. Split combined tensor into its 5 components
    # --------------------------------------------------------------
    split_tensors = torch.split(combined, hidden_dim, dim=-1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = split_tensors

    # --------------------------------------------------------------
    # 5. Compute out_gate (sigmoid) – used after the contraction
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)  # FP16, shape [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6. Apply mask and sigmoid gates to left/right projections
    # --------------------------------------------------------------
    if nomask:
        # Simple element‑wise path – no mask to load
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Convert mask to FP16 and ensure contiguity
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        left_proj_raw = left_proj_raw.contiguous()
        right_proj_raw = right_proj_raw.contiguous()
        left_gate_raw = left_gate_raw.contiguous()
        right_gate_raw = right_gate_raw.contiguous()

        total = left_proj_raw.numel()
        BLOCK = 1024  # ~1 K threads per block (tuned for H100)
        grid = ((total + BLOCK - 1) // BLOCK,)

        _gate_mask_fusion_kernel[grid](
            left_ptr=left_proj_raw,
            right_ptr=right_proj_raw,
            left_gate_ptr=left_gate_raw,
            right_gate_ptr=right_gate_raw,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            HAS_MASK=1,
            num_warps=4,
            num_stages=1,
        )
        # Unpack the in‑place results
        left = left_proj_raw
        right = right_proj_raw

    # --------------------------------------------------------------
    # 7. O(N³) contraction: batched GEMM (left @ rightᵀ) per hidden head
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape: [B, N, N, hidden_dim]

    # Reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # Right transposed (swap the two N dimensions before flattening)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication using Tensor‑Core (FP16)
    contracted = torch.bmm(left_flat, right_flat)  # (B*H, N, N)

    # Restore the original layout: [B, N, N, hidden_dim]
    out = contracted.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Second LayerNorm on the hidden dimension (still FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9. Apply the final out‑gate and linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate  # element‑wise gating (FP16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = F.linear(out, to_out_w)  # FP16 -> shape [B, N, N, dim]

    # --------------------------------------------------------------
    # 10. Cast result back to FP32 for the caller
    # --------------------------------------------------------------
    return out.to(dtype_fp32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_fusion_kernel(
    left_ptr,            # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,           # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,       # *fp16  [B,N,N,H]  (pre‑sigmoid)
    right_gate_ptr,      # *fp16  [B,N,N,H]  (pre‑sigmoid)
    mask_ptr,            # *fp16  [B,N,N]    (broadcasted, optional)
    total_elements,      # int    total number of elements (B·N·N·H)
    H,                   # int    hidden dimension
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Apply sigmoid gates and (optionally) a binary mask to the left/right
    projections.  The mask is broadcast over the hidden dimension: each
    mask element corresponds to ``H`` consecutive entries in the left/right
    tensors.  The result overwrites the left/right tensors in‑place.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load the raw projections and gate tensors
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Sigmoid (1/(1+exp(-x))) – cheap elementwise op
    left_gate  = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

    # Mask handling (if present)
    if HAS_MASK:
        # each mask value spans H hidden entries
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)
    else:
        mask_val = 1.0

    # Apply gates and mask
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store results back in‑place
    tl.store(left_ptr  + offs, left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)

) -> torch.Tensor:

    Outgoing TriMul forward pass (no gradients).

    This implementation fuses all linear projections (left/right value
    projections, their gates, and the output gate) into a single matmul,
    applies sigmoid + optional mask with a custom Triton kernel, and
    performs the O(N³) contraction via a batched FP16 GEMM (torch.bmm).
    The routine stays entirely on‑device and returns a float32 tensor.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    device = input_tensor.device
    fp16   = torch.float16
    fp32   = torch.float32

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on input (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # [B, N, N, dim] -> FP32
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy kernels
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five linear maps into a single matmul
    # ------------------------------------------------------------------
    # Each weight has shape (hidden_dim, dim).  Concatenating on dim=0 gives
    # shape (5*hidden_dim, dim).  The order must match the later split.
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(device=device, dtype=fp16)

    # Combined projection: shape [B, N, N, 5*hidden_dim] (FP16)
    combined = F.linear(x, combined_weight)

    # ------------------------------------------------------------------
    # 4️⃣ Split into the five components
    # ------------------------------------------------------------------
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = \
        torch.split(combined, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 5️⃣ Compute the output gate (sigmoid) – used after the contraction
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)   # [B, N, N, hidden_dim], FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply mask + sigmoid gates to left/right projections
    # ------------------------------------------------------------------
    if nomask:
        # Simple element‑wise gating, no mask load required.
        left_gate  = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left  = left_proj_raw * left_gate          # [B,N,N,H]
        right = right_proj_raw * right_gate        # [B,N,N,H]
    else:
        # Convert mask to FP16 (0/1) and make everything contiguous.
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        left_proj_raw = left_proj_raw.contiguous()
        right_proj_raw = right_proj_raw.contiguous()
        left_gate_raw = left_gate_raw.contiguous()
        right_gate_raw = right_gate_raw.contiguous()

        total = left_proj_raw.numel()
        BLOCK = 1024            # 1 K threads per block – good for H100
        grid  = ((total + BLOCK - 1) // BLOCK,)

        # Triton kernel performs: left = left_proj * sigmoid(left_gate) * mask
        #                         right= right_proj * sigmoid(right_gate) * mask
        _gate_mask_fusion_kernel[grid](
            left_ptr=left_proj_raw,
            right_ptr=right_proj_raw,
            left_gate_ptr=left_gate_raw,
            right_gate_ptr=right_gate_raw,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            HAS_MASK=1,
            num_warps=4,
            num_stages=1,
        )
        # The kernel overwrites left_proj_raw / right_proj_raw with gated data.
        left  = left_proj_raw
        right = right_proj_raw

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction – batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left: [B, N, N, hidden_dim]

    # Reshape to (B*H, N, N) for torch.bmm.
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # Right needs its inner N‑dimension transposed before flattening.
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication using Tensor‑cores (FP16)
    contracted = torch.bmm(left_flat, right_flat)                     # (B*H, N, N)

    # Restore layout: [B, N, N, hidden_dim]
    out = contracted.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on the hidden dimension (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 9️⃣ Apply the final out‑gate and project back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise gate (FP16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)                # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(fp32)
```
