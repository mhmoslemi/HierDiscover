Pseudo-code for kernel:

```
@triton.jit
def _mask_gate_sigmoid_kernel(
    left_ptr,
    right_ptr,
    left_gate_raw_ptr,
    right_gate_raw_ptr,
    mask_ptr,
    total_elements,
    H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
    EPS: tl.constexpr = 1e-6,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    mask_offs = offs // H

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate_raw = tl.load(left_gate_raw_ptr + offs, mask=valid, other=0.0)
    right_gate_raw = tl.load(right_gate_raw_ptr + offs, mask=valid, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # Sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # LayerNorm (FP32)
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # Combine weights
    left_combined_weight = torch.cat([weights['left_proj.weight'],
                                      weights['left_gate.weight'],
                                      weights['out_gate.weight']], dim=0).to(dtype=torch.float16, device=x.device)
    right_combined_weight = torch.cat([weights['right_proj.weight'],
                                      weights['right_gate.weight']], dim=0).to(torch.float16)

    # Linear passes
    left_combined = F.linear(x_fp16, left_combined_weight)   # shape B,N,N,3*hidden_dim
    right_combined = F.linear(x_fp16, right_combined_weight) # shape B,N,N,2*hidden_dim

    # Split
    left_proj_raw = left_combined[..., :hidden_dim]
    left_gate_raw = left_combined[..., hidden_dim:2*hidden_dim]
    out_gate_raw = left_combined[..., 2*hidden_dim:3*hidden_dim]

    right_proj_raw = right_combined[..., :hidden_dim]
    right_gate_raw = right_combined[..., hidden_dim:2*hidden_dim]

    # Gating + mask
    if nomask:
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        left = left_proj_raw
        right = right_proj_raw
        # Ensure contiguous
        left = left.contiguous()
        right = right.contiguous()
        left_gate_raw = left_gate_raw.contiguous()
        right_gate_raw = right_gate_raw.contiguous()
        total = left.numel()
        BLOCK = 1024  # adjustable
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_sigmoid_kernel[grid](left_ptr=left,
                                         right_ptr=right,
                                         left_gate_raw_ptr=left_gate_raw,
                                         right_gate_raw_ptr=right_gate_raw,
                                         mask_ptr=mask_fp16,
                                         total_elements=total,
                                         H=hidden_dim,
                                         BLOCK_SIZE=BLOCK,
                                         num_warps=4,
                                         num_stages=1)
        left_gate = None; right_gate = None  # not needed further
    # out_gate sigmoid
    out_gate = torch.sigmoid(out_gate_raw)

    # Contraction (batched gemm)
    B, N, _, H = left.shape
    # Reshape for bmm
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_mat.transpose(1,2))
    out = C.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # Second layernorm on hidden dimension
    out = F.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'].to(torch.float16), bias=weights['to_out_norm.bias'].to(torch.float16))

    # Apply out_gate
    out = out * out_gate

    # final linear projection 
    to_out_weight = weights['to_out.weight'].to(torch.float16)
    out = F.linear(out, to_out_weight)

    # back to fp32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward‐pass with heavy use of fused fp16 kernels.

Algorithm
---------
1) LayerNorm over the channel dimension `dim` (FP32 for numerical stability).
2) Cast to fp16 for the compute‑intensive part.
3) Fuse the five linear projections into two GEMM calls:
      * left_proj  + left_gate + out_gate  → weight shape (3*H, dim)
      * right_proj + right_gate          → weight shape (2*H, dim)
   where H = hidden_dim.  This eliminates repeated loads of the large
   input tensor `x`.
4) Split the fused tensors and apply the sigmoid gates.
   When a mask is present we fuse the masking, sigmoid and element‑wise
   multiplication into a single Triton kernel; otherwise we use simple
   in‑place multiplication.
5) Perform the O(N³) contraction as a batched matrix‑multiplication:
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   By reshaping to (B*H, N, N) this becomes a single torch.bmm call
   (accelerated by Tensor Cores).
6) LayerNorm on the hidden dimension, element‑wise out‑gate, and final
   linear projection back to `dim`.
7) Cast the result back to FP32.

Performance notes
-----------------
* Combining the projections reduces memory traffic on the massive
  (B·N²·dim) input tensor.
* The mask‑gate kernel also computes the sigmoid, removing a whole
  torch.sigmoid pass.
* All heavy arithmetic uses fp16 with Tensor‑core friendly GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: mask + sigmoid gating for left/right projections.
#   left   = left   * sigmoid(left_gate)   * mask
#   right  = right  * sigmoid(right_gate)  * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_sigmoid_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]  (pre‑sigmoid)
    right_gate_ptr,    # *fp16  [B,N,N,H]  (pre‑sigmoid)
    mask_ptr,          # *fp16  [B,N,N]
    total_elements,    # int   total number of elements (B·N·N·H)
    H,                 # int   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Index of the mask element that corresponds to this hidden entry
    mask_offs = offs // H

    # Load values (zero for OOB)
    left   = tl.load(left_ptr + offs,    mask=valid, other=0.0)
    right  = tl.load(right_ptr + offs,   mask=valid, other=0.0)
    lgate  = tl.load(left_gate_ptr + offs,   mask=valid, other=0.0)
    rgate  = tl.load(right_gate_ptr + offs,  mask=valid, other=0.0)

    # Load mask (neutral 1.0 for OOB)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # Sigmoid (1/(1+exp(-x))) – safe in fp16 range
    lgate_sig = 1.0 / (1.0 + tl.exp(-lgate))
    rgate_sig = 1.0 / (1.0 + tl.exp(-rgate))

    # Apply mask and gates
    left_out  = left  * lgate_sig * mask_val
    right_out = right * rgate_sig * mask_val

    tl.store(left_ptr + offs,  left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul (AlphaFold3 style) forward pass – no gradients.

    Parameters
    ----------
    data : (x, mask, weights, config)
        x         : torch.Tensor [B, N, N, dim]            (float32)
        mask      : torch.Tensor [B, N, N]                (bool / 0‑1 float)
        weights   : dict of model weight tensors (float32)
        config    : dict with keys ``dim`` and ``hidden_dim``;
                    optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack input
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = x.device
    dtype_fp16 = torch.float16
    dtype_fp32 = torch.float32

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stable stats)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse linear projections (no bias)
    # ------------------------------------------------------------------
    #   left_combined: [B,N,N, 3*H]   → left_proj, left_gate, out_gate
    #   right_combined: [B,N,N, 2*H]  → right_proj, right_gate
    left_combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # concatenate on out_features dimension
    ).to(device=device, dtype=dtype_fp16)

    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)

    # Perform fused matmuls
    left_combined = F.linear(x, left_combined_weight)   # [B,N,N,3*H] FP16
    right_combined = F.linear(x, right_combined_weight) # [B,N,N,2*H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Split fused tensors
    # ------------------------------------------------------------------
    # Note: these are views; we will make them contiguous where needed.
    left_proj_raw = left_combined[..., :hidden_dim]                # [B,N,N,H]
    left_gate_raw = left_combined[..., hidden_dim:2*hidden_dim]   # [B,N,N,H]
    out_gate_raw  = left_combined[..., 2*hidden_dim:3*hidden_dim] # [B,N,N,H]

    right_proj_raw = right_combined[..., :hidden_dim]              # [B,N,N,H]
    right_gate_raw = right_combined[..., hidden_dim:2*hidden_dim] # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates (and optional mask) – fused kernel when mask present
    # ------------------------------------------------------------------
    if nomask:
        # No mask: simple element‑wise gating
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Ensure contiguous inputs for Triton
        left = left_proj_raw.contiguous()
        right = right_proj_raw.contiguous()
        left_gate_raw = left_gate_raw.contiguous()
        right_gate_raw = right_gate_raw.contiguous()

        mask_fp16 = mask.to(dtype_fp16).contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_sigmoid_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate_raw,
            right_gate_ptr=right_gate_raw,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
            num_stages=1,
        )
        # Gating done in‑place; no extra tensors needed
        # out_gate will be computed later
    # --------------------------------------------------------------
    # Sigmoid for the out‑gate (always required)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction → batched GEMM (fp16)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape for torch.bmm: (B*H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_mat.transpose(1, 2))                # (B*H, N, N)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype_fp16),
        bias=weights["to_out_norm.bias"].to(dtype_fp16),
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype_fp16))

    # ------------------------------------------------------------------
    # 9️⃣ Return FP32 result
    # ------------------------------------------------------------------
    return out.to(dtype_fp32)
