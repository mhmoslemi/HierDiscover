---------
1️⃣  Layer‑Norm (fp32) on the raw pairwise tensor.
2️⃣  Cast to fp16 for the heavy path.
3️⃣  Linear projections (left/right) and three gates (left, right, out),
    all in fp16 (bias=False).
4️⃣  Fuse mask, left‑gate and right‑gate element‑wise multiplications
    with a tiny Triton kernel (avoid extra passes).
5️⃣  O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    implemented as a *batched* GEMM in Triton.  The original
    [B,N,N,H] layout is kept (no costly permute+contiguous copies);
    each hidden slice is treated as a separate batch.
6️⃣  Second Layer‑Norm (fp16) over the hidden dimension.
7️⃣  Multiply by the output‑gate and a final linear projection back to
    the original dimension.
8️⃣  Cast the result to fp32 (required by the harness).

Only the forward pass is provided; gradients are not needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1: fuse mask and gating (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : hidden dimension H
    size,           # i32   : total number of elements in x (= B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    off = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = off < size                                 # guard

    # the hidden dimension is the innermost stride, therefore the mask
    # index is obtained by integer division.
    mask_off = off // hidden_dim

    x   = tl.load(x_ptr   + off, mask=mask, other=0.0)
    g   = tl.load(gate_ptr+ off, mask=mask, other=0.0)
    m   = tl.load(mask_ptr+ mask_off, mask=mask, other=1.0)   # mask=1 outside

    tl.store(x_ptr + off, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: batched matrix‑multiply for the O(N³) contraction.
#    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# The layout of left/right/out is [B, N, N, H] (contiguous, H innermost).
# Each (b, h) pair is treated as a separate batch entry.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                                   # dimensions (all = seq_len)
    # strides for A = left tensor
    a_batch_stride, a_i_stride, a_k_stride, a_h_stride,
    # strides for B = right tensor (transposed on‑the‑fly)
    b_batch_stride, b_k_stride, b_j_stride, b_h_stride,
    # strides for C = out tensor
    c_batch_stride, c_i_stride, c_j_stride, c_h_stride,
    hidden_dim,                                 # integer = H
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # 1️⃣  Determine which (batch, hidden) we are processing.
    # ---------------------------------------------------------------
    batch_id = tl.program_id(0)           # combined index = b*H + h
    pid_m   = tl.program_id(1)           # row‑tile index
    pid_n   = tl.program_id(2)           # col‑tile index

    b_idx = batch_id // hidden_dim
    h_idx = batch_id % hidden_dim

    # Base pointer for this (b, h) pair
    a_batch = a_ptr + b_idx * a_batch_stride + h_idx * a_h_stride
    b_batch = b_ptr + b_idx * b_batch_stride + h_idx * b_h_stride
    c_batch = c_ptr + b_idx * c_batch_stride + h_idx * c_h_stride

    # ---------------------------------------------------------------
    # 2️⃣  Tile offsets for the output block
    # ---------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # ---------------------------------------------------------------
    # 3️⃣  Accumulator (fp32) – we will cast back to fp16 at the end
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # 4️⃣  Loop over the reduction dimension K in BLOCK_K tiles
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load a tile from A (shape BLOCK_M x BLOCK_K) ----
        a_off = (offs_i[:, None] * a_i_stride) + (offs_k[None, :] * a_k_stride)
        a = tl.load(a_batch + a_off,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ---- load a tile from Bᵀ (shape BLOCK_K x BLOCK_N) ----
        # Bᵀ is realised by swapping the j/k axes of the original tensor.
        b_off = (offs_k[:, None] * b_k_stride) + (offs_j[None, :] * b_j_stride)
        b = tl.load(b_batch + b_off,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ---- multiply‑add over k (tl.dot uses Tensor‑Core if possible) ----
        acc += tl.dot(a, b)       # fp16 * fp16 → fp32 accumulation

    # ---------------------------------------------------------------
    # 5️⃣  Store the result back to C (cast to fp16)
    # ---------------------------------------------------------------
    c_off = (offs_i[:, None] * c_i_stride) + (offs_j[None, :] * c_j_stride)
    tl.store(c_batch + c_off, acc.to(tl.float16),
            mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# The public entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float or None)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` key.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # True means *no* mask is present

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stabilise statistics
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # -> [B, N, N, dim]  (fp32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the work
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast all model weights once (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 5️⃣ Compute gates (sigmoid) – also fp16
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fused multiplication (left/right * gate * mask)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → fp16, contiguous, broadcast over H
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()                # B·N·N·H
        BLOCK = 1024                               # must be ≤ max threads/block
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # Fuse: left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse: right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple elementwise multiplication (in‑place)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ O(N³) contraction via a custom batched GEMM kernel (no data copy)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                     # N = seq_len, H = hidden_dim

    # Output buffer
    out = torch.empty_like(left)                # [B, N, N, H] fp16

    # Strides for left/right/out (all share the same layout)
    a_batch_stride, a_i_stride, a_k_stride, a_h_stride = left.stride()
    b_batch_stride, b_k_stride, b_j_stride, b_h_stride = right.stride()
    c_batch_stride, c_i_stride, c_j_stride, c_h_stride = out.stride()

    total_batches = B * H                        # combined (b, h) dimension

    # Choose tile sizes (tuned for H100, but fall back safely for any N)
    BLOCK_M = 128 if N >= 128 else N
    BLOCK_N = 128 if N >= 128 else N
    BLOCK_K = 64 if N >= 64 else N

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        left,
        right,
        out,
        N, N, N,                                 # M, N, K  (all seq_len)
        a_batch_stride, a_i_stride, a_k_stride, a_h_stride,
        b_batch_stride, b_k_stride, b_j_stride, b_h_stride,
        c_batch_stride, c_i_stride, c_j_stride, c_h_stride,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                             # enough warps for H100 matmul
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Output gating + final projection back to original dim (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # -> [B, N, N, dim]

    # -----------------------------------------------------------------
    # 🔟 Cast back to fp32 (required by the external harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
