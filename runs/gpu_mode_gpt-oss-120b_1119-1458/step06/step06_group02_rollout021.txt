def batched_matmul_triton(left, right):
    # left, right shape [B,N,N,H] float16
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] (contiguous)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat)
    # Compute strides
    stride_batch_a, stride_m_a, stride_k_a = left_flat.stride()
    stride_batch_b, stride_k_b, stride_n_b = right_flat.stride()
    stride_batch_c, stride_m_c, stride_n_c = out_flat.stride()
    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Compute grid dims
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * left_flat.shape[0], num_blocks_n) # flatten batch dimension within axis0
    # Launch kernel
    batched_matmul_triton_kernel[grid](
        left_flat, right_flat, out_flat,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        stride_m_a, stride_k_a,
        stride_k_b, stride_n_b,
        stride_m_c, stride_n_c,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4, # maybe 8 for larger N
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1. Fuse masking, gating and projection (mask optional)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,               # *fp16 : projection output (left or right)
    gate_ptr,               # *fp16 : gate output (same shape as proj)
    mask_ptr,               # *fp16 : mask (flattened B·N·N)
    hidden_dim,             # i32   : size of hidden dimension H
    total_elems,            # i32   : total number of elements in proj/gate
    BLOCK: tl.constexpr,    # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    active = offs < total_elems                         # OOB guard

    # mask is broadcast over hidden dim → each hidden element shares the same mask entry
    mask_offs = offs // hidden_dim
    p = tl.load(proj_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)   # 1 outside range → no effect
    out = p * g * m
    tl.store(proj_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel (B*H independent GEMMs)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_triton_kernel(
    a_ptr, b_ptr, c_ptr,                         # tensor pointers
    stride_batch_a, stride_batch_b, stride_batch_c,
    stride_am, stride_ak,                        # A‑strides (M‑row, K‑col)
    stride_bk, stride_bn,                        # B‑strides (K‑row, N‑col) – B already transposed
    stride_cm, stride_cn,                        # C‑strides (M‑row, N‑col)
    M, N, K,                                     # matrix sizes (square N×N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B   where
        A : [batch, M, K]
        B : [batch, K, N]   (B is already the transposed operand)
        C : [batch, M, N]
    """
    pid = tl.program_id(axis=0)          # flatten (batch, block_M) into one dimension
    pid_n = tl.program_id(axis=1)        # block index along N

    # number of output‑row tiles per batch
    num_block_m = tl.cdiv(M, BLOCK_M)

    batch_idx = pid // num_block_m
    block_i = pid % num_block_m

    # coordinates of this tile
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the start of the tile (k‑offset = 0)
    a_ptrs = a_ptr + batch_idx * stride_batch_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_idx * stride_batch_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k_idx in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for partial tiles at the matrix boundaries
        mask_a = (offs_m[:, None] < M) & (k_off[None, :] < K)
        mask_b = (k_off[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)                     # tensor‑core dot

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write out the tile
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_idx * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask_c)


def batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM based on the Triton kernel above.
    left, right : [B, N, N, H]  (fp16)
    Returns out  : [B, N, N, H]  (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]; left: (i,k) ; right needs to be transposed on (k,j)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous()          # still (B,H,i,j)
    right_flat = right_flat.transpose(2, 3)                     # -> (B,H,j,i)  i.e. (B*H, K, N)

    right_flat = right_flat.view(B * H, N, N)  # now shape [B*H, K, N] where K==N

    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_batch_a, stride_am, stride_ak = left_flat.stride()
    stride_batch_b, stride_bk, stride_bn = right_flat.stride()
    stride_batch_c, stride_cm, stride_cn = out_flat.stride()

    # tile sizes – tuned for H100 (adjustable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_blocks_m * (B * H), num_blocks_n)   # flatten batch dimension into axis‑0

    batched_matmul_triton_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N,                # M
        N,                # N
        N,                # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # 8 warps (256 threads) works well on H100
    )

    # restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------- unpack ------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------- Layer‑Norm (FP32) ------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  FP32

    # ------------------- cast to FP16 -----------------------------------------
    x = x.to(torch.float16)

    # ------------------- fp16 weight copies ------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------- left / right projection + gating ----------------------
    # linear projection
    left_proj = torch.nn.functional.linear(x, left_proj_w)      # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # gating (sigmoid of separate linear)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))

    # ------------------- mask & gating fusion (Triton) ------------------------
    if not nomask:
        # mask may be bool or float; bring to fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total = left_proj.numel()
        BLOCK = 1024
        grid_mask = ((total + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid_mask](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid_mask](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # no mask – fuse only the elementwise product
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- heavy O(N³) contraction -----------------------------
    # out shape: [B,N,N,hidden_dim] (fp16)
    out = batched_matmul_triton(left, right)

    # ------------------- second Layer‑Norm (FP16) over hidden dim ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------- output gate ------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))
    out = out * out_gate

    # ------------------- final linear projection -------------------------------
    out = torch.nn.functional.linear(out, to_out_w)          # back to [B,N,N,dim] (fp16)

    # ------------------- cast back to FP32 ------------------------------------
    return out.to(torch.float32)
