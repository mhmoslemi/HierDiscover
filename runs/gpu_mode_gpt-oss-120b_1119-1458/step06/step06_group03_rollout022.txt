```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH: tl.constexpr, N: tl.constexpr,
    stride_left_b, stride_left_m, stride_left_k,
    stride_right_b, stride_right_k, stride_right_n,
    stride_out_b, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)    # 0..BATCH-1
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute global offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_iter = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_iter):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers for A and B (B_T)
        a_ptrs = left_ptr + batch_id * stride_left_b + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr + batch_id * stride_right_b + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product (accumulator in FP32)
        acc += tl.dot(a, b)
    # Cast to FP16
    out = acc.to(tl.float16)

    # store output
    out_ptrs = out_ptr + batch_id * stride_out_b + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – high‑performance Triton kernel implementation.
-----------------------------------------------------------------
Algorithm Overview
------------------
1️⃣ First LayerNorm (FP32 → FP16) – the heavy part of the model works in
   half precision while the statistics are computed in full precision.
2️⃣ Fuse the three linear projections for the *left* and *right* paths:
   – combine (proj, gate) weights into a single matrix‑multiply so that the
     input tensor is read only once.
   – apply a sigmoid to the gate half of the result.
3️⃣ If a mask is present we fuse element‑wise multiplication
   (proj * gate * mask) with a tiny Triton kernel (`fused_gate_mask_kernel`);
   otherwise we multiply only by the gate.
4️⃣ Contract the two 4‑D tensors:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is an O(N³) batched GEMM.  We launch a custom Triton kernel
   (`tri_mul_outgoing_kernel`) that treats each (b,h) as a separate batch
   and performs the matrix multiplication using Tensor‑Core‑enabled
   `tl.dot`.  The kernel tiles the output with BLOCK_M×BLOCK_N tiles and
   iterates over K in blocks of size BLOCK_K.
5️⃣ Second LayerNorm (FP16) over the hidden dimension.
6️⃣ Apply the output gate (sigmoid of a linear projection of the original
   normalized input) and the final linear projection back to `dim`.
7️⃣ Cast back to float32 for the harness.

The implementation keeps the memory footprint low (in‑place operations,
view‑based reshapes) and uses mixed‑precision everywhere except the
first LayerNorm where FP32 stability is useful.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: fused (tensor * gate * mask) – in‑place on FP16 data.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated (`proj`)
    gate_ptr,             # *fp16 : gate tensor (same shape as `proj`)
    mask_ptr,             # *fp16 : mask tensor (shape = B*N*N, broadcast)
    total_elts: tl.int32, # total elements = B*N*N*hidden_dim
    hidden_dim: tl.int32, # hidden dimension (size of the last axis)
    BLOCK: tl.constexpr,  # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    valid = offs < total_elts

    # broadcast the mask over the hidden dimension
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM that implements the O(N³) contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# The input tensors are flattened to shape (B*H, N, N) where the hidden
# channel is fused into the batch dimension.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,        # FP16 pointers (shape = (B*H, N, N))
    batch_dim: tl.int32,                 # B*H
    N: tl.int32,                         # sequence length
    stride_l_batch, stride_l_row, stride_l_col,   # strides for left (rows = i, cols = k)
    stride_r_batch, stride_r_row, stride_r_col,   # strides for right (rows = k, cols = j) – transposed
    stride_o_batch, stride_o_row, stride_o_col,   # strides for output
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    batch_id = tl.program_id(0)   # 0 … batch_dim-1   (each hidden channel of each batch)
    pid_m    = tl.program_id(1)   # tile index along i (rows)
    pid_n    = tl.program_id(2)   # tile index along j (cols)

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑axis
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑axis
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base offsets for this “batch” (b,h) pair
    base_l = left_ptr + batch_id * stride_l_batch
    base_r = right_ptr + batch_id * stride_r_batch
    base_o = out_ptr + batch_id * stride_o_batch

    # accumulator (use FP32 for numerics)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the k‑dimension in BLOCK_K sized tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)           # k‑axis
        mask_k = offs_k < N

        # A tile: left[batch_id, i, k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_l + offs_m[:, None] * stride_l_row + offs_k[None, :] * stride_l_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile (transposed view of right): right[batch_id, k, j]   shape (BLOCK_K, BLOCK_N)
        b_ptrs = base_r + offs_k[:, None] * stride_r_row + offs_n[None, :] * stride_r_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # tile‑level matrix multiplication (Tensor‑Core friendly)
        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # write back FP16 result
    out = acc.to(tl.float16)
    o_ptrs = base_o + offs_m[:, None] * stride_o_row + offs_n[None, :] * stride_o_col
    tl.store(o_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict  : name → torch.Tensor (float32)
        - config_dict   : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack and move to device
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # FP32 layer‑norm, then cast to FP16 for the heavy path
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)  # shape [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (once)
    # ------------------------------------------------------------------
    # combine left proj + left gate, and right proj + right gate to halve
    # the number of reads of `x_norm`.
    left_proj_w  = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w     = weights["to_out.weight"]

    # Convert to FP16
    left_proj_w  = left_proj_w.to(torch.float16)
    right_proj_w = right_proj_w.to(torch.float16)
    left_gate_w  = left_gate_w.to(torch.float16)
    right_gate_w = right_gate_w.to(torch.float16)
    out_gate_w  = out_gate_w.to(torch.float16)
    to_out_norm_w = to_out_norm_w.to(torch.float16)
    to_out_norm_b = to_out_norm_b.to(torch.float16)
    to_out_w    = to_out_w.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused Linear + Gate for left and right paths
    # ------------------------------------------------------------------
    # concatenate weights for proj and gate (hidden_dim, dim) → (2*hidden_dim, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # One linear pass per side
    left_combined = torch.nn.functional.linear(x_norm, left_combined_w)   # [B,N,N,2*H] (fp16)
    right_combined = torch.nn.functional.linear(x_norm, right_combined_w)

    # Split back into proj and gate halves
    left  = left_combined[..., :hidden_dim]               # [B,N,N,H]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])   # [B,N,N,H]
    right = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – fuse (proj * gate * mask) via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast across hidden_dim – flatten + cast to FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)   # (B*N*N,)
        total_elts = left.numel()                # B*N*N*H
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # no mask – just multiply by gate
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via custom Triton GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten the first two dimensions
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    left_flat = left_perm.view(batch_dim, N, N)   # (B*H, N, N) – already contiguous
    right_flat = right_perm.view(batch_dim, N, N)

    # Prepare output buffer
    out_flat = torch.empty_like(left_flat)       # (B*H, N, N)  FP16

    # Strides (in elements) for the flattened tensors
    stride_l_batch = left_flat.stride(0)   # N*N
    stride_l_row   = left_flat.stride(1)   # N
    stride_l_col   = left_flat.stride(2)   # 1

    stride_r_batch = right_flat.stride(0)
    # For the transposed view we treat 'row' as k and 'col' as j
    stride_r_row   = right_flat.stride(1)   # N   (k‑stride after transpose)
    stride_r_col   = right_flat.stride(2)   # 1   (j‑stride after transpose)

    stride_o_batch = out_flat.stride(0)
    stride_o_row   = out_flat.stride(1)     # N
    stride_o_col   = out_flat.stride(2)     # 1

    # Tiling parameters – chosen for H100 Tensor‑Core efficiency
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_outgoing_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        batch_dim,
        N,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to original layout: [B, N, N, H]
    out_perm = out_flat.view(B, hidden_dim, N, N)               # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)   FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Compute out‑gate (use original normed tensor `x_norm`)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_w))  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  FP16

    # ------------------------------------------------------------------
    # 9️⃣ Return FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
