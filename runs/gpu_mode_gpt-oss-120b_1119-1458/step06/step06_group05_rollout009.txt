"""
TriMul (outgoing) forward pass – mixed‑precision, Triton‑accelerated mask & gate fusion.

Algorithm
---------
1️⃣ First LayerNorm (FP32) → cast to FP16.
2️⃣ Fuse the two linear projections per side (proj + gate) by concatenating
   the weights and performing ONE matmul.  This halves the number of
   cuBLAS calls for the (proj, gate) pair.
3️⃣ If a mask is present, a Triton kernel fuses          proj * sigmoid(gate) * mask
   (sigmoid is computed inside the kernel) to avoid extra memory traffic.
   When there is no mask we fall back to a cheap PyTorch element‑wise product.
4️⃣ Heavy O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (C = A @ Bᵀ) by flattening the
   (batch, hidden) dimensions → a single `torch.bmm` call that leverages
   H100 tensor‑cores.
5️⃣ Second LayerNorm (FP16) → element‑wise multiplication with the out‑gate.
6️⃣ Final linear projection back to the model dimension.
7️⃣ Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion kernel is written in Triton; all large matmuls use
high‑performance cuBLAS kernels.  The implementation works for any
B∈{1,2}, N∈{128,256,512,1024}, dim∈{128,384}, hidden_dim=128 and respects the
optional mask.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * sigmoid(gate) * mask   (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,               # *fp16  [B, N, N, H]
    gate_ptr,               # *fp16  [B, N, N, H]   (raw, not sigmoid‑ed)
    mask_ptr,               # *fp16  [B, N, N]
    out_ptr,                # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * sigmoid(gate) * mask    (mask broadcast over hidden dimension H)
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # i / j coordinates for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later across H.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )                                   # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles.
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16, shape [I,J,H]

        # Load raw gate tile
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16

        # Sigmoid (stable version)
        gate_sig = 1.0 / (1.0 + tl.exp(-gate_tile))

        # Fuse: proj * sigmoid(gate) * mask
        out_tile = proj_tile * gate_sig * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

# ----------------------------------------------------------------------
# Helper: batched GEMM for O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using cuBLAS.
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns out  : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)            # (BH, N, N)
    right_flat = right_perm.view(B * H, N, N)          # (BH, N, N)

    # Batched GEMM: C = A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        left = _apply_proj_gate_mask(left_proj_raw, left_gate_raw, mask_fp16)
        right = _apply_proj_gate_mask(right_proj_raw, right_gate_raw, mask_fp16)
        # No need to keep the raw tensors around
        del left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw
    else:
        # No mask ⇒ regular sigmoid + element‑wise multiplication
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
        del left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw

    # -----------------------------------------------------------------
    # 5️⃣ Compute out‑gate (sigmoid) – same trajectory as the reference
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))   # [B,N,N,H]

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS tensor cores)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) → out‑gate multiplication
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
