"""
custom_kernel implements the outgoing TriMul operation with an optimized Triton
kernel for the O(N³) contraction. The algorithm consists of:

1. First LayerNorm (float32) → cast to fp16.
2. Linear projections (left / right) to hidden_dim (fp16).
3. Compute three gating tensors (sigmoid of linear transforms) (fp16).
4. Optional mask‑gate fusion via a small Triton kernel (mask_gate_fuse_kernel).
5. Heavy contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented with a custom batched GEMM kernel (tri_mul_gemm_kernel) that
   works directly on the original [B, N, N, H] layout, thus avoiding the
   costly permute‑contiguous copies required by the reference implementation.
6. Second LayerNorm (fp16) on the contraction result.
7. Element‑wise multiplication with out_gate and final linear projection back
   to the original channel dimension.
8. Cast back to float32 (the expected output dtype).

The GEMM kernel tiles over the N dimension (i, j) and over the combined
batch‑head axis (B*H). Operands are fp16; accumulation is performed in fp32
using `tl.dot`, which maps to tensor‑core matmul on H100. This removes the
large memory copies of the reference code and yields a significant speed‑up.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and element‑wise multiplication.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated (flattened)
    gate_ptr,       # *fp16  gate tensor (same shape as x)
    mask_ptr,       # *fp16  mask tensor (flattened, shape [B*N*N])
    hidden_dim,     # i32   hidden dimension (H)
    size,           # i32   total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands (fp16)
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # mask index = element index // hidden_dim
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fuse multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
# Implements:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#
# left, right, out are stored in the original layout [B, N, N, H]
# (contiguous, with H as the innermost dimension).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,      # *fp16   left tensor   [B, N, N, H]
    right_ptr,     # *fp16   right tensor  [B, N, N, H]
    out_ptr,       # *fp16   output tensor [B, N, N, H]
    B,             # i32   batch size
    N,             # i32   sequence length
    H,             # i32   hidden dimension
    BLOCK_M: tl.constexpr,  # tile size in the i dimension
    BLOCK_N: tl.constexpr,  # tile size in the j dimension
    BLOCK_K: tl.constexpr,  # tile size in the reduction (k) dimension
):
    pid_i = tl.program_id(0)      # tile index along i (rows)
    pid_j = tl.program_id(1)      # tile index along j (cols)
    pid_bh = tl.program_id(2)     # combined batch * hidden index

    # -------------------------------------------------------------
    # Decode batch and hidden indices from the combined dimension
    # -------------------------------------------------------------
    b = pid_bh // H               # batch index 0 .. B-1
    h = pid_bh % H                # hidden index 0 .. H-1

    # -------------------------------------------------------------
    # Base offset for this (b, h) slice.
    # Layout: offset = b*N*N*H + i*N*H + j*H + h
    # -------------------------------------------------------------
    base_offset = b * N * N * H + h

    # -------------------------------------------------------------
    # Coordinate ranges for the current tile
    # -------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # rows   (0..N-1)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # cols   (0..N-1)

    mask_i = i < N
    mask_j = j < N

    # Strides in the contiguous layout
    stride_i = N * H   # advance one row (i or j)  -> N*H
    stride_k = H       # advance along reduction k -> H

    # Broadcasted offsets (in element units, not bytes)
    i_off = i[:, None] * stride_i           # (BLOCK_M, 1)
    j_off = j[None, :] * stride_k           # (1, BLOCK_N)

    # -------------------------------------------------------------
    # Accumulator in float32
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---------------------------------------------------------
        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        #   left[b, i, k, h]
        # ---------------------------------------------------------
        a_ptr = left_ptr + base_offset + i_off + k[None, :] * stride_k
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16

        # ---------------------------------------------------------
        # Load a tile from right transposed (k, j) -> shape (BLOCK_K, BLOCK_N)
        #   right[b, j, k, h]   (we use the transpose implicitly)
        # ---------------------------------------------------------
        b_ptr = right_ptr + base_offset + k[:, None] * stride_k + j_off
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # fp16

        # ---------------------------------------------------------
        # Accumulate with a tensor‑core dot product
        # ---------------------------------------------------------
        acc += tl.dot(a, b)                   # (BLOCK_M, BLOCK_N), fp32 accumulation

    # -------------------------------------------------------------
    # Store the result back to out (fp16)
    # -------------------------------------------------------------
    c_ptr = out_ptr + base_offset + i_off + j_off
    tl.store(c_ptr,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd, highly optimized):
      - first LayerNorm in FP32,
      - linear embeddings & gating in FP16,
      - optional mask‑gate fusion (Triton),
      - O(N³) contraction via a custom batched GEMM kernel (Triton, avoids copies),
      - second LayerNorm, out‑gate, final projection,
      - result returned as FP32.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → no mask, False → mask present

    device = input_tensor.device
    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → stable for large N
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the compute‑heavy part
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast all model weights once to fp16 (to avoid repeated casts)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    # input shape: [B, N, N, dim] → flatten to (B*N*N, dim) internally
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (fp16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16 and flatten (shape [B*N*N])
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)

        size = left.numel()                 # B * N * N * hidden_dim
        BLOCK = 1024                         # chosen for high occupancy
        grid_mask = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid_mask](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid_mask](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: just apply the gates (element‑wise)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – custom batched GEMM (tri_mul_gemm_kernel)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # allocate output in fp16

    # Choose tile sizes based on N (tuned for H100)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,                 # combined batch‑head dimension
    )
    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # use 8 warps per block on H100
        num_stages=2,
    )
    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over the hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                                 # element‑wise out‑gate
    out = torch.nn.functional.linear(out, to_out_w)      # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
