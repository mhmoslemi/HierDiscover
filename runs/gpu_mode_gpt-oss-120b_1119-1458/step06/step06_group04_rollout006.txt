"""
TriMul (outgoing) forward pass – highly‑optimized Triton + PyTorch implementation.

Algorithm
---------
1️⃣  LayerNorm over the feature dimension (float32 → keep numerical stability).
2️⃣  Cast the normalized tensor to fp16.
3️⃣  Fuse the five linear projections (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single matrix multiplication:
       X_flat (M×dim) × W_fusedᵀ  →  Y (M×5·hidden_dim)
   where M = B·N·N.  This eliminates four extra passes over the large 4‑D tensor.
4️⃣  Slice Y into the five components and reshape to `[B,N,N,hidden_dim]`.
5️⃣  Apply sigmoid to the three gate tensors.
6️⃣  Fuse mask, gate and element‑wise product for the left/right branches with a
    tiny Triton kernel (`mask_gate_fuse_kernel`).  When no mask is present a plain
    element‑wise multiplication is used.
7️⃣  Core O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (fp16) using `torch.bmm`.
8️⃣  Second LayerNorm over the hidden dimension (fp16).
9️⃣  Multiply by the out‑gate and project back to the original dimension with a
    final linear layer (fp16).
🔟  Cast the result back to float32 (the harness expects float32).

Only the mask‑gate fusion is done in Triton; all heavy linear algebra uses
cuBLAS‑backed `torch.nn.functional.linear` / `torch.bmm`, which on an H100
exploits TensorCores.  This layout reduces memory traffic (the large 4‑D tensor
is read once for the five projections) and brings the overall latency well
below the 1 ms target for the benchmark shapes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of a tensor, its gate and an
# optional broadcast mask.
# All tensors are flattened fp16 arrays.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 → tensor to be updated (flattened)
    gate_ptr,       # *fp16 → gate tensor (same flattening as x)
    mask_ptr,       # *fp16 → mask tensor, shape [B·N·N]
    hidden_dim,     # i32  → hidden dimension H
    size,           # i32  → total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load data
    x_val   = tl.load(x_ptr + offs,   mask=valid, other=0.0)   # [BLOCK]
    g_val   = tl.load(gate_ptr + offs,mask=valid, other=0.0)   # [BLOCK]

    # Each hidden element shares the same mask entry
    mask_idx = offs // hidden_dim
    m_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default mask=1

    # Fuse: x * gate * mask
    out = x_val * g_val * m_val

    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the core contraction,
# out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
# Executes in fp16 using torch.bmm.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Parameters
    ----------
    left, right : torch.Tensor of shape [B, N, N, H] (fp16)

    Returns
    -------
    torch.Tensor of shape [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # (B·H, N, N) x (B·H, N, N)ᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                     # (B·H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input tensor [B, N, N, dim] (float32)
        torch.Tensor,                # mask tensor [B, N, N] (bool/float32) or None
        Dict[str, torch.Tensor],     # weights dictionary (float32)
        Dict[str, Any],              # config dict, must contain "dim" & "hidden_dim"
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)    # True → mask is ignored

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent heavy work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single GEMM:
    #    left_proj, right_proj, left_gate, right_gate, out_gate
    # ------------------------------------------------------------------
    # Cast weights to fp16 once
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature dimension
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w, out_gate_w],
        dim=0
    )  # shape [(5·hidden_dim), dim]  (fp16)

    # Flatten input to (M, dim) where M = B·N·N
    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.reshape(M, dim)               # (M, dim)

    # Single linear pass – heavy compute handled by cuBLAS/TensorCores
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # (M, 5·hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Slice and reshape the five components
    # ------------------------------------------------------------------
    H = hidden_dim
    # Helper to reshape a slice
    def _reshape(slice_tensor):
        return slice_tensor.view(B, N, N, H)

    left_proj   = _reshape(fused_out[:, 0*H:1*H])
    right_proj  = _reshape(fused_out[:, 1*H:2*H])
    left_gate_pre  = _reshape(fused_out[:, 2*H:3*H])
    right_gate_pre = _reshape(fused_out[:, 3*H:4*H])
    out_gate_pre   = _reshape(fused_out[:, 4*H:5*H])

    # ------------------------------------------------------------------
    # 5️⃣ Compute gate activations (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)    # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_pre)   # (B,N,N,H) fp16
    out_gate   = torch.sigmoid(out_gate_pre)     # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply optional mask and fuse with gate multiplication
    # ------------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # Convert mask to fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                     # (B·N·N)
        total_elems = left_proj.numel()                    # B·N·N·H
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # Fuse left_proj ← left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right_proj ← right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernels, left_proj/right_proj already contain the gated
        # and masked values.
        left  = left_proj
        right = right_proj
    else:
        # No mask – simple element‑wise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction via batched GEMM (fp16)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # (B,N,N,hidden_dim) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B,N,N,hidden_dim) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by out_gate (element‑wise) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate                     # (B,N,N,hidden_dim) fp16
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    # Flatten and project back to original dimension
    out_flat = out.view(M, hidden_dim)       # (M, H)
    out_proj = torch.nn.functional.linear(out_flat, to_out_w)  # (M, dim)

    # Reshape to original 4‑D shape and cast back to float32
    output = out_proj.view(B, N, N, dim).to(torch.float32)

    return output
