    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # hidden dimension (= size of last axis of x)
    BLOCK: tl.constexpr,  # tile size (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elts

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2 – batched GEMM for the heavy O(N³) contraction
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#   left/right/out shape: [B, N, N, H]  (H == hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_gemm_kernel(
    left_ptr,    # *fp16  left  tensor  [B,N,N,H]
    right_ptr,   # *fp16  right tensor  [B,N,N,H]
    out_ptr,     # *fp16  output tensor [B,N,N,H]

    B: tl.int32,
    N: tl.int32,
    H: tl.int32,

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    BLOCK_M: tl.constexpr,  # rows of the output tile
    BLOCK_N: tl.constexpr,  # cols of the output tile
    BLOCK_K: tl.constexpr,  # reduction tile size
):
    # -------------- program ids ---------------------------------------
    batch_id = tl.program_id(0)   # encodes (b, h)
    pid_i    = tl.program_id(1)   # tile index along i
    pid_j    = tl.program_id(2)   # tile index along j

    # decode batch offset
    b = batch_id // H
    h = batch_id % H

    # starting coordinates of the output tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # -------------- index vectors ------------------------------------
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols (handles N not divisible by block)
    mask_i = i_range < N
    mask_j = j_range < N

    # base pointer for this (b,h) slice of the output tensor
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------- reduction over k ---------------------------------
    for k0 in range(0, N, BLOCK_K):
        k_range = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ------- load left tile (shape BLOCK_M x BLOCK_K) -------------
        # left[b, i, k, h]
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            h * stride_l_h + \
            i_range[:, None] * stride_l_i + \
            k_range[None, :] * stride_l_k
        mask_left = (mask_i[:, None] & mask_k[None, :])
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)

        # ------- load right tile (shape BLOCK_K x BLOCK_N) ------------
        # right[b, j, k, h] -> we need (k, j) ordering for tl.dot
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            h * stride_r_h + \
            k_range[:, None] * stride_r_k + \
            j_range[None, :] * stride_r_i
        mask_right = (mask_k[:, None] & mask_j[None, :])
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # promote to FP32 and accumulate
        acc += tl.dot(left_tile.to(tl.float32), right_tile.to(tl.float32))

    # -------------- write back ----------------------------------------
    out_ptrs = out_base + i_range[:, None] * stride_o_i + j_range[None, :] * stride_o_j
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------
# Helper: Triton‑based batched GEMM for TriMul
# ----------------------------------------------------------------------
def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a custom Triton kernel.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides are expressed in element counts (as expected by Triton)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – tuned for H100 Tensor‑Core performance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid: (batch*hidden, i‑tiles, j‑tiles)
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_batch_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,  # H100 prefers 8 warps for this tile size
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : map name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim]  FP32

    x = x_norm.to(torch.float16)  # FP16 for the heavy part

    # --------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 once (no bias in any Linear)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Gating logits + sigmoid (FP16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5️⃣ Optional mask – fuse (proj * gate * mask) in‑place via Triton
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – convert to FP16 (1 ⇒ keep, 0 ⇒ mask‑out)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()                 # B * N * N * H
        BLOCK = 1024                               # fits within a thread block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via custom Triton batched GEMM
    # --------------------------------------------------------------
    # Ensure tensors are contiguous for the kernel
    left  = left.contiguous()
    right = right.contiguous()
    out = _batched_gemm_triton(left, right)   # [B,N,N,H]  FP16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) over hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  FP16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
