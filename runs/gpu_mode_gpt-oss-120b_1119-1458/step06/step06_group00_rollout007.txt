```
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements,
    H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # compute mask offset: each hidden dimension element corresponds to exactly one mask entry
    # Since mask shape is [B,N,N], each mask entry broadcast over H hidden dims
    mask_offs = offs // H

    # Load
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out = left * tl.sigmoid(left_gate) * mask_val
    right_out = right * tl.sigmoid(right_gate) * mask_val

    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)

@triton.jit
def _mask_gate_fusion_kernel(left_ptr, right_ptr,
                             left_gate_ptr, right_gate_ptr,
                             mask_ptr,
                             total_elements, H,
                             BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # compute mask index
    mask_index = offs // H

    # load values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_index, mask=mask, other=1.0)

    # compute sigmoid
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    left_out = left * left_gate_sig * mask_val
    right_out = right * right_gate_sig * mask_val

    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)
```

"""
TriMul “outgoing” forward pass (no gradient) – highly‑optimised Triton implementation.

Algorithm
---------
1. Layer‑norm the input (float32 → stable statistics).
2. Cast to fp16 for the heavy compute.
3. Fuse *all* linear projections (left/right values and three gates) into a
   single matmul by concatenating their weight matrices:
   y = x @ W_concat.T → shape (B,N,N,5,hidden_dim)
   → splits into left_proj, left_gate, right_proj, right_gate, out_gate.
4. Apply sigmoid to the three gate tensors.
5. Apply the (optional) mask and the left/right gates in‑place with a tiny
   Triton kernel (mask is broadcast over the hidden dimension).  This kernel
   also computes the sigmoid, so the gating cost is hidden in the memory‑bound
   mask‐fusion step.
6. Contract over the “k” dimension:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched GEMM of shape (B·hidden, N, N) × (B·hidden, N, N)ᵀ.
   The highly‑optimised torch.bmm (Tensor‑core) is used here.
7. Second LayerNorm over the hidden dimension.
8. Apply the out‑gate and a final linear projection back to `dim`.
9. Cast the result back to float32.

Key performance tricks
----------------------
* **Projection fusion** – the input is read only once (instead of five separate
  matmuls) by concatenating the five weight matrices.
* **Mask‑gate fusion kernel** – a single Triton kernel does mask loading,
  sigmoid, and element‑wise multiplication for both left/right branches, saving
  extra memory passes.
* All heavy work is performed in fp16, exploiting Tensor‑cores on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional), sigmoid gates, and in‑place
# multiplication for the left‑ and right‑projections.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # int   total number of fp16 elements = B·N·N·H
    H,                 # int   hidden dimension (size of gating dim)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat index
    valid = offs < total_elements

    # Each mask element corresponds to H consecutive hidden entries.
    mask_offs = offs // H

    # Load everything (out‑of‑bounds entries get zero or 1.0 for the mask).
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # Sigmoid (numerically stable enough for fp16)
    left_gate_sig  = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    left_out  = left  * left_gate_sig  * mask_val
    right_out = right * right_gate_sig * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool / 0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)                               # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the computation
    # ------------------------------------------------------------------
    x = x.to(fp16)                                            # [B,N,N,dim] FP16
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 3️⃣ Concatenate projection + gate weights (5·hidden_dim × dim)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(device=device, dtype=fp16)
    w_left_gate   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    w_right_proj  = weights["right_proj.weight"].to(device=device, dtype=fp16)
    w_right_gate  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    w_out_gate    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    concat_w = torch.cat([w_left_proj,
                          w_left_gate,
                          w_right_proj,
                          w_right_gate,
                          w_out_gate], dim=0)                      # (5·H, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Single fused linear projection
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                               # (B·N·N, dim)
    y_flat = torch.nn.functional.linear(x_flat, concat_w)   # (B·N·N, 5·H)
    y = y_flat.view(B, N, N, 5, hidden_dim)                  # (B,N,N,5,H)

    # ------------------------------------------------------------------
    # 5️⃣ Split tensors
    # ------------------------------------------------------------------
    left_proj = y[..., 0, :]          # (B,N,N,H)
    left_gate = y[..., 1, :]          # (B,N,N,H) – raw (pre‑sigmoid)
    right_proj = y[..., 2, :]         # (B,N,N,H)
    right_gate = y[..., 3, :]         # (B,N,N,H) – raw
    out_gate = torch.sigmoid(y[..., 4, :])  # (B,N,N,H) – already sigmoid

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask + sigmoid gates (always via Triton; use all‑ones mask when nomask)
    # ------------------------------------------------------------------
    # In‑place: left_proj becomes final left, right_proj becomes final right.
    left = left_proj
    right = right_proj

    # Prepare mask (flattened to 1‑D).  If no mask, use an all‑ones tensor.
    if nomask:
        mask_flat = torch.ones(B * N * N, dtype=fp16, device=device)
    else:
        mask_flat = mask_tensor.to(fp16).reshape(-1).contiguous()

    # Ensure contiguous tensors for Triton
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_flat = mask_flat.contiguous()

    total = left.numel()
    BLOCK = 8192                      # ~8 K threads per block – fits H100 well
    grid = ((total + BLOCK - 1) // BLOCK,)

    _mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_flat,
        total_elements=total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
        num_warps=8,
        num_stages=1,
    )
    # Release temporarily‑used tensors
    del left_gate, right_gate, mask_flat

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction: batch GEMM (left @ rightᵀ) for each hidden channel
    # ------------------------------------------------------------------
    # left/right shape now = (B,N,N,H)
    # Rearrange to (B·H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * hidden_dim, N, N)
    B_mat = right_perm.view(B * hidden_dim, N, N)

    C = torch.bmm(A, B_mat.transpose(1, 2))               # (B·H, N, N)

    # Restore original layout: (B,N,N,H)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 9️⃣ Apply the final out‑gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                                  # element‑wise gating
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = torch.nn.functional.linear(out, to_out_w)       # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
