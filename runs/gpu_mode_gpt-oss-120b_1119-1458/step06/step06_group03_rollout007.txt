```
"""TriMul outgoing layer optimized forward pass.

The implementation fuses all five linear projections (left/right projections
and three gating linear layers) into a single half‑precision GEMM, then uses a
tiny Triton kernel to fuse element‑wise mask and gate multiplication.  The
heavy O(N³) contraction is performed as a batched matrix‑multiply using
torch.bmm (Tensor‑Core).  LayerNorms and gating are carried out in FP16,
and the final result is cast back to FP32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # hidden dimension (= size of last axis of x)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elts

    # Compute index into the mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=valid)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right: [B, N, N, H] (FP16)
    Returns out: [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # (B, H, N, N) view -> (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict  : name → torch.Tensor (float32)
        - config_dict   : includes ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm in FP32 (numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the forward
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)   # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Cast weights to FP16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse the five linear layers (left/right projections + gates)
    # ------------------------------------------------------------------
    # Stack weight matrices along the output‑feature dimension
    weight_concat = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    )  # shape (5*hidden_dim, dim)

    # Single GEMM (batch of B*N*N vectors) – returns everything at once
    proj = torch.nn.functional.linear(x, weight_concat)  # [B, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Slice the concatenated result and apply sigmoid to the gates
    # ------------------------------------------------------------------
    H = hidden_dim
    left            = proj[..., 0:H]
    right           = proj[..., H:2*H]
    left_gate_raw   = proj[..., 2*H:3*H]
    right_gate_raw  = proj[..., 3*H:4*H]
    out_gate_raw    = proj[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_raw)
    right_gate  = torch.sigmoid(right_gate_raw)
    out_gate    = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 6️⃣ (Optional) mask – fuse element‑wise multiplication with gate
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 and make it contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()            # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Gating and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 🔟 Return FP32 tensor (as required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  First LayerNorm in FP32 (numerical stability).
2️⃣  Cast to FP16 – the bulk of the computation runs in half‑precision.
3️⃣  Fuse the five linear layers (left/right projections + three gates) into a
    single GEMM by concatenating their weight matrices.  This eliminates five
    separate reads of the large input tensor.
4️⃣  Slice the concatenated output, apply sigmoid to the three gating parts.
5️⃣  If a mask is present, fuse element‑wise mask, gate and projection multiplication
    with a tiny Triton kernel (`fused_gate_mask_kernel`).  When `nomask=True` we
    simply multiply left/right with their gates.
6️⃣  Perform the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched matrix‑multiply using `torch.bmm` (Tensor‑Core).
7️⃣  Second LayerNorm (FP16), apply the output‑gate, and a final linear
    projection back to the original feature dimension.
8️⃣  Cast the result back to FP32 for the external harness.

The implementation requires at least one Triton kernel (the fused
gate‑mask kernel) and achieves a large speed‑up by reducing memory traffic
and kernel launches while keeping the heavy matrix‑multiplications on cuBLAS.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # hidden dimension (= size of last axis of x)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear indices
    valid = offs < total_elts

    # Broadcast mask over the hidden dimension.
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=valid)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    Shapes
    ------
    left, right : [B, N, N, H] (FP16)

    Returns
    -------
    out : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # Move hidden dimension into the batch axis for a single bmm call.
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (B·H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)         # (B·H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                                   # (B·H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – all subsequent heavy ops use half‑precision
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)   # [B, N, N, dim]   FP16

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (no bias in any Linear)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse the five linear layers into a single GEMM
    # ------------------------------------------------------------------
    weight_concat = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # One large matmul: (B·N·N, dim) @ (dim, 5·hidden_dim)ᵀ → (B·N·N, 5·hidden_dim)
    proj = torch.nn.functional.linear(x, weight_concat)   # [B, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Slice the fused projection and compute gating sigmoids
    # ------------------------------------------------------------------
    H = hidden_dim
    left          = proj[..., 0:H]
    right         = proj[..., H:2*H]
    left_gate_raw = proj[..., 2*H:3*H]
    right_gate_raw= proj[..., 3*H:4*H]
    out_gate_raw  = proj[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_raw)
    right_gate  = torch.sigmoid(right_gate_raw)
    out_gate    = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 6️⃣ (Optional) mask – fuse element‑wise mask & gate via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 and make it contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()               # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply output‑gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 🔟 Return FP32 (as required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
