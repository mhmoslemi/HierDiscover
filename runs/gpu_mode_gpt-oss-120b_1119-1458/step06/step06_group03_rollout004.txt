import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for fused gate and mask multiplication
@triton.jit
def fused_gate_mask_kernel(x_ptr, gate_ptr, mask_ptr,
                           total_elts: tl.int32,
                           hidden_dim: tl.int32,
                           BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # Global index for mask (broadcast across hidden_dim)
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    tl.store(x_ptr + offs, x * g * m, mask=mask)

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).
    This implementation fuses the five linear transforms that depend on the
    same input tensor by concatenating their weight matrices and performing a
    single matmul, then splits the result.  The mask‑gate multiplication for
    ``left`` and ``right`` is fused in a small Triton kernel.  The O(N³) tensor
    contraction is carried out as a batched GEMM using torch.bmm (Tensor‑Core).
    All heavy operations are performed in FP16; LayerNorms run in FP32
    (first) and FP16 (second).  The final result is returned in float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------
    # First LayerNorm (FP32) and cast to FP16
    # ------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32
    x = x_norm.to(torch.float16)  # FP16 for the heavy part

    # ------------------------------------------------------------
    # Combine the five linear projections (left_proj, left_gate,
    # right_proj, right_gate, out_gate) into one matrix multiply.
    # ------------------------------------------------------------
    # Cast all linear weights to FP16
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the out‑feature dimension → shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # [5*hidden_dim, dim]

    # Single linear projection
    combined = torch.nn.functional.linear(x, combined_weight)  # [B,N,N,5*hidden_dim]

    # ------------------------------------------------------------
    # Slice the combined result
    # ------------------------------------------------------------
    h = hidden_dim
    left_proj     = combined[..., :h]
    left_gate_pre = combined[..., h:2*h]
    right_proj    = combined[..., 2*h:3*h]
    right_gate_pre= combined[..., 3*h:4*h]
    out_gate_pre  = combined[..., 4*h:5*h]

    # Gating sigmoids
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------
    # Apply mask (if any) and element‑wise gate multiplication
    # ------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → FP16 and contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()
        BLOCK = 8192
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left_proj  = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ------------------------------------------------------------
    # Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)  # [B,N,N,hidden_dim] FP16

    # ------------------------------------------------------------
    # Second LayerNorm (FP16)
    # ------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------
    # Apply out_gate and final projection to original dim
    # ------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] FP16

    # ------------------------------------------------------------
    # Return FP32 tensor (as expected by the harness)
    # ------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold3) – fast fused forward pass.

Algorithm Overview
------------------
1️⃣  FP32 LayerNorm on the [B,N,N,dim] tensor → cast to FP16.
2️⃣  The five linear layers that share the same input (`x`) are fused:
    * left_proj , left_gate , right_proj , right_gate , out_gate
    → their weight matrices (each [hidden_dim, dim] or [dim, hidden_dim] for
      torch.nn.Linear) are concatenated along the output‑feature axis,
      yielding a single weight of shape (5*hidden_dim, dim).  A single
      FP16 matmul produces a tensor of shape [B,N,N,5*hidden_dim] that is
      then sliced into the five needed tensors.
    * The gating tensors are passed through sigmoid.
3️⃣  If a mask is present we fuse element‑wise multiplication
    (tensor * gate * mask) for both left and right tensors in a tiny
    Triton kernel (`fused_gate_mask_kernel`).  In the “nomask” case the
    multiplication is performed with a regular PyTorch element‑wise op.
4️⃣  The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is realized as a batched GEMM: each hidden channel becomes a
    separate batch element, and we call `torch.bmm` (Tensor‑Core).
5️⃣  A second FP16 LayerNorm, gating by `out_gate`, and the final linear
    projection back to `dim` complete the forward pass.
6️⃣  The result is cast back to FP32 for the evaluation harness.

All heavy arithmetic is performed in FP16; only the first LayerNorm runs
in FP32 for numerical stability.  The implementation uses at least one
Triton kernel (mask‑gate fusion) as required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden
# dimension `hidden_dim`.  `total_elts` = B * N * N * hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # hidden dimension (= size of last axis of x)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤8192 on H100)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear indices
    valid = offs < total_elts

    # The mask is broadcast over `hidden_dim` entries.
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Batched GEMM via torch.bmm (cuBLAS/Tensor‑Core).  This implements the
# O(N³) contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (FP16).

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, H] (FP16) where H = hidden_dim.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – every hidden channel becomes a separate batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N).T → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` (int), ``hidden_dim`` (int),
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16 for the heavy ops
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # still float32
    x = x_norm.to(torch.float16)   # FP16 from here on

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers that depend on the same input:
    #    left_proj, left_gate, right_proj, right_gate, out_gate.
    # ------------------------------------------------------------------
    # Cast all five weight matrices to FP16 once.
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature axis → shape (5*hidden_dim, dim)
    # (torch.nn.Linear stores weights as (out_features, in_features))
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # [5*hidden_dim, dim]

    # Single matmul produces all intermediate tensors at once.
    # Result shape: [B, N, N, 5*hidden_dim] (FP16)
    combined = torch.nn.functional.linear(x, combined_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Slice the combined tensor into the five components.
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj     = combined[..., :h]                # [B,N,N,h]
    left_gate_pre = combined[..., h:2*h]            # [B,N,N,h]
    right_proj    = combined[..., 2*h:3*h]          # [B,N,N,h]
    right_gate_pre= combined[..., 3*h:4*h]          # [B,N,N,h]
    out_gate_pre  = combined[..., 4*h:5*h]          # [B,N,N,h]

    # Gate activations (sigmoid)
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and fuse element‑wise multiplication:
    #    left = left_proj * left_gate * mask
    #    right = right_proj * right_gate * mask
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 and ensure contiguity.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                     # B * N * N * hidden_dim
        BLOCK = 8192                                       # tile size (tunable)
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # Left side
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Right side
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication
        left_proj  = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # [B,N,N,hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Gating by out_gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
