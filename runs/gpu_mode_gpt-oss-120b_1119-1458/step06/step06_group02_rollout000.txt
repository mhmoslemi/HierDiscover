"""
TriMul outgoing forward pass (AlphaFold‑3) – highly‑optimized Triton + PyTorch.

Algorithm
---------
1️⃣  LayerNorm on the raw 4‑D input (float32) → stabilises statistics.
2️⃣  Cast to fp16 for the bulk of the computation.
3️⃣  Linear projections (`left_proj`, `right_proj`) and three gate projections
    (`left_gate`, `right_gate`, `out_gate`) are performed with fp16 weights.
4️⃣  If a mask is present we fuse element‑wise multiplication of the left/right
    tensors with their corresponding gates and the mask in a single Triton
    kernel (`mask_gate_fuse_kernel`).  This eliminates three separate memory
    passes.
5️⃣  The cubic contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is implemented as a batched GEMM after reshaping the tensors from
    [B,N,N,H] → [B·H, N, N] – each hidden channel becomes a separate batch.
    cuBLAS/Tensor‑Cores on the H100 compute this extremely fast.
6️⃣  A second LayerNorm (fp16) over the hidden dimension.
7️⃣  Multiply by the output gate and finally project back to the original
    dimension with a linear layer.
8️⃣  Cast the result back to float32 (the harness expects float32).

Only the forward pass is implemented – no autograd is needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and element‑wise product.
# x and gate have shape (B, N, N, H)  → flattened stride H (innermost).
# mask has shape (B, N, N) and is broadcast over H.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (flattened B·N·N)
    hidden_dim,     # i32   : H
    size,           # i64   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)    # linear indices for this block
    mask = offs < size                           # guard against OOB

    # Index into the mask (broadcast over hidden_dim)
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=mask, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=mask, other=1.0)  # mask=1 outside valid range

    # fused multiplication
    y = x * g * m
    tl.store(x_ptr + offs, y, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using a batched fp16 GEMM.
    Input/Output shape: [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # (B, N, N, H) → (B·H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)^T  → (B·H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # Back to original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input tensor  [B, N, N, dim] (float32)
        torch.Tensor,                # mask tensor   [B, N, N] (bool/float or None)
        Dict[str, torch.Tensor],     # weights dict (float32)
        Dict[str, Any]               # config dict {dim, hidden_dim, nomask?}
    ]
) -> torch.Tensor:
    """Forward pass of the outgoing TriMul module (no autograd)."""
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy region
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast all model weights to fp16 (once)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 5️⃣ Gate tensors (sigmoid) – fp16
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Mask & gate fusion (if present) via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and make it contiguous  (shape B·N·N)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()          # B·N·N·H
        BLOCK = 1024                         # tile size, fits H100 warp schedulers
        grid = ( (total_elems + BLOCK - 1) // BLOCK, )

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: simple in‑place elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Cubic contraction via batched fp16 GEMM (cuBLAS/Tensor‑Core)
    # -----------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # -----------------------------------------------------------------
    # 9️⃣ Final gating and projection back to the original dim
    # -----------------------------------------------------------------
    out = out * out_gate                      # elementwise gating (fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
