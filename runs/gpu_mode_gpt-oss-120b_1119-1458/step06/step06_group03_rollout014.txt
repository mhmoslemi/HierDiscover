def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B,N,_,H = left.shape
    a = left.permute(0,3,1,2).reshape(B*H, N, N)
    b = right.permute(0,3,1,2).reshape(B*H, N, N)
    c = torch.bmm(a, b.transpose(1,2)) # shape B*H,N,N
    out = c.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

"""
TriMul (outgoing) – fused implementation.

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input (FP32 → FP16).
2️⃣  Fuse the four Linear+Gate projections (left/right proj & gates) and the
    out‑gate in a *single* linear layer (weight shape 5·hidden_dim × dim).  The
    result is split into 5 tensors and the gates are passed through sigmoid.
3️⃣  If a mask is supplied we fuse the three multiplicative factors
    (proj * gate * mask) with a tiny Triton kernel (`fused_gate_mask_kernel`);
    otherwise we simply multiply the projection by its gate.
4️⃣  The heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM using `torch.bmm` (Tensor‑Core on H100).
5️⃣  A second Layer‑Norm (hidden dimension) is applied, followed by the out‑gate.
6️⃣  The hidden‑dim tensor is projected back to the original `dim` with one
    more linear layer.
7️⃣  Cast back to FP32 for the harness.

The kernel fuses the element‑wise mask‑gate‑projection step, while the large
matrix‑multiplications stay in cuBLAS, giving a fast mixed‑precision forward
pass without autograd overhead.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) in‑place.
# The mask is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated (proj * gate * mask)
    gate_ptr,             # *fp16 : gate tensor (same shape as x)
    mask_ptr,             # *fp16 : mask tensor (broadcast over hidden dim)
    total_elts: tl.int32, # total number of elements in x/gate (B·N·N·H)
    hidden_dim: tl.int32, # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # mask index: each hidden_dim entries share the same mask element
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H] (FP16)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # Collapse batch and hidden dimensions: each hidden channel becomes its own batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 / bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32, then FP16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32
    x = x.to(torch.float16).contiguous()  # FP16 for the heavy part

    # --------------------------------------------------------------
    # 2️⃣ Cast all model weights to FP16 (once per call)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fused linear for proj & gate + out‑gate
    #    Input shape: [B, N, N, dim]  -->  output shape [B, N, N, 5·hidden_dim]
    # --------------------------------------------------------------
    combined_weight = torch.cat(
        [
            left_proj_w,   # (H, D)
            right_proj_w,  # (H, D)
            left_gate_w,   # (H, D)
            right_gate_w,  # (H, D)
            out_gate_w,    # (H, D)
        ],
        dim=0,  # concatenate on output‑feature dimension
    )  # shape (5*hidden_dim, dim)

    # Linear works on the last dimension; no bias in these layers
    lin_out = torch.nn.functional.linear(x, combined_weight)  # [B, N, N, 5·H]

    # --------------------------------------------------------------
    # 4️⃣ Split and apply sigmoids to the gate tensors
    # --------------------------------------------------------------
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        lin_out, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Ensure contiguous tensors for the Triton kernel (if needed)
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Mask + gate fusion (optional)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                # B·N·N·H
        BLOCK = 1024                                   # tile size
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # [B, N, N, H] (FP16)

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden dimension)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final projection back to ``dim``
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
