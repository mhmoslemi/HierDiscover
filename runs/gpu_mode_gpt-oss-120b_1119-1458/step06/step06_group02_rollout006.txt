def apply_gate_mask_kernel(
    x_ptr, gate_ptr, mask_ptr, hidden_dim, size,
    BLOCK: tl.constexpr,
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < size

    mask_offs = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=active, other=0.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)

"""
TriMul (outgoing) forward pass – highly‑optimized Triton + PyTorch implementation.

Algorithm
---------
1️⃣  Layer‑Norm over the input channels (float32 for numerical stability).
2️⃣  Cast to fp16 – the heavy part of the kernel runs in half‑precision.
3️⃣  **Fused linear projection**: a single matmul produces
        left_proj, left_gate, right_proj, right_gate and out_gate
    in one pass (weights are concatenated along the output dimension).
4️⃣  **Fused gate + mask** (when a mask is present) using a Triton kernel.
    The kernel applies the sigmoid non‑linearity to the gate, multiplies the
    projection, and optionally the mask – all in‑place without extra temporaries.
5️⃣  O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    realized as a batched GEMM (`torch.bmm`) after reshaping to
    [B·H, N, N] → [B·H, N, N] @ [B·H, N, N]ᵀ.
6️⃣  Second Layer‑Norm on the hidden dimension.
7️⃣  Apply the output gate → linear projection back to the original `dim`.
8️⃣  Cast the result back to float32 (the harness expects float32).

Only the forward pass is implemented – no autograd support is needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def gate_mask_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right projection)
    gate_ptr,       # *fp16 : raw gate tensor (pre‑sigmoid)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N), broadcast over hidden dim
    hidden_dim,     # i32   : size of the hidden dimension H
    size,           # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time block size (≤1024)
):
    """In‑place: x ← x * sigmoid(gate) * mask  (mask broadcast over hidden dim)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear indices for this block
    active = offs < size                           # bounds check

    # mask index = element index divided by hidden_dim (since hidden is the innermost dim)
    mask_offs = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # sigmoid on the gate (computed in fp16)
    g = 1.0 / (1.0 + tl.exp(-g))

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using a batched GEMM.
    left/right: [B, N, N, H] (fp16)
    Returns:    [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # Shape to [B*H, N, N] where each hidden channel is its own batch.
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))

    # Reshape back to original layout.
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) or None
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain “dim” and “hidden_dim”; optional “nomask”

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast and concatenate all projection / gate weights (fp16)
    # -----------------------------------------------------------------
    # Shapes:
    #   left_proj, left_gate, right_proj, right_gate, out_gate -> each [hidden_dim, dim]
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along output dimension: shape [(5*hidden_dim), dim]
    combined_weight = torch.cat(
        [left_proj_w,
         left_gate_w,
         right_proj_w,
         right_gate_w,
         out_gate_w],
        dim=0,
    )  # fp16

    # -----------------------------------------------------------------
    # 4️⃣ Single linear projection producing all required tensors
    # -----------------------------------------------------------------
    # combined: [B, N, N, 5*hidden_dim] (fp16)
    combined = torch.nn.functional.linear(x, combined_weight)  # bias=False

    # -----------------------------------------------------------------
    # 5️⃣ Slice the combined tensor
    # -----------------------------------------------------------------
    H = hidden_dim
    left_proj      = combined[..., 0:H]                 # [B, N, N, H]
    left_gate_raw  = combined[..., H:2*H]               # pre‑sigmoid
    right_proj     = combined[..., 2*H:3*H]              # [B, N, N, H]
    right_gate_raw = combined[..., 3*H:4*H]              # pre‑sigmoid
    out_gate_raw   = combined[..., 4*H:5*H]              # pre‑sigmoid

    # output gate (sigmoid) – needed after the second LayerNorm
    out_gate = torch.sigmoid(out_gate_raw)               # [B, N, N, H]

    # -----------------------------------------------------------------
    # 6️⃣ Apply gate (and mask) to left and right projections
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → fp16 contiguous flattened view
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()                 # B·N·N·H
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # In‑place: left_proj ← left_proj * sigmoid(left_gate_raw) * mask
        gate_mask_fuse_kernel[grid](
            left_proj,
            left_gate_raw,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # In‑place: right_proj ← right_proj * sigmoid(right_gate_raw) * mask
        gate_mask_fuse_kernel[grid](
            right_proj,
            right_gate_raw,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel left_proj/right_proj already contain the gated values.
        # No separate tensors for left_gate/right_gate are kept.
    else:
        # No mask → simple element‑wise multiplication with sigmoid‑gated values.
        left_proj = left_proj * torch.sigmoid(left_gate_raw)
        right_proj = right_proj * torch.sigmoid(right_gate_raw)

    # Alias for clarity (these tensors are now the gated, masked values)
    left = left_proj   # [B, N, N, H] (fp16)
    right = right_proj # [B, N, N, H] (fp16)

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H] (fp16)

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # -----------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to original dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16)  # shape [dim, hidden_dim]
    )  # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the caller and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
