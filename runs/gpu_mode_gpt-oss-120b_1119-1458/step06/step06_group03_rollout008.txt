"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D pairwise tensor (FP32 → FP16 for speed).
2️⃣ Fuse the two linear projections (proj + gate) for the *left* and *right*
   branches by concatenating their weight matrices.  A single matmul per
   branch produces the projection and the pre‑gate values in one go.
3️⃣ Apply sigmoid to the gate tensors.
4️⃣ If a mask is present we fuse ``tensor * gate * mask`` in‑place with a tiny
   Triton kernel (`fused_gate_mask_kernel`).  This eliminates two extra
   element‑wise passes.
5️⃣ The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed with a custom Triton batched‑GEMM kernel (`trimul_outgoing_gemm_kernel`)
   that treats the hidden dimension as an extra batch axis.  The kernel
   uses block‑wise loads and `tl.dot` (FP16 × FP16 → FP32 accumulation) so
   that Tensor‑Core instructions are fully utilised on H100.
6️⃣ A second Layer‑norm (over the hidden dimension) normalises the result,
   then the gating tensor `out_gate` (sigmoid(linear(x))) is applied.
7️⃣ The final linear projection brings the hidden dimension back to ``dim``.
8️⃣ Output is cast to FP32 for the harness.

All heavy arithmetic runs in FP16 (except the accumulators which are FP32),
while the two LayerNorms stay in FP16 (the first starts from FP32 for
numerical stability).  The implementation works for both masked and
un‑masked inputs, and for any batch size ≤2, sequence length up to 1024,
and hidden dimensions 128/384.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1: fused (tensor * gate * mask) – inplace FP16 multiplication.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,           # *fp16  : tensor to be updated (total_elts)
    gate_ptr,        # *fp16  : gate tensor, same shape as x
    mask_ptr,        # *fp16  : mask tensor (broadcast over hidden_dim)
    total_elts: tl.int32,   # total number of scalars in x / gate
    hidden_dim: tl.int32,   # size of hidden dimension
    BLOCK: tl.constexpr,    # tile size (<= 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elts

    # mask is broadcast across hidden_dim entries
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the TriMul contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# Tensors are stored as [B, N, N, H] (C‑contiguous, H fastest).
# The kernel treats (B*H) as the batch dimension, and performs a
# standard 2‑D GEMM (M=N, K=N, N=N) for every hidden channel.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_gemm_kernel(
    left_ptr,   # *fp16 : left tensor  [B,N,N,H]
    right_ptr,  # *fp16 : right tensor [B,N,N,H]
    out_ptr,    # *fp16 : output tensor [B,N,N,H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    # strides for left
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right
    stride_rb, stride_rj, stride_rk, stride_rh,
    # strides for out
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_bn = tl.program_id(0)        # combined batch (B) * hidden (H) index
    pid_i  = tl.program_id(1)        # tile row   (i)
    pid_j  = tl.program_id(2)        # tile col   (j)

    # Decompose pid_bn into batch and hidden indices
    b = pid_bn // H
    h = pid_bn % H

    # Tile start offsets
    i_off = pid_i * BLOCK_M
    j_off = pid_j * BLOCK_N

    # Range vectors
    offs_i = i_off + tl.arange(0, BLOCK_M)
    offs_j = j_off + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers (including batch & hidden offsets)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # Accumulator (FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the contracted dimension K (= N)
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k_off = k_iter * BLOCK_K
        cur_k = k_off + offs_k
        mask_k = cur_k < N

        # ----- Load left tile : shape (BLOCK_M, BLOCK_K) -----
        left_ptrs = left_base \
                    + (offs_i[:, None] * stride_li) \
                    + (cur_k[None, :] * stride_lk)   # (i,k)
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- Load right tile (transposed) : shape (BLOCK_K, BLOCK_N) -----
        # Note: right is indexed as (b, j, k, h); we need (k, j) order.
        right_ptrs = right_base \
                     + (cur_k[:, None] * stride_rk) \
                     + (offs_j[None, :] * stride_rj)   # (k,j)
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # ----- Block dot product (FP16 × FP16 → FP32) -----
        acc += tl.dot(a, b_tile)   # (BLOCK_M, BLOCK_N)

    # ----- Store the result -----
    out_ptrs = out_base \
               + (offs_i[:, None] * stride_oi) \
               + (offs_j[None, :] * stride_oj)   # (i,j)
    out_tile = acc.to(tl.float16)
    tl.store(out_ptrs,
             out_tile,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Trimul GEMM kernel.
# ----------------------------------------------------------------------
def _trimul_outgoing_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction with a Triton GEMM kernel.
    Arguments:
        left, right : torch tensors of shape [B, N, N, H] (dtype fp16, contiguous)
    Returns:
        out : torch tensor of shape [B, N, N, H] (dtype fp16, contiguous)
    """
    B, N, _, H = left.shape
    device = left.device

    # Allocate output
    out = torch.empty_like(left)

    # Strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes – tuned for H100 Tensor‑Core
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions: (batch*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_outgoing_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block works well on H100
    )
    # Synchronise before returning (optional – Triton kernels are synchronous with the stream)
    return out


# ----------------------------------------------------------------------
# Entry point for the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict mapping parameter names to tensors (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; may also contain ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm in FP32 (more stable) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (FP32)
    x = x_norm.to(torch.float16)        # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for *left* and *right* branches
    # ------------------------------------------------------------------
    # Concatenate projection and gate weight matrices (out_features stacked)
    left_proj_w   = weights["left_proj.weight"]   # (hidden_dim, dim)
    left_gate_w   = weights["left_gate.weight"]   # (hidden_dim, dim)
    combined_left_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)   # (2*hidden_dim, dim)

    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    combined_right_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # (2*hidden_dim, dim)

    # Single matmul per side (produces [B,N,N,2*hidden_dim])
    left_combined = torch.nn.functional.linear(x, combined_left_w)   # fp16
    right_combined = torch.nn.functional.linear(x, combined_right_w)

    # Split projection & gate tensors
    left = left_combined[..., :hidden_dim]                     # proj
    left_gate_pre = left_combined[..., hidden_dim:]            # pre‑gate
    right = right_combined[..., :hidden_dim]
    right_gate_pre = right_combined[..., hidden_dim:]

    # Apply sigmoid to the gate tensors
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 3️⃣ Out‑gate (independent linear) – computed early for later use
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ (Optional) mask & gate fusion via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()                # B * N * N * hidden_dim
        BLOCK = 1024
        grid_mask = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid_mask](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid_mask](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # Simple element‑wise multiplication when mask is absent
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (blocked GEMM via Triton)
    # ------------------------------------------------------------------
    out = _trimul_outgoing_gemm(left, right)   # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
