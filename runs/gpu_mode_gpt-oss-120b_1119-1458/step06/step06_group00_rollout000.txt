   logits.  The gate is activated with a sigmoid.
3️⃣ Optional mask‑gate fusion: a Triton kernel multiplies
   proj × gate × mask for both left and right tensors in‑place.
   When no mask is supplied a simple point‑wise multiply is used.
4️⃣ The core O(N³) contraction is performed as a batched matrix‑multiplication:
   for each batch and hidden channel we compute  L @ Rᵀ  where
   L = left[b,…,h] and R = right[b,…,h].  This is an efficient
   Tensor‑Core GEMM (torch.bmm) on tensors of shape (B·hidden_dim, N, N).
5️⃣ A second Layer‑Norm on the hidden dimension, followed by the out‑gate.
6️⃣ Final linear projection back to ``dim`` and cast to FP32.

Only the mask‑gate fusion is implemented in Triton, satisfying the
“at least part of the operations in a kernel’’ requirement while keeping
the heavy N³ contraction on cuBLAS/Tensor‑Core for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with the per‑head gates.
#   left  <- left_proj  * left_gate  * mask
#   right <- right_proj * right_gate * mask
# All tensors are contiguous FP16.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]   – in‑place output (projected left)
    right_ptr,         # *fp16  [B,N,N,H]   – in‑place output (projected right)
    left_gate_ptr,     # *fp16  [B,N,N,H]   – gate logits (sigmoid already applied)
    right_gate_ptr,    # *fp16  [B,N,N,H]   – gate logits (sigmoid already applied)
    mask_ptr,          # *fp16  [B,N,N]    – broadcasted mask (0/1)
    total_elements,    # i64   total number of elements in left/right (B·N·N·H)
    H,                 # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    """
    Apply mask (if any) and the sigmoid gates to the left/right projections
    in‑place.  The mask is broadcast over the hidden dimension; each mask
    element corresponds to ``H`` consecutive entries in the left/right tensors.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat index
    valid = offs < total_elements

    # Compute the index of the mask element that corresponds to this hidden‑dim entry.
    mask_offs = offs // H

    # Load data (out‑of‑bounds entries are ignored via the ``mask`` argument)
    left = tl.load(left_ptr + offs,      mask=valid, other=0.0)
    right = tl.load(right_ptr + offs,    mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Load mask: for OOB we use ``1.0`` (neutral element)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # Fuse: (proj * gate) * mask
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # Store the results back in‑place
    tl.store(left_ptr + offs,  left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor
            Tensor of shape [B, N, N, dim] (float32) on the target device.
        - mask_tensor  : torch.Tensor
            Tensor of shape [B, N, N] (bool/float) – may be all‑ones when
            ``nomask`` is True.
        - weights_dict : mapping name → torch.Tensor (float32)
            Must contain the same keys as the reference ``TriMul`` module.
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32), same device as input.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B,N,N,dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy path
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16
    # ------------------------------------------------------------------
    # ---- combine left projection + left gate --------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)  # (H, dim)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)  # (H, dim)
    # concatenate along the output dimension → (2·H, dim)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # shape (2*H, dim)

    # ---- combine right projection + right gate -------------------------------
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # out‑gate (separate)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    # final linear (hidden → dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (proj + gate fused)
    # ------------------------------------------------------------------
    # left_comb: (B,N,N,2*H)
    left_comb = F.linear(x, left_comb_w)   # FP16
    # right_comb: (B,N,N,2*H)
    right_comb = F.linear(x, right_comb_w)  # FP16

    # Split projection and gate halves
    left_proj = left_comb[..., :hidden_dim]          # (B,N,N,H)
    left_gate = left_comb[..., hidden_dim:]          # (B,N,N,H)
    right_proj = right_comb[..., :hidden_dim]        # (B,N,N,H)
    right_gate = right_comb[..., hidden_dim:]        # (B,N,N,H)

    # Activate gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask‑gate fusion (Triton) – in place on left/right proj.
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 (0.0 / 1.0) and make everything contiguous.
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_elements = left_proj.numel()                      # B·N·N·H
        BLOCK = 1024                                            # ~1 K threads per block
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left_proj,
            right_ptr=right_proj,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total_elements,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates are no longer required
        del left_gate, right_gate, mask_fp16
    else:
        # Simple element‑wise fusion without a mask.
        left_proj.mul_(left_gate)
        right_proj.mul_(right_gate)
        # Release gate tensors
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Compute out‑gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction: batched GEMM L @ Rᵀ  (FP16, Tensor‑Core)
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape  # H == hidden_dim

    # Rearrange to (B, H, N, N) and make contiguous – this copy is cheap compared
    # to the O(N³) work and enables an efficient batched GEMM.
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse batch & head dimensions → (B·H, N, N)
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    # Batched matrix multiply: for each (b, h) compute A @ Bᵀ
    out_batch = torch.bmm(A, B_.transpose(1, 2))  # (B·H, N, N)

    # Restore original layout → (B, N, N, H)
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate               # element‑wise gating (FP16)
    out = F.linear(out, to_out_w)      # (B,N,N,dim)  FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
