def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    # config
    dim = config["dim"]      # original channel dimension
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. First LayerNorm (float32)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # cast to fp16
    x = x_norm.to(torch.float16)

    B, N, _, _ = x.shape

    # 2. Fused linear projection for all 5 pathways
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # cast all to fp16 (they're usually fp32)
    weight_concat = torch.cat([
        w_left, w_right, w_left_gate, w_right_gate, w_out_gate
    ], dim=0).to(torch.float16)   # shape (5*hidden_dim, dim)

    # flatten x to (B*N*N, dim )
    x_flat = x.view(-1, dim)
    fused = torch.nn.functional.linear(x_flat, weight_concat)  # (B*N*N, 5*hidden_dim)
    fused = fused.view(B, N, N, 5 * hidden_dim)

    # split
    left_proj_raw  = fused[..., 0:hidden_dim]
    right_proj_raw = fused[..., hidden_dim:2*hidden_dim]
    left_gate_raw  = fused[..., 2*hidden_dim:3*hidden_dim]
    right_gate_raw = fused[..., 3*hidden_dim:4*hidden_dim]
    out_gate_raw   = fused[..., 4*hidden_dim:5*hidden_dim]

    # free fused after splitting partially? keep separate to free memory after gating
    del fused

    # 3. Sigmoid on gating paths
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # free raw gates maybe keep for possible reuse
    del left_gate_raw, right_gate_raw, out_gate_raw

    # 4. Prepare mask (fp16) or all-ones
    if nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=x.device)
    else:
        # mask_tensor may be bool or float
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()

    # 5. Reorder + gating + mask fusion via Triton kernel
    total = B * hidden_dim * N * N
    BLOCK = 1024  # can be tuned
    grid = (triton.cdiv(total, BLOCK),)

    left_out  = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=x.device)
    right_out = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=x.device)

    # kernel: each thread writes one element of left_out / right_out

        right_out,
        B, N, hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # free large intermediates
    del left_proj_raw, right_proj_raw, left_gate, right_gate

    # 6. Heavy O(N^3) contraction using batched GEMM
    # left_out, right_out shape (B*H,N,N)
    # Compute left_out @ right_out^T for each batch
    out_flat = torch.bmm(left_out, right_out.transpose(1, 2))   # (B*H,N,N)

    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    del left_out, right_out, out_flat

    out_norm = out_norm * out_gate

    # 9. Final linear projection to original dim
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32) and cast to fp16 for the compute‑heavy part.
2. Fuse the five linear projections/gates (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single matmul by concatenating the weight matrices (shape
   (5*hidden_dim, dim)).  This reduces the number of reads of the large input
   tensor from five to one.
3. Split the fused result, apply a sigmoid on the three gating tensors.
4. Fuse the element‑wise multiplication (proj * gate * mask) *and* the layout
   transposition needed for the O(N³) contraction with a Triton kernel.
   The kernel reads the raw projection and gate tensors of shape (B,N,N,hidden_dim),
   multiplies them together (optionally applying a mask) and writes the result
   directly into a contiguous buffer of shape (B*hidden_dim, N, N)
   required for the batched GEMM.
5. Heavy O(N³) contraction is performed as a batched matrix multiplication:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  ⇔
   left_out @ right_outᵀ for each (b,h) pair.
6. Reshape the result back to (B,N,N,hidden_dim) and apply the second LayerNorm.
7. Multiply by the out‑gate and project back to the original channel dimension.
8. Cast the final output to float32 (the harness expectation).

The implementation keeps all large tensors in fp16, uses a single fused
projection, and replaces the two costly permute+copy operations with a single
Triton kernel, giving a large reduction in memory traffic and runtime.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def reorder_mul_kernel(
    proj_ptr,     # *fp16  projection tensor   (B,N,N,H)
    gate_ptr,     # *fp16  gate tensor        (B,N,N,H)
    mask_ptr,     # *fp16  mask tensor        (B*N*N)
    out_ptr,      # *fp16  output tensor      (B*H,N,N)
    B: tl.int32,  # batch size
    N: tl.int32,  # sequence length
    H: tl.int32,  # hidden dimension
    BLOCK: tl.constexpr,
):
    """
    Fuse (proj * gate * mask) and rearrange from [B,N,N,H] to [B*H,N,N].

    Each program instance processes a contiguous block of elements in the
    output tensor.  The mapping from output index to input indices is:

        out_idx = (b*H + h) * N * N + i * N + k
        src_idx = ((b * N + i) * N + k) * H + h
        mask_idx = (b * N + i) * N + k          (broadcast mask)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * H * N * N
    mask = offs < total

    # decompose linear output index
    bh = offs // (N * N)                 # bh = b*H + h
    ik = offs % (N * N)                  # ik = i*N + k
    i = ik // N
    k = ik % N

    b = bh // H
    h = bh % H

    # source offsets (same for proj and gate)
    src_offset = ((b * N + i) * N + k) * H + h
    # mask offset (mask is broadcast across H)
    mask_offset = (b * N + i) * N + k

    proj_val = tl.load(proj_ptr + src_offset, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + src_offset, mask=mask, other=0.0)
    # default mask = 1.0 (all‑ones buffer is passed when no mask is used)
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=1.0)

    out_val = proj_val * gate_val * mask_val
    tl.store(out_ptr + offs, out_val, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Args:
        data: (input_tensor, mask_tensor, weights, config)
            input_tensor : [B, N, N, dim]  (float32)
            mask_tensor  : [B, N, N]      (bool/float)
            weights      : dict of model weights
            config       : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32, shape [B,N,N,dim]
    x = x_norm.to(torch.float16)  # fp16 for the heavy part

    B, N, _, _ = x.shape  # x has shape [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Fuse all five linear maps into one matmul
    # -----------------------------------------------------------------
    # Cast all required weight matrices to fp16 once
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*H, dim)

    # Flatten input to (B*N*N, dim) and apply the fused matmul
    x_flat = x.view(-1, dim)  # contiguous view
    fused = torch.nn.functional.linear(x_flat, weight_concat)  # (B*N*N, 5*H)
    fused = fused.view(B, N, N, 5 * hidden_dim)  # [B,N,N,5*H]

    # -----------------------------------------------------------------
    # 3️⃣ Split the fused tensor and apply the sigmoids on the gates
    # -----------------------------------------------------------------
    left_proj_raw  = fused[..., 0:hidden_dim]
    right_proj_raw = fused[..., hidden_dim : 2 * hidden_dim]
    left_gate_raw  = fused[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate_raw = fused[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate_raw   = fused[..., 4 * hidden_dim : 5 * hidden_dim]

    # Free fused to release memory
    del fused

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # Release raw gate tensors
    del left_gate_raw, right_gate_raw, out_gate_raw

    # -----------------------------------------------------------------
    # 4️⃣ Prepare mask (fp16) – a tensor of ones when no mask is used
    # -----------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        # mask may be bool or int/float; ensure fp16 and contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()

    # Flatten mask for the kernel (linear index)
    mask_fp16 = mask_fp16.view(-1)

    # -----------------------------------------------------------------
    # 5️⃣ Fuse element‑wise multiplication (proj * gate * mask) and
    #     layout transposition (B,N,N,H) → (B*H,N,N) via Triton
    # -----------------------------------------------------------------
    total_elems = B * hidden_dim * N * N
    BLOCK = 1024  # threads per block; can be tuned
    grid = (triton.cdiv(total_elems, BLOCK),)

    left_out  = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=device)
    right_out = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=device)

    # Left side kernel launch
    reorder_mul_kernel[grid](
        left_proj_raw,
        left_gate,
        mask_fp16,
        left_out,
        B,
        N,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # Right side kernel launch
    reorder_mul_kernel[grid](
        right_proj_raw,
        right_gate,
        mask_fp16,
        right_out,
        B,
        N,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # Release intermediate tensors that are no longer needed
    del left_proj_raw, right_proj_raw, left_gate, right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    # left_out/right_out have shape (B*H, N, N)
    contracted = torch.bmm(left_out, right_out.transpose(1, 2))  # (B*H, N, N)

    # Restore the 4‑D shape expected by the later layers
    out = contracted.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Release buffers that held the large transposed data
    del left_out, right_out, contracted

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) over the hidden dimension
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 8️⃣ Apply the output gate
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate  # element‑wise gating

    # -----------------------------------------------------------------
    # 9️⃣ Final projection back to original channel dimension
    # -----------------------------------------------------------------
    out_final = torch.nn.functional.linear(
        out_norm, weights["to_out.weight"].to(torch.float16)
    )  # (B,N,N,dim), fp16

    # -----------------------------------------------------------------
    # 10️⃣ Cast back to float32 (required by harness) and return
    # -----------------------------------------------------------------
    return out_final.to(torch.float32)
