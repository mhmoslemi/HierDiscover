def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    b, n, _, _ = input_tensor.shape
    device = input_tensor.device

    # First LayerNorm (float32)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)  # convert to fp16

    # Flatten
    x_flat = x_norm.reshape(-1, dim)  # [B*N*N, dim]

    # Concatenate weights for proj, gate, out_gate
    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)

    # Concatenate along out_features dimension
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape [5*H, dim]

    # Linear projection: resulting shape [M, 5*H]
    proj_gate = torch.nn.functional.linear(x_flat, combined_weight)  # fp16

    # Reshape back
    proj_gate = proj_gate.view(b, n, n, -1)  # shape [B,N,N,5*H]

    # Split
    H = hidden_dim
    left_proj = proj_gate[..., 0:H]
    left_gate_raw = proj_gate[..., H:2*H]
    right_proj = proj_gate[..., 2*H:3*H]
    right_gate_raw = proj_gate[..., 3*H:4*H]
    out_gate_raw = proj_gate[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # Mask
    if not nomask:
        # mask shape [B,N,N], unsqueeze to broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # O(N³) contraction (batched GEMM)
    out = _batched_matmul_transpose_fp16(left, right)

    # Second LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device, dtype=torch.float16),
    )

    # Apply out_gate and final projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(device, dtype=torch.float16))

    # cast to FP32
    return out.to(torch.float32)

def _gate_mask_kernel(
    proj_a_ptr, gate_a_ptr, proj_b_ptr, gate_b_ptr,
    mask_ptr,  # shape [B, N, N], broadcast over H
    out_a_ptr, out_b_ptr,
    B, N, H,
    stride_proj_a_b, stride_proj_a_i, stride_proj_a_j, stride_proj_a_h,
    stride_gate_a_b, stride_gate_a_i, stride_gate_a_j, stride_gate_a_h,
    stride_proj_b_b,...,
    stride_gate_b_*,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_a_b, stride_out_a_i, stride_out_a_j, stride_out_a_h,
    stride_out_b_b, stride_out_b_i, stride_out_b_j, stride_out_b_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Mask loading (scalar)
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=1.0)  # default 1 if no mask
    # mask_tile shape (BLOCK_I, BLOCK_J)

    # Loop over hidden dim
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load proj and gate for left
        proj_a_ptrs = proj_a_ptr + pid_b * stride_proj_a_b + offs_i[:, None, None] * stride_proj_a_i + offs_j[None, :, None] * stride_proj_a_j + offs_h[None, None, :] * stride_proj_a_h
        proj_a_tile = tl.load(proj_a_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        gate_a_ptrs = gate_a_ptr + pid_b * stride_gate_a_b + offs_i[:, None, None] * stride_gate_a_i + offs_j[None, :, None] * stride_gate_a_j + offs_h[None, None, :] * stride_gate_a_h
        gate_a_raw_tile = tl.load(gate_a_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)
        gate_a_tile = 1.0 / (1.0 + tl.exp(-gate_a_raw_tile))

        # Right side
        proj_b_ptrs = proj_b_ptr + ... same but use stride/proj_b...
        proj_b_tile = tl.load(proj_b_ptrs, mask=..., other=0.0)

        gate_b_ptrs = gate_b_ptr + ... and load raw gate B
        gate_b_raw_tile = tl.load(gate_b_ptrs, mask=..., other=0.0)
        gate_b_tile = 1.0 / (1.0 + tl.exp(-gate_b_raw_tile))

        # mask broadcast over H dimension
        mask_tile_h = mask_tile[:, :, None]  # shape (BLOCK_I, BLOCK_J, 1)

        # Compute outputs
        out_a_tile = proj_a_tile * gate_a_tile * mask_tile_h
        out_b_tile = proj_b_tile * gate_b_tile * mask_tile_h

        # Store
        out_a_ptrs = out_a_ptr + ...

        tl.store(out_b_ptrs, out_b_tile, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
Outgoing TriMul (AlphaFold‑3) forward‑pass – fast fused implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣  Fuse **all** linear projections that depend on the same input (left/right
    projections, their gates, and the out‑gate) into a *single* GEMM:
       proj_gate = X @ W_catᵀ            # X : [B·N·N, dim] , W_cat : [5·H, dim]
   where H = hidden_dim and the concatenated weight is
       [left_proj; left_gate; right_proj; right_gate; out_gate].
3️⃣  Split the result and, with a tiny Triton kernel, apply the sigmoid gate,
    optional mask (broadcast over the hidden dimension) and finally write the
    gated tensors `left` and `right`.  This kernel fuses three element‑wise
    ops (sigmoid, mask, multiplication) and saves an extra memory round‑trip.
4️⃣  The O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (C = A @ Bᵀ) using cuBLAS tensor‑cores.
5️⃣  Second LayerNorm → multiply with the previously computed `out_gate`.
6️⃣  Final linear projection back to model dimension.
7️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate‑sigmoid fusion (step 3) is written in Triton; all heavy
matrix‑multiplications reuse cuBLAS, giving the best of both worlds.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse sigmoid‑gate, mask and element‑wise multiplication
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_a_ptr, gate_a_ptr,                     # left projection & gate (raw)
    proj_b_ptr, gate_b_ptr,                     # right projection & gate (raw)
    mask_ptr,                                   # [B, N, N] mask  (fp16)
    out_a_ptr, out_b_ptr,                       # left, right output tensors
    B, N, H,
    # strides for the four input tensors
    stride_pa_b, stride_pa_i, stride_pa_j, stride_pa_h,
    stride_ga_b, stride_ga_i, stride_ga_j, stride_ga_h,
    stride_pb_b, stride_pb_i, stride_pb_j, stride_pb_h,
    stride_gb_b, stride_gb_i, stride_gb_j, stride_gb_h,
    # stride for mask (no hidden dimension)
    stride_mask_b, stride_mask_i, stride_mask_j,
    # strides for output tensors
    stride_oa_b, stride_oa_i, stride_oa_j, stride_oa_h,
    stride_ob_b, stride_ob_i, stride_ob_j, stride_ob_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute, for a tile (i, j, h):
        out_a = proj_a * sigmoid(gate_a) * mask
        out_b = proj_b * sigmoid(gate_b) * mask
    Mask is broadcast over the hidden dimension (H).
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # Tile coordinates in the N‑dimension
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – will be broadcast later
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=1.0,                     # default to 1.0 (no masking)
    )                                 # fp16

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load left projection
        proj_a_ptrs = (
            proj_a_ptr
            + pid_b * stride_pa_b
            + offs_i[:, None, None] * stride_pa_i
            + offs_j[None, :, None] * stride_pa_j
            + offs_h[None, None, :] * stride_pa_h
        )
        proj_a = tl.load(
            proj_a_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )                               # fp16

        # Load left gate (raw) and apply sigmoid
        gate_a_ptrs = (
            gate_a_ptr
            + pid_b * stride_ga_b
            + offs_i[:, None, None] * stride_ga_i
            + offs_j[None, :, None] * stride_ga_j
            + offs_h[None, None, :] * stride_ga_h
        )
        gate_a_raw = tl.load(
            gate_a_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        gate_a = 1.0 / (1.0 + tl.exp(-gate_a_raw))

        # Load right projection
        proj_b_ptrs = (
            proj_b_ptr
            + pid_b * stride_pb_b
            + offs_i[:, None, None] * stride_pb_i
            + offs_j[None, :, None] * stride_pb_j
            + offs_h[None, None, :] * stride_pb_h
        )
        proj_b = tl.load(
            proj_b_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Load right gate (raw) and apply sigmoid
        gate_b_ptrs = (
            gate_b_ptr
            + pid_b * stride_gb_b
            + offs_i[:, None, None] * stride_gb_i
            + offs_j[None, :, None] * stride_gb_j
            + offs_h[None, None, :] * stride_gb_h
        )
        gate_b_raw = tl.load(
            gate_b_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        gate_b = 1.0 / (1.0 + tl.exp(-gate_b_raw))

        # Broadcast mask over H dimension
        mask_h_tile = mask_tile[:, :, None]          # shape (BLOCK_I, BLOCK_J, 1)

        # Fuse the three operations
        out_a = proj_a * gate_a * mask_h_tile
        out_b = proj_b * gate_b * mask_h_tile

        # Store left result
        out_a_ptrs = (
            out_a_ptr
            + pid_b * stride_oa_b
            + offs_i[:, None, None] * stride_oa_i
            + offs_j[None, :, None] * stride_oa_j
            + offs_h[None, None, :] * stride_oa_h
        )
        tl.store(
            out_a_ptrs,
            out_a,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

        # Store right result
        out_b_ptrs = (
            out_b_ptr
            + pid_b * stride_ob_b
            + offs_i[:, None, None] * stride_ob_i
            + offs_j[None, :, None] * stride_ob_j
            + offs_h[None, None, :] * stride_ob_h
        )
        tl.store(
            out_b_ptrs,
            out_b,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_gate_mask(left_proj: torch.Tensor,
                     left_gate_raw: torch.Tensor,
                     right_proj: torch.Tensor,
                     right_gate_raw: torch.Tensor,
                     mask: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Triton‑fused sigmoid‑gate, optional mask and element‑wise product.
    Returns:
        left  = left_proj * sigmoid(left_gate_raw) * mask
        right = right_proj * sigmoid(right_gate_raw) * mask
    All tensors must be on the same CUDA device and have dtype torch.float16.
    """
    B, N, _, H = left_proj.shape
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Strides expressed in number of elements
    s_lp = left_proj.stride()
    s_lg = left_gate_raw.stride()
    s_rp = right_proj.stride()
    s_rg = right_gate_raw.stride()
    s_m = mask.stride()
    s_lo = left.stride()
    s_ro = right.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _gate_mask_kernel[grid](
        left_proj,
        left_gate_raw,
        right_proj,
        right_gate_raw,
        mask,
        left,
        right,
        B,
        N,
        H,
        s_lp[0], s_lp[1], s_lp[2], s_lp[3],
        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_rp[0], s_rp[1], s_rp[2], s_rp[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_m[0], s_m[1], s_m[2],
        s_lo[0], s_lo[1], s_lo[2], s_lo[3],
        s_ro[0], s_ro[1], s_ro[2], s_ro[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return left, right


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM using cuBLAS.
    Input:
        left, right – [B, N, N, H] (fp16, contiguous)
    Output:
        out – [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B·H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # right is transposed in the contraction
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32), same device as input.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True ⇒ no mask is applied
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # FP32
    x = x.to(torch.float16)                      # FP16 for the rest

    # ------------------------------------------------------------------
    # Prepare weight tensors (FP16, on‑device)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    w_to_out      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear projections that share the same input
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                 # [B·N·N, dim]

    # Concatenate the 5 weight matrices: [left_proj, left_gate,
    # right_proj, right_gate, out_gate] → shape [5·H, dim]
    cat_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).contiguous()

    # Single fused linear call (cuBLAS)
    fused = torch.nn.functional.linear(x_flat, cat_weight)   # [B·N·N, 5·H]
    fused = fused.view(B, N, N, -1)                         # [B, N, N, 5·H]

    H = hidden_dim
    left_proj_raw   = fused[..., 0:H]
    left_gate_raw   = fused[..., H:2 * H]
    right_proj_raw  = fused[..., 2 * H:3 * H]
    right_gate_raw  = fused[..., 3 * H:4 * H]
    out_gate_raw    = fused[..., 4 * H:5 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Fuse sigmoid, mask and element‑wise product (Triton)
    # ------------------------------------------------------------------
    if nomask:
        # mask of all ones – cheap to create once
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16)          # [B,N,N]

    left, right = _apply_gate_mask(
        left_proj_raw, left_gate_raw,
        right_proj_raw, right_gate_raw,
        mask_fp16,
    )  # both [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Compute the O(N³) contraction via batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)  # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate (sigmoid) and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)              # [B, N, N, H] FP16
    out = out * out_gate                                 # gating (fp16)

    out = torch.nn.functional.linear(out, w_to_out)      # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
