@triton.jit
def batched_gemm_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    P, M, N, K,
    stride_l_p, stride_l_m, stride_l_k,
    stride_r_p, stride_r_n, stride_r_k,
    stride_o_p, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
    eps: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)  # block index across batch and M,N tiles
    # compute block coordinates
    # number of blocks in m, n
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # number of blocks per batch
    total_blocks_per_batch = num_blocks_m * num_blocks_n
    batch_id = pid // total_blocks_per_batch
    block_idx = pid % total_blocks_per_batch
    pid_m = block_idx // num_blocks_n

def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    P, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    total_blocks_per_batch = num_blocks_m * num_blocks_n
    batch_id = pid // total_blocks_per_batch
    tile_index = pid % total_blocks_per_batch
    pid_m = tile_index // num_blocks_n
    pid_n = tile_index % num_blocks_n

    # offsets for m and n:
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the batch
    a_batch_ptr = A_ptr + batch_id * stride_a_batch
    b_batch_ptr = B_ptr + batch_id * stride_b_batch
    c_batch_ptr = C_ptr + batch_id * stride_c_batch

    # init accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        A = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_T tile: shape (BLOCK_K, BLOCK_N)
        # B is stored as [batch, N, K] -> need to index B[n,k].
        # We load as B_T[k, n] = B[n,k]. compute pointers:
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        B = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate product
        acc += tl.dot(A, B)            # A: (M,K), B: (K,N) -> (M,N)

    # Write back to C
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)

def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left,right shape [B, N, N, hidden_dim] FP16
    B, N, N2, H = left.shape
    assert N == N2
    # permute and view
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (P, N, N)
    P = B * H
    a = left_perm.view(P, N, N)
    b = right_perm.view(P, N, N)
    # allocate output
    c = torch.empty_like(a, dtype=torch.float16)   # composition maybe same dtype
    # kernel launch
    # compute stride arguments
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_n, stride_b_k = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()
    # constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # compute grid size
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = P * num_blocks_m * num_blocks_n
    grid = (total_blocks,)
    batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        P=P,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_n=stride_b_n,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # reshape back: c shape (P,N,N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

"""
TriMul (outgoing) – fast forward pass with Triton‑fused mask‑gate and
batched GEMM.

Algorithm
---------
1️⃣  Layer‑Norm on the 4‑D input (FP32) → cast to FP16.
2️⃣  Fuse the two projection Linear layers into one Linear (weight concatenation).
    Fuse the two gating Linear layers likewise.
3️⃣  If a mask is present, fuse (proj * gate * mask) for the left and right
    streams with a tiny Triton kernel (`mask_gate_fuse_kernel`).
   – Otherwise just multiply the projection by its gate in‑place.
4️⃣  Heavy O(N³·c) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched half‑precision GEMM in Triton
   (`batched_gemm_kernel`).  The reduction is accumulated in FP32.
5️⃣  Second Layer‑Norm on the hidden dimension (FP16),
    out‑gate (sigmoid) multiplication and final linear projection back to `dim`.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the forward pass is required; no gradients are computed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise fuse (tensor * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16: tensor to be updated, length = total_elements
    gate_ptr,       # *fp16: gate tensor (same shape as x)
    mask_ptr,       # *fp16: mask flattened of shape (B·N·N,)
    total_elements,  # i32: total number of elements in x
    hidden_dim,      # i32: size of the hidden dimension (gate repeats per entry)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)        # linear offsets for this block
    mask = offs < total_elements                    # guard for tail

    # Load x and gate
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Every `hidden_dim` consecutive elements share the same mask entry.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default 1.0 when mask absent

    # Fused multiply
    out = x * g * m

    # Store in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ
#   A, B : (P, M, K)   – here P = B·hidden_dim, M = N, K = N
#   C    : (P, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # *fp16 pointers
    P, M, N, K,                         # i32 sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                        # global block id

    # --- compute block coordinates -------------------------------------------------
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m_blocks * num_n_blocks

    batch_id = pid // blocks_per_batch
    tile_index = pid % blocks_per_batch
    pid_m = tile_index // num_n_blocks
    pid_n = tile_index % num_n_blocks

    # offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds for the tile (handles tails)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- loop over K dimension ----------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m
                            + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # fp16

        # Load Bᵀ tile: (BLOCK_K, BLOCK_N)
        #   B is stored as [batch, N, K]; we need Bᵀ => index (k, n) = B[n, k]
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k
                            + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # fp16

        # Matrix multiply tile (FP16 inputs → FP32 accumulation)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # --- write back ---------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m
                        + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper wrapping the Triton GEMM
# ----------------------------------------------------------------------
def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³·c) contraction using a custom Triton kernel.
    Input tensors must have shape [B, N, N, C] (FP16) and be contiguous.
    Returns tensor of the same shape and dtype.
    """
    B, N, _, C = left.shape
    # permute to [B, C, N, N] and view as (P, N, N) where P = B·C
    left_perm = left.permute(0, 3, 1, 2).contiguous().view(B * C, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().view(B * C, N, N)

    P, M, K = left_perm.shape
    _, N2, _ = right_perm.shape
    assert N == N2 and K == N, "Inconsistent dimensions for GEMM"

    # allocate output
    out = torch.empty_like(left_perm)

    # strides (int64 → int32 for Triton)
    stride_a_batch, stride_a_m, stride_a_k = left_perm.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_perm.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    # Tuning parameters – these work well on H100 for the target sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid = total number of (batch, tile_m, tile_n) blocks
    num_m_blocks = (M + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = P * num_m_blocks * num_n_blocks
    grid = (total_blocks,)

    batched_gemm_kernel[grid](
        a_ptr=left_perm,
        b_ptr=right_perm,
        c_ptr=out,
        P=P,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_n=stride_b_n,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # reshape back to [B, N, N, C]
    out = out.view(B, C, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point – called by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : dict(str → torch.Tensor) (float32, same device)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]        # projection/gating width (c)
    nomask = config.get("nomask", True)     # True → treat mask as all‑ones

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, dim]  FP32
    x = x_norm.half()                             # work in FP16 from now on

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 4️⃣ Fused projections & gates (two concatenated Linear calls)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(B * N * N, dim)                      # [B·N², dim]

    # 4a. Concatenated projection (left & right)
    proj_weight_cat = torch.cat([left_proj_w, right_proj_w], dim=0)   # (2·c, dim)
    proj = torch.nn.functional.linear(x_flat, proj_weight_cat)        # (B·N², 2·c)
    left  = proj[:, :hidden_dim].view(B, N, N, hidden_dim).contiguous()
    right = proj[:, hidden_dim:].view(B, N, N, hidden_dim).contiguous()

    # 4b. Concatenated gating (left_gate & right_gate)
    gate_weight_cat = torch.cat([left_gate_w, right_gate_w], dim=0)    # (2·c, dim)
    gate_logits = torch.nn.functional.linear(x_flat, gate_weight_cat)  # (B·N², 2·c)
    gate = torch.sigmoid(gate_logits)
    left_gate  = gate[:, :hidden_dim].view(B, N, N, hidden_dim).contiguous()
    right_gate = gate[:, hidden_dim:].view(B, N, N, hidden_dim).contiguous()

    # 4c. out_gate (used after the second Norm)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    ).view(B, N, N, hidden_dim).contiguous()   # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask + gate (using Triton when mask is present)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
    else:
        # Ensure contiguous layout for the Triton kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # [B·N²]

        total_elements = left.numel()               # B·N²·c
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Fuse (proj * gate * mask) for left stream (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse for right stream
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    out = _batched_gemm_triton(left, right)   # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                            # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication + final projection
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
