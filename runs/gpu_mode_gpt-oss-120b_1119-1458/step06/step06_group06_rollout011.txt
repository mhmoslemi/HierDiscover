"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. FP32 layer‑norm on the raw [B,N,N,dim] tensor.
2. Cast to FP16 and perform a *single* fused linear projection that
   simultaneously computes:
        - left_proj, left_gate, right_proj, right_gate, out_gate
   (5 × hidden_dim outputs).  This reduces the number of large GEMM
   launches from five to one.
3. Apply sigmoid to the three gating streams.
4. Fuse (proj * gate * mask) for the left and right streams with a tiny
   Triton kernel (`mask_gate_fuse_kernel`).  When there is no mask a
   single in‑place multiplication suffices.
5. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is realized as a batched GEMM (B·hidden_dim independent matrix
   multiplies of size N×N) using torch.bmm, which maps to a single
   strided‑batched cuBLAS Tensor‑Core kernel.
6. Second layer‑norm (FP16) → element‑wise out‑gate multiplication.
7. Final linear projection back to the original channel dimension.
8. Cast the result back to FP32 (expected by the harness).

All heavy matrix multiplications run in FP16 with FP32 accumulation,
leveraging H100 Tensor‑Cores.  The only Triton kernel is the mask‑gate
fusion, satisfying the requirement to use custom kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask); mask is broadcast over the
# hidden dimension. Operates on a flat view of a contiguous tensor
# of shape [B, N, N, hidden_dim] (size = B·N·N·hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (in‑place)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,)
    total_elements,  # i32 : total number of elements in x
    hidden_dim,      # i32 : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elements                       # guard for tail

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)    # proj
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0) # gate

    # every `hidden_dim` entries share the same mask entry
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # cast 1.0 for no‑mask

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via
    a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]   (batch dimension = B·H)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # bmm does (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection size (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32
    x = x_norm.half()  # work in FP16 from now on

    # --------------------------------------------------------------
    # 2️⃣ Fuse *all* linear projections & gates into one matmul
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each of shape [hidden_dim, dim])
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # out_features dimension
    ).contiguous()  # shape (5*hidden_dim, dim)

    # Cast once to FP16
    combined_weight_fp16 = combined_weight.half()

    # Perform the fused linear transform
    # Result shape: [B, N, N, 5*hidden_dim]
    fused = torch.nn.functional.linear(x, combined_weight_fp16)

    # Split into the five streams
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = torch.split(
        fused, hidden_dim, dim=-1
    )

    # Apply sigmoid to the gating streams
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and element‑wise gating.
    #     Fuse proj * gate * mask with a tiny Triton kernel.
    # --------------------------------------------------------------
    if nomask:
        # Simple in‑place element‑wise product when no mask is needed
        left = left_proj.mul_(left_gate)   # [B, N, N, hidden_dim]
        right = right_proj.mul_(right_gate)
    else:
        # Flatten mask once (B·N·N) and cast to FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)

        total_elements = left_proj.numel()          # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Ensure all tensors are contiguous for Triton
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # Fuse projection, gate and mask for the left stream (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse projection, gate and mask for the right stream (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj

    # --------------------------------------------------------------
    # 4️⃣ Heavy contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM on FP16 (Tensor‑Core friendly).
    # --------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)          # [B, N, N, hidden_dim]  FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # --------------------------------------------------------------
    # 6️⃣ Out‑gate multiplication (element‑wise) + final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # broadcasted element‑wise
    out_final = torch.nn.functional.linear(
        out_norm, weights["to_out.weight"].half()
    )                                            # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out_final.float()
