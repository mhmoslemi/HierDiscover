"""
Optimized Triton‑accelerated “outgoing” TriMul forward pass.
-----------------------------------------------------------------
Algorithm
---------
1️⃣  Layer‑Norm on the raw input (float32) – stabilises statistics.
2️⃣  Cast to fp16 for all heavy arithmetic.
3️⃣  Fuse the five linear projections (left/right, three gates) into a
    single GEMM:   X·W (X∈ℝ^{B·N·N×dim}, W∈ℝ^{5·H×dim}).
    The result is reshaped to [B, N, N, 5·H] and sliced into
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre.
4️⃣  Apply sigmoid to the three gating slices.
5️⃣  If a mask is present, fuse the element‑wise product
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    into a single Triton kernel (mask_gate_fuse_kernel).  When there is
    no mask we simply multiply left_proj * left_gate (and similarly for
    right).
6️⃣  Perform the O(N³) contraction via a batched GEMM:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    by reshaping to (B·H, N, N) and calling torch.bmm (cuBLAS/Tensor‑Core).
7️⃣  Second Layer‑Norm (fp16) over the hidden dimension.
8️⃣  Multiply by the output gate, project back to the original dimension
    with a final linear layer and cast back to float32.

Only the forward pass is required – no autograd support.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ---------------------------------------------------------------
# Triton kernel: fuse mask, gate and element‑wise multiplication.
# ---------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : hidden dimension H
    size,           # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # linear offsets
    active = offs < size                                 # bounds check

    # Convert a flat offset in the (B,N,N,H) tensor to the corresponding mask offset.
    mask_offs = offs // hidden_dim                       # integer division

    # Load data
    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)  # mask defaults to 1 (no effect)

    # Fuse: x = x * g * m
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)

# --------------------------------------------------------------------
# Helper: batched matrix multiplication implementing the O(N³) contraction.
# --------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a batched GEMM.
    All tensors are expected to be fp16 with shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] → each hidden channel becomes a separate batch entry
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))       # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# --------------------------------------------------------------------
# Main entry point required by the harness.
# --------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask

    # -------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32)
    # -------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim] – float32

    # -------------------------------------------------------------
    # 2️⃣  Cast to FP16 for the heavy parts
    # -------------------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------------------
    # 3️⃣  Fuse the five linear projections into one GEMM
    # -------------------------------------------------------------
    B, N, _, _ = x.shape
    # Concatenate weights (shape [(5*hidden_dim), dim]) and cast once to FP16
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(device=x.device, dtype=torch.float16)   # [(5*H), dim]

    # Flatten batch & spatial dimensions → shape (B·N·N, dim)
    x_flat = x.view(-1, dim)

    # Single linear projection
    combined_proj = torch.nn.functional.linear(x_flat, combined_weight)  # (B·N·N, 5·H)

    # Reshape to (B, N, N, 5·H) so the hidden dimension is innermost (contiguous)
    combined_proj = combined_proj.view(B, N, N, 5 * hidden_dim)

    # -----------------------------------------------------------------
    # 4️⃣  Slice the combined result and apply sigmoid to gates
    # -----------------------------------------------------------------
    # slices along the last dimension
    offset = hidden_dim
    left_proj    = combined_proj[..., :offset]                        # [B,N,N,H]
    right_proj   = combined_proj[..., offset:2*offset]                # [B,N,N,H]
    left_gate_pre = combined_proj[..., 2*offset:3*offset]
    right_gate_pre = combined_proj[..., 3*offset:4*offset]
    out_gate_pre  = combined_proj[..., 4*offset:5*offset]

    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # -------------------------------------------------------------
    # 5️⃣  Apply mask + gate (fused via Triton when a mask is present)
    # -------------------------------------------------------------
    if not nomask:
        # Ensure contiguous fp16 tensors for the kernel
        left_proj   = left_proj.contiguous()
        right_proj  = right_proj.contiguous()
        left_gate   = left_gate.contiguous()
        right_gate  = right_gate.contiguous()
        mask_fp16   = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()               # B·N·N·H
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left  = left_proj   # already fused
        right = right_proj
    else:
        # No mask – simple element‑wise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------------------
    # 6️⃣  Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # -------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H]  fp16

    # -------------------------------------------------------------
    # 7️⃣  Second LayerNorm over hidden dimension (fp16)
    # -------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------------------
    # 8️⃣  Output gate and final projection back to original dim
    # -------------------------------------------------------------
    out = out * out_gate                               # gated hidden state
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # -------------------------------------------------------------
    # 9️⃣  Cast back to float32 (required by the test harness)
    # -------------------------------------------------------------
    return out.to(torch.float32)
