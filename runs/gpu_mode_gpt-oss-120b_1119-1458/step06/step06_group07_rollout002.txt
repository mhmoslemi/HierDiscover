def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr, hidden_dim, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size
    # compute idx = offs // hidden_dim
    idx = offs // hidden_dim
    # mask validity:
    mask_valid = idx < (size // hidden_dim)   # mask_len = size/hidden_dim = B*N*N
    # load
    x = tl.load(x_ptr + offs, mask=valid, other=0.0).to(tl.float16)   # Actually pointer is half.
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    # mask: load as half, default 1.0 (no mask)
    m = tl.load(mask_ptr + idx, mask=mask_valid, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)
```

"""
high‑performance outgoing TriMul implementation (AlphaFold3)

Algorithm
---------
1.  Layer‑norm the input tensor (float32) → fp16.
2.  Fuse the five linear projections (left, right, left‑gate,
    right‑gate, out‑gate) into a single GEMM:
       X_flat @ W_concat^T   where  W_concat ∈ ℝ^{5·H×C}.
3.  Slice the concatenated result into tensors
    `left_proj, right_proj, left_gate, right_gate, out_gate`.
    Apply sigmoid to the three gating tensors.
4.  If a mask is present, fuse `tensor = tensor * gate * mask`
    with a small Triton kernel (in‑place).
    Otherwise multiply `tensor = tensor * gate`.
5.  Heavy O(N³) update:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    is realised as a batched GEMM:
    for each hidden channel `h` compute `L_h @ R_hᵀ`.
6.  Layer‑norm over the hidden dimension (fp16).
7.  Multiply by `out_gate` and project back to the original
    channel size with a final linear.
8.  Cast the result back to float32 (required by the harness).
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask·gate·tensor multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16   tensor to be updated (flattened)
    gate_ptr,       # *fp16   gate tensor (same shape as `x_ptr`)
    mask_ptr,       # *fp16   flattened mask (shape = B·N·N)
    hidden_dim,     # i32    hidden dimension (H)
    size,           # i32    total number of elements in `x_ptr` (B·N·N·H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load data & gate
    x = tl.load(x_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0).to(tl.float16)

    # Compute mask index and load mask element (broadcast over H)
    idx = offs // hidden_dim                # which (b,i,k) index we are at
    mask_len = size // hidden_dim           # = B·N·N
    mask_valid = idx < mask_len
    m = tl.load(mask_ptr + idx, mask=mask_valid, other=1.0)   # default = 1

    # Fuse multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM (fp16) for the O(N³) contraction.
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H) in fp16
    Returns:
        out: (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]  (contiguous layout required for torch.bmm)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # batch‑matrix‑multiply L * Rᵀ
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Args:
        data:
            - input_tensor: [B, N, N, C] (float32)
            - mask_tensor : [B, N, N] (float32 or bool) – may be all‑ones when nomask=True
            - weights      : dict of model weights (float32)
            - config       : dict containing 'dim', 'hidden_dim' and optional 'nomask'

    Returns:
        Tensor [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → fp16
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C] float32
    x = x_norm.to(torch.float16)      # fp16 for heavy work

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # -----------------------------------------------------------------
    # Concatenate weight matrices (out_features = 5*H, in_features = C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()  # (5*H, C)

    B, N, _, _ = x.shape
    M = B * N * N                     # number of (i,k) pairs per batch

    # Flatten to (M, C) for the fused linear projection
    x_flat = x.reshape(M, dim)       # contiguous view, fp16

    # Single large GEMM: (M, C) @ (C, 5*H)ᵀ → (M, 5*H)
    combined_out = torch.nn.functional.linear(x_flat, concat_w)  # fp16
    combined_out = combined_out.view(B, N, N, 5 * hidden_dim)    # [B,N,N,5H]

    # -----------------------------------------------------------------
    # 3️⃣ Slice and apply sigmoid to the three gating tensors
    # -----------------------------------------------------------------
    left_proj = combined_out[..., :hidden_dim]
    right_proj = combined_out[..., hidden_dim:2 * hidden_dim]
    left_gate = combined_out[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = combined_out[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate = combined_out[..., 4 * hidden_dim:5 * hidden_dim]

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton)
    # -----------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N] – broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)               # shape = B*N*N

        total_elements = left_proj.numel()           # = B*N*N*H
        BLOCK = 1024  # max threads per block (fits on H100)
        grid = (triton.cdiv(total_elements, BLOCK),)

        # left_proj = left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right_proj = right_proj * right_gate * mask (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # No mask → plain element‑wise gating
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)          # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16) over hidden_dim
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # -----------------------------------------------------------------
    # 7️⃣ Gating + final projection
    # -----------------------------------------------------------------
    # out = out * out_gate  (element‑wise)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)  # (C, H)
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,C] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
