    left_ptr,
    left_gate_ptr,
    right_ptr,

import triton
import triton.language as tl

# Triton kernel for fused gate and mask multiplication for left/right tensors.
@triton.jit
def fused_gate_mask_two_tensor(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    total_elts: tl.int32,
    hidden_dim: tl.constexpr,

"""
TriMul (outgoing) – fused‑linear + fused‑gate‑mask Triton implementation.

Algorithm
---------
1️⃣  Layer‑Norm the input in FP32 (stable) → cast to FP16.
2️⃣  Fuse the projection and its gate (left / right) into a single matmul
    by concatenating the two weight matrices (proj‑weight|gate‑weight).
3️⃣  Apply sigmoid to the gate tensors.
4️⃣  Fuse the element‑wise multiplication “proj × gate × mask” for the
    left and right branches into **one** Triton kernel (the mask is broadcast
    across the hidden dimension).
5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    performed as a batched GEMM using torch.bmm (Tensor‑Cores).  The hidden
    dimension is folded into the batch dimension (B·H) so that each GEMM
    is a standard N×N matmul.
6️⃣  Layer‑Norm over the hidden dimension, multiply by the output‑gate,
    and a final linear projection back to the original channel dimension.
7️⃣  Cast the result back to FP32 (the harness expects float32).

Benefits over the reference implementation
-----------------------------------------
* 2× fewer `torch.nn.functional.linear` calls (projection+gate are fused).
* 1× fewer Triton launches (left‑right gate‑mask fusion).
* All large‑scale arithmetic stays in FP16 / Tensor‑Core and memory traffic
  is reduced (no extra temporary tensors for the gate tensors).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) for left and right tensors.
# The mask has shape (B, N, N) and is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_two_tensor(
    left_ptr,          # *fp16 : left projection tensor (B,N,N,H)
    left_gate_ptr,     # *fp16 : left gate tensor (same shape as left)
    right_ptr,         # *fp16 : right projection tensor (B,N,N,H)
    right_gate_ptr,    # *fp16 : right gate tensor (same shape as right)
    mask_ptr,          # *fp16 : mask tensor (B,N,N), broadcast over H
    total_elts: tl.int32,          # total number of elements in a single tensor
    hidden_dim: tl.constexpr,      # size of the hidden dimension H
    BLOCK: tl.constexpr,           # tile size (≤ 1024)
):
    pid = tl.program_id(0)
    # Linear index of the elements this program will handle
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # map the flat index to the mask index (broadcast across H)
    mask_idx = offs // hidden_dim

    # Load the four tensors and the mask
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr      + offs, mask=mask, other=0.0)
    right_g   = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m         = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # In‑place fusion: proj * gate * mask
    tl.store(left_ptr  + offs, left  * left_g  * m, mask=mask)
    tl.store(right_ptr + offs, right * right_g * m, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   (FP16)

    Parameters
    ----------
    left, right : torch.Tensor of shape [B, N, N, H] (FP16)

    Returns
    -------
    out : torch.Tensor of shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) – each hidden slice becomes a separate batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul (outgoing) forward pass.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {'dim': int, 'hidden_dim': int, 'nomask': bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # --------------------------------------------------------------
    # Unpack everything
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for everything else
    # --------------------------------------------------------------
    x = x_norm.to(torch.float16)  # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 3️⃣ Prepare FP16 weights – fuse projection + gate matrices
    # --------------------------------------------------------------
    # left side: concat([proj, gate]) → shape (2*H, dim)
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)      # (2*H, dim)

    # right side: concat([proj, gate]) → shape (2*H, dim)
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)      # (2*H, dim)

    # out‑gate (dim → H)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    # final projection (H → dim) and its LayerNorm parameters
    to_out_weight      = weights["to_out.weight"].to(torch.float16, non_blocking=True)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16, non_blocking=True)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16, non_blocking=True)

    # --------------------------------------------------------------
    # 4️⃣ Fused projection + gate (one matmul per side)
    # --------------------------------------------------------------
    # left side: first hidden_dim → left_proj, last hidden_dim → left_gate_pre
    left_combined = F.linear(x, left_combined_weight)            # [B,N,N,2*H]  FP16
    left_proj = left_combined[..., :hidden_dim]                 # [B,N,N,H]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])  # [B,N,N,H]

    # right side
    right_combined = F.linear(x, right_combined_weight)          # [B,N,N,2*H]  FP16
    right_proj = right_combined[..., :hidden_dim]               # [B,N,N,H]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])# [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Optional mask – fuse (proj * gate * mask) in‑place via Triton
    # --------------------------------------------------------------
    if not nomask:
        # Broadcast mask to hidden_dim and convert to FP16
        mask_fp16 = mask_tensor.to(torch.float16, non_blocking=True).contiguous()

        total_elts = left_proj.numel()           # B * N * N * H
        BLOCK = 1024                              # ≤ max threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        fused_gate_mask_two_tensor[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,          # compile‑time constant for the kernel
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Compute the output‑gate (dim → H) – used after the contraction
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # [B,N,N,H]  FP16

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Cores)
    # --------------------------------------------------------------
    # left_proj/right_proj now contain: proj * gate * mask
    out = _batched_matmul_torch(left_proj, right_proj)  # [B,N,N,H]  FP16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm on the hidden dimension (FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # --------------------------------------------------------------
    # 9️⃣ Apply the output‑gate and final linear projection
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]  FP16

    # --------------------------------------------------------------
    # 🔟 Return FP32 for the evaluation harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
