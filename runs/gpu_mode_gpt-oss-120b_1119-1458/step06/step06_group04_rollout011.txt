@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,       # dtype fp16
    gate_ptr,    # dtype fp16
    mask_ptr,    # dtype fp16
    hidden_dim: tl.constexpr, # i32
    total_elems: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Calculate mask index: each hidden_dim values share same mask.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)
```

import torch
import triton
import triton.language as tl
from torch.nn import functional as F
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    hidden_dim: tl.constexpr,
    total_elems: tl.constexpr,
    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load values
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (mask per position, broadcast across hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Efficient forward pass of the outgoing TriMul module.
    Steps:
      1. LayerNorm over the last dimension (float32)
      2. Cast to fp16 and flatten
      3. Single fused linear to compute left/right projections and three gates
      4. Sigmoid on gate tensors
      5. Optional mask‑gate fusion using a tiny Triton kernel
      6. Reshape and batched GEMM (B*H) @ (B*H)ᵀ via torch.bmm
      7. Second LayerNorm (fp16), out‑gate multiplication
      8. Final linear back to `dim`
      9. Cast back to float32
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device
    eps = 1e-5

    # ---------------------------
    # 1. First LayerNorm (float32)
    # ---------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim] float32

    # ---------------------------
    # 2. Cast to fp16 and flatten
    # ---------------------------
    x = x_norm.to(torch.float16)
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # [B*N*N, dim]

    # -------------------------------------------------
    # 3. Fuse all linear projections into a single matmul
    # -------------------------------------------------
    # Gather weights (float16)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along out_features dimension: order matters!
    combined_weight = torch.cat(
        [left_proj_w,
         right_proj_w,
         left_gate_w,
         right_gate_w,
         out_gate_w], dim=0)  # shape (5*hidden_dim, dim)

    # Single linear: (B*N*N, dim) @ (dim, 5*hidden_dim).T → (B*N*N, 5*hidden_dim)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # fp16

    # -----------------------------------------------------------------
    # 4. Split combined tensor into projections and gate pre‑activations
    # -----------------------------------------------------------------
    h = hidden_dim
    left_proj = combined[:, :h]                     # (M, h)
    right_proj = combined[:, h:2*h]                # (M, h)
    left_gate_pre = combined[:, 2*h:3*h]           # (M, h)
    right_gate_pre = combined[:, 3*h:4*h]          # (M, h)
    out_gate_pre = combined[:, 4*h:5*h]            # (M, h)

    # -------------------------------------------------------
    # 5. Apply sigmoid to obtain gates (fp16)
    # -------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # -------------------------------------------------------
    # 6. Mask‑gate fusion (if mask present)
    # -------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # flatten mask to shape (B*N*N,)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # shape [M]
        total_elems = left_proj.numel()   # M * h
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # fuse left
        mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16,
            hidden_dim, total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # fuse right
        mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16,
            hidden_dim, total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: keep left/right as proj * gate
        left_proj.mul_(left_gate)   # in‑place
        right_proj.mul_(right_gate)

    # At this point left_proj and right_proj hold the final left/right tensors
    # -----------------------------------------------------------------
    # 7. Reshape to 4‑D tensors for the contraction
    # -----------------------------------------------------------------
    left = left_proj.view(B, N, N, hidden_dim)      # [B, N, N, h]
    right = right_proj.view(B, N, N, hidden_dim)

    # -------------------------------------------------
    # 8. Core O(N³) contraction via batched GEMM (fp16)
    # -------------------------------------------------
    # Bring hidden_dim to batch dimension: (B, h, N, N) → (B*h, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiply: (B*h, N, N) @ (B*h, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*h, N, N)

    # reshape back to (B, N, N, h)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 9. Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # -------------------------------------------------
    # 10. Out‐gate multiplication
    # -------------------------------------------------
    # out_gate currently shape (M, h) → reshape to (B,N,N,h)
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate

    # -------------------------------------------------
    # 11. Final projection back to dim
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim) fp16

    # -------------------------------------------------
    # 12. Cast back to float32
    # -------------------------------------------------
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – highly‑optimized Triton‑augmented implementation.

Algorithm
---------
1. First LayerNorm over the last dimension (float32) – stable for the raw input.
2. Cast to fp16 and flatten the tensor to (B·N·N, dim).
3. Fuse *five* separate linear layers (left/right projections + three gates) into a
   single matrix multiplication:
       [B·N·N, dim] @ [out_features, dim].T  →  [B·N·N, 5·hidden_dim]
   The weight matrix is the vertical concatenation of the five PyTorch weight
   tensors (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`).
4. Split the fused result, apply `sigmoid` to the three gate tensors.
5. Optional mask‑gate fusion: a tiny Triton kernel multiplies
   `proj * gate * mask` in‑place for `left` and `right`.  If no mask is used,
   we simply perform `proj *= gate`.
6. Reshape `left`/`right` to (B, N, N, hidden_dim) and reorganise to
   (B·hidden_dim, N, N) for a batched GEMM:
       out[b·h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   Implemented as `torch.bmm` on fp16 tensors (uses cuBLAS/Tensor‑cores).
7. Second LayerNorm (fp16) on the contraction result.
8. Multiply by `out_gate` (produced in step 4) and apply the final linear
   projection back to the original `dim`.
9. Cast the final tensor back to float32 (the harness expects float32).

All heavy arithmetic runs in fp16, the only Triton part is the mask‑gate
fusion kernel (required by the problem statement).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F
from typing import Tuple, Dict, Any


@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 – tensor to be updated (flattened)
    gate_ptr,       # *fp16 – gate tensor (flattened)
    mask_ptr,       # *fp16 – mask tensor (flattened, length = B·N·N)
    hidden_dim: tl.constexpr,   # i32 – hidden dimension (H)
    total_elems: tl.constexpr,  # i32 – total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,        # compile‑time block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load the projection and gate values
    proj = tl.load(x_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Each `hidden_dim` consecutive entries share the same mask entry
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1

    # Fuse multiplication
    out = proj * gate * m

    # Store result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)

    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication: (B·h, N, N) @ (B·h, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))      # (B·h, N, N)

    # Restore shape (B, N, N, h)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)

    out = out * out_gate

    # 1️⃣1️⃣ Final projection back to original dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # (B, N, N, dim), fp16

    # -----------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
