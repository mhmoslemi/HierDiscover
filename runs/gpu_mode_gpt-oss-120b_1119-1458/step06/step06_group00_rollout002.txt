import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements, H,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    mask_offs = offs // H

    left       = tl.load(left_ptr       + offs, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + offs, mask=valid, other=0.0)
    left_gate  = tl.load(left_gate_ptr  + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val   = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    The implementation fuses:
    * LayerNorm (FP32)
    * Two fused linear projections (proj + gate) per side via a stacked weight.
    * Optional mask+gate fusion (Triton kernel).
    * Batched GEMM for the N³ contraction using a strided‑batched cuBLAS call.
    * Second LayerNorm, out‑gate and final projection.

    All heavy arithmetic runs in fp16 for peak H100 throughput.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stable statistics)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias
    )
    # Cast to fp16 for all subsequent ops
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left and left‑gate linear projections
    #    Output shape: [B,N,N,2*hidden_dim] → split → proj & gate
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=dtype_fp16)
    left_fused = F.linear(x, left_fused_w)                         # [B,N,N,2*H]
    left_proj = left_fused[..., :hidden_dim]                      # proj part
    left_gate = torch.sigmoid(left_fused[..., hidden_dim:])       # gate part

    # ------------------------------------------------------------------
    # 3️⃣ Fuse right and right‑gate linear projections
    # ------------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=dtype_fp16)
    right_fused = F.linear(x, right_fused_w)                       # [B,N,N,2*H]
    right_proj = right_fused[..., :hidden_dim]
    right_gate = torch.sigmoid(right_fused[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) and element‑wise gating
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate       # in‑place not required
        right = right_proj * right_gate
    else:
        # Convert mask to fp16 and broadcast over hidden dim
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        # Ensure all tensors are contiguous before Triton launch
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        total = left_proj.numel()
        # Choose a block size that gives ≈2 K threads per block (tuned for H100)
        BLOCK = 2048
        grid = ((total + BLOCK - 1) // BLOCK,)
        _mask_gate_fusion_kernel[grid](
            left_ptr=left_proj,
            right_ptr=right_proj,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=2,
        )
        left = left_proj
        right = right_proj
        # Free temporaries early
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Compute out‑gate (from the normalized input)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction: batched GEMM (left @ rightᵀ)
    #    Collapse batch & head dimensions → (B·H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange (B,N,N,H) → (B·H, N, N) by moving H before the two N's
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (in fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate application and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                              # element‑wise gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = F.linear(out, to_out_w)                     # [B,N,N,dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the consumer
    # ------------------------------------------------------------------
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate element‑wise.
# This kernel reads 4 fp16 values (left, right, left_gate, right_gate)
# and one broadcasted mask value, then writes the gated‑masked results
# back to left and right.  The mask is broadcast over the hidden‑dim.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # int   total number of fp16 elems in left/right
    H,                 # int   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # each hidden‑dim element belongs to the same mask entry
    mask_offs = offs // H

    left       = tl.load(left_ptr       + offs, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + offs, mask=valid, other=0.0)
    left_gate  = tl.load(left_gate_ptr  + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val   = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Algorithm
    ---------
    1️⃣  FP32 LayerNorm on the input.
    2️⃣  Fuse `proj` and `gate` linear layers per side:
        - Stack the projection and gate weight matrices (2·H × dim).
        - A single matmul yields both projection and pre‑gate values.
        - Apply sigmoid to the gate part.
    3️⃣  Fuse optional mask, projection‑gate multiplication in a Triton kernel.
    4️⃣  Compute the out‑gate (sigmoid of another linear projection on the normed input).
    5️⃣  O(N³) contraction implemented as a strided‑batched GEMM
        (left @ rightᵀ) with hidden‑dim fused into the batch dimension.
    6️⃣  FP16 LayerNorm on the hidden representation.
    7️⃣  Apply the out‑gate and the final linear projection back to `dim`.
    8️⃣  Cast result back to FP32.

    All heavy arithmetic runs in fp16 to exploit H100 Tensor Cores.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → stable statistics)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B,N,N,dim] FP32
    x = x.to(fp16)                     # cast to FP16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left projection + left gate (single matmul)
    # ------------------------------------------------------------------
    left_fused_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=fp16)   # (2·H, dim)
    left_fused = F.linear(x, left_fused_w)           # [B,N,N,2·H] FP16
    left_proj = left_fused[..., :hidden_dim]         # projection part
    left_gate = torch.sigmoid(left_fused[..., hidden_dim:])  # gate part

    # ------------------------------------------------------------------
    # 3️⃣ Fuse right projection + right gate (single matmul)
    # ------------------------------------------------------------------
    right_fused_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=fp16)   # (2·H, dim)
    right_fused = F.linear(x, right_fused_w)         # [B,N,N,2·H] FP16
    right_proj = right_fused[..., :hidden_dim]
    right_gate = torch.sigmoid(right_fused[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and element‑wise gating
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask shape: [B,N,N] → broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left_proj.numel()
        BLOCK = 2048                       # tuned for H100
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left_proj,
            right_ptr=right_proj,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=2,
        )
        left = left_proj
        right = right_proj
        # free temporaries
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Out‑gate (computed from the normed input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device=device, dtype=fp16))
    )                                   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM for the N³ contraction
    #    Collapse (batch, hidden) → (B·H, N, N) and use a strided‑batched
    #    cuBLAS call (torch.bmm) which is Tensor‑core accelerated.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden_dim before the two N dimensions, then reshape.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_mat[b·H, i, j] = Σ_k left_mat[b·H, i, k] * right_mat[b·H, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=fp16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=fp16),
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating (FP16)
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=fp16)
    )                                          # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
