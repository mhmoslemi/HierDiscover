"""
TriMul (outgoing) forward pass – highly‑optimized mixed‑precision
implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣  Fuse each pair of (proj, gate) linear layers into a single matmul
    (weight stacked) → split into projection & gate, apply sigmoid to gate.
3️⃣  If a mask is present, fuse mask·gate·proj element‑wise with a small
    Triton kernel (mask broadcast over the hidden dimension).  When no mask
    is used we simply multiply the projection by its gate.
4️⃣  The O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM (C = A @ Bᵀ) using cuBLAS tensor‑cores.
    The hidden dimension is folded into the batch dimension (B·H).
5️⃣  Apply a second LayerNorm (FP16) → mask the result with the out‑gate →
    final linear projection back to the model dimension.
6️⃣  Cast the final result back to FP32 (the harness expects FP32).

Only step 3 is written in Triton – this satisfies the “must use Triton”
requirement while keeping the heavy O(N³) work on the highly‑optimized
cuBLAS kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask is broadcast over hidden dimension H)
    """
    pid_b = tl.program_id(0)  # batch index
    pid_i = tl.program_id(1)  # i‑tile
    pid_j = tl.program_id(2)  # j‑tile

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store output tile
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim)

    All tensors must be FP16, CUDA, contiguous.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides expressed in elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction:
        out[b, i, j, h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batched GEMM (C = A @ Bᵀ) using cuBLAS tensor‑core kernels.
    Arguments
    ---------
    left, right : torch.Tensor [B, N, N, H] (FP16, contiguous)

    Returns
    -------
    out : torch.Tensor [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # Fold batch and hidden dims → shape [B·H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply with right transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool, or int)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # True => treat mask as all‑ones

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # result in FP32
    x = x.to(torch.float16)  # FP16 for the remaining ops

    # ------------------------------------------------------------------
    # 2️⃣ Prepare FP16 model weights (device‑resident)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections + gates (FP16)
    #     We fuse each (proj, gate) pair into one matmul by stacking weights.
    # ------------------------------------------------------------------
    # Left side: stacked weight [2*H, dim]
    left_stacked_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)
    left_combined = torch.nn.functional.linear(x, left_stacked_w)  # [B,N,N,2*H]
    left_proj, left_gate = torch.split(left_combined, [hidden_dim, hidden_dim], dim=-1)
    left_gate = torch.sigmoid(left_gate)  # gating sigmoid

    # Right side: stacked weight
    right_stacked_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)
    right_combined = torch.nn.functional.linear(x, right_stacked_w)  # [B,N,N,2*H]
    right_proj, right_gate = torch.split(right_combined, [hidden_dim, hidden_dim], dim=-1)
    right_gate = torch.sigmoid(right_gate)

    # Out gate (separate)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Mask‑gate fusion – Triton kernel or plain multiply
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is FP16 (values 0/1)
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # No mask – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate and final projection back to model dim (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast result to FP32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
