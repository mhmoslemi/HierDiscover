"""
Highly‑optimized forward pass of the **outgoing** Triangle Multiplicative Update (TriMul)
used in AlphaFold3 and similar protein‑structure models.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32 for numerical stability).
2️⃣  Cast to FP16 for the compute‑heavy part.
3️⃣  Fuse the two linear projections (proj + gate) for the *left* and *right*
    tensors by stacking their weight matrices (shape [2·hidden_dim, dim]).
    A single half‑precision matmul yields both the projection and the pre‑sigmoid
    gate in one pass.
4️⃣  Apply the sigmoid to the gate tensors.
5️⃣  If a mask is supplied, fuse the element‑wise multiplication of
    (projection * gate * mask) for *left* and *right* via a tiny Triton kernel.
    This avoids three separate memory passes.
6️⃣  Perform the O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM: reshape to (B·hidden_dim, N, N) and call ``torch.bmm``.
7️⃣  Second Layer‑norm on the hidden dimension (FP16).
8️⃣  Apply the output‑gate (sigmoid of a linear on the normalized input) and
    a final linear projection back to ``dim``.
9️⃣  Cast the result back to FP32 (the harness expects float32).

The implementation uses mixed‑precision (FP16) for all heavy tensor algebra,
fuses the projection+gate+mask step into a Triton kernel, and relies on
cuBLAS/Tensor‑Core kernels for the batched matrix multiplication.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) in‑place.
# -------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,            # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor (no hidden dimension)
    total_elts: tl.int32,   # total number of elements in x/gate (B·N·N·H)
    hidden_dim: tl.int32,   # hidden dimension (size of final axis of x)
    BLOCK: tl.constexpr,    # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_elts

    # mask is broadcast across the hidden dimension
    mask_idx = offs // hidden_dim

    x_val   = tl.load(x_ptr   + offs, mask=mask, other=0.0)
    g_val   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m_val   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x_val * g_val * m_val, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the heavy N³ contraction.
# ----------------------------------------------------------------------
def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] (FP16).

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] in half precision.

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) – each hidden channel becomes a separate batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* Triangle Multiplicative Update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict  : dict of model weights (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``.
          Optional key ``nomask`` (True → ignore mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] in float32.
    """
    # ---------------------------- unpack ------------------------------
    input_tensor, mask_tensor, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    device = input_tensor.device

    # --------------------- 1️⃣ First LayerNorm (FP32) -----------------
    norm_w = weights["norm.weight"]          # (dim,)
    norm_b = weights["norm.bias"]            # (dim,)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # FP32, shape [B,N,N,dim]

    # --------------------- 2️⃣ Cast to FP16 ---------------------------
    x = x_norm.to(torch.float16)   # [B,N,N,dim] in half

    # --------------------- 3️⃣ Prepare FP16 weights -------------------
    # Fuse (proj + gate) for left
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden_dim, dim)

    # Fuse (proj + gate) for right
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Out‑gate (dim → hidden_dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Second (to‑out) layer‑norm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # Final Linear (hidden_dim → dim)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # --------------------- 4️⃣ Linear + Gate for left ----------------
    left_combined = torch.nn.functional.linear(x, left_combined_w)     # [B,N,N,2*hidden_dim]
    left_proj = left_combined[..., :hidden_dim]                       # proj part
    left_gate_pre = left_combined[..., hidden_dim:]                   # gate pre‑sigmoid
    left_gate = torch.sigmoid(left_gate_pre)                          # sigmoid

    # --------------------- 5️⃣ Linear + Gate for right ---------------
    right_combined = torch.nn.functional.linear(x, right_combined_w)
    right_proj = right_combined[..., :hidden_dim]
    right_gate_pre = right_combined[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # --------------------- 6️⃣ Output gate (dim → hidden_dim) ----------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden_dim]

    # --------------------- 7️⃣ Mask & Gate Fusion (Triton) ------------
    if not nomask:
        # Convert mask to half and ensure contiguous layout.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                     # B·N·N·hidden_dim
        BLOCK = 1024                                        # tile size (tuned)
        grid = ( (total_elts + BLOCK - 1) // BLOCK, )

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel, left_proj/right_proj already contain the gated & masked values.
        left = left_proj
        right = right_proj
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------- 8️⃣ Heavy O(N³) contraction ----------------
    # Shape after step 7: [B, N, N, hidden_dim] (FP16)
    out = _batched_matmul_fp16(left, right)   # [B, N, N, hidden_dim] FP16

    # --------------------- 9️⃣ Second LayerNorm (FP16) ----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # --------------------- 🔟 Apply out‑gate and final linear ---------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] (FP16)

    # --------------------- 📐 Cast back to FP32 ----------------------
    return out.to(torch.float32)
