@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)   # combined batch and hidden
    pid_i = tl.program_id(1) # i tile
    pid_j = tl.program_id(2) # j tile

    # flatten batch*hidden dimension into single dimension = pid
    b = pid // H
    h = pid % H

    # tile offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension tiles
    for k_offset in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_offset * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        # left is [i,k]
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_J)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)

        # Multiply-accumulate
        # left: (I,K), right: (K,J)
        acc += tl.dot(left, right)  # returns (I,J) in fp32 (if left,right fp16?)
    # after loop, cast to fp16 (or keep fp32 for later)
    out = acc.to(tl.float16)

    # Store out tile
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

    pid_j = tl.program_id(2)

    h = pid_bh % H

    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_j = offs_j < N

    # Loop over K dimension
    for k_offset in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_offset * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Load right tile transposed: shape (BLOCK_K, BLOCK_J)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Accumulate dot product left @ right (left I x K; right K x J)
        acc += tl.dot(left, right)

    # Write out
    out = acc.to(tl.float16)

    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated O(N³) contraction.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input in FP32, cast to FP16 for the rest.
2️⃣ Fuse each side’s projection + gate into a single linear call
   (weights are concatenated → one matmul, later split & sigmoid).
3️⃣ Apply the optional mask (broadcast over the hidden dimension)
   and the gates by cheap element‑wise products.
4️⃣ Perform the heavy O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that loops over K‑tiles and
   accumulates with `tl.dot`.  The kernel directly works on the
   [B,N,N,H] layout, avoiding costly permutes.
5️⃣ Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear
   projection back to model‑dim.
6️⃣ Cast the result to FP32 (the harness expects FP32).

All major work after the linear projections is executed inside
the Triton kernel, meeting the “use Triton” requirement while
keeping the implementation simple and fast on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
@triton.jit
def _tr_mul_kernel(
    left_ptr,                          # *fp16  [B, N, N, H]
    right_ptr,                         # *fp16  [B, N, N, H]
    out_ptr,                           # *fp16  [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """out = Σₖ left[...,i,k,h] * right[...,j,k,h]."""
    pid_bh = tl.program_id(0)   # combined batch & hidden index
    pid_i = tl.program_id(1)    # tile over i
    pid_j = tl.program_id(2)    # tile over j

    # Derive batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (FP32 for precision)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over K dimension tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile   shape (BLOCK_I, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Load right tile transposed   shape (BLOCK_K, BLOCK_J)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Multiply‑accumulate: (I×K) @ (K×J) -> (I×J)
        acc += tl.dot(left_tile, right_tile)

    # Write back FP16 result
    out_tile = acc.to(tl.float16)
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _tr_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched GEMM over the hidden dimension using the Triton kernel."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # [B, N, N, H] fp16

    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # Tile sizes – tuned for H100 Tensor‑core kernels
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _tr_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor   [B, N, N, dim]  (float32)
        torch.Tensor,          # mask tensor    [B, N, N]      (float32/bool) or None
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor   : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict  : name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``.
                          Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Weights (move to FP16 once)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate per side (single matmul each)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights for the “left” side
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # [2*H, dim]
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Linear: produces [B, N, N, 2*hidden_dim]
    left_combined = torch.nn.functional.linear(x, left_combined_w)   # fp16
    right_combined = torch.nn.functional.linear(x, right_combined_w)

    # Split projection and gate, apply sigmoid to the gate part
    left_proj = left_combined[..., :hidden_dim]                     # fp16
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])      # fp16

    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # Out‑gate (independent linear + sigmoid)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 3️⃣ Mask & gate fusion (element‑wise, cheap)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; broadcast over hidden dim
        mask_fp16 = mask_tensor.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via Triton GEMM kernel
    # ------------------------------------------------------------------
    out = _tr_mul(left, right)   # shape [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) and out‑gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # element‑wise gating (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Final linear back to model dim (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
