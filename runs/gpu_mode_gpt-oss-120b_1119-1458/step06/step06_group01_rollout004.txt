"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input in FP32 (numerically stable) → FP16.
2️⃣  Fuse the four linear + sigmoid “gate” projections (left_proj,
    left_gate, right_proj, right_gate) into a single FP16 matmul.
   The result is sliced into the four tensors, the gate tensors are
   passed through sigmoid.
3️⃣  If a mask is present, fuse mask·gate·projection with a tiny Triton
   kernel (one kernel per left/right tensor).  Otherwise a simple
   element‑wise multiplication is used.
4️⃣  Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM C = A @ Bᵀ using cuBLAS (torch.bmm).  The hidden
   dimension is folded into the batch dimension (B·H) for maximal Tensor‑Core
   throughput.
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear
   projection back to the model dimension.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

Only step 3 is written in Triton – everything else uses highly‑optimized
PyTorch kernels.  The fusion in step 2 removes three extra matmul launches
and reduces memory traffic of the 4‑D input, giving a substantial speed‑up.
"""

import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: fuse projection, gate and mask (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (mask is broadcast over the hidden dimension H)
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16 scalar mask

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask (mask broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim)

    All tensors must be fp16 CUDA tensors.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides expressed in number of elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tuned tile sizes – keep per‑block shared memory moderate while
    # preserving high occupancy.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
) -> torch.Tensor:
    """
    Compute the O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batched GEMM C = A @ Bᵀ using cuBLAS Tensor‑Core
    kernels.  The hidden dimension is folded into the batch dimension
    (B·H) to maximise throughput.

    Arguments
    ---------
    left, right : torch.Tensor of shape [B, N, N, H] and dtype fp16.

    Returns
    -------
    out : torch.Tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) – need contiguous layout for bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched matrix multiply A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore original layout: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 / bool) – may be ignored
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True ⇒ ignore mask
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # FP32
    x_fp16 = x.to(torch.float16)   # FP16 for the remainder

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear projections (left/right + their gates)
    # ------------------------------------------------------------------
    # Cast weights once to FP16 on the target device
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)

    # Stack the four weight matrices (row‑wise concatenation)
    # Shape after stacking: [4*hidden_dim, dim]
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )   # fp16

    # Single linear call producing all four tensors at once
    fused_out = torch.nn.functional.linear(x_fp16, fused_weight)  # [B, N, N, 4*hidden_dim]
    H = hidden_dim
    left_proj      = fused_out[..., :H]
    left_gate_pre  = fused_out[..., H:2*H]
    right_proj     = fused_out[..., 2*H:3*H]
    right_gate_pre = fused_out[..., 3*H:4*H]

    # Apply sigmoid to gate pre‑activations
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask–gate fusion (Triton) or simple element‑wise mul
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 (0.0 / 1.0) – broadcasted over hidden dim later
        mask_fp16 = mask_tensor.to(torch.float16)
        left  = _apply_mask_gate(left_proj,  left_gate,  mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # No mask: just multiply by the gates
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Clean up large intermediate tensors early to reduce peak memory
    del fused_out, left_proj, left_gate_pre, right_proj, right_gate_pre, left_gate, right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate (sigmoid) – linear on the original normalized input
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_pre = torch.nn.functional.linear(x_fp16, out_gate_w)   # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 7️⃣ Gate + final linear projection back to model dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                 # gating (FP16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    final_fp16 = torch.nn.functional.linear(gated, to_out_w)   # [B,N,N,dim] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the external harness)
    # ------------------------------------------------------------------
    return final_fp16.to(torch.float32)
