    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b)

    out = out * out_gate

    # Final projection
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast to fp32 and return
    return out.to(torch.float32)
```

Implementation details for kernel:

When we have left_comb shape [B,N,N,2H] containing the packed raw projection (first H) and gate (second H). We need to produce out shape [B,N,N,H] = proj * sigmoid(gate). For no mask version: no mask multiplication.

Since bottleneck might be memory loads for each tile, but we can have each thread block load the tile of combined projection (both proj and gate). Or we can load left_comb as two halves.

- Input: combined_ptr [B,N,N,2*H]. We'll treat stride_comb similarly.

- Output: out_ptr [B,N,N,H].

- Parameters: B,N,H combined (2*H). We'll need to know hidden_dim, but tile size for inner block is BLOCK_H.

- We'll compute offsets: offsets for combined dimensions: we need to split at index offset H.

Implementation:

```
@triton.jit
def _gate_kernel_no_mask(proj_comb_ptr, out_ptr,
                         B,N, H,
                         stride_comb_b, stride_comb_i, stride_comb_j, stride_comb_d,
                         stride_out_b, stride_out_i, stride_out_j, stride_out_d,
                         BLOCK_I: tl.constexpr,
                         BLOCK_J: tl.constexpr,
                         BLOCK_H: tl.constexpr):
    pid = ...
    # same block tiling over (i,j)
    # i and j offsets: offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    # offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N
    # for each block of H (i.e., hidden dimension):
    for tile_h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = tile_h * BLOCK_H + tl.arange(0, BLOCK_H)
        active_h = offs_h < H
        # compute pointer for projection part:
        # proj_comb index: (b, i, j, ptr) where ptr = offs_h
        proj_ptrs = proj_comb_ptr + pid_b * stride_comb_b + offs_i[:,None,None] * stride_comb_i + offs_j[None,:,None] * stride_comb_j + offs_h[None,None,:] * stride_comb_d
        proj_tile = tl.load(proj_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & active_h[None,None,:], other=0.0)
        # compute pointer for gate part:
        gate_ptrs = proj_comb_ptr + pid_b * stride_comb_b + offs_i[:,None,None] * stride_comb_i + offs_j[None,:,None] * stride_comb_j + (offs_h + H)[None,None,:] * stride_comb_d
        gate_tile = tl.load(gate_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & active_h[None,None,:], other=0.0)

        # apply sigmoid to gate tile
        gate_sig = 1.0 / (1.0 + tl.exp(-gate_tile))
        out_tile = proj_tile * gate_sig

        # store out tile
        out_ptrs = out_ptr + pid_b * stride_out_b + offs_i[:,None,None] * stride_out_i + offs_j[None,:,None] * stride_out_j + offs_h[None,None,:] * stride_out_d
        tl.store(out_ptrs, out_tile, mask=mask_i[:,None,None] & mask_j[None,:,None] & active_h[None,None,:])

**Kernel definitions:**

```
@triton.jit
def _gate_mask_kernel(comb_ptr, mask_ptr, out_ptr,
                      B, N, H,
                      stride_comb_b, stride_comb_i, stride_comb_j, stride_comb_h,
                      stride_mask_b, stride_mask_i, stride_mask_j,
                      stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                      BLOCK_I: tl.constexpr,
                      BLOCK_J: tl.constexpr,
                      BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load mask tile and broadcast to hidden dim
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

    # number of H tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection portion
        proj_ptrs = comb_ptr + pid_b * stride_comb_b + offs_i[:, None, None] * stride_comb_i + offs_j[None, :, None] * stride_comb_j + offs_h[None, None, :] * stride_comb_h
        proj_tile = tl.load(proj_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # Load gate (raw) portion: offset by H
        gate_offs = offs_h + H
        gate_ptrs = comb_ptr + pid_b * stride_comb_b + offs_i[:, None, None] * stride_comb_i + offs_j[None, :, None] * stride_comb_j + gate_offs[None, None, :] * stride_comb_h
        gate_raw_tile = tl.load(gate_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # sigmoid
        gate_sig_tile = 1.0 / (1.0 + tl.exp(-gate_raw_tile))

        # out tile = proj_tile * gate_sig_tile * mask
        out_tile = proj_tile * gate_sig_tile * mask_tile[:, :, None]

        # Store

"""
Outgoing TriMul (AlphaFold‑3) forward pass – fused linear+gate (+mask) Triton kernels
and a single batched GEMM for the O(N³) contraction.

Algorithm
---------
1️⃣  Layer‑norm the input (FP32) → cast to FP16.
2️⃣  Fuse each projection + its gate into a single Linear call
    (weights concatenated along the output axis).  This removes two
    matrix‑multiply launches per side and reduces the amount of data read
    from the large `[B,N,N,dim]` tensor.
3️⃣  A custom Triton kernel splits the concatenated tensor, applies a
    sigmoid to the gate, multiplies with the projection and (optionally)
    a pairwise mask.  The kernel works on 2‑D tiles of the `(i,j)` pair
    and a small hidden‑dim tile, keeping the data in registers and
    avoiding extra kernel launches.
4️⃣  The O(N³) contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    is performed as a batched GEMM `C = A @ Bᵀ`, where the hidden
    dimension is folded into the batch dimension: `[B*H, N, N]`.  This
    uses cuBLAS Tensor‑Core kernels (torch.bmm) – the fastest implementation
    for this operation on an H100.
5️⃣  A second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection
    back to the model dimension.
6️⃣  Cast the result to FP32 (the harness expects FP32).

All heavy element‑wise work (projection‑gate‑mask) is written in Triton;
the remaining linear and GEMM work uses highly‑optimised cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: split concatenated proj+gate, apply sigmoid(gate),
# optionally multiply by a mask which is broadcast over the hidden axis.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    comb_ptr,            # *fp16  [B, N, N, 2*H] (proj || gate_raw)
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_comb_b, stride_comb_i, stride_comb_j, stride_comb_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * sigmoid(gate_raw) * mask      (mask is broadcast over H)
    where comb = concat(proj, gate_raw) along the last axis.
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # Tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ---------- Projection part (first H entries) ----------
        proj_ptrs = (
            comb_ptr
            + pid_b * stride_comb_b
            + offs_i[:, None, None] * stride_comb_i
            + offs_j[None, :, None] * stride_comb_j
            + offs_h[None, None, :] * stride_comb_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # ---------- Gate raw part (second H entries) ----------
        gate_offs = offs_h + H
        gate_ptrs = (
            comb_ptr
            + pid_b * stride_comb_b
            + offs_i[:, None, None] * stride_comb_i
            + offs_j[None, :, None] * stride_comb_j
            + gate_offs[None, None, :] * stride_comb_h
        )
        gate_raw_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Apply sigmoid to gate
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_raw_tile))

        # Fusion: proj * sigmoid(gate) * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )

# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    The hidden dimension `h` is folded into the batch dimension so that
    the operation becomes a batched matrix multiply C = A @ Bᵀ.
    """
    B, N, _, H = left.shape
    # [B, H, N, N] → view as [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM (right transposed on the last two axes)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # FP32
    x = x.to(dtype_fp16)  # FP16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate weights (concat) and cast to FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(
        device=device, dtype=dtype_fp16
    )  # shape (2*hidden_dim, dim)

    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(
        device=device, dtype=dtype_fp16
    )

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Combined Linear + gate (with optional mask) for left/right sides
    # ------------------------------------------------------------------
    left_comb = torch.nn.functional.linear(x, left_comb_w)   # [B,N,N,2*hidden_dim]
    right_comb = torch.nn.functional.linear(x, right_comb_w)

    # out_gate (separate linear, then sigmoid)
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)
    out_gate = torch.sigmoid(out_gate_raw)

    if not nomask:
        # Convert mask to FP16 (broadcast over hidden)
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left = _apply_gate_and_mask(left_comb, mask_fp16, hidden_dim)
        right = _apply_gate_and_mask(right_comb, mask_fp16, hidden_dim)
    else:
        left = _apply_gate_no_mask(left_comb, hidden_dim)
        right = _apply_gate_no_mask(right_comb, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via a single batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden_dim] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) → out_gate → final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Cast back to FP32 (expected by caller)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
