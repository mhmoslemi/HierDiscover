"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1️⃣ Layer‑norm the input (FP32 → FP16) for stability.
2️⃣ Five bias‑less linear projections (left_proj, right_proj, left_gate,
   right_gate, out_gate) are performed in FP16.
3️⃣ If a mask is present we fuse mask·proj·gate in a Triton kernel,
   eliminating two extra memory passes.
4️⃣ Heavy O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is realized as a batched GEMM (C = A @ Bᵀ) with a custom Triton kernel.
   The batch dimension is B·H, so a single launch computes all hidden
   channels in parallel.
5️⃣ Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear
   projection back to the model dimension (FP16).
6️⃣ Cast the result back to FP32 (expected by the harness).

The implementation contains two Triton kernels:
* `_mask_gate_kernel`  – element‑wise fusion of projection, gating and mask.
* `_batched_matmul_transpose_kernel` – batched matrix multiplication of
  left @ rightᵀ across the hidden channels.
Both kernels are small, use shared‑memory tiling and Tensor‑Core‑friendly
BLOCK sizes and therefore provide a substantial speed‑up over the pure
PyTorch reference.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask   (mask broadcast over hidden dim H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H]  (already sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute: out = proj * gate * mask   (mask broadcast over hidden dim H)
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # tile over dim‑i
    pid_j = tl.program_id(2)          # tile over dim‑j

    # i / j coordinates for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H.
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16, shape [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles.
    # ------------------------------------------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16, shape [I,J,H]

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Fuse: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gating and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim H)

    Args:
        proj: [B, N, N, H] (fp16)
        gate: [B, N, N, H] (fp16, already sigmoid)
        mask: [B, N, N]    (fp16)

    Returns:
        out: fp16 tensor of same shape as proj/gate.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 8   # works well for H up to 384

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for out = left @ rightᵀ (both fp16).
# Batch dimension = B·H, matrices are N×N.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr,               # *fp16   [batch, M, K]
    b_ptr,               # *fp16   [batch, K, N]   (already transposed view of right)
    c_ptr,               # *fp16   [batch, M, N]
    M, N, K,
    stride_am, stride_ak,          # A: row‑stride, col‑stride
    stride_bk, stride_bn,          # B (transposed): row‑stride (K), col‑stride (N)
    stride_cm, stride_cn,          # C: row‑stride, col‑stride
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b] = A[b] @ B[b]ᵀ   for all batches simultaneously.
    All tensors are fp16; accumulation uses fp32 for numerical stability.
    """
    pid_batch = tl.program_id(0)   # batch index (0 … batch‑1)
    pid_m = tl.program_id(1)       # tile over rows   (M axis)
    pid_n = tl.program_id(2)       # tile over columns (N axis)

    # ------------------------------------------------------------------
    # Compute global offsets for the current tile.
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the batch‑specific slices.
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # ------------------------------------------------------------------
    # Accumulator in FP32.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles.
    for k in range(tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape [BLOCK_M, BLOCK_K]
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile: shape [BLOCK_K, BLOCK_N]
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate the partial product.
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result (cast back to fp16).
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_transpose_triton(
    left: torch.Tensor,   # [B, N, N, H] fp16
    right: torch.Tensor,  # [B, N, N, H] fp16
) -> torch.Tensor:
    """
    Batched GEMM using a custom Triton kernel.
    Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    Returns:
        out : [B, N, N, H] fp16 (contiguous)
    """
    B, N, _, H = left.shape
    # ------------------------------------------------------------------
    # Rearrange to shape [batch, M, K] with batch = B*H.
    # ------------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output.
    out_flat = torch.empty_like(left_flat)

    # Strides in elements (not bytes).
    s_left = left_flat.stride()
    s_right = right_flat.stride()
    s_out = out_flat.stride()

    # A = left_flat  : [batch, M=N, K=N]
    stride_am = s_left[1]      # stride from row (i) to next row
    stride_ak = s_left[2]      # stride from col (k) to next col
    stride_batch_a = s_left[0] # stride between batch entries

    # B = right_flat transposed (K,N) -> we treat it as transposed by swapping stride indices.
    # The original stride for dim2 (col) becomes the row‑stride after transposition.
    stride_bk = s_right[2]     # row‑stride after transpose (K dimension)
    stride_bn = s_right[1]     # column‑stride after transpose (N dimension)
    stride_batch_b = s_right[0]

    # C = output flat.
    stride_cm = s_out[1]
    stride_cn = s_out[2]
    stride_batch_c = s_out[0]

    # ------------------------------------------------------------------------------
    # Launch parameters – tuned for H100 (fp16 Tensor‑cores).
    # ------------------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H].
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient computation).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # Cast to fp16 for the rest of the pipeline

    # -----------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 on the target device
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # -----------------------------------------------------------------
    # 3️⃣ Projections (FP16)
    # -----------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 4️⃣ Gates (sigmoid, FP16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 5️⃣ Fuse mask and gates (if a mask is provided)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and broadcastable.
        mask_fp16 = mask_tensor.to(dtype_fp16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # No mask – simple element‑wise multiplication with gates.
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via a Triton batched GEMM.
    # -----------------------------------------------------------------
    # Result shape: [B, N, N, hidden_dim] (fp16)
    out = _batched_matmul_transpose_triton(left, right)

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final projection back to model dimension.
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (the harness expects float32).
    # -----------------------------------------------------------------
    return out.to(torch.float32)
