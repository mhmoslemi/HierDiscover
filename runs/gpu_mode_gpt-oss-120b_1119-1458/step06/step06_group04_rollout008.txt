"""
TriMul (outgoing) forward pass – highly‑fused implementation.

Algorithm
---------
1️⃣  LayerNorm over the last dimension (dim) – performed in FP32 for numerical
    stability (torch.nn.functional.layer_norm).
2️⃣  Cast the normalized tensor to FP16.
3️⃣  Fuse the five independent linear projections (left_proj, left_gate,
    right_proj, right_gate, out_gate) into a single GEMM by concatenating the
    weight matrices (shape (5*hidden_dim, dim)).  This reduces kernel launch
    overhead and maximises Tensor‑Core utilisation.
4️⃣  Split the concatenated output, apply the sigmoid to the three gating
    tensors.
5️⃣  If a mask is present, fuse the element‑wise multiplication of
    `left * left_gate` and `right * right_gate` with the mask using a tiny
    Triton kernel (`mask_gate_fuse_kernel`).  When there is no mask we simply
    multiply the tensors.
6️⃣  Core O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is equivalent to a batched GEMM (A @ Bᵀ) over the hidden dimension.
    We reshape the tensors to (B*hidden_dim, N, N) and use `torch.bmm`,
    which runs on cuBLAS with Tensor‑Cores.
7️⃣  Second LayerNorm over the hidden dimension (hidden_dim) – still in FP16.
8️⃣  Apply the output gate (`out_gate`) to the normalised tensor.
9️⃣  Final linear projection back to the original channel dimension (dim)
    – a single GEMM with weight `to_out.weight`.
🔟  Cast the result back to FP32 (the harness expects float32).

Only the mask‑gate fusion is written in Triton; all heavy matrix
multiplications use cuBLAS/Tensor‑Cores.  The implementation works for
any batch size (1 or 2), sequence length up to 1024, and both masked /
unmasked inputs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused multiplication with gating and optional mask.
# Input tensors are flattened half‑precision arrays; mask is a half‑array
# of shape [B * N * N] (broadcast over the hidden dimension).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → hidden dimension (H)
    total_elems,    # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load operands (zero where out‑of‑bounds)
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Derive the mask index (broadcast across H)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1.0

    # Fuse the three multiplications
    out = x * g * m

    # Store the result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the core O(N³) contraction.
# Operates entirely in FP16 using cuBLAS via torch.bmm.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the Tensor‑Mul contraction

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor, shape [B, N, N, H] (dtype=torch.float16)

    Returns
    -------
    out : torch.Tensor, shape [B, N, N, H] (dtype=torch.float16)
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched GEMM: A @ Bᵀ  (each slice is N×N @ N×Nᵀ → N×N)
    c = torch.bmm(a, b.transpose(1, 2))                    # [B*H, N, N]
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        left = left * left_gate

    # --------------------------------------------------------------
