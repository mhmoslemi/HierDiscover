```
def _batched_gemm_fp16(left, right):
    B, N, _, H = left.shape
    # left: [B,N,N,H]; after gating and mask
    # Reorder to (B*H,N,N) for bmm
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # shape: [B*H,N,N]
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # shape: [B*H,N,N]
    # compute bmm: a @ b^T => shape [B*H,N,N]
    c = torch.bmm(a, b.transpose(1, 2))  # [B*H, N, N]
    # reshape back
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel to fuse multiplication of tensor, gate and mask.
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 tensor
    gate_ptr,       # *fp16 tensor
    mask_ptr,       # *fp16 mask tensor
    hidden_dim,     # i32
    size,           # i32 total elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # indices
    mask = offs < size

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # compute mask index as integer division: each mask entry repeats hidden_dim times.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] using batched GEMM.
    All tensors are assumed to be fp16 and contiguous.
    """
    B, N, _, H = left.shape
    # shape left: (B,N,N,H) -> (B*H,N,N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # compute left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)
    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd) with fused projections and mask·gate kernel.
    Steps:
    1. Layer‑norm (float32) of the input and cast to fp16.
    2. Fuse the four linear projections (left/right) and three gates into a single matmul.
    3. Apply sigmoid on gates.
    4. Optional mask·gate fusion using a Triton kernel.
    5. Perform the cubic contraction via a batched GEMM (fp16).
    6. Second layer‑norm (fp16) on the hidden dimension.
    7. Apply the out‑gate and a final linear projection back to `dim`.
    8. Cast the result back to float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First layer‑norm (float32) → fp16
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_norm.to(torch.float16)   # [B,N,N,dim]

    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (2 proj + 3 gates) into a single matmul
    # -----------------------------------------------------------------
    # Collect and concatenate weight matrices (out_features, in_features)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden_dim, dim)
            weights["right_proj.weight"],  # (hidden_dim, dim)
            weights["left_gate.weight"],   # (hidden_dim, dim)
            weights["right_gate.weight"],  # (hidden_dim, dim)
            weights["out_gate.weight"],    # (hidden_dim, dim)
        ],
        dim=0,
    )  # shape (5*hidden_dim, dim)
    fused_weight = fused_weight.to(torch.float16)

    # Flatten x to (B*N*N, dim) and apply single linear operation
    x_flat = x.view(-1, dim)                         # (B*N*N, dim)
    fused_out = torch.nn.functional.linear(
        x_flat, fused_weight, bias=None
    )  # (B*N*N, 5*hidden_dim)

    # Reshape back to [B,N,N,5*hidden_dim]
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)

    # Slice the concatenated result
    left = fused_out[..., 0:hidden_dim]
    right = fused_out[..., hidden_dim:2 * hidden_dim]
    left_gate = fused_out[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = fused_out[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate = fused_out[..., 4 * hidden_dim:5 * hidden_dim]

    # Apply sigmoid to gating tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 3️⃣ Mask & gate fusion (in‑place) if mask is provided
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and flatten it – mask has shape [B,N,N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)           # shape (B*N*N)
        size = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)
        # In‑place fusion for left
        mask_gate_fuse_kernel[grid](
            left, left_gate, mask_flat,
            hidden_dim, size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # In‑place fusion for right
        mask_gate_fuse_kernel[grid](
            right, right_gate, mask_flat,
            hidden_dim, size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:

        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B,N,N,hidden_dim] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # -----------------------------------------------------------------
    # 6️⃣ Gating and final projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                               # element‑wise gating
    # Flatten and apply final linear map to go back to `dim`
    out_flat = out.view(-1, hidden_dim)                # (B*N*N, hidden_dim)
    out_proj = torch.nn.functional.linear(
        out_flat, weights["to_out.weight"].to(torch.float16), bias=None
    )  # (B*N*N, dim)
    out_tensor = out_proj.view(B, N, N, dim)           # reshape

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out_tensor.to(torch.float32)

"""
custom_kernel – Outgoing TriMul forward (no autograd)

Algorithm (high‑level):
1️⃣  First LayerNorm on the input (float32) → cast to FP16.
2️⃣  Fuse the five linear maps (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single matmul:
       x_flat @ W_fused.T   where  W_fused ∈ ℝ^{5·H×D}.
    Slice the result to obtain the four tensors and three gates.
3️⃣  Apply sigmoid to the three gate tensors.
4️⃣  If a mask is supplied, fuse ``tensor * gate * mask`` in‑place
    with a Triton kernel (mask_gate_fuse_kernel).  Otherwise do a
    simple element‑wise multiplication.
5️⃣  Heavy O(N³) contraction (out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h])
    via a batched GEMM on FP16 (torch.bmm on a reshaped view).
6️⃣  Second LayerNorm over the hidden dimension (FP16).
7️⃣  Multiply by the out‑gate, flatten and apply the final linear
    projection back to the original channel dimension.
8️⃣  Cast the result to float32 (required by the harness).

All heavy compute is performed in FP16.  The mask·gate fusion is the
only custom Triton kernel, which eliminates a separate launch for each
tensor.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse tensor * gate * mask (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 input tensor (flattened)
    gate_ptr,       # *fp16 gate tensor (same shape as x)
    mask_ptr,       # *fp16 mask tensor (flattened, shape [B·N·N])
    hidden_dim,     # i32 hidden dimension H
    size,           # i32 total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load a tile of the tensors
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # One mask entry corresponds to `hidden_dim` contiguous elements in x/g.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fuse the three multiplications
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the cubic contraction (left @ rightᵀ)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
    Inputs are (B, N, N, H) fp16. Returns (B, N, N, H) fp16.
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] where each (b,h) pair is a separate matrix
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # matrix multiply a @ bᵀ for each batch entry
    c = torch.bmm(a, b.transpose(1, 2))                     # (B·H, N, N)
    # reshape back to (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Parameters
    ----------
    data : Tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights      : dict of model parameters (FP32)
        - config       : dict containing "dim", "hidden_dim", and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)
    x = x_norm.to(torch.float16)                        # [B, N, N, dim]

    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (proj + gates) into one matmul
    # -----------------------------------------------------------------
    # Concatenate the five weight matrices (shape: hidden_dim × dim each)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden_dim, dim)
            weights["right_proj.weight"],  # (hidden_dim, dim)
            weights["left_gate.weight"],   # (hidden_dim, dim)
            weights["right_gate.weight"],  # (hidden_dim, dim)
            weights["out_gate.weight"],    # (hidden_dim, dim)
        ],
        dim=0,
    ).to(torch.float16)                                 # (5·H, dim)

    # Flatten x to (B·N·N, dim) and apply the single linear operation
    x_flat = x.view(-1, dim)                            # (B·N·N, dim)
    fused_out = torch.nn.functional.linear(
        x_flat, fused_weight, bias=None
    )                                                   # (B·N·N, 5·H)

    # Reshape back to 4‑D and slice
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)
    left       = fused_out[..., 0:hidden_dim]
    right      = fused_out[..., hidden_dim:2 * hidden_dim]
    left_gate  = fused_out[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = fused_out[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate   = fused_out[..., 4 * hidden_dim:5 * hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣ Apply sigmoid to gate tensors
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – fuse multiplication in‑place via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and flatten (one entry per (b,i,j) pair)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                # (B·N·N)
        size = left.numel()                           # B·N·N·H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (batched GEMM, FP16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # (B, N, N, hidden_dim) fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # -----------------------------------------------------------------
    # 7️⃣ Gating and final projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                                   # element‑wise
    out_flat = out.view(-1, hidden_dim)                    # (B·N·N, hidden_dim)

    out_proj = torch.nn.functional.linear(
        out_flat,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )                                                       # (B·N·N, dim)

    out_tensor = out_proj.view(B, N, N, dim)                # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out_tensor.to(torch.float32)
