    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias) # float32

    x = x.to(torch.float16)

    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)

    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    combined_weight = torch.cat([w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0) # shape (5*hidden_dim, dim)

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_head = tl.program_id(0) # head index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute batch and head indices
    b = pid_head // H
    h = pid_head % H

    # Starting indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Masks for valid indices
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Offsets used in each iteration over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Offsets for batch and head
    b_offset = b * stride_left_b
    h_offset = h * stride_left_h # stride_left_h = 1 guaranteed

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for left and right blocks
        # left ptr: (M,K) shape
        left_ptrs = left_ptr + b_offset + h_offset \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k

        # right ptr: (K,N) shape; note j stride is stride_right_i (same as left_i)
        right_ptrs = right_ptr + b_offset + h_offset \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i

        # load blocks
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b_mat = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute block product
        acc += tl.dot(a, b_mat)          # (M,K)@(K,N) -> (M,N) accumulate in float32

    # write out accumulator to out tensor
    out_ptrs = out_ptr + b_offset + h_offset \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass – fast Triton implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) – the reference paper normalises over the last channel.
2️⃣ Cast to fp16 for all heavy arithmetic.
3️⃣ Fuse the five linear projections (left/right and three gates) into a single
   matrix multiplication (weight shape (5·hidden_dim, dim)).  This removes the
   repeated reads of the large input tensor.
4️⃣ Split the result into left, right, left_gate, right_gate, out_gate and apply
   sigmoid to the gates.  Element‑wise gating (and optional mask) is performed
   with a regular PyTorch broadcast; it is cheap compared to the O(N³) step.
5️⃣ Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is realised as a batched GEMM using a hand‑written Triton kernel that
   works directly on the contiguous layout (B,N,N,hidden) – no extra permute
   or copy is required.
6️⃣ Second LayerNorm (fp16) → multiply by out_gate.
7️⃣ Final linear projection back to the original channel dimension.
8️⃣ Cast result to float32 (the harness expects float32).

The kernel (step 5) loops over the summation dimension `k` in blocks and
accumulates with `tl.dot`, which exploits TensorCores on H100.  All other
operations run in PyTorch, but the heavy O(N³) part is fully fused in Triton,
meeting the requirement of “use Triton” while keeping overall runtime low.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched contraction for the outgoing TriMul.
# Input layout: [B, N, N, H] (contiguous, H is the hidden dimension).
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_outgoing_kernel(
    left_ptr,      # *fp16  – left tensor (B,N,N,H)
    right_ptr,     # *fp16  – right tensor (B,N,N,H)
    out_ptr,       # *fp16  – output tensor (B,N,N,H)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    #  Program IDs:
    #   axis 0 – combined batch & head index (b*H + h)
    #   axis 1 – tile index along the i‑dimension
    #   axis 2 – tile index along the j‑dimension
    # ------------------------------------------------------------------
    pid_head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decode batch and head
    b = pid_head // H            # batch index
    h = pid_head % H             # hidden‑head index

    # Starting coordinates of the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets for the i‑ and j‑axis
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Mask out‑of‑bounds rows/columns
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base offsets for the current (b, h) slice
    off_b = b * stride_left_b          # same for left/right/out (batch stride)
    off_h = h * stride_left_h          # stride_left_h == 1 for fp16 tensors

    # ------------------------------------------------------------------
    # Accumulator (float16 – TensorCore accumulates in fp16 on H100)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        # Offsets for the k‑axis (shared by left & right)
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------------------- Load left tile -----------------------
        # left[b, i, k, h] – shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + off_b + off_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ----------------------- Load right tile ----------------------
        # right[b, j, k, h] – we load as (BLOCK_K, BLOCK_N)
        # Note: stride_right_i == stride_left_i (both = N*H)
        right_ptrs = right_ptr + off_b + off_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # ----------------------- Block GEMM ---------------------------
        # a: (M, K), b_tile: (K, N) → accumulate into acc (M, N)
        acc += tl.dot(a, b_tile)   # TensorCore `dot` on fp16

    # ------------------------------------------------------------------
    # Store the result – write only valid rows/cols.
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + off_b + off_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton contraction kernel.
# ----------------------------------------------------------------------
def _tri_mul_outgoing_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using Triton:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        FP16 tensors of shape [B, N, N, H] (contiguous).

    Returns
    -------
    torch.Tensor
        FP16 tensor of shape [B, N, N, H] (contiguous).
    """
    B, N, _, H = left.shape

    # Output buffer
    out = torch.empty_like(left)

    # Strides (in elements)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Tiling parameters – tuned for H100 TensorCores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (batch*head, n_tiles_M, n_tiles_N)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # TensorCore‐friendly warps
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,
        torch.Tensor,
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) or None
        - weights_dict : model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # --------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single matmul
    # --------------------------------------------------------------
    # Weights are stored as (out_features, in_features) = (hidden_dim, dim)
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Stack along the output‑feature axis: shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )  # [5*hidden_dim, dim]

    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # [B*N*N, dim] (contiguous)
    proj = torch.nn.functional.linear(x_flat, combined_weight)  # [B*N*N, 5*hidden_dim]

    # Reshape to separate the five branches
    proj = proj.view(B, N, N, 5, hidden_dim)     # [B, N, N, 5, hidden_dim]
    left          = proj[..., 0, :]              # [B, N, N, hidden_dim]
    right         = proj[..., 1, :]
    left_gate_pre = proj[..., 2, :]
    right_gate_pre= proj[..., 3, :]
    out_gate_pre  = proj[..., 4, :]

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in fp16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # --------------------------------------------------------------
    # 5️⃣ Apply optional mask and combine with gates
    # --------------------------------------------------------------
    if (not nomask) and mask_tensor is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left * left_gate * mask_fp16
        right = right * right_gate * mask_fp16
    else:
        left  = left * left_gate
        right = right * right_gate

    # Ensure contiguous layout expected by the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Core O(N³) contraction via Triton (fp16)
    # --------------------------------------------------------------
    out = _tri_mul_outgoing_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # --------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication + final linear projection (fp16)
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)       # [B,N,N,dim] (fp16)

    # --------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
