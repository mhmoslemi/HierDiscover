"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input in FP32 → cast to FP16.
2️⃣ Compute linear projections (`left_proj`, `right_proj`) and gating
   projections (`left_gate`, `right_gate`, `out_gate`) in FP16.
3️⃣ If a mask is supplied we fuse the three element‑wise ops
   `proj * gate * mask` with a tiny Triton kernel (`mask_gate_fuse_kernel`);
   otherwise we just multiply the projection with its gate.
4️⃣ Heavy contraction  
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  
   is performed as a batched GEMM (A @ Bᵀ) using cuBLAS Tensor‑Core kernels.
   The tensors are reshaped to `[B·H, N, N]` (H = hidden_dim) and a single
   `torch.bmm` call does all heads at once.
5️⃣ Second Layer‑norm (FP16) → element‑wise out‑gate multiplication.
6️⃣ Final linear projection back to the original channel dimension.
7️⃣ Cast the result back to FP32 (the harness expects FP32).

Only the forward pass is required; no gradients are computed.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, size = total_elements
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor, shape (B·N·N,)
    total_elements,  # i32 : number of elements in x
    hidden_dim,      # i32 : width of the hidden dimension (last dim of x)
    BLOCK: tl.constexpr,
):
    """In‑place fuse `x = x * gate * mask`.  The mask is broadcast over the
    hidden dimension, therefore each `hidden_dim` consecutive entries share the
    same mask value."""
    pid = tl.program_id(0)                         # block id
    offs = pid * BLOCK + tl.arange(0, BLOCK)       # linear offsets handled by this block
    mask = offs < total_elements                    # guard for tail

    # Load projection and gate values
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index: every `hidden_dim` elements share the same mask entry
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1.0 → no masking

    # Fuse multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    via a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape                       # left.shape = [B, N, N, H]

    # reshape to [B*H, N, N] where each batch item is a single head
    # we need a contiguous layout for the batched GEMM
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_ = torch.bmm(left_, right_.transpose(1, 2))

    # Restore original 4‑D shape [B, N, N, H]
    out = out_.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      shape [B, N, N]    (float32/bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`; optional

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"] # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm expects the normalized shape,
    # i.e. the last dimension.
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)
    x = x_norm.half()                # continue in FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights once (avoid many .half() calls)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Gating streams (sigmoid, still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gating (Triton kernel) if a mask is present
    # ------------------------------------------------------------------
    if nomask:

        left  = left.mul_(left_gate)       # in‑place
        right = right.mul_(right_gate)
    else:
        # Ensure tensors are contiguous for the Triton kernel.
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        # mask → broadcasted over hidden_dim, flatten to 1‑D
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)   # (B·N·N,)

        total_elements = left.numel()                # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Fuse left stream: left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right stream: right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed after fusion
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction (batched GEMM) – cuBLAS Tensor‑Core kernel
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the evaluation harness
    # ------------------------------------------------------------------
    return out.float()
```
