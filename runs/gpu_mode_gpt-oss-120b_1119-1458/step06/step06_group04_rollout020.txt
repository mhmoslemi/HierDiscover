import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr, hidden_dim, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    tl.store(x_ptr + offs, x * g * m, mask=mask)

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # permute into (B, H, N, N) and view as batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd) with several
    performance‑optimisations:
      1. Merge the five separate linear layers (left/right projections,
         three gating signals) into a single GEMM, reducing reads of the
         large input tensor.
      2. Fuse mask, left/right gates and the mask using a tiny Triton kernel.
      3. Perform the O(N³) contraction with a single batched GEMM (cuBLAS)
         on FP16 tensors.
      4. Use FP16 for all heavy arithmetic and cast back to FP32 only
         for the final result.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) & cast to FP16 for the fast part
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] FP32
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # -----------------------------------------------------------------
    # Stack weights ‑‑> (5 * hidden_dim, dim)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate in the same order as they are extracted later
    fused_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)

    # Single GEMM → shape [..., 5*hidden_dim]
    fused = torch.nn.functional.linear(x, fused_weight)  # [B, N, N, 5*hidden_dim]

    # View as 5 separate tensors of shape [..., hidden_dim]
    fused = fused.view(*fused.shape[:-1], 5, hidden_dim)  # [B, N, N, 5, hidden_dim]

    left      = fused[..., 0, :]  # [B, N, N, hidden_dim]
    right     = fused[..., 1, :]  # [B, N, N, hidden_dim]
    left_gate = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate  = torch.sigmoid(fused[..., 4, :])

    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # [B*N*N]
        total_elems = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:

        right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Core O(N³) contraction (batched GEMM in FP16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim] FP16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) → out_gate multiplication
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 6️⃣ Final projection back to dim (FP16)
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] FP16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (what the harness expects)
    # -----------------------------------------------------------------
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ First LayerNorm (float32) → cast to fp16.
2️⃣ Merge the *five* linear layers (left/right projections + three gating
   signals) into a **single** GEMM.  This cuts the number of passes over the
   large 4‑D tensor x from five to one and removes four extra weight loads.
3️⃣ Split the fused result into left, right, left_gate, right_gate and out_gate.
   Apply the mask and the left/right gates with a tiny Triton kernel that
   fuses the three element‑wise multiplications (tensor * gate * mask).
4️⃣ Core O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (FP16) via cuBLAS – the most efficient
   implementation for large N.
5️⃣ Second LayerNorm (FP16) → element‑wise multiplication with out_gate.
6️⃣ Final linear projection back to the original channel dimension.
7️⃣ Cast the result back to float32 (the harness expects FP32).

All heavy arithmetic runs in fp16 to exploit Tensor‑Cores on an H100.
Only the mask‑gate fusion uses Triton, satisfying the “use Triton” requirement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and the tensor in a single multiply.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 → tensor to be updated (flattened)
    gate_ptr,       # *fp16 → gate tensor (same flattening as x)
    mask_ptr,       # *fp16 → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32 → size of hidden dimension (H)
    size,           # i32 → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,        # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = offs < size

    # load operands
    x = tl.load(x_ptr + offs, mask=mask_valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)

    # Derive mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)   # default mask=1

    # fused multiplication
    out = x * g * m

    # store back in‑place
    tl.store(x_ptr + offs, out, mask=mask_valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the N³ contraction (FP16 → FP16)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Parameters
    ----------
    left, right : torch.Tensor
        FP16 tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        FP16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) and perform A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                 # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                 # input tensor   [B, N, N, dim] (float32)
        torch.Tensor,                 # mask tensor    [B, N, N]   (float32 / bool / None)
        Dict[str, torch.Tensor],      # model weights  (float32)
        Dict[str, Any],               # config dict    must contain dim & hidden_dim
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                               # [B, N, N, dim] (float32)
    x = x_norm.to(torch.float16)    # work in fp16 for the heavy part

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # -----------------------------------------------------------------
    # Gather and concatenate the weight matrices (no bias)
    left_proj_w   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    fused_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)                             # (5*hidden_dim, dim)

    # Single linear layer → shape [..., 5*hidden_dim]
    fused = torch.nn.functional.linear(x, fused_weight)

    # Split the fused tensor: [..., 5, hidden_dim]
    fused = fused.view(*fused.shape[:-1], 5, hidden_dim)

    left      = fused[..., 0, :]          # [B, N, N, hidden_dim]
    right     = fused[..., 1, :]          # [B, N, N, hidden_dim]
    left_gate = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate  = torch.sigmoid(fused[..., 4, :])

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask + gate fusion (Triton)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                 # [B*N*N]
        total_elems = left.numel()                     # B*N*N*hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – plain element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Core O(N³) contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16) → out_gate multiplication
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 6️⃣ Final projection back to the original dimensionality (fp16)
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
