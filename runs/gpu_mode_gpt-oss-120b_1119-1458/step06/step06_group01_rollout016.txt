@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    # shape parameters
    B, H, N,
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr 
):
    pid = tl.program_id(0) # batch = combined B*H
    pid_i = tl.program_id(1) # output i tile
    pid_j = tl.program_id(2) # output j tile

    # Map pid into batch and hidden indices
    total_batch = B * H
    b_idx = pid // H
    h_idx = pid % H

    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks for valid i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Offsets for batch
    base_batch_offset = b_idx * stride_left_b + h_idx * stride_left_h
    base_right_batch_offset = b_idx * stride_right_b + h_idx * stride_right_h

    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers for left tile (A)
        left_ptrs = left_ptr + base_batch_offset + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute pointers for right tile (B)
        # Note: right shape: [B, N_j, N_k, H]; we treat rows as k (offs_k) and columns as j (offs_j)
        right_ptrs = right_ptr + base_right_batch_offset + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)   # shape (BLOCK_K, BLOCK_N)

        # Compute tile multiplication
        acc += tl.dot(a, b)

    # Write results to output
    out_ptrs = out_ptr + b_idx * stride_out_b + h_idx * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm Overview
------------------
1️⃣  Layer‑norm the input (FP32) → cast to FP16.
2️⃣  Fuse the five linear projections (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a **single** GEMM:
    a single weight matrix of shape (5·hidden_dim, dim) is
    multiplied by the input, then the result is split.
3️⃣  Apply sigmoid to the three gate tensors.
4️⃣  If a mask is present, fuse mask, projection and gate
    multiplication in a tiny Triton kernel (`_mask_gate_kernel`);
    otherwise perform simple element‑wise multiplication.
5️⃣  Perform the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    with a custom Triton kernel (`_tri_mul_contract_kernel`) that
    works directly on the original [B,N,N,H] layout, avoiding the
    costly `permute‑>contiguous` copies required by a naïve torch
    implementation.
6️⃣  Second LayerNorm (FP16) → gate‑scale with `out_gate`.
7️⃣  Final linear projection back to `dim` (FP16).
8️⃣  Cast the result to float32 (required by the harness).

Only the mask‑gate fusion and the cubic contraction are implemented
in Triton, satisfying the “must‑use‑Triton” requirement while keeping
the heavy computation on the GPU tensor‑core pipeline.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuses projection * gate * mask (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H]
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load mask (BLOCK_I × BLOCK_J), broadcast later over H
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

    # Loop over hidden dim
    num_tiles_h = tl.cdiv(H, BLOCK_H)
    for t_h in range(num_tiles_h):
        offs_h = t_h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection slice
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Load gate slice (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask   (mask broadcast over hidden dim)

    All tensors must be FP16, CUDA, same shape [B, N, N, H] for proj/gate
    and [B, N, N] for mask.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched O(N³) contraction left * rightᵀ
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_contract_kernel(
    left_ptr,   # *fp16  [B, N, N, H]
    right_ptr,  # *fp16  [B, N, N, H]
    out_ptr,    # *fp16  [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    for all b ∈ [0,B), h ∈ [0,H).
    """
    pid_batch = tl.program_id(0)   # combined batch*hidden index
    pid_i = tl.program_id(1)       # i‑tile
    pid_j = tl.program_id(2)       # j‑tile

    # Decode batch and hidden indices
    b_idx = pid_batch // H
    h_idx = pid_batch % H

    # Tile offsets in i and j dimensions
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers that include batch and hidden offsets
    left_base = left_ptr + b_idx * stride_left_b + h_idx * stride_left_h
    right_base = right_ptr + b_idx * stride_right_b + h_idx * stride_right_h
    out_base = out_ptr + b_idx * stride_out_b + h_idx * stride_out_h

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k‑dimension in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_k_tiles):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = (
            left_base
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load right tile (transposed on the fly): shape (BLOCK_K, BLOCK_J)
        right_ptrs = (
            right_base
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        b = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate the product
        acc += tl.dot(a, b)

    # Write result (cast back to FP16)
    out_ptrs = (
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _contract_left_right(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched O(N³) contraction implemented in Triton.
    left/right: [B, N, N, H] (FP16, contiguous)
    Returns: out of shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides (in elements, not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        B * H,                      # combined batch*hidden dimension
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _tri_mul_contract_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0],  # stride_left_b
        s_left[1],  # stride_left_i
        s_left[2],  # stride_left_k
        s_left[3],  # stride_left_h
        s_right[0], # stride_right_b
        s_right[1], # stride_right_j (same as i stride)
        s_right[2], # stride_right_k
        s_right[3], # stride_right_h
        s_out[0],   # stride_out_b
        s_out[1],   # stride_out_i
        s_out[2],   # stride_out_j
        s_out[3],   # stride_out_h
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool) – may be all‑ones
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # still FP32
    x = x.to(torch.float16)   # FP16 for the rest of the forward pass

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # ------------------------------------------------------------------
    # Gather weight tensors and cast to FP16
    w_left_proj  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    w_left_gate  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_out_gate   = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Concatenate along the out_features dimension: shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )

    # Single linear projection (no bias)
    combined = torch.nn.functional.linear(x, combined_weight)  # [B, N, N, 5*hidden_dim] (FP16)

    # Split into the five tensors we need
    H = hidden_dim
    left_proj   = combined[..., 0 * H : 1 * H]
    right_proj  = combined[..., 1 * H : 2 * H]
    left_gate_z = combined[..., 2 * H : 3 * H]  # pre‑sigmoid
    right_gate_z = combined[..., 3 * H : 4 * H]
    out_gate_z   = combined[..., 4 * H : 5 * H]

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_z)
    right_gate = torch.sigmoid(right_gate_z)
    out_gate  = torch.sigmoid(out_gate_z)

    # --------------------------------------------------------------
    # 4️⃣ Apply optional mask (fused with projection & gate)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # Simple element‑wise gating when no mask is present
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via custom Triton kernel
    # --------------------------------------------------------------
    # left/right are [B, N, N, H] FP16
    out_mid = _contract_left_right(left, right)   # [B, N, N, H] FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) → out‑gate scaling
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out_mid = torch.nn.functional.layer_norm(
        out_mid,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out_mid = out_mid * out_gate

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to model dimension
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out_mid, w_to_out)   # [B, N, N, dim] FP16

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
