Define kernel:

```
@triton.jit
def tri_mul_batched_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # batch dimension (B*H)
    pid_i = tl.program_id(1)      # block i
    pid_j = tl.program_id(2)      # block j

    # Compute b and h from pid_batch
    batch_idx = pid_batch // H
    h_idx = pid_batch % H

    # Compute offsets to the start of the left and right matrices for this batch & hidden channel
    left_base = left_ptr + batch_idx * stride_left_batch + h_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_batch + h_idx * stride_right_h
    out_base = out_ptr + batch_idx * stride_out_batch + h_idx * stride_out_h

    # Compute starting row and col for left/out
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Compute range within N
    i_range = i_offset + tl.arange(0, BLOCK_M)
    j_range = j_offset + tl.arange(0, BLOCK_N)

    # Mask within bounds
    mask_i = i_range < N
    mask_j = j_range < N

    # Compute pointers for left (i rows) and right (j rows)
    # left pointer: offset for i and k
    # left[i_range, k_range] shape (BLOCK_M, BLOCK_K)
    # right pointer: for j rows and k columns (but we treat as transposed)
    # right_T's i = j_original, j = k_original after swapping stripes.

    # allocate accumulator for each i,j element (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_offset in range(0, N, BLOCK_K):
        # range for k block
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # compute offsets pointers for left and right
        # left offset: left_base + i_range[:,None] * stride_left_i + k_range[None,:] * stride_left_k
        left_ptrs = left_base + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        # Right as transposed: right_base + k_range[:, None] * stride_right_i + j_range[None, :] * stride_right_k (swap i and j)
        # Because originally right stride_i is row j stride, stride_k is col k stride. For transposed, row stride becomes original col stride, column stride becomes original row stride.
        right_ptrs = right_base + k_range[:, None] * stride_right_i + j_range[None, :] * stride_right_k

        # Load sub-blocks
        left_block = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right_block = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate multiplication
        acc += tl.dot(left_block, right_block)  # left: (M,K), right: (K,N) -> (M,N)

    # Write out to out tensor (per hidden channel)
    out_ptrs = out_base + i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

@triton.jit
def tri_mul_batched_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute b and hidden channel
    b = pid_batch // H
    h = pid_batch % H

    # base pointers
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # offsets for i and j
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    i_range = i_offset + tl.arange(0, BLOCK_M)
    j_range = j_offset + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of blocks along K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k_blocks):
        k_offset = k_iter * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # pointers
        # left: rows i, cols k
        a_ptrs = left_base + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        # right transposed: rows k (original col), cols j (original row)
        b_ptrs = right_base + k_range[:, None] * stride_right_i + j_range[None, :] * stride_right_k

        # load with masks
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, eviction_policy='evict_last')
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, eviction_policy='evict_last')

        # promote to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # matrix multiply tile (M,K)*(K,N) -> (M,N)
        acc += tl.dot(a, b)

    # write to output
    out_ptrs = out_base + i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass – highly optimized version.

Algorithm
---------
1. Layer‑norm the input (float32 → fp32 stability) and cast to fp16.
2. Linear projections → `left`, `right` (fp16, shape [B,N,N,H]).
3. Compute gating tensors `left_gate`, `right_gate`, `out_gate` (fp16 sigmoid).
4. If a mask is present, fuse ``tensor * gate * mask`` via a tiny Triton kernel
   (in‑place on `left` and `right`).  Otherwise do a plain element‑wise multiply.
5. Heavy O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom batched‑matmul Triton kernel that works on the
   original `[B,N,N,H]` layout – no costly permutes or copies.
6. Second LayerNorm (fp16) over the hidden dimension.
7. Element‑wise gating with `out_gate`.
8. Final linear projection back to the original channel dimension.
9. Cast the result to float32 (required by the harness).

All heavy compute (steps 4‑5) is done on the GPU with Triton, while
lightweight PyTorch ops (layer‑norm, sigmoid, final linear) are kept in
FP16 for speed and accuracy.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1 – fuse mask * gate * tensor (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated (flattened)
    gate_ptr,       # *fp16  gate tensor (same shape as x)
    mask_ptr,       # *fp16  mask tensor (flattened, shape [B*N*N])
    hidden_dim,     # i32   hidden dimension H
    size,           # i32   total number of elements in x = B*N*N*H
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands (fp16)
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Each mask entry corresponds to `hidden_dim` consecutive elements.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fuse multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2 – batched GEMM for the O(N³) contraction
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batched_kernel(
    left_ptr,       # *fp16 left  tensor, shape [B,N,N,H]
    right_ptr,      # *fp16 right tensor, shape [B,N,N,H]
    out_ptr,        # *fp16 output tensor, shape [B,N,N,H]
    B, N, H,        # i32 problem size
    # strides for left   (batch, i, k, h)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides for right  (batch, j, k, h) – we will read it transposed
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    # strides for out    (batch, i, j, h)
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)   # combined batch * hidden channel
    pid_i = tl.program_id(1)       # tile row (i)
    pid_j = tl.program_id(2)       # tile col (j)

    # Decompose batch & hidden‑channel
    b = pid_batch // H
    h = pid_batch % H

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # Global coordinates of the tile
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    i_range = i_offset + tl.arange(0, BLOCK_M)
    j_range = j_offset + tl.arange(0, BLOCK_N)

    i_valid = i_range < N
    j_valid = j_range < N

    # Accumulator (fp32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k_offset = k_iter * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        k_valid = k_range < N

        # Load A = left[i,k]  – shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i_range[:, None] * stride_left_i \
                              + k_range[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=i_valid[:, None] & k_valid[None, :],
                    other=0.0)
        a = a.to(tl.float32)

        # Load Bᵀ = rightᵀ[k,j]  – shape (BLOCK_K, BLOCK_N)
        #   right is stored as (b, j, k, h); we need it transposed.
        b_ptrs = right_base + k_range[:, None] * stride_right_i \
                                 + j_range[None, :] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & j_valid[None, :],
                    other=0.0)
        b = b.to(tl.float32)

        # Rank‑K update
        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)

    # Write result (cast back to fp16)
    out_ptrs = out_base + i_range[:, None] * stride_out_i \
                          + j_range[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=i_valid[:, None] & j_valid[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd). Returns a float32 tensor
    of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B,N,N,dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Switch to fp16 for the compute‑heavy part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights once to fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w) # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (fp16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask is boolean / 0‑1 – cast to fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                       # [B*N*N]
        size = left.numel()                                   # B*N*N*hidden_dim
        BLOCK = 1024                                          # ≤ 1024 threads per block
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left, left_gate, mask_flat,
            hidden_dim, size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask  (in‑place)
        mask_gate_fuse_kernel[grid](
            right, right_gate, mask_flat,
            hidden_dim, size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just apply the gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – custom Triton batched GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden_dim
    out = torch.empty_like(left)          # allocate output ([B,N,N,H] fp16)

    # Strides (int) – left/right share the same layout
    s_left = left.stride()                # (batch, i, k, h)
    s_right = right.stride()
    s_out = out.stride()

    # Choose tile sizes (tuned for typical N, H)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:  # N up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64

    grid = (
        B * hidden_dim,                     # batch * hidden-channel dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles along j
    )

    tri_mul_batched_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # enough warps to keep the tensor cores busy
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Gating + final projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # shape [B,N,N,dim]

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
