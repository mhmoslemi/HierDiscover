"""
TriMul (outgoing) – fast Triton‑fused implementation
====================================================

The forward pass consists of three stages:

1️⃣  **First LayerNorm + linear projections** – performed with PyTorch
    (highly‑optimised cuBLAS kernels).  This gives the tensors
    `left_proj`, `right_proj`, `left_gate`, `right_gate`,
    `out_gate` in fp16.

2️⃣  **Mask‑gate fusion** – a tiny Triton kernel fuses
    ``proj * gate * mask`` for *left* and *right* (mask is broadcast
    over the hidden dimension).  When ``nomask=True`` the kernel is
    skipped and the element‑wise product is performed directly in‑place.

3️⃣  **Heavy O(N³) contraction** – realised as a batched GEMM
    ``C = A @ Bᵀ`` where ``A = left`` and ``B = right``.
    The hidden dimension is folded into the batch dimension
    (shape ``[B·H, N, N]``) and the multiplication is performed by
    ``torch.bmm`` (cuBLAS tensor‑core kernel).

4️⃣  **Second LayerNorm + out‑gate + final linear** – a second Triton
    kernel fuses the three operations:
        out = LayerNorm(out) ⊙ out_gate
        out = out @ to_out.weightᵀ
    The kernel processes one (batch, i, j) element at a time,
    computes mean/variance across the hidden dimension in fp32,
    applies the learned LN scale/bias, multiplies by the out‑gate,
    and finally projects back to ``dim``.  The result is returned as
    fp32.

The implementation respects the original PyTorch semantics (including
eps=1e‑5 for LayerNorm and sigmoid for gates) while dramatically reducing
the number of memory traversals.  No gradients are computed – only the
forward pass is required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: mask * gate * projection (used for left/right tensors)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]
    gate_ptr,            # *fp16  [B, N, N, H]
    mask_ptr,            # *fp16  [B, N, N]
    out_ptr,             # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Fuse `proj * gate * mask` where mask is broadcast over the hidden dim H.
    Each program processes a tile (BLOCK_I × BLOCK_J) of positions and loops
    over the hidden dimension.
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # tile i coordinate
    pid_j = tl.program_id(2)   # tile j coordinate

    # ------------------------------------------------------------------
    # Offsets for the spatial dimensions
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load a (BLOCK_I × BLOCK_J) mask tile – broadcast over H later
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # ------------------------------------------------------------------
    # Loop over hidden dimension tiles
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ----- projection -------------------------------------------------
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # ----- gate -------------------------------------------------------
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # ----- fuse -------------------------------------------------------
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # ----- store -------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask:
        out = proj * gate * mask          (mask broadcast over hidden dim)

    All arguments must be fp16 CUDA tensors.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm (over hidden dim) + out‑gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def _final_norm_gate_proj_kernel(
    out_ptr,          # *fp16   [B, N, N, H]   intermediate tensor
    out_gate_ptr,     # *fp16   [B, N, N, H]   gate generated after first norm
    norm_w_ptr,       # *fp16   [H]           LN scale
    norm_b_ptr,       # *fp16   [H]           LN bias
    to_out_w_ptr,     # *fp16   [Dim, H]      final projection weight
    out_res_ptr,      # *fp16   [B, N, N, Dim] result (fp16, later cast to fp32)
    B, N, H, Dim,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_w_h, stride_norm_b_h,
    stride_to_out_w_h, stride_to_out_w_dim,
    stride_res_b, stride_res_i, stride_res_j, stride_res_dim,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    """
    For every position (b,i,j):
        1.  Layer‑norm over the hidden dimension H (FP32 accumulation)
        2.  Multiply element‑wise by the pre‑computed out‑gate
        3.  Project back to the model dimension Dim:
                y = (gate ⊙ norm) @ to_out.weightᵀ
    The kernel processes one (b,i,j) element per program and tiles
    over the hidden and output dimensions.
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # ------------------------------------------------------------------
    # Compute base pointers for a single (b,i,j) element
    # ------------------------------------------------------------------
    out_base = out_ptr + pid_b * stride_out_b + pid_i * stride_out_i + pid_j * stride_out_j
    gate_base = out_gate_ptr + pid_b * stride_gate_b + pid_i * stride_gate_i + pid_j * stride_gate_j

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean / variance of `out` over H (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    num_h_tiles = tl.cdiv(H, BLOCK_H)

    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        val = tl.load(out_base + offs_h * stride_out_h,
                      mask=mask_h,
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val)
        sum_sq += tl.sum(val * val)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ------------------------------------------------------------------
    # 2️⃣ Loop over the output‑dimension tiles (Dim) and accumulate
    # ------------------------------------------------------------------
    num_d_tiles = tl.cdiv(Dim, BLOCK_D)

    for d_tile in range(num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < Dim

        # accumulator for the dot‑product contribution of this output tile
        acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over hidden tiles: apply LN, gate and multiply with weight
        # ------------------------------------------------------------------
        for h_tile in range(num_h_tiles):
            offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # ----- load raw output value (fp16 → fp32) -----------------
            out_val = tl.load(out_base + offs_h * stride_out_h,
                              mask=mask_h,
                              other=0.0).to(tl.float32)

            # ----- layer‑norm (scale & bias) --------------------------
            norm_w = tl.load(norm_w_ptr + offs_h * stride_norm_w_h,
                             mask=mask_h,
                             other=0.0)
            norm_b = tl.load(norm_b_ptr + offs_h * stride_norm_b_h,
                             mask=mask_h,
                             other=0.0)
            # normalize
            out_norm = (out_val - mean) * inv_std
            out_norm = out_norm * norm_w + norm_b   # still fp32 after multiplication

            # ----- apply out‑gate ------------------------------------
            gate_val = tl.load(gate_base + offs_h * stride_gate_h,
                               mask=mask_h,
                               other=0.0).to(tl.float32)

            gated = out_norm * gate_val   # fp32

            # ----- weight tile (H × D) – note the weight is stored as
            #       [Dim, H] (row‑major).  We load it as a transposed view
            #       [H, D] via swapped strides.
            # ------------------------------------------------------------------
            w_ptrs = (
                to_out_w_ptr
                + (offs_h[:, None] * stride_to_out_w_h)          # rows = hidden dimension
                + (offs_d[None, :] * stride_to_out_w_dim)        # cols = output dimension
            )
            w_tile = tl.load(w_ptrs,
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0)   # shape (BLOCK_H, BLOCK_D)

            # ----- contribution to the final output -------------------
            # gated has shape (BLOCK_H,); we transform to (1, BLOCK_H) for dot
            gated_row = gated[None, :]       # (1, BLOCK_H)
            contrib = tl.dot(gated_row, w_tile)   # (1, BLOCK_D)
            acc += contrib[0]  # accumulate over hidden tiles

        # ------------------------------------------------------------------
        # Store the final result for this output tile
        # ------------------------------------------------------------------
        out_res_base = out_res_ptr + pid_b * stride_res_b + pid_i * stride_res_i + pid_j * stride_res_j
        out_res_ptrs = out_res_base + offs_d * stride_res_dim
        tl.store(out_res_ptrs,
                 acc.to(tl.float16),
                 mask=mask_d)
                

def _final_norm_gate_proj(out: torch.Tensor,
                          out_gate: torch.Tensor,
                          norm_w: torch.Tensor,
                          norm_b: torch.Tensor,
                          to_out_w: torch.Tensor) -> torch.Tensor:
    """
    Fuse the second LayerNorm, out‑gate multiplication and the final
    linear projection.  All tensors are fp16 on CUDA.
    Returns a fp16 tensor of shape [B, N, N, Dim].
    """
    B, N, _, H = out.shape
    Dim = to_out_w.shape[0]

    res = torch.empty((B, N, N, Dim), dtype=torch.float16, device=out.device)

    # strides expressed in number of elements (not bytes)
    s_out = out.stride()
    s_gate = out_gate.stride()
    s_norm_w = norm_w.stride()
    s_norm_b = norm_b.stride()
    s_to_out = to_out_w.stride()
    s_res = res.stride()

    BLOCK_H = 32          # tile size for the hidden dimension
    BLOCK_D = 32          # tile size for the projection dimension

    grid = (B, N, N)
    _final_norm_gate_proj_kernel[grid](
        out,
        out_gate,
        norm_w,
        norm_b,
        to_out_w,
        res,
        B,
        N,
        H,
        Dim,
        s_out[0], s_out[1], s_out[2], s_out[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_norm_w[0], s_norm_b[0],
        s_to_out[0], s_to_out[1],
        s_res[0], s_res[1], s_res[2], s_res[3],
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        EPS=1e-5,
        num_warps=8,
        num_stages=2,
    )
    return res


# ----------------------------------------------------------------------
# Helper: batched matmul (left @ rightᵀ) with hidden dim folded into batch
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Implemented as a batched GEMM C = A @ Bᵀ where
        A = left.permute(0, 3, 1, 2).reshape(B·H, N, N)
        B = right.permute(0, 3, 1, 2).reshape(B·H, N, N)
    Both tensors must be contiguous fp16.
    """
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``. May contain
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # FP32
    x = x.to(torch.float16)   # everything after this uses FP16

    # --------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (on‑device)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Gate values (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5️⃣ Mask‑gate fusion (Triton) or simple element‑wise multiplication
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS tensor‑core)
    # --------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm + out‑gate + final linear (all fused in Triton)
    # --------------------------------------------------------------
    out_fp16 = _final_norm_gate_proj(
        out,                     # intermediate tensor after contraction
        out_gate,                # gating tensor (from step 4)
        to_out_norm_w,           # LN2 scale
        to_out_norm_b,           # LN2 bias
        to_out_w,                # final projection weight
    )   # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 8️⃣ Cast result back to FP32 (as expected by the harness)
    # --------------------------------------------------------------
    return out_fp16.to(torch.float32)
