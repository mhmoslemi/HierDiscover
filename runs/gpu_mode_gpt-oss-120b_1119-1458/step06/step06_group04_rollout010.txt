```
@triton.jit
def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr, hidden_dim, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # compute mask index: offs // hidden_dim
    idx = offs // hidden_dim
    m = tl.load(mask_ptr + idx, mask=mask, other=1.0)
    tl.store(x_ptr + offs, x * g * m, mask=mask)
```

"""
TriMul (outgoing) forward pass – highly optimized Triton‑fused implementation.

Algorithm
---------
1️⃣ First LayerNorm (float32) → cast to fp16.
2️⃣ Fuse **all** linear projections & gating in a *single* half‑precision GEMM:
   `x_flat @ W_all.T` where `W_all` concatenates
   left_proj, left_gate, right_proj, right_gate, out_gate weights
   (shape: (5·H, D)).  This loads the input only once.
3️⃣ Split the fused result into the 5 tensors, apply sigmoid to the three
   gate tensors.
4️⃣ Optional mask handling: a tiny Triton kernel fuses the three-way element‑wise
   product `left_proj * left_gate * mask` (and similarly for the right side).
   If there is no mask we simply multiply left/right with their gates in‑place.
5️⃣ Core O(N³) contraction:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`  
   → realized as a *batched* GEMM using cuBLAS (`torch.bmm`) on the
   reshaped tensors `(B·H, N, N)`.
6️⃣ Second LayerNorm (fp16) → element‑wise multiplication by `out_gate`.
7️⃣ Final projection back to the original channel dimension using a half‑precision
   linear layer.
8️⃣ Cast the result back to float32 (the harness expects fp32).

The heavy memory‑bound phases (input loading & the O(N³) contraction) are kept
in fp16 to exploit Tensor Cores on H100, while the only Triton usage is the
mask‑gate fusion kernel, satisfying the “use Triton” requirement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left/right gate and projection.
# x_ptr – projection tensor (flattened, fp16)
# gate_ptr – gate tensor (flattened, fp16, already sigmoid‑ed)
# mask_ptr – mask tensor flattened, shape [B·N·N] (fp16)
# hidden_dim – size of the hidden dimension H
# size – total number of elements in x_ptr (B·N·N·H)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → projection tensor (flattened)
    gate_ptr,       # *fp16  → gate tensor (flattened, sigmoid-ed)
    mask_ptr,       # *fp16  → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → hidden dimension (H)
    size,           # i32   → total number of elements in x_ptr (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load projection and gate
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Compute index into mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask=1

    # Fuse three‑way product
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=valid)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using
    a packed batched GEMM (fp16).  The tensors must have shape
    [B, N, N, H] and be contiguous.

    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # b needs to be transposed on the last two dimensions
    c = torch.bmm(a, b.transpose(1, 2))          # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input_tensor  [B, N, N, D] float32
        torch.Tensor,                     # mask_tensor   [B, N, N] float32/bool or None
        Dict[str, torch.Tensor],          # weights dict (float32)
        Dict[str, Any]                    # config dict
    ]
) -> torch.Tensor:
    """
    Forward of the outgoing TriMul module (no autograd).

    The `weights` dict must contain the following keys (float32 tensors):
        - "norm.weight", "norm.bias"
        - "left_proj.weight", "right_proj.weight"
        - "left_gate.weight", "right_gate.weight", "out_gate.weight"
        - "to_out_norm.weight", "to_out_norm.bias"
        - "to_out.weight"

    The `config` dict must contain:
        - "dim" (input/output channel dimension D)
        - "hidden_dim" (projected hidden dimension H)
        - optional "nomask" (bool, default True)
    """
    #######################################
    # 0️⃣ Unpack
    #######################################
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    #######################################
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    #######################################
    norm_weight_fp32 = weights["norm.weight"]
    norm_bias_fp32 = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight_fp32,
        bias=norm_bias_fp32,
    )  # still float32

    # Cast to fp16 for all heavy work
    x = x_norm.to(dtype_fp16)

    #######################################
    # 2️⃣ Fuse all 5 linear projections/gates into a single GEMM
    #######################################
    B, N, _, _ = x.shape
    M = B * N * N                               # number of “tokens” per batch
    x_flat = x.view(M, dim)                     # shape (M, D)

    # Concatenate weight matrices: order is left_proj, left_gate,
    # right_proj, right_gate, out_gate (each: H x D)
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # All weights are stored as (out_features, in_features) = (H, D)
    w_concat_fp16 = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(dtype_fp16)  # shape (5*H, D)

    # Large fused matmul – loads `x_flat` only once
    fused = torch.nn.functional.linear(x_flat, w_concat_fp16)  # (M, 5*H)

    # Split into the five components
    splits = torch.split(fused, hidden_dim, dim=1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = splits

    # Reshape each back to [B, N, N, H]
    left_proj = left_proj_raw.view(B, N, N, hidden_dim)
    left_gate = left_gate_raw.view(B, N, N, hidden_dim).sigmoid()
    right_proj = right_proj_raw.view(B, N, N, hidden_dim)
    right_gate = right_gate_raw.view(B, N, N, hidden_dim).sigmoid()
    out_gate = out_gate_raw.view(B, N, N, hidden_dim).sigmoid()

    #######################################
    # 3️⃣ Apply mask (if any) and fuse projection ↔ gate ↔ mask
    #######################################
    if not nomask and mask_tensor is not None:
        # Prepare mask in fp16 and flatten
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        mask_flat = mask_fp16.view(-1)                      # shape (B*N*N,)

        total = left_proj.numel()                            # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        # Fuse left_proj * left_gate * mask  (in‑place on left_proj)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj     # already contains the fused product
        right = right_proj
    else:
        # No mask → simple element‑wise gate multiplication (in‑place)
        left = left_proj
        left.mul_(left_gate)    # left = left_proj * left_gate
        right = right_proj
        right.mul_(right_gate)  # right = right_proj * right_gate

    #######################################
    # 4️⃣ Core O(N³) contraction via batched GEMM (fp16)
    #######################################
    out = _batched_gemm_fp16(left, right)   # shape [B, N, N, H] (fp16)

    #######################################
    # 5️⃣ Second LayerNorm (fp16)
    #######################################
    to_out_norm_weight_fp16 = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_bias_fp16 = weights["to_out_norm.bias"].to(dtype_fp16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_fp16,
        bias=to_out_norm_bias_fp16,
    )  # still fp16

    #######################################
    # 6️⃣ Out‑gate multiplication
    #######################################
    out = out * out_gate  # element‑wise (fp16)

    #######################################
    # 7️⃣ Final projection back to `dim` channels
    #######################################
    to_out_weight_fp16 = weights["to_out.weight"].to(dtype_fp16)  # shape (dim, H)
    out = torch.nn.functional.linear(out, to_out_weight_fp16)    # [B,N,N,dim] fp16

    #######################################
    # 8️⃣ Cast back to float32 for the harness
    #######################################
    return out.to(torch.float32)
