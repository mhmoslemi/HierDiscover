"""
Optimized outgoing TriMul forward pass (no autograd).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣ Fuse the five linear transforms (left_proj, left_gate, right_proj, right_gate,
   out_gate) into a single GEMM by concatenating the weight matrices.
   This eliminates 4 extra passes over the massive (B·N·N·dim) tensor.
3️⃣ Apply the sigmoid to the three gating tensors.
4️⃣ Fuse element‑wise multiplication with the optional mask via a tiny Triton
   kernel (`fused_gate_mask_kernel`).  This updates the projected tensors
   in‑place:  proj ← proj * gate * mask.
5️⃣ Heavy O(N³) contraction is performed as a batched GEMM:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   (implemented with `torch.bmm`, which maps to Tensor‑Core kernels on H100).
6️⃣ Second Layer‑norm (FP16) normalises the hidden dimension.
7️⃣ Apply the output‑gate and a final linear projection back to `dim`.
8️⃣ Cast the result back to FP32 for the harness.

Only the forward pass is required; all heavy work stays in FP16 for speed.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across `hidden_dim`.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # hidden dimension (= size of last axis of x)
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # mask index is the linear index of the (B,N,N) tensor,
    # each entry is repeated `hidden_dim` times in x/gate.
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1 ⇒ no mask

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left, right : [B, N, N, H]   (FP16)
    Returns
    -------
    out : [B, N, N, H]  (FP16)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore layout to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool)
        - weights_dict : model weights (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the computation
    # --------------------------------------------------------------
    x = x_norm.to(torch.float16)  # [B, N, N, dim]   FP16

    # --------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 once (no bias in any Linear)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # Cast everything to FP16
    left_proj_w   = left_proj_w.to(torch.float16)
    right_proj_w  = right_proj_w.to(torch.float16)
    left_gate_w   = left_gate_w.to(torch.float16)
    right_gate_w  = right_gate_w.to(torch.float16)
    out_gate_w    = out_gate_w.to(torch.float16)
    to_out_norm_w = to_out_norm_w.to(torch.float16)
    to_out_norm_b = to_out_norm_b.to(torch.float16)
    to_out_w      = to_out_w.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Fuse the five linear transforms into a single GEMM
    # --------------------------------------------------------------
    # weight layout: [5*hidden_dim, dim]
    combined_weight = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)  # shape (5*hidden_dim, dim)

    # Linear:   [B, N, N, dim] @ [5*hidden_dim, dim].T → [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_weight)  # FP16

    # --------------------------------------------------------------
    # 5️⃣ Slice the fused result into its constituents
    # --------------------------------------------------------------
    h = hidden_dim
    left_proj   = combined[..., 0:h]                       # [B,N,N,H]
    left_gate_p = combined[..., h:2*h]                     # pre‑sigmoid
    right_proj  = combined[..., 2*h:3*h]
    right_gate_p= combined[..., 3*h:4*h]
    out_gate_p  = combined[..., 4*h:5*h]

    # --------------------------------------------------------------
    # 6️⃣ Apply sigmoid to gating tensors
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_p)   # [B,N,N,H] FP16
    right_gate = torch.sigmoid(right_gate_p)
    out_gate  = torch.sigmoid(out_gate_p)    # will be used later

    # --------------------------------------------------------------
    # 7️⃣ Optional mask – fuse (proj * gate * mask) in‑place via Triton
    # --------------------------------------------------------------
    if not nomask:
        # mask is broadcast across hidden_dim – convert to FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()               # B * N * N * H
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # Fuse left_proj, left_gate, mask → left_proj (in‑place)
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right_proj, right_gate, mask → right_proj (in‑place)
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # --------------------------------------------------------------
    out = _batched_matmul(left, right)   # [B, N, N, H]  FP16

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 🔟 Gating by out_gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 🏁 Return FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
