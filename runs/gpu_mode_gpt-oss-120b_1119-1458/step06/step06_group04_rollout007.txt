def _batched_gemm_fp16(left, right):
    B,N,_,H = left.shape
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # Now shape is [B, H, N, N]
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

"""
TriMul (outgoing) forward pass – accelerated Triton implementation.

Algorithm
---------
1. First LayerNorm (float32) of the 4‑D input.
2. Cast to fp16 for the compute‑intensive region.
3. Fuse the *projection* and its *gate* linear layers:
   - Concatenate the projection‑weight and gate‑weight (both shape
     [hidden_dim, dim]) → [2*hidden_dim, dim].
   - One `torch.nn.functional.linear` call yields both the projected
     tensor and the pre‑sigmoid gate tensor.
   - Apply the sigmoid to the gate tensor.
   - If a mask is present, fuse ``proj * gate * mask`` with a tiny
     Triton kernel (`mask_gate_fuse_kernel`).  When there is no mask,
     a simple element‑wise product is used.
4. The same fusion is performed for the right side.
5. Compute the heavy O(N³) contraction:
   ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``,
   realized as a batched GEMM on fp16 tensors with cuBLAS (`torch.bmm`).
6. Second LayerNorm over the hidden dimension.
7. Multiply by the *output gate* (computed by another linear+sigmoid).
8. Final linear projection back to the original channel dimension.
9. Cast the result back to float32 (the harness expects float32).

The implementation reduces the number of heavy linear layers from six
to four (two fused proj/gate, one out‑gate, one final projection),
applies the mask and gating element‑wise in a fused Triton kernel,
and keeps all arithmetic in fp16 to exploit TensorCores on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# All tensors are flattened fp16 arrays; ``mask`` has shape [B*N*N].
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    tensor_ptr,          # *fp16 – tensor to be updated (flattened)
    gate_ptr,           # *fp16 – gate tensor (same flattening as tensor)
    mask_ptr,           # *fp16 – mask tensor (flattened) [B*N*N]
    hidden_dim,         # i32   – hidden dimension size
    total_elems,        # i32   – total number of elements in ``tensor``
    BLOCK: tl.constexpr,   # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = offs < total_elems

    # Load the two operands
    t = tl.load(tensor_ptr + offs, mask=mask_valid, other=0.0)
    g = tl.load(gate_ptr + offs,   mask=mask_valid, other=0.0)

    # Compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)   # default mask = 1

    # Fuse multiplication
    out = t * tl.sigmoid(g) * m        # sigmoid is inside the kernel to avoid a separate launch

    # Store back in‑place
    tl.store(tensor_ptr + offs, out, mask=mask_valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the core contraction (left @ rightᵀ).
# Operates in fp16 and relies on cuBLAS via torch.bmm.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.

    Parameters
    ----------
    left, right : torch.Tensor
        fp16 tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        fp16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input  [B, N, N, dim]  (float32)
        torch.Tensor | None,              # mask   [B, N, N]      (float32 / bool / None)
        Dict[str, torch.Tensor],          # weights dict
        Dict[str, Any],                   # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, dtype float32, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor or None, shape [B, N, N]
        - weights_dict : mapping from layer‑name to weight tensors (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool) indicates whether a mask
                         will ever be supplied.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → stable)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x = x.to(dtype_fp16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(dtype_fp16)

    out_gate_w    = weights["out_gate.weight"].to(dtype_fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(dtype_fp16)

    # -----------------------------------------------------------------
    # 4️⃣ Fuse projection + gate (one linear each side)
    # -----------------------------------------------------------------
    # Concatenate projection and gate weights for the left side: shape [2*H, dim]
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Linear combos (produces both proj and pre‑sigmoid gate)
    left_cat = torch.nn.functional.linear(x, left_cat_w)   # [B, N, N, 2*H]
    right_cat = torch.nn.functional.linear(x, right_cat_w) # [B, N, N, 2*H]

    # Split into projection and gate pre‑sigmoid tensors
    left_proj = left_cat[..., :hidden_dim].contiguous()
    left_gate_pre = left_cat[..., hidden_dim:].contiguous()
    right_proj = right_cat[..., :hidden_dim].contiguous()
    right_gate_pre = right_cat[..., hidden_dim:].contiguous()

    # Apply sigmoid to the gate tensors
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fuse (tensor * gate * mask) via Triton
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Mask is a float/bool tensor of shape [B, N, N]; broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        mask_flat = mask_fp16.view(-1)                    # shape [B*N*N]
        total_elements = left_proj.numel()                # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        # left = left_proj * sigmoid(left_gate_pre) * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate_pre,
            mask_flat,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * sigmoid(right_gate_pre) * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate_pre,
            mask_flat,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel the tensors already contain the gated+masked values.
        left = left_proj
        right = right_proj
    else:
        # No mask – simple element‑wise multiplication with the sigmoid gates
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release intermediate tensors early to free memory
    del left_proj, left_gate_pre, right_proj, right_gate_pre, left_gate, right_gate
    torch.cuda.empty_cache()

    # -----------------------------------------------------------------
    # 6️⃣ Compute the output‑gate (linear + sigmoid) – still fp16
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)    # [B, N, N, H]

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over the hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Apply the out‑gate and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
