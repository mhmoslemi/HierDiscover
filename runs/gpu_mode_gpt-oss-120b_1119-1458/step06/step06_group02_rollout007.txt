"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑Norm over the last feature dimension (float32 → half).
2️⃣  Cast activations to fp16 (greatly speeds up the heavy ops).
3️⃣  Fuse the *projection* and *gate* linear layers for the left and right
    tensors by concatenating the two weight matrices and performing a single
    matmul per side.  This reduces the number of full‑precision matmuls from
    five to three (left, right, out‑gate).
4️⃣  Apply the sigmoid gate and optional mask in a tiny Triton kernel
    (`mask_gate_fuse_kernel`).  The kernel fuses the element‑wise multiply
    of the projected tensor, its gate, and the broadcast mask, cutting
    memory traffic and launch overhead.
5️⃣  Contract the gated left/right tensors with a batched GEMM
    (`torch.bmm`).  The data are reshaped to [B·H, N, N] so each hidden
    channel becomes an independent batch – this lets cuBLAS / TensorCore
    handle the O(N³) work at near‑peak throughput.
6️⃣  Second Layer‑Norm over the hidden dimension (fp16).
7️⃣  Multiply by the pre‑computed `out_gate` (sigmoid) and apply the final
    output projection (`to_out`).  Both are cheap element‑wise ops and a
    single linear layer.
8️⃣  Cast the result back to float32 – the surrounding model expects FP32.

Only the forward pass is implemented; no gradients are required.
A Triton kernel is used for the mask‑gate fusion step.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel that fuses element‑wise mask, gate and multiplication.
# It works on a flattened view of a tensor of shape [B, N, N, H] where
# the hidden dimension H is the innermost stride.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (projected values)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor of shape B·N·N (broadcast over H)
    hidden_dim,     # i32   : size H
    size,           # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)     # linear offsets for this block
    mask = offs < size                            # guard for OOB

    # map linear index to mask index (mask repeats every hidden_dim elems)
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr + offs,   mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right with an O(N³) batched GEMM.
    Both tensors have shape [B, N, N, H] (fp16).  The contraction is
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    which is equivalent to a batched matmul of left and rightᵀ per (b,h).
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] → matmul over the two N dimensions
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float or None)
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; may contain
                         ``nomask`` (True/False).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32), residing on the same device
        as the input.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True => no mask (skip mask steps)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Prepare FP16 weights (concatenated where possible)
    # ------------------------------------------------------------------
    # left projection + left gate
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_weights_cat = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden_dim, dim)

    # right projection + right gate
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_weights_cat = torch.cat([right_proj_w, right_gate_w], dim=0)

    # out gate (uses original dim -> hidden_dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # final output Linear and its LayerNorm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Compute out_gate (sigmoid of a linear transform of x)
    # ------------------------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)    # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(out_gate_pre)                     # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Compute left and right projections + gates in one matmul each
    # ------------------------------------------------------------------
    # left: concat projection + gate → shape [..., 2*hidden_dim]
    left_concat = torch.nn.functional.linear(x, left_weights_cat)   # [B,N,N,2*hidden_dim]
    left_proj = left_concat[..., :hidden_dim]       # projected values
    left_gate_pre = left_concat[..., hidden_dim:]   # logits for gate
    left_gate = torch.sigmoid(left_gate_pre)        # gate values (fp16)

    # right side (same pattern)
    right_concat = torch.nn.functional.linear(x, right_weights_cat)
    right_proj = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask, gate and element‑wise multiply (Triton kernel)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask is [B, N, N]; broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()                     # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple elementwise product
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Apply out_gate and final linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                                 # elementwise gating
    out = torch.nn.functional.linear(out, to_out_w)     # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
