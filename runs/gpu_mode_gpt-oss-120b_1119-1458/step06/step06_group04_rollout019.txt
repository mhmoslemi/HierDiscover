def gated_proj_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim, total_elem,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elem
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left * left_gate * m
    right_out = right * right_gate * m

    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)
```

"""
Highly‑optimized forward implementation of the “outgoing” TriMul operator.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single matrix multiplication by concatenating the corresponding
   weight matrices.  This reduces the number of reads of the large input
   tensor from four to one.
3️⃣ Split the fused output into the four components, apply a sigmoid to the
   two gate slices and fuse the element‑wise multiplication with an optional
   mask using a single Triton kernel (`gate_mask_fuse_kernel`).  The kernel
   updates *both* left and right tensors in‑place, guaranteeing that at least
   one Triton kernel is always launched.
4️⃣ Compute the heavy O(N³) contraction
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM on fp16 via cuBLAS (torch.bmm).  The tensors are
   reshaped to (B·H, N, N) which contains H independent matrix‑multiplications.
5️⃣ Apply the second LayerNorm (fp16) to the contracted tensor.
6️⃣ Multiply by the out‑gate (sigmoid(linear(x))) and apply the final linear
   projection back to `dim` channels.
7️⃣ Cast the result to float32 for the harness.

The key speed‑ups over the reference implementation are:
* One fused linear for the projection+gate weights instead of four separate
  linear calls (dramatically reduces memory traffic of the large input tensor).
* One Triton kernel that simultaneously applies both gates and the mask,
  halving kernel launch overhead.
* All heavy arithmetic is performed in fp16 to exploit Tensor Cores on an H100.

"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse gating (sigmoid already applied) and optional mask.
# Updates left_proj and right_proj in‑place.
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,               # *fp16  → left projection (flattened)
    right_ptr,              # *fp16  → right projection (flattened)
    left_gate_ptr,          # *fp16  → left gate (flattened)
    right_gate_ptr,         # *fp16  → right gate (flattened)
    mask_ptr,               # *fp16  → mask vector of shape [B * N * N]
    hidden_dim,             # i32   → hidden dimension (H)
    total_elems,            # i32   → total number of elements (B·N·N·H)
    BLOCK: tl.constexpr,    # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load data
    left  = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    # Derive mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1

    # Fuse multiplication
    left_out  = left  * lgate * m
    right_out = right * rgate * m

    # Store back in‑place
    tl.store(left_ptr  + offs, left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# Operates in fp16 and relies on cuBLAS via torch.bmm.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Parameters
    ----------
    left, right : torch.Tensor
        fp16 tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        fp16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N) contiguous layout required by cuBLAS
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                # [B*H, N, N]
    # reshape back to (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – keep it in fp32 for numerical stability
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 – the heavy part of the model runs in half‑precision
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)   # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 3️⃣ Prepare fused projection+gate weight (4·hidden_dim × dim)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    # Concatenate in the order required for later slicing
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )  # shape (4*H, dim)

    # -----------------------------------------------------------------
    # 4️⃣ Single linear projection that yields proj, gate for left/right
    # -----------------------------------------------------------------
    proj_gate = torch.nn.functional.linear(x, fused_weight)      # (B,N,N,4*H)

    # Slice the four components
    h = hidden_dim
    left_proj  = proj_gate[..., :h]               # (B,N,N,H)
    right_proj = proj_gate[..., h:2*h]            # (B,N,N,H)
    left_gate_raw  = proj_gate[..., 2*h:3*h]      # (B,N,N,H)
    right_gate_raw = proj_gate[..., 3*h:4*h]      # (B,N,N,H)

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – always use the Triton kernel (mask of 1 when unused)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        # all‑ones mask
        b, n, _ = input_tensor.shape[:3]
        mask_fp16 = torch.ones(b * n * n, dtype=torch.float16, device=device)

    mask_flat = mask_fp16.view(-1)                # shape [B·N·N]

    total_elems = left_proj.numel()               # B·N·N·H
    BLOCK = 1024                                  # ≤ max threads per block
    grid = (triton.cdiv(total_elems, BLOCK),)

    # Fuse gate multiplication and mask in a single Triton kernel
    gate_mask_fuse_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_flat,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # after the kernel left_proj and right_proj hold the gated & masked tensors

    # -----------------------------------------------------------------
    # 6️⃣ Out‑gate computation (separate linear + sigmoid)
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7️⃣ Core O(N³) contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left_proj, right_proj)    # (B,N,N,H) fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                                    # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
