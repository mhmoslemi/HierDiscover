def _batched_matmul_transpose_fp16(left, right):
    B,N,_,H = left.shape
    left_t = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_t = right.permute(0,3,2,1).contiguous().view(B*H, N, N)

    out = torch.empty_like(left_t, dtype=torch.float16) # shape M,N,N

    stride_am, stride_ai, stride_ak = left_t.stride()
    stride_bm, stride_bk, stride_bj = right_t.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tile sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128  # K reduction tile
    # Validate that N % BLOCK_* == 0; fallback to torch.einsum if not.
    if N % BLOCK_M != 0 or N % BLOCK_N != 0 or N % BLOCK_K != 0:
        # fallback
        out = torch.einsum('b i k h, b j k h -> b i j h', left, right)
        return out

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B*H )
    _trimul_matmul[grid](
        left_t, right_t, out,
        B*H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # reshape back
    out = out.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1 LayerNorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16 for the heavy path
    x = x.to(torch.float16)

    # Cast model weights to fp16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections (fp16)
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w) # [B,N,N,H]

    # Apply optional mask before gating (broadcast)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # Gating (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # O(N³) contraction using Triton batched GEMM
    out = _batched_matmul_transpose_fp16(left, right)  # [B,N,N,hidden_dim] fp16

    # Second LayerNorm (fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply out gate and final linear projection to dim
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # fp16

    return out.to(torch.float32)

Outgoing TriMul (AlphaFold‑3) – forward pass.

The heavy O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is rewritten as a batch of independent matrix multiplications
    C_h = A_h @ B_hᵀ  (A_h = left[...,i,k,h] , B_h = right[...,j,k,h])
for every hidden slice h.  By folding the batch and hidden axes we obtain a
tensor of shape [M, N, N] with M = B·hidden_dim.  A custom Triton kernel
computes C = A @ Bᵀ with fp16 inputs, fp32 accumulation (tensor‑cores) and
a 64 × 64 × 128 tiling (no boundary checks – all test lengths are multiples of
64/128).  All surrounding operations (LayerNorm, Linear, Sigmoid, gating)
are performed in PyTorch using fp16 arithmetic; the final result is cast back
to float32 as required by the harness.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : [M, N, K]   (M = batch * hidden, N = seq_len, K = seq_len)
#   B : [M, K, N]   (right already transposed)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]  (right already transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bk, stride_bj,   # strides for B (K‑N swapped)
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) tile indices
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_z = tl.program_id(2)          # batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j
    # No need for boundary masks – all test N are multiples of the block sizes

    # ------------------------------------------------------------------
    # 2) accumulator (FP32 for numeric stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) reduction over K (seq_len)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)          # k

        # load A tile  (BLOCK_M x BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak

        a = tl.load(a_ptrs)                             # fp16, no mask

        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk                  # K dim
            + offs_n[None, :] * stride_bj                  # N dim (post‑transpose)
        )
        b = tl.load(b_ptrs)                             # fp16, no mask

        # FP16×FP16 → FP32 accumulation using tensor‑cores
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 4) write back (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16))


# ----------------------------------------------------------------------
# Helper: reshape tensors, launch the Triton kernel, restore original layout
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H] (fp16)

    Returns
    -------
    out : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape

    # pack batch & hidden together → [M, N, N] with M = B*H
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)       # A : (M, i, k)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)     # B : (M, k, j)

    out = torch.empty_like(left_, dtype=torch.float16)                     # (M, i, j)

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # Tiling configuration – all test seq_len are multiples of 64/128
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128                         # larger K‑tile reduces loop overhead

    # sanity check – fall back to torch.einsum if the static tile sizes do not divide N
    if (N % BLOCK_M) != 0 or (N % BLOCK_N) != 0 or (N % BLOCK_K) != 0:
        # Very rare for the provided test suite; keep a correct fallback
        return torch.einsum('b i k h, b j k h -> b i j h', left, right)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,      # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,      # j‑tiles
        B * H,                             # batch*hidden (M)
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,                     # double‑buffered pipeline
    )

    # reshape back to [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical safety)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 – the expensive part runs in half‑precision
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 (once per forward)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates in‑place to keep memory traffic low
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via a Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gating and projection back to the original dimensionality
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)    # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
