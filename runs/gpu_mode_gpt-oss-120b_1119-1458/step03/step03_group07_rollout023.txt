Implementation details for custom Triton batched GEMM:

Goal: compute out[b,i,j,h] = sum_{k} left[b,i,k,h] * right[b,j,k,h].

Approach: Use a 3D grid of program IDs: (i_tile, j_tile, bh). Where bh encodes batch * hidden_dim.

We'll implement a kernel with blocks (BLOCK_M, BLOCK_N) for out tile.

Pseudo:

```
@triton.jit
def trilmul_outgoing_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_li, stride_lk, stride_lh, stride_lb,  # left strides for dimensions: i, k, h, batch
    stride_ri, stride_rk, stride_rh, stride_rb,  # similarly for right
    stride_oi, stride_oj, stride_oh, stride_ob,  # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_i = tl.program_id(0) # i_tile
    pid_j = tl.program_id(1) # j_tile
    pid_bh = tl.program_id(2) # batch_head

    total_bh = B * H
    b = pid_bh // H
    h = pid_bh % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in BLOCK_K increments
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left pointer: b at batch, i, k, h
        # left_ptr shape: B, N (i), N (k), H
        # stride formula: offset = b*stride_lb + offs_i*stride_li + offs_k*stride_lk + h*stride_lh
        left_ptrs = left_ptr + b*stride_lb + offs_i[:, None]*stride_li + offs_k[None, :]*stride_lk + h*stride_lh

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right pointer: we need right component transposed: want right[j, k, h] but we need matrix multiplication left[i,k] * right[j,k]^T
        # We'll load right tile as (BLOCK_K, BLOCK_N) with rows k, cols j.
        right_ptrs = right_ptr + b*stride_rb + offs_j[None, :]*stride_ri + offs_k[:, None]*stride_rk + h*stride_rh

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Here left shape (M,K), right shape (K,N). Compute matmul.
        acc += tl.dot(left, right)   # shapes (M,K) x (K,N) => (M,N)
    # After loop, we have acc
    # Write to output out[b,i,j,h]
    out_ptrs = out_ptr + b*stride_ob + offs_i[:, None]*stride_oi + offs_j[None, :]*stride_oj + h*stride_oh

    # store
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
 
    # Cast to fp16 for performance after first layernorm
    # First LN
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # float32
    x_norm_fp16 = x.to(torch.float16)   # shape [B,N,N,dim]

    # Save normalized input for out_gate later
    x_for_out_gate = x_norm_fp16  # same

    # Apply mask early if needed
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        x_proj_input = x_norm_fp16 * mask_fp16
    else:
        x_proj_input = x_norm_fp16

    # Load and cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16) # shape [hidden_dim, dim]
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape [dim, hidden_dim]

    # Fuse projection + gate
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # shape [2*hidden_dim, dim]
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # shape [2*hidden_dim, dim]

    # Perform linear
    left_comb = torch.nn.functional.linear(x_proj_input, left_comb_w)   # [B,N,N,2*hidden_dim]
    right_comb = torch.nn.functional.linear(x_proj_input, right_comb_w)  # same

    # split
    left_proj = left_comb[..., :hidden_dim]   # [B,N,N,hidden_dim]
    left_gate = left_comb[..., hidden_dim:]   # [B,N,N,hidden_dim]

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    stride_lk = left.stride(2)  # k stride
    stride_lh = left.stride(3)  # h stride

    stride_rb = right.stride(0)
    stride_ri = right.stride(1)  # j? But right's "i" axis actually j dimension
    stride_rk = right.stride(2)
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input tensor (FP32 → FP16 for speed).
2️⃣ Fuse the two projection + two gate Linear layers into two Linear calls with
   concatenated weight matrices (2 × matmul instead of 4).
   * The mask is applied **only** to the projection part (exactly as the
     reference implementation) – this saves two extra element‑wise kernels.
3️⃣ Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   with a custom Triton kernel that:
   * treats each pair (batch, hidden‑dim) as an independent program id,
   * tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N) and the reduction axis k
     (BLOCK_K),
   * uses FP32 accumulation for numerical stability,
   * writes the result directly in FP16, thus avoiding the costly
     permute‑contiguous reshapes used in the naïve implementation.
4️⃣ Apply the second Layer‑Norm (still FP16), compute the output gate,
   multiply element‑wise, and project back to the original channel dimension
   with a final Linear.
5️⃣ Cast the result back to float32 (the API expectation).

Only the O(N³) contraction is performed in Triton; all other operations use
high‑performance PyTorch kernels and the fused linear layers minimise kernel
launch overhead.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the hidden dimension (the N³ part)
# Computes: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trilmul_outgoing_matmul_kernel(
    left_ptr,          # *fp16 – left tensor   [B, N, N, H]
    right_ptr,         # *fp16 – right tensor  [B, N, N, H]
    out_ptr,           # *fp16 – output tensor [B, N, N, H]
    B, N, H,           # i32 scalar dimensions
    # left strides: (batch, i, k, h)
    stride_lb, stride_li, stride_lk, stride_lh,
    # right strides: (batch, j, k, h)
    stride_rb, stride_ri, stride_rk, stride_rh,
    # out strides: (batch, i, j, h)
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile along the i‑axis
    pid_j = tl.program_id(1)          # tile along the j‑axis
    pid_bh = tl.program_id(2)         # merged (batch, hidden) dimension

    # ---- decode batch and hidden‑dim from pid_bh ----
    b = pid_bh // H                  # batch index
    h = pid_bh - b * H                # hidden‑dim index (0 ≤ h < H)

    # ---- start offsets for i and j ----
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over k ----
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left tile (M × K) ----------
        # left[b, i, k, h]
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # -------- load right tile (K × N) ----------
        # we need the transpose: right[b, j, k, h] -> right_T[k, j]
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_j[None, :] * stride_ri   # j dimension (output column)
            + offs_k[:, None] * stride_rk   # k dimension (reduction row)
            + h * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # -------- accumulate dot product ----------
        acc += tl.dot(left, right)   # (M, K) × (K, N) → (M, N)

    # ---- store the result (cast back to fp16) ----
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool) – may be unused
        - weights_dict : mapping name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                               # [B, N, N, dim] – float32
    x_fp16 = x_norm.to(torch.float16)   # work in FP16 from now on
    # keep a copy for out‑gate (mask‑independent)
    x_for_out_gate = x_fp16

    # -------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # -------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)    # (H, dim)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)    # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)   # (H, dim)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)   # (H, dim)

    out_gate_w   = weights["out_gate.weight"].to(torch.float16)     # (H, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w     = weights["to_out.weight"].to(torch.float16)      # (dim, H)

    # -------------------------------------------------
    # 3️⃣ Fuse projection + gate (2 linear ops instead of 4)
    # -------------------------------------------------
    # concatenate proj‑ and gate‑weights → shape (2*H, dim)
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # combined linear (biases are zero in the reference model)
    left_comb = torch.nn.functional.linear(x_fp16, left_comb_w)   # [B,N,N,2H]
    right_comb = torch.nn.functional.linear(x_fp16, right_comb_w)  # [B,N,N,2H]

    # split projection and gate parts (views, no copy)
    left_proj = left_comb[..., :hidden_dim]      # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:]      # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]    # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:]    # [B,N,N,H]

    # -------------------------------------------------
    # 4️⃣ Apply mask *only* to the projection tensors
    # -------------------------------------------------
    if not nomask:
        # mask shape → [B,N,N,1] broadcast over hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate      # [B,N,N,H]
    right = right_proj * right_gate   # [B,N,N,H]

    # -------------------------------------------------
    # 6️⃣ O(N³) contraction via a custom Triton kernel
    # -------------------------------------------------
    Bsz, Nsz, _, Hsz = left.shape
    out = torch.empty((Bsz, Nsz, Nsz, Hsz), dtype=torch.float16, device=device)

    # strides in number of elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes – tuned for H100 tensor‑core kernels
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (Nsz + BLOCK_M - 1) // BLOCK_M,
        (Nsz + BLOCK_N - 1) // BLOCK_N,
        Bsz * Hsz,
    )
    _trilmul_outgoing_matmul_kernel[grid](
        left,
        right,
        out,
        Bsz,
        Nsz,
        Hsz,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Out‑gate (sigmoid) and element‑wise multiplication
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_for_out_gate, out_gate_w))
    out = out * out_gate          # cheap FP16 element‑wise product

    # -------------------------------------------------
    # 9️⃣ Final linear projection back to ``dim``
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 🔚 Cast back to float32 (API requirement)
    # -------------------------------------------------
    return out.to(torch.float32)
