"""
TriMul (outgoing) – fast forward implementation for AlphaFold‑3.

The heavy‑weight step is the O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
which is a batched matrix multiplication of shape
    (B·H, N, N) @ (B·H, N, N)ᵀ.
We implement this contraction with a custom Triton GEMM that
processes all batches in a single launch, avoiding the overhead of
torch.bmm's many small kernels.  All other linear layers are fused
and performed in FP16, while the two LayerNorms stay in FP32 for
numerical stability.  The final element‑wise out‑gate multiplication
is fused inside Triton as well (optional – the main speedup comes from
the batched GEMM).

The function `custom_kernel` receives
    (input, mask, weights, config)
and returns the processed tensor in FP32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
# Computes C = A @ Bᵀ  where
#   A : [batch, M, K]   (row‑major)
#   B : [batch, K, N]   (row‑major) → we treat it as (K,N) in the dot
#   C : [batch, M, N]
# All tensors are fp16; accumulation is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_aa, stride_ab, stride_ac,         # batch stride for A/B/C
    M, N, K,                                 # matrix dimensions (runtime)
    stride_am, stride_ak,                    # A row/col strides
    stride_bk, stride_bn,                    # B row/col strides   (B is row‑major K×N)
    stride_cm, stride_cn,                    # C row/col strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    batch_id = tl.program_id(1)                     # batch (B*H) dimension
    pid = tl.program_id(0)                          # tile index over M & N

    # ------------------------------------------------------------------
    # tile offsets (M‑axis)
    # ------------------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # base pointers for the current batch
    # ------------------------------------------------------------------
    a_base = a_ptr + batch_id * stride_aa
    b_base = b_ptr + batch_id * stride_ab
    c_base = c_ptr + batch_id * stride_ac

    # ------------------------------------------------------------------
    # accumulation buffer (fp32 for accuracy)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # B tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base
            + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # fp16

        # Tensor‑core multiplication (fp16 × fp16 → fp32)
        acc = acc + tl.dot(a, b)

    # ------------------------------------------------------------------
    # write out (convert back to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (dst *= src)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr, src_ptr, N,
    dst_stride: tl.constexpr, src_stride: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)
    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on input (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                    # [B, N, N, dim]  FP32
    x = x.to(torch.float16)               # work in FP16 from now on

    # ------------------------------------------------------------------
    # 2️⃣  Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Fuse projection + gate (2 linear calls instead of 4)
    # ------------------------------------------------------------------
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2·H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2·H, dim)

    # combined linear → [B,N,N,2·H]
    left_comb  = torch.nn.functional.linear(x, left_comb_w)   # fp16
    right_comb = torch.nn.functional.linear(x, right_comb_w)  # fp16

    # split back into projection and gate
    left_proj = left_comb[..., :hidden_dim]           # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:]           # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]        # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:]        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣  Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣  Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate          # [B,N,N,H]
    right = right_proj * right_gate       # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣  Heavy O(N³) contraction via a custom Triton batched GEMM
    #    left : [B,N,N,H] → [B, H, N, N] (row‑major)
    #    right: same shape but with the two inner dimensions swapped
    # ------------------------------------------------------------------
    # permute to batch‑major layout
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k as row)

    B, H, N, _ = left_perm.shape
    batch = B * H

    # flatten batch dimension for the kernel
    left_flat  = left_perm.reshape(batch, N, N).contiguous()
    right_flat = right_perm.reshape(batch, N, N).contiguous()
    out_flat   = torch.empty_like(left_flat)               # [batch, N, N] fp16

    # strides for the kernel (row‑major layout)
    stride_aa = left_flat.stride(0)      # batch stride for A
    stride_ab = right_flat.stride(0)     # batch stride for B (same as A)
    stride_ac = out_flat.stride(0)       # batch stride for C

    stride_am = left_flat.stride(1)      # A row stride (N)
    stride_ak = left_flat.stride(2)      # A col stride (1)

    stride_bk = right_flat.stride(1)     # B row stride (N)  (since B is row‑major K×N)
    stride_bn = right_flat.stride(2)     # B col stride (1)

    stride_cm = out_flat.stride(1)       # C row stride
    stride_cn = out_flat.stride(2)       # C col stride

    # tile sizes – tuned for H100 Tensor‑Core utilization
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m * num_pid_n, batch)   # (tile grid, batch)

    _batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        stride_aa=stride_aa,
        stride_ab=stride_ab,
        stride_ac=stride_ac,
        M=N, N=N, K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣  Second LayerNorm (still FP16 weights)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )    # fp16 → fp16

    # ------------------------------------------------------------------
    # 8️⃣  Out‑gate (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # Fuse multiplication with a tiny Triton kernel (in‑place)
    total_elems = out.numel()
    BLOCK_SIZE = 1024
    grid_mul = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elemwise_mul_kernel[grid_mul](
        out.contiguous().view(-1),
        out_gate.contiguous().view(-1),
        total_elems,
        1, 1,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)    # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 🔚 Cast back to float32 (the API expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
