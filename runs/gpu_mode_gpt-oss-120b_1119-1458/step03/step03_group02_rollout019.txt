"""
TriMul outgoing (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
* LayerNorm on the feature axis (float32 for stability)
* Fuse the five linear projections (two projections + three gates) into a single
  batched GEMM (`x @ W_stack`).  This reduces the number of passes over the huge
  input tensor `x` from five to one and saves both memory traffic and kernel
  launch overhead.
* Apply sigmoid gates and optional mask.
* Heavy O(N³) contraction is performed by a custom batched‑matrix‑multiply
  kernel that computes `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
  The kernel tiles the two N‑axes with 64×64 or 128×128 blocks (chosen at
  runtime) and uses Tensor‑Core MMA (`tl.dot`) with FP16 inputs and FP32
  accumulation.
* Second LayerNorm (hidden_dim) + output gate + final linear projection.
* Output is returned in float32.

The implementation keeps the heavy contraction in Triton while all other
operations are done with high‑performance PyTorch primitives.
"""

import math
from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch*hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                # *fp16  [M, N, K]
    B_ptr,                # *fp16  [M, N, K]
    C_ptr,                # *fp16  [M, N, N]
    M, N, K,              # i32 sizes
    stride_am, stride_ai, stride_ak,  # A strides (M,N,K) in elements
    stride_bm, stride_bi, stride_bk,  # B strides (M,N,K) in elements
    stride_cm, stride_ci, stride_cj,  # C strides (M,N,N) in elements
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matmul for many independent N×N matrices.
    Each program (identified by pid_z) processes one (M‑index) = one
    (batch * hidden) slice.  Tiling over the two N‑axes uses blocks of
    size BLOCK_M × BLOCK_N.
    """
    pid_i = tl.program_id(0)   # tile along first N dimension
    pid_j = tl.program_id(1)   # tile along second N dimension
    pid_z = tl.program_id(2)   # index over M (= batch * hidden)

    # -------------------------------------------------
    # Offsets for this tile
    # -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # -------------------------------------------------
    # Main loop over K
    # -------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile:   A[pid_z, i, k]
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile (transposed view): B[pid_z, k, j]
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Tensor‑core MMA (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Store the result tile
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper: batched matmul C = A @ Bᵀ   (FP16 in, FP16 out)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int,
    block_n: int,
    block_k: int,
) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Left/right are FP16 tensors of shape [B, N, N, H].
    The function packs (batch, hidden) → M and runs the Triton kernel.
    """
    B, N, _, H = left.shape
    # pack B and H together: (B*H, N, N) contiguous
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bi, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # grid definition
    grid = (
        (N + block_m - 1) // block_m,   # i‑tiles
        (N + block_n - 1) // block_n,   # j‑tiles
        B * H,                          # M‑tiles (batch * hidden)
    )

    # Choose number of warps: 8 for 64‑tile, 4 for 128‑tile (empirical)
    num_warps = 8 if block_m <= 64 else 4

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bi,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=num_warps,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask_tensor  : torch.Tensor [B, N, N]          (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical safety)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # -------------------------------------------------
    # Cast to FP16 for the rest of the pipeline
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Fuse all five linear projections into a single GEMM
    #    (proj_left, proj_right, gate_left, gate_right, gate_out)
    # -------------------------------------------------
    # Pack weights: each has shape (hidden_dim, dim)
    weight_stack = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden, dim)
            weights["right_proj.weight"],  # (hidden, dim)
            weights["left_gate.weight"],   # (hidden, dim)
            weights["right_gate.weight"],  # (hidden, dim)
            weights["out_gate.weight"],    # (hidden, dim)
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden, dim)

    # Flatten the leading three dimensions (B,N,N)
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    y = torch.nn.functional.linear(x_flat, weight_stack)  # (B*N*N, 5*hidden)
    y = y.view(B, N, N, 5 * hidden_dim)         # (B,N,N,5*hidden)

    # ------------------------------------------------------------------
    # Split the packed tensor and apply sigmoid where needed
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj = y[..., 0:h]                     # (B,N,N,hidden)
    right_proj = y[..., h : 2 * h]              # (B,N,N,hidden)
    left_gate = torch.sigmoid(y[..., 2 * h : 3 * h])
    right_gate = torch.sigmoid(y[..., 3 * h : 4 * h])
    out_gate = torch.sigmoid(y[..., 4 * h : 5 * h])

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Apply hidden‑dim gates
    # -------------------------------------------------
    left = left_proj * left_gate      # (B,N,N,hidden)  fp16
    right = right_proj * right_gate   # (B,N,N,hidden)  fp16

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton
    # -------------------------------------------------
    # Dynamically pick block sizes (larger tiles for large seq_len)
    if N <= 256:
        blk_m, blk_n, blk_k = 64, 64, 64
    else:
        blk_m, blk_n, blk_k = 128, 128, 64

    out = _batched_matmul_transpose_fp16(left, right, blk_m, blk_n, blk_k)
    # out: (B,N,N,hidden) fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden_dim)
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -------------------------------------------------
    # 7️⃣ Output gating + final projection back to original dim
    # -------------------------------------------------
    out = out * out_gate                         # element‑wise gate (fp16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim) fp16

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 for the final result
    # -------------------------------------------------
    return out.to(torch.float32)
