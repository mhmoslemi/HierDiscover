```python
"""
Custom Triton‑accelerated implementation of the outgoing TriMul operation.

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – we keep the heavy arithmetic in FP16.
2️⃣ Fuse the five linear transformations (left/right projection, left/right/out gate)
   into a single GEMM:  x́ = LayerNorm(x)  →  x̃ = x́.view(-1, dim)
   y = x̃ @ W_fused.T   (W_fused stacks the five weight matrices)
   The result is reshaped to (B, N, N, 5·hidden) and sliced.
3️⃣ Apply sigmoid to the three gate slices.
4️⃣ If a mask is present we launch a Triton kernel that fuses
   (proj * gate * mask) for both the left and right paths in one pass,
   drastically reducing memory traffic.  When no mask is given we fall back
   to a simple element‑wise product.
5️⃣ Core contraction (outgoing update) – compute for every hidden channel
   a batched matrix multiplication   left[b,h,i,k] × right[b,h,k,j].
6️⃣ Output LayerNorm, out‑gate multiplication and final linear projection,
   all performed in FP16.  The result is cast back to the input dtype.

The Triton kernel handles the broadcast of the 2‑D mask over the hidden
dimension, performing two independent element‑wise products per thread
and writing the gated tensors back to memory.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: (proj * gate * mask) for left and right tensors.
# All inputs are dense FP16 with shape (B, N, N, hidden) and the mask has
# shape (B, N, N).  The kernel processes a flat index `offs = b·N·N·hidden + i·N·hidden + j·hidden + d`.
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_mul_kernel(
    left_proj_ptr,   # float16 * (B,N,N,hidden)
    left_gate_ptr,   # float16 *
    right_proj_ptr,  # float16 *
    right_gate_ptr,  # float16 *
    mask_ptr,        # float16 * (B,N,N)
    left_out_ptr,    # float16 *
    right_out_ptr,   # float16 *
    hidden: tl.constexpr,      # hidden dimension (compile‑time)
    total: tl.int32,           # total number of elements in a/c
    BLOCK_SIZE: tl.constexpr,  # #threads per block
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total                # guard for out‑of‑range threads

    # load the four FP16 tensors
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)   # left proj
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)   # left gate
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)  # right proj
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)  # right gate

    # each hidden element belongs to the same (b,i,j), compute the mask index.
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)     # broadcasted mask value

    left_val  = lp * lg * m
    right_val = rp * rg * m

    tl.store(left_out_ptr + offs,  left_val,  mask=mask)
    tl.store(right_out_ptr + offs, right_val, mask=mask)

def gated_mask_mul(left_proj, left_gate, right_proj, right_gate, mask):
    """
    Fuse (proj * gate * mask) for left and right tensors using Triton.

    Args:
        left_proj, left_gate, right_proj, right_gate: torch.Tensor
            Contiguous FP16 tensors of shape (B, N, N, hidden).
        mask: torch.Tensor of shape (B, N, N), already FP16 and contiguous.

    Returns:
        left_out, right_out: torch.Tensor of shape (B, N, N, hidden), FP16.
    """
    assert left_proj.shape == left_gate.shape == right_proj.shape == right_gate.shape
    B, N, _, hidden = left_proj.shape
    # Ensure contiguity – Triton only works on flat memory.
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    total = left_proj.numel()
    BLOCK = 8192  # empirically good for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    left_out  = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    gated_mask_mul_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask,
        left_out, right_out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return left_out, right_out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple containing (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float, CUDA)
            - weights_dict: mapping of weight names to torch.Tensors (CUDA)
            - config_dict : contains keys "dim", "hidden_dim", optional "eps", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse five linear layers into one GEMM
    # ------------------------------------------------------------------
    # Stack the five weight matrices: each is (hidden, dim)
    weight_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # concatenate along the out_features dimension
    ).to(torch.float16)  # shape (5*hidden, dim)

    # Flatten the normalized input for the GEMM
    x_flat = x_norm.view(-1, dim)            # (B*N*N, dim)
    fused_out_flat = F.linear(x_flat, weight_fused)  # (B*N*N, 5*hidden)
    fused_out = fused_out_flat.view(B, N, N, hidden * 5)

    # Slice the fused tensor back into the original streams
    left_proj_raw  = fused_out[..., 0               : hidden]
    right_proj_raw = fused_out[..., hidden          : 2*hidden]
    left_gate_raw  = fused_out[..., 2*hidden        : 3*hidden]
    right_gate_raw = fused_out[..., 3*hidden        : 4*hidden]
    out_gate_raw   = fused_out[..., 4*hidden        : 5*hidden]

    # ---------------------------------------------------------------
    # 3) Sigmoid gates
    # ---------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)   # shape (B,N,N,hidden)

    # ---------------------------------------------------------------
    # 4) Apply mask and gate to the projection tensors.
    # ---------------------------------------------------------------
    if nomask:
        # No mask -> simple element‑wise product (no extra kernel)
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask to FP16 once; Triton kernel will broadcast it internally.
        mask_fp16 = mask.to(torch.float16)
        # Ensure contiguity for the kernel (prev slices are still views into fused_out)
        left_proj_contig  = left_proj_raw.contiguous()
        right_proj_contig = right_proj_raw.contiguous()
        left_gate_contig  = left_gate.contiguous()
        right_gate_contig = right_gate.contiguous()

        left, right = gated_mask_mul(
            left_proj_contig, left_gate_contig,
            right_proj_contig, right_gate_contig,
            mask_fp16,
        )
        # Free up the intermediate views that are no longer needed.
        del left_proj_contig, right_proj_contig, left_gate_contig, right_gate_contig

    # ---------------------------------------------------------------
    # 5) Core contraction: (B, N, N, hidden) → (B, hidden, N, N) for batched GEMM.
    # ---------------------------------------------------------------
    # Note: .permute creates a view, no memory copy.
    left_perm  = left.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)  # (B, hidden, k, j)

    # Batched matrix multiplication over the hidden channels.
    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # Restore the original layout.
    out = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # ---------------------------------------------------------------
    # 6) Output layer‑norm, out‑gate, final projection.
    # ---------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                     # apply output gate (element‑wise)

    # Final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to original dtype (normally float32) before returning.
    return out_final.to(inp.dtype)

"""
Custom Triton‑accelerated “outgoing” TriMul implementation.

Algorithm Overview
------------------
1️⃣ Input LayerNorm (float32 → float16) – heavy arithmetic stays in FP16.
2️⃣ Fuse the five linear transforms (left/right projection, left/right/out gate)
   into a single GEMM:  x̂ = LayerNorm(x).view(-1, dim)
   y = x̂ @ W_fused.T where W_fused stacks the five weight matrices; result is
   reshaped to (B, N, N, 5·hidden) and sliced.
3️⃣ Apply sigmoid to the three gate slices.
4️⃣ If a mask is present we launch a Triton kernel that fuses
   (proj * gate * mask) for the left and right paths in one pass,
   broadcasting the 2‑D mask over the hidden dimension without materialising
   an expanded mask.  Without a mask we simply use element‑wise multiplication.
5️⃣ Core contraction (outgoing update) – batched GEMM over the hidden channel:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
6️⃣ Output LayerNorm → out‑gate multiplication → final linear projection.
   The result is cast back to the original dtype (usually float32).

The Triton kernel reduces memory traffic for the masked case and is the only
non‑trivial kernel required; all other ops are delegated to highly‑optimized
torch/CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (proj * gate * mask) → left & right tensors.
# All tensors are FP16, left/right have shape (B, N, N, hidden),
# mask has shape (B, N, N) and is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_mul_kernel(
    left_proj_ptr,   # float16 * (B,N,N,hidden)
    left_gate_ptr,   # float16 *
    right_proj_ptr,  # float16 *
    right_gate_ptr,  # float16 *
    mask_ptr,        # float16 * (B,N,N)
    left_out_ptr,    # float16 *
    right_out_ptr,   # float16 *
    hidden: tl.constexpr,      # hidden dimension (compile‑time)
    size,                     # total number of elements in left/right tensors
    BLOCK_SIZE: tl.constexpr   # threads per block
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size                      # out‑of‑range guard

    # Load values from global memory
    lp = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute index into the 2‑D mask (broadcast over hidden)
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse the three multiplications
    left_val  = lp * lg * m
    right_val = rp * rg * m

    tl.store(left_out_ptr + offs,  left_val,  mask=mask)
    tl.store(right_out_ptr + offs, right_val, mask=mask)


def gated_mask_mul(left_proj, left_gate, right_proj, right_gate, mask):
    """
    Fuse (proj * gate * mask) for left and right tensors using Triton.

    Args:
        left_proj, left_gate, right_proj, right_gate: torch.Tensor
            Contiguous FP16 tensors of shape (B, N, N, hidden).
        mask: torch.Tensor of shape (B, N, N), FP16 and contiguous.

    Returns:
        left_out, right_out: torch.Tensor of shape (B, N, N, hidden), FP16.
    """
    assert left_proj.shape == left_gate.shape == right_proj.shape == right_gate.shape
    B, N, _, hidden = left_proj.shape
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    total = left_proj.numel()
    BLOCK = 8192                                     # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    left_out  = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    gated_mask_mul_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask,
        left_out, right_out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return left_out, right_out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor:  [B, N, N, dim]  (float32, CUDA)
            - mask_tensor :  [B, N, N]       (bool/float, CUDA)
            - weights_dict: model weights (see keys below)
            - config_dict : must contain "dim" and "hidden_dim";
                            optional "eps" (default 1e-5) and
                            "nomask" (default False).

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep result in FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse the five linear layers into a single GEMM
    # ------------------------------------------------------------------
    # Each weight: (hidden, dim).  Stack along the out_features axis.
    weight_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                              # (5*hidden, dim)
    ).to(torch.float16)

    # Flatten for the GEMM
    x_flat = x_norm.view(-1, dim)          # (B·N·N, dim)
    fused_out_flat = F.linear(x_flat, weight_fused)   # (B·N·N, 5·hidden)
    fused_out = fused_out_flat.view(B, N, N, hidden * 5)

    # ------------------------------------------------------------------
    # Slice the fused tensor back into the original streams
    # ------------------------------------------------------------------
    left_proj_raw  = fused_out[..., 0               : hidden]
    right_proj_raw = fused_out[..., hidden          : 2*hidden]
    left_gate_raw  = fused_out[..., 2*hidden        : 3*hidden]
    right_gate_raw = fused_out[..., 3*hidden        : 4*hidden]
    out_gate_raw   = fused_out[..., 4*hidden        : 5*hidden]

    # ------------------------------------------------------------------
    # 3) Apply sigmoid to the three gate tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4) Mask & gate the projection tensors
    # ------------------------------------------------------------------
    if nomask:
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask.to(torch.float16)   # broadcasted mask in FP16
        # Make the projection tensors contiguous for the Triton kernel
        left_proj_contig  = left_proj_raw.contiguous()
        right_proj_contig = right_proj_raw.contiguous()
        left_gate_contig  = left_gate.contiguous()
        right_gate_contig = right_gate.contiguous()

        left, right = gated_mask_mul(
            left_proj_contig, left_gate_contig,
            right_proj_contig, right_gate_contig,
            mask_fp16,
        )
        # Release temporaries early
        del left_proj_contig, right_proj_contig, left_gate_contig, right_gate_contig

    # ------------------------------------------------------------------
    # 5) Core contraction (batched matrix multiplication)
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)  # (B, hidden, k, j)

    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # Restore original layout
    out = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                       # element‑wise out‑gate

    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to the original dtype (normally float32) before returning
    return out_final.to(inp.dtype)
