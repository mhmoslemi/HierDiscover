```python
@triton.jit
def fused_proj_gate_mask_kernel(
    left_proj_ptr,
    right_proj_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    left_out_ptr,
    right_out_ptr,
    hidden: tl.constexpr,
    size: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_range = offs < size
    
    # Load values
    left_proj = tl.load(left_proj_ptr + offs, mask=mask_range, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask_range, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask_range, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask_range, other=0.0)
    
    # Compute sigmoid for gates
    # sigmoid(x) = 1 / (1 + exp(-x))
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
    
    # mask index
    mask_idx = offs // hidden
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_range, other=0.0)
    
    left_out = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val
    
    tl.store(left_out_ptr + offs, left_out, mask=mask_range)
    tl.store(right_out_ptr + offs, right_out, mask=mask_range)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_proj_gate_mask_kernel(
    left_proj_ptr,
    right_proj_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    left_out_ptr,
    right_out_ptr,
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_off = offs < total

    # Load values
    left_proj = tl.load(left_proj_ptr + offs, mask=mask_off, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask_off, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask_off, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask_off, other=0.0)

    # sigmoid for gates
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # compute mask index (broadcast over hidden dim)
    mask_idx = offs // hidden
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_off, other=0.0)

    left_out = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_out, mask=mask_off)
    tl.store(right_out_ptr + offs, right_out, mask=mask_off)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Algorithm:
      1. Input LayerNorm (float32 → fp16).
      2. Fuse the five linear layers (left_proj, right_proj, left_gate,
         right_gate, out_gate) into a single batched GEMM.
      3. Split the fused result; compute out_gate (sigmoid) and apply mask,
         gate and mask to left/right using a custom Triton kernel.
      4. Compute the core contraction with a batched matrix‑multiply:
            out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
      5. Output LayerNorm, multiply by out_gate, and a final linear projection.
      All heavy arithmetic is performed in fp16; the final result is cast back
      to the input dtype (usually float32).
    """

    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep fp32 for numeric stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------

    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # shape (5*hidden, dim)
    fused_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0
    )

    fused = F.linear(x, fused_weight)  # (B, N, N, 5*hidden)

    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(

            right_gate_raw,

            BLOCK_SIZE=BLOCK,
        )

    # --------------------------------------------------------------

        eps=eps,
    )

    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to original dtype (usually float32)
    return out_final.to(inp.dtype)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fused projection, gate (sigmoid) and mask
# --------------------------------------------------------------
@triton.jit
def fused_proj_gate_mask_kernel(
    left_proj_ptr,   # left_proj (B,N,N,hidden)
    right_proj_ptr,  # right_proj (B,N,N,hidden)
    left_gate_ptr,   # left_gate (B,N,N,hidden) – raw (pre‑sigmoid)
    right_gate_ptr,  # right_gate (B,N,N,hidden) – raw (pre‑sigmoid)
    mask_ptr,        # mask (B,N,N) – broadcasted over hidden
    left_out_ptr,    # output tensor for left side (B,N,N,hidden)
    right_out_ptr,   # output tensor for right side (B,N,N,hidden)
    hidden: tl.constexpr,     # hidden dimension (compile‑time constant)
    total: tl.constexpr,      # total number of elements (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr, # block size for the kernel
):
    """
    Computes, for every element (b,i,j,h):
        left_out  = left_proj  * sigmoid(left_gate)  * mask
        right_out = right_proj * sigmoid(right_gate) * mask
    The mask is broadcast over the hidden dimension; the sigmoid is
    evaluated inside the kernel to avoid an extra pass over memory.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total

    # Load the four tensors (projected values and raw gates)
    left_proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Sigmoid (1/(1+exp(-x))) – Triton provides exp for half‑precision
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # Compute broadcasted mask index and load the mask value
    mask_idx = offs // hidden                     # integer division
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)

    # Fuse the three multiplies
    left_out = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)

        total = left_proj_raw.numel()                      # B*N*N*hidden
        BLOCK = 16384                                       # tuned for H100
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        fused_proj_gate_mask_kernel[grid](
            left_proj_raw,
            right_proj_raw,
            left_gate_raw,
            right_gate_raw,
            mask_fp16,
            left,
            right,
            hidden,
            total,
            BLOCK_SIZE=BLOCK,
        )

    # ------------------------------------------------------------------
    # 4) Core contraction: batched matrix multiplication over the hidden dim
    # ------------------------------------------------------------------
    # Transform to (B, hidden, N, N) layout required for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, k, j)

    # Batched GEMM → (B, hidden, i, j)
    out_hidden = torch.matmul(left_perm, right_perm)

    # Back to (B, N, N, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # Multiply by the output gate (element‑wise)
    out_norm = out_norm * out_gate

    # Final projection back to the original feature dimension
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to the original dtype (commonly float32) before returning
    return out_final.to(inp.dtype)
