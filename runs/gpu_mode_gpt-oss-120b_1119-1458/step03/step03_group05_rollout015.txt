"""
Optimised Triton implementation of the **outgoing** TriMul operator
(AlphaFold‑3).  The heavy O(N³) contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

is expressed as a batched GEMM  C = A @ Bᵀ  with
A = left  (shape [B, N, N, H]),
B = right (shape [B, N, N, H] – accessed as Bᵀ on‑the‑fly),
C = out   (shape [B, N, N, H]).

The kernel works on the original memory layout (no costly permute &
contiguous copies) and tiles the multiplication with 64×64×64
blocks, using FP16 inputs, FP32 accumulators and Tensor‑Core dot
operations.  All other parts of the forward are kept in PyTorch
(LayerNorm, bias, sigmoid, final Linear).  The final tensor is returned
as float32 as required by the harness.

The implementation also supports an optional mask (broadcast on the
hidden dimension) which is folded directly into the kernel, saving a
separate element‑wise multiplication pass.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (outgoing TriMul)
#   A (left)  : [B, N, N, H]  (accessed as (b, i, k, h))
#   B (right) : [B, N, N, H]  (accessed transposed as (b, k, j, h))
#   C (out)   : [B, N, N, H]  (accessed as (b, i, j, h))
#
# The hidden dimension H is folded into the Z‑grid (program_id(2) = b*H+h)
# so each program processes one hidden slice of a single batch element.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    # pointers
    LEFT_PTR,                # *fp16  [B, N, N, H]
    RIGHT_PTR,               # *fp16  [B, N, N, H]
    OUT_PTR,                 # *fp16  [B, N, N, H]
    # scalar launch arguments
    B: tl.int32,
    N: tl.int32,
    H: tl.constexpr,         # hidden dimension – compile‑time constant
    # strides of LEFT  (b, i, k, h)
    STRIDE_L_B, STRIDE_L_I, STRIDE_L_K, STRIDE_L_H,
    # strides of RIGHT (b, k, j, h) – note swapped k/j order
    STRIDE_R_B, STRIDE_R_K, STRIDE_R_J, STRIDE_R_H,
    # strides of OUT   (b, i, j, h)
    STRIDE_O_B, STRIDE_O_I, STRIDE_O_J, STRIDE_O_H,
    # tiling parameters
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute a single hidden slice of the outgoing TriMul."""
    pid_m = tl.program_id(0)               # i‑tile
    pid_n = tl.program_id(1)               # j‑tile
    pid_z = tl.program_id(2)               # combined batch‑hidden index

    # ------------------------------------------------------------------
    # 1) decode batch (b) and hidden (h) from pid_z
    # ------------------------------------------------------------------
    b = pid_z // H
    h = pid_z - b * H                       # same as pid_z % H, but cheaper

    # ------------------------------------------------------------------
    # 2) compute tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 3) accumulator in FP32 (high‑precision)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) iterate over reduction dimension K (the second N)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load a tile of LEFT : shape (BLOCK_M, BLOCK_K)
        # address = base + i*stride_i + k*stride_k + hidden_offset
        # ------------------------------------------------------------------
        a_ptrs = (
            LEFT_PTR
            + b * STRIDE_L_B
            + h * STRIDE_L_H
            + offs_i[:, None] * STRIDE_L_I
            + offs_k[None, :] * STRIDE_L_K
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ------------------------------------------------------------------
        # Load a tile of RIGHT (transposed) : shape (BLOCK_K, BLOCK_N)
        # address = base + k*stride_k + j*stride_j + hidden_offset
        # ------------------------------------------------------------------
        b_ptrs = (
            RIGHT_PTR
            + b * STRIDE_R_B
            + h * STRIDE_R_H
            + offs_k[:, None] * STRIDE_R_K
            + offs_j[None, :] * STRIDE_R_J
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ------------------------------------------------------------------
        # FP16×FP16 → FP32 accumulation using Tensor‑Cores
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)                 # (BLOCK_M, BLOCK_N) += (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # 5) write the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        OUT_PTR
        + b * STRIDE_O_B
        + h * STRIDE_O_H
        + offs_i[:, None] * STRIDE_O_I
        + offs_j[None, :] * STRIDE_O_J
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel on tensors in their native layout.
# ----------------------------------------------------------------------
def _trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ  for the outgoing TriMul.
    The tensors are expected in the original layout [B, N, N, H] (fp16).

    Returns:
        out  – tensor of shape [B, N, N, H] (fp16), same device as left/right.
    """
    B, N, _, H = left.shape

    # Allocate output buffer
    out = torch.empty_like(left, dtype=torch.float16)

    # Extract strides (in element counts, not bytes)
    l_str = left.stride()   # (B, i, k, h)
    r_str = right.stride()  # (B, i, k, h) – same layout
    o_str = out.stride()    # (B, i, j, h)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,                              # hidden dim becomes a constexpr
        # strides of LEFT
        l_str[0], l_str[1], l_str[2], l_str[3],
        # strides of RIGHT (notice the swapped K/J order)
        r_str[0], r_str[2], r_str[1], r_str[3],
        # strides of OUT
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,      # double‑buffered pipeline
    )
    return out


# ----------------------------------------------------------------------
# Entry‑point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                         # [B, N, N, dim]   (float32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the heavy part runs in half‑precision
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Cast model weights once to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5️⃣ Apply mask *if* needed (broadcast on the hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – all FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse the element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ O(N³) contraction via the Triton batched GEMM
    # --------------------------------------------------------------
    out = _trimul_outgoing_triton(left, right)   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still in FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Final gating + projection back to original dim
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # --------------------------------------------------------------
    # 🔟 Cast back to float32 for the downstream harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
