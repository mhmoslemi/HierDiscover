    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h, 
    stride_right_b, ..., 
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    # program_id(1) = tile_i
    # program_id(2) = tile_j

    head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute b and d from head
    b = head // H
    d = head % H

    # compute offsets for left/right and output start positions
    # Use local base pointers:
    a_ptr = left_ptr + b*stride_left_b + d*stride_left_h
    b_ptr = right_ptr + b*stride_right_b + d*stride_right_h
    c_ptr = out_ptr + b*stride_out_b + d*stride_out_h

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute head index
    head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = head // H
    h = head % H

    # Compute offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base addresses
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k)

"""
Outgoing TriMul (AlphaFold‑3 style) – Triton‑accelerated forward pass.

The implementation follows the reference PyTorch module but moves the
dominant N³ einsum into a custom Triton batched GEMM kernel that works on
the native layout (B, N, N, H).  This eliminates the costly
`permute(...).contiguous()` reshapes required for the naïve `torch.bmm`
approach.  Small element‑wise kernels fuse the mask and gate multiplications.
All heavy arithmetic runs in FP16 with FP32 accumulation; final output is
returned in FP32 to match the reference model.
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Element‑wise kernels (gate + optional mask, and output‑gate)
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place: left *= left_gate ; right *= right_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(left_ptr + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def apply_gates_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    numel: tl.int64,      # total elements of left/right
    hidden_dim: tl.int64, # size of the last dimension (H)
    BLOCK: tl.constexpr,
):
    """In‑place: left *= left_gate * mask ; right *= right_gate * mask.
       `mask` has shape (B, N, N) and is broadcast over the hidden dimension."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # broadcasted mask index: each hidden entry repeats the same mask value
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    tl.store(left_ptr + offs, l * lg * m, mask=mask)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr, b_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place element‑wise multiplication a *= b."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Core N³ kernel – batched matrix multiplication for every head
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # fp16*, shape (B, N, N, H)
    right_ptr,         # fp16*, same shape
    out_ptr,           # fp16*, same shape
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Performs   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for every head `h` (hidden_dim) and batch `b`.  The computation is a
    standard GEMM per head, tiled with BLOCK_M×BLOCK_N output tiles and a
    BLOCK_K reduction dimension.
    """
    head_id = tl.program_id(0)   # 0 ≤ head_id < B*H
    tile_i   = tl.program_id(1)   # 0 ≤ tile_i < ceil(N / BLOCK_M)
    tile_j   = tl.program_id(2)   # 0 ≤ tile_j < ceil(N / BLOCK_N)

    # Decode batch and head index
    b = head_id // H
    h = head_id % H

    # Starting coordinates of the current tile
    i0 = tile_i * BLOCK_M
    j0 = tile_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (M x K)  – rows i, cols k
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i
                                 + offs_k[None, :] * stride_l_k)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # right tile (K x N) – rows k, cols j   (note the transpose)
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k
                                   + offs_j[None, :] * stride_r_i)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # GEMM: (M,K) × (K,N) → (M,N)
        acc += tl.dot(left_tile, right_tile)

    # Store the result back as FP16
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i
                           + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data :
        (input, mask, weights, config)
        - input : torch.Tensor   [B, N, N, dim]   (float32)
        - mask  : torch.Tensor   [B, N, N]       (bool/float) – can be None
        - weights : dict of model parameters (LayerNorm, Linear, etc.)
        - config : dict with keys 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 (the heavy part runs in half precision)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Convert weights to fp16 (no bias terms in the model)
    # ------------------------------------------------------------------
    w_left_proj   = w["left_proj.weight"].to(torch.float16)
    w_right_proj  = w["right_proj.weight"].to(torch.float16)
    w_left_gate   = w["left_gate.weight"].to(torch.float16)
    w_right_gate  = w["right_gate.weight"].to(torch.float16)
    w_out_gate    = w["out_gate.weight"].to(torch.float16)
    w_to_out_norm = w["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = w["to_out_norm.bias"].to(torch.float16)
    w_to_out      = w["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, w_left_proj)    # [B, N, N, H]  fp16
    right = F.linear(x_h, w_right_proj)  # same shape

    # ------------------------------------------------------------------
    # 6️⃣ Compute gate tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 7️⃣ Fuse mask (if present) and gate multiplications with a tiny Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    numel = left.numel()                     # B * N * N * H
    BLOCK = 65536                             # ~64 k elements per launch
    grid = ((numel + BLOCK - 1) // BLOCK,)

    if nomask:
        # Fast path: only gate multiplication
        apply_gates_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            right.view(-1),
            right_gate.view(-1),
            numel,
            BLOCK=BLOCK,
        )
    else:
        # Prepare mask in fp16 and flatten.  The mask is broadcast over `hidden_dim`.
        mask_h = mask.to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)                     # shape (B*N*N)
        apply_gates_mask_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            right.view(-1),
            right_gate.view(-1),
            mask_flat,
            numel,
            hidden_dim,
            BLOCK=BLOCK,
        )
    # Gates (and optional mask) are no longer needed
    del left_gate, right_gate, mask_h if not nomask else None

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation – custom batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # underscore = hidden_dim (H)
    out = torch.empty_like(left)                # allocate [B, N, N, H] fp16

    # fetch strides (in element counts, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # one program per head (batch*hidden_dim)
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles in i‑direction
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles in j‑direction
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # We can now free the projected tensors
    del left, right

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm (half‑precision weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 🔟 Multiply by the output gate (in‑place) using Triton
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)   # [B, N, N, dim] fp16

    # Return in the expected float32 precision
    return out.to(torch.float32)
