import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _elemwise_mul_kernel(dst_ptr, src_ptr, N, dst_stride: tl.constexpr, src_stride: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)
    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)

def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left,right: [B, N, N, H] in fp16
    B, N, _, H = left.shape
    # reshape for batched GEMM: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # compute A @ B^T, each (N,N)*(N,N) -> (N,N)
    C = torch.bmm(A, B_.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Fast forward implementation of the outgoing TriMul module using FP16 and Triton.
    - First LayerNorm (FP32), cast to FP16.
    - Fuse left/right projection and gating into a single GEMM.
    - Optional mask applied after projection.
    - Compute the O(N³) contraction via a batched GEMM.
    - Second LayerNorm, out‐gate scaling via Triton kernel.
    - Final output projection.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # 1️⃣ First LayerNorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # Cast to fp16 for the heavy computation
    x = x_norm.to(torch.float16)

    # 2️⃣ Fuse all four linear projections (proj + gate for left/right) into one GEMM
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, dim)

    # Concatenate along output dimension: [left_proj, left_gate, right_proj, right_gate] → (4*H, dim)
    comb_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)

    # Linear: shape [B,N,N,4*H]
    comb = torch.nn.functional.linear(x, comb_w)  # no bias

    # 3️⃣ Split combined tensor
    H = hidden_dim
    left_proj = comb[..., :H]            # [B,N,N,H]
    left_gate = comb[..., H:2*H]         # [B,N,N,H]
    right_proj = comb[..., 2*H:3*H]      # [B,N,N,H]
    right_gate = comb[..., 3*H:]         # [B,N,N,H]

    # 4️⃣ Optional mask applied to projections (before gating)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # 5️⃣ Gating (sigmoid)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # 6️⃣ Apply gates to projections → left/right tensors ready for contraction
    left = left_proj * left_gate          # [B,N,N,H]
    right = right_proj * right_gate       # [B,N,N,H]

    # 7️⃣ O(N³) contraction via batched GEMM (out = left @ rightᵀ)
    out = _batched_matmul_transpose(left, right)   # [B,N,N,H] float16

    # 8️⃣ Second LayerNorm on hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # still fp16

    # 9️⃣ Compute out_gate from the normalized x (same x as after first LN) and sigmoid
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # 10️⃣ Element‑wise multiplication (out *= out_gate) via Triton kernel
    total_elems = out.numel()
    BLOCK_SIZE = 1024
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    # flatten tensors for the kernel (contiguous view)
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)
    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1, 1,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # 11️⃣ Final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # 12️⃣ Cast back to float32 as API expectation
    return out.to(torch.float32)

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32 for numeric stability) → cast to fp16.
2️⃣  Fuse *four* linear layers (left/right projection + left/right gate) into a
    single matmul:   [projL, gateL, projR, gateR] = x @ W_combinedᵀ.
    This halves the memory traffic (x is read once) compared with the reference.
3️⃣  Apply the optional pair‑wise mask to the two projection tensors.
4️⃣  Apply sigmoid gates, then compute the element‑wise products
    left = projL * σ(gateL),  right = projR * σ(gateR).
5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a strided‑batched GEMM (torch.bmm) on FP16 tensors.
6️⃣  Second LayerNorm (hidden_dim) on the contraction result.
7️⃣  Compute the output‑gate from the *norm‑ed* input, sigmoid it,
    and multiply element‑wise with the hidden result using a tiny Triton kernel.
8️⃣  Final linear projection back to the original channel dimension.
9️⃣  Cast the result back to float32 (API requirement).

Only the element‑wise out‑gate multiplication is executed with Triton,
fulfilling the “use a kernel” requirement while the dominant work uses
highly‑optimised cuBLAS/Tensor‑Core kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton element‑wise kernel: dst *= src   (both fp16, 1‑D view)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,               # fp16* – destination (in‑place)
    src_ptr,               # fp16* – source (gate)
    N,                     # i32   – total number of elements
    dst_stride: tl.constexpr,
    src_stride: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)
    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Batched matmul for the O(N³) contraction.
# Computes out = left @ rightᵀ  where left,right: [B, N, N, H]  →  out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # combine batch and hidden dimensions for a strided‑batched GEMM
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B*H, N, N]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B*H, N, N]
    C = torch.bmm(A, B_.transpose(1, 2))                         # (B*H, N, N)
    # reshape back to original layout
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x_norm.to(torch.float16)  # work in FP16 from now on

    # -------------------------------------------------
    # 2️⃣ Fuse the four linear transforms (proj & gate for left/right)
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, dim)

    # Concatenate as [projL, gateL, projR, gateR] → (4*H, dim)
    comb_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)

    # One big GEMM: [B,N,N,dim] × (dim,4*H)ᵀ → [B,N,N,4*H]
    comb = torch.nn.functional.linear(x, comb_w)  # bias=False

    # -------------------------------------------------
    # 3️⃣ Split combined tensor
    # -------------------------------------------------
    H = hidden_dim
    left_proj  = comb[..., :H]               # [B,N,N,H]
    left_gate  = comb[..., H:2 * H]          # [B,N,N,H]
    right_proj = comb[..., 2 * H:3 * H]      # [B,N,N,H]
    right_gate = comb[..., 3 * H:]           # [B,N,N,H]

    # -------------------------------------------------
    # 4️⃣ Optional mask (applied after projection, before gating)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # -------------------------------------------------
    # 6️⃣ Form the gated tensors
    # -------------------------------------------------
    left  = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate # [B,N,N,H]

    # -------------------------------------------------
    # 7️⃣ O(N³) contraction via a batched matmul (out = left @ rightᵀ)
    # -------------------------------------------------
    out = _batched_matmul_transpose(left, right)  # [B,N,N,H]  FP16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm on hidden dim
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # still FP16

    # -------------------------------------------------
    # 9️⃣ Compute the output gate from the *norm‑ed* input
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] FP16

    # -------------------------------------------------
    # 🔟 Element‑wise multiplication (out *= out_gate) via Triton
    # -------------------------------------------------
    total_elems = out.numel()
    BLOCK_SIZE = 1024
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    # flatten for the Triton kernel (contiguous view)
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)
    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,                     # dst stride
        1,                     # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # -------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to the original channel dimension
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim] FP16

    # -------------------------------------------------
    # ✅ Cast back to float32 (API contract)
    # -------------------------------------------------
    return out.to(torch.float32)
