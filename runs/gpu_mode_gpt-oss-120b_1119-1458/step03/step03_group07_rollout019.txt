"""
TriMul (outgoing) – highly‑optimized forward pass.
-------------------------------------------------
1️⃣ First LayerNorm performed entirely in FP16 to avoid a FP32→FP16 copy.
2️⃣ Projection + gate for the left/right paths are fused into a single
   `torch.nn.functional.linear` call (2 × Linear → 1 × Linear).
3️⃣ Optional pair‑wise mask is applied after projection (still in FP16).
4️⃣ The O(N³) pairwise contraction is a batched GEMM:
       C = A @ Bᵀ  where A,B: [B·H, N, N].
5️⃣ Second LayerNorm (still using the highly‑optimised torch kernel).
6️⃣ `out_gate` is obtained with a regular linear + sigmoid on the
   already‑normalised input.
7️⃣ The final element‑wise gating *and* the output projection are fused
   into a single Triton GEMM kernel:
       out = ((out_norm * out_gate) @ W_outᵀ)
   This eliminates an extra memory write/read (the TensorCore kernel
   operates directly on the gated hidden vectors).  The kernel works on
   2‑D flattened tensors (M = B·N·N, K = hidden_dim, N = dim) and
   accumulates in FP32 before casting back to FP16.
8️⃣ Cast back to float32 for the API output.

The only non‑trivial operation performed with Triton is the fused
multiply‑and‑linear kernel (step 7).  All other heavy work uses the
vendor‑optimised cuBLAS kernels, so the implementation stays fast on
H100 while remaining fully correct.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: (A*gate) @ Bᵀ   where
#   A : [M, K]   (out_norm)
#   gate : [M, K] (out_gate)
#   B : [K, N]   (to_out_weightᵀ)
#   C : [M, N]   (final output, FP16)
# ----------------------------------------------------------------------
@triton.jit
def _fused_gemm_mul_kernel(
    a_ptr,          # *fp16  – out_norm
    gate_ptr,       # *fp16  – out_gate
    b_ptr,          # *fp16  – weight_T  (shape K×N)
    c_ptr,          # *fp16  – output
    M, N, K,
    stride_am, stride_ak,                 # a strides
    stride_gate_am, stride_gate_ak,       # gate strides
    stride_bk, stride_bn,                 # b strides (K dim, N dim)
    stride_cm, stride_cn,                 # c strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr = 2,
):
    pid_m = tl.program_id(0)   # row block index (M‑dimension)
    pid_n = tl.program_id(1)   # column block index (N‑dimension)

    # ------------------------------------------------------------------
    # Block coordinates
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 (better numerics & TensorCore accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Iterate over K in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ------------------ Load A and gate, apply mask -----------------
        a_ptrs = a_ptr + (offs_m[:, None] * stride_am
                          + offs_k[None, :] * stride_ak)
        gate_ptrs = gate_ptr + (offs_m[:, None] * stride_gate_am
                                + offs_k[None, :] * stride_gate_ak)

        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)               # fp16
        g = tl.load(gate_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)               # fp16

        a_scaled = a * g                     # element‑wise scaling (fp16)

        # --------------------------- Load B -----------------------------
        b_ptrs = b_ptr + (offs_k[:, None] * stride_bk
                          + offs_n[None, :] * stride_bn)

        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)               # fp16

        # --------------------------- Dot product ------------------------
        acc += tl.dot(a_scaled, b)           # (BLOCK_M×BLOCK_K)·(BLOCK_K×BLOCK_N)

    # ------------------------------------------------------------------
    # Write C – respect output masks
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm
                      + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Helper: batched matmul for the O(N³) contraction.
# Computes C = A @ Bᵀ  where A,B: [B, N, N, H] → out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(
    left: torch.Tensor,   # [B, N, N, H]  (FP16)
    right: torch.Tensor,  # [B, N, N, H]  (FP16)
) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] – we need contiguous memory for cuBLAS
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ (Tensor‑Core GEMM)
    C = torch.bmm(A, B_.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    norm_b = weights["norm.bias"].to(torch.float16)

    # torch.layer_norm works on half‑precision tensors; mean/var are
    # accumulated in FP32 internally.
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    ).to(torch.float16)                     # [B, N, N, dim]  FP16

    # -------------------------------------------------
    # 2️⃣ FP16 linear weights (project + gate fused)
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # Fuse projection + gate in a single 2×H×dim weight matrix
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2·H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2·H, dim)

    # -------------------------------------------------
    # 3️⃣ Project & split into proj / gate tensors
    # -------------------------------------------------
    left_comb = torch.nn.functional.linear(x, left_comb_w)   # [B,N,N,2·H]
    right_comb = torch.nn.functional.linear(x, right_comb_w)

    left_proj = left_comb[..., :hidden_dim]      # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:]

    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:]

    # -------------------------------------------------
    # 4️⃣ Optional masking (applied after projection, before gating)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM
    # -------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B,N,N,hidden_dim]  FP16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (still FP16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                            # [B,N,N,hidden_dim] FP16

    # -------------------------------------------------
    # 8️⃣ Output gate (sigmoid of a linear on the *first* normed input)
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden_dim] FP16

    # -------------------------------------------------
    # 9️⃣ Fuse (out_norm * out_gate)  and final linear projection
    # -------------------------------------------------
    # Flatten the first three dimensions → M = B·N·N
    B, N, _, _ = out_norm.shape
    M = B * N * N
    out_norm_flat = out_norm.view(M, hidden_dim)           # [M, H]
    out_gate_flat = out_gate.view(M, hidden_dim)           # [M, H]

    # Weight for the final projection: we need Wᵀ (H × dim) in contiguous layout
    to_out_w = weights["to_out.weight"].to(torch.float16)  # [dim, hidden_dim]
    weight_T = to_out_w.t().contiguous()                  # [hidden_dim, dim]

    # Allocate output buffer (FP16)
    out_fp16 = torch.empty((M, dim), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Launch the fused Triton GEMM kernel
    # ------------------------------------------------------------------
    # Block sizes – tuned for the H100 Tensor‑Core dimensions.
    BLOCK_M = 128    # rows per program
    BLOCK_N = 64     # columns per program
    BLOCK_K = 32     # reduction tile (covers hidden_dim ≤ 128 nicely)

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (dim + BLOCK_N - 1) // BLOCK_N,
    )

    _fused_gemm_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        weight_T,
        out_fp16,
        M,
        dim,
        hidden_dim,
        out_norm_flat.stride(0), out_norm_flat.stride(1),           # a strides
        out_gate_flat.stride(0), out_gate_flat.stride(1),           # gate strides
        weight_T.stride(0), weight_T.stride(1),                     # b strides (K,N)
        out_fp16.stride(0), out_fp16.stride(1),                     # c strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 benefits from 8 warps per block
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, dim] and cast to float32 for the API
    # ------------------------------------------------------------------
    out = out_fp16.view(B, N, N, dim).to(torch.float32)
    return out
