import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # batch*head index

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & (offs_k[None, :] < K),
                    other=0.0)

        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                  right: torch.Tensor,
                                  tile: int = 64) -> torch.Tensor:
    """
    Compute C = left @ rightᵀ for each (batch, hidden) pair.
    left/right shape: [B, N, N, H] (FP16), N already padded to multiple of tile.
    Returns C of same shape (FP16).
    """
    B, N, _, H = left.shape
    # Transform to [B*H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)
    C = torch.empty_like(A)

    # Strides in element units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * H)

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,
        num_stages=3,
    )
    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (gradient‑free).

    The implementation fuses all per‑pair linear projections into a single
    FP16 GEMM, applies the gating and optional mask, then performs the heavy
    ``C[i,j] = Σ_k left[i,k] * right[j,k]`` contraction using a custom
    Triton kernel that executes a batched matrix‑multiply (A @ Bᵀ) for each
    hidden dimension.  All heavy work runs in FP16; only the final result is
    cast back to FP32.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # First LayerNorm (FP32 -> FP16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # Fuse all linear projections (left_proj, right_proj,
    # left_gate, right_gate, out_gate) into a single GEMM.
    # --------------------------------------------------------------
    # Gather fp16 weight matrices
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along output dimension (out_features)
    # Shape after cat: (5 * hidden_dim, dim)
    concat_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # [5*H, dim]

    # Flatten spatial dimensions
    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.reshape(M, dim)   # [M, dim]

    # Single FP16 GEMM: Y = x_flat @ concat_weight.T
    Y = torch.nn.functional.linear(x_flat, concat_weight)  # [M, 5*H] in fp16

    # Split the concatenated output
    H = hidden_dim
    left_proj = Y[:, :H]                     # [M, H]
    right_proj = Y[:, H:2 * H]               # [M, H]
    left_gate_raw = Y[:, 2 * H:3 * H]        # [M, H]
    right_gate_raw = Y[:, 3 * H:4 * H]       # [M, H]
    out_gate_raw = Y[:, 4 * H:5 * H]         # [M, H]

    # --------------------------------------------------------------
    # Apply sigmoid gates (still FP16).  Mask is applied later.
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # Fuse projection & gate (mask will be applied after if needed)
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        # Mask has shape (B, N, N); broadcast over hidden
        mask_fp16 = mask_tensor.to(torch.float16).reshape(M, 1)  # [M,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # Reshape to original 4‑D layout: [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # Pad spatial dimensions to a multiple of TILE to simplify the kernel.
    # --------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # --------------------------------------------------------------
    # Heavy O(N³) contraction via Triton (batched matmul A @ Bᵀ).
    # --------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, H]

    # Remove padding if applied
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # --------------------------------------------------------------
    # Second LayerNorm on hidden dimension (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # Output gating (sigmoid already computed) + final projection.
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] in fp16

    # --------------------------------------------------------------
    # Cast back to FP32 (expected downstream dtype)
    # --------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32 → FP16).
2️⃣ Fuse *all* per‑pair linear projections (left/right proj, left/right
   gate, out‑gate) into a **single** FP16 GEMM:
       Y = X_flat @ W_concat.T      (W_concat has shape (5·H, dim))
   Afterwards split Y into the five tensors and apply sigmoid to the
   three gate tensors.
3️⃣ Fuse the mask (if present) and the two projection‑gate products:
       left  = (X·W_left)   * sigmoid(X·W_left_gate)   * mask
       right = (X·W_right)  * sigmoid(X·W_right_gate)  * mask
   and keep the output‑gate tensor for later use.
4️⃣ Pad the sequence dimension to a multiple of 64 (the Triton tile size).
5️⃣ Heavy O(N³) contraction
       out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
   is performed as a batched matrix‑multiply C = A @ Bᵀ for each
   (batch, head) pair using a custom Triton kernel (64×64×64 tiles,
   FP16×FP16→FP32 accumulation, 16 warps, 3‑stage pipeline).
6️⃣ Remove the padding and apply the second LayerNorm (hidden dim).
7️⃣ Multiply by the previously computed output‑gate and apply the final
   linear projection back to `dim`.
8️⃣ Cast the result to FP32 (the expected external dtype).

Only the O(N³) contraction is written in Triton; all surrounding
operations use high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]    (M = batch * hidden, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,
    B_ptr,
    C_ptr,
    M,
    N,
    K,
    stride_am,
    stride_ai,
    stride_ak,
    stride_bm,
    stride_bj,
    stride_bk,
    stride_cm,
    stride_ci,
    stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile along the first N dimension (i)
    pid_n = tl.program_id(1)  # tile along the second N dimension (j)
    pid_z = tl.program_id(2)  # batch * head index

    # ------------------- tile offsets -------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # FP32 accumulator (higher precision for the reduction)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------- main k‑loop --------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A[i, k] tile
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # FP16

        # Bᵀ[k, j] tile = B[k, j] (implicit transpose)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # FP16

        # tensor‑core dot (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # ------------------- write result --------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiply ``C = left @ rightᵀ`` for each (batch, hidden)
    pair.  All tensors must be FP16 and ``N`` already padded to a multiple
    of ``tile``.
    """
    B, N, _, H = left.shape

    # Bring the hidden dimension into the batch dimension:
    #   left_perm / right_perm : [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)   # [B*H, N, N]
    B_ = right_perm.view(B * H, N, N)  # [B*H, N, N]

    C = torch.empty_like(A)  # FP16 output

    # Strides (in element units)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,
        num_stages=3,
    )

    # Reshape back to the original 4‑D layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim] (float32)
        - mask_tensor  : [B, N, N] (bool or float) – unused if config["nomask"]=True
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, same device as the input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – float32
    x = x_norm.to(torch.float16)  # heavy work will use FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into ONE GEMM
    # ------------------------------------------------------------------
    # Cast all required weight matrices once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the *output* dimension → (5·H, dim)
    concat_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*H, dim)

    B, N, _, _ = x.shape
    M = B * N * N                       # number of (i,k) pairs per batch
    x_flat = x.reshape(M, dim)          # [M, dim]

    # Single large FP16 GEMM: Y = x_flat @ concat_w.T  → shape [M, 5·H]
    Y = torch.nn.functional.linear(x_flat, concat_w)  # FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split and apply gates (sigmoid).  Mask (if any) is applied later.
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj      = Y[:, :H]                # [M, H]
    right_proj     = Y[:, H:2 * H]           # [M, H]
    left_gate_raw  = Y[:, 2 * H:3 * H]       # [M, H]
    right_gate_raw = Y[:, 3 * H:4 * H]       # [M, H]
    out_gate_raw   = Y[:, 4 * H:5 * H]       # [M, H]

    left_gate  = torch.sigmoid(left_gate_raw)   # FP16
    right_gate = torch.sigmoid(right_gate_raw)  # FP16
    out_gate   = torch.sigmoid(out_gate_raw)    # FP16

    # Fuse projection & gate (mask will be inserted afterwards)
    left = left_proj * left_gate                 # [M, H]
    right = right_proj * right_gate              # [M, H]

    if not nomask:
        # mask shape: (B, N, N) → flatten to (M, 1) and broadcast over H
        mask_fp16 = mask_tensor.to(torch.float16).reshape(M, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Restore the original 4‑D layout
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # will be used after the O(N³) contraction

    # ------------------------------------------------------------------
    # 4️⃣ Pad spatial dimensions to a multiple of TILE (64) to drop bounds checks
    # ------------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – batched matmul A @ Bᵀ (Triton)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, H]

    # Strip padding if it was added
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gating + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)       # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the downstream model
    # ------------------------------------------------------------------
    return out.to(torch.float32)
