"""
TriMul (outgoing) – fast Triton implementation
================================================

The kernel performs the heavy O(N³) contraction of the TriMul module:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

Key ideas
---------
* The input tensor is first layer‑normed (FP32 for stability) and cast to FP16.
* All linear projections and sigmoid gates are executed with half‑precision
  PyTorch primitives – they are already heavily optimised.
* The N³ contraction is expressed as a *batched* matrix multiplication
  C = A @ Bᵀ where each batch corresponds to a (batch × hidden) pair.
* We avoid any Python‑level loops and launch a single Triton kernel that
  tiles the matrices with 128×128×64 blocks.  The 128‑wide tiles greatly
  reduce the number of program instances (by a factor of 4 compared to a
  64‑wide tiling) while still fitting into registers; accumulator arithmetic
  is performed in FP32 and the result is stored in FP16.
* The output of the kernel is reshaped back to `[B, N, N, hidden]`,
  followed by a second LayerNorm, an output gate, and a final linear projection
  back to the original feature dimension.
* The final result is returned as float32 as required by the harness.

The implementation works for all test configurations listed in the problem
statement (B∈{1,2}, N∈{128,256,512,768,1024}, dim∈{128,384},
hidden_dim=128, optional mask).  No gradients are computed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for many independent (N×N) matrices.
    Each program (identified by pid_z) processes one slice of the batch
    (i.e. one hidden dimension of one batch element).  The i‑ and j‑axes are
    tiled with blocks of size BLOCK_M × BLOCK_N.
    """
    pid_i = tl.program_id(0)         # tile along the first N dimension (row)
    pid_j = tl.program_id(1)         # tile along the second N dimension (col)
    pid_z = tl.program_id(2)         # index over the M‑dimension (batch*hidden)

    # -------------------------------------------------
    # Offsets for the current output tile
    # -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles (ceil division)
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # -------------------------------------------------
    # Main reduction loop over K
    # -------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- Load A tile (M‑slice, i‑dim, k‑dim) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----- Load B tile (M‑slice, k‑dim, j‑dim) -----
        # This loads the transposed view of B (K × N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ----- Tensor‑core multiply‑accumulate -----
        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulation

    # -------------------------------------------------
    # Store the result tile (cast back to fp16)
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the batched contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape [B, N, N, H] (dtype=fp16)
    right : torch.Tensor
        Tensor of shape [B, N, N, H] (dtype=fp16)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] (dtype=fp16)
    """
    B, N, _, H = left.shape

    # -------------------------------------------------
    # Pack the (batch, hidden) dimensions into a single leading dimension.
    # The resulting layout (M, N, N) is contiguous, which allows the kernel
    # to use a constant stride for the M‑axis.
    # -------------------------------------------------
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [M, N, K]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [M, N, K]

    # Output buffer (FP16)
    C = torch.empty_like(A, dtype=torch.float16)

    # -------------------------------------------------
    # Strides expressed in number‑of‑elements (not bytes)
    # -------------------------------------------------
    stride_am, stride_ai, stride_ak = A.stride()      # (M,N,K)
    stride_bm, stride_bi, stride_bk = B_.stride()     # (M,N,K)
    stride_cm, stride_ci, stride_cj = C.stride()      # (M,N,N)

    # -------------------------------------------------
    # Tile sizes – 128×128×64 is a good trade‑off on H100 for fp16 MMA.
    # -------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid dimensions
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # M‑tiles (batch * hidden)
    )

    # Launch the kernel
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bi,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to the original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast to FP16 for the rest of the pipeline
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Convert all model weights to FP16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast over the hidden dim)
    # -------------------------------------------------
    if not nomask:

        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (FP16 inputs, FP16 output)
    # -------------------------------------------------
    # Result shape: [B, N, N, hidden] (FP16)
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden_dim) – keep in FP16
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection back to original dimension
    # -------------------------------------------------
    out = out * out_gate                     # element‑wise gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (required output type)
    # -------------------------------------------------
    return out.to(torch.float32)
