@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          stride_ah, stride_am, stride_ak,
                          stride_bh, stride_bk, stride_bn,
                          stride_ch, stride_cm, stride_cn,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # linear offsets for A and B blocks
    a_ptrs = a_ptr + pid_batch * stride_ah + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * stride_bh + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute all k blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs)
        b = tl.load(b_ptrs)
        # dot product: a shape (M, K), b shape (K, N)
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result back as fp16 (or float32 -> convert)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c)

"""
high‑performance Triton implementation of the "outgoing" TriMul operator (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input (fp32 for stability).
2. Cast to fp16 and compute the four linear projections:
      left  = x @ W_left_proj
      right = x @ W_right_proj
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
3. (optional) apply the pairwise mask to ``left`` and ``right``.
4. Fuse element‑wise multiplication of ``left``/``right`` with their gates
   using a tiny Triton kernel (``apply_gates_kernel``).  This replaces several
   separate ``*`` and ``sigmoid`` calls and works in‐place.
5. Reshape the tensors to (batch*head, N, N) and perform the dominant
   N³ operation: a batched matrix multiplication ``C = A @ Bᵀ`` where
   ``A = left`` and ``B = right``.  This is implemented with a custom
   Triton GEMM kernel (``batched_matmul_kernel``) that works directly on
   fp16 data and accumulates in fp32, fully exploiting H100 Tensor‑Cores.
6. Restore the original layout (B, N, N, hidden_dim) and apply a second
   layer‑norm (fp16 weights).
7. Fuse the final element‑wise multiplication by ``out_gate`` with a
   Triton kernel (``mul_inplace_kernel``).
8. Linear projection back to the original ``dim`` and cast the result to
   float32.

The kernels start from a flat view of the tensors, use large block sizes
(128×128×64) for maximal Tensor‑Core occupancy, and avoid unnecessary
copies.  The whole forward pass runs without autograd and returns a
float32 tensor, matching the reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Tiny Triton kernels for element‑wise fusions
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,      # fp16* flattened left tensor
    left_gate_ptr, # fp16* flattened left_gate
    right_ptr,     # fp16* flattened right tensor
    right_gate_ptr,# fp16* flattened right_gate
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place:
        left  = left  * left_gate
        right = right * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def mul_inplace_kernel(
    out_ptr,     # fp16* flattened output tensor (post‑norm)
    gate_ptr,    # fp16* flattened out_gate tensor
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        out = out * gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    o = tl.load(out_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, o * g, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM kernel (left @ rightᵀ) – the N³ core of TriMul
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B where
        A : (batch, M, K)
        B : (batch, K, N)   (already transposed w.r.t. the original
                              right matrix, i.e. B = right.transpose(1,2))
        C : (batch, M, N)

    All tensors are fp16; accumulation is in fp32.
    """

    pid_batch = tl.program_id(0)   # which head (batch*head)
    pid_m     = tl.program_id(1)   # row tile
    pid_n     = tl.program_id(2)   # col tile

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch/head
    a_batch = pid_batch * stride_ah
    b_batch = pid_batch * stride_bh
    c_batch = pid_batch * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        # k‑axis offsets
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current k‑block
        a_ptrs = a_ptr + a_batch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + b_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # masked loads (out‑of‑bounds elements are zero)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)   # fp32 accumulation

    # write result
    c_ptrs = c_ptr + c_batch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn

    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))

# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float), may be ignored
        - weights      : dict of model Parameter tensors
        - config       : dict containing at least 'dim' and 'hidden_dim'.
                         May also contain 'nomask' (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """

    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                         # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)      # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, w_right_proj)     # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gate pre‑activations (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates (in‑place) using Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK_EL = 65536            # ~64k elements per program (~256 KB for fp16)
    grid_el = ((numel + BLOCK_EL - 1) // BLOCK_EL,)

    apply_gates_kernel[grid_el](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK_EL,
    )
    # Gates no longer needed – free memory
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation – batched matmul via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden_dim
    total_heads = B * H

    # reshape to (total_heads, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # we need Bᵀ => transpose last two dims of right_mat
    right_mat_T = right_mat.transpose(1, 2)   # (total_heads, N, N) with swapped strides

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # stride information (int64)
    stride_ah, stride_am, stride_ak = left_mat.stride()
    stride_bh, stride_bk, stride_bn = right_mat_T.stride()
    stride_ch, stride_cm, stride_cn = out_mat.stride()

    # Tuned block sizes – chosen to match H100 Tensor‑Core tiles
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure out_mat is materialized before further ops
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, hidden_dim] (still FP16)
    # ------------------------------------------------------------------
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # shape (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 🔟 Post‑norm over hidden_dim (FP16 parameters, FP32 accumulation)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )   # fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Multiply by the output gate (in‑place) via Triton
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK_EL - 1) // BLOCK_EL,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK_EL,
    )
    del out_gate   # free GPU memory

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # Return in the dtype expected by downstream code (float32)
    # ------------------------------------------------------------------
    return final.to(torch.float32)
