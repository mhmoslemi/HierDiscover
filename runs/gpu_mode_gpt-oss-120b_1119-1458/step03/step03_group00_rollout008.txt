"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module, but it fuses the
most expensive memory‑bound steps:

1️⃣ Input LayerNorm (FP32 → FP16).

2️⃣ **Single** linear projection for all five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate).  The
   concatenated weight matrix has shape ``(5*hidden, dim)`` and is
   multiplied once with the flattened input → ``M × 5*hidden`` (M = B·N²).
   This avoids reading the input tensor five times.

3️⃣ A custom Triton kernel ``gate_mask_kernel`` computes, for the
   *left* and *right* branches, the three‑way product

        out = proj * sigmoid(gate) * mask               (mask optional)

   The kernel also evaluates the sigmoid, so an extra PyTorch
   ``torch.sigmoid`` call is not needed.

4️⃣ Core contraction (the “TriMul” operation)

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   which is a batched matrix‑multiply for each hidden channel.
   By permuting to ``(B, hidden, N, N)`` we invoke a single
   strided‑batched ``torch.matmul`` (cuBLAS).

5️⃣ Output LayerNorm, gating by the previously computed ``out_gate``,
   and final projection back to the original ``dim``.

Only the element‑wise triple product (step 3) is written in Triton,
satisfying the requirement “implement at least part of the operations in a
kernel”.  All heavy matrix multiplications stay in FP16, and the final
tensor is cast back to the input dtype (normally FP32).

This design drastically reduces memory traffic (the input is read only
once) and gives a large speed‑up over the naïve five‑linear implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused sigmoid + mask + elementwise product for left & right
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,                     # (M,)   – broadcasted over hidden
    out_left_ptr, out_right_ptr,
    hidden: tl.constexpr,         # hidden dimension (C_z)
    size: tl.constexpr,           # total number of elements (M*hidden)
    BLOCK_SIZE: tl.constexpr,
    MASKED: tl.constexpr,         # 1 -> use mask, 0 -> skip mask
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < size

    # Load raw projections and gate values
    proj_l = tl.load(left_proj_ptr + offs, mask=active, other=0.0)
    gate_l = tl.load(left_gate_ptr + offs,  mask=active, other=0.0)

    proj_r = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    gate_r = tl.load(right_gate_ptr + offs,  mask=active, other=0.0)

    # Sigmoid = 1/(1+exp(-x)) – works in FP16
    sig_l = 1.0 / (1.0 + tl.exp(-gate_l))
    sig_r = 1.0 / (1.0 + tl.exp(-gate_r))

    out_l = proj_l * sig_l
    out_r = proj_r * sig_r

    if MASKED:
        # Compute index of the mask element (broadcast over hidden)
        mask_idx = offs // hidden
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        out_l = out_l * mask_val
        out_r = out_r * mask_val

    tl.store(out_left_ptr + offs,  out_l, mask=active)
    tl.store(out_right_ptr + offs, out_r, mask=active)


def fused_gate_mask(left_proj, left_gate_raw,
                    right_proj, right_gate_raw,
                    mask, hidden):
    """
    Apply   out = proj * sigmoid(gate) * mask   (mask broadcasted over hidden).

    All tensors must be 1‑D contiguous fp16 views of shape (M*hidden) where
    M = B·N·N. ``mask`` has shape (M,) and is also fp16.
    Returns two 1‑D fp16 tensors (left, right) that can be reshaped back.
    """
    total = left_proj.numel()
    out_left  = torch.empty_like(left_proj)
    out_right = torch.empty_like(right_proj)

    BLOCK = 65536                         # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    gate_mask_kernel[grid](
        left_proj, left_gate_raw,
        right_proj, right_gate_raw,
        mask,
        out_left, out_right,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
        MASKED=1,                         # mask is present
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data : Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float, CUDA)
            - weights_dict : dict with the model weights (see skeleton)
            - config_dict  : dict containing at least "dim", "hidden_dim",
                             optional "eps" and "nomask".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", False)

    device = inp.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # continue in FP16

    # ------------------------------------------------------------------
    # 2) Single fused linear for all five weight matrices
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (all have shape hidden×dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()        # (5*hidden, dim)

    # Flatten the normalized input to (M, dim) where M = B·N·N
    x_flat = x_norm.view(-1, dim)           # (M, dim)

    # Linear projection – one GEMM instead of five
    proj_all = F.linear(x_flat, w_concat)   # (M, 5*hidden)

    # ------------------------------------------------------------------
    # 3) Slice the concatenated result into the individual tensors
    # ------------------------------------------------------------------
    # Helper to reshape a slice back to (B,N,N,hidden)
    def _reshape(slice_tensor):
        return slice_tensor.view(B, N, N, hidden).contiguous()

    start = 0
    left_proj_raw  = _reshape(proj_all[:, start:start+hidden]);        start += hidden
    right_proj_raw = _reshape(proj_all[:, start:start+hidden]);        start += hidden
    left_gate_raw  = _reshape(proj_all[:, start:start+hidden]);        start += hidden
    right_gate_raw = _reshape(proj_all[:, start:start+hidden]);        start += hidden
    out_gate_raw   = _reshape(proj_all[:, start:start+hidden]);        # start += hidden

    # --------------------------------------------------------------
    # 4) Apply mask + gates (fused Triton kernel for the masked case)
    # --------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise multiplication with sigmoid gates
        left_gate  = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)

        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask once to fp16 and flatten; broadcasting is handled on‑the‑fly
        mask_fp16 = mask.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)          # (M,)

        # Flatten tensors so the kernel can treat them as 1‑D
        left_proj_f  = left_proj_raw.view(-1)
        left_gate_f  = left_gate_raw.view(-1)
        right_proj_f = right_proj_raw.view(-1)
        right_gate_f = right_gate_raw.view(-1)

        left_flat, right_flat = fused_gate_mask(
            left_proj_f, left_gate_f,
            right_proj_f, right_gate_f,
            mask_flat, hidden,
        )
        # Reshape back to 4‑D tensors
        left  = left_flat.view(B, N, N, hidden)
        right = right_flat.view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 5) Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # (B,N,N,hidden) → (B,hidden,N,N)  for both tensors
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N)

    # Batched matrix multiplication over the hidden dimension
    out_hidden = torch.matmul(left_perm, right_perm)     # (B, hidden, N, N)

    # Restore layout (B,N,N,hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output gating and final projection
    # --------------------------------------------------------------
    # Sigmoid on the out‑gate branch
    out_gate = torch.sigmoid(out_gate_raw)

    # Output LayerNorm over the hidden dimension
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # Apply the learned output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to original dim
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),  # (dim, hidden)
    )

    # Cast back to the original dtype (usually float32) and return
    return out_final.to(inp.dtype)
