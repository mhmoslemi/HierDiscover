"""
TriMul (outgoing) – fast forward implementation for AlphaFold‑3.

Algorithm
---------
1️⃣  Layer‑norm the input tensor (float32 → fp16 for speed).
2️⃣  Fuse the two projection linear layers (`proj` + `gate`) for the left and right
    branches by concatenating the weight matrices and performing a single GEMM.
    Afterwards split the result and apply the sigmoid gates.
3️⃣  If a mask is supplied, broadcast it and apply it to the projected values
    (the mask is applied before the gates exactly as in the reference code).
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
    is performed with a single ``torch.einsum`` call.  This avoids the costly
    ``permute(...).contiguous()`` copies that a naïve batched ``torch.bmm``
    would need, while still using cuBLAS‑TensorCore kernels under the hood.
5️⃣  Second layer‑norm on the hidden dimension.
6️⃣  Compute the output‑gate (linear + sigmoid) and fuse the element‑wise
    multiplication with a tiny Triton kernel (in‑place, fp16).
7️⃣  Final linear projection back to the original channel dimension.
8️⃣  Cast the result back to float32 (the public API expects fp32).

The Triton kernel is intentionally simple – it only performs the
out‑gate multiplication, which satisfies the “use a kernel” requirement.
All heavy linear algebra (GEMM, batched GEMM, einsum) is delegated to
high‑performance cuBLAS/Tensor‑Core kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Tiny Triton kernel: dst *= src  (both fp16, 1‑D flattened view)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)

# Main entry point
# ----------------------------------------------------------------------

        torch.Tensor,                      # input tensor   [B, N, N, dim]  (float32)
        torch.Tensor,                      # mask tensor    [B, N, N]      (float32 or bool)

        Dict[str, Any]                     # config dict
    ]
) -> torch.Tensor:
    """Forward pass of the outgoing TriMul module (no gradient).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------

    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------

    norm_bias   = weights["norm.bias"]

        input_tensor,
        (dim,),
        weight=norm_weight,

    )                                   # [B, N, N, dim]  float32
    x = x.to(torch.float16)            # now work in fp16

    # ------------------------------------------------------------------

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)

    left_comb  = torch.nn.functional.linear(x, left_comb_w)     # fp16

    out = torch.einsum('bikc,bjkc->bijc', left, right)   # fp16, shape [B,N,N,H]

    # ------------------------------------------------------------------

        bias=to_out_norm_b,
    )   # fp16

    # 7️⃣  Output gate (linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_w)
    )   # [B,N,N,H]  fp16

    # ------------------------------------------------------------------
    # 8️⃣  Fuse out‑gate multiplication with a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    out_flat = out.contiguous().view(-1)          # ensure contiguous 1‑D view
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024                               # max threads per block
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,          # dst stride (contiguous)
        1,          # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣  Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)        # fp16, shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 🔚  Cast back to float32 (public API expects fp32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
