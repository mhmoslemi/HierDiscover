"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

The module computes
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where
    left  = (x @ W_left )      ⊙ sigmoid(x @ W_left_gate)
    right = (x @ W_right)      ⊙ sigmoid(x @ W_right_gate)

The heavy O(N³·H) contraction is performed with a custom Triton kernel
that evaluates a batched matrix multiply C = A @ Bᵀ with tensor‑core
MMA (fp16 → fp32 accumulation).  The kernel works on tiles of 128×128
in the output dimensions and 64 in the reduction dimension (all input
sizes in the test suite are multiples of 64, so no boundary‑mask is
required).

All other operations (layer‑norms, linear projections, gating,
masking) are carried out with high‑performance PyTorch primitives in
half‑precision; the final result is returned in float32.  At least one
operation (the contraction) is therefore realised in Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16; accumulator is fp32.
#   The kernel assumes N and K are multiples of BLOCK_M/N/K.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    N,                   # i32    sequence length (also K)
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch * hidden dimension packed into `M`.
    """
    pid_m = tl.program_id(0)   # tile index for the first N dimension (i)
    pid_n = tl.program_id(1)   # tile index for the second N dimension (j)
    pid_z = tl.program_id(2)   # index over M = batch * hidden

    # ------------------------------------------------------------------
    # Tile offsets (no bounds checks – N is a multiple of BLOCK_*)
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over reduction dimension K in BLOCK_K‑sized steps
    # ------------------------------------------------------------------
    num_k_tiles = N // BLOCK_K
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs)                     # fp16

        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs)                     # fp16

        # Mixed‑precision matrix multiply (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result tile as fp16
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16))


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.
    Args:
        left, right – torch.Tensor of shape [B, N, N, H] (dtype=fp16, contiguous)
    Returns:
        out – torch.Tensor of shape [B, N, N, H] (dtype=fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Pack batch and hidden dimension into a leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides expressed in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()    # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()   # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()    # (N*N, N, 1)

    # Tile sizes – chosen to be multiples of the test sequence lengths.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Sanity check (the kernel does not handle remainders)
    assert N % BLOCK_M == 0 and N % BLOCK_N == 0 and N % BLOCK_K == 0, (
        f"Sequence length {N} must be a multiple of {BLOCK_M}/{BLOCK_N}/{BLOCK_K}"
    )

    grid = (
        N // BLOCK_M,           # i‑tiles
        N // BLOCK_N,           # j‑tiles
        B * H,                  # packed batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float32)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``, optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] in float32.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣  LayerNorm on the full dim (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣  Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣  Cast model weights once to fp16
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣  Linear projections (fp16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w) # [B,N,N,hidden]

    # -------------------------------------------------
    # 5️⃣  Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣  Gating (sigmoid) – keep everything in fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gate into projections to minimise memory traffic
    torch.mul(left, left_gate, out=left)
    torch.mul(right, right_gate, out=right)

    # -------------------------------------------------
    # 7️⃣  Heavy O(N³) contraction via Triton (fp16 in, fp16 out)
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 8️⃣  Second LayerNorm on the hidden dimension (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣  Final gating + projection back to `dim` (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)     # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 🔟  Cast result back to float32 for the downstream harness
    # -------------------------------------------------
    return out.to(torch.float32)
