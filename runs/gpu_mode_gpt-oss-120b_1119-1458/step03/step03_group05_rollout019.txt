```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for batched matmul left @ right^T
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j]
    All tensors are stored in fp16. Accumulator is fp32.
    """
    pid_m = tl.program_id(0)   # i tile id
    pid_n = tl.program_id(1)   # j tile id
    pid_z = tl.program_id(2)   # batch*hidden id

    # offsets for current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile (M,N,K) -> (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # B tile (M,K,N) -> (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        acc += tl.dot(a, b)

    # write back
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM `out = left @ right.transpose(-2, -1)` for tensors
    of shape [B, N, N, H] (fp16). Returns tensor of same shape.
    The implementation packs the (B, H) dimensions into a single batch
    dimension and then launches a 3‑D Triton kernel.
    """
    B, N, _, H = left.shape
    # pack batch and hidden
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [M, N, K]
    # right must be transposed on the last two axes before view
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # [M, K, N]

    out = torch.empty_like(left_, dtype=torch.float16)

    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tunable block sizes – empirically good for H100
    # We keep a single good configuration but allow a bigger tile for large N
    if N <= 256:
        BLOCK_M = BLOCK_N = BLOCK_K = 64
    elif N <= 512:
        BLOCK_M = BLOCK_N = BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        left_,
        right_,
        out,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # Restore shape
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32/bool) – optional
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # layer‑norm before any heavy work (float32 for stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # --------------------------------------------------------------
    # compute gating signals in float32 (more accurate)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # down‑cast to fp16 for the expensive contraction
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # linear projections (fp16)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_f16, left_proj_w)   # [B, N, N, H], fp16
    right = torch.nn.functional.linear(x_f16, right_proj_w)  # same

    # --------------------------------------------------------------
    # optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_expand = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_expand
        right = right * mask_expand

    # --------------------------------------------------------------
    # apply gates (fp16)
    # --------------------------------------------------------------
    left_gate_f16 = left_gate.to(torch.float16)
    right_gate_f16 = right_gate.to(torch.float16)
    left = left * left_gate_f16
    right = right * right_gate_f16

    # --------------------------------------------------------------
    # O(N³) contraction via Triton
    # --------------------------------------------------------------
    # left/right are [B, N, N, H] → perform out = Σ_k left[:, i, k, :] * right[:, j, k, :]
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # second LayerNorm (fp16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # --------------------------------------------------------------
    # final gating + projection back to dim
    # --------------------------------------------------------------
    out_gate_f16 = out_gate.to(torch.float16)
```

"""
Highly‑optimized forward pass of the **outgoing** TriMul module (AlphaFold‑3).

Algorithm
---------
1. **LayerNorm** on the input (float32) – this stabilises the statistics.
2. **Gating signals** (left‑gate, right‑gate, out‑gate) are computed in
   float32 for numerical accuracy.
3. The input is down‑cast to fp16 and projected with two linear layers
   (`left_proj`, `right_proj`).  Optional pairwise mask is applied here.
4. Gating scalars are cast to fp16 and fused element‑wise into the projected
   tensors.
5. The expensive cubic contraction  

      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   is performed by a custom Triton kernel that treats the batch‑and‑hidden
   dimensions as a single batch (M = B·H) and executes a batched GEMM
   C = A @ Bᵀ with FP16 operands and FP32 accumulation.  The kernel uses a
   64 × 64 × 64 tile, double‑buffered pipeline (`num_stages=3`) and 8 warps.
6. **LayerNorm** on the hidden tensor (fp16), followed by the out‑gate.
7. Final linear projection back to the original feature dimension.
8. Cast the result back to float32 (required by the harness).

The implementation keeps all heavy data in fp16, limits memory traffic by
in‑place operations and avoids any gradient computation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : [M, N, K]   (left tensor, A[M,i,k])
#   B : [M, K, N]   (right tensor transposed, B[M,k,j])
#   C : [M, N, N]   (output, C[M,i,j])
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                 # *fp16 pointers
    M, N, K,                             # i32 problem sizes
    stride_am, stride_ai, stride_ak,     # strides of A
    stride_bm, stride_bk, stride_bj,     # strides of B (note swapped K‑N order)
    stride_cm, stride_ci, stride_cj,     # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one tile of the batched matmul.
    Accumulation is performed in fp32 to retain accuracy.
    """
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M dimension)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---------- load A tile (M,i,k) ----------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---------- load B tile (M,k,j) ----------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # fp16 × fp16 → fp32 accumulation (tensor‑core)
        acc += tl.dot(a, b)

    # ---------- write result ----------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction for the TriMul module:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left, right : torch.Tensor of shape [B, N, N, H] (fp16)

    Returns
    -------
    out : torch.Tensor of shape [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape

    # Pack batch & hidden dimensions → [M, N, K] where M = B*H
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Right must be transposed on the last two axes before flattening
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out = torch.empty_like(left_, dtype=torch.float16)

    # strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tile size – 64 works well for all supported N on H100.
    BLOCK_M = BLOCK_N = BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        left_,
        right_,
        out,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,          # double‑buffered pipeline
    )

    # Restore original layout [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : dict of model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → stable statistics)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Compute gating signals in float32 (more accurate)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the heavy linear projections + O(N³) step
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H] (fp16)
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # same

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left.mul_(mask_fp16)
        right.mul_(mask_fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gating (cast gates to fp16, in‑place multiply)
    # ------------------------------------------------------------------
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)
    left.mul_(left_gate_fp16)
    right.mul_(right_gate_fp16)

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm on the hidden dimension (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + projection back to original dim
    # ------------------------------------------------------------------
    out_gate_fp16 = out_gate.to(torch.float16)
    out.mul_(out_gate_fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (harness expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
