@triton.jit
def fused_mul3_masked_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,
    hidden: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    mask_idx = offs // hidden
    b = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)

    out = a * c * b
    tl.store(out_ptr + offs, out, mask=mask)
```

    # detect data types
    device = inp.device
    dtype = inp.dtype

    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # layernorm weights
    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias = to_fp16(weights["norm.bias"])

    # concat linear weights
    left_proj_weight = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])
    left_gate_weight = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight = to_fp16(weights["out_gate.weight"])
    # Concatenate along out_features dimension (dim 0)
    all_proj_weight = torch.cat([left_proj_weight,
                                 right_proj_weight,
                                 left_gate_weight,
                                 right_gate_weight,
                                 out_gate_weight], dim=0)  # shape (5*hidden, dim)

    # to_out_norm weights
    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias = to_fp16(weights["to_out_norm.bias"])
    # to_out weight
    to_out_weight = to_fp16(weights["to_out.weight"])

    # Input LayerNorm
    x = F.layer_norm(inp, normalized_shape=(inp.shape[-1],),
                     weight=norm_weight, bias=norm_bias, eps=eps)
    x = x.to(torch.float16)

    B, N, _, dim = x.shape

    # fused linear projection for all 5 sub-modules
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)
    proj_flat = torch.matmul(x_flat, all_proj_weight.t())  # (B*N*N, 5*hidden)
    proj = proj_flat.view(B, N, N, 5*hidden)

    # split
    h = hidden
    left_proj_raw = proj[..., :h]
    right_proj_raw = proj[..., h:2*h]
    left_gate_raw = proj[..., 2*h:3*h]
    right_gate_raw = proj[..., 3*h:4*h]
    out_gate_raw = proj[..., 4*h:]

    # apply sigmoid gate
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # mask and gate fusion
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask.to(torch.float16)
        left = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # core contraction
    # Arrange for batched matmul: shape (B*hidden, N, N)
    # left: (B, hidden, i, k) ; right: (B, hidden, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    left_view = left_perm.view(B*hidden, N, N)
    right_view = right_perm.view(B*hidden, N, N)
    out_hidden = torch.bmm(left_view, right_view)  # (B*hidden, N, N)
    out = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # output layernorm
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)
    # apply out gate
    out_norm = out_norm * out_gate

    # final linear projection
    out_final = F.linear(out_norm, to_out_weight)

    # cast back to input dtype
    return out_final.to(dtype)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – improves speed while keeping
    numerical stability.

2️⃣ **Fused linear projection** – The five linear layers
    (left_proj, right_proj, left_gate, right_gate, out_gate) are
    merged into a single GEMM:
        (B·N·N, dim) × (5·hidden, dim)ᵀ → (B·N·N, 5·hidden)
    The result is reshaped and sliced into the five tensors.
    This replaces five separate GEMMs with one, cutting the cost of
    the projection stage by ~5×.

3️⃣ Gate activations (sigmoid) are applied in FP16.

4️⃣ **Mask‑gate fusion** – When a mask is present the element‑wise
    product `proj * gate * mask` is fused into a Triton kernel
    (`fused_mul3_masked`).  The mask is broadcast on‑the‑fly,
    avoiding an explicit expansion to the hidden dimension.

5️⃣ Core contraction (the “TriMul” operation) is
    `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`.  By
    permuting to `(B, hidden, N, N)` we can use a single
    strided‑batched GEMM (`torch.bmm`) which is executed by cuBLAS’s
    tensor cores in FP16.

6️⃣ Output LayerNorm, output gate, and final projection back to the
    original `dim` are performed in FP16.  The final tensor is cast
    back to the input dtype (normally float32).

Only the mask‑gate multiplication is custom‑written in Triton, while
the heavy linear algebra relies on highly‑optimised PyTorch / cuBLAS
kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise a * b * c   (c is a 3‑D mask broadcasted
# over the hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,                # left_proj or right_proj   (B,N,N,hidden)
    b_ptr,                # mask                     (B,N,N)
    c_ptr,                # gate                      (B,N,N,hidden)
    out_ptr,              # destination
    hidden: tl.constexpr, # hidden dimension size
    size:   tl.constexpr, # total number of elements in a/c (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load a (proj) and c (gate) – they are contiguous in the hidden dim.
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    # Compute the mask index (broadcast over hidden).
    mask_idx = offs // hidden                # integer division
    b = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)

    out = a * c * b
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(a: torch.Tensor,
                      mask: torch.Tensor,
                      c: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """
    Elementwise product a * c * mask where ``mask`` has shape (B,N,N) and
    is broadcast over the trailing hidden dimension.
    All tensors must be contiguous and ``dtype`` float16.
    """
    assert a.shape == c.shape, "a and c must have identical shapes"
    B, N, _, H = a.shape
    assert H == hidden, "hidden dimension mismatch"

    a = a.contiguous()
    c = c.contiguous()
    mask = mask.contiguous()  # shape (B,N,N)

    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192  # tuned for H100; adjust if needed
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        a_ptr=a,
        b_ptr=mask,
        c_ptr=c,
        out_ptr=out,
        hidden=hidden,
        size=total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor   shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor : torch.Tensor   shape [B, N, N]   (bool/float, CUDA)
            - weights_dict: model weights (see keys in skeleton)
            - config_dict : contains at least "dim", "hidden_dim";
                            optional "eps" and "nomask".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    dtype_in = inp.dtype

    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Helper to cast weights to FP16 on‑device
    # --------------------------------------------------------------
    to_fp16 = lambda t: t.to(device=device, dtype=torch.float16)

    # LayerNorm weights (input)
    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])

    # Concatenated projection weights (5 × hidden × dim)
    left_proj_weight  = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])
    left_gate_weight  = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight   = to_fp16(weights["out_gate.weight"])

    all_proj_weight = torch.cat([
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
    ], dim=0)                               # shape (5*hidden, dim)

    # Output LayerNorm weights
    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])

    # Final linear projection weight (dim × hidden)
    to_out_weight = to_fp16(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32 → FP16 for the rest)
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    x = x.to(torch.float16)   # keep the heavy part in FP16

    # --------------------------------------------------------------
    # 2) Fused linear projection for all five sub‑modules
    # --------------------------------------------------------------
    # Flatten to (B·N·N, dim) and apply one big GEMM.
    x_flat = x.view(-1, dim)                               # (B*N*N, dim)
    proj_flat = torch.matmul(x_flat, all_proj_weight.t()) # (B*N*N, 5*hidden)
    proj = proj_flat.view(B, N, N, 5 * hidden)             # (B,N,N,5*hidden)

    h = hidden
    left_proj_raw  = proj[..., :h]            # (B,N,N,hidden)
    right_proj_raw = proj[..., h:2*h]
    left_gate_raw  = proj[..., 2*h:3*h]
    right_gate_raw = proj[..., 3*h:4*h]
    out_gate_raw   = proj[..., 4*h:]

    # --------------------------------------------------------------
    # 3) Gate activations (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 4) Mask & gate fusion
    # --------------------------------------------------------------
    if nomask:
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Convert mask to FP16 once; Triton kernel will broadcast it.
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # --------------------------------------------------------------
    # 5) Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM over the hidden dimension.
    # --------------------------------------------------------------
    # Permute to (B, hidden, N, N) where left = (i,k) and right = (k,j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, k, j)

    # Reshape to (B*hidden, N, N) and run a single batched matmul.
    left_mat  = left_perm.view(B * hidden, N, N)   # (B*hidden, N, K)
    right_mat = right_perm.view(B * hidden, N, N)  # (B*hidden, K, N)
    out_hidden = torch.bmm(left_mat, right_mat)    # (B*hidden, N, N)

    # Restore the original layout (B,N,N,hidden)
    out = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate                # apply output gate

    out_final = F.linear(out_norm, to_out_weight)  # bias‑free projection

    # Cast back to the original input dtype (usually float32)
    return out_final.to(dtype_in)
```
