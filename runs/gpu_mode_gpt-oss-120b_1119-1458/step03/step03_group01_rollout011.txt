```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def mul_inplace_kernel(a_ptr, b_ptr, numel: tl.int64, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(a_ptr + offs, a * b, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (AlphaFold3 style) accelerated by
    - a single fused linear projection that computes left, right and the three gates
      in one GEMM,
    - a simple Triton kernel for the final element‑wise multiplication with the
      output gate.
    All heavy matrix multiplications are performed in FP16 on Tensor‑Cores.
    Input: (input_tensor, mask, weights, config)
      - input_tensor: [B, N, N, dim] (float32)
      - mask: [B, N, N] (bool/float)
      - weights: dict of model parameters (float32)
      - config: dict containing "dim", "hidden_dim", and optional "nomask"
    Output: tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Initial LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] fp32

    # ------------------------------------------------------------------
    # 2️⃣ FP16 conversion (for all GEMMs)
    # ------------------------------------------------------------------
    x_h = x.half()  # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all per‑position linear projections into a single GEMM
    #    Output layout: [left, left_gate, right, right_gate, out_gate]
    # ------------------------------------------------------------------
    # Cast weight matrices to fp16 and concatenate
    w_left_proj = weights["left_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()
    # Concatenate along the output dimension (dim=1)
    w_concat = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate], dim=1
    )  # shape (dim, hidden_dim*5)

    # Perform fused linear: (B*N*N, dim) @ (dim, hidden_dim*5) -> (B*N*N, hidden_dim*5)
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)  # (B*N*N, dim)
    proj_flat = F.linear(x_flat, w_concat)  # (B*N*N, hidden_dim*5)
    proj = proj_flat.view(B, N, N, -1)  # (B,N,N, hidden_dim*5)

    # Split the concatenated result
    left_raw, left_gate_raw, right_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )  # each (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # (B,N,N,1) fp16
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and element‑wise application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left_raw * left_gate  # (B,N,N,hidden_dim) fp16
    right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ operation: batched matmul per head
    # ------------------------------------------------------------------
    # Rearrange to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # Compute left @ rightᵀ for each head
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Restore (B,N,N,hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Post‑projection LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )  # (B,N,N,hidden_dim) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by output gate (in‑place) via Triton
    # ------------------------------------------------------------------
    out = out.contiguous()

"""
Optimized “outgoing” TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Input LayerNorm (FP32) → `x`.
2. Cast `x` to FP16 for all heavy arithmetic.
3. Fuse the five per‑position linear maps (left_proj, left_gate,
   right_proj, right_gate, out_gate) into **one** GEMM by concatenating
   the weight matrices along the output dimension.  The result is split
   into the five tensors and the three gates are obtained with a sigmoid.
4. Optional mask (broadcast on the hidden dimension) is applied to the
   left/right projections before gating.
5. Element‑wise gating (`proj * gate`) is performed with ordinary PyTorch
   ops (very cheap compared to the N³ phase).
6. Core N³ computation: for every head (`hidden_dim`) we compute a
   batched matrix multiplication `left @ rightᵀ` using `torch.bmm`.  This
   is the dominant O(N³) work and runs on Tensor‑Cores in FP16.
7. Post‑projection LayerNorm over the hidden dimension (FP16), then an
   **in‑place** Triton kernel multiplies the result by the output gate.
8. Final linear projection back to `dim` (FP16) and cast to FP32.

Only one Triton kernel (`mul_inplace_kernel`) is required, satisfying the
“must use a kernel” rule while keeping the hot paths in highly‑optimized
PyTorch/CUDA primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def mul_inplace_kernel(
    a_ptr,               # fp16* – tensor to be multiplied in‑place
    b_ptr,               # fp16* – multiplication operand (same shape)
    numel: tl.int64,     # total number of elements
    BLOCK: tl.constexpr, # tile size
):
    """Element‑wise in‑place a *= b (both half‑precision)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(a_ptr + offs, a * b, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
        - mask         : torch.Tensor  [B, N, N]       (bool/float)
        - weights      : dict of torch.Tensor (float32) containing all model
                        parameters as in the reference PyTorch module.
        - config       : dict with at least ``dim`` and ``hidden_dim``.
                        Optional ``nomask`` (bool), default ``True``.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for all dense ops
    # ------------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim] – fp16
    B, N, _, _ = x_h.shape

    # ------------------------------------------------------------------
    # 3️⃣ Fuse five linear maps (left/right projections + three gates)
    # ------------------------------------------------------------------
    # Cast weight matrices to fp16
    w_left_proj   = weights["left_proj.weight"].half()   # (hidden, dim)
    w_left_gate   = weights["left_gate.weight"].half()
    w_right_proj  = weights["right_proj.weight"].half()
    w_right_gate  = weights["right_gate.weight"].half()
    w_out_gate    = weights["out_gate.weight"].half()

    # Concatenate along the *output* dimension (rows) → shape (5*hidden, dim)
    w_concat = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )  # (5*hidden_dim, dim)

    # Flatten spatial dimensions to a 2‑D matrix for a single GEMM
    x_flat = x_h.view(-1, dim)                      # (B*N*N, dim)
    proj_flat = F.linear(x_flat, w_concat)          # (B*N*N, 5*hidden_dim)
    proj = proj_flat.view(B, N, N, -1)              # (B, N, N, 5*hidden_dim)

    # Split back into the five components
    left_raw, left_gate_raw, right_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=3
    )  # each: (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # (B, N, N, 1) – fp16
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and apply them
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)   # (B, N, N, hidden_dim) fp16
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)    # needed later

    left  = left_raw * left_gate               # (B, N, N, hidden_dim) fp16
    right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ operation – batched matrix multiplication per head
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout (B, N, N, hidden_dim)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Post‑projection LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ In‑place multiplication with the output gate (Triton)
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel = out.numel()
    BLOCK = 65536  # ~64 k elements per program; good balance of occupancy vs. latency
    grid = ((numel + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid](
        out.view(-1),
        out_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # `out_gate` is no longer needed
    del out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    final = F.linear(out, weights["to_out.weight"].half())  # (B, N, N, dim) fp16

    # Return in the expected float32 precision
    return final.to(torch.float32)
