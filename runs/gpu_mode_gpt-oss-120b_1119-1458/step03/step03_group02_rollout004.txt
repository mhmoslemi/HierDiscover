   A and B have shape (M, N, K) with M = B·hidden_dim, N = seq_len,
   K = seq_len.  The custom Triton kernel tiles the multiplication with
   64×64×64 blocks, uses FP16 inputs and accumulates in FP32 via `tl.dot`
   (Tensor‑Core MMA).  Double‑buffering (`num_stages=2`) and eight warps
   per program hide memory latency.
4️⃣ A second LayerNorm (hidden_dim) is applied, followed by the output‑gate
   and a final linear projection back to the original feature dimension.
   The result is returned as FP32.

The heavy contraction is the only part implemented in Triton; all other
operations rely on high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for many independent (N×N) matrices.
    The program grid is (i_tile, j_tile, m_tile) where:
      - i_tile / j_tile tile the output matrix along its two N‑axes,
      - m_tile indexes the flattened batch‑hidden dimension (M = B·H).
    """
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # index over M (batch * hidden)

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    in_i = offs_i < N
    in_j = offs_j < N

    # Accumulator in FP32 (higher precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # Loop over K dimension
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=(in_i[:, None] & (offs_k[None, :] < K)),
            other=0.0,
        )   # fp16

        # Load B tile (transposed view of B): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=((offs_k[:, None] < K) & in_j[None, :]),
            other=0.0,
        )   # fp16

        # Multiply‑accumulate using Tensor‑Core MMA (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # Store the result tile (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(in_i[:, None] & in_j[None, :]),
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing TriMul contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ----------
    left, right : torch.Tensor
        Tensors of shape [B, N, N, H] (FP16).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] (FP16) containing the contracted result.
    """
    B, N, _, H = left.shape

    # Pack batch and hidden dimensions into one leading dimension.
    #   (B, N, N, H) -> (B*H, N, N)   (contiguous)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element units (Triton expects 32‑bit ints)
    stride_am, stride_ai, stride_ak = A.stride()      # (M,N,K)
    stride_bm, stride_bi, stride_bk = B_.stride()    # (M,N,K)
    stride_cm, stride_ci, stride_cj = C.stride()     # (M,N,N)

    # Tile sizes – 64×64×64 works well on H100 with fp16 MMA.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # M‑tiles (batch * hidden)
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numeric stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to FP16 for the rest of the pipeline
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Convert model weights once to FP16
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # -------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w)     # left projection
    right = torch.nn.functional.linear(x, right_proj_w)   # right projection

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden_dim)
    # -------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] (FP16)
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gating (sigmoid) – still FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via the Triton kernel
    # -------------------------------------------------
    # Result shape: [B, N, N, hidden_dim] (FP16)
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden_dim) in FP16
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Output gate + final linear projection back to original dim
    # -------------------------------------------------
    out = out * out_gate                     # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # -------------------------------------------------
    # 🔟 Cast back to float32 (required output type)
    # -------------------------------------------------
    return out.to(torch.float32)
```
