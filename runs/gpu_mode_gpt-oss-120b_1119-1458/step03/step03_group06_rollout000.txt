"""
TriMul (outgoing) – fast Triton implementation.

The heavy O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is executed as a batched matrix‑multiply with the right operand transposed.
Both inputs are kept in FP16; the dot product uses tensor‑core FP16×FP16→FP32
accumulation for maximal throughput and numerical stability.
All surrounding linear layers, layer‑norms and sigmoids are performed with
PyTorch primitives in FP16 (except the first LayerNorm, which stays FP32).
The result is finally cast back to FP32.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ
#   A, B : [M, N, K]  (FP16)    where M = batch * hidden,
#   C    : [M, N, N]  (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16        [M, N, K]
    B_ptr,               # *fp16        [M, N, K]
    C_ptr,               # *fp16        [M, N, N]
    M, N, K,             # i32 scalars (only N,K are used for tiling)
    stride_am, stride_ai, stride_ak,   # strides of A
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,   # tile size along i (rows)
    BLOCK_N: tl.constexpr,   # tile size along j (cols)
    BLOCK_K: tl.constexpr,   # reduction tile size
):
    pid_i = tl.program_id(0)        # i‑tile index
    pid_j = tl.program_id(1)        # j‑tile index
    pid_z = tl.program_id(2)        # batch * hidden index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (Tensor‑core dot writes to FP32 automatically)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over K
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load a tile of A (shape: BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # FP16

        # Load a tile of B transposed (shape: BLOCK_K × BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # FP16

        # Multiply‑accumulate on tensor‑cores (FP16×FP16 → FP32)
        acc += tl.dot(a, b)        # no explicit cast – Triton handles it

    # ------------------------------------------------------------------
    # Write result back
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a custom Triton kernel.  The tensors must be contiguous,
    FP16 and located on the same CUDA device.
    """
    B, N, _, H = left.shape                         # left.shape == right.shape
    # reshape to (M, N, K) where M = B*H, K = N
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # element‑wise strides (Triton works with element strides)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # -----------------------------------------------------------------------------
    # Tile size selection – 256×256 works well for N≥512, otherwise fall back
    # to a smaller tile to avoid excessive out‑of‑bounds work.
    # -----------------------------------------------------------------------------
    if N >= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 256, 256, 32
        num_warps = 16
    elif N >= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
        num_warps = 8
    else:   # very small sequences – keep low register pressure
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
        num_warps = 4

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth dimension (batch * hidden)
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,               # dummy M (not used inside kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,        # double‑buffering
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask  : torch.Tensor, shape [B, N, N] (bool or float, optional)
        - weights: dict name → torch.Tensor (float32)
        - config : dict with keys ``dim`` and ``hidden_dim``.
                    Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] and dtype float32.
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 -> remain float32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Switch to FP16 for the heavy path
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast all linear weights once (FP16)
    # -------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16) → [B,N,N,H]
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_w)   # (B,N,N,hidden)

    # -------------------------------------------------
    # 5️⃣ Optional masking (broadcast on hidden)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gating (sigmoid) – stays FP16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (still FP16)
    # -------------------------------------------------
    # out shape: [B, N, N, hidden]
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Apply output gate and final projection
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to dim

    # -------------------------------------------------
    # 10️⃣ Cast back to float32 for the caller
    # -------------------------------------------------
    return out.to(torch.float32)
