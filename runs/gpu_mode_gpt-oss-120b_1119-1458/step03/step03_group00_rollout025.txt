Pseudo-code:

```
@triton.jit
def fused_input_kernel(
    x_ptr, mask_ptr,
    # LN params
    ln_weight_ptr, ln_bias_ptr,
    # proj/gate weight pointers
    w_lp_ptr, w_lg_ptr,
    w_rp_ptr, w_rg_ptr,
    w_og_ptr,
    # outputs
    left_ptr, right_ptr, out_gate_ptr,
    # strides
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_m_b, stride_m_i, stride_m_j,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_gate_b, stride_out_gate_i, stride_out_gate_j, stride_out_gate_h,
    # weight strides (for weight matrices)
    stride_w_h, stride_w_d,
    # LN weight stride (should be 1)
    stride_ln_w,
    stride_ln_b,
    B, N, dim, hidden, eps,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    # Determine position (b,i,j)
    pos = tl.program_id(0)
    h_start = tl.program_id(1) * BLOCK_H
    # compute hidden idx range
    h_range = tl.arange(0, BLOCK_H) + h_start
    valid_h = h_range < hidden
    # compute b,i,j
    b = pos // (N * N)
    residual = pos % (N * N)
    i = residual // N
    j = residual % N

    # Compute mask value
    if HAS_MASK:
        mask_offset = b * stride_m_b + i * stride_m_i + j * stride_m_j
        mask_val = tl.load(mask_ptr + mask_offset)
    else:
        mask_val = tl.full((1,), 1.0, dtype=tl.float16)[0]

    # compute mean & var of x for LN
    sum_x = tl.zeros((1,), tl.float32)
    sum_x2 = tl.zeros((1,), tl.float32)

    # loop over dim
    for d in range(0, dim, BLOCK_D):
        cur_block = tl.minimum(BLOCK_D, dim - d)
        d_range = tl.arange(0, cur_block)
        offs = b * stride_x_b + i * stride_x_i + j * stride_x_j + (d + d_range) * stride_x_d
        mask_d = d_range < cur_block
        x = tl.load(x_ptr + offs, mask=mask_d, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_x += tl.sum(x_f32)
        sum_x2 += tl.sum(x_f32 * x_f32)

    mean = sum_x / dim
    var = sum_x2 / dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Initialize accumulators for projections
    acc_left_proj = tl.zeros((BLOCK_H,), tl.float32)
    acc_left_gate = tl.zeros((BLOCK_H,), tl.float32)
    acc_right_proj = tl.zeros((BLOCK_H,), tl.float32)
    acc_right_gate = tl.zeros((BLOCK_H,), tl.float32)
    acc_out_gate = tl.zeros((BLOCK_H,), tl.float32)

    # second loop over dim for dot products
    for d in range(0, dim, BLOCK_D):
        cur_block = tl.minimum(BLOCK_D, dim - d)
        d_range = tl.arange(0, cur_block)
        offs = b * stride_x_b + i * stride_x_i + j * stride_x_j + (d + d_range) * stride_x_d
        mask_d = d_range < cur_block
        x = tl.load(x_ptr + offs, mask=mask_d, other=0.0)
        # normalize
        x_norm = ((x.to(tl.float32) - mean) * inv_std).to(tl.float16)
        # weight slices for each linear
        # note weight layout: (hidden, dim); stride across hidden = dim, across dim = 1
        w_off = (h_range[:, None] * stride_w_h) + (d + d_range)[None, :] * stride_w_d
        # left proj weight
        w_lp = tl.load(w_lp_ptr + w_off, mask=valid_h[:, None] & mask_d[None, :], other=0.0)
        w_lg = tl.load(w_lg_ptr + w_off, mask=valid_h[:, None] & mask_d[None, :], other=0.0)
        w_rp = tl.load(w_rp_ptr + w_off, mask=valid_h[:, None] & mask_d[None, :], other=0.0)
        w_rg = tl.load(w_rg_ptr + w_off, mask=valid_h[:, None] & mask_d[None, :], other=0.0)
        w_og = tl.load(w_og_ptr + w_off, mask=valid_h[:, None] & mask_d[None, :], other=0.0)

        # compute dot products using tl.dot
        # x_norm shape (cur_block) -> cast to float16
        # but tl.dot expects at least 2D; we use x_norm[None,:] shape (1,cur_block)
        # ensure dtype consistency: use same dtype for weight and x (float16)
        acc_left_proj += tl.dot(x_norm[None, :], w_lp)[0]
        acc_left_gate  += tl.dot(x_norm[None, :], w_lg)[0]
        acc_right_proj += tl.dot(x_norm[None, :], w_rp)[0]
        acc_right_gate += tl.dot(x_norm[None, :], w_rg)[0]
        acc_out_gate   += tl.dot(x_norm[None, :], w_og)[0]

    # compute sigmoid for gates
    left_gate_sig = 1.0 / (1.0 + tl.exp(-acc_left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-acc_right_gate))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-acc_out_gate))

    # apply mask to left/right projections
    left_proj_masked = acc_left_proj * mask_val
    right_proj_masked = acc_right_proj * mask_val

    # final left and right
    left_out = left_proj_masked * left_gate_sig
    right_out = right_proj_masked * right_gate_sig

    # store outputs
    offs_left = b * stride_left_b + i * stride_left_i + j * stride_left_j + h_range * stride_left_h
    offs_right = b * stride_right_b + i * stride_right_i + j * stride_right_j + h_range * stride_right_h
    offs_gate = b * stride_out_gate_b + i * stride_out_gate_i + j * stride_out_gate_j + h_range * stride_out_gate_h

    tl.store(left_ptr + offs_left, left_out.to(tl.float16), mask=valid_h)
    tl.store(right_ptr + offs_right, right_out.to(tl.float16), mask=valid_h)
    tl.store(out_gate_ptr + offs_gate, out_gate_sig.to(tl.float16), mask=valid_h)
```

```
@triton.jit
def fused_output_kernel(
    hidden_ptr, gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    weight_T_ptr,
    out_ptr,
    # strides for hidden_ptr (B,N,N,hidden)
    stride_h_b, stride_h_i, stride_h_j, stride_h_h,
    # strides for gate_ptr similarly
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    # stride for norm weight: stride_norm = 1
    # stride for weight_T: stride_T_h = dim, stride_T_d = 1
    stride_T_h, stride_T_d,
    # strides for out_ptr (B,N,N,dim)
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    B, N, hidden, dim, eps,
    BLOCK_H: tl.constexpr,
):
    # compute position indices
    pos = tl.program_id(0)
    b = pos // (N * N)
    residual = pos % (N * N)
    i = residual // N
    j = residual % N

    # h_range
    h_range = tl.arange(0, BLOCK_H)
    mask_h = h_range < hidden

    # offsets for hidden and gate
    offs_h = b * stride_h_b + i * stride_h_i + j * stride_h_j + h_range * stride_h_h
    h_vals = tl.load(hidden_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
    g_vals = tl.load(gate_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

    # compute mean and var over hidden
    mean = tl.sum(h_vals) / hidden
    var = tl.sum(h_vals * h_vals) / hidden - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # load layernorm weight and bias (hidden)
    ln_w = tl.load(norm_weight_ptr + h_range, mask=mask_h, other=0.0).to(tl.float32)
    ln_b = tl.load(norm_bias_ptr + h_range, mask=mask_h, other=0.0).to(tl.float32)

    # normalized hidden
    h_norm = ((h_vals - mean) * inv_std) * ln_w + ln_b

    # apply gate
    v = h_norm * g_vals

    # load weight_T slice (hidden, dim)
    d_range = tl.arange(0, dim)
    # offsets = h_range[:, None] * stride_T_h + d_range[None, :] * stride_T_d
    w_T = tl.load(weight_T_ptr + (h_range[:, None] * stride_T_h + d_range[None, :] * stride_T_d), mask=mask_h[:, None], other=0.0)

    # compute final output: dot(v[None,:], w_T) -> shape (1, dim)
    out_vec = tl.dot(v[None, :], w_T)[0]  # shape (dim)

    # store output
    offs_o = b * stride_o_b + i * stride_o_i + j * stride_o_j + d_range * stride_o_d
    tl.store(out_ptr + offs_o, out_vec.to(tl.float16))
```

def fused_output_kernel(
    hidden_ptr, gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    w_T_ptr,
    out_ptr,
    # strides for hidden/gate/out...
    stride_h_b, stride_h_i, stride_h_j, stride_h_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    stride_w_T_h, stride_w_T_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    B, N, hidden, dim, eps,
    BLOCK_H: tl.constexpr,
    DIM: tl.constexpr,
):
    pos = tl.program_id(0)
    b = pos // (N * N)
    tmp = pos % (N * N)
    i = tmp // N
    j = tmp % N

    h_range = tl.arange(0, BLOCK_H)
    mask_h = h_range < hidden

    # offsets for hidden and gate
    off_h = b * stride_h_b + i * stride_h_i + j * stride_h_j + h_range * stride_h_h
    off_g = b * stride_g_b + i * stride_g_i + j * stride_g_j + h_range * stride_g_h

    h_vals = tl.load(hidden_ptr + off_h, mask=mask_h, other=0.0).to(tl.float32)
    g_vals = tl.load(gate_ptr + off_g, mask=mask_h, other=0.0).to(tl.float32)

    # compute mean and variance
    mean = tl.sum(h_vals) / hidden
    var = tl.sum(h_vals * h_vals) / hidden - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # LN weight and bias
    ln_w = tl.load(norm_weight_ptr + h_range, mask=mask_h, other=0.0).to(tl.float32)
    ln_b = tl.load(norm_bias_ptr + h_range, mask=mask_h, other=0.0).to(tl.float32)

    # normalized hidden then gating
    h_norm = ((h_vals - mean) * inv_std) * ln_w + ln_b
    v = h_norm * g_vals

    # load weight_T slice (hidden x dim)
    d_range = tl.arange(0, DIM)
    # weight_T shape (hidden, dim) => offsets:
    w_off = h_range[:, None] * stride_w_T_h + d_range[None, :] * stride_w_T_d
    w_T = tl.load(w_T_ptr + w_off, mask=mask_h[:, None], other=0.0)

    # compute final linear: v (1, hidden) dot w_T (hidden, dim) -> (1, dim)
    out_vec = tl.dot(v[None, :], w_T)[0]  # shape (DIM,)

    # store output
    out_off = b * stride_out_b + i * stride_out_i + j * stride_out_j + d_range * stride_out_d
    tl.store(out_ptr + out_off, out_vec.to(tl.float16))

    norm_bias = weights["norm.bias"].to(torch.float16).contiguous()

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm Overview
------------------
1️⃣ Input LayerNorm + all five linear projections (left_proj,
   left_gate, right_proj, right_gate, out_gate) are fused into a
   single Triton kernel.  This kernel reads each input vector
   (shape=[dim]) exactly once, computes the LayerNorm statistics,
   applies the LayerNorm affine transformation, performs the five
   matrix‑vector products and the required mask & sigmoid fusions.
   The result are three tensors:
        left_out  = left_proj * mask * sigmoid(left_gate)
        right_out = right_proj * mask * sigmoid(right_gate)
        out_gate  = sigmoid(out_gate_raw)

2️⃣ Core contraction:
        out[b,i,j,d] = Σₖ left_out[b,i,k,d] * right_out[b,j,k,d]
   This is a batched matrix‑multiply in FP16 using the highly
   optimized cuBLAS GEMM (torch.matmul).  The tensors are first
   permuted to (B, hidden, N, N) and the multiplication is performed
   once.

3️⃣ The output LayerNorm, output‑gate multiplication and final linear
   projection are also fused into a single Triton kernel.
   For each pair (b,i,j) the kernel computes the mean/variance over
   the hidden dimension, applies the LayerNorm affine transform,
   multiplies by the previously computed out_gate, and finally
   projects back to `dim` using the transposed output weight.

All heavy arithmetic stays in FP16, while reductions (mean/var,
sigmoid) use FP32 for numerical stability.  The final tensor is cast
back to the input dtype (usually float32).

The implementation respects the “no‑mask” case (nomask=True) and
handles arbitrary batch sizes (B∈{1,2}) and sequence lengths up to 1024.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Fused input LayerNorm + 5 linear layers (mask & gate fused)
# ----------------------------------------------------------------------
@triton.jit
def fused_input_kernel(
    # ------------------ data ------------------
    x_ptr, mask_ptr,                      # input tensor & optional mask
    ln_w_ptr, ln_b_ptr,                   # LayerNorm weight & bias (shape [dim])
    w_lp_ptr, w_lg_ptr,                   # left_proj   weight, left_gate   weight
    w_rp_ptr, w_rg_ptr,                   # right_proj  weight, right_gate  weight
    w_og_ptr,                             # out_gate weight
    # ------------------ outputs ------------------
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    # ------------------ strides ------------------
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    # ------------------ weight strides ------------------
    stride_w_h, stride_w_d,               # weight layout: (hidden, dim)
    stride_ln_w, stride_ln_b,             # LayerNorm weight/bias stride (usually 1)
    # ------------------ scalars ------------------
    B, N, DIM, hidden, eps,
    # ------------------ compile‑time consts ------------------
    BLOCK_H: tl.constexpr,                # tile size in hidden dimension
    BLOCK_D: tl.constexpr,                # tile size in dim dimension
    HAS_MASK: tl.constexpr,               # 1 -> mask present, 0 -> no mask
):
    """
    Computes:
        left_out  = (x·W_left_proj + ln_b)·mask·sigmoid(x·W_left_gate + ln_b)
        right_out = (x·W_right_proj + ln_b)·mask·sigmoid(x·W_right_gate + ln_b)
        out_gate  = sigmoid(x·W_out_gate + ln_b)
    where the input `x` is first LayerNormed.
    All tensors are in fp16, reductions are in fp32.
    """

    # ----- program ids & hidden‑tile -----
    pos = tl.program_id(0)                      # spatial position id = b*N*N + i*N + j
    tile_h = tl.program_id(1) * BLOCK_H         # starting hidden index for this tile
    h_range = tl.arange(0, BLOCK_H) + tile_h     # hidden indices handled by this program
    mask_h = h_range < hidden                    # out‑of‑bounds guard for hidden

    # ----- decode (b,i,j) from flat position -----
    b = pos // (N * N)
    residual = pos % (N * N)
    i = residual // N
    j = residual % N

    # ----- mask value (broadcast over hidden) -----
    if HAS_MASK:
        mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
        mask_val = tl.load(mask_ptr + mask_offset)          # fp16 scalar
    else:
        mask_val = tl.full((1,), 1.0, dtype=tl.float16)[0]

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean / variance of the input vector (LayerNorm)
    # ------------------------------------------------------------------
    sum_x = tl.zeros((1,), dtype=tl.float32)
    sum_x2 = tl.zeros((1,), dtype=tl.float32)

    # first pass: accumulate sums over the dim axis
    for d in range(0, DIM, BLOCK_D):
        cur = tl.minimum(BLOCK_D, DIM - d)                     # actual block size
        d_range = tl.arange(0, cur)
        # load raw input (fp16)
        x_offset = b * stride_x_b + i * stride_x_i + j * stride_x_j + (d + d_range) * stride_x_d
        mask_d = (d + d_range) < DIM
        x_vec = tl.load(x_ptr + x_offset, mask=mask_d, other=0.0).to(tl.float32)
        sum_x  += tl.sum(x_vec)
        sum_x2 += tl.sum(x_vec * x_vec)

    mean = sum_x / DIM
    var  = sum_x2 / DIM - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Second pass: LayerNorm + five matrix‑vector products
    # ------------------------------------------------------------------
    # accumulators for the five linear results (float32)
    acc_lp = tl.zeros((BLOCK_H,), dtype=tl.float32)   # left_proj
    acc_lg = tl.zeros((BLOCK_H,), dtype=tl.float32)   # left_gate
    acc_rp = tl.zeros((BLOCK_H,), dtype=tl.float32)   # right_proj
    acc_rg = tl.zeros((BLOCK_H,), dtype=tl.float32)   # right_gate
    acc_og = tl.zeros((BLOCK_H,), dtype=tl.float32)   # out_gate

    for d in range(0, DIM, BLOCK_D):
        cur = tl.minimum(BLOCK_D, DIM - d)
        d_range = tl.arange(0, cur)

        # ----- raw input vector -----
        x_offset = b * stride_x_b + i * stride_x_i + j * stride_x_j + (d + d_range) * stride_x_d
        mask_d = (d + d_range) < DIM
        x_vec = tl.load(x_ptr + x_offset, mask=mask_d, other=0.0)        # fp16
        x_f32 = x_vec.to(tl.float32)

        # ----- LayerNorm affine transformation -----
        ln_w = tl.load(ln_w_ptr + (d + d_range) * stride_ln_w,
                       mask=mask_d, other=0.0).to(tl.float32)
        ln_b = tl.load(ln_b_ptr + (d + d_range) * stride_ln_b,
                       mask=mask_d, other=0.0).to(tl.float32)
        x_norm_f32 = ((x_f32 - mean) * inv_std) * ln_w + ln_b
        x_norm = x_norm_f32.to(tl.float16)          # back to fp16 for dot products

        # ----- weight slices (shape: [BLOCK_H, cur]) -----
        w_off = h_range[:, None] * stride_w_h + (d + d_range)[None, :] * stride_w_d
        w_lp = tl.load(w_lp_ptr + w_off,
                       mask=mask_h[:, None] & (d_range[None, :] < cur),
                       other=0.0)
        w_lg = tl.load(w_lg_ptr + w_off,
                       mask=mask_h[:, None] & (d_range[None, :] < cur),
                       other=0.0)
        w_rp = tl.load(w_rp_ptr + w_off,
                       mask=mask_h[:, None] & (d_range[None, :] < cur),
                       other=0.0)
        w_rg = tl.load(w_rg_ptr + w_off,
                       mask=mask_h[:, None] & (d_range[None, :] < cur),
                       other=0.0)
        w_og = tl.load(w_og_ptr + w_off,
                       mask=mask_h[:, None] & (d_range[None, :] < cur),
                       other=0.0)

        # ----- dot products (fp16·fp16 → fp32 accumulator) -----
        acc_lp += tl.dot(x_norm[None, :], w_lp)[0]
        acc_lg += tl.dot(x_norm[None, :], w_lg)[0]
        acc_rp += tl.dot(x_norm[None, :], w_rp)[0]
        acc_rg += tl.dot(x_norm[None, :], w_rg)[0]
        acc_og += tl.dot(x_norm[None, :], w_og)[0]

    # ----- sigmoid for the three gates -----
    left_gate_sig  = 1.0 / (1.0 + tl.exp(-acc_lg))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-acc_rg))
    out_gate_sig   = 1.0 / (1.0 + tl.exp(-acc_og))

    # ----- apply mask (if present) and combine projection + gate -----
    mask_val_f32 = mask_val.to(tl.float32)
    left_proj_masked  = acc_lp * mask_val_f32
    right_proj_masked = acc_rp * mask_val_f32

    left_res  = (left_proj_masked * left_gate_sig).to(tl.float16)
    right_res = (right_proj_masked * right_gate_sig).to(tl.float16)
    out_gate_res = out_gate_sig.to(tl.float16)

    # ----- store results -----
    off_left = (b * stride_left_b + i * stride_left_i + j * stride_left_j
                + h_range * stride_left_h)
    off_right = (b * stride_right_b + i * stride_right_i + j * stride_right_j
                 + h_range * stride_right_h)
    off_gate = (b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
                + h_range * stride_gate_h)

    tl.store(left_out_ptr + off_left, left_res, mask=mask_h)
    tl.store(right_out_ptr + off_right, right_res, mask=mask_h)
    tl.store(out_gate_out_ptr + off_gate, out_gate_res, mask=mask_h)


# ----------------------------------------------------------------------
# 3️⃣ Fused output LayerNorm + out‑gate + final linear projection
# ----------------------------------------------------------------------
@triton.jit
def fused_output_kernel(
    # ------------------ inputs ------------------
    hidden_ptr, gate_ptr,                     # (B,N,N,hidden) tensors
    out_norm_w_ptr, out_norm_b_ptr,           # LayerNorm weight/bias (shape [hidden])
    w_out_T_ptr,                              # transposed output weight (hidden, dim)
    # ------------------ output ------------------
    out_ptr,
    # ------------------ strides ------------------
    stride_h_b, stride_h_i, stride_h_j, stride_h_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    stride_wout_h, stride_wout_d,             # weight_T layout: (hidden, dim)
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    # ------------------ scalars ------------------
    B, N, hidden, eps,
    # ------------------ compile‑time consts ------------------
    BLOCK_H: tl.constexpr,    # we set BLOCK_H = hidden (full hidden vector)
    OUT_DIM: tl.constexpr,    # output channel dimension (dim)
):
    """
    Computes final output:
        hidden = out_hidden        # (B,N,N,hidden)
        gate   = out_gate          # (B,N,N,hidden)
        hidden_norm = LayerNorm(hidden)
        gated       = hidden_norm * gate
        out = gated @ W_out.T      # (B,N,N,dim)
    All arithmetic stays in fp16; reductions use fp32.
    """

    # ----- program id maps to a single (b,i,j) location -----
    pos = tl.program_id(0)
    b = pos // (N * N)
    residual = pos % (N * N)
    i = residual // N
    j = residual % N

    # ----- hidden indices -----
    h_range = tl.arange(0, BLOCK_H)
    mask_h = h_range < hidden

    # ----- load hidden & gate vectors -----
    off_h = (b * stride_h_b + i * stride_h_i + j * stride_h_j
             + h_range * stride_h_h)
    hidden_vec = tl.load(hidden_ptr + off_h, mask=mask_h, other=0.0).to(tl.float32)
    off_g = (b * stride_g_b + i * stride_g_i + j * stride_g_j
             + h_range * stride_g_h)
    gate_vec = tl.load(gate_ptr + off_g, mask=mask_h, other=0.0).to(tl.float32)

    # ----- mean / variance over hidden -----
    mean = tl.sum(hidden_vec) / hidden
    var  = tl.sum(hidden_vec * hidden_vec) / hidden - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ----- LayerNorm affine (weight & bias over hidden) -----
    ln_w = tl.load(out_norm_w_ptr + h_range, mask=mask_h, other=0.0).to(tl.float32)
    ln_b = tl.load(out_norm_b_ptr + h_range, mask=mask_h, other=0.0).to(tl.float32)

    hidden_norm = ((hidden_vec - mean) * inv_std) * ln_w + ln_b
    gated_vec = hidden_norm * gate_vec               # still fp32

    # ----- convert to fp16 for the final GEMM -----
    gated_fp16 = gated_vec.to(tl.float16)

    # ----- load output weight slice (shape: [hidden, OUT_DIM]) -----
    d_range = tl.arange(0, OUT_DIM)
    w_off = h_range[:, None] * stride_wout_h + d_range[None, :] * stride_wout_d
    w_T = tl.load(w_out_T_ptr + w_off, mask=mask_h[:, None], other=0.0)   # fp16

    # ----- final linear projection (fp16·fp16 → fp32 accumulator) -----
    out_vec = tl.dot(gated_fp16[None, :], w_T)[0]   # shape (OUT_DIM,)

    # ----- store -------------------------------------------------------
    off_o = (b * stride_o_b + i * stride_o_i + j * stride_o_j
             + d_range * stride_o_d)
    tl.store(out_ptr + off_o, out_vec.to(tl.float16))


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32 CUDA)
            - mask_tensor : shape [B, N, N]  (bool/float CUDA) – optional
            - weights_dict : model weights
            - config_dict : contains "dim", "hidden_dim", optional "eps", "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32 CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    device = inp.device
    # ------------------------------------------------------------------
    # Cast everything to fp16 (the kernels work in half precision)
    # ------------------------------------------------------------------
    x_fp16 = inp.to(torch.float16).contiguous()
    if nomask:
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)   # dummy, never read
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    # ----------------- weight tensors (fp16, contiguous) -----------------
    norm_w = weights["norm.weight"].to(torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(torch.float16).contiguous()

    left_proj_w   = weights["left_proj.weight"].to(torch.float16).contiguous()   # (hidden, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16).contiguous()
    right_proj_w  = weights["right_proj.weight"].to(torch.float16).contiguous()
    right_gate_w  = weights["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w    = weights["out_gate.weight"].to(torch.float16).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).contiguous()
    # transpose to (hidden, dim) for the final GEMM
    to_out_w_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # (hidden, dim)

    # ----------------- allocate intermediate tensors -----------------
    left   = torch.empty((B, N, N, hidden), dtype=torch.float16, device=device)
    right  = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # ------------------------------------------------------------------
    # Gather strides (in element units, not bytes) for all tensors
    # ------------------------------------------------------------------
    # input
    sx_b, sx_i, sx_j, sx_d = x_fp16.stride()
    # mask (if present)
    sm_b, sm_i, sm_j = (mask_fp16.stride() if not nomask else (0, 0, 0))
    # left / right / out_gate outputs
    sl_b, sl_i, sl_j, sl_h = left.stride()
    sr_b, sr_i, sr_j, sr_h = right.stride()
    sg_b, sg_i, sg_j, sg_h = out_gate.stride()
    # weight matrices (all have layout (hidden, dim))
    stride_w_h = left_proj_w.stride(0)      # dim
    stride_w_d = left_proj_w.stride(1)      # 1
    # LayerNorm weight / bias (dim,)
    stride_ln_w = norm_w.stride(0)
    stride_ln_b = norm_b.stride(0)

    # ------------------------------------------------------------------
    # 1️⃣ Run fused input kernel (LayerNorm + all five linear layers)
    # ------------------------------------------------------------------
    BLOCK_H = 32          # tile along hidden dimension
    BLOCK_D = 32          # tile along dim dimension
    hidden_tiles = (hidden + BLOCK_H - 1) // BLOCK_H
    grid = (B * N * N, hidden_tiles)          # (positions, hidden‑tiles)

    fused_input_kernel[grid](
        # data
        x_fp16, mask_fp16,
        norm_w, norm_b,
        left_proj_w, left_gate_w,
        right_proj_w, right_gate_w,
        out_gate_w,
        # outputs
        left, right, out_gate,
        # strides
        sx_b, sx_i, sx_j, sx_d,
        sm_b, sm_i, sm_j,
        sl_b, sl_i, sl_j, sl_h,
        sr_b, sr_i, sr_j, sr_h,
        sg_b, sg_i, sg_j, sg_h,
        # weight strides
        stride_w_h, stride_w_d,
        stride_ln_w, stride_ln_b,
        # scalars
        B, N, dim, hidden, eps,
        # compile‑time constants
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        HAS_MASK=int(not nomask),
        DIM=dim,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Core contraction: batched matrix multiplication
    # ------------------------------------------------------------------
    # Shape transformation: (B,N,N,hidden) -> (B, hidden, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # Batched GEMM over the hidden dimension
    contracted = torch.matmul(left_perm, right_perm)               # (B, hidden, N, N)
    # Back to original layout (B,N,N,hidden)
    contracted = contracted.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 3️⃣ Fused output LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, dim), dtype=torch.float16, device=device)

    # Strides for the tensors used in the final kernel
    sh_b, sh_i, sh_j, sh_h = contracted.stride()
    sg_b, sg_i, sg_j, sg_h = out_gate.stride()
    so_b, so_i, so_j, so_d = out.stride()
    # weight_T strides (hidden, dim) layout
    swout_h = to_out_w_T.stride(0)      # dim
    swout_d = to_out_w_T.stride(1)      # 1

    # Use a single program id per spatial position
    grid_out = (B * N * N,)

    fused_output_kernel[grid_out](
        contracted, out_gate,
        to_out_norm_w, to_out_norm_b,
        to_out_w_T,
        out,
        # strides
        sh_b, sh_i, sh_j, sh_h,
        sg_b, sg_i, sg_j, sg_h,
        swout_h, swout_d,
        so_b, so_i, so_j, so_d,
        # scalars
        B, N, hidden, eps,
        # compile‑time consts
        BLOCK_H=hidden,            # we process the whole hidden vector at once
        OUT_DIM=dim,
    )

    # ------------------------------------------------------------------
    # Cast back to the original dtype (normally float32) and return
    # ------------------------------------------------------------------
    return out.to(inp.dtype)
