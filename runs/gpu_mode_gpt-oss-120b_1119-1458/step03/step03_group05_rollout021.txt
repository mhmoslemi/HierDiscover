@triton.jit
def _trimul_gated_matmul_kernel(
    A_ptr, B_ptr, G1_ptr, G2_ptr, mask_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_g1m, stride_g1i, stride_g1k,
    stride_g2m, stride_g2k, stride_g2j,
    stride_mask_batch, stride_mask_i, stride_mask_j,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_z = tl.program_id(2) # batch*hidden index
    pid_i = tl.program_id(0) # i tile
    pid_j = tl.program_id(1) # j tile

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute batch index from pid_z
    if HAS_MASK:
        batch_id = pid_z // hidden_dim
        mask_batch_offset = batch_id * stride_mask_batch
    else:
        mask_batch_offset = 0

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < K), other=0.0)

        # Load G1 tile (left gate)
        g1_ptrs = (
            G1_ptr
            + pid_z * stride_g1m
            + offs_i[:, None] * stride_g1i
            + offs_k[None, :] * stride_g1k
        )
        g1 = tl.load(g1_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < K), other=0.0)

        # Scale A
        a = a * g1
        if HAS_MASK:
            mask_a_ptrs = (
                mask_ptr
                + mask_batch_offset
                + offs_i[:, None] * stride_mask_i
                + offs_k[None, :] * stride_mask_j
            )
            mask_a = tl.load(mask_a_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < K), other=0.0)
            a = a * mask_a

        # Load B tile (right, transposed)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_j[None, :], other=0.0)

        # Load G2 tile (right gate)
        g2_ptrs = (
            G2_ptr
            + pid_z * stride_g2m
            + offs_k[:, None] * stride_g2k
            + offs_j[None, :] * stride_g2j
        )
        g2 = tl.load(g2_ptrs, mask=(offs_k[:, None] < K) & mask_j[None, :], other=0.0)

        # Scale B
        b = b * g2
        if HAS_MASK:
            mask_b_ptrs = (
                mask_ptr
                + mask_batch_offset
                + offs_j[:, None] * stride_mask_i
                + offs_k[None, :] * stride_mask_j
            )
            mask_b = tl.load(mask_b_ptrs, mask=mask_j[:, None] & (offs_k[None, :] < K), other=0.0)
            b = b * mask_b

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write output
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

- For right_gate: stride_g2m = stride for M dimension (same as left_gate). We need stride for the K and J dims after transposition, which are same as for right (since shape after permute and view is same). So G2_ptr uses strides (stride_g2m, stride_g2k, stride_g2j).

"""
Optimised outgoing TriMul (AlphaFold‑3) forward pass.

The heavy O(N³) contraction:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched GEMM C = A @ Bᵀ where
    A = left_proj   (shape [M, N, K])   with M = B·H,
    B = right_projᵀ (shape [M, K, N]).
A single Triton kernel fuses the element‑wise mask and gate scalings
into the GEMM, thereby removing the extra write‑back of the gated tensors
and reducing global‑memory traffic.  The kernel works with FP16
operands and accumulates in FP32, exploiting Tensor‑Core support on
H100 (64×64×64 tiling, 3‑stage double‑buffering).

All other layers (LayerNorm, Linear, Sigmoid) are kept in PyTorch.
The final output is cast back to float32 as required by the harness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with fused left/right gating and optional mask.
#   A : [M, N, K]   (left_proj   – already in (i,k) order)
#   B : [M, K, N]   (right_projᵀ – (k,j) order)
#   G1: [M, N, K]  (left_gate)
#   G2: [M, K, N]  (right_gate)
#   mask : [B, N, N]  (broadcast over H)
#   C : [M, N, N]   (output)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gated_matmul_kernel(
    A_ptr, B_ptr,                # left_proj, right_projᵀ
    G1_ptr, G2_ptr,              # left_gate, right_gate
    mask_ptr, C_ptr,             # mask (optional) and output
    M, N, K,                     # dimensions (M = B*H)
    stride_am, stride_ai, stride_ak,               # A strides
    stride_bm, stride_bk, stride_bj,               # B strides (K,N order)
    stride_g1m, stride_g1i, stride_g1k,            # G1 strides
    stride_g2m, stride_g2k, stride_g2j,            # G2 strides (K,N order)
    stride_mask_batch, stride_mask_i, stride_mask_j,  # mask strides
    stride_cm, stride_ci, stride_cj,               # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.int32,         # H
    HAS_MASK: tl.constexpr,       # compile‑time flag
):
    # ------------------------------------------------------------------
    # 1. Tile indices
    # ------------------------------------------------------------------
    pid_z = tl.program_id(2)          # batch*hidden index   (M dimension)
    pid_i = tl.program_id(0)          # i‑tile (rows)
    pid_j = tl.program_id(1)          # j‑tile (cols)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 2. Accumulator (FP32 for numerical safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3. Batch (and hence mask) offset
    # ------------------------------------------------------------------
    if HAS_MASK:
        batch_id = pid_z // hidden_dim          # integer division
        mask_batch_off = batch_id * stride_mask_batch
    else:
        mask_batch_off = 0

    # ------------------------------------------------------------------
    # 4. Loop over reduction dimension K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- Load left projection -------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- Load left gate -------------------------------------------------------
        g1_ptrs = (
            G1_ptr
            + pid_z * stride_g1m
            + offs_i[:, None] * stride_g1i
            + offs_k[None, :] * stride_g1k
        )
        g1 = tl.load(
            g1_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        a = a * g1
        if HAS_MASK:
            mask_a_ptrs = (
                mask_ptr
                + mask_batch_off
                + offs_i[:, None] * stride_mask_i
                + offs_k[None, :] * stride_mask_j
            )
            mask_a = tl.load(
                mask_a_ptrs,
                mask=mask_i[:, None] & (offs_k[None, :] < K),
                other=0.0,
            )
            a = a * mask_a

        # ---- Load right projection (already transposed) ---------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # K‑dimension (k)
            + offs_j[None, :] * stride_bj   # N‑dimension (j)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ---- Load right gate ------------------------------------------------------
        g2_ptrs = (
            G2_ptr
            + pid_z * stride_g2m
            + offs_k[:, None] * stride_g2k
            + offs_j[None, :] * stride_g2j
        )
        g2 = tl.load(
            g2_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        b = b * g2
        if HAS_MASK:
            mask_b_ptrs = (
                mask_ptr
                + mask_batch_off
                + offs_j[:, None] * stride_mask_i   # j plays role of i for mask
                + offs_k[None, :] * stride_mask_j   # k
            )
            mask_b = tl.load(
                mask_b_ptrs,
                mask=mask_j[:, None] & (offs_k[None, :] < K),
                other=0.0,
            )
            b = b * mask_b

        # ---- Accumulate -----------------------------------------------------------
        acc += tl.dot(a, b)   # FP16×FP16 → FP32

    # ------------------------------------------------------------------
    # 5. Write result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors, launches the Triton kernel and restores the
# original layout [B, N, N, hidden_dim].
# ----------------------------------------------------------------------
def _batched_gated_matmul_fp16(
    left_proj: torch.Tensor,          # [B, N, N, H]   raw projection
    left_gate: torch.Tensor,          # [B, N, N, H]   sigmoid gate
    right_proj: torch.Tensor,         # [B, N, N, H]   raw projection
    right_gate: torch.Tensor,         # [B, N, N, H]   sigmoid gate
    mask: torch.Tensor | None,        # [B, N, N]  (fp16) or None
    hidden_dim: int,
    has_mask: bool,
) -> torch.Tensor:
    """
    Batched matmul: out = (left_proj * left_gate * mask) @ (right_proj * right_gate * mask)ᵀ
    Returns FP16 tensor of shape [B, N, N, hidden_dim].
    """
    B, N, _, H = left_proj.shape   # H == hidden_dim
    assert H == hidden_dim

    # ---- Permute + view -------------------------------------------------
    # left and left_gate keep (i,k) order
    left_proj = left_proj.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    left_proj_view = left_proj.view(B * H, N, N)                    # [M, N, K]
    left_gate_view = left_gate.view(B * H, N, N)                    # [M, N, K]

    # right and right_gate need (k,j) order → transpose the last two dims
    right_proj = right_proj.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N]  (k,j)
    right_gate = right_gate.permute(0, 3, 2, 1).contiguous()
    right_proj_view = right_proj.view(B * H, N, N)                  # [M, K, N]
    right_gate_view = right_gate.view(B * H, N, N)                  # [M, K, N]

    # ---- Output buffer ---------------------------------------------------
    out = torch.empty_like(left_proj_view, dtype=torch.float16)

    # ---- Strides (element counts) ----------------------------------------
    stride_am, stride_ai, stride_ak = left_proj_view.stride()
    stride_g1m, stride_g1i, stride_g1k = left_gate_view.stride()
    stride_bm, stride_bk, stride_bj = right_proj_view.stride()
    stride_g2m, stride_g2k, stride_g2j = right_gate_view.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    if has_mask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        stride_mask_batch, stride_mask_i, stride_mask_j = mask_fp16.stride()
        mask_ptr = mask_fp16
    else:
        # Dummy tensor – never read when HAS_MASK=False
        mask_ptr = torch.empty(0, dtype=torch.float16, device=left_proj.device)
        stride_mask_batch = stride_mask_i = stride_mask_j = 0

    # --------------------------------------------------------------
    # Tile sizes – good for H100 Tensor‑Cores
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # M tiles (batch * hidden)
    )

    _trimul_gated_matmul_kernel[grid](
        left_proj_view,
        right_proj_view,
        left_gate_view,
        right_gate_view,
        mask_ptr,
        out,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_g1m,
        stride_g1i,
        stride_g1k,
        stride_g2m,
        stride_g2k,
        stride_g2j,
        stride_mask_batch,
        stride_mask_i,
        stride_mask_j,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
        HAS_MASK=has_mask,
        num_warps=8,
        num_stages=3,               # double‑buffered pipeline
    )

    # ---- Restore original layout [B, N, N, H] -------------------------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor[B, N, N]   (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(dtype_fp16)

    left_gate_w   = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)      # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (keep it for the kernel – no extra write‑back)
    # ------------------------------------------------------------------
    has_mask = not nomask
    mask = None
    if has_mask:
        mask = mask_tensor.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via Triton (gated + optional mask)
    # ------------------------------------------------------------------
    out = _batched_gated_matmul_fp16(
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        hidden_dim=hidden_dim,
        has_mask=has_mask,
    )                                                    # [B, N, N, hidden] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                                    # [B, N, N, hidden] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Final gating + projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                                 # elementwise gate
    out = torch.nn.functional.linear(out, to_out_w)      # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32
    # ------------------------------------------------------------------
    return out.to(dtype_fp32)
