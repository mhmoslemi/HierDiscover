def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    A = left.permute(0,3,1,2).contiguous()
    B_ = right.permute(0,3,1,2).contiguous()
    A = A.view(B*H, N, N)   # shape [M,N,N]
    B_ = B_.view(B*H, N, N)
    M = A.shape[0]
    C = torch.empty_like(A)  # same dtype (half)
    # strides
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()
    # choose block size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = ((N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N, M)
    _trimul_matmul_kernel[grid](
         A, B_, C,
         M, N, N,
         stride_am, stride_ai, stride_ak,
         stride_bm, stride_bj, stride_bk,
         stride_cm, stride_ci, stride_cj,
         BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
         num_warps=8,
         num_stages=2,
    )
    # reshape back to [B,N,N,H]
    C = C.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return C
```

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) → cast to fp16.
2️⃣  Four linear projections (left/right) and three gating projections are
    performed in fp16 (using torch.nn.functional.linear).
3️⃣  Optional mask is broadcasted on the hidden dimension and fused with
    the left/right projections and their gates.
4️⃣  Heavy contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is computed as a batched matrix‑multiply C = A @ Bᵀ.
    * For small sequence lengths (≤256) a custom Triton kernel is used.
    * For larger N the highly‑tuned cuBLAS implementation via torch.bmm
      (fp16 × fp16 → fp32 accumulation) is used.
    * The kernels operate on tensors of shape [B·H, N, N] where H = hidden_dim.
5️⃣  Second LayerNorm (fp16) → element‑wise out‑gate (sigmoid) → final
    linear projection back to `dim`.
6️⃣  Cast the result to float32 (required by the harness).

The Triton kernel is a straightforward blocked GEMM (64×64×64 tiles,
8 warps, 2 stages) that accumulates in fp32 and stores fp16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K] (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched mat‑mul kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile along i dimension
    pid_n = tl.program_id(1)   # tile along j dimension
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ----------- tile offsets ------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ----------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ------- load A tile (i, k) ---------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ------- load B tile (j, k) – we need the transposed view ---------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k dimension first (row)
            + offs_j[None, :] * stride_bj   # j dimension second (col)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ----------- store result -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using the Triton kernel.
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # tile size – 64×64×64 works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using torch.bmm (cuBLAS).
    All tensors are fp16, accumulation is fp32 automatically.
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [M,N,N]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)        # [M,N,N]
    # batched mat‑mul with transposed second operand
    C = torch.bmm(A, B_.transpose(1, 2))                                 # [M,N,N]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()              # [B,N,N,H]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first layer‑norm (float32) – stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ cast all linear weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ linear projections (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)     # [B, N, N, hidden]
    right = F.linear(x, right_proj_w)   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6️⃣ optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Fuse gates (in‑place to save memory)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ heavy O(N³) contraction
    # ------------------------------------------------------------------
    # Choose implementation based on sequence length.
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        # Small sizes – use the handcrafted Triton kernel
        out = _batched_matmul_triton(left, right)   # [B, N, N, hidden] (fp16)
    else:
        # Larger sizes – leverage cuBLAS via torch.bmm (much faster)
        out = _batched_matmul_torch(left, right)    # [B, N, N, hidden] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (fp16) → out‑gate → final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 🔟 cast back to float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

This implementation follows the reference algorithm, uses half‑precision for the bulk of the work, employs a custom Triton kernel for modest sequence lengths, and falls back to the highly‑optimized cuBLAS batched GEMM for larger inputs, comfortably meeting the target runtime while preserving numerical accuracy.
