Kernel:

```
@triton.jit
def fused_mul_gate_mask_kernel(
    left_raw_ptr, left_gate_ptr, right_raw_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    hidden: tl.constexpr,
    size: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_block = offs < size

    # load raw and gate
    left_raw = tl.load(left_raw_ptr + offs, mask=mask_block, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_block, other=0.0)
    right_raw = tl.load(right_raw_ptr + offs, mask=mask_block, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_block, other=0.0)
    
    # compute mask index: (offs // hidden) ; integer division
    mask_idx = offs // hidden
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_block, other=0.0)

    left_out = left_raw * left_gate * mask_val
    right_out = right_raw * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_out, mask=mask_block)
    tl.store(right_out_ptr + offs, right_out, mask=mask_block)
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (high‑level)
----------------------
1️⃣ Input LayerNorm (float32 → float16) – all later arithmetic is FP16.
2️⃣ **Three** fused Linear layers:
   * (left_proj , left_gate)   → weight shape (2*hidden, dim)
   * (right_proj, right_gate) → weight shape (2*hidden, dim)
   * out_gate                  → weight shape (hidden, dim)
   This reduces the number of passes over the huge tensor `x_norm`
   from five to three.
3️⃣ Fuse mask, left/right gating and mask‑application in a single
   Triton kernel.  The kernel computes

        left  = left_proj_raw  * left_gate  * mask
        right = right_proj_raw * right_gate * mask

   (mask is broadcast over the hidden dimension).  When `nomask` is
   true the kernel is skipped and a simple element‑wise product is
   used.
4️⃣ Core contraction – the outgoing update

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   is performed as a batched matrix multiply:

        left_perm  = left.permute(0,3,1,2)   # (B, hidden, i, k)
        right_perm = right.permute(0,3,2,1)  # (B, hidden, k, j)
        out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

   which maps to a strided‑batched GEMM that fully exploits Tensor
   Cores on H100.
5️⃣ Output LayerNorm → multiply by the output gate → final Linear
   projection back to `dim`.  All operations stay in FP16 and the
   result is cast back to the original dtype (usually float32).

Only the mask/gate fusion step is written in Triton, satisfying the
requirement to include a custom kernel while keeping the heavy
computations in highly‑optimized cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (a * b * mask) for left/right branches
# ----------------------------------------------------------------------
@triton.jit
def fused_mul_gate_mask_kernel(
    left_raw_ptr,   # FP16, shape (B,N,N,hidden)
    left_gate_ptr,  # FP16, same shape
    right_raw_ptr,  # FP16, same shape
    right_gate_ptr, # FP16, same shape
    mask_ptr,       # FP16, shape (B,N,N) – broadcasted over hidden
    left_out_ptr,   # FP16, same shape as left_raw
    right_out_ptr,  # FP16, same shape as right_raw
    hidden: tl.constexpr,
    total: tl.constexpr,          # total number of elements = B*N*N*hidden
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load raw projections and gates
    l_raw = tl.load(left_raw_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_raw = tl.load(right_raw_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (mask is expanded over the hidden dim)
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Fuse the three‑way multiplication
    left_out = l_raw * l_gate * m
    right_out = r_raw * r_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def fused_mul_gate_mask(
    left_raw: torch.Tensor,
    left_gate: torch.Tensor,
    right_raw: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    hidden: int,
) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, left/right projections and gating into two output tensors.
    All tensors must be contiguous, half‑precision and on the same CUDA device.
    """
    total = left_raw.numel()
    out_left = torch.empty_like(left_raw)
    out_right = torch.empty_like(right_raw)

    BLOCK = 65536  # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul_gate_mask_kernel[grid](
        left_raw,
        left_gate,
        right_raw,
        right_gate,
        mask,
        out_left,
        out_right,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out_left, out_right

        bias=weights["norm.bias"].to(device, torch.float16),
        eps=eps,
    ).to(torch.float16)  # now (B,N,N,dim) half‑precision

    # --------------------------------------------------------------
    # 2) Fused linear layers (three matmuls instead of five)
    # --------------------------------------------------------------
    # left_proj + left_gate → (2*hidden)-wide projection
    left_cat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)        # (2*hidden, dim)
    left_cat = F.linear(x_norm, left_cat_weight)  # (B,N,N,2*hidden)
    left_proj_raw, left_gate_raw = torch.split(left_cat, hidden, dim=-1)

    # right_proj + right_gate → (2*hidden)-wide projection
    right_cat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)        # (2*hidden, dim)
    right_cat = F.linear(x_norm, right_cat_weight)  # (B,N,N,2*hidden)
    right_proj_raw, right_gate_raw = torch.split(right_cat, hidden, dim=-1)

    # out_gate (single projection)
    out_gate_raw = F.linear(
        x_norm, weights["out_gate.weight"].to(device, torch.float16)
    )  # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 3) Gating + optional mask – fused Triton kernel
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,hidden), FP16
    right_gate = torch.sigmoid(right_gate_raw) # (B,N,N,hidden), FP16
    out_gate = torch.sigmoid(out_gate_raw)     # (B,N,N,hidden), FP16

    if nomask:
        # No mask → simple element‑wise product
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask once to FP16 and make it contiguous
        mask_fp16 = mask.to(device, torch.float16).contiguous()
        left, right = fused_mul_gate_mask(
            left_proj_raw.contiguous(),
            left_gate.contiguous(),
            right_proj_raw.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            hidden,
        )
        # `left` and `right` already have shape (B,N,N,hidden)

    # --------------------------------------------------------------
    # 4) Core contraction (batched GEMM over the ‘k’ dimension)
    # --------------------------------------------------------------
    # Rearrange for batched matmul: (B, hidden, i, k) @ (B, hidden, k, j)
    left_perm = left.permute(0, 3, 1, 2)          # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)        # (B, hidden, k, j)
    out_hidden = torch.matmul(left_perm, right_perm)  # (B, hidden, i, j)

    # Bring back to (B, i, j, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 5) Output LayerNorm → gate → final projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(device, torch.float16),
        bias=weights["to_out_norm.bias"].to(device, torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                       # apply output gate

    out_final = F.linear(
        out_norm, weights["to_out.weight"].to(device, torch.float16)
    )  # (B, N, N, dim), bias‑free

    # Cast back to the original dtype (normally float32)
    return out_final.to(dtype_in)
```
