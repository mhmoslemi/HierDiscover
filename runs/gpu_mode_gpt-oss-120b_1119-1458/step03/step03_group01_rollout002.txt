"""
TriMul (outgoing) – high‑performance forward kernel.

The main bottleneck of the reference implementation is the four separate
linear projections (left/right + their gates) which read the large
[ B, N, N, dim ] tensor four times.  This version
* fuses each pair of projection+gate into a *single* GEMM
  (x @ [W_proj ; W_gate]) so the input is streamed only twice;
* keeps the heavy matrix‑multiply (∑ₖ left[b,i,k,d]·right[b,j,k,d])
  as a batched GEMM on fp16 using torch.bmm (cuBLAS/Tensor‑Cores);
* uses two tiny Triton kernels to fuse the element‑wise gate
  multiplication (and an optional mask) and the final out‑gate
  multiplication;
* performs the two LayerNorms in fp32 (or fp16 where safe) for
  numerical stability and returns the result in float32.

The overall data flow:

    x_norm    = LayerNorm(x)                              # fp32
    x_h       = x_norm.half()                             # fp16
    left+gate = x_h @ [W_left ; W_left_gate]               # fp16
    right+gate= x_h @ [W_right; W_right_gate]              # fp16
    left      = left_proj * sigmoid(left_gate) * mask?     # fused (Triton)
    right     = right_proj* sigmoid(right_gate) * mask?   # fused (Triton)
    out       = bmm(left, right^T)  (B*H, N, N)           # fp16
    out       = LayerNorm(out)                            # fp16
    out      *= sigmoid(out_gate)                         # fused (Triton)
    out       = out @ W_to_out                              # fp16
    return out.float()                                    # fp32
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_mask_kernel(
    left_ptr,            # fp16*  shape (total_el,)
    left_gate_ptr,       # fp16*  same shape as left_ptr
    right_ptr,           # fp16*  shape (total_el,)
    right_gate_ptr,      # fp16*  same shape as right_ptr
    mask_ptr,            # fp16*  shape (mask_el,)  (mask_el = total_el // hidden_dim)
    hidden_dim: tl.constexpr,
    has_mask: tl.constexpr,   # 1 if mask is present, else 0
    numel: tl.int64,          # total number of elements per tensor (left/right)
    BLOCK: tl.constexpr,
):
    """
    In‑place multiplication:
        left  = left  * left_gate  * mask   (if mask is present)
        right = right * right_gate * mask   (if mask is present)

    The tensors are flattened before launch.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # load left, gate
    l      = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg     = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)

    # load right, gate
    r      = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg     = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if has_mask:
        # mask index = element // hidden_dim (integer division)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # broadcast 1.0 if out‑of‑range
        l = l * lg * m
        r = r * rg * m
    else:
        l = l * lg
        r = r * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,        # fp16*  (output tensor to be multiplied in‑place)
    b_ptr,        # fp16*  (same shape as a_ptr)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        a = a * b
    Both tensors are flattened.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]    (float32)
        - mask         : torch.Tensor  [B, N, N]         (bool/float) – may be None
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack & basic config
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # if True, ignore mask
    eps = 1e-5

    device = input_tensor.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all dense work
    # ------------------------------------------------------------------
    x_h = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Prepare fused weight matrices (dim, 2*hidden_dim) for left/right
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(dtype_fp16)          # (dim, hidden_dim)
    w_left_gate = weights["left_gate.weight"].to(dtype_fp16)          # (dim, hidden_dim)
    w_right_proj = weights["right_proj.weight"].to(dtype_fp16)        # (dim, hidden_dim)
    w_right_gate = weights["right_gate.weight"].to(dtype_fp16)        # (dim, hidden_dim)

    # concatenate projection + gate along output dim
    w_left_fused = torch.cat([w_left_proj, w_left_gate], dim=1)      # (dim, 2*hidden_dim)
    w_right_fused = torch.cat([w_right_proj, w_right_gate], dim=1)   # (dim, 2*hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Fused linear transforms (input read only twice)
    # ------------------------------------------------------------------
    # shape after linear: [B, N, N, 2*hidden_dim]
    left_fused = F.linear(x_h, w_left_fused)   # fp16
    right_fused = F.linear(x_h, w_right_fused) # fp16

    # split projection and gate pre‑activations
    left_proj = left_fused[..., :hidden_dim]          # [B, N, N, hidden_dim]
    left_gate_pre = left_fused[..., hidden_dim:]      # same shape
    right_proj = right_fused[..., :hidden_dim]
    right_gate_pre = right_fused[..., hidden_dim:]

    # apply sigmoid to the gate tensors
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 5️⃣ Compute the output gate (separate linear)
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(dtype_fp16)   # (dim, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))    # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Mask handling (optional)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.to(dtype_fp16).unsqueeze(-1)  # [B, N, N, 1]
    else:
        mask_h = None

    # ------------------------------------------------------------------
    # 7️⃣ Fuse element‑wise gate multiplication (and mask if present)
    # ------------------------------------------------------------------
    # make tensors contiguous for safe flattening
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()
    if mask_h is not None:
        mask_h = mask_h.contiguous()

    numel = left_proj.numel()          # total elements = B*N*N*hidden_dim
    BLOCK = 65536                      # ~64k elements per program (tuned)

    grid = ((numel + BLOCK - 1) // BLOCK,)

    # Triton does not support python booleans as constexpr directly,
    # so we convert to int.
    has_mask = 0 if mask_h is None else 1

    apply_gates_mask_kernel[grid](
        left_proj.view(-1),
        left_gate.view(-1),
        right_proj.view(-1),
        right_gate.view(-1),
        mask_h.view(-1) if mask_h is not None else torch.empty(0, device=device, dtype=dtype_fp16).view(-1),
        hidden_dim,
        has_mask,
        numel,
        BLOCK=BLOCK,
    )
    # Gates no longer needed; free memory
    del left_gate, right_gate, left_fused, right_fused, left_gate_pre, right_gate_pre

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation – batched matrix multiplication
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape   # H == hidden_dim
    total_heads = B * H            # treat each head as a separate batch

    # transform to (total_heads, N, N) layout expected by torch.bmm
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)    # (B*H, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)  # (B*H, N, N)

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm (fp16 weights)
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(dtype_fp16)
    b_to_out_norm = weights["to_out_norm.bias"].to(dtype_fp16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )   # fp16

    # ------------------------------------------------------------------
    # 🔟 Multiply by the output gate (in‑place) via Triton
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    # Gate tensor can be released now
    del out_gate

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(dtype_fp16)   # (hidden_dim, dim)
    final = F.linear(out, w_to_out)                      # [B, N, N, dim] fp16

    # Return as float32 to match the reference module's output
    return final.to(dtype_fp32)
