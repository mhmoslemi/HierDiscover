"""
TriMul (outgoing) – fused linear + Triton mask‑gate + batched GEMM
=================================================================
Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16).  
2️⃣ Fuse the five linear transforms (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a *single* GEMM by concatenating
   the weight matrices along the output dimension.  This reduces the
   memory traffic of the input tensor (read once instead of five times)
   and eliminates five separate kernel launches.
3️⃣ Apply the three sigmoid gates.  
4️⃣ If a mask is present we fuse the triple product  
        left = left_proj * left_gate * mask,  
        right = right_proj * right_gate * mask  
   with a small Triton kernel (`fused_mul3_masked`).  This avoids an
   explicit broadcast of the 3‑D mask over the hidden dimension.
5️⃣ Core contraction (outgoing update)  
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  
   is performed with a single batched matrix‑multiply (cuBLAS) by
   reshaping to (B·hidden, N, N).  cuBLAS automatically uses Tensor
   Cores for FP16.
6️⃣ Output LayerNorm, output‑gate multiplication and final projection
   back to the original channel dimension (dim).  All heavy arithmetic
   stays in FP16; the final result is cast back to the input dtype.

The implementation keeps at least one Triton kernel (step 4) as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: a * b * c  (b is a 3‑D mask broadcasted over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,               # left_proj_raw or right_proj_raw   (B,N,N,hidden)
    b_ptr,               # mask                              (B,N,N)
    c_ptr,               # gate (already sigmoid)            (B,N,N,hidden)
    out_ptr,             # destination
    hidden: tl.constexpr,
    size:   tl.constexpr,          # total number of elements in a/c (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load the two FP16 tensors (contiguous in hidden dim)
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    # Derive the index into the 3‑D mask (broadcast over hidden)
    mask_idx = offs // hidden          # integer division
    b = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)

    out = a * c * b
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(a: torch.Tensor,
                      mask: torch.Tensor,
                      gate: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """
    Element‑wise product a * gate * mask where ``mask`` is broadcast over the
    last (hidden) dimension.  All tensors must be contiguous, half‑precision.
    """
    assert a.shape == gate.shape, "proj and gate must share shape"
    B, N, _, H = a.shape
    assert H == hidden, "hidden dimension mismatch"

    a = a.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()               # (B,N,N)

    out = torch.empty_like(a)
    total = a.numel()                      # B * N * N * hidden
    BLOCK = 8192                            # good for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        a, mask, gate, out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]  (float32, CUDA)
            - mask_tensor  : [B, N, N]       (bool/float, CUDA)
            - weights_dict : model weight tensors (see skeleton)
            - config_dict  : contains "dim", "hidden_dim", optional "eps"
                             and "nomask" flag.

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # now FP16 for everything else

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection + gates (single GEMM)
    # ------------------------------------------------------------------
    # Flatten the batch‑sequence dimensions for a single matmul
    x_flat = x_norm.view(-1, dim)           # shape (B*N*N, dim)

    # Concatenate the five weight matrices (each: hidden x dim)
    # Linear expects weight shape (out_features, in_features)
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Stack along the output dimension (0) → (5*hidden, dim)
    w_combined = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate,
         w_right_gate, w_out_gate],
        dim=0,
    ).to(torch.float16)

    # Single linear (bias‑free) → shape (B*N*N, 5*hidden)
    y = F.linear(x_flat, w_combined)                # FP16

    # Reshape to 5‑way tensor
    y = y.view(B, N, N, 5, hidden)                  # (B,N,N,5,hidden)

    left_proj_raw = y[..., 0, :]   # (B,N,N,hidden)
    right_proj_raw = y[..., 1, :]
    left_gate_raw  = y[..., 2, :]
    right_gate_raw = y[..., 3, :]
    out_gate_raw   = y[..., 4, :]

    # Free intermediate y (optional, helps peak memory)
    del y

    # --------------------------------------------------------------
    # 3️⃣ Sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,hidden)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)     # used after output norm

    # --------------------------------------------------------------
    # 4️⃣ Apply mask + gate (mask fusion via Triton)
    # --------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise product
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask once to half; Triton kernel broadcasts it on‑the‑fly
        mask_fp16 = mask.to(torch.float16)
        left = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # --------------------------------------------------------------
    # 5️⃣ Core contraction (outgoing update)
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, i, k) and (B, hidden, k, j) for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)  # (B, hidden, k, j)

    # Batched matmul across (B*hidden) → cuBLAS uses Tensor Cores (FP16)
    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # Restore original layout
    out = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm, output‑gate, final projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # Multiply by the output gate (broadcast on hidden dim)
    out_norm = out_norm * out_gate

    # Final linear projection back to original `dim`
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)   # bias‑free
    )   # (B, N, N, dim), still FP16

    # Cast back to the input dtype (usually float32)
    return out_final.to(inp.dtype)
