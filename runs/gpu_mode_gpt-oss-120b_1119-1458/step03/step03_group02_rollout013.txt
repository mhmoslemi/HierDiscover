"""
TriMul outgoing (AlphaFold‑3) – high‑performance Triton implementation
=====================================================================

This module implements the forward‑only “outgoing” TriMul operation described
in the AlphaFold‑3 paper.  The expensive O(N³) contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

is realised as a *batched* matrix multiplication C = A @ Bᵀ where
A and B have shape (M, N, N) with M = batch × hidden_dim.  The heavy
contraction is off‑loaded to a custom Triton kernel that tiles the
multiplication with 128 × 128 × 32 blocks, accumulates in FP32 with
tensor‑core MMA (`tl.dot`) and writes the result back in FP16.  By
fusing the two linear projections per side (proj + gate) into a single
matrix multiplication we halve the number of dense GEMMs, keep most
operations in FP16 for bandwidth, and only cast back to FP32 at the end.

The implementation supports optional masking (applied in‑place after the
gates) and works for all test configurations (B ∈ {1,2}, N ∈ {128,…,1024},
dim ∈ {128,384}, hidden_dim = 128, with or without a mask).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, N]   (M = batch * hidden_dim)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, N]
    B_ptr,               # *fp16  [M, N, N]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes (K == N)
    stride_am, stride_ai, stride_ak,   # A strides (M,N,N)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,N)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batched matrix product for many independent (N×N) matrices.
    Each program (pid_z) processes one (M‑index) i.e. one (batch, hidden) pair.
    The computation is tiled over the two N‑axes (i‑ and j‑axes) with
    blocks of size BLOCK_M × BLOCK_N.
    """
    pid_i = tl.program_id(0)      # tile along first N dimension (i)
    pid_j = tl.program_id(1)      # tile along second N dimension (j)
    pid_z = tl.program_id(2)      # index over M (= batch * hidden_dim)

    # -----------------------------------------------------------------
    # Offsets for this tile
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (more numerically stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # -----------------------------------------------------------------
    # Loop over K dimension (the contracted dimension)
    # -----------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile: A[pid_z, i, k]
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # B tile (transposed view): B[pid_z, j, k]  -> we need B[j, k]
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Tensor‑core MMA:  fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Store FP16 result
    # -----------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape [B, N, N, H] (fp16).
    right : torch.Tensor
        Tensor of shape [B, N, N, H] (fp16).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension.
    #   (B, N, N, H) → (B*H, N, N)   (contiguous)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides in element units (triton expects i32)
    # ------------------------------------------------------------------
    stride_am, stride_ai, stride_ak = A.stride()          # (M, N, K)
    stride_bm, stride_bi, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # ------------------------------------------------------------------
    # Block sizes – tuned for H100 (FP16 tensor‑core MMA)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid dimensions
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # M‑tiles (batch × hidden)
    )

    # Launch kernel
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to the original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # Cast to FP16 for the remaining operations
    # -----------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the two projection + gate linear layers per side
    # -----------------------------------------------------------------
    # left side:  proj + gate weight concatenation
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2*H, dim)

    left_comb = torch.nn.functional.linear(x_fp16, left_comb_w)  # [B,N,N,2H] fp16
    left = left_comb[..., :hidden_dim]                         # proj part
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])     # gate part (fp16)

    # right side:  proj + gate weight concatenation
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)

    right_comb = torch.nn.functional.linear(x_fp16, right_comb_w)  # [B,N,N,2H] fp16
    right = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (applied in‑place to avoid extra allocations)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left *= mask_fp16
        right *= mask_fp16

    # -----------------------------------------------------------------
    # 4️⃣ Output gate (computed once, later applied after the second LN)
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))  # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (fp16 in, fp16 out)
    # -----------------------------------------------------------------
    # Result shape: [B, N, N, hidden_dim] (fp16)
    contracted = _batched_matmul_transpose_fp16(left, right)

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (still fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Final gating + projection back to the original dimension
    # -----------------------------------------------------------------
    contracted = contracted * out_gate                     # element‑wise gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out_fp16 = torch.nn.functional.linear(contracted, to_out_w)  # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the required output dtype
    # -----------------------------------------------------------------
    return out_fp16.to(torch.float32)
```
