"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1.  Layer‑norm the 4‑D input (float‑32 → FP16).
2.  Apply all linear projections & gates in FP16, fuse mask & gating.
3.  (Optional) pad the sequence dimension to a multiple of the tile size
    (the test shapes are already multiples of 64, padding is a no‑op there).
4.  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a custom Triton kernel that performs a batched matrix multiply
    C = A @ Bᵀ for each (batch, hidden) pair.  The kernel tiles the i‑ and
    j‑axes with 64×64×64 blocks, uses fp16 values, fp32 accumulation, 8 warps
    and a 3‑stage pipeline.
5.  Slice away any padding, apply the second LayerNorm, the output gate,
    and the final linear projection back to the original `dim`.
6.  Return the result in FP32 (the surrounding framework expects FP32).

Only the O(N³) part lives in Triton; all surrounding ops are performed with
high‑performance PyTorch primitives (layer_norm, linear, sigmoid, etc.).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]  (M = batch * hidden, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton kernel that computes, for each (batch, hidden) slice:
        C[i, j] = Σₖ A[i, k] * B[j, k]    (B is accessed transposed)
    """
    pid_m = tl.program_id(0)  # tile on i (first N axis)
    pid_n = tl.program_id(1)  # tile on j (second N axis)
    pid_z = tl.program_id(2)  # batch * hidden index

    # ------------------------------------------------------------------
    # 1) Compute global offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)      # (BLOCK_M,)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)      # (BLOCK_N,)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 2) Accumulator in FP32 (to avoid overflow)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)                # (BLOCK_K,)

        # ----- load A[i, k] -------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ----- load Bᵀ[k, j] (i.e. B[k, j]) ---------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # ----- dot product (fp16 * fp16 -> fp32) ----------------------------
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 4) Write back the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_trimul_matmul_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Perform the contraction `out = left @ rightᵀ` per (batch, hidden) using Triton.
    Input:
        left, right – fp16 tensors of shape [B, N, N, H] (H = hidden_dim)
        N is already padded to a multiple of `tile`.
    Output:
        Tensor of shape [B, N, N, H] (still padded).
    """
    B, N, _, H = left.shape

    # Pack batch & hidden into a single leading dimension so each program works on
    # a (N, N) matrix pair.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)    # [B*H, N, N]
    B_ = right_perm.view(B * H, N, N)  # [B*H, N, N]
    C = torch.empty_like(A)            # fp16 output

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden slices
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Restore original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # still float32
    x = x_norm.to(dtype_fp16)  # fp16 for everything that follows

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once (save repeated casts)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(dtype_fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections & gates (all in fp16, no extra flattening)
    # ------------------------------------------------------------------
    # Projections
    left = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # Optional mask (broadcast over hidden dim)
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates (sigmoid)
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, left_gate_w)
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, right_gate_w)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_w)
    )

    # Fuse gates into the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Pad sequence length to a multiple of TILE (64) to drop bounds checks.
    #     (All test dimensions are already multiples of 64, but we keep the
    #      logic for completeness.)
    # ------------------------------------------------------------------
    TILE = 64
    Bsz, N, _, _ = left.shape
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((Bsz, N_pad, N_pad, hidden_dim), dtype=dtype_fp16)
        right_pad = right.new_zeros((Bsz, N_pad, N_pad, hidden_dim), dtype=dtype_fp16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (fp16 in/out)
    # ------------------------------------------------------------------
    out = _batched_trimul_matmul_fp16(left_pad, right_pad, tile=TILE)   # [B,N_pad,N_pad,H]

    # Remove any padding that was added
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (still in fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (both fp16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (the caller expects FP32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
