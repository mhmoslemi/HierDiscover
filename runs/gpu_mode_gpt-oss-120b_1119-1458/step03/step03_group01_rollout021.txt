@triton.jit
def trmul_outgoing_kernel(
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr, mask_ptr,
    out_ptr,
    batch_dim,      # total_batches
    N,              # dimension size
    hidden_dim,     # hidden_dim maybe not needed compute B = batch_dim // hidden_dim but we may not need it as mask already expanded
    stride_left_batch, stride_left_i, stride_left_k,
    stride_left_gate_batch, stride_left_gate_i, stride_left_gate_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_right_gate_batch, stride_right_gate_k, stride_right_gate_j,
    stride_mask_batch, stride_mask_i, stride_mask_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # grid: (batch_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    # each program processes a tile of out: [BLOCK_M, BLOCK_N] for given batch.

    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute base offsets for out
    out_ptrs = out_ptr + pid_batch * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, N, BLOCK_K):
        # Offsets for left_proj: shape (BLOCK_M, BLOCK_K)
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left_ptrs = left_proj_ptr + pid_batch * stride_left_batch + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_gate_ptrs = left_gate_ptr + pid_batch * stride_left_gate_batch + offs_m[:, None] * stride_left_gate_i + offs_k[None, :] * stride_left_gate_k

        # load and multiply
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        ag = tl.load(left_gate_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a * ag
        # optional mask multiplication:
        if mask_ptr is not None:
            mask_ptrs = mask_ptr + pid_batch * stride_mask_batch + offs_m[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
            mask_a = tl.load(mask_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=1.0) # mask default 1.0
            a = a * mask_a

        # Right side tile: we need B tile shape (BLOCK_K, BLOCK_N) from right_proj and right_gate and mask
        # Right proj shape is (total_batches, N_k, N_j) where first axis is batch, second is k, third is j
        # Offsets: stride_right_batch + offs_k * stride_right_k + offs_n[None,:] * stride_right_j

        right_ptrs = right_proj_ptr + pid_batch * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right_gate_ptrs = right_gate_ptr + pid_batch * stride_right_gate_batch + offs_k[:, None] * stride_right_gate_k + offs_n[None, :] * stride_right_gate_j

        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        bg = tl.load(right_gate_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b * bg
        if mask_ptr is not None:
            mask_ptrs_b = mask_ptr + pid_batch * stride_mask_batch + offs_k[:, None] * stride_mask_i + offs_n[None, :] * stride_mask_k
            mask_b = tl.load(mask_ptrs_b, mask=mask_k[:, None] & mask_n[None, :], other=1.0)
            b = b * mask_b

        # Now compute dot: A tile (M,K) times B tile (K,N) -> (M,N)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write out output
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    import torch.nn.functional as F
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    eps = 1e-5

    # 1. LayerNorm over dim (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_h = x.to(torch.float16)

    # Linear projections
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))    # shape [B,N,N,hidden_dim]
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gate linear layers
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    # Optional mask to float16
    if not nomask:
        mask_h = mask.to(torch.float16)
    else:
        # use ones mask
        mask_h = torch.ones(input_tensor.shape[:3], dtype=torch.float16, device=input_tensor.device)

    # Permute for matmul
    left_proj_perm = left_proj.permute(0,3,1,2).contiguous()  # [B, H, N, N]
    right_proj_perm = right_proj.permute(0,3,2,1).contiguous()  # [B, H, N, N] (k,j)
    left_gate_perm = left_gate.permute(0,3,1,2).contiguous()
    right_gate_perm = right_gate.permute(0,3,2,1).contiguous()

    batch, H, N, _ = left_proj_perm.shape

    total_batches = batch * H

    # Flatten to shape (total_batches, N, N)
    left_proj_flat = left_proj_perm.view(total_batches, N, N)
    right_proj_flat = right_proj_perm.view(total_batches, N, N)
    left_gate_flat = left_gate_perm.view(total_batches, N, N)
    right_gate_flat = right_gate_perm.view(total_batches, N, N)

    # Build mask_flat expanded across heads:
    if not nomask:
        # mask shape [B, N, N]
        mask_expand = mask_h.unsqueeze(1).expand(-1, H, -1, -1)  # shape (B, H, N, N)
        mask_flat = mask_expand.contiguous().view(total_batches, N, N)
    else:
        mask_flat = torch.ones((total_batches, N, N), dtype=torch.float16, device=input_tensor.device)

    # allocate output tensor
    out_flat = torch.empty_like(left_proj_flat, dtype=torch.float16)  # shape (total_batches,N,N)

    # Compute strides in elements
    stride_a_batch = left_proj_flat.stride(0)
    stride_a_i = left_proj_flat.stride(1)
    stride_a_k = left_proj_flat.stride(2)

    stride_ag_batch = left_gate_flat.stride(0)
    stride_ag_i = left_gate_flat.stride(1)
    stride_ag_k = left_gate_flat.stride(2)

    stride_b_batch = right_proj_flat.stride(0)
    stride_b_k = right_proj_flat.stride(1)
    stride_b_j = right_proj_flat.stride(2)

    stride_bg_batch = right_gate_flat.stride(0)
    stride_bg_k = right_gate_flat.stride(1)
    stride_bg_j = right_gate_flat.stride(2)

    stride_mask_batch = mask_flat.stride(0)
    stride_mask_i = mask_flat.stride(1)
    stride_mask_k = mask_flat.stride(2)

    stride_out_batch = out_flat.stride(0)
    stride_out_i = out_flat.stride(1)
    stride_out_j = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (total_batches, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    trmul_outgoing_kernel[grid](
        left_proj_flat,
        left_gate_flat,
        right_proj_flat,
        right_gate_flat,
        mask_flat,
        out_flat,
        total_batches,
        N,
        # strides
        stride_a_batch, stride_a_i, stride_a_k,
        stride_ag_batch, stride_ag_i, stride_ag_k,
        stride_b_batch, stride_b_k, stride_b_j,
        stride_bg_batch, stride_bg_k, stride_bg_j,
        stride_mask_batch, stride_mask_i, stride_mask_k,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape out_flat back to (B,N,N,H)
    out = out_flat.view(batch, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16), eps=eps)

    # Multiply out_gate
    out = out * out_gate   # maybe fused kernel but easy with multiplication

    # Final linear projection
    final = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return final.to(torch.float32)

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_left_gate_batch, stride_left_gate_i, stride_left_gate_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_right_gate_batch, stride_right_gate_k, stride_right_gate_j,
    stride_mask_batch, stride_mask_i, stride_mask_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offset_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offset_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offset_m < N
    mask_n = offset_n < N
    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        offset_k = k + tl.arange(0, BLOCK_K)
        mask_k = offset_k < N

        # Load left projection and left gate
        a_ptrs = left_ptr + pid_batch * stride_left_batch + offset_m[:, None] * stride_left_i + offset_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        ag_ptrs = left_gate_ptr + pid_batch * stride_left_gate_batch + offset_m[:, None] * stride_left_gate_i + offset_k[None, :] * stride_left_gate_k
        ag = tl.load(ag_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a * ag

        # mask multiplication for A side
        mask_a_ptrs = mask_ptr + pid_batch * stride_mask_batch + offset_m[:, None] * stride_mask_i + offset_k[None, :] * stride_mask_k
        mask_a = tl.load(mask_a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=1.0)
        a = a * mask_a

        # Load right projection and right gate (note dimensions: K x N)
        b_ptrs = right_ptr + pid_batch * stride_right_batch + offset_k[:, None] * stride_right_k + offset_n[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        bg_ptrs = right_gate_ptr + pid_batch * stride_right_gate_batch + offset_k[:, None] * stride_right_gate_k + offset_n[None, :] * stride_right_gate_j
        bg = tl.load(bg_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b * bg

        # mask multiplication for B side
        mask_b_ptrs = mask_ptr + pid_batch * stride_mask_batch + offset_k[:, None] * stride_mask_i + offset_n[None, :] * stride_mask_k
        mask_b = tl.load(mask_b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=1.0)
        b = b * mask_b

        # dot product
        acc += tl.dot(a, b, out_dtype=tl.float32)  # a: (M,K), b: (K,N) -> (M,N)

    # Store result
    out_ptrs = out_ptr + pid_batch * stride_out_batch + offset_m[:, None] * stride_out_i + offset_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1. Layer‑norm the input (FP32 for numerical safety).
2. Cast to FP16 and compute the four linear projections:
   * left_proj, right_proj  – shape [B,N,N,H]
   * left_gate, right_gate  – shape [B,N,N,H] (sigmoid)
   * out_gate               – shape [B,N,N,H] (sigmoid)
3. If a mask is supplied it is broadcast over the hidden dimension.
4. Permute tensors to [B,H,N,N] and flatten the first two axes,
   giving a batched GEMM layout (batch = B·H, M = N, K = N, N = N).
5. **Fused Triton kernel**:
   * Loads the projection, gate and mask values,
   * Multiplies `proj * gate * mask` on‑the‑fly,
   * Accumulates the product `∑_k left_i_k * right_j_k`
     with FP16 matrix‑core arithmetic (accumulated in FP32).
   The kernel computes the full O(B·H·N³) operation without any
   intermediate materialisation of gated tensors.
6. Reshape the result back to [B,N,N,H].
7. Apply the output LayerNorm (FP32), multiply by `out_gate`
   (element‑wise) and a final linear projection back to the original
   channel dimension `dim`.
8. Return the result in FP32.

The kernel fuses the element‑wise gating & masking with the N³
multiplication, dramatically reducing memory traffic compared to a
two‑step approach (separate gating + `torch.bmm`).  All heavy compute
runs in FP16 on H100 Tensor‑Cores; only the two LayerNorms run in
FP32 for stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fused batched GEMM with per‑element gates & mask
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    # tensor pointers (fp16)
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr, mask_ptr, out_ptr,
    # problem size
    N,
    # strides (in elements)
    stride_left_batch, stride_left_i, stride_left_k,
    stride_left_gate_batch, stride_left_gate_i, stride_left_gate_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_right_gate_batch, stride_right_gate_k, stride_right_gate_j,
    stride_mask_batch, stride_mask_i, stride_mask_k,
    stride_out_batch, stride_out_i, stride_out_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out[b,i,j] = Σ_k (left[b,i,k] * left_gate[b,i,k] * mask[b,i,k]) *
                     (right[b,k,j] * right_gate[b,k,j] * mask[b,k,j])
    where the batch dimension b = batch*head (total_batches = B·H).

    All tensors are fp16; accumulation is performed in fp32.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets in the M and N dimensions
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tile size = BLOCK_K)
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------ A side (left * left_gate * mask) ------------------
        a_ptrs = left_ptr + pid_batch * stride_left_batch \
                 + offs_m[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        ag_ptrs = left_gate_ptr + pid_batch * stride_left_gate_batch \
                  + offs_m[:, None] * stride_left_gate_i \
                  + offs_k[None, :] * stride_left_gate_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        ag = tl.load(ag_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a * ag

        mask_a_ptrs = mask_ptr + pid_batch * stride_mask_batch \
                      + offs_m[:, None] * stride_mask_i \
                      + offs_k[None, :] * stride_mask_k
        mask_a = tl.load(mask_a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=1.0)
        a = a * mask_a

        # ------------------ B side (right * right_gate * mask) ------------------
        b_ptrs = right_ptr + pid_batch * stride_right_batch \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_j
        bg_ptrs = right_gate_ptr + pid_batch * stride_right_gate_batch \
                  + offs_k[:, None] * stride_right_gate_k \
                  + offs_n[None, :] * stride_right_gate_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        bg = tl.load(bg_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b * bg

        mask_b_ptrs = mask_ptr + pid_batch * stride_mask_batch \
                      + offs_k[:, None] * stride_mask_i \
                      + offs_n[None, :] * stride_mask_k
        mask_b = tl.load(mask_b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=1.0)
        b = b * mask_b

        # ------------------ Matrix multiplication ------------------
        acc += tl.dot(a, b, out_dtype=tl.float32)   # (BLOCK_M,BLOCK_N) += (M,K)·(K,N)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + pid_batch * stride_out_batch \
               + offs_m[:, None] * stride_out_i \
               + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Front‑end wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim]    (float32)
        - mask         : torch.Tensor      [B, N, N]         (bool/float, optional)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict containing at least 'dim', 'hidden_dim' and optional
                         'nomask' (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                      # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_h, w_left_proj)    # [B, N, N, H]  fp16
    right_proj = F.linear(x_h, w_right_proj) # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gate pre‑activations (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16)           # [B, N, N]
    else:
        # a tensor of ones – cheap to broadcast later
        mask_h = torch.ones_like(input_tensor[..., 0], dtype=torch.float16)  # [B,N,N]

    # ------------------------------------------------------------------
    # 7️⃣ Permute to (B, H, N, N) and flatten batch × head
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    left_proj = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_proj = right_proj.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – k/j swapped
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 2, 1).contiguous()
    total_batches = B * hidden_dim

    # flatten batch*head → single batch dimension used by the kernel
    left_proj = left_proj.view(total_batches, N, N)
    right_proj = right_proj.view(total_batches, N, N)
    left_gate = left_gate.view(total_batches, N, N)
    right_gate = right_gate.view(total_batches, N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Broadcast mask to (total_batches, N, N)  (no extra copy if nomask)
    # ------------------------------------------------------------------
    mask_flat = mask_h.unsqueeze(1).expand(-1, hidden_dim, -1, -1).contiguous().view(total_batches, N, N)

    # ------------------------------------------------------------------
    # 9️⃣ Allocate output buffer (fp16)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_proj, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 10️⃣ Compute strides (in elements) for all tensors
    # ------------------------------------------------------------------
    s_l_b, s_l_i, s_l_k = left_proj.stride()
    s_lg_b, s_lg_i, s_lg_k = left_gate.stride()
    s_r_b, s_r_k, s_r_j = right_proj.stride()
    s_rg_b, s_rg_k, s_rg_j = right_gate.stride()
    s_m_b, s_m_i, s_m_k = mask_flat.stride()
    s_o_b, s_o_i, s_o_j = out_flat.stride()

    # ------------------------------------------------------------------
    # 11️⃣ Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # ------------------------------------------------------------------
    # 12️⃣ Launch the fused kernel
    # ------------------------------------------------------------------
    trmul_outgoing_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_flat,
        out_flat,
        N,
        s_l_b, s_l_i, s_l_k,
        s_lg_b, s_lg_i, s_lg_k,
        s_r_b, s_r_k, s_r_j,
        s_rg_b, s_rg_k, s_rg_j,
        s_m_b, s_m_i, s_m_k,
        s_o_b, s_o_i, s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 13️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 14️⃣ Post‑processing LayerNorm (FP32)
    # ------------------------------------------------------------------
    out_fp32 = F.layer_norm(
        out.to(torch.float32),
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 15️⃣ Multiply by the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 16️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim]  fp16

    # Return to expected float32 precision
    return final.to(torch.float32)
