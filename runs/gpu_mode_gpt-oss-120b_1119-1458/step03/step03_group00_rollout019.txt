Implementation details in Triton:

```
@triton.jit
def fused_mul3_gate_mask_kernel(
    left_ptr,
    left_gate_ptr,
    right_ptr,
    right_gate_ptr,
    mask_ptr,
    left_out_ptr,
    right_out_ptr,
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0) # 1D
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # compute mask idx
    mask_idx = offs // hidden

    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # Cast to float32 for input layernorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # combine linear weights
    w_comb = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)  # shape (5*hidden, dim)

    # combine linear
    proj = torch.nn.functional.linear(x_norm.view(-1, dim), w_comb)  # shape (B*N*N, 5*hidden)
    B, N, _, _ = input_tensor.shape
    proj = proj.view(B, N, N, 5 * hidden)

    split = torch.split(proj, hidden, dim=-1)
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = split

    # compute out_gate
    out_gate = torch.sigmoid(out_gate_raw)  # shape B,N,N,hidden

    # apply mask and gates to left and right
    if nomask:
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # mask is shape (B,N,N), cast to float16
        mask_fp16 = mask.to(torch.float16)
        # allocate output tensors
        left = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        total = left_proj_raw.numel()
        BLOCK = 65536  # may tune
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        fused_mul3_gate_mask_kernel[grid](
            left_proj_raw, left_gate_raw,
            right_proj_raw, right_gate_raw,
            mask_fp16,
            left, right,
            hidden,
            total,
            BLOCK_SIZE=BLOCK,
        )
        # No need to compute left_gate and right_gate separately

    # Contraction
    # left: (B,N,N,hidden) -> (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # need j as second dimension

    # reshape for batched matmul
    left_mat = left_perm.view(B * hidden, N, N)
    right_mat = right_perm.view(B * hidden, N, N)

    out_hidden = torch.bmm(left_mat, right_mat)  # shape (B*hidden, N, N)
    # reshape back
    out_hidden = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # Output layernorm
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # gating
    out_norm = out_norm * out_gate

    # final projection
    out = torch.nn.functional.linear(out_norm.view(-1, hidden), weights["to_out.weight"].to(torch.float16))
    out = out.view(B, N, N, -1)  # dim

    # cast back to input dtype
    return out.to(input_tensor.dtype)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul3_gate_mask_kernel(
    proj_ptr,               # left_proj raw [B,N,N,H]
    gate1_ptr,              # left_gate raw [B,N,N,H]
    proj2_ptr,              # right_proj raw [B,N,N,H]
    gate2_ptr,              # right_gate raw [B,N,N,H]
    mask_ptr,               # mask [B,N,N]
    out1_ptr,               # left out (in-place)
    out2_ptr,               # right out (in-place)
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load data
    proj1 = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate1_raw = tl.load(gate1_ptr + offs, mask=mask, other=0.0)
    proj2 = tl.load(proj2_ptr + offs, mask=mask, other=0.0)
    gate2_raw = tl.load(gate2_ptr + offs, mask=mask, other=0.0)

    # Load mask (broadcast over hidden)
    mask_idx = offs // hidden
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Sigmoid
    gate1 = 1.0 / (1.0 + tl.exp(-gate1_raw))
    gate2 = 1.0 / (1.0 + tl.exp(-gate2_raw))

    out1 = proj1 * gate1 * mask_val
    out2 = proj2 * gate2 * mask_val

    tl.store(out1_ptr + offs, out1, mask=mask)
    tl.store(out2_ptr + offs, out2, mask=mask)

def fused_mul3_gate_mask(proj1, gate1_raw, proj2, gate2_raw, mask, hidden):
    """Fuse projection * sigmoid(gate) * mask for left and right tensors."""
    assert proj1.shape == gate1_raw.shape == proj2.shape == gate2_raw.shape
    B, N, _, H = proj1.shape
    assert H == hidden
    total = proj1.numel()
    BLOCK = 8192  # can be tuned; larger block may increase occupancy
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
    fused_mul3_gate_mask_kernel[grid](
        proj1, gate1_raw,
        proj2, gate2_raw,
        mask,
        proj1,  # write in-place
        proj2,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    # In‑place modification; return the same tensors for convenience
    return proj1, proj2

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.
    ...
    """
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (FP32)
    x = F.layer_norm(inp,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)
    x = x.to(torch.float16)

    # 2. Combined linear projection for left/right and gates
    w_comb = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)  # shape (5*hidden, dim)
    # flatten input
    B, N, _, _ = inp.shape
    proj = F.linear(x.view(-1, dim), w_comb)      # (B*N*N, 5*hidden)
    proj = proj.view(B, N, N, 5 * hidden)

    # split the combined projection
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden, dim=-1)

    # sigmoid for out_gate (used later)
    out_gate = torch.sigmoid(out_gate_raw)

    if nomask:
        # Simple element‑wise multiply with sigmoid‑gated values
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask.to(torch.float16)  # shape (B,N,N)
        # Fuse mask, sigmoid(gate) and projection for left/right in a Triton kernel
        left, right = fused_mul3_gate_mask(left_proj_raw,
                                           left_gate_raw,
                                           right_proj_raw,
                                           right_gate_raw,
                                           mask_fp16,
                                           hidden)

    # 3. Core outgoing contraction (batched GEMM per hidden channel)
    # Bring hidden dimension to batch axis for better GEMM layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N)

    left_mat = left_perm.view(B * hidden, N, N)
    right_mat = right_perm.view(B * hidden, N, N)

    out_hidden = torch.bmm(left_mat, right_mat)               # (B*hidden, N, N)
    out_hidden = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # 4. Output layer‑norm, gate and final projection
    out_norm = F.layer_norm(out_hidden,
                            normalized_shape=(hidden,),
                            weight=weights["to_out_norm.weight"].to(torch.float16),
                            bias=weights["to_out_norm.bias"].to(torch.float16),
                            eps=eps)

    out_norm = out_norm * out_gate

    out = F.linear(out_norm.view(-1, hidden), weights["to_out.weight"].to(torch.float16))
    out = out.view(B, N, N, -1)

    return out.to(inp.dtype)

"""TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input (FP32 → FP16).  
2️⃣ Fuse the five linear projections (left/right + four gates) into a single
   GEMM by concatenating the weight matrices along the output dimension.
3️⃣ Fuse mask broadcasting, sigmoid‑gating and the two projections into a
   single Triton kernel (the mask is loaded only once for a given (b,i,k)).  
4️⃣ Core contraction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  
   is realized as a batched GEMM (one GEMM per hidden channel) after
   permuting the tensors to a contiguous (B·hidden, N, N) layout.
5️⃣ Output layer‑norm, multiplication by the output gate and the final
   linear projection (hidden → dim) are done with regular PyTorch kernels.
   The result is cast back to the original dtype (usually FP32).

The kernel in step 3 is the only Triton kernel required, satisfying the
specification while keeping the heavy O(N³) contraction on cuBLAS,
which on an H100 is already near the hardware peak."""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (proj * sigmoid(gate) * mask) for *both* left and right
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_gate_mask_kernel(
    proj_l_ptr,      # left projection (B,N,N,hidden)
    gate_l_ptr,      # left gate pre‑sigmoid (B,N,N,hidden)
    proj_r_ptr,      # right projection (B,N,N,hidden)
    gate_r_ptr,      # right gate pre‑sigmoid (B,N,N,hidden)
    mask_ptr,        # mask (B,N,N) – broadcast over hidden
    out_l_ptr,       # output left (in‑place)
    out_r_ptr,       # output right (in‑place)
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total

    # Load tensors
    proj_l    = tl.load(proj_l_ptr + offs, mask=valid, other=0.0)
    gate_l_raw = tl.load(gate_l_ptr + offs, mask=valid, other=0.0)
    proj_r    = tl.load(proj_r_ptr + offs, mask=valid, other=0.0)
    gate_r_raw = tl.load(gate_r_ptr + offs, mask=valid, other=0.0)

    # Mask index (same for the whole hidden slice)
    mask_idx = offs // hidden
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)

    # Sigmoid (1 / (1+exp(-x))) – no bias term, cheap enough for fp16
    gate_l = 1.0 / (1.0 + tl.exp(-gate_l_raw))
    gate_r = 1.0 / (1.0 + tl.exp(-gate_r_raw))

    out_l = proj_l * gate_l * mask_val
    out_r = proj_r * gate_r * mask_val

    tl.store(out_l_ptr + offs, out_l, mask=valid)
    tl.store(out_r_ptr + offs, out_r, mask=valid)


def fused_mul3_gate_mask(proj_l, gate_l_raw, proj_r, gate_r_raw, mask, hidden):
    """
    In‑place fuse ``proj * sigmoid(gate) * mask`` for the left/right tensors.
    All tensors must be contiguous and of dtype torch.float16.
    Returns the (now overwritten) ``proj_l`` and ``proj_r`` tensors.
    """
    total = proj_l.numel()
    BLOCK = 8192  # tuned for H100; can be increased for larger tensors
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_gate_mask_kernel[grid](
        proj_l,
        gate_l_raw,
        proj_r,
        gate_r_raw,
        mask,
        proj_l,          # write in‑place
        proj_r,          # write in‑place
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return proj_l, proj_r

# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  [B, N, N] (bool/float, CUDA) – ignored when config['nomask']==True
            - weights_dict: model weights (see skeleton)
            - config_dict: { "dim": int, "hidden_dim": int, "eps": float (opt.), "nomask": bool (opt.) }

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 → FP16 for the rest)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Combined linear projection for left/right & gates
    # ------------------------------------------------------------------
    # Stack the five weight matrices: [left_proj, right_proj, left_gate, right_gate, out_gate]
    w_comb = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5*hidden, dim)

    # Linear: (B·N·N, dim) @ (5*hidden, dim).T → (B·N·N, 5*hidden)
    proj = F.linear(x.view(-1, dim), w_comb)                # (B*N*N, 5*hidden)
    proj = proj.view(B, N, N, 5 * hidden)                  # (B, N, N, 5*hidden)

    # ------------------------------------------------------------------
    # 3) Slice the combined projection
    # ------------------------------------------------------------------
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden, dim=-1
    )   # each : (B, N, N, hidden)

    # Output gate (used after the contraction)
    out_gate = torch.sigmoid(out_gate_raw)                  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4) Apply mask & sigmoid‑gate to the two projections
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise product
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask to half‑precision once; Triton kernel will broadcast it.
        mask_fp16 = mask.to(torch.float16)
        left, right = fused_mul3_gate_mask(
            left_proj_raw,
            left_gate_raw,
            right_proj_raw,
            right_gate_raw,
            mask_fp16,
            hidden,
        )
        # left/right are now the gated & masked tensors (in‑place)

    # ------------------------------------------------------------------
    # 5) Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented via a batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    # (B, N, N, hidden) → (B, hidden, N, N) – make it contiguous for cuBLAS
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, hidden, N, N)

    # Collapse the leading dimensions so that we have a single batch axis
    left_mat = left_perm.view(B * hidden, N, N)
    right_mat = right_perm.view(B * hidden, N, N)

    # Batched matrix multiplication (one GEMM per hidden channel)
    out_hidden = torch.bmm(left_mat, right_mat)                     # (B*hidden, N, N)

    # Restore original layout: (B, N, N, hidden)
    out_hidden = (
        out_hidden.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    out_norm = out_norm * out_gate                     # apply output gate

    # Final linear projection hidden → dim
    out = F.linear(out_norm.view(-1, hidden), weights["to_out.weight"].to(torch.float16))
    out = out.view(B, N, N, -1)                        # (B, N, N, dim)

    # Cast back to the original dtype (usually float32)
    return out.to(inp.dtype)
```
