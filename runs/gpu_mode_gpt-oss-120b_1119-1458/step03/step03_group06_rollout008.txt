@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    MASKED: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        a_ptrs = A_ptr + pid_z*stride_am + offs_i[:,None]*stride_ai + offs_k[None,:]*stride_ak
        if MASKED:
            a = tl.load(a_ptrs, mask=mask_i[:,None] & (offs_k[None,:] < K), other=0.0)
        else:
            a = tl.load(a_ptrs)

        b_ptrs = B_ptr + pid_z*stride_bm + offs_k[:,None]*stride_bk + offs_j[None,:]*stride_bj
        if MASKED:
            b = tl.load(b_ptrs, mask=(offs_k[:,None] < K) & mask_j[None,:], other=0.0)
        else:

"""
TriMul (outgoing) – Triton‑accelerated implementation
=====================================================

The operator works on a 4‑D tensor *x* of shape [B, N, N, C] and computes

    out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]

where the “left’’ and “right’’ tensors are obtained from a single fused
linear projection of *x* (including three gating projections).  The heavy
contraction is a batch of independent matrix‑multiplications
(A @ Bᵀ) with A,B∈[B·H, N, N] (H = hidden_dim).  It is implemented with a
Triton kernel that tiles the i‑, j‑ and k‑axes (128×128×64) and uses
FP16 arithmetic with FP32 accumulation (Tensor‑Core friendly).  When the
sequence length N is a multiple of the tile sizes the kernel runs without
any masking, which removes the overhead of bounds checks.

The surrounding linear layers, layer‑norms and sigmoid gates are executed
with fast PyTorch primitives in FP16 (except the first LayerNorm, which is
kept in FP32 for numerical stability).  The final result is returned in
float32.

The implementation follows the reference PyTorch module:

    x_norm   = LayerNorm(x)
    fused    = Linear(x_norm, [left_proj, left_gate,
                               right_proj, right_gate,
                               out_gate])               # shape [...,5·H]
    left  = (fused[...,0:H]   * sigmoid(fused[...,H:2H]))   * mask
    right = (fused[...,2H:3H] * sigmoid(fused[...,3H:4H])) * mask
    out_gate = sigmoid(fused[...,4H:5H])

    out = batch_matmul_transpose(left, right)               # Triton
    out = LayerNorm(out)
    out = out * out_gate
    out = Linear(out, to_out.weight)                        # FP16 → FP32
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (batched over depth dimension)
#   A, B : [M, N, K]  (M = B*H, N = seq_len, K = seq_len) – fp16
#   C    : [M, N, N]  – fp16
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalars (K == N)
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bj, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,   # tile size along i (rows)
    BLOCK_N: tl.constexpr,   # tile size along j (cols)
    BLOCK_K: tl.constexpr,   # tile size along reduction k
    MASKED: tl.constexpr,    # compile‑time flag: need bounds checks?
):
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_z = tl.program_id(2)          # depth (batch * hidden)

    # Offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for the tails (only needed when MASKED=True)
    if MASKED:
        mask_i = offs_i < N
        mask_j = offs_j < N

    # Accumulator in FP32 (high‑precision reduction)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile  (BLOCK_M × BLOCK_K)
        a_ptrs = (A_ptr
                  + pid_z * stride_am
                  + offs_i[:, None] * stride_ai
                  + offs_k[None, :] * stride_ak)
        if MASKED:
            a = tl.load(a_ptrs,
                        mask=mask_i[:, None] & (offs_k[None, :] < K),
                        other=0.0)                # fp16
        else:
            a = tl.load(a_ptrs)                         # fp16

        # Load B tile, transposed (BLOCK_K × BLOCK_N)
        b_ptrs = (B_ptr
                  + pid_z * stride_bm
                  + offs_k[:, None] * stride_bk
                  + offs_j[None, :] * stride_bj)
        if MASKED:
            b = tl.load(b_ptrs,
                        mask=(offs_k[:, None] < K) & mask_j[None, :],
                        other=0.0)                # fp16
        else:
            b = tl.load(b_ptrs)                         # fp16

        # FP16 × FP16 → FP32 accumulate (Tensor‑Core friendly)
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + pid_z * stride_cm
              + offs_i[:, None] * stride_ci
              + offs_j[None, :] * stride_cj)
    if MASKED:
        mask = mask_i[:, None] & mask_j[None, :]
        tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
    else:
        tl.store(c_ptrs, acc.to(tl.float16))


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel (batched matmul A @ Bᵀ)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] – fp16, contiguous
    right: torch.Tensor,  # [B, N, N, H] – fp16, contiguous
) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.  The tensors must be on the same
    CUDA device and have dtype torch.float16.
    """
    B, N, _, H = left.shape
    # Pack (batch, hidden) into a single leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # Strides in element units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile size – tuned for H100 Tensor‑Core pipeline
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Determine whether we need bounds‑checking masks
    masked = (N % BLOCK_M != 0) or (N % BLOCK_N != 0) or (N % BLOCK_K != 0)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth (batch * hidden)
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,            # dummy M (not used inside the kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        MASKED=masked,
        num_warps=8,        # 256 threads per block (8 warps)
        num_stages=2,       # double‑buffering for latency hiding
    )

    # Restore original layout [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] with dtype float32 on the same device
        as the input.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # hidden dimension H
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, C]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy path
    # -------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fuse all linear projections (left/right + three gates)
    #    shape after linear : [B, N, N, 5*H]
    # -------------------------------------------------
    # Stack the five weight matrices (no bias)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,  # concatenate on the output‑feature axis
    ).to(torch.float16)  # (5*H, C)

    fused = torch.nn.functional.linear(x_fp16, fused_weight)  # [B, N, N, 5*H]

    # -------------------------------------------------
    # 4️⃣ Slice and apply sigmoids (gates)
    # -------------------------------------------------
    # Note: mask is applied after gating – order does not matter for a binary mask
    split = torch.split(fused, hidden_dim, dim=-1)
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = split

    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton
    # -------------------------------------------------
    # left/right are now [B, N, N, H] fp16
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # -------------------------------------------------
    # 8️⃣ Gate with out_gate and final projection back to dim
    # -------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (C, H)
    out = torch.nn.functional.linear(out, to_out_w)        # fp16, shape [B,N,N,C]

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (as required by harness)
    # -------------------------------------------------
    return out.to(torch.float32)
