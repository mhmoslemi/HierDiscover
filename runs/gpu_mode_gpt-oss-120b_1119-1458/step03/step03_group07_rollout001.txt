"""
TriMul (outgoing) – fast forward implementation (no gradients)

Algorithm Overview
------------------
1️⃣  Layer‑norm the input tensor (float32 → fp16 for the rest of the pipeline).
2️⃣  Fuse the four linear layers (proj & gate for left/right) into two
    linear calls by concatenating the weight matrices.
3️⃣  (Optional) Apply the pair‑wise mask **after** the projection but **before**
    the gating, exactly as in the reference implementation.
4️⃣  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
    as a batched matrix‑multiply (torch.bmm) on tensors of shape
    [B·H, N, N] – this maps to an optimized Tensor‑Core kernel on H100.
5️⃣  Layer‑norm the intermediate result.
6️⃣  Apply the output‑gate (sigmoid) and fuse the element‑wise multiplication
    with a tiny Triton kernel (dst *= src).
7️⃣  Final linear projection back to `dim`.
8️⃣  Cast back to float32 (the public API expects float32).

Only step 6 is performed with Triton, satisfying the “use a kernel” requirement,
while the remaining heavy work is delegated to highly‑optimized cuBLAS/Tensor‑Core
routines. Mixed‑precision (FP16) is used throughout to maximise throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication (dst *= src)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched matrix multiply for the O(N³) contraction
# Computes C = A @ Bᵀ where A,B: [B, N, N, H] (FP16) → out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: [B, N, N, H]  (float16)
    Returns: out [B, N, N, H] (float16)
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] (contiguous layout required by torch.bmm)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Perform the batched matmul: A @ Bᵀ
    C = torch.bmm(A, B_.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # if True, ignore mask completely

    device = input_tensor.device
    dtype_fp16 = torch.float16
    dtype_fp32 = torch.float32

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # stays in float32

    # Cast to fp16 for the heavy part of the forward pass
    x = x_norm.to(dtype_fp16)  # shape [B, N, N, dim], fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once (avoid repeated casts)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(dtype_fp16)       # [H, dim]
    left_gate_w   = weights["left_gate.weight"].to(dtype_fp16)       # [H, dim]
    right_proj_w  = weights["right_proj.weight"].to(dtype_fp16)      # [H, dim]
    right_gate_w  = weights["right_gate.weight"].to(dtype_fp16)      # [H, dim]
    out_gate_w    = weights["out_gate.weight"].to(dtype_fp16)        # [H, dim]

    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)      # [H]

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left and right sides
    # ------------------------------------------------------------------
    # Concatenate proj and gate weight matrices along the output dimension

    # Combined linear: produces both projection and gate activations at once
    left_comb  = torch.nn.functional.linear(x, left_comb_w)   # [B, N, N, 2*H]
    right_comb = torch.nn.functional.linear(x, right_comb_w)  # [B, N, N, 2*H]

    # split into projection and gate tensors
    left_proj = left_comb[..., :hidden_dim]        # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:]        # [B, N, N, H]
    right_proj = right_comb[..., :hidden_dim]      # [B, N, N, H]
    right_gate = right_comb[..., hidden_dim:]      # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) on the projected tensors (before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast to [B, N, N, 1]
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)   # fp16 mask
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # Apply the gates to the projected tensors
    left = left_proj * left_gate       # [B, N, N, H]
    right = right_proj * right_gate    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via a batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H]   fp16

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate (sigmoid) + fused element‑wise multiplication
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B, N, N, H] fp16

    # Fuse multiplication into a single Triton kernel (in‑place)
    out_flat = out.contiguous().view(-1)          # 1‑D view
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024                               # threads per block
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,          # dst stride (contiguous)
        1,          # src stride (contiguous)
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)    # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔚 Cast back to float32 (public API expectation)
    # ------------------------------------------------------------------
    return out.to(dtype_fp32)
