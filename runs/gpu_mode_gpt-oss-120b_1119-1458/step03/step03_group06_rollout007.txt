"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32).
2️⃣ Compute all linear maps (proj + gates) in FP16.
   - Optional binary mask is applied after the projections.
   - Gating (sigmoid) is fused with the projected tensors.
3️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply   left @ rightᵀ   with
   torch.bmm (cuBLAS tensor‑core path) on FP16 tensors.
4️⃣ Fuse the FP16→FP32 cast and the element‑wise multiplication with the
   output‑gate using a tiny Triton kernel (no extra memory traffic).
5️⃣ Layer‑norm (FP32), then final linear projection.
6️⃣ Return FP32 result.

Only the element‑wise cast‑and‑multiply kernel is required to “use Triton”.
The rest of the computation relies on highly‑optimised cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse FP16→FP32 cast and element‑wise multiply with
# a FP32 gate tensor.
# ----------------------------------------------------------------------
@triton.jit
def _fp16_mul_fp32_kernel(
    a_ptr,               # *fp16  input  (out_fp16)
    b_ptr,               # *fp32  gate   (out_gate_fp32)
    c_ptr,               # *fp32  output (out_fp32)
    size,                # i32 total number of elements
    stride_a: tl.constexpr,
    stride_b: tl.constexpr,
    stride_c: tl.constexpr,
    BLOCK: tl.constexpr, # tile size (elements per program)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)        # linear offset
    mask = offs < size

    # Load FP16 value, promote to FP32
    a = tl.load(a_ptr + offs * stride_a, mask=mask).to(tl.float32)
    # Load FP32 gate
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    # Multiply
    c = a * b
    tl.store(c_ptr + offs * stride_c, c, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device
        as the input.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Compute out_gate (float32) – required later
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float32)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, out_gate_w)
    )  # [B, N, N, hidden_dim] float32

    # -------------------------------------------------
    # 3️⃣ Cast to FP16 for heavy work
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Cast linear weights once to FP16
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]      # keep FP32 for LN
    to_out_norm_b = weights["to_out_norm.bias"]

    to_out_w = weights["to_out.weight"]                # keep FP32 for final Linear

    # -------------------------------------------------
    # 5️⃣ Linear projections (FP16)
    # -------------------------------------------------
    # shapes: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w)
    right = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------
    # 6️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 7️⃣ Gating (sigmoid) – still in FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via cuBLAS batched GEMM (FP16)
    # -------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to [B*hidden, N, N] for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    # i.e. left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 9️⃣ Fuse FP16→FP32 cast & multiplication by out_gate (Triton)
    # -------------------------------------------------
    # out_gate is FP32, out_fp16 is FP16
    out_fp32 = torch.empty_like(out_gate, dtype=torch.float32, device=device)

    total_elements = out_fp16.numel()
    BLOCK = 1024  # each program processes 1024 elements
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    _fp16_mul_fp32_kernel[
        grid
    ](
        out_fp16,
        out_gate,
        out_fp32,
        total_elements,
        1, 1, 1,                 # all tensors are 1‑D contiguous → stride = 1
        BLOCK=BLOCK,
        num_warps=4,
    )

    # -------------------------------------------------
    # 10️⃣ Second LayerNorm (float32)
    # -------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 11️⃣ Final linear projection back to original dim (float32)
    # -------------------------------------------------
    out = torch.nn.functional.linear(out_norm, to_out_w)   # [B, N, N, dim]

    # -------------------------------------------------
    # 12️⃣ Return FP32 result
    # -------------------------------------------------
    return out
