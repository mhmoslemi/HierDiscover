"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → float32 for stability).
2. Cast to fp16 and perform **one** fused linear projection that yields the
   two hidden‑dim projections (left/right) and the three gates
   (left_gate, right_gate, out_gate) in a single GEMM.  This avoids loading the
   large [B,N,N,dim] tensor five times, drastically cutting memory traffic.
3. Apply the (optional) mask to the two projected tensors, then fuse the
   element‑wise gates.
4. Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix‑multiply A @ Bᵀ with a custom Triton kernel that
   works on fp16 inputs and accumulates in FP32.  The kernel uses a
   128×128×64 tile and 8 warps with double‑buffering.
5. Second LayerNorm (fp16), multiply with the output‑gate and apply the final
   linear projection back to the original dimension.
6. Return the result in float32.

All heavy arithmetic stays in fp16 except the inner reduction, keeping the
runtime low on H100 while preserving numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A,B fp16, C fp16, acc FP32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]   (M = B*H, N = seq_len, K = seq_len)
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalars
    stride_am, stride_ai, stride_ak,   # strides of A (in elements)
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,   # tile rows
    BLOCK_N: tl.constexpr,   # tile cols
    BLOCK_K: tl.constexpr,   # reduction tile
):
    pid_i = tl.program_id(0)   # tile index along first seq‑len axis
    pid_j = tl.program_id(1)   # tile index along second seq‑len axis
    pid_z = tl.program_id(2)   # combined batch*hidden dimension

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (prevent overflow in long reductions)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # --------------------------------------------------------------
    # Main reduction loop over K
    # --------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile   (BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile transposed (BLOCK_K × BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Multiply‑accumulate (fp16 x fp16 → fp32)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # --------------------------------------------------------------
    # Write out the result
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] via the Triton kernel.
    The tensors must be contiguous fp16 on the same CUDA device.
    Returns a tensor of the same shape, fp16.
    """
    B, N, _, H = left.shape
    # --------------------------------------------------------------
    # Pack batch and hidden dims together → [B*H, N, N]
    # --------------------------------------------------------------
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[
        grid
    ](
        A,
        B_,
        C,
        B * H,               # dummy M (not used inside the kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Gather weight tensors (all bias‑less)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature dimension
    W_concat = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape [5*hidden_dim, dim]  fp16

    # Flatten spatial dimensions, perform a single linear
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)
    y_flat = torch.nn.functional.linear(x_flat, W_concat)  # (B*N*N, 5*hidden_dim)
    y = y_flat.view(B, N, N, 5 * hidden_dim)        # (B, N, N, 5*hidden)

    # Split projections & gate pre‑activations
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        y, hidden_dim, dim=3
    )  # each: (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Apply hidden‑dim gates (the mask is already merged)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 in/out)
    # ------------------------------------------------------------------
    # out shape: (B, N, N, hidden_dim)
    out = _batched_matmul_transpose_fp16(left, right)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm on hidden dimension (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gating & final projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)        # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
