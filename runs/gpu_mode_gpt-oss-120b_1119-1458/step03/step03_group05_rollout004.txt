"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1.  Layer‑norm on the input tensor (float32 for numerical stability).
2.  Cast to fp16 and apply the four linear projections (left/right and their
    gates).  All linear ops stay in fp16 (torch.nn.functional.linear).
3.  Optional mask is broadcast to the hidden dimension and applied to the
    projected tensors.
4.  Element‑wise sigmoid on the gating projections and fuse the gates:
    left  ← left  * sigmoid(left_gate)
    right ← right * sigmoid(right_gate)
5.  Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched GEMM C = A @ Bᵀ for each hidden slice.
    The kernel packs the (batch * hidden) dimension into the program‑id‑2
    axis, then tiles the two sequence axes with a 64×64×128 tile
    (M‑tile = rows i, N‑tile = cols j, K‑tile = reduction k).  Accumulation
    is performed in fp32, while the data stays in fp16, letting the H100
    Tensor‑Cores work at full speed.  Double‑buffered pipeline (num_stages=3)
    hides global‑memory latency.
6.  Second Layer‑norm on the hidden dimension (still fp16) and final gating.
7.  Linear projection back to the original feature dimension and cast to
    float32 for the output.

The implementation keeps the only memory‑intensive step (the cubic contraction)
inside a custom Triton kernel, while all lightweight ops remain in PyTorch.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (batched GEMM)
#   A : [M, N, K]  (M = batch * hidden, N = seq_len, K = seq_len)
#   B : [M, K, N]  (right already transposed)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # *fp16  [M, N, K]
    B_ptr,                     # *fp16  [M, K, N]   (right already transposed)
    C_ptr,                     # *fp16  [M, N, N]
    M, N, K,                   # i32 scalars
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bk, stride_bj,   # strides for B (note swapped K‑N order)
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,     # rows per block (i‑axis)
    BLOCK_N: tl.constexpr,     # cols per block (j‑axis)
    BLOCK_K: tl.constexpr,     # reduction tile size (k‑axis)
):
    pid_m = tl.program_id(0)          # i‑tile index
    pid_n = tl.program_id(1)          # j‑tile index
    pid_z = tl.program_id(2)          # batch*hidden index

    # ------------------------------------------------------------------
    # Tile offsets and masks
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator in FP32 (better numeric stability on H100 Tensor‑Cores)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over reduction dimension K in BLOCK_K‑sized chunks
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- Load A tile (M‑block, K‑tile) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---- Load B tile (K‑tile, N‑block) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # K‑dimension stride
            + offs_n[None, :] * stride_bj   # N‑dimension stride (post‑transpose)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ---- FP16×FP16 → FP32 accumulation (Tensor‑Core dot) ----
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write out C (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors, launches the Triton kernel and restores the
# original layout [B, N, N, hidden].
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]   (fp16)
    right: torch.Tensor,  # [B, N, N, H]   (fp16)
) -> torch.Tensor:
    """
    Computes, for every hidden slice h,
        out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]

    The contraction is performed as a batched GEMM C = A @ Bᵀ.
    """
    B, N, _, H = left.shape

    # Pack batch & hidden together → shape (M, N, N) where M = B*H
    #   A (left)  : [M, i, k]
    #   B (right) : [M, k, j]  (transposed)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides in **elements** (not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – empirically good on H100
    BLOCK_M = 64   # rows i
    BLOCK_N = 64   # cols j
    BLOCK_K = 128  # reduction k (twice the previous 64)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Restore the original [B, N, N, H] layout
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point for the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no grads).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once (weights are provided in float32)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; ensure fp16 and add channel dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates in‑place to keep memory traffic low
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Cubic contraction via the custom Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gating + projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
