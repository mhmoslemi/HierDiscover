"""
Outgoing TriMul (AlphaFold‑3) forward pass – Triton‑accelerated.
----------------------------------------------------------------
Algorithm
1. LayerNorm over the last channel (dim) in fp32 → cast to fp16.
2. Linear projections (left/right) and gating linear layers in fp16.
3. Optional mask multiplication (broadcast on the hidden dim).
4. Apply sigmoid gates (left_gate, right_gate) → element‑wise multiply.
5. Compute the O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM C = A @ Bᵀ where
        A = left .permute(0,3,1,2).view(B*H, N, N)   (i‑k)
        B = right.permute(0,3,2,1).view(B*H, N, N)  (k‑j)
   The Triton kernel tiles over (i, j) with a 64×64×64 tile, accumulates
   in fp32 and writes back fp16.
6. Second LayerNorm (hidden_dim) in fp16.
7. Multiply by out_gate (sigmoid) and final linear projection back to dim.
8. Cast the result to fp32 (required by the harness).

Both mask‑aware and mask‑free variants are supported.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : [M, N, K]   (M = B*H, N = seq_len, K = seq_len)
#   B : [M, K, N]   (right already transposed)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]   (right transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar dimensions
    stride_am, stride_ai, stride_ak,   # strides of A
    stride_bm, stride_bk, stride_bj,   # strides of B (note swapped K‑N)
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Tile indices
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (M, i, k)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load B tile (M, k, j) – already transposed
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k‑dim stride
            + offs_n[None, :] * stride_bj   # j‑dim stride (post‑transpose)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # FP16×FP16 → FP32 accumulation (Tensor‑Core friendly)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write output (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper: launch the batched GEMM with transpose on the right operand.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] fp16
    right: torch.Tensor,  # [B, N, N, H] fp16
) -> torch.Tensor:
    """
    Perform C = left @ rightᵀ for each (batch, hidden) pair.
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape

    # Pack batch & hidden together → shape [M, N, K] with M = B*H
    # left  : (i, k)   → [M, N, K]
    # right : (k, j)   → [M, K, N] (transpose the last two dims)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # k‑j

    # Output buffer (fp16)
    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides in elements
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tile sizes – empirically good on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        left_,
        right_,
        out,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,          # double‑buffered pipeline
    )

    # Restore original layout [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(

) -> torch.Tensor:
    """

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor[B, N, N]   (float32 or bool)
        - weights_dict : dict of model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → better stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy sub‑graph
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once (saved memory traffic)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask if required (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:

        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – built on the *unmasked* normalized input
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate       # fuse left gating
    right = right * right_gate    # fuse right gating

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via the Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden_dim) in fp16
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gating + linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                       # broadcast on hidden dim
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (required by downstream code)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
