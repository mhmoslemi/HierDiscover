"""
Trident TriMul (outgoing) – Triton‑accelerated forward pass
========================================================

The module receives a 4‑D tensor `x` of shape [B, N, N, D] (float32) and
an optional mask [B, N, N].  The forward pass consists of:

1️⃣ Layer‑norm over the feature dimension D (computed in FP32 for stability).
2️⃣ All linear projections and sigmoid gates are executed in FP16.
3️⃣ If a mask is present it is broadcast over the hidden dimension H.
4️⃣ Hidden‑dim gates (`left_gate`, `right_gate`) are fused into the projected
   tensors.
5️⃣ Heavy O(N³) contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = A @ Bᵀ with A and B of shape
   (B·H, N, N).  The custom Triton kernel tiles the multiplication with
   64 × 64 × 64 blocks, uses FP16 inputs and accumulates in FP32 via
   `tl.dot` (tensor‑core MMA), double‑buffered with `num_stages=2` and
   8 warps per program.
6️⃣ A second LayerNorm over the hidden dimension H.
7️⃣ Element‑wise output gate (`out_gate`) followed by a final linear
   projection back to the original dimension D.
8️⃣ Cast the result back to float32.

Only the O(N³) contraction is performed in Triton; the remaining
operations use highly‑optimized PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bi, stride_bk,   # strides for B (M,N,K)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication with the second operand transposed.
    The program IDs (pid_i, pid_j, pid_z) tile the output matrix (i,j)
    and the batch dimension (hidden·batch).
    """
    pid_i = tl.program_id(0)   # tile along output‑row (i) axis
    pid_j = tl.program_id(1)   # tile along output‑col (j) axis
    pid_z = tl.program_id(2)   # index over M = batch*hidden

    # -------------------------------------------------
    # Compute row / column offsets for this tile
    # -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator uses fp32 for numerical accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # -------------------------------------------------
    # Loop over K (the contraction dimension)
    # -------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load a tile from A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load a tile from Bᵀ: we load B indexed as (k, j) → shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k dimension first
            + offs_j[None, :] * stride_bi   # j dimension second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Tensor‑core MMA: fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Store the result tile (cast back to fp16)
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    where ``left`` and ``right`` are FP16 tensors of shape [B, N, N, H].

    Returns:
        Tensor of shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape

    # reshape to (M, N, K) where M = B*H, K = N
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    C = torch.empty_like(A, dtype=torch.float16)

    # strides in element units (triton expects i32)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bi, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # tile sizes – 64³ works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden tiles
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# Entrypoint required by the evaluation harness
# ----------------------------------------------------------------------

) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool, or None)
        - weights_dict : dict of model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as ``input_tensor``).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device

    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # Cast to FP16 for the remainder of the forward
    # -------------------------------------------------

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] in FP16

        left  = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (FP16 → FP16)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension (FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Output gating and final linear projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)    # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to FP32 (target dtype)
    # -------------------------------------------------
    return out.to(torch.float32)
