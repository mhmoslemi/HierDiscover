"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The module computes for each batch `b`, head `h` and pair of positions `(i,j)`:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are linear projections of the layer‑normed input
followed by element‑wise sigmoid gates (the masks are applied before the
gates).  The contraction is an O(N³) batched matrix‑multiply
`C = A @ Bᵀ` with `A`/`B` shaped `(B·H, N, N)`; `H` is the hidden
dimension (typically 128).  The heavy part is executed in a single
custom Triton kernel that:

* works directly on the original `[B, N, N, H]` layout (no
  expensive permute‑and‑copy),
* tiles the multiplication with 64×64×64 blocks,
* uses FP16 inputs and accumulates in FP32 via `tl.dot` (Tensor‑Core MMA),
* double‑buffers the K‑tiles (`num_stages=2`) and runs with 8 warps.

All other operations (layer‑norms, linear layers, sigmoids, masking) are
performed with high‑performance PyTorch primitives in FP16, while the
final result is returned in FP32.

The implementation works for the full range of test configurations:
    B ∈ {1,2}, N ∈ {128,256,512,768,1024},
    dim ∈ {128,384}, hidden_dim = 128,
    optional mask, and both normal / Cauchy input distributions.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A: left  [B, N, N, H]  (FP16)
#   B: right [B, N, N, H]  (FP16)
#   C: out   [B, N, N, H]  (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16   left tensor  (base pointer)
    B_ptr,               # *fp16   right tensor (base pointer)
    C_ptr,               # *fp16   output tensor (base pointer)

    # sizes
    BATCH: tl.constexpr,     # batch size (int)
    N: tl.constexpr,         # sequence length
    K: tl.constexpr,         # K dimension (= N)

    # strides of A (left) in *elements*
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,
    # strides of B (right)
    stride_Bb, stride_Bi, stride_Bk, stride_Bh,
    # strides of C (output)
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,

    HIDDEN_DIM: tl.constexpr,   # hidden dimension (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One Triton program processes a tile of shape (BLOCK_M, BLOCK_N) of the
    output matrix for a single (batch, hidden) pair, i.e. for a specific
    ``pid_z = b * HIDDEN_DIM + h``.
    """
    pid_i = tl.program_id(0)        # tile index along first N dimension (i)
    pid_j = tl.program_id(1)        # tile index along second N dimension (j)
    pid_z = tl.program_id(2)        # combined (batch, hidden) index

    # ------------------------------------------------------------------
    # Decode the combined index into batch and head dimension
    # ------------------------------------------------------------------
    b = pid_z // HIDDEN_DIM                 # batch idx
    h = pid_z - b * HIDDEN_DIM               # head idx (0 <= h < HIDDEN_DIM)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over K dimension (the shared dimension of the two matrices)
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ------------------------------------------------------------------
        # Base pointers for the current (batch, head) slice
        # ------------------------------------------------------------------
        a_base = A_ptr + b * stride_Ab + h * stride_Ah
        b_base = B_ptr + b * stride_Bb + h * stride_Bh
        c_base = C_ptr + b * stride_Cb + h * stride_Ch

        # ------------------------------------------------------------------
        # Load a tile of A = left[b,i,k,h]
        #   shape (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        a_ptrs = (
            a_base
            + offs_i[:, None] * stride_Ai
            + offs_k[None, :] * stride_Ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ------------------------------------------------------------------
        # Load a transposed tile of B = right[b,j,k,h]  (we need Bᵀ)
        #   shape (BLOCK_K, BLOCK_N)
        # ------------------------------------------------------------------
        b_ptrs = (
            b_base
            + offs_k[:, None] * stride_Bk
            + offs_j[None, :] * stride_Bi
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ------------------------------------------------------------------
        # Multiply‑accumulate (fp16·fp16 → fp32)
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_N) fp32

    # ------------------------------------------------------------------
    # Store the resulting tile to C
    # ------------------------------------------------------------------
    c_ptrs = (
        c_base
        + offs_i[:, None] * stride_Ci
        + offs_j[None, :] * stride_Cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel without any data copies
# ----------------------------------------------------------------------
def _trimul_batch_matmul_fp16(left: torch.Tensor, right: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Perform the heavy O(N³) contraction for the outgoing TriMul.

    Parameters
    ----------
    left, right : torch.Tensor
        Tensors of shape [B, N, N, H] with dtype=torch.float16.
        They are expected to be contiguous in the standard (B,N,N,H) layout.
    hidden_dim : int
        Size of the hidden feature dimension (H).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] (fp16) containing the result.
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2 and H == hidden_dim

    # ------------------------------------------------------------------
    # Output allocation
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Extract strides (in elements, not bytes)
    # ------------------------------------------------------------------
    sAb, sAi, sAk, sAh = left.stride()      # (B, N, N, H) → (M,i,k,h)
    sBb, sBi, sBk, sBh = right.stride()
    sCb, sCi, sCj, sCh = out.stride()      # same layout as left/right

    # ------------------------------------------------------------------
    # Tile sizes – 64×64×64 works well on H100 for fp16 MMA.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # ------------------------------------------------------------------
    # Grid definition
    # ------------------------------------------------------------------
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * hidden_dim,                        # (batch * head) tiles
    )

    # ------------------------------------------------------------------
    # Launch the kernel
    # ------------------------------------------------------------------
    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        BATCH=B,
        N=N,
        K=N,
        stride_Ab=sAb,
        stride_Ai=sAi,
        stride_Ak=sAk,
        stride_Ah=sAh,
        stride_Bb=sBb,
        stride_Bi=sBi,
        stride_Bk=sBk,
        stride_Bh=sBh,
        stride_Cb=sCb,
        stride_Ci=sCi,
        stride_Cj=sCj,
        stride_Ch=sCh,
        HIDDEN_DIM=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps = 256 threads per program
        num_stages=2,         # double‑buffering
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # -> [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the remainder of the pipeline
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1]
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left  = left  * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Compute hidden‑dimension gates (sigmoid) and apply them
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via custom Triton kernel
    # ------------------------------------------------------------------
    # The kernel works directly on the (B,N,N,H) layout – no data copy.
    out_hidden = _trimul_batch_matmul_fp16(left, right, hidden_dim)   # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (still FP16)
    # ------------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + projection back to the original dimension
    # ------------------------------------------------------------------
    out_hidden = out_hidden * out_gate                      # element‑wise gate (fp16)
    out = torch.nn.functional.linear(out_hidden, to_out_w)  # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (required output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
