def fused_left_right_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,

    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < size
    # Load from global memory
    lp = tl.load(left_proj_ptr + offsets, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offsets, mask=mask, other=0.0)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Layer‑norm on the input (FP32 → FP16).  
2️⃣ Fuse the five linear transforms (two projections, three gates) into a
   *single* GEMM by concatenating their weight matrices.  This avoids
   re‑reading the large (B·N²·dim) tensor five times and lets cuBLAS
   process a larger output matrix (5·hidden) for better occupancy.
3️⃣ Split the GEMM result into the five tensors, apply a sigmoid to the
   three gate tensors and fuse (proj * gate * mask) for the left/right
   sides with one Triton kernel (mask broadcasted over the hidden dim).
4️⃣ Core outgoing update:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matrix‑multiply over the hidden dimension
   using `torch.matmul` (strided‑batched GEMM – the fastest path on H100).
5️⃣ Layer‑norm over the hidden dimension, element‑wise multiply with the
   output gate and a final linear projection back to `dim`.  All heavy
   arithmetic stays in FP16; the result is cast back to the input dtype
   (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fused element‑wise product (proj * gate * mask) for both
# left and right tensors in a single launch.
# ----------------------------------------------------------------------
@triton.jit
def fused_left_right_mask_kernel(
    left_proj_ptr,        # left projection   (B,N,N,hidden)
    right_proj_ptr,       # right projection  (B,N,N,hidden)
    left_gate_ptr,        # left gate (sigmoid)   (B,N,N,hidden)
    right_gate_ptr,       # right gate (sigmoid)  (B,N,N,hidden)
    mask_ptr,             # mask (B,N,N) flattened
    left_out_ptr,         # left result    (B,N,N,hidden)
    right_out_ptr,        # right result   (B,N,N,hidden)
    hidden: tl.constexpr, # hidden dimension
    size:   tl.constexpr, # total number of elements (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # load the four tensors (projected value + gate) and the mask (broadcast)
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask index = element // hidden (broadcast over hidden)
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    tl.store(left_out_ptr + offs,  lp * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=mask)


def fused_left_right_mask(left_proj, right_proj, left_gate, right_gate, mask, hidden):
    """Apply (proj*gate*mask) for left and right tensors in one Triton launch."""
    B, N, _, _ = left_proj.shape
    total = left_proj.numel()
    out_left = torch.empty_like(left_proj)
    out_right = torch.empty_like(right_proj)
    BLOCK = 8192  # works well on H100

    # flatten mask for broadcasting
    mask_flat = mask.reshape(-1).contiguous()

    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_left_right_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_flat,
        out_left,
        out_right,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the **outgoing** TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float, CUDA)
        - weights_dict : mapping of weight names → torch.Tensor (FP32)
        - config_dict  : contains ``dim`` (int) and ``hidden_dim`` (int);
                         optional ``eps`` (float) and ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # 0) Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (keep in FP32 for statistics, then down‑cast)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # shape [B,N,N,dim] in fp16

    # ------------------------------------------------------------------
    # 2) One‑shot GEMM for all five linear layers
    # ------------------------------------------------------------------
    # reshape to (B*N*N, dim)
    x_flat = x_norm.view(-1, dim)                     # (M, dim), M = B*N*N
    # concatenate weight matrices vertically -> (5*hidden, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # (5*hidden, dim)

    # fused linear: (M, dim) @ (dim, 5*hidden).T  -> (M, 5*hidden)
    combined_out = F.linear(x_flat, combined_weight)  # fp16
    # reshape back to (B, N, N, 5*hidden)
    combined_out = combined_out.view(B, N, N, 5 * hidden)

    # ------------------------------------------------------------------
    # 3) Slice and apply gate sigmoid
    # ------------------------------------------------------------------
    h = hidden
    left_proj_raw = combined_out[..., 0:h]
    right_proj_raw = combined_out[..., h:2 * h]
    left_gate_raw = combined_out[..., 2 * h:3 * h]
    right_gate_raw = combined_out[..., 3 * h:4 * h]
    out_gate_raw = combined_out[..., 4 * h:5 * h]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)            # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Mask + gate fusion (or plain multiplication if mask omitted)
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = fused_left_right_mask(
            left_proj_raw,
            right_proj_raw,
            left_gate,
            right_gate,
            mask_fp16,
            hidden,
        )

    # ------------------------------------------------------------------
    # 5) Core outgoing contraction: batched matrix multiply per hidden channel
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # change layout to (B, hidden, N, N) for both tensors
    left_perm = left.permute(0, 3, 1, 2)    # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)  # (B, hidden, k, j)

    # batched GEMM: (B, hidden, i, k) @ (B, hidden, k, j) -> (B, hidden, i, j)
    out_hidden = torch.matmul(left_perm, right_perm)   # fp16

    # back to (B, N, N, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # multiply by the output gate
    out_gated = out_norm * out_gate

    # final linear projection back to `dim`
    out_final = F.linear(
        out_gated,
        weights["to_out.weight"].to(torch.float16),  # (dim, hidden)
    )

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype (float32) and return
    # ------------------------------------------------------------------
    return out_final.to(inp.dtype)
