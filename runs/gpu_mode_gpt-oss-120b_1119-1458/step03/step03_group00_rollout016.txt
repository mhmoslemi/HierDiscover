@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    gamma_ptr, beta_ptr,
    eps: tl.constexpr,
    C: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute offset for row
    offs_c = tl.arange(0, BLOCK_C)
    # mask for short C
    mask = offs_c < C
    # compute row base offset
    base = pid * C
    x = tl.load(x_ptr + base + offs_c, mask=mask, other=0.0)
    # compute mean
    m = tl.mean(x, axis=0) # or sum / C
    # compute variance
    diff = x - m
    var = tl.mean(diff * diff, axis=0)
    inv_std = 1.0 / tl.sqrt(var + eps)
    # apply gamma and beta
    gamma = tl.load(gamma_ptr + offs_c, mask=mask, other=0.0)
    beta = tl.load(beta_ptr + offs_c, mask=mask, other=0.0)
    y = (diff) * inv_std * gamma + beta
    tl.store(y_ptr + base + offs_c, y, mask=mask)

def layernorm_gate_kernel(
    out_ptr, out_gate_ptr,
    gamma_ptr, beta_ptr,

@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    eps: tl.constexpr,
    C: tl.constexpr,
):
    pid = tl.program_id(0) # each row
    # Compute offset start for row
    base = pid * C
    offs = tl.arange(0, C)   # exactly C threads
    x = tl.load(inp_ptr + base + offs, other=0.0)
    # Compute mean
    m = tl.sum(x) / C
    diff = x - m
    var = tl.sum(diff * diff) / C
    inv_std = 1.0 / tl.sqrt(var + eps)
    gamma = tl.load(weight_ptr + offs, other=0.0)
    beta = tl.load(bias_ptr + offs, other=0.0)
    y = diff * inv_std * gamma + beta
    tl.store(out_ptr + base + offs, y)
```

@triton.jit
def fused_proj_gate_mask_kernel(
    left_proj_ptr,
    right_proj_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    out_left_ptr,
    out_right_ptr,
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    left = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_g = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # compute index for mask (broadcast over hidden);
    mask_idx = offs // hidden
    b = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out_l = left * left_g * b
    out_r = right * right_g * b

    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)
```

"""
TriMul (outgoing) – fused Triton implementation
================================================

Algorithm
---------
1️⃣ Input LayerNorm (over the last channel dimension) is performed by a
   custom Triton kernel.  All operations stay in FP16 after the
   initial FP32→FP16 cast.

2️⃣ The five linear layers (left_proj, right_proj, left_gate, right_gate,
   out_gate) are *fused* into a single matrix multiplication.  The
   weight matrices are concatenated (shape (5·H, C)) and applied to the
   flattened input tensor, producing one large tensor that is split into
   the five required parts.

3️⃣ When a mask is present a second Triton kernel computes the element‑wise
   product ``proj * gate * mask`` for the left and right branches in a
   single pass, avoiding an extra load of the mask.

4️⃣ The core contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is a batched GEMM over the hidden dimension.  Each hidden slice is a
   matrix multiply of shape (N,N) and is executed with cuBLAS’s
   strided‑batched GEMM via ``torch.bmm`` – this is the fastest way to
   realise the O(N³) operation on an H100.

5️⃣ A fused Triton kernel performs the output LayerNorm **and** the
   output gate multiplication in one pass, again staying in FP16.

6️⃣ The final linear projection (hidden → C) is a regular bias‑free GEMM.
   The result is cast back to the original dtype (usually float32).

Only the element‑wise mask/gate fusion and the two LayerNorms are custom
Triton kernels; the heavy matrix multiplications use highly‑optimised
cuBLAS kernels.  This reduces the memory‑traffic from five separate
linear layers to a single GEMM and removes an extra kernel launch for
masking, achieving a large speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: input LayerNorm (over the last channel dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    inp_ptr,               # input   [P, C]
    out_ptr,               # output  [P, C]
    weight_ptr,            # gamma   [C]
    bias_ptr,              # beta    [C]
    eps: tl.constexpr,
    C: tl.constexpr,       # channel dimension (compile‑time)
    BLOCK_SIZE: tl.constexpr,
):
    # one program per position (P = B·N·N)
    pid = tl.program_id(0)
    base = pid * C                        # start of the row
    offs = tl.arange(0, BLOCK_SIZE)       # thread offsets within row
    mask = offs < C                       # guard for possible padding

    # ------------------------------------------------------------------
    # Load the row (FP16) and promote to FP32 for numerics
    # ------------------------------------------------------------------
    x = tl.load(inp_ptr + base + offs, mask=mask, other=0.0)
    x_f = tl.cast(x, tl.float32)

    # ------------------------------------------------------------------
    # Compute mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_x = tl.sum(x_f, axis=0)
    mean = sum_x / C
    diff = x_f - mean
    var = tl.sum(diff * diff, axis=0) / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Load layer‑norm parameters (FP16 -> FP32)
    # ------------------------------------------------------------------
    gamma = tl.load(weight_ptr + offs, mask=mask, other=0.0)
    beta = tl.load(bias_ptr + offs, mask=mask, other=0.0)
    gamma_f = tl.cast(gamma, tl.float32)
    beta_f = tl.cast(beta, tl.float32)

    # ------------------------------------------------------------------
    # Normalise → scale + shift, then cast back to FP16
    # ------------------------------------------------------------------
    y_f = diff * inv_std * gamma_f + beta_f
    y = tl.cast(y_f, tl.float16)

    tl.store(out_ptr + base + offs, y, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float) -> torch.Tensor:
    """
    Triton‑based LayerNorm over the last dimension.
    x must be contiguous, dtype torch.float16.
    Returns tensor of the same shape and dtype.
    """
    B, N, N2, C = x.shape
    assert N == N2
    total_rows = B * N * N
    out = torch.empty_like(x)

    # Compile‑time block size equals C (C≤384 in all test configs)
    grid = (total_rows,)
    layernorm_kernel[grid](
        x, out,
        weight, bias,
        eps,
        C,
        BLOCK_SIZE=C,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcasted over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_mask_kernel(
    left_proj_ptr,     # left projection   (P·H)
    right_proj_ptr,    # right projection  (P·H)
    left_gate_ptr,     # left gate         (P·H)
    right_gate_ptr,    # right gate        (P·H)
    mask_ptr,          # mask (P)
    out_left_ptr,      # output left       (P·H)
    out_right_ptr,     # output right      (P·H)
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total

    # load tensors (FP16)
    left = tl.load(left_proj_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    left_g = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_g = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    # mask is broadcast over hidden, compute its index
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)

    out_l = left * left_g * m
    out_r = right * right_g * m

    tl.store(out_left_ptr + offs, out_l, mask=active)
    tl.store(out_right_ptr + offs, out_r, mask=active)


# ----------------------------------------------------------------------
# Triton kernel: output LayerNorm + output gate (combined)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    inp_ptr,           # input tensor (P·H)
    gate_ptr,          # output gate (P·H)
    out_ptr,           # destination (P·H)
    weight_ptr,        # LN weight (H)
    bias_ptr,          # LN bias   (H)
    eps: tl.constexpr,
    H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    base = pid * H
    offs = tl.arange(0, BLOCK_SIZE)
    mask = offs < H

    # Load data (FP16) and promote to FP32 for reduction
    x = tl.load(inp_ptr + base + offs, mask=mask, other=0.0)
    x_f = tl.cast(x, tl.float32)

    sum_x = tl.sum(x_f, axis=0)
    mean = sum_x / H
    diff = x_f - mean
    var = tl.sum(diff * diff, axis=0) / H
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Load LN parameters (FP16 → FP32)
    gamma = tl.load(weight_ptr + offs, mask=mask, other=0.0)
    beta = tl.load(bias_ptr + offs, mask=mask, other=0.0)
    gamma_f = tl.cast(gamma, tl.float32)
    beta_f = tl.cast(beta, tl.float32)

    # Normalise
    y_f = diff * inv_std * gamma_f + beta_f

    # Apply output gate (gate is already FP16 → promote)
    gate = tl.load(gate_ptr + base + offs, mask=mask, other=0.0)
    gate_f = tl.cast(gate, tl.float32)

    y_f = y_f * gate_f

    # Cast back to FP16 and store
    y = tl.cast(y_f, tl.float16)
    tl.store(out_ptr + base + offs, y, mask=mask)


def layernorm_gate_triton(x: torch.Tensor,
                         gate: torch.Tensor,
                         weight: torch.Tensor,
                         bias: torch.Tensor,
                         eps: float) -> torch.Tensor:
    """
    LayerNorm over the hidden dimension followed by an element‑wise
    multiplication with the output gate.  All tensors are FP16.
    """
    B, N, N2, H = x.shape
    assert N == N2
    total_positions = B * N * N
    out = torch.empty_like(x)

    grid = (total_positions,)
    layernorm_gate_kernel[grid](
        x, gate, out,
        weight, bias,
        eps,
        H,
        BLOCK_SIZE=H,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C]  (float32, CUDA)
            - mask_tensor : [B, N, N]    (bool/float, CUDA) – may be empty
            - weights_dict: model weights (see skeleton)
            - config_dict : contains at least "dim", "hidden_dim" and
                            optional "eps", "nomask".

    Returns:
        Tensor of shape [B, N, N, C] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack and cast
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dtype_in = inp.dtype
    device = inp.device

    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    x_fp16 = inp.to(torch.float16)
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    x_norm = layernorm_triton(x_fp16, norm_weight, norm_bias, eps)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into one GEMM
    # ------------------------------------------------------------------
    # flatten [B,N,N,dim] → [B·N·N, dim]
    x_flat = x_norm.reshape(-1, dim)   # (P, dim)  where P = B·N·N

    # concat the five weight matrices (all shape [hidden, dim])
    w_left    = weights["left_proj.weight"].to(torch.float16)
    w_right   = weights["right_proj.weight"].to(torch.float16)
    w_lgate   = weights["left_gate.weight"].to(torch.float16)
    w_rgate   = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)

    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)  # (5*H, dim)

    # single linear projection (bias=False)
    linear_out = F.linear(x_flat, w_concat)          # (P, 5*H)

    # reshape back to [B,N,N,5*H] and split
    linear_out = linear_out.view(B, N, N, hidden * 5)
    left_proj_raw = linear_out[..., 0:hidden]
    right_proj_raw = linear_out[..., hidden:2*hidden]
    left_gate_raw = linear_out[..., 2*hidden:3*hidden]
    right_gate_raw = linear_out[..., 3*hidden:4*hidden]
    out_gate_raw = linear_out[..., 4*hidden:5*hidden]

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)   # shape (B,N,N,hidden)

    # --------------------------------------------------------------
    # 4️⃣ Masked projection + gating (fused kernel when a mask exists)
    # --------------------------------------------------------------
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # ensure mask is FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        total = left_proj_raw.numel()                 # B·N·N·hidden
        BLOCK = 8192                                   # tuned for H100
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        fused_proj_gate_mask_kernel[grid](
            left_proj_raw, right_proj_raw,
            left_gate, right_gate,
            mask_fp16,
            left, right,
            hidden,
            total,
            BLOCK_SIZE=BLOCK,
        )

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction: batched GEMM over hidden dimension
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    left_mat = left_perm.view(B * hidden, N, N)   # (B*H, i, k)
    right_mat = right_perm.view(B * hidden, N, N) # (B*H, k, j)

    out_hidden = torch.bmm(left_mat, right_mat)   # (B*H, i, j)

    # restore to (B, N, N, hidden)
    out = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + output gate (fused Triton kernel)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    out = layernorm_gate_triton(
        out, out_gate,
        to_out_norm_weight, to_out_norm_bias,
        eps,
    )  # shape (B,N,N,hidden), still FP16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)

    out_flat = out.view(-1, hidden)        # (B·N·N, hidden)
    final = F.linear(out_flat, to_out_weight)  # (B·N·N, dim)
    final = final.view(B, N, N, dim)

    # Cast back to original dtype (usually float32)
    return final.to(dtype_in)
