@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    N_ROWS, C,
    stride_in_row, stride_in_col,
    stride_out_row, stride_out_col,
    eps: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    row = tl.program_id(0)
    if row >= N_ROWS:
        return
    col = tl.arange(0, BLOCK_SIZE)
    mask = col < C

    inp_offset = inp_ptr + row * stride_in_row + col * stride_in_col
    x = tl.load(inp_offset, mask=mask, other=0.0)
    # cast to fp32 for accumulation
    x_fp32 = x.to(tl.float32)

    sum_val = tl.sum(x_fp32, axis=0)

"""
TriMul (outgoing) – fast forward pass for AlphaFold‑3 (no gradients).

Algorithm (FP16‑only except the very first LayerNorm which stays FP32 for
stability):
1️⃣ Layer‑norm the input tensor.
2️⃣ Cast to FP16 and fuse each “proj + gate” pair of Linear layers into a
   single Linear (cat‑weight → 2 × hidden_dim outputs) → split.
3️⃣ (Optional) Apply the pair‑wise mask to the projected tensors.
4️⃣ Apply the sigmoid gates and obtain the gated tensors *left* and *right*.
5️⃣ Heavy O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   computed with a single `torch.einsum` (no costly permute‑/contiguous‑copy).
6️⃣ Second LayerNorm over the hidden dimension.
7️⃣ Compute the output‑gate (linear + sigmoid) and fuse the element‑wise
   multiplication `out *= out_gate` with a tiny Triton kernel.
8️⃣ Final linear projection back to the original channel dimension.
9️⃣ Cast the result back to float32 (the public API type).

Only the element‑wise out‑gate multiplication is performed with Triton,
fulfilling the “kernel‑required” part while the dominant compute (the
contraction) uses the highly‑optimized Tensor‑Core GEMM implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication out *= src  (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    # Load, multiply and store
    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)
    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32, bool or None)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    x = x_norm.to(torch.float16)  # work in FP16 from now on

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left and right (2 × Linear instead of 4)
    # ------------------------------------------------------------------
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # Linear projection + gate in one go (produces [B,N,N,2*H])
    left_comb   = torch.nn.functional.linear(x, left_comb_w)      # fp16
    right_comb  = torch.nn.functional.linear(x, right_comb_w)     # fp16

    # Split back into projection and gate tensors
    left_proj = left_comb[..., :hidden_dim]          # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:]          # [B,N,N,H]

    right_proj = right_comb[..., :hidden_dim]        # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:]        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim) – applied BEFORE gating
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask assumed to be 0/1 (float or bool)
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still FP16) and fuse
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left  = left_proj * left_gate          # [B,N,N,H]
    right = right_proj * right_gate        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via a single fused einsum (no data copy)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # [B,N,N,H]  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate (sigmoid) and fused element‑wise multiplication
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] fp16

    # Flatten for Triton kernel (in‑place dst ← dst * src)
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024  # max threads per block on H100
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,          # dst stride
        1,          # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 🔚 Cast back to float32 (public API)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
