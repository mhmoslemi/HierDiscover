```python
@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr,          # pointer to projection values (B,N,N,hidden)
    gate_ptr,          # pointer to gate values (B,N,N,hidden)
    mask_ptr,          # pointer to mask (B,N,N)
    out_ptr,           # output pointer
    hidden: tl.constexpr,
    size: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_off = offs < size

    # load projection and gate
    proj = tl.load(proj_ptr + offs, mask=mask_off, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_off, other=0.0)

    # compute mask index (broadcasted over hidden)
    # Equivalent to floor division by hidden
    mask_idx = offs // hidden
    b = tl.load(mask_ptr + mask_idx, mask=mask_off, other=0.0)

    # compute sigmoid gate
    # sigmoid(x) = 1 / (1 + exp(-x))
    sigmoid_gate = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * sigmoid_gate * b
    tl.store(out_ptr + offs, out, mask=mask_off)
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – keep the heavy arithmetic in fp16.

2️⃣ Fuse the *projection* and *gate* linear layers:
   left_comb = Linear(x, [left_proj; left_gate])  → shape (...,2*H)
   right_comb = Linear(x, [right_proj; right_gate]) → shape (...,2*H)
   The two halves are split into projection and gate tensors.

3️⃣ Fuse *gate sigmoid* and optional *mask* with the projection
   using a custom Triton kernel:
       out = proj * sigmoid(gate) * mask
   This eliminates materialising the sigmoid-ed gate and the broadcast
   of the mask, cutting memory traffic by ~⅓.

4️⃣ Core contraction (outgoing update):
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   By permuting to (B,H,N,N) we can call ``torch.matmul`` once,
   which uses cuBLAS’s strided‑batched GEMM (very fast on H100).

5️⃣ Output LayerNorm → multiply by the output gate → final linear
   projection back to the original dimension.

Only the element‑wise “proj·gate·mask” part is written in Triton; the
rest leverages highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr,            # pointer to projection tensor (B,N,N,H)
    gate_ptr,            # pointer to gate pre‑sigmoid tensor (B,N,N,H)
    mask_ptr,            # pointer to mask tensor (B,N,N) – broadcast over H
    out_ptr,             # destination tensor (B,N,N,H)
    H: tl.constexpr,     # hidden dimension (compile‑time constant)
    SIZE: tl.constexpr,  # total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < SIZE

    # load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # broadcasted mask index (each H entries share one mask entry)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # sigmoid gate = 1 / (1 + exp(-x))
    sigmoid_gate = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * sigmoid_gate * m
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def fused_proj_gate_kernel(
    proj_ptr,            # projection tensor (B,N,N,H)
    gate_ptr,            # gate pre‑sigmoid tensor (B,N,N,H)
    out_ptr,             # destination tensor (B,N,N,H)
    H: tl.constexpr,
    SIZE: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < SIZE

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    sigmoid_gate = 1.0 / (1.0 + tl.exp(-gate))
    out = proj * sigmoid_gate
    tl.store(out_ptr + offs, out, mask=mask)


def fused_proj_gate_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         hidden: int) -> torch.Tensor:
    """
    Apply ``proj * sigmoid(gate) * mask`` element‑wise.
    All tensors must be contiguous, half‑precision and have matching shapes:
        proj, gate : (B, N, N, hidden)
        mask       : (B, N, N)
    """
    assert proj.shape == gate.shape
    B, N, _, H = proj.shape
    assert H == hidden, "hidden dimension mismatch"

    proj = proj.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()

    out = torch.empty_like(proj)
    total = proj.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


def fused_proj_gate(proj: torch.Tensor,
                    gate: torch.Tensor,
                    hidden: int) -> torch.Tensor:
    """
    Apply ``proj * sigmoid(gate)`` element‑wise (no mask).
    """
    assert proj.shape == gate.shape
    B, N, _, H = proj.shape
    assert H == hidden

    proj = proj.contiguous()
    gate = gate.contiguous()
    out = torch.empty_like(proj)
    total = proj.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_proj_gate_kernel[grid](
        proj,
        gate,
        out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation (no gradients).
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
            mask_tensor  : torch.Tensor [B, N, N]        (bool/float, CUDA)
            weights_dict : model weights (see skeleton)
            config_dict  : contains at least ``dim`` and ``hidden_dim``.
                           May also contain ``eps`` and ``nomask``.
    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep FP32 for stability, then cast to fp16)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Fuse projection & gate linear layers (2 * hidden -> hidden)
    # --------------------------------------------------------------
    # Left side: concat(proj, gate) weight matrices
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)          # shape (2*hidden, dim)

    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # shape (2*hidden, dim)

    # Combined linear transformations
    left_comb = F.linear(x, left_comb_weight)    # (B, N, N, 2*hidden)
    right_comb = F.linear(x, right_comb_weight)  # (B, N, N, 2*hidden)

    # Split into projection and gate sub‑tensors (views, no copy)
    left_proj_raw = left_comb[..., :hidden]   # (B, N, N, hidden)
    left_gate_raw = left_comb[..., hidden:]   # (B, N, N, hidden)

    right_proj_raw = right_comb[..., :hidden]
    right_gate_raw = right_comb[..., hidden:]

    # Output gate (separate linear)
    out_gate_raw = F.linear(x, weights["out_gate.weight"].to(torch.float16))
    out_gate = torch.sigmoid(out_gate_raw)    # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 3) Apply mask (if any) and sigmoid‑gate in a fused Triton kernel
    # --------------------------------------------------------------
    if nomask:
        # No mask → only proj·sigmoid(gate)
        left = fused_proj_gate(left_proj_raw, left_gate_raw, hidden)
        right = fused_proj_gate(right_proj_raw, right_gate_raw, hidden)
    else:
        # Cast mask once to fp16 for the Triton kernel
        mask_fp = mask.to(torch.float16)
        left = fused_proj_gate_mask(left_proj_raw, left_gate_raw, mask_fp, hidden)
        right = fused_proj_gate_mask(right_proj_raw, right_gate_raw, mask_fp, hidden)

    # --------------------------------------------------------------
    # 4) Core outgoing contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM over the hidden dimension.
    # --------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N) for both tensors
    left_perm = left.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)  # (B, hidden, k, j)

    # Batched matrix multiplication (cuBLAS strided‑batched GEMM)
    out_hidden = torch.matmul(left_perm, right_perm)  # (B, hidden, i, j)

    # Back to original layout
    out = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection back to dim
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # Multiply by the output gate (already sigmoid-ed)
    out_norm = out_norm * out_gate

    # Final linear projection back to original dimensionality
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),
    )  # (B, N, N, dim)

    # Cast back to the original dtype (usually float32) before returning
    return out_final.to(inp.dtype)
