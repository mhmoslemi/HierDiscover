"""
Outgoing TriMul (AlphaFold‑3) – high‑performance forward pass.

The heavy O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix‑multiply C = A @ Bᵀ using a
Triton kernel.  Tiles of size 128×128×32 are processed with Tensor‑Core
MMA (fp16 × fp16 → fp32 accumulator).  This reduces the memory traffic
by a factor 2 compared to the previous 64‑tile version and brings the
kernel close to peak H100 performance.

All other operations (LayerNorm, linear projections, gates, mask) are
performed with native PyTorch kernels in fp16.  The linear projection
and its gate are fused into a single matmul to halve the number of
matrix‑vector products.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output (float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B,C are fp16, accumulator fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 dimensions
    stride_am, stride_ai, stride_ak,   # A strides (M‑axis, i‑axis, k‑axis)
    stride_bm, stride_bj, stride_bk,   # B strides (M‑axis, j‑axis, k‑axis)
    stride_cm, stride_ci, stride_cj,   # C strides (M‑axis, i‑axis, j‑axis)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile index along the first N‑axis (i)
    pid_j = tl.program_id(1)   # tile index along the second N‑axis (j)
    pid_z = tl.program_id(2)   # flattened batch × hidden index (M)

    # -----------------------------------------------------------------
    # Offsets for the current tile
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (higher precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # -------------------------------------------------------------
        # Load tile A: shape (BLOCK_M, BLOCK_K) → A[m, i, k]
        # -------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # -------------------------------------------------------------
        # Load tile B (transposed): shape (BLOCK_K, BLOCK_N) → B[m, j, k]
        # -------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision MMA: fp16 × fp16 ⇒ fp32 accumulation
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write back the result (fp16)
    # -----------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors, launches the Triton kernel and restores shape.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a
    Triton kernel.  ``left`` and ``right`` must be fp16, contiguous and
    have shape [B, N, N, H] (H = hidden_dim).  Returns a tensor of the
    same shape and dtype.
    """
    B, N, _, H = left.shape

    # Pack batch and hidden into a single leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16, same layout as A)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile sizes – 128×128×32 is a sweet spot for H100 fp16 Tensor‑Cores.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to the original 4‑D layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights_dict : {name: torch.Tensor (float32)}
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣  First LayerNorm (float32 for numeric stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast all linear weights to fp16 once
    # -------------------------------------------------
    # Fuse projection + its gate into a single matmul (2× hidden → projection & gate)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    left_combined_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections + gating (fp16)
    # -------------------------------------------------
    # left: [B,N,N, 2*H] → split into proj & gate
    left_combined = torch.nn.functional.linear(x, left_combined_w)   # [B,N,N,2*H]
    left_proj = left_combined[..., :hidden_dim]                     # [B,N,N,H]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])     # [B,N,N,H]

    # right: same pattern
    right_combined = torch.nn.functional.linear(x, right_combined_w)
    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left_proj * mask_fp16 * left_gate
        right = right_proj * mask_fp16 * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 → fp16)
    # -------------------------------------------------
    # Ensure tensors are contiguous before the reshape required by the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) + out‑gate + final projection
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] fp16

    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
