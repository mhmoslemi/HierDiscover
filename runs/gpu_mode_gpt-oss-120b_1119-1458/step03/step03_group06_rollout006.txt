"""
Optimized outgoing TriMul implementation (AlphaFold‑3).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input in FP32 (numerical stability).
2️⃣ Cast the normalized tensor to FP16.
3️⃣ Fuse the four linear projections (left/right + their gates) into a single
   matrix‑multiply by concatenating the four weight matrices → one call to
   `torch.nn.functional.linear`.  The result is split into left_proj,
   right_proj, left_gate_logit, right_gate_logit.
4️⃣ Apply sigmoid to the gate logits and the optional mask (broadcasted on the
   hidden dimension) → `left = left_proj * left_gate * mask`,
   `right = right_proj * right_gate * mask`.
5️⃣ Compute the heavy O(N³) contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   with a custom Triton kernel that performs a batched GEMM
   `C = A @ Bᵀ` on tensors of shape `[B·H, N, N]`.  The kernel tiles the
   i‑j plane with 128 × 128 blocks and the reduction axis k with 64,
   uses FP16 loads, FP32 accumulation (tensor‑core friendly) and double‑
   buffering (`num_stages=2`).  8 warps (256 threads) give high occupancy
   on an H100.
6️⃣ Second Layer‑norm (still in FP16) on the contraction result.
7️⃣ Compute the output gate `out_gate = sigmoid(out_gate_linear(x))`,
   apply it element‑wise and project back to the original dimension with a
   final FP16 linear layer.
8️⃣ Cast the result to FP32 (the harness expects float32).

Only the O(N³) part is written in Triton; the surrounding lightweight
operations stay in PyTorch.  This layout reduces the number of separate
linear calls (four → two) and keeps all heavy tensors in FP16, dramatically
cutting memory traffic and runtime.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply C = A @ Bᵀ
#   A, B : [M, N, K]  (M = B·H, N = seq_len, K = seq_len)   fp16
#   C    : [M, N, N]  fp16
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalars (M is unused, kept for API compatibility)
    stride_am, stride_ai, stride_ak,   # strides of A
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,   # tile size along the i axis
    BLOCK_N: tl.constexpr,   # tile size along the j axis
    BLOCK_K: tl.constexpr,   # tile size along the reduction axis k
):
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # combined (batch * hidden) index

    # --------------------------------------------------------------
    # Offsets for this tile (including boundary masks)
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 (more stable for long reductions)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # --------------------------------------------------------------
    # Main k‑reduction loop
    # --------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i,k] tile   (shape: BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B[k,j] tile   (shape: BLOCK_K × BLOCK_N) – this is the transpose
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk          # k‑dimension (inner)
            + offs_j[None, :] * stride_bj          # j‑dimension (outer)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # FP16 × FP16 → FP32 accumulate (tensor‑core friendly)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # --------------------------------------------------------------
    # Write the result back (FP32 → FP16)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]   fp16, contiguous
    right: torch.Tensor,  # [B, N, N, H]   fp16, contiguous
) -> torch.Tensor:
    """
    Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.  Both inputs must be contiguous fp16
    tensors on the same CUDA device.
    """
    B, N, _, H = left.shape                # left.shape == right.shape
    # Pack batch and hidden into a single leading dimension: [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element‑units (required by Triton)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tuning parameters (empirically good on H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth dimension (batch * hidden)
    )

    _trimul_matmul_kernel[
        grid
    ](
        A,
        B_,
        C,
        B * H,          # dummy M (not used inside kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 256 threads per block
        num_stages=2,       # double‑buffering hides global‑mem latency
    )

    # Reshape back to the original [B, N, N, H] layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entrypoint expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 / bool)
        - weights_dict : mapping from weight names to float32 tensors
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optionally
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast to fp16 – all heavy work now stays in half‑precision
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Stack the four projection / gate weights into one matrix
    # -------------------------------------------------
    # All weights are bias‑free linear layers
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    stacked_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )  # shape [4*H, dim], fp16

    # -------------------------------------------------
    # 3️⃣ One fused linear that yields all 4 tensors at once
    # -------------------------------------------------
    # Result shape: [B, N, N, 4*H] (contiguous)
    fused = torch.nn.functional.linear(x, stacked_w)  # fp16

    # Split the concatenated dimension
    split_tensors = torch.split(fused, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_logit, right_gate_logit = split_tensors

    # -------------------------------------------------
    # 4️⃣ Compute gates (sigmoid) – still fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)

    # Apply gates (and optional mask) to the projected tensors
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via the Triton kernel
    # -------------------------------------------------
    # out has shape [B, N, N, H] (fp16)
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # -------------------------------------------------
    # 7️⃣ Output gate (computed from the same normalized input)
    # -------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x,
            weights["out_gate.weight"].to(torch.float16),
        )
    )  # [B,N,N,H] fp16

    out = out * out_gate

    # -------------------------------------------------
    # 8️⃣ Final projection back to the original dimension
    # -------------------------------------------------
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
    )  # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
