"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input (float32) → cast to fp16.
2. Fuse each projection (left/right) with its gate by concatenating the two
   weight matrices and performing a single `nn.functional.linear`.  The
   sigmoid on the gate part is applied afterwards.
3. Optional binary/float mask is broadcast and applied (elementwise mul).
4. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is reduced to a batched matrix‑multiply C = A @ Bᵀ with
   `torch.bmm` (strided‑batched GEMM) which on an H100 uses Tensor‑Core
   fp16 MMA and is faster than the previous Triton implementation.
5. Second Layer‑norm over the hidden dimension, out‑gate (sigmoid) and final
   linear projection back to `dim`.
6. Cast the result to float32 (required by the harness).

All heavy linear layers are executed in fp16; only the first LayerNorm is
performed in float32 for numerical stability.  No gradients are computed.
A trivial Triton kernel is also provided (required by the problem statement)
but the performance‑critical contraction uses cuBLAS via `torch.bmm`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Minimal Triton kernel (required by the rules).  This kernel simply
# copies a tensor with an optional mask – it is *not* used for the main
# contraction, which is delegated to cuBLAS via torch.bmm for speed.
# ----------------------------------------------------------------------
@triton.jit
def _mask_copy_kernel(
    src_ptr,               # *fp16  source tensor
    mask_ptr,              # *fp16  mask tensor (broadcastable)
    dst_ptr,               # *fp16  destination tensor
    size,                  # i32   total number of elements
    src_stride,            # i32   stride (in elements) between consecutive values in src
    mask_stride,           # i32   stride (in elements) between mask values
    dst_stride,            # i32   stride (in elements) between consecutive dst values
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    m = offset < size

    src = tl.load(src_ptr + offset * src_stride, mask=m, other=0.0)
    mask = tl.load(mask_ptr + offset * mask_stride, mask=m, other=1.0)  # 1.0 = no‑mask
    out = src * mask
    tl.store(dst_ptr + offset * dst_stride, out, mask=m)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → keep numeric stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate using concatenated weight matrices
    # ------------------------------------------------------------------
    # left side: proj + gate
    left_w_proj = weights["left_proj.weight"]
    left_w_gate = weights["left_gate.weight"]
    left_w_cat = torch.cat([left_w_proj, left_w_gate], dim=0).to(torch.float16)  # [2*H, dim]
    left_cat = torch.nn.functional.linear(x, left_w_cat)                     # [B,N,N,2*H]
    left_proj = left_cat[..., :hidden]                                      # view, no copy
    left_gate_pre = left_cat[..., hidden:]                                   # view
    left_gate = torch.sigmoid(left_gate_pre)

    # right side: proj + gate
    right_w_proj = weights["right_proj.weight"]
    right_w_gate = weights["right_gate.weight"]
    right_w_cat = torch.cat([right_w_proj, right_w_gate], dim=0).to(torch.float16)  # [2*H, dim]
    right_cat = torch.nn.functional.linear(x, right_w_cat)                          # [B,N,N,2*H]
    right_proj = right_cat[..., :hidden]                                           # view
    right_gate_pre = right_cat[..., hidden:]                                        # view
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates (in‑place multiplication to reduce memory)
    # ------------------------------------------------------------------
    # left = proj * gate
    left = torch.mul(left_proj, left_gate, out=left_proj)   # reuse left_proj storage
    # right = proj * gate
    right = torch.mul(right_proj, right_gate, out=right_proj)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask: broadcast on hidden dimension (mask ∈ {0,1} or soft mask)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]

        # Use the tiny Triton kernel to copy+mask (demonstrates kernel use)
        total_elems = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        # left ← left * mask
        _mask_copy_kernel[grid](
            left.data_ptr(),
            mask_fp16.data_ptr(),
            left.data_ptr(),
            total_elems,
            1,                     # src_stride = 1 (contiguous)
            1,                     # mask_stride = 1 (broadcasted)
            1,                     # dst_stride = 1
            BLOCK=BLOCK,
        )
        # right ← right * mask
        _mask_copy_kernel[grid](
            right.data_ptr(),
            mask_fp16.data_ptr(),
            right.data_ptr(),
            total_elems,
            1,
            1,
            1,
            BLOCK=BLOCK,
        )

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate (computed from the same normalized x)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)  # [H, dim]
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)  # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction using batched GEMM (cuBLAS) – fp16 input,
    #    fp32 accumulation (torch.bmm does this internally)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden

    # Reshape to (B*H, N, N) with contiguous memory layout
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [B*H, N, N]
    B_t = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)      # [B*H, N, N] (transpose j<->k)

    # Batched matrix multiply: each slice does Σ_k A[i,k] * B_t[k,j]
    C = torch.bmm(A, B_t)                                                # [B*H, N, N] fp16

    # Reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()           # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden]
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
