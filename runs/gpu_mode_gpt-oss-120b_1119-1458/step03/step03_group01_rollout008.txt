"""
Optimized Triton implementation of the “outgoing” TriMul forward pass (AlphaFold3).
Key optimisations:
* Fuse the four linear projections (left/right and their gates) into a single `F.linear`
  by concatenating the four weight matrices – reduces 4 matmul calls to 1.
* Fuse left‑/right‑gate element‑wise multiplications (and the optional mask) in a
  small Triton kernel, eliminating extra memory passes.
* All heavy arithmetic (projections, gates, core N³ bilinear product) runs in
  FP16 on Tensor‑Cores. LayerNorms keep FP32/FP16 weights for numerical stability.
* The N³ operation is performed with a single batched `torch.bmm`, which maps to
  cuBLAS’s highly‑optimised strided‑batched GEMM.
* The final gate multiplication is fused in‑place with another Triton kernel.
* The whole forward runs without any Python‑level loops and returns FP32 to match
  the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernels
# -------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,   # fp16 *
    right_ptr, right_gate_ptr, # fp16 *
    numel: tl.int64,           # total number of elements (B * N * N * H)
    BLOCK: tl.constexpr,       # block size
):
    """In‑place gate multiplication without mask:
       left  = left  * left_gate
       right = right * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def apply_gates_masked_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,                 # fp16 *, shape (B * N * N)
    hidden_dim: tl.int64,     # size of the channel dimension
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place mask + gate multiplication:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # each block of `hidden_dim` elements share the same mask entry
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg * m, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr, b_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place element‑wise multiplication: a = a * b"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# -------------------------------------------------------------
# Forward entry point
# -------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    TriMul “outgoing” forward.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N]       (bool/float) – may be None when nomask=True
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict containing at least 'dim' and 'hidden_dim';
                         optional key 'nomask' (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 0️⃣ Unpack & config
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy work
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the four linear projections (left/right + gates)
    # -----------------------------------------------------------------
    # shape checks: each .weight is (out_features, in_features) == (hidden_dim, dim)
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    # concatenate along the output dimension (0)
    w_combined = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate], dim=0
    ).to(torch.float16)              # (4*hidden_dim, dim)

    # one big linear: output = [Lproj, Rproj, Lgate_logits, Rgate_logits]
    combined = F.linear(x_h, w_combined)  # [B, N, N, 4*hidden_dim]  fp16

    # split
    Lproj = combined[..., 0:hidden_dim]
    Rproj = combined[..., hidden_dim:2 * hidden_dim]
    Lgate_logit = combined[..., 2 * hidden_dim:3 * hidden_dim]
    Rgate_logit = combined[..., 3 * hidden_dim:4 * hidden_dim]

    # gates (sigmoid)
    Lgate = torch.sigmoid(Lgate_logit)
    Rgate = torch.sigmoid(Rgate_logit)

    # -----------------------------------------------------------------
    # 4️⃣ Out‑gate (separate linear)
    # -----------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))      # [B,N,N,hidden_dim] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Fuse mask (if any) + gate multiplication via Triton
    # -----------------------------------------------------------------
    # Ensure contiguous for the kernels
    Lproj = Lproj.contiguous()
    Rproj = Rproj.contiguous()
    Lgate = Lgate.contiguous()
    Rgate = Rgate.contiguous()

    total_elems = Lproj.numel()          # B * N * N * hidden_dim
    BLOCK = 65536                         # tunable; ~64k elements per program
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    if nomask:
        apply_gates_kernel[grid](
            Lproj.view(-1),
            Lgate.view(-1),
            Rproj.view(-1),
            Rgate.view(-1),
            total_elems,
            BLOCK=BLOCK,
        )
    else:
        # mask broadcasted over hidden_dim
        mask_f = mask.to(torch.float16)                 # [B, N, N]
        mask_f = mask_f.contiguous()
        # flatten mask (no channel dim)
        apply_gates_masked_kernel[grid](
            Lproj.view(-1),
            Lgate.view(-1),
            Rproj.view(-1),
            Rgate.view(-1),
            mask_f.view(-1),
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
        )
    # gates are no longer needed
    del Lgate, Rgate

    # -----------------------------------------------------------------
    # 6️⃣ Core N³ bilinear product (batched GEMM)
    #    out[b,i,j,c] = Σ_k Lproj[b,i,k,c] * Rproj[b,j,k,c]
    # -----------------------------------------------------------------
    B, N, _, H = Lproj.shape  # H == hidden_dim
    batch_heads = B * H

    # reshape to (B*H, N, N)
    Lmat = Lproj.permute(0, 3, 1, 2).reshape(batch_heads, N, N).contiguous()
    Rmat = Rproj.permute(0, 3, 1, 2).reshape(batch_heads, N, N).contiguous()

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    Omat = torch.bmm(Lmat, Rmat.transpose(1, 2))

    # restore original layout [B, N, N, hidden_dim]
    out = (
        Omat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # -----------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP16 weights)
    # -----------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # -----------------------------------------------------------------
    # 8️⃣ Multiply by output gate (in‑place Triton kernel)
    # -----------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free

    # -----------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # -----------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = F.linear(out, w_to_out)          # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 10️⃣ Cast back to FP32 for API compatibility
    # -----------------------------------------------------------------
    return final.to(torch.float32)
