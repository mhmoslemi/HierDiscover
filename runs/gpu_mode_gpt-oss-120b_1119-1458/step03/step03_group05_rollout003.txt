"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1.  Layer‑norm the input (float32 → fp32 for stability).
2.  Cast to float16 for the heavy part of the pipeline.
3.  Fuse the optional mask into the linear projections:
        left = (x * mask) @ W_left   (bias‑free)
        right = (x * mask) @ W_right
    This replaces the two separate element‑wise mask multiplications.
4.  Compute the three gating vectors (sigmoid) from the *unmasked* x.
5.  Apply gates to the projected tensors (in‑place).
6.  Perform the O(N³) contraction
        out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]
    as a batched GEMM C = A @ Bᵀ with a custom Triton kernel.
    Tensor shapes are reshaped to [M, N, N] where M = B·H (batch × hidden).
    The kernel uses 128×128 tiles for the output dimensions and 64 for the
    reduction dimension, loads FP16 data, accumulates in FP32 and employs a
    3‑stage double‑buffered pipeline with 8 warps.
7.  Second Layer‑Norm, output‑gate multiplication and final linear
    projection back to the original feature dimension.
8.  Cast the result back to float32 (required by the harness).

Only the contraction (step 6) is implemented in Triton; all other
operations stay in PyTorch (fp16 where possible) to keep the code
compact while still achieving a large speed‑up over a pure PyTorch
implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : [M, N, K]      (left)
#   B : [M, K, N]      (right already transposed)
#   C : [M, N, N]      (output)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]   (right transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar tensor sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bk, stride_bj,   # strides for B (M,K,N)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                # tile row (i)
    pid_n = tl.program_id(1)                # tile col (j)
    pid_z = tl.program_id(2)                # batch*hidden index (M)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- load tile of A (M,N,K) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ----- load tile of B (M,K,N) -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # reduction dimension
            + offs_n[None, :] * stride_bj   # column dimension (post‑transpose)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ----- dot product (FP16×FP16 → FP32) -----
        acc += tl.dot(a, b)

    # Write back, cast to fp16
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors, launches the Triton kernel and restores
# the original layout [B, N, N, hidden].
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] (fp16)
    right: torch.Tensor,  # [B, N, N, H] (fp16)
) -> torch.Tensor:
    """
    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel defined above.
    """
    B, N, _, H = left.shape

    # Pack batch and hidden together → [M, N, K] with M = B * H
    # left   : (i, k) → keep as is
    # right  : (j, k) → we need it transposed → (k, j)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [M, N, K]
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)         # [M, K, N] (transposed)

    out = torch.empty_like(left_, dtype=torch.float16)  # [M, N, N]

    # Extract strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tunable block sizes – 128×128 for rows/cols, 64 for reduction.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,       # double‑buffered pipeline
    )

    # Reshape back to [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradient).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # First LayerNorm (float32 → better numerics)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x_fp32.to(torch.float16)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # Cast model weights to fp16 once (bias‑free linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Apply mask *before* the projections (fused).  Gating uses the
    # original (unmasked) tensor, so keep `x` unchanged for that.
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        proj_input = x * mask_fp16
    else:
        proj_input = x

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(proj_input, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(proj_input, right_proj_w)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Gating (sigmoid) – computed from the *unmasked* tensor `x`
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # O(N³) contraction via the Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # Final gating + projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # Cast to float32 as expected by the benchmark harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
