def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps=1e-5

    # 1 layernorm
    x = torch.nn.functional.layer_norm(...)

    # cast to half
    x_h = x.half()   # x.to(torch.float16)

    B, N, _, _ = x_h.shape
    M = B * N * N  # number of (i,k) pairs per batch

    # flatten for linear
    x_flat = x_h.reshape(M, dim)          # shape (M, dim)

    # prepare weight concat
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    weight_concat = torch.cat([w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0).to(torch.float16)

    # linear
    proj = torch.nn.functional.linear(x_flat, weight_concat)   # shape (M, 5*hidden_dim)
    # split into components
    proj = proj.view(M, 5, hidden_dim)   # (M,5,hidden_dim)
    left = proj[:,0,:].view(B, N, N, hidden_dim)
    right = proj[:,1,:].view(B, N, N, hidden_dim)
    left_gate = proj[:,2,:].view(B, N, N, hidden_dim)
    right_gate = proj[:,3,:].view(B, N, N, hidden_dim)
    out_gate = proj[:,4,:].view(B, N, N, hidden_dim)

    # sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # apply gates and mask (if any)
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 32768  # or 65536

    grid = ((numel + BLOCK - 1) // BLOCK,)

    if nomask:
        # put mask factor = 1 (no mask)
        apply_gates_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            right.view(-1),
            right_gate.view(-1),
            numel,
            BLOCK=BLOCK,
        )
    else:
        # mask flatten for replication across hidden_dim
        mask_h = mask.to(torch.float16).contiguous()            # [B,N,N]
        mask_h = mask_h.view(-1)       # shape M

        apply_gates_masked_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            right.view(-1),
            right_gate.view(-1),
            mask_h,
            numel,
            hidden_dim,
            BLOCK=BLOCK,
        )

    # Free gates tensors no longer needed
    del left_gate, right_gate

    # 8 core N³ matmul
    # reshape to (total_heads, N, N)
    total_heads = B * hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))   # shape (total_heads, N, N)
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # to_out_norm
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(out,
         normalized_shape=(hidden_dim,),
         weight=w_to_out_norm,
         bias=b_to_out_norm,
         eps=eps)
    
    # multiply by out_gate (inplace)
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    # delete out_gate to free
    del out_gate

    # final projection
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = torch.nn.functional.linear(out, w_to_out)   # [B,N,N,dim]

    # output to float32
    return final.to(torch.float32)

"""
Outgoing TriMul (AlphaFold3) – highly‑optimized forward pass.

Key ideas
---------
* Fuse the five independent Linear layers (`left_proj`, `right_proj`,
  `left_gate`, `right_gate`, `out_gate`) into a single matrix‑multiply
  by concatenating their weight matrices.  This reduces the number of
  passes over the input tensor from five to one and cuts the memory
  traffic dramatically.
* Fuse the element‑wise mask (when present) together with the left/right
  gate multiplication in a single Triton kernel (`apply_gates_{masked}_kernel`).
* Fuse the final “output gate” multiplication in another tiny Triton
  kernel (`mul_inplace_kernel`).
* Do the bulk of the arithmetic in FP16 (or BF16) while keeping the
  LayerNorms in the original dtype for numerical stability.
* All heavy operations (the N³ interaction) are performed with a
  batched `torch.bmm`, which on an H100 dispatches to cuBLAS tensor‑core
  GEMMs.

The implementation follows the reference PyTorch module but is
substantially faster because it dramatically reduces memory traffic
and fuses element‑wise work into custom Triton kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,        # fp16* – left tensor (flattened)
    left_gate_ptr,   # fp16* – left gate (flattened)
    right_ptr,       # fp16* – right tensor (flattened)
    right_gate_ptr,  # fp16* – right gate (flattened)
    numel: tl.int64,          # total number of elements per tensor
    BLOCK: tl.constexpr,      # block size (elements per program)
):
    """
    In‑place element‑wise multiplication:
        left  = left  * left_gate
        right = right * right_gate
    No mask is applied (used when ``nomask == True``).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # load
    l  = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,      mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # store in‑place
    tl.store(left_ptr  + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def apply_gates_masked_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,                       # fp16* – mask of shape (M,) where M = B*N*N
    numel: tl.int64,
    H: tl.constexpr,                # hidden dimension (size of the innermost axis)
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication with a broadcasted mask:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    ``mask_ptr`` stores one float16 value per (b,i,k) entry (0 or 1).
    The mask is broadcast over the hidden dimension ``H``.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < numel

    # load the raw tensors
    l  = tl.load(left_ptr + offs,      mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r  = tl.load(right_ptr + offs,      mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # compute mask index (broadcast over hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default 1.0 (no‑mask)

    # store the gated & masked results
    tl.store(left_ptr  + offs, l * lg * m, mask=valid)
    tl.store(right_ptr + offs, r * rg * m, mask=valid)


@triton.jit
def mul_inplace_kernel(
    a_ptr,               # fp16* – tensor to be multiplied (flattened)
    b_ptr,               # fp16* – multiplier (flattened, same shape as a_ptr)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        a = a * b
    Used for the final ``out_gate`` scaling.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < numel

    a = tl.load(a_ptr + offs, mask=valid, other=0.0)
    b = tl.load(b_ptr + offs, mask=valid, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=valid)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]   (float32)
        - mask         : torch.Tensor   [B, N, N]        (bool/float)
        - weights      : dict of model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape
    M = B * N * N                # number of (b, i, k) positions

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five Linear projections into a single matmul
    # ------------------------------------------------------------------
    # flatten to (M, dim)
    x_flat = x_h.reshape(M, dim)

    # concatenate all five weight matrices (no bias)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device).to(torch.float16)               # shape (5*hidden_dim, dim)

    # single fused linear
    proj = F.linear(x_flat, w_concat)             # (M, 5*hidden_dim)

    # split & reshape
    proj = proj.view(M, 5, hidden_dim)            # (M, 5, hidden_dim)

    left      = proj[:, 0, :].view(B, N, N, hidden_dim)
    right     = proj[:, 1, :].view(B, N, N, hidden_dim)
    left_gate = proj[:, 2, :].view(B, N, N, hidden_dim)
    right_gate= proj[:, 3, :].view(B, N, N, hidden_dim)
    out_gate  = proj[:, 4, :].view(B, N, N, hidden_dim)

    # free the concatenated projection tensor
    del proj, w_concat, x_flat, x_h

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid on the gate tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse “gate * mask” for left/right via Triton
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 32768                             # a good trade‑off for H100

    grid = ((numel + BLOCK - 1) // BLOCK,)

    if nomask:
        # No mask – simple element‑wise multiplication
        apply_gates_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            right.view(-1),
            right_gate.view(-1),
            numel,
            BLOCK=BLOCK,
        )
    else:
        # Broadcast mask over the hidden dimension inside the kernel
        mask_h = mask.to(torch.float16).contiguous()          # (B, N, N)
        mask_flat = mask_h.view(-1)                           # (M,)
        apply_gates_masked_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            right.view(-1),
            right_gate.view(-1),
            mask_flat,
            numel,
            hidden_dim,
            BLOCK=BLOCK,
        )
    # Gates no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ interaction – batched matmul per hidden channel
    # ------------------------------------------------------------------
    total_heads = B * hidden_dim                     # B * H
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # (total_heads, N, N) @ (total_heads, N, N)ᵀ -> (total_heads, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Post‑matmul LayerNorm (half‑precision)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device).to(torch.float16),
        bias=weights["to_out_norm.bias"].to(device).to(torch.float16),
        eps=eps,
    )  # still fp16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by the output gate (in‑place) via Triton
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate   # free memory

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out,
        weights["to_out.weight"].to(device).to(torch.float16),
    )  # [B, N, N, dim] (fp16)

    # Return in the expected float32 precision
    return final.to(torch.float32)
