"""
TriMul (outgoing) – fast forward pass for AlphaFold‑3 style pairwise update.

Algorithm
---------
1️⃣  Layer‑norm the input (float32 for stability).
2️⃣  Cast to fp16 and fuse **four** linear projections (left_proj,
    left_gate, right_proj, right_gate) into a *single* GEMM
    `x @ W_concat.T`.  The result is split into the four tensors,
    the gates are passed through a sigmoid and the masked projections
    are multiplied by their gates → `left` and `right` (shape [B,N,N,H]).
3️⃣  The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM C = A @ Bᵀ using a custom Triton
    kernel.  The input tensors are reshaped to [M,N,K] with
    M = B·H, N = seq_len, K = seq_len and the kernel tiles the
    multiplication with a 64×64×128 (or 128×128×128) layout,
    accumulating in FP32 while the data stays in FP16 (Tensor‑Core
    friendly).  Double‑buffering (`num_stages=4`) hides global‑memory
    latency.
4️⃣  Apply the second LayerNorm, the output‑gate (sigmoid) and the
    final linear projection back to `dim`.
5️⃣  Cast the result back to float32 (required by the harness).

The implementation keeps the low‑dimensional ops in native PyTorch
(easily vectorised) and isolates the cubic contraction – the true
bottleneck – in Triton.  By fusing the four projections we cut the
memory traffic to the input tensor by 4× and by using a larger K‑tile
the reduction loop count is halved, giving a substantial speed‑up
over a naïve implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A:[M,N,K], B:[M,K,N], C:[M,N,N])
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M,N,K]
    B_ptr,               # *fp16  [M,K,N]   (right already transposed)
    C_ptr,               # *fp16  [M,N,N]
    M, N, K,             # scalar problem sizes (int32)
    stride_am, stride_ai, stride_ak,   # strides of A
    stride_bm, stride_bk, stride_bj,   # strides of B (K‑N swapped)
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiply with the reduction over the K dimension.
    Each program instance (identified by pid_z) handles one slice of the
    hidden dimension (M = batch * hidden).  The kernel uses FP16 data,
    accumulates in FP32 and writes back FP16.
    """
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_z = tl.program_id(2)          # batch * hidden index

    # --------------------------------------------------------------
    # Offsets for the current i‑tile and j‑tile
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Accumulator in FP32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K‑tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load B tile (BLOCK_K × BLOCK_N) – note the swapped K/N order
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write out C tile (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors, launches the Triton kernel and restores
# the original layout [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] (fp16)
    right: torch.Tensor,  # [B, N, N, H] (fp16)
) -> torch.Tensor:
    """
    Performs `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    using the Triton kernel above.
    """
    B, N, _, H = left.shape

    # Pack batch & hidden (M = B*H) and transpose right for the dot product.
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [M,N,K]
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # [M,K,N]

    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides (in element units, not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # --------------------------------------------------------------
    # Choose tile sizes based on sequence length.
    # Larger sequences benefit from bigger M/N tiles and a larger K‑tile.
    # --------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 128
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        left_,
        right_,
        out,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,               # double‑buffered pipeline + one extra stage
    )

    # Recover original layout [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → more stable)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the work
    # ------------------------------------------------------------------
    x_fp16 = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the four linear projections (left/right + gates) into one
    # ------------------------------------------------------------------
    # Concatenate weights: left_proj, left_gate, right_proj, right_gate
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]

    # (4*H, dim) matrix in fp16, on the same device as x_fp16
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    ).to(device, dtype_fp16)

    # Single fused linear: shape [B, N, N, 4*H]
    combined = torch.nn.functional.linear(x_fp16, combined_w)  # no bias

    # Split the concatenated result
    H = hidden_dim
    left_proj  = combined[..., :H]          # [B,N,N,H]
    left_gate  = combined[..., H:2 * H]     # [B,N,N,H]
    right_proj = combined[..., 2 * H:3 * H] # [B,N,N,H]
    right_gate = combined[..., 3 * H:]      # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – kept in fp32 for numerical stability then cast
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate).to(dtype_fp16)
    right_gate = torch.sigmoid(right_gate).to(dtype_fp16)

    # Fuse gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (still fp16, hidden_dim = H)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype_fp16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final gating (out_gate) – compute from the *original* normed x
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device)          # keep fp32
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_w)
    )  # [B,N,N,H] float32
    out_gate_fp16 = out_gate.to(dtype_fp16)

    out = out * out_gate_fp16

    # ------------------------------------------------------------------
    # 9️⃣ Projection back to original dimension `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (as required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
