"""
Optimized “outgoing” TriMul implementation (AlphaFold‑3).

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32 → float32 for stability).
2.  Cast to fp16 and apply the four linear projections (left/right + three gates)
    using cuBLAS FP16 kernels.
3.  Optional mask is fused element‑wise after the projections.
4.  Fuse the hidden‑dim gates with the projected tensors (still fp16).
5.  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    This is a batched mat‑mul A @ Bᵀ with a custom Triton kernel.  The kernel
    works entirely in fp16 (dot product and accumulation) to minimise
    conversion overhead – the sum over ≤1024 terms safely fits in fp16.
    Tile size 128 × 128 × 64 and 8 warps give good occupancy on an H100.
6.  Apply the second LayerNorm (fp16), the output gate and the final linear
    projection back to the original model dimension.
7.  Cast the result back to float32 for the harness.

Only the O(N³) mat‑mul is written in Triton; all surrounding linear layers,
layer‑norms and sigmoids are performed with fast PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [B*H, N, N]  (fp16)
#   C    : [B*H, N, N]  (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]   (M = B*H, N = seq_len, K = seq_len)
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes (i32)
    stride_am, stride_ai, stride_ak,   # strides of A (in elements)
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile index for i dimension
    pid_j = tl.program_id(1)   # tile index for j dimension
    pid_z = tl.program_id(2)   # batch*hidden index

    # ------------------------------------------------------------------
    # Offsets for this tile (i, j)
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp16 (fp16 accumulation is safe for N ≤ 1024)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=tl.float16(0.0),
        )

        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=tl.float16(0.0),
        )

        # Multiply‑accumulate (fp16×fp16 → fp16)
        acc += tl.dot(a, b)   # dot returns fp16 because inputs are fp16

    # ------------------------------------------------------------------
    # Write back the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]   fp16
    right: torch.Tensor,  # [B, N, N, H]   fp16
) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.  Input tensors must be contiguous
    fp16 on the same CUDA device.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into the leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element units (Triton works with element strides)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tunable tile sizes – 128×128×64 works well on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth dimension (batch * hidden)
    )

    _trimul_matmul_kernel[
        grid
    ](
        A,
        B_,
        C,
        B * H,               # dummy M – not used inside kernel
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 256 threads per block
        num_stages=2,       # double‑buffering to hide latency
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

        - input_tensor : torch.Tensor, [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor, [B, N, N]      (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numeric stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                       # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast to fp16 for the heavy stages
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Convert all model weights to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    # Result shape: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)    # (B,N,N,hidden_dim)

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 in/out)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection (fp16)
    # -------------------------------------------------
    out = out * out_gate                              # gate after norm
    out = torch.nn.functional.linear(out, to_out_w)   # back to dim

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
