@triton.jit
def apply_gates_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    numel, hidden_dim,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # Compute mask offset: mask_idx = offs // hidden_dim
    # Use integer division that yields floor division; but we need to convert to integer type
    mask_idx = offs // hidden_dim  # integer division

    # Load values
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_ptr + offs, l * lg * m, mask=mask)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)

"""
Optimized outgoing TriMul forward pass (AlphaFold3 style).

Algorithm
---------
1. Input LayerNorm (FP16) – cheap fused kernel inside PyTorch.
2. Fuse the two projection linear layers (left/right) into a single
   GEMM → shape [..., 2*hidden_dim] → split.
3. Fuse the two gate linear layers (left/right) into a single GEMM,
   apply sigmoid, split into left/right gates.
4. Compute the out‑gate (single GEMM + sigmoid).
5. One Triton kernel multiplies left/right by mask (if present) *their
   respective gates in‑place, saving separate mask & gate kernels.
6. Batched matrix multiplication (BMM) implements the N³ einsum:
   left @ rightᵀ for each head (hidden channel) → shape [B,N,N,hidden_dim].
7. Post‑norm on the hidden dimension (FP16) using PyTorch’s fused LN.
8. One Triton kernel multiplies the result by the out‑gate in‑place.
9. Final linear projection back to `dim` (FP16) and cast to FP32.

All heavy arithmetic runs in float16 to exploit H100 Tensor‑Cores.
The kernel work‑loads are large (≈64 k elements per program) to keep launch
overhead low.  The implementation works for any of the test configurations,
including optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_mask_kernel(
    left_ptr,         # fp16* left  tensor  (flattened)
    left_gate_ptr,    # fp16* left_gate tensor (flattened)
    right_ptr,        # fp16* right tensor (flattened)
    right_gate_ptr,   # fp16* right_gate tensor (flattened)
    mask_ptr,         # fp16* mask tensor (flattened, shape = B*N*N)
    numel: tl.int64,          # total number of elements in left/right tensors
    hidden_dim: tl.int64,     # size of the hidden dimension (gate dimension)
    BLOCK: tl.constexpr,      # tile size
):
    """
    In‑place:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # index of the corresponding mask element (broadcast over hidden_dim)
    mask_off = offs // hidden_dim

    # load tensors
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)   # 1.0 for nomask

    # write back fused result
    tl.store(left_ptr + offs, l * lg * m, mask=mask)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,      # fp16* operand that will be overwritten
    b_ptr,      # fp16* multiplier (same shape)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    Element‑wise in‑place multiplication:
        a = a * b
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]   (float32)
        - mask         : torch.Tensor   [B, N, N]        (bool/float)
        - weights      : dict of torch.Tensor model parameters
        - config       : dict containing 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    dtype = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    # The LayerNorm weights are stored as float32 – cast to float16 for speed.
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim] float32
    x = x.to(dtype)  # FP16 for all subsequent work

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections: (left,right) = x @ W_proj.T
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(dtype)   # [hidden, dim]
    w_right_proj = weights["right_proj.weight"].to(dtype) # [hidden, dim]
    w_proj_cat = torch.cat([w_left_proj, w_right_proj], dim=0)  # [2*hidden, dim]

    proj = F.linear(x, w_proj_cat)  # [B,N,N,2*hidden] FP16
    left, right = torch.split(proj, hidden_dim, dim=-1)  # each [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Fused gate projections: (left_gate,right_gate) = sigmoid(x @ W_gate.T)
    # ------------------------------------------------------------------
    w_left_gate = weights["left_gate.weight"].to(dtype)   # [hidden, dim]
    w_right_gate = weights["right_gate.weight"].to(dtype)  # [hidden, dim]
    w_gate_cat = torch.cat([w_left_gate, w_right_gate], dim=0)  # [2*hidden, dim]

    gates = torch.sigmoid(F.linear(x, w_gate_cat))  # [B,N,N,2*hidden] FP16
    left_gate, right_gate = torch.split(gates, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate (independent linear + sigmoid)
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(dtype)    # [hidden, dim]
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))   # [B,N,N,hidden] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask (if any) and gate multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(dtype)                   # [B,N,N,1]
    else:
        # tensor of ones – broadcast will be a no‑op
        mask_h = torch.ones_like(left_gate[..., :1], device=device, dtype=dtype)

    # Ensure contiguous memory layout for vectorised loads/stores
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_h = mask_h.contiguous()   # shape [B,N,N,1] → view as [B,N,N]

    # Flatten tensors
    total_elems = left.numel()          # B * N * N * hidden_dim
    BLOCK = 65536                       # ~64 k elements per program
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    # mask pointer operates on the broadcasted mask (size B*N*N)
    apply_gates_mask_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        mask_h.view(-1),          # flatten mask (no hidden dim)
        total_elems,
        hidden_dim,
        BLOCK=BLOCK,
    )
    # Release gate tensors early
    del left_gate, right_gate, mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ operation: batched matrix multiplication per hidden head
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore shape to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Post‑norm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by the out‑gate (in‑place) via Triton
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(dtype)  # [dim, hidden]
    final = F.linear(out, w_to_out)                # [B,N,N,dim] FP16

    # Return in the dtype expected by the reference implementation
    return final.to(torch.float32)
