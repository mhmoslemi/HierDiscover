"""
TriMul (outgoing) – fused forward pass
--------------------------------------
The implementation follows the reference PyTorch module but reduces the
number of kernel launches and intermediate tensors:

1️⃣ Layer‑norm the input tensor (float32 → fp16 for the heavy work).
2️⃣ Fuse the four projection/gate linear layers *and* the output‑gate linear
   into a single matrix‑multiply:  
   `combined = Linear(x_norm, [left_proj; left_gate; right_proj; right_gate; out_gate])`.
3️⃣ Split the combined tensor, apply the (optional) pair‑wise mask, the
   sigmoid gates and obtain the final `left` and `right` tensors.
4️⃣ Perform the O(N³) contraction `out = Σₖ left[...,k,:]·right[...,k,:]ᵀ`
   with a strided batched GEMM (torch.bmm).  This is the dominant compute.
5️⃣ Layer‑norm the intermediate result, multiply element‑wise by the
   output‑gate (sigmoid) using a tiny Triton kernel.
6️⃣ Final linear projection back to the original channel dimension.
7️⃣ Cast the result back to float32 (the expected API dtype).

Only the element‑wise multiplication (step 5) is performed with a custom
Triton kernel; all large GEMMs are delegated to cuBLAS/Tensor‑Core for
maximum throughput on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication (dst *= src)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # fp16* – destination (in‑place)
    src_ptr,            # fp16* – source (gate)
    N,                  # i32 – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched matmul for the O(N³) contraction
#   Computes C = A @ Bᵀ  where A,B: [B, N, N, H] → out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] for a strided‑batched GEMM
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ   (fast Tensor‑Core implementation)
    C = torch.bmm(A, B_.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32, on same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5  # default eps used by torch.nn.LayerNorm

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32
    x = x_norm.to(torch.float16)  # work in FP16 from here on

    # -------------------------------------------------
    # 2️⃣ Fuse all three linear layers into one matmul
    # -------------------------------------------------
    #   left_proj, left_gate, right_proj, right_gate, out_gate
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    # concatenate on the output‑feature dimension
    combined_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)   # shape (5*hidden_dim, dim)

    # combined linear output: [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_weight)

    H = hidden_dim
    # split the combined tensor
    left_proj_raw = combined[..., 0:H]                 # [B,N,N,H]
    left_gate_raw = combined[..., H:2*H]              # [B,N,N,H]
    right_proj_raw = combined[..., 2*H:3*H]           # [B,N,N,H]
    right_gate_raw = combined[..., 3*H:4*H]           # [B,N,N,H]
    out_gate_raw = combined[..., 4*H:5*H]             # [B,N,N,H]

    # -------------------------------------------------
    # 3️⃣ Optional mask (applied before gating, exactly as reference)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Apply sigmoid gates and construct the final tensors
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left = left_proj_raw * left_gate    # [B,N,N,H]
    right = right_proj_raw * right_gate # [B,N,N,H]

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via a strided‑batched GEMM
    # -------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B,N,N,H] (FP16)

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B,N,N,H]

    # -------------------------------------------------
    # 7️⃣ Output‑gate (sigmoid) – element‑wise multiply via Triton
    # -------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)  # [B,N,N,H]  FP16

    # Flatten for the Triton kernel (in‑place)
    out_flat = out_norm.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)
    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,               # dst stride
        1,               # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # -------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    final = torch.nn.functional.linear(out_norm, to_out_w)  # [B,N,N,dim]  FP16

    # -------------------------------------------------
    # 9️⃣ Cast back to FP32 (API requirement)
    # -------------------------------------------------
    return final.to(torch.float32)
