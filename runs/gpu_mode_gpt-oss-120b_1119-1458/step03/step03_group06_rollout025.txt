    A_ptr, B_ptr, C_ptr,
    B, N, H,

    stride_br, stride_ri, stride_rk, stride_rh,
    stride_cb, stride_ci, stride_cj, stride_ch,

):

    pid_j = tl.program_id(1)

    a_base = A_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = B_ptr + b_idx * stride_br + h_idx * stride_rh
    c_base = C_ptr + b_idx * stride_cb + h_idx * stride_ch

"""
Optimized Triton implementation of the outgoing TriMul module (AlphaFold‑3).

Key optimisations
-----------------
* Fuse left/right projection and gating linear layers into two
  concatenated matmuls (proj+gate) → reduces memory traffic by ~2×.
* Eliminate the `permute(...).contiguous()` copy that was required to
  reshape tensors for the batched matmul.  The Triton kernel works
  directly on tensors of shape (B,N,N,H) by computing the batch/hid
  index inside the kernel.
* Use a 128×128×64 tile (i,j,k) with 8 warps, 4 pipeline stages and
  fp32 accumulation for numerical stability.
* All heavy work is performed in half‑precision; only the first
  LayerNorm and final cast are done in fp32.
* Optional mask is applied after the fused projection/gate step,
  avoiding a separate masking pass when `nomask=True`.

The forward pass reproduces the reference implementation in pure
PyTorch (no gradients) while substantially reducing runtime on an
H100 GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [B, N, N, H] (fp16)
#   C    : [B, N, N, H] (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,            # *fp16  [B, N, N, H]
    B_ptr,            # *fp16
    C_ptr,            # *fp16
    batch,            # i32  B
    seq_len,          # i32  N
    hidden_dim,       # i32  H
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides of A
    stride_br, stride_ri, stride_rk, stride_rh,   # strides of B
    stride_cb, stride_ci, stride_cj, stride_ch,   # strides of C
    BLOCK_M: tl.constexpr,   # tile size in i (rows)
    BLOCK_N: tl.constexpr,   # tile size in j (cols)
    BLOCK_K: tl.constexpr,   # reduction tile size
):
    """
    Compute out[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h].

    A, B are stored as (B,N,N,H) with H as the innermost dimension.
    The kernel treats the (b,h) pair as a single logical program-ID.
    """
    pid_i = tl.program_id(0)   # tile along i (first N axis)
    pid_j = tl.program_id(1)   # tile along j (second N axis)
    pid_z = tl.program_id(2)   # combined batch*head

    # Decode batch and head from pid_z
    b_idx = pid_z // hidden_dim
    h_idx = pid_z - b_idx * hidden_dim   # equivalent to pid_z % hidden_dim

    # Base pointers for this (b, h) slice
    a_base = A_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = B_ptr + b_idx * stride_br + h_idx * stride_rh
    c_base = C_ptr + b_idx * stride_cb + h_idx * stride_ch

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # Accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(seq_len, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load a tile of A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base \
                + offs_i[:, None] * stride_ai \
                + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < seq_len),
            other=0.0,
        )   # fp16

        # Load a tile of B, transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base \
                + offs_k[:, None] * stride_rk \
                + offs_j[None, :] * stride_ri
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < seq_len) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Multiply‑accumulate in FP32
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ------------------------------------------------------------------
    # Write results back
    # ------------------------------------------------------------------
    c_ptrs = c_base \
            + offs_i[:, None] * stride_ci \
            + offs_j[None, :] * stride_cj

    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] (fp16, contiguous)
    right: torch.Tensor,  # [B, N, N, H] (fp16, contiguous)
) -> torch.Tensor:
    """
    Batched matmul for the TriMul contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Both inputs must be on the same CUDA device and have the same
    shape.  Output is returned in fp16.
    """
    B, N, _, H = left.shape

    # Allocate output buffer
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides in *elements* (not bytes)
    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()
    stride_br, stride_ri, stride_rk, stride_rh = right.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out.stride()

    # Tile sizes – these values were found to be a good trade‑off on
    # NVIDIA H100 (Tensor‑core friendly 128×128×64 tiles).
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth dimension (batch * head)
    )

    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_br,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_cb,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=H,
        num_warps=8,
        num_stages=4,
    )

    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast to fp16 for the heavy work
    # -------------------------------------------------
    x = x_norm.to(torch.float16)  # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 2️⃣ Fuse projection + gating linear layers (fp16)
    # -------------------------------------------------
    # Concatenate left projection & left gate weights (shape: [2*H, dim])
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # Linear maps: input [B,N,N,dim] -> [B,N,N,2*H]
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # fp16
    right_concat = torch.nn.functional.linear(x, right_concat_w) # fp16

    # Split concatenated tensors back into projection & gate parts
    left_proj, left_gate_raw = left_concat.split([hidden_dim, hidden_dim], dim=-1)
    right_proj, right_gate_raw = right_concat.split([hidden_dim, hidden_dim], dim=-1)

    # Apply sigmoid to gate tensors (in‑place where possible)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # Fuse gating into the projected tensors (in‑place to avoid extra alloc)
    left = left_proj
    left.mul_(left_gate)               # left = left_proj * left_gate
    right = right_proj
    right.mul_(right_gate)             # right = right_proj * right_gate

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Output‑gate (independent linear + sigmoid)
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] fp16

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (fp16 in / fp16 out)
    # -------------------------------------------------
    # Ensure tensors are contiguous for the Triton kernel.
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 7️⃣ Apply out‑gate and final linear projection
    # -------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)        # back to dim (fp16)

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)

"""
Optimized Triton implementation of the *outgoing* TriMul operator 
(AlphaFold‑3). 

Key optimisations
-----------------
1️⃣ Fuse the left/right projection and gating linear layers into two
   concatenated matmuls, halving the number of passes over the large
   input tensor.

2️⃣ Eliminate the costly `permute(...).contiguous()` copy that was
   previously required to reshape tensors for the batched matmul.
   The Triton kernel works directly on tensors of shape
   (B, N, N, H) by decoding the batch‑and‑head index inside the kernel.

3️⃣ Use a 128 × 128 × 64 tile (i, j, k) with 8 warps and a 4‑stage
   pipeline.  Accumulation is performed in FP32 for numerical stability,
   while all heavy arithmetic stays in FP16 to minimise memory traffic.

4️⃣ Apply the optional mask after the fused projection/gate step, and
   perform the final layer‑norm, output‑gate and projection with fast
   PyTorch primitives.

The forward pass reproduces the reference PyTorch implementation
(excluding gradients) while significantly reducing runtime on an H100
GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [B, N, N, H] (fp16)
#   C    : [B, N, N, H] (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,              # *fp16  [B, N, N, H]
    B_ptr,              # *fp16
    C_ptr,              # *fp16
    batch,              # i32  B
    seq_len,            # i32  N
    hidden_dim,         # i32  H
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides of A
    stride_br, stride_ri, stride_rk, stride_rh,   # strides of B
    stride_cb, stride_ci, stride_cj, stride_ch,   # strides of C
    BLOCK_M: tl.constexpr,   # tile size along i (rows)
    BLOCK_N: tl.constexpr,   # tile size along j (cols)
    BLOCK_K: tl.constexpr,   # reduction tile size
):
    """
    Compute out[b,i,j,h] = Σₖ A[b,i,k,h] * B[b,j,k,h].

    The tensors are stored as (B, N, N, H) with H as the innermost
    dimension.  The program‑ID in the third dimension indexes the
    combined (batch, head) pair, which is decoded inside the kernel.
    """
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # combined batch*head

    # Decode batch index and head index from pid_z
    b_idx = pid_z // hidden_dim                # batch index
    h_idx = pid_z - b_idx * hidden_dim         # head index (equiv. pid_z % hidden_dim)

    # Base pointers for this (b, h) slice
    a_base = A_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = B_ptr + b_idx * stride_br + h_idx * stride_rh
    c_base = C_ptr + b_idx * stride_cb + h_idx * stride_ch

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k_tiles = tl.cdiv(seq_len, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile : shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base \
                + offs_i[:, None] * stride_ai \
                + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < seq_len),
            other=0.0,
        )  # fp16

        # Load B tile (transposed) : shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base \
                + offs_k[:, None] * stride_rk \
                + offs_j[None, :] * stride_ri
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < seq_len) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Multiply‑accumulate (fp16×fp16 → fp32)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = c_base \
            + offs_i[:, None] * stride_ci \
            + offs_j[None, :] * stride_cj
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] (fp16)
    right: torch.Tensor,  # [B, N, N, H] (fp16)
) -> torch.Tensor:
    """
    Batched matrix multiplication for the TriMul contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Both inputs must be contiguous fp16 tensors on the same CUDA device.
    """
    B, N, _, H = left.shape

    # Allocate output
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides in *elements* (Triton works with element strides)
    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()
    stride_br, stride_ri, stride_rk, stride_rh = right.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out.stride()

    # Tile configuration – tuned for H100 Tensor‑Core pipelines
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth (batch × head)
    )

    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_br,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_cb,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast to fp16 for the heavy work
    # -------------------------------------------------
    x = x_norm.to(torch.float16)  # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 2️⃣ Fuse projection & gating (fp16)
    # -------------------------------------------------
    # Concatenate projection and gate weight matrices
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                                   # [2*H, dim]
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                                   # [2*H, dim]

    # Perform the fused linear maps
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # [B,N,N,2*H]
    right_concat = torch.nn.functional.linear(x, right_concat_w) # [B,N,N,2*H]

    # Split back into projection and gate tensors (views, no copy)
    left_proj, left_gate_raw = left_concat.split([hidden_dim, hidden_dim], dim=-1)
    right_proj, right_gate_raw = right_concat.split([hidden_dim, hidden_dim], dim=-1)

    # Apply sigmoid gating
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # Fuse gate into projection (in‑place to avoid extra allocation)
    left = left_proj
    left.mul_(left_gate)     # left = left_proj * left_gate
    right = right_proj
    right.mul_(right_gate)   # right = right_proj * right_gate

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcast over heads)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left.mul_(mask_fp16)
        right.mul_(mask_fp16)

    # -------------------------------------------------
    # 4️⃣ Output‑gate (independent linear + sigmoid)
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H]

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (fp16 in / fp16 out)
    # -------------------------------------------------
    # Ensure contiguous layout for the kernel (still fp16)
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # -------------------------------------------------
    # 7️⃣ Apply output‑gate and final linear projection
    # -------------------------------------------------
    out.mul_(out_gate)                         # in‑place gating
    to_out_w = weights["to_out.weight"].to(torch.float16)  # [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)        # back to dim (fp16)

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
