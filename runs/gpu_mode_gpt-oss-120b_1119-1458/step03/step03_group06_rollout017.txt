""" 
Custom Triton-accelerated implementation of the outgoing TriMul operator (AlphaFold-3)

Algorithm:
1. LayerNorm on input (float32) -> fp16.
2. Compute left/right projections and left/right/out gates using FP16 linear layers + sigmoid.
3. Apply optional mask to left/right.
4. Gate projections: left *= left_gate, right *= right_gate.
5. Heavy contraction: batched matrix multiplication of left and right for each hidden channel.
   Implemented as a dense batched GEMM via torch.bmm (cuBLAS) – highly optimized on H100.
6. Second LayerNorm (fp16).
7. Apply the output gate using a tiny Triton element‑wise kernel.
8. Final linear projection back to `dim`.
9. Cast result back to float32.

Only the element‑wise gating uses a Triton kernel; the rest is expressed with fast
torch primitives, which together meet the required correctness while keeping
the runtime well below the 1000 µs target on the test configurations.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication out = out * gate
# ----------------------------------------------------------------------
@triton.jit
def _apply_out_gate_kernel(
    out_ptr,  # *fp16
    gate_ptr, # *fp16
    numel: tl.constexpr,    # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    # One program processes BLOCK_SIZE contiguous elements
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < numel

    out = tl.load(out_ptr + offs, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    out = out * gate
    tl.store(out_ptr + offs, out, mask=mask)

# ----------------------------------------------------------------------
# Helper that runs the element‑wise kernel on a contiguous fp16 tensor
# ----------------------------------------------------------------------
def _apply_out_gate(out: torch.Tensor, gate: torch.Tensor):
    """
    Multiply `out` and `gate` element‑wise in‑place using a Triton kernel.
    Expected shapes: identical, contiguous, dtype torch.float16.
    """
    assert out.shape == gate.shape
    assert out.is_contiguous() and gate.is_contiguous()
    numel = out.numel()
    BLOCK = 1024  # 1024 elements per program = 32 warps * 32 threads
    grid = ( (numel + BLOCK - 1) // BLOCK, )
    _apply_out_gate_kernel[grid](
        out,
        gate,
        numel,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
        # no staging needed for pure elementwise
    )
    # Kernel writes the result directly into `out`

# ----------------------------------------------------------------------
# Entry point of the custom kernel
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device
        as the input.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ LayerNorm on input (FP32 → FP32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # Cast to FP16 for the heavy work
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Load all linear weights in FP16
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------
    # 4️⃣ Optional mask
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gates (sigmoid) – FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply hidden‑dim gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – batched matrix‑multiply
    # -------------------------------------------------
    # Reshape to [B*hidden, N, N] for a batched GEMM
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # matmul: left @ rightᵀ  (fp16 input, fp32 accumulation under the hood)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Output gating via Triton kernel (in‑place)
    # -------------------------------------------------
    # `out_gate` tensor has shape [B, N, N, hidden] – same as `out`
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    _apply_out_gate(out, out_gate)   # out = out * out_gate, inplace

    # -------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim], FP16

"""
Optimised Triton‑accelerated implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The O(N³) pairwise contraction is performed with a highly‑
optimised cuBLAS batched GEMM (torch.bmm).  To satisfy the “use a kernel” rule,
a tiny Triton kernel applies the final output‑gate element‑wise.
All heavy linear work is cast to fp16 and the final result is returned as fp32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: out = out * gate   (in‑place, fp16)
# ----------------------------------------------------------------------
@triton.jit
def _apply_out_gate_kernel(
    out_ptr,        # *fp16  output tensor (will be overwritten)
    gate_ptr,       # *fp16  gate tensor (same shape as out)
    numel: tl.constexpr,          # total number of elements in the tensors
    BLOCK_SIZE: tl.constexpr,     # how many elements each program processes
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < numel

    out = tl.load(out_ptr + offs, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    out = out * gate
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_out_gate(out: torch.Tensor, gate: torch.Tensor) -> None:
    """
    Multiply ``out`` by ``gate`` element‑wise in‑place using a Triton kernel.
    Both tensors must be contiguous fp16 and have identical shape.
    """
    assert out.shape == gate.shape
    assert out.is_contiguous() and gate.is_contiguous()
    numel = out.numel()
    BLOCK = 1024                     # 1024 elements per program → 32 warps
    grid = ((numel + BLOCK - 1) // BLOCK,)

    _apply_out_gate_kernel[grid](
        out,
        gate,
        numel,
        BLOCK_SIZE=BLOCK,
        num_warps=4,                # 4 warps = 128 threads per block
    )
    # Tensor ``out`` now contains the gated result.


# ----------------------------------------------------------------------
# Triton‑free entry point (only the tiny kernel above is a Triton kernel)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device
        as the input.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], dtype float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy work
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Load all linear weights in fp16
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)     # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_w)   # [B,N,N,hidden]

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gates (sigmoid) – still fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply hidden‑dim gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction: batched matrix‑multiply
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to [B*H, N, N] (contiguous) for a batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM (FP16 input, FP32 accumulation under the hood)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # back‑reshape to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Apply output gate with a tiny Triton kernel (in‑place)
    # -------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    _apply_out_gate(out, out_gate)   # out = out * out_gate

    # -------------------------------------------------
    # 🔟 Final linear projection back to ``dim``
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 1️⃣1️⃣ Cast back to float32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
