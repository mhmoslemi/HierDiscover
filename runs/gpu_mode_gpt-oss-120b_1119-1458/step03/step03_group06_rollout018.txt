"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1.   Layer‑norm the 4‑D input in float32.
2.   Cast to fp16 and apply the four linear layers (left/right projections and
     three gates).  The optional binary mask is applied after the projections.
3.   Fuse the hidden‑dim gates with the projected tensors (elementwise mul).
4.   Contract over the third axis with the O(N³) operation
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
     implemented as a batched matrix‑multiply (A @ Bᵀ) in a single Triton
     kernel.  The kernel works directly on the original layout
     `[B, N, N, H]` (no expensive permute/copy) and accumulates in FP32 while
     reading/writing FP16.
5.   Apply a second LayerNorm (fp16), the output gate (sigmoid) and the final
     linear projection back to `dim`.
6.   Return the result in float32.

Only the heavy O(N³) contraction is performed in Triton; all other
operations use fast PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply C = A @ Bᵀ
#   A, B : [B, N, N, H] (fp16)
#   C    : [B, N, N, H] (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,          # *fp16 tensors
    B, N, H,                               # i32 scalars
    stride_bl, stride_ai, stride_ak,       # strides for left (batch, i, k)
    stride_br, stride_ri, stride_rk,       # strides for right (batch, j, k)
    stride_bo, stride_oi, stride_oj,       # strides for output (batch, i, j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction.

    For each (batch, hidden) pair we compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    """
    pid_i = tl.program_id(0)          # tile index along i (first N axis)
    pid_j = tl.program_id(1)          # tile index along j (second N axis)
    pid_z = tl.program_id(2)          # combined batch*hidden index

    # Decompose pid_z -> (batch, hidden)
    batch_idx = pid_z // H
    hidden_idx = pid_z % H

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_bl + hidden_idx
    right_base = right_ptr + batch_idx * stride_br + hidden_idx
    out_base   = out_ptr   + batch_idx * stride_bo + hidden_idx

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (i, k) – shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # Load right tile (k, j) – shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Multiply‑accumulate (fp16×fp16 → fp32).  The `out=` argument makes the
        # dot product accumulate into `acc` directly.
        tl.dot(a, b, out=acc)

    # Write the result back (fp16) with proper masking
    c_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_outgoing_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using the Triton
    kernel above.  The input tensors must be contiguous fp16 on the same GPU.
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    # Output buffer
    out = torch.empty_like(left, dtype=torch.float16)

    # Element‑wise strides (in elements, not bytes)
    stride_bl, stride_ai, stride_ak, stride_bh = left.stride()
    stride_br, stride_ri, stride_rk, stride_rh = right.stride()
    stride_bo, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile size – tuned for H100 tensor‑core pipelines
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[
        grid
    ](
        left,
        right,
        out,
        B, N, H,
        stride_bl,
        stride_ai,
        stride_ak,
        stride_br,
        stride_ri,
        stride_rk,
        stride_bo,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out

# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float32)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast to fp16 for the heavy part
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once
    # -------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_w)   # (B,N,N,hidden)

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 in / fp16 out)
    # -------------------------------------------------
    out = _batched_matmul_outgoing_fp16(left, right)   # [B,N,N,hidden]

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)      # back to dim

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
