def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Compute offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    h_start = (pid_z % ((H + BLOCK_H - 1)//BLOCK_H)) * BLOCK_H
    b = pid_z // ((H + BLOCK_H - 1)//BLOCK_H)

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # define ranges
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)
    h_range = tl.arange(0, BLOCK_H)

    # loop over K blocks
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        # mask for k tile
        k_valid = k_range < cur_k

        # compute offsets pointers
        # left memory offset: for each i, k, h
        left_ptrs = left_ptr \
            + b * stride_lb \
            + (i_start + i_range[:, None, None]) * stride_li \
            + (k_start + k_range[None, :, None]) * stride_lk \
            + (h_start + h_range[None, None, :]) * stride_lh

        # load left
        left = tl.load(left_ptrs, mask=(i_range[:, None, None] < N) &
                                      (k_range[None, :, None] < cur_k) &
                                      (h_range[None, None, :] < H), other=0.0)

        # right memory offset: for each j, k, h
        right_ptrs = right_ptr \
            + b * stride_rb \
            + (j_start + j_range[:, None, None]) * stride_ri \
            + (k_start + k_range[None, :, None]) * stride_rk \
            + (h_start + h_range[None, None, :]) * stride_rh

        right = tl.load(right_ptrs, mask=(j_range[:, None, None] < N) &
                                      (k_range[None, :, None] < cur_k) &
                                      (h_range[None, None, :] < H), other=0.0)

        # left shape: (M, cur_k, H)
        # right shape: (N, cur_k, H)
        # need to broadcast left over j and right over i to compute elementwise product.
        # Expand dimensions to (M,1,cur_k,H) and (1,N,cur_k,H)
        left_exp = left[:, None, :, :]   # shape (M, 1, K, H)
        right_exp = right[None, :, :, :] # shape (1, N, K, H)

        # elementwise multiply and sum over K
        # product shape (M,N,K,H)
        prod = left_exp * right_exp

        # sum over K axis (axis=2)
        acc += tl.sum(prod, axis=2)  # result shape (M,N,H)

    # after K loop, store acc to output
    # output pointer layout: out[b, i, j, h] (original order)
    # compute store offsets
    # output offsets: base + b*stride_ob + (i_start + i_idx)*stride_oi + (j_start + j_idx)*stride_oj + (h_start + h_idx)*stride_oh

    out_ptrs = out_ptr \
        + b * stride_ob \
        + (i_start + i_range[:, None, None]) * stride_oi \
        + (j_start + j_range[None, :, None]) * stride_oj \
        + (h_start + h_range[None, None, :]) * stride_oh

    # store using masks for boundaries
    mask_i = i_start + i_range < N
    mask_j = j_start + j_range < N
    mask_h = h_start + h_range < H
    mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

"""
Optimized outgoing TriMul forward pass (AlphaFold3 style).

The implementation follows the reference PyTorch module but avoids the costly
permute‑and‑contiguous copy of the [B, N, N, H] tensors before the cubic
einsum.  A custom Triton kernel computes the N³ contraction directly on the
original layout:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for every batch *b*, sequence indices *i, j* and hidden channel *h*.
The kernel tiles the i, j and hidden dimensions (32×32×8) and the reduction
dimension k (32) and accumulates in float‑32 registers for numerical
stability.  After the contraction we apply the remaining layer‑norms,
gates and final linear projection using lightweight Triton kernels for the
element‑wise fusions.

All heavy arithmetic runs in FP16 (Tensor‑Core friendly) while the final
output is returned in FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Element‑wise kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,   # fp16*  (flattened tensors)
    right_ptr, right_gate_ptr, # fp16*
    numel: tl.int64,           # total elements per tensor
    BLOCK: tl.constexpr,       # block size
):
    """
    In‑place fuse left/right linear output with their respective gates.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l  = tl.load(left_ptr + offs,  mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs,  l * lg, mask=mask)

    r  = tl.load(right_ptr + offs,  mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,      # fp16* (output tensor to be multiplied in‑place)
    b_ptr,      # fp16* (multiplicand)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication a = a * b.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Core cubic contraction kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B,                          # batch size (runtime)
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    num_h_blocks: tl.int64,     # = ceil(H / BLOCK_H)
    N: tl.constexpr,            # sequence length (compile‑time)
    H: tl.constexpr,            # hidden dimension (compile‑time)
    BLOCK_M: tl.constexpr,      # tile size for i axis
    BLOCK_N: tl.constexpr,      # tile size for j axis
    BLOCK_K: tl.constexpr,      # tile size for reduction (k) axis
    BLOCK_H: tl.constexpr,      # tile size for hidden axis
):
    """
    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    directly on the (B,N,N,H) layout without permuting.
    """
    pid_i = tl.program_id(0)                     # i‑tile
    pid_j = tl.program_id(1)                     # j‑tile
    pid_z = tl.program_id(2)                     # combines batch & hidden block

    # ------------------------------------------------------------------
    # Decode batch index and hidden‑block index
    # ------------------------------------------------------------------
    num_hb = num_h_blocks
    batch_idx = pid_z // num_hb
    hb_idx    = pid_z %  num_hb          # hidden‑block index

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    h_start = hb_idx * BLOCK_H

    # ------------------------------------------------------------------
    # Ranges (compile‑time constants)
    # ------------------------------------------------------------------
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)
    h_range = tl.arange(0, BLOCK_H)

    # ------------------------------------------------------------------
    # Accumulator in FP32 (no overflow)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k tiles (compile‑time unrolled)
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K

        # ----------------------------------------------------------------
        # Masks for edge handling
        # ----------------------------------------------------------------
        i_valid = i_start + i_range < N
        j_valid = j_start + j_range < N
        k_valid = k_start + k_range < N
        h_valid = h_start + h_range < H

        mask_left  = i_valid[:, None, None] & k_valid[None, :, None] & h_valid[None, None, :]
        mask_right = j_valid[:, None, None] & k_valid[None, :, None] & h_valid[None, None, :]

        # ----------------------------------------------------------------
        # Load left tile   : shape (M, K, H)
        # ----------------------------------------------------------------
        left_base = left_ptr + batch_idx * stride_lb
        left_offs = (i_start + i_range[:, None, None]) * stride_li \
                  + (k_start + k_range[None, :, None]) * stride_lk \
                  + (h_start + h_range[None, None, :]) * stride_lh
        left_tile = tl.load(left_base + left_offs, mask=mask_left, other=0.0)

        # ----------------------------------------------------------------
        # Load right tile transposed (K, N, H) so that we can call tl.dot
        # ----------------------------------------------------------------
        right_base = right_ptr + batch_idx * stride_rb
        right_offs = (j_start + j_range[None, :, None]) * stride_ri \
                   + (k_start + k_range[:, None, None]) * stride_rk \
                   + (h_start + h_range[None, None, :]) * stride_rh
        right_tile = tl.load(right_base + right_offs, mask=mask_right, other=0.0)

        # Cast to FP32 for accumulation
        left_f32  = left_tile.to(tl.float32)
        right_f32 = right_tile.to(tl.float32)

        # ----------------------------------------------------------------
        # Compute dot products for every hidden slice in the block
        # ----------------------------------------------------------------
        for hh in range(0, BLOCK_H):
            # left_slice: (M, K) ; right_slice: (K, N)
            left_slice  = left_f32[:, :, hh]    # (M, K)
            right_slice = right_f32[:, :, hh]   # (K, N)

            # Accumulate into the corresponding hidden slice
            acc[:, :, hh] += tl.dot(left_slice, right_slice)   # (M,N)

    # ------------------------------------------------------------------
    # Write the result back (with masks for tail tiles)
    # ------------------------------------------------------------------
    i_valid = i_start + i_range < N
    j_valid = j_start + j_range < N
    h_valid = h_start + h_range < H
    mask_out = i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]

    out_base = out_ptr + batch_idx * stride_ob
    out_offs = (i_start + i_range[:, None, None]) * stride_oi \
             + (j_start + j_range[None, :, None]) * stride_oj \
             + (h_start + h_range[None, None, :]) * stride_oh

    tl.store(out_base + out_offs, acc.to(tl.float16), mask=mask_out)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (AlphaFold3) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask         : torch.Tensor      [B, N, N]      (bool/float)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all dense work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Convert model weights to FP16
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, hidden]
    right = F.linear(x_h, w_right_proj)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gate pre‑activations (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)      # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates (one Triton kernel)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536                     # tune; ~64 k elements per kernel launch
    grid_apply = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid_apply](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Gates are no longer needed → free memory
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ contraction – custom Triton kernel
    # ------------------------------------------------------------------
    B, N_, _, H = left.shape
    assert N_ == N, "sequence length mismatch"
    assert H == hidden_dim, "hidden dimension mismatch"

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left.device)

    # strides for the original (B, N, N, H) layout
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile configuration – chosen to fit in registers and exploit Tensor‑Cores
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 8                         # hidden is processed in blocks of 8

    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M ),   # i‑tiles
        ( (N + BLOCK_N - 1) // BLOCK_N ),   # j‑tiles
        B * num_h_blocks                    # batch × hidden‑block dimension
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        num_h_blocks,
        N=N,                 # compile‑time constant
        H=hidden_dim,        # compile‑time constant
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm (FP16 weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # FP16

    # ------------------------------------------------------------------
    # 🔟 Multiply by the output gate (in‑place) via Triton
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_mul = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_mul](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # release memory

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim] (FP16)

    # Return as FP32 to match the reference implementation
    return final.to(torch.float32)
