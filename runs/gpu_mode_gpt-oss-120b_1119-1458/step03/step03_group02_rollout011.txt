"""
TriMul (outgoing) – Triton based forward pass
================================================

The outgoing TriMul module computes a dense pairwise contraction
over a 4‑D tensor `x` of shape [B, N, N, C] (B = batch, N = seq_len,
C = feature dim).  The operation consists of:

1️⃣ Layer‑norm over the feature axis (float32 for stability).
2️⃣ Linear projections to a hidden dimension `H` (fp16) and element‑wise
   sigmoid gates (fp16).  Optional masking is applied after the projections.
3️⃣ Heavy O(N³) contraction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply C = A @ Bᵀ where each (b,h) pair
   defines an independent N×N multiplication.  The contraction is
   performed by a custom Triton kernel that tiles the multiplication
   with 64×64×64 blocks, accumulates in FP32 (tensor‑core MMA) and
   stores the result in FP16.
4️⃣ A second LayerNorm over the hidden dimension, an output gate (sigmoid)
   and a linear projection back to the original dimension.
5️⃣ The final result is cast to float32.

The implementation avoids any intermediate copies of the 4‑D tensors.
Instead of reshaping `[B,N,N,H] → [B*H,N,N]` (which would require a
large memory copy), the Triton kernel works directly on the original
layout and computes the per‑pair offset from the program id.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A, B : [B, N, N, H]  (fp16)
#   C    : [B, N, N, H]  (fp16)   # same layout as A/B
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                         # *fp16  [B, N, N, H]
    B_ptr,                         # *fp16  [B, N, N, H]
    C_ptr,                         # *fp16  [B, N, N, H]
    M, N, K,                       # i32 – M = B*H, N = seq_len, K = seq_len
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,   # strides of A (in elements)
    stride_b_b, stride_b_i, stride_b_k, stride_b_h,   # strides of B
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,   # strides of C
    hidden_dim: tl.constexpr,                          # H (known at launch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batched product for every (batch, hidden) pair:
        C[b,i,j,h] = Σₖ A[b,i,k,h] * B[b,j,k,h]

    The kernel tiles over the two N‑axes (i/j) with 2‑D blocks of size
    BLOCK_M × BLOCK_N and reduces over K in blocks of size BLOCK_K.
    """
    pid_i = tl.program_id(0)   # tile along output‑row axis   (i)
    pid_j = tl.program_id(1)   # tile along output‑col axis   (j)
    pid_z = tl.program_id(2)   # combined (batch, hidden) index = b*H + h

    # ------------------------------------------------------------------
    # Derive batch (b) and hidden (h) indices from pid_z
    # ------------------------------------------------------------------
    b = pid_z // hidden_dim
    h = pid_z % hidden_dim

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Main K‑reduction loop
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- Load A tile: shape (BLOCK_M, BLOCK_K) ----
        a_ptrs = (
            A_ptr
            + b * stride_a_b
            + h * stride_a_h
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- Load B tile (transposed view) shape (BLOCK_K, BLOCK_N) ----
        # We need B[b, j, k, h] -> k is the row dimension, j the column
        b_ptrs = (
            B_ptr
            + b * stride_b_b
            + h * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_i
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ---- Multiply‑accumulate (tensor‑core MMA) ----
        # tl.dot promotes fp16 multiplication to fp32 accumulation.
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + b * stride_c_b
        + h * stride_c_h
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel on tensors of shape [B,N,N,H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor   (B, N, N, H)  fp16
    right: torch.Tensor   (B, N, N, H)  fp16

    Returns
    -------
    torch.Tensor of shape (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # stride values are expressed in *elements* (not bytes)
    stride_a_b, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_b, stride_b_i, stride_b_k, stride_b_h = right.stride()
    # output buffer – same layout as inputs (contiguous after linear layers)
    out = torch.empty_like(left, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Tile sizes – 64×64×64 works well on H100 for FP16 MMA.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (tiles over i, tiles over j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B * H,              # M = total (batch * hidden)
        N,
        N,
        stride_a_b,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_b,
        stride_b_i,
        stride_b_k,
        stride_b_h,
        stride_a_b,         # C has same strides as A (both are [B,N,N,H])
        stride_a_i,
        stride_a_k,         # NOTE: for C the third stride is the same as A's K‑stride (i.e. H)
        stride_a_h,
        H,                  # hidden_dim (constexpr inside kernel)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numeric stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype float32

    # -------------------------------------------------
    # Cast to FP16 for the rest of the pipeline
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Convert all linear weights to FP16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -------------------------------------------------
    # torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # -------------------------------------------------
    # 4️⃣ Optional mask – broadcast over hidden dimension
    # -------------------------------------------------
    if not nomask:
        # mask shape: [B,N,N] → [B,N,N,1] → broadcast with hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gating to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ O(N³) contraction via Triton (FP16 inputs, FP16 output)
    # -------------------------------------------------
    # Result shape: [B, N, N, hidden_dim] (FP16)
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden_dim) – still in FP16
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate                                 # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)      # [B,N,N,dim] (FP16)

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (required output type)
    # -------------------------------------------------
    return out.to(torch.float32)
