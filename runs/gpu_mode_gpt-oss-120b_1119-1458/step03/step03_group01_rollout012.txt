    def mul_inplace_kernel(...): ...

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # Input LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # cast to fp16
    x_h = x.to(torch.float16)

    # cast weights
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # Projections
    left = F.linear(x_h, w_left_proj)      # [B,N,N,hidden_dim]
    right = F.linear(x_h, w_right_proj)

    # Gate raw and sigmoid
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # Flatten
    n_elem = left.numel()
    BLOCK = 65536
    grid = ((n_elem + BLOCK - 1) // BLOCK,)

    # Prepare mask pointer
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous()  # shape [B,N,N]
    else:
        mask_h = torch.tensor([1.0], device=left.device, dtype=left.dtype)  # dummy

    apply_gate_mask_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        mask_h.view(-1),
        mask_flag=0 if nomask else 1,
        hidden_dim=hidden_dim,
        numel=n_elem,
        BLOCK=BLOCK,
    )
    del left_gate, right_gate

    # Core computation
    out = torch.einsum('bikd,bjkd->bijd', left, right)  # [B,N,N,hidden_dim] in fp16

"""
Optimized outgoing TriMul forward (AlphaFold3 style) with Triton fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) – the only floating‑point operation that
   needs high numerical stability.
2. Cast to FP16 for the heavy work.
3. Project the tensor to the hidden dimension with three linear layers:
   * left_proj, right_proj  – no bias
   * left_gate, right_gate, out_gate – followed by sigmoid.
4. Fuse “mask × left_gate” and “mask × right_gate” into a single Triton kernel.
   The kernel works on the flattened tensors and (optionally) applies the mask
   broadcasted over the hidden dimension.
5. Compute the core N³ contraction with a single `torch.einsum` call:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This avoids the expensive permute‑contiguous copy required for batched
   `torch.bmm`.
6. Post‑norm the result (LayerNorm over the hidden dimension).
7. Fuse the element‑wise multiplication with `out_gate` using another Triton
   kernel (in‑place).
8. Final linear projection back to `dim` and cast the output to float32.

The implementation keeps the dominant N³ work in half‑precision, eliminates
the large permute/contiguous copy, and fuses all cheap element‑wise ops into
just two Triton kernels, yielding a considerable speed‑up on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    left_ptr,          # fp16*, flatten left   shape (B*N*N*H,)
    left_gate_ptr,     # fp16*, flatten left_gate, same shape
    right_ptr,         # fp16*, flatten right
    right_gate_ptr,    # fp16*, flatten right_gate
    mask_ptr,          # fp16*, flatten mask (B*N*N,) – dummy when mask_flag==0
    mask_flag: tl.constexpr,   # 0 = no mask, 1 = mask present
    hidden_dim: tl.constexpr,  # size of the hidden dimension (H)
    numel: tl.int64,           # total number of elements in left/right
    BLOCK: tl.constexpr,       # tile size
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask (if present)
        right = right * right_gate * mask (if present)
    All tensors are flattened. The mask is broadcasted over the hidden
    dimension; its flattened length is B * N * N.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < numel

    # Index in the mask array (one entry per (b,i,k) location)
    mask_index = offs // hidden_dim

    # ----- LEFT -----
    l  = tl.load(left_ptr + offs,       mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    if mask_flag:
        m = tl.load(mask_ptr + mask_index, mask=valid, other=1.0)
        l = l * m
    tl.store(left_ptr + offs, l * lg, mask=valid)

    # ----- RIGHT -----
    r  = tl.load(right_ptr + offs,       mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs,  mask=valid, other=0.0)
    if mask_flag:
        m = tl.load(mask_ptr + mask_index, mask=valid, other=1.0)
        r = r * m
    tl.store(right_ptr + offs, r * rg, mask=valid)


@triton.jit
def mul_inplace_kernel(
    a_ptr,          # fp16*, flatten tensor to be multiplied in‑place
    b_ptr,          # fp16*, flatten multiplier (same shape)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """Element‑wise in‑place multiplication: a = a * b."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)

# Triton‑accelerated outgoing TriMul forward

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        - mask         : torch.Tensor      [B, N, N]         (bool/float)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   must contain 'dim', 'hidden_dim' and may contain

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32, stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy math
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (no bias on the projection layers)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, hidden_dim] (fp16)
    right = F.linear(x_h, w_right_proj)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gate pre‑activations (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask (if present) and gate multiplications in‑place
    # ------------------------------------------------------------------
    # Ensure contiguous layout for the flattening that Triton expects
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # Flatten tensors for the kernel
    numel = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 65536
    grid = ((numel + BLOCK - 1) // BLOCK,)

    # Prepare mask pointer (broadcasted over hidden_dim) – dummy when mask is absent
    if not nomask:
        # Cast boolean / float mask to fp16 and flatten
        mask_h = mask.to(torch.float16).contiguous()
        mask_ptr = mask_h.view(-1)
        mask_flag = 1
    else:
        # Dummy 1‑element tensor – never accessed because mask_flag=0
        mask_ptr = torch.tensor([1.0], device=left.device, dtype=left.dtype).view(-1)
        mask_flag = 0

    apply_gate_mask_kernel[grid](
        left.view(-1),          # left_ptr
        left_gate.view(-1),     # left_gate_ptr
        right.view(-1),         # right_ptr
        right_gate.view(-1),    # right_gate_ptr
        mask_ptr,               # mask_ptr
        mask_flag,              # mask_flag constexpr
        hidden_dim,             # hidden_dim constexpr
        numel,
        BLOCK=BLOCK,
    )
    # Release gate tensors – they are no longer needed
    del left_gate, right_gate, mask_ptr

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction (batched “einsum”)
    # ------------------------------------------------------------------
    # The operation: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('bikd,bjkd->bijd', left, right)   # [B, N, N, hidden_dim] fp16

    # Left/right are no longer needed
    del left, right

    # ------------------------------------------------------------------
    # 8️⃣ Post‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (in‑place Triton kernel)
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim] (fp16)

    # Return in the expected Float32 precision
    return final.to(torch.float32)
