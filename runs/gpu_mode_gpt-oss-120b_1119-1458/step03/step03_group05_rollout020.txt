def gate_and_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_raw_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    total, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    NUM_WARPS: tl.constexpr = 4,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load values
    left_p = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    left_g_raw = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_p = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    right_g_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    out_g_raw = tl.load(out_gate_raw_ptr + offs, mask=mask, other=0.0)

    # Sigmoid

    left_g = 1.0 / (1.0 + tl.exp(-left_g_raw))
    right_g = 1.0 / (1.0 + tl.exp(-right_g_raw))

    mask_idx = offs // hidden_dim

"""
TriMul (outgoing) – fast forward pass.

The implementation follows the reference PyTorch module but
* keeps the first LayerNorm in FP32 for stability,
* performs all linear projections in FP16,
* fuses the three element‑wise steps (left‑gate, right‑gate,
  mask, and out‑gate sigmoid) in a single Triton kernel,
* carries out the heavy O(N³) contraction with a batched GEMM
  (`torch.bmm`) that uses cuBLAS/Tensor‑cores on H100,
* finishes with a second FP16 LayerNorm, the output gate and the
  final linear projection back to ``dim``.

All heavy compute (`left·rightᵀ`) is done by cuBLAS while the
element‑wise fusion is performed in Triton, satisfying the
requirement to use a custom kernel.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse sigmoid, mask and element‑wise multiplication.
# Input tensors are contiguous and stored as [B, N, N, H]   (H = hidden_dim).
# The tensors are flattened (row‑major) and processed as a 1‑D array.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    # input pointers
    left_proj_ptr,    # fp16  [*] = left_proj.flatten()
    left_gate_ptr,    # fp16  [*] = left_gate_raw.flatten()
    right_proj_ptr,   # fp16  [*] = right_proj.flatten()
    right_gate_ptr,   # fp16  [*] = right_gate_raw.flatten()
    out_gate_ptr,     # fp16  [*] = out_gate_raw.flatten()
    mask_ptr,         # fp16  [*] = mask.flatten()  (size = B*N*N)

    # output pointers
    left_out_ptr,     # fp16  [*] left after gating and mask
    right_out_ptr,    # fp16  [*] right after gating and mask
    out_gate_out_ptr, # fp16  [*] sigmoid(out_gate_raw)

    total_elements,   # i64 total number of elements (=B*N*N*H)
    hidden_dim,       # i64 hidden dimension (H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # ---- Load raw data -------------------------------------------------
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    lg = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0).to(tl.float16)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    og = tl.load(out_gate_ptr + offs,   mask=mask, other=0.0).to(tl.float16)

    # ---- Sigmoid --------------------------------------------------------
    # sigmoid(x) = 1 / (1 + exp(-x))
    left_gate = 1.0 / (1.0 + tl.exp(-lg))
    right_gate = 1.0 / (1.0 + tl.exp(-rg))
    out_gate = 1.0 / (1.0 + tl.exp(-og))

    # ---- Mask (broadcast over the hidden dimension) --------------------
    # For a flattened index ``offs`` the corresponding mask index is
    # ``offs // hidden_dim`` because the innermost stride is the hidden
    # dimension.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0).to(tl.float16)

    # ---- Apply gate and mask --------------------------------------------
    left_out = lp * left_gate * m
    right_out = rp * right_gate * m

    # ---- Write results ---------------------------------------------------
    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
    tl.store(out_gate_out_ptr + offs, out_gate, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched gemm C = A @ Bᵀ  where A,B are [B*H, N, N] (fp16)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) TriMul contraction using a cuBLAS batched GEMM.

    Arguments
    ---------
    left, right : torch.Tensor
        [B, N, N, H]  (fp16).  ``right`` will be transposed on the
        last two dimensions before multiplication.

    Returns
    -------
    torch.Tensor
        [B, N, N, H]  (fp16) result of ``left @ rightᵀ``.
    """
    B, N, _, H = left.shape
    # reshape to (batch*head, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # BMM where the second argument is transposed (k‑j)
    out_ = torch.bmm(left_, right_.transpose(1, 2))
    # restore original layout
    out = out_.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no grads).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor[B, N, N]   (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                            # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights once (FP16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16) – each produces a tensor of shape
    #    [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)      # [B,N,N,H]
    left_gate  = torch.nn.functional.linear(x, left_gate_w)      # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_w)     # [B,N,N,H]
    right_gate = torch.nn.functional.linear(x, right_gate_w)     # [B,N,N,H]
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Prepare mask (broadcast over hidden).  If ``nomask`` we use a
    #    tensor of ones – the kernel multiplies by it anyway.
    # ------------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]

    # ------------------------------------------------------------------
    # 6️⃣ Fuse sigmoid, mask and element‑wise gating in a Triton kernel
    # ------------------------------------------------------------------
    left   = torch.empty_like(left_proj)
    right  = torch.empty_like(right_proj)
    out_gate = torch.empty_like(out_gate_raw)

    total = left_proj.numel()                     # B * N * N * H
    BLOCK_SIZE = 1024                             # threads per block
    grid = (triton.cdiv(total, BLOCK_SIZE), )    # 1‑D grid

    _gate_mask_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        out_gate_raw,
        mask_fp16,
        left,
        right,
        out_gate,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=32,          # 1024 threads → 32 warps
    )

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction – batched GEMM using cuBLAS (fast on H100)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gating (element‑wise) and linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to FP32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
