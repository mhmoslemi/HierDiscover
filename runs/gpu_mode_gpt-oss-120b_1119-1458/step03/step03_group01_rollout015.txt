@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    mask_ptr, stride_mask_b, stride_mask_i, stride_mask_k,
    has_mask: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # pid example
    pid = tl.program_id(0)
    total_batch = B * H
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    num_tiles = num_blocks_i * num_blocks_j
    batch_head = pid // num_tiles
    tile_id = pid % num_tiles
    i_block = tile_id // num_blocks_j
    j_block = tile_id % num_blocks_j

    # compute b and h
    b = batch_head // H
    h = batch_head % H

    # offsets for i and j
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute masks for i, j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointers for left and right for this batch_head
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None,:],
            other=0.0,
        ).to(tl.float32)  # convert to fp32

        # load right tile: shape (BLOCK_K, BLOCK_N)
        # note: right row index is j (offset on stride_ri), column index is k (stride_rk)
        # but we want shape (BLOCK_K, BLOCK_N)
        right = tl.load(
            right_base + k_offset[:, None] * stride_rk + j_offset[None, :] * stride_ri,
            mask=mask_k[:, None] & mask_j[None,:],
            other=0.0,
        ).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # Now apply layernorm, out_gate multiplication, maybe after compute out.
    # For now store raw output:
    out_base = out_ptr + b * stride_ob + h * stride_oh
    # store only if i and j valid
    tl.store(
        out_base + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None,:]
    )

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    epsilon: tl.constexpr = 1e-5,
):
   # decode pid
   pid = tl.program_id(0)
   total_heads = B * H
   num_blocks_i = tl.cdiv(N, BLOCK_M)
   num_blocks_j = tl.cdiv(N, BLOCK_N)
   tiles_per_head = num_blocks_i * num_blocks_j
   head_id = pid // tiles_per_head
   tile_id = pid % tiles_per_head
   i_block = tile_id // num_blocks_j
   j_block = tile_id % num_blocks_j

   # compute batch and hidden index
   b = head_id // H
   h = head_id % H

   # offsets
   i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
   j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

   # mask for i/j
   mask_i = i < N
   mask_j = j < N

   # base ptr for left/right/out for this batch/head
   left_base = left_ptr + b * stride_lb + h * stride_lh
   right_base = right_ptr + b * stride_rb + h * stride_rh
   out_base = out_ptr + b * stride_ob + h * stride_oh

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over k
   for k in range(0, tl.cdiv(N, BLOCK_K)):
       kk = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = kk < N

       # left tile (i,k)
       left = tl.load(left_base + i[:, None] * stride_li + kk[None, :] * stride_lk,
                      mask=mask_i[:, None] & mask_k[None, :],
                      other=0.0).to(tl.float32)

       # right tile (j,k) transposed: shape (BLOCK_K, BLOCK_N)
       right = tl.load(right_base + kk[:, None] * stride_rk + j[None, :] * stride_ri,
                       mask=mask_k[:, None] & mask_j[None, :],
                       other=0.0).to(tl.float32)

       # accumulate dot product
       acc += tl.dot(left, right)

   # store result
   # convert to half
   out_tile = acc.to(tl.float16)
   # compute store pointer offsets:
   # out offset: out_base + i * stride_oi + j * stride_oj

   tl.store(out_base + i[:, None] * stride_oi + j[None, :] * stride_oj,
            out_tile,
            mask=mask_i[:, None] & mask_j[None, :])

"""
Optimized Triton implementation of the *outgoing* TriMul operator (AlphaFold3).

The algorithm follows the reference implementation but moves the cubic
N³ “einsum” (left @ rightᵀ across the pair‑wise axis) to a hand‑written
batched matrix‑multiply kernel.  This eliminates the costly
`permute(...).contiguous()` copies required by the naïve `torch.bmm`
approach while still using FP16 + Tensor‑Core acceleration.  The heavy
element‑wise steps (masking, gating, final linear) are kept in native
PyTorch which already provides highly‑optimized kernels.  Only the
core O(B·H·N³) work is performed in Triton.

Key ideas
---------
* Input LayerNorm in FP32 → cast to FP16.
* Five linear projections (two for the pairwise tensors, three for the
  sigmoid gates) are computed with `torch.nn.functional.linear`.
* Optional mask is applied by a simple broadcasted multiply.
* Gating (`left = left * left_gate`, `right = right * right_gate`) is
  performed with a single in‑place torch multiplication.
* Core compute: for each head `h` (hidden_dim) and batch `b`,
  compute `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]` using a
  custom Triton kernel that tiles the `i`, `j` and `k` dimensions.
  The kernel works on half‑precision inputs, accumulates in FP32,
  and stores the result back in FP16.
* Post‑processing: LayerNorm over the hidden dimension, multiplication
  with the output gate, and a final linear projection back to `dim`.
  The final result is returned in FP32 to match the reference module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul N³ term
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # tensor pointers
    left_ptr,      # fp16*, shape [B, N, N, H]
    right_ptr,     # fp16*, shape [B, N, N, H]
    out_ptr,       # fp16*, shape [B, N, N, H]

    # shape & dimension constants
    B: tl.int64,
    N: tl.int64,
    H: tl.int64,

    # stride information for left/right/out (in elements, NOT bytes)
    stride_lb, stride_li, stride_lk, stride_lh,   # left
    stride_rb, stride_ri, stride_rk, stride_rh,   # right
    stride_ob, stride_oi, stride_oj, stride_oh,   # out

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute (B*H) independent matrix multiplications:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    The input tensors are layout‑contiguous [B,N,N,H]; the kernel
    accesses them via the supplied strides.
    """
    pid = tl.program_id(0)

    # ---- decode program ID into (batch_head, tile_i, tile_j) ----
    total_heads = B * H
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    tiles_per_head = num_blocks_i * num_blocks_j

    head_id = pid // tiles_per_head                # 0 … (B*H)-1
    tile_id = pid % tiles_per_head

    i_block = tile_id // num_blocks_j
    j_block = tile_id % num_blocks_j

    # batch and hidden indices from flattened head_id
    b = head_id // H
    h = head_id % H

    # ---- tile offsets ----
    i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ---- base pointers for this (b, h) -------------------------------------------------
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # ---- accumulator (FP32 for numerical stability) -----------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- tiled K‑loop -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile:   shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base + i[:, None] * stride_li + k_off[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile:  we need (BLOCK_K, BLOCK_N) = transpose of (j,k)
        right = tl.load(
            right_base + k_off[:, None] * stride_rk + j[None, :] * stride_ri,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # matrix‑multiply‑accumulate
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---- write the result ----------------------------------------------------------------
    out_tile = acc.to(tl.float16)
    tl.store(
        out_base + i[:, None] * stride_oi + j[None, :] * stride_oj,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right tensors with their corresponding gates
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        left  = left  * left_gate
        right = right * right_gate
    Operates on flattened views.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # left side
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    # right side
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], dtype=float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the heavy compute
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Load model weights (half‑precision)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, hidden_dim], fp16
    right = F.linear(x_h, w_right_proj)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gate pre‑activations (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [B, N, N, 1] for broadcasting
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates (in‑place) via a tiny Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK_GATE = 65536  # ~64k elements per program
    grid_gate = ((numel + BLOCK_GATE - 1) // BLOCK_GATE,)

    apply_gates_kernel[grid_gate](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK_GATE,
    )
    # free gates that are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation – batched GEMM across (batch × hidden_dim) heads
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # [B, N, N, hidden_dim] fp16

    # strides for indexed loads/stores (in elements)
    s_lb, s_li, s_lk, s_lh = left.stride()   # (B, i, k, h)
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # ------------------------------------------------------------------
    # Choose tile sizes based on sequence length (tuned for H100)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:   # up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    n_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    n_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    total_heads = B * hidden_dim
    grid = (total_heads * n_blocks_i * n_blocks_j,)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 🔟 Multiply by the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim], fp16

    # Return float32 to match the reference module
    return final.to(torch.float32)
```
