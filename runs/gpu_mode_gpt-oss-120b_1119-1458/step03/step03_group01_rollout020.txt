@triton.jit
def gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.constexpr,
    apply_mask: tl.constexpr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    # compute mask factor
    if apply_mask:
        mask_idx = offs // hidden_dim
        mask_factor = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
    else:
        mask_factor = 1.0

    # left side
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg * mask_factor, mask=mask)

    # right side
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg * mask_factor, mask=mask)

"""
Highly‑optimized forward pass of the outgoing TriMul operator (AlphaFold3 style).

Algorithm
---------
1. Input LayerNorm (float32 -> half)                         (F.layer_norm)
2. Fuse the two linear projections (proj & gating) per side   (F.linear)
   * left  : concat(left_proj, left_gate)  →  split → sigmoid(gate)
   * right : concat(right_proj, right_gate) →  split → sigmoid(gate)
3. Compute the output‑gate (sigmoid)                           (F.linear)
4. Fuse masking (if present) and gate multiplication for *both*
   sides in a single Triton kernel to avoid extra memory traffic.
5. Core N³ operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matrix‑multiply (torch.bmm) after
   reshaping to (B*H, N, N) where H = hidden_dim.
6. Output LayerNorm (half‑precision weights)               (F.layer_norm)
7. Scale by the output‑gate in‑place (Triton kernel).
8. Final linear projection back to `dim`.                     (F.linear)
9. Cast the result to float32 (the reference implementation works
   in float32).

Both Triton kernels are deliberately lightweight:
* `gate_fuse_kernel` multiplies *left*/*right* with their gates and
  optionally applies the broadcast mask in‑place.
* `mul_inplace_kernel` multiplies the post‑norm tensor with the
  output‑gate in‑place.

All heavy matrix‑multiplications use FP16 (Tensor‑Core friendly) while
the LayerNorm statistics are computed in the same dtype for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def gate_fuse_kernel(
    left_ptr,                # fp16*   left tensor   (flattened)
    left_gate_ptr,           # fp16*   left gate   (flattened)
    right_ptr,               # fp16*   right tensor (flattened)
    right_gate_ptr,          # fp16*   right gate  (flattened)
    mask_ptr,                # fp16*   mask (flattened, size B*N*N)
    hidden_dim: tl.constexpr,    # compile‑time hidden dimension
    apply_mask: tl.constexpr,    # 0 / 1 – whether the mask is active
    numel: tl.int64,            # total number of elements per tensor
    BLOCK: tl.constexpr,        # block size (tiles per program)
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask (if present)
        right = right * right_gate * mask (if present)
    The tensors are already flattened, the mask is broadcast across
    `hidden_dim` and stored as a flat vector of size B*N*N.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # ---------- mask factor ----------
    mask_factor = 1.0
    if apply_mask:
        # each group of `hidden_dim` contiguous entries share one mask value
        mask_idx = offs // hidden_dim
        mask_factor = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # ---------- left side ----------
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg * mask_factor, mask=mask)

    # ---------- right side ----------
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg * mask_factor, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,        # fp16*   tensor to be scaled   (flattened)
    b_ptr,        # fp16*   scaling tensor        (flattened)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    Element‑wise in‑place multiplication:
        a = a * b
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        - mask         : torch.Tensor  [B, N, N]        (bool/float)
        - weights      : dict of torch.Tensor model parameters
        - config       : dict containing at least
                         `'dim'`, `'hidden_dim'` and optional `'nomask'`

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack & config
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → half for the rest)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    x = F.layer_norm(
        input_tensor.to(torch.float16),          # convert once
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse proj + gate linear layers per side (half‑precision)
    # ------------------------------------------------------------------
    # left: [proj | gate] → split
    w_left_combined = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    left_combined = F.linear(x, w_left_combined)               # (B,N,N,2H)
    left = left_combined[..., :hidden_dim]                    # (B,N,N,H)
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])  # (B,N,N,H)

    # right: [proj | gate] → split
    w_right_combined = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)   # (2*H, dim)

    right_combined = F.linear(x, w_right_combined)              # (B,N,N,2H)
    right = right_combined[..., :hidden_dim]                   # (B,N,N,H)
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])  # (B,N,N,H)

    # 3️⃣ Output‑gate (sigmoid)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask (if any) and gate multiplications via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # broadcast mask over the hidden dimension and flatten
        mask_flat = mask_tensor.unsqueeze(-1).to(torch.float16).view(-1)  # [B*N*N]
    else:
        # dummy placeholder – never read because `apply_mask=False`
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)

    # Ensure tensors are contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elems = left.numel()
    BLOCK = 65536
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    gate_fuse_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        mask_flat,
        hidden_dim,
        int(not nomask),      # apply_mask compile‑time flag
        total_elems,
        BLOCK=BLOCK,
    )
    # Gates are no longer needed
    del left_gate, right_gate, mask_flat

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ operation (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim
    total_heads = B * H

    # reshape to (total_heads, N, N)   where the hidden dimension is the batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # (total_heads, N, N) @ (total_heads, N, N)^T → (total_heads, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore layout [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (half‑precision weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Scale by the output gate (in‑place Triton)
    # ------------------------------------------------------------------
    out_gate = out_gate.contiguous()
    out = out.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, dim]  fp16

    # Return with the same dtype as the reference implementation
    return final.to(torch.float32)
```
