@triton.jit
def linear_gate_mask_kernel(
    x_ptr, w_proj_ptr, w_gate_ptr, mask_ptr, out_ptr,
    M, K, H,
    stride_xm, stride_xk,
    stride_wp_h, stride_wp_k,
    stride_wg_h, stride_wg_k,
    stride_mask_m,
    stride_out_m, stride_out_h,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,  # hidden tile size
):
    pid = tl.program_id(axis=0)
    # define offsets for rows
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_mask = offs_m < M

    # loop over K dimension tiles
    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # load x tile
        a = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk),
                    mask=row_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # load w_proj tile (H, K) we need (BLOCK_H, BLOCK_K) where hidden dimension block size is BLOCK_H
        w_p = tl.load(w_proj_ptr + (tl.arange(0, BLOCK_H)[:, None] * stride_wp_h
                                    + offs_k[None, :] * stride_wp_k),
                      mask=(tl.arange(0, BLOCK_H)[:, None] < H) & k_mask[None, :],
                      other=0.0)

        w_g = tl.load(w_gate_ptr + (tl.arange(0, BLOCK_H)[:, None] * stride_wg_h
                                    + offs_k[None, :] * stride_wg_k),
                      mask=(tl.arange(0, BLOCK_H)[:, None] < H) & k_mask[None, :],
                      other=0.0)

        acc_proj += tl.dot(a, w_p) # (BLOCK_M, BLOCK_H) from (M,K)*(K,H)
        acc_gate += tl.dot(a, w_g)
    # after loops: convert to FP16
    acc_proj = acc_proj.to(tl.float16)
    acc_gate = acc_gate.to(tl.float16)

    # apply sigmoid to gate
    acc_gate = tl.sigmoid(acc_gate) # maybe need custom compute: 1/(1+exp(-x))

    # mask scaling
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m, mask=row_mask, other=1.0)  # default mask=1 (no effect)
        acc_proj = acc_proj * mask_val[:, None]

    # final out = proj * gate
    out = acc_proj * acc_gate
    # store out
    tl.store(out_ptr + (offs_m[:, None] * stride_out_m + tl.arange(0, BLOCK_H)[None, :] * stride_out_h),
             out,
             mask=row_mask[:, None] & (tl.arange(0, BLOCK_H)[None, :] < H))
```

    x_norm = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)
    B, N, _, _ = x_norm.shape
    M = B * N * N

    # flatten x_norm to (M, dim)
    x_flat = x_norm.view(M, dim)

    # Get weight tensors and cast to float16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)

    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Flatten mask if needed
    if not nomask:
        mask_fp16 = mask.to(torch.float16).view(M)
    else:
        mask_fp16 = None

    left_flat = torch.empty((M, hidden), dtype=torch.float16, device=device)
    right_flat = torch.empty((M, hidden), dtype=torch.float16, device=device)

    # fused linear+gate+mask for left
    fused_linear_gate_mask(x_flat, left_proj_w, left_gate_w,
                           mask_fp16, left_flat, has_mask=not nomask)

    # fused linear+gate+mask for right
    fused_linear_gate_mask(x_flat, right_proj_w, right_gate_w,
                           mask_fp16, right_flat, has_mask=not nomask)

    # Reshape left and right to original shape
    left = left_flat.view(B, N, N, hidden)
    right = right_flat.view(B, N, N, hidden)

    # out_gate
    out_gate_raw = F.linear(x_norm, out_gate_w)
    out_gate = torch.sigmoid(out_gate_raw)  # shape [B,N,N,hidden]

    # Contraction
    left_perm = left.permute(0, 3, 1, 2)   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, hidden, N, N)

    # Batched matmul
    out_hidden = torch.matmul(left_perm, right_perm)  # (B, hidden, N, N)

    # Restore layout
    out = out_hidden.permute(0, 2, 3, 1).contiguous() # (B, N, N, hidden)

    # Output layer norm
    out_norm = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"].to(torch.float16),
                            bias=weights["to_out_norm.bias"].to(torch.float16), eps=eps)

    # Gating
    out_norm = out_norm * out_gate

    # Final projection: fmap to dim
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to original dtype
    return out_final.to(inp.dtype)
```

```
@triton.jit
def fused_proj_gate_mask_kernel(
    x_ptr, w_proj_ptr, w_gate_ptr, mask_ptr, out_ptr,
    M, K, H,
    stride_xm, stride_xk,
    stride_wp_h, stride_wp_k,
    stride_wg_h, stride_wg_k,
    stride_mask_m,
    stride_out_m, stride_out_h,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of x
        a = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load weight tiles (K, H) transposed
        w_proj_tile = tl.load(w_proj_ptr + offs_k[:, None] * stride_wp_k + offs_h[None, :] * stride_wp_h,
                              mask=mask_k[:, None] & mask_h[None, :], other=0.0)
        w_gate_tile = tl.load(w_gate_ptr + offs_k[:, None] * stride_wg_k + offs_h[None, :] * stride_wg_h,
                              mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        acc_proj += tl.dot(a, w_proj_tile)
        acc_gate += tl.dot(a, w_gate_tile)

    # Convert to fp16
    proj = acc_proj.to(tl.float16)
    gate = acc_gate.to(tl.float16)

    # Sigmoid activation
    gate = tl.sigmoid(gate)    # approx tanh maybe uses built-in
    # Equivalent to 1/(1+exp(-x)). Use tl.sigmoid.

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m,
                           mask=mask_m,
                           other=1.0)
        proj = proj * mask_val[:, None]

    out = proj * gate

    out_ptrs = out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_h[None, :])

def fused_five_linear_kernel(
    x_ptr,
    w_lp_ptr, w_lg_ptr, w_rp_ptr, w_rg_ptr, w_og_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    M, K, H,
    stride_xm, stride_xk,
    stride_wlp_h, stride_wlp_k,
    stride_wlg_h, stride_wlg_k,
    stride_wrp_h, stride_wrp_k,
    stride_wrg_h, stride_wrg_k,
    stride_wog_h, stride_wog_k,
    stride_mask_m,
    stride_left_out_m, stride_left_out_h,
    stride_right_out_m, stride_right_out_h,
    stride_out_gate_m, stride_out_gate_h,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # accumulator
    acc_lp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_k[None,:] * stride_xk,
                    mask=mask_m[:, None] & mask_k[None,:], other=0.0)

        # Load weight tiles: shape (BLOCK_K, BLOCK_H)
        w_lp_tile = tl.load(w_lp_ptr + offs_k[:, None] * stride_wlp_k + offs_h[None,:] * stride_wlp_h,
                            mask=mask_k[:, None] & mask_h[None,:], other=0.0)
        w_lg_tile = tl.load(w_lg_ptr + offs_k[:, None] * stride_wlg_k + offs_h[None,:] * stride_wlg_h,
                            mask=mask_k[:, None] & mask_h[None,:], other=0.0)
        w_rp_tile = tl.load(w_rp_ptr + offs_k[:, None] * stride_wrp_k + offs_h[None,:] * stride_wrp_h,
                            mask=mask_k[:, None] & mask_h[None,:], other=0.0)
        w_rg_tile = tl.load(w_rg_ptr + offs_k[:, None] * stride_wrg_k + offs_h[None,:] * stride_wrg_h,
                            mask=mask_k[:, None] & mask_h[None,:], other=0.0)
        w_og_tile = tl.load(w_og_ptr + offs_k[:, None] * stride_wog_k + offs_h[None,:] * stride_wog_h,
                            mask=mask_k[:, None] & mask_h[None,:], other=0.0)

        acc_lp += tl.dot(a, w_lp_tile)
        acc_lg += tl.dot(a, w_lg_tile)
        acc_rp += tl.dot(a, w_rp_tile)
        acc_rg += tl.dot(a, w_rg_tile)
        acc_og += tl.dot(a, w_og_tile)

    # Convert to fp16
    lp = acc_lp.to(tl.float16)
    lg = acc_lg.to(tl.float16)
    rp = acc_rp.to(tl.float16)
    rg = acc_rg.to(tl.float16)
    og = acc_og.to(tl.float16)

    # sigmoid activation
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))
    og_sig = 1.0 / (1.0 + tl.exp(-og))

    # Apply mask if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)   # shape (BLOCK_M)
        mask_val = mask_val[:, None]   # broadcast
        lp = lp * mask_val
        rp = rp * mask_val

    left = lp * lg_sig
    right = rp * rg_sig

    # Store left, right, out_gate
    left_ptr = left_out_ptr + offs_m[:, None] * stride_left_out_m + offs_h[None,:] * stride_left_out_h
    tl.store(left_ptr, left, mask=mask_m[:, None] & mask_h[None,:])

    right_ptr = right_out_ptr + offs_m[:, None] * stride_right_out_m + offs_h[None,:] * stride_right_out_h
    tl.store(right_ptr, right, mask=mask_m[:, None] & mask_h[None,:])

    og_ptr = out_gate_out_ptr + offs_m[:, None] * stride_out_gate_m + offs_h[None,:] * stride_out_gate_h
    tl.store(og_ptr, og_sig, mask=mask_m[:, None] & mask_h[None,:])
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The dominant cost of the outgoing triangle update is the series of
5 linear layers (left/right projections and their gates and the output
gate) followed by a massive N³ contraction.  The original reference
implementation performs each linear layer with a separate torch call,
causing the 4‑D input tensor (B·N²·C) to be streamed from memory five
times.

We replace those five matmuls with **one fused Triton kernel** that:
* loads each row of the flattened input only once,
* multiplies it with all five weight matrices (left‑proj, left‑gate,
  right‑proj, right‑gate, out‑gate),
* applies a sigmoid to the three gate results,
* multiplies the two projections by their gates and by the optional mask,
* writes three output buffers (left, right, out_gate).

All arithmetic is performed in FP16 with FP32 accumulation for
numerical stability.  The large N³ contraction is kept as a batched
matrix‑multiply (`torch.matmul`), which cuBLAS executes extremely fast
on an H100.  The remaining element‑wise operations (output LayerNorm,
output gating and final projection) are left to PyTorch because they are
cheap relative to the contraction.

Only the fused linear‑gate‑mask kernel is custom‑written, satisfying the
requirement that “part of the operations” run in Triton while keeping
the code simple and correct for all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: 5 linear layers + gate + mask in a single pass
# ----------------------------------------------------------------------
@triton.jit
def fused_five_linear_kernel(
    # Pointers
    x_ptr,                     # (M, K)                – flattened input (B·N·N, dim)
    wlp_ptr, wlg_ptr,          # (H, K)                – left proj / gate
    wrp_ptr, wrg_ptr,          # (H, K)                – right proj / gate
    wog_ptr,                   # (H, K)                – out gate
    mask_ptr,                  # (M,)                  – optional mask (broadcasted)
    left_out_ptr, right_out_ptr, out_gate_ptr,   # (M, H) results
    # Shapes
    M, K, H,
    # Strides
    stride_xm, stride_xk,
    stride_wlp_h, stride_wlp_k,
    stride_wlg_h, stride_wlg_k,
    stride_wrp_h, stride_wrp_k,
    stride_wrg_h, stride_wrg_k,
    stride_wog_h, stride_wog_k,
    stride_mask_m,
    stride_lo_m, stride_lo_h,
    stride_ro_m, stride_ro_h,
    stride_og_m, stride_og_h,
    # Compile‑time constants
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_K: tl.constexpr,    # reduction dimension tile
    BLOCK_H: tl.constexpr,    # hidden‑dim tile
):
    pid_m = tl.program_id(0)   # block over rows (M)
    pid_h = tl.program_id(1)   # block over hidden dimension (H)

    # ------------------------------------------------------------------
    # Offsets and masks for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Accumulators (FP32 for accuracy)
    # ------------------------------------------------------------------
    acc_lp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop: multiply the input row with each weight matrix
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of the input (M×K)
        a = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # shape (BLOCK_M, BLOCK_K)

        # Load transposed weight tiles (K×H) for the five linear maps
        wlp = tl.load(wlp_ptr + offs_k[:, None] * stride_wlp_k + offs_h[None, :] * stride_wlp_h,
                      mask=mask_k[:, None] & mask_h[None, :],
                      other=0.0)                      # (BLOCK_K, BLOCK_H)
        wlg = tl.load(wlg_ptr + offs_k[:, None] * stride_wlg_k + offs_h[None, :] * stride_wlg_h,
                      mask=mask_k[:, None] & mask_h[None, :],
                      other=0.0)
        wrp = tl.load(wrp_ptr + offs_k[:, None] * stride_wrp_k + offs_h[None, :] * stride_wrp_h,
                      mask=mask_k[:, None] & mask_h[None, :],
                      other=0.0)
        wrg = tl.load(wrg_ptr + offs_k[:, None] * stride_wrg_k + offs_h[None, :] * stride_wrg_h,
                      mask=mask_k[:, None] & mask_h[None, :],
                      other=0.0)
        wog = tl.load(wog_ptr + offs_k[:, None] * stride_wog_k + offs_h[None, :] * stride_wog_h,
                      mask=mask_k[:, None] & mask_h[None, :],
                      other=0.0)

        # Accumulate the four projections and the out‑gate
        acc_lp += tl.dot(a, wlp)   # (BLOCK_M, BLOCK_H)
        acc_lg += tl.dot(a, wlg)
        acc_rp += tl.dot(a, wrp)
        acc_rg += tl.dot(a, wrg)
        acc_og += tl.dot(a, wog)

    # ------------------------------------------------------------------
    # Convert to FP16 and apply sigmoid where needed
    # ------------------------------------------------------------------
    lp = acc_lp.to(tl.float16)
    lg = acc_lg.to(tl.float16)
    rp = acc_rp.to(tl.float16)
    rg = acc_rg.to(tl.float16)
    og = acc_og.to(tl.float16)

    # sigmoid(x) = 1 / (1 + exp(-x))
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))
    og_sig = 1.0 / (1.0 + tl.exp(-og))

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m,
                           mask=mask_m,
                           other=1.0)            # (BLOCK_M,)
        mask_val = mask_val[:, None]                # broadcast to (BLOCK_M, 1)
        lp = lp * mask_val
        rp = rp * mask_val

    # Apply gates
    left  = lp * lg_sig
    right = rp * rg_sig

    # ------------------------------------------------------------------
    # Store the three results (left, right, out_gate)
    # ------------------------------------------------------------------
    left_ptr = left_out_ptr + offs_m[:, None] * stride_lo_m + offs_h[None, :] * stride_lo_h
    tl.store(left_ptr, left, mask=mask_m[:, None] & mask_h[None, :])

    right_ptr = right_out_ptr + offs_m[:, None] * stride_ro_m + offs_h[None, :] * stride_ro_h
    tl.store(right_ptr, right, mask=mask_m[:, None] & mask_h[None, :])

    og_ptr = out_gate_ptr + offs_m[:, None] * stride_og_m + offs_h[None, :] * stride_og_h
    tl.store(og_ptr, og_sig, mask=mask_m[:, None] & mask_h[None, :])


def fused_five_linear(x, wlp, wlg, wrp, wrg, wog, mask, left_out, right_out, out_gate_out,
                      has_mask: bool):
    """
    Wrapper around `fused_five_linear_kernel`.  Computes:
        left  = (x @ wlp.T) * sigmoid(x @ wlg.T) * mask
        right = (x @ wrp.T) * sigmoid(x @ wrg.T) * mask
        out_gate = sigmoid(x @ wog.T)

    Arguments:
        x            – (M, K) flattened input   (float16)
        wlp,wlg,…   – weight matrices (H, K)   (float16)
        mask         – (M,) or None (float16)
        left_out, right_out, out_gate_out – pre‑allocated (M, H) buffers
        has_mask     – whether to apply the mask
    """
    M, K = x.shape
    H = wlp.shape[0]

    # Ensure contiguity (required for correct stride handling)
    x = x.contiguous()
    wlp = wlp.contiguous()
    wlg = wlg.contiguous()
    wrp = wrp.contiguous()
    wrg = wrg.contiguous()
    wog = wog.contiguous()
    if has_mask:
        mask = mask.contiguous()
    else:
        # Dummy tensor – never read because HAS_MASK=False
        mask = torch.empty(0, dtype=x.dtype, device=x.device)

    # Strides (in elements, not bytes)
    stride_xm, stride_xk = x.stride()
    stride_wlp_h, stride_wlp_k = wlp.stride()
    stride_wlg_h, stride_wlg_k = wlg.stride()
    stride_wrp_h, stride_wrp_k = wrp.stride()
    stride_wrg_h, stride_wrg_k = wrg.stride()
    stride_wog_h, stride_wog_k = wog.stride()
    stride_mask_m = mask.stride(0) if has_mask else 0

    stride_lo_m, stride_lo_h = left_out.stride()
    stride_ro_m, stride_ro_h = right_out.stride()
    stride_og_m, stride_og_h = out_gate_out.stride()

    # Tile sizes – tuned for H100, adjust if needed
    BLOCK_M = 128
    BLOCK_K = 32
    BLOCK_H = 128  # covers most hidden sizes; larger hidden dims get split

    grid = ((tl.cdiv(M, BLOCK_M), tl.cdiv(H, BLOCK_H)))   # 2‑D grid

    fused_five_linear_kernel[grid](
        x,
        wlp, wlg,
        wrp, wrg,
        wog,
        mask,
        left_out,
        right_out,
        out_gate_out,
        M, K, H,
        stride_xm, stride_xk,
        stride_wlp_h, stride_wlp_k,
        stride_wlg_h, stride_wlg_k,
        stride_wrp_h, stride_wrp_k,
        stride_wrg_h, stride_wrg_k,
        stride_wog_h, stride_wog_k,
        stride_mask_m,
        stride_lo_m, stride_lo_h,
        stride_ro_m, stride_ro_h,
        stride_og_m, stride_og_h,
        HAS_MASK=has_mask,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    # No return – the three output buffers are filled in‑place.


# ----------------------------------------------------------------------
# TriMul entry point (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]   (float32, CUDA)
            - mask_tensor  : [B, N, N]        (bool/float, CUDA) – optional
            - weights_dict : model parameters (see skeleton)
            - config_dict  : contains at least "dim", "hidden_dim"
                             optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    eps = config.get("eps", 1e-5)
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    has_mask = not nomask

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32 → float16 for the heavy parts)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)                     # (B, N, N, dim)

    B, N, _, _ = x_norm.shape
    M = B * N * N                                          # number of (b,i,k) rows

    # ------------------------------------------------------------------
    # 3) Flatten the input to (M, dim) – each row corresponds to (b,i,k)
    # ------------------------------------------------------------------
    x_flat = x_norm.view(M, dim).contiguous()             # (M, dim)

    # ------------------------------------------------------------------
    # 4) Load all five weight matrices (float16, contiguous)
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Prepare output buffers for the fused kernel
    # ------------------------------------------------------------------
    left_flat      = torch.empty((M, hidden), dtype=torch.float16, device=device)
    right_flat     = torch.empty((M, hidden), dtype=torch.float16, device=device)
    out_gate_flat  = torch.empty((M, hidden), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Optional mask – broadcasted over the hidden dimension inside the kernel
    # ------------------------------------------------------------------
    if has_mask:
        mask_flat = mask.to(torch.float16).view(M).contiguous()
    else:
        mask_flat = None

    # ------------------------------------------------------------------
    # 7) One‐shot fused linear + gate + mask kernel (left, right, out_gate)
    # ------------------------------------------------------------------
    fused_five_linear(
        x_flat,
        w_left_proj, w_left_gate,
        w_right_proj, w_right_gate,
        w_out_gate,
        mask_flat,
        left_flat, right_flat, out_gate_flat,
        has_mask=has_mask,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to the 4‑D layout
    # ------------------------------------------------------------------
    left  = left_flat .view(B, N, N, hidden)
    right = right_flat.view(B, N, N, hidden)
    out_gate = out_gate_flat.view(B, N, N, hidden)   # already sigmoid‑ed

    # ------------------------------------------------------------------
    # 9) Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matmul over the hidden dimension.
    # ------------------------------------------------------------------
    left_perm  = left .permute(0, 3, 1, 2)    # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)    # (B, hidden, k, j)

    # Batched matrix multiplication (cuBLAS does the heavy lifting)
    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # Restore original layout (B, N, N, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 10) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # Apply the output gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate

    # Final projection back to the original `dim`
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)
    )   # bias‑free as in the reference implementation

    # Cast back to the input dtype (usually float32) before returning
    return out_final.to(inp.dtype)
