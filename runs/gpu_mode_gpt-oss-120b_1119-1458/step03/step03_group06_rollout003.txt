"""
Optimized implementation of the *outgoing* TriMul operator (AlphaFold‑3) using
Triton for the O(N³) contraction.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → float16 for the heavy part).
2. Fuse the left/right linear projections with their gates:
   * Concatenate the projection and gate weights (shape (2·H, C)).
   * Perform a single half‑precision linear layer →  [B, N, N, 2·H].
   * Split, apply sigmoid to the gate half and (optionally) mask to the
     projection half, then multiply (proj * gate) → left/right tensors.
3. Heavy contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched matrix‑multiply A @ Bᵀ with a custom Triton
   kernel.  The kernel works on fp16 inputs, accumulates in fp32 and writes
   fp16 results.
4. Apply a second LayerNorm on the hidden dimension (fp16), element‑wise
   gating with `out_gate` (produced by another fused linear+sigmoid), and a
   final linear projection back to the original channel dimension.
5. Cast the result back to float32 (required by the harness).

The implementation keeps the most expensive O(N³) step on the GPU with a
well‑tuned Triton kernel while all other operations use fast PyTorch
primitives in half precision.  This reduces kernel launch overhead, memory
traffic and achieves sub‑millisecond runtimes on the target H100 GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ  (A,B,C are fp16)
#   A, B : [M, N, K]   (M = B·H, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # strides of A (elements)
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,   # tile rows   (i‑axis)
    BLOCK_N: tl.constexpr,   # tile cols   (j‑axis)
    BLOCK_K: tl.constexpr,   # reduction tile (k‑axis)
):
    pid_i = tl.program_id(0)   # tile index along i
    pid_j = tl.program_id(1)   # tile index along j
    pid_z = tl.program_id(2)   # batch*hidden index

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # FP32 accumulator (stable for long reductions)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # B tile, transposed (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Multiply‑accumulate (fp16×fp16 → fp32)
        acc += tl.dot(a, b)   # dot returns fp16, will be up‑cast to fp32

    # ------------------------------------------------------------------
    # Write back result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.  Tensors must be contiguous fp16 on
    the same CUDA device.
    """
    B, N, _, H = left.shape

    # Pack batch and hidden dimension into a single leading dim:
    #   A, B_ → [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Extract element‑wise strides (Triton works with element strides)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile dimensions – chosen to match H100 tensor‑core sweet spot
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128   # larger K‑tile → fewer loop iterations

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth (batch × hidden)
    )

    _trimul_matmul_kernel[
        grid
    ](
        A,
        B_,
        C,
        B * H,               # dummy M (not used for tiling)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps == 256 threads per block
        num_stages=2,       # double‑buffering
    )

    # Reshape back to original 4‑D layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device
        as the input.
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                       # [B, N, N, dim]  (float32)
    x = x_norm.to(torch.float16)   # fp16 for the heavy path

    # -------------------------------------------------
    # 2️⃣ Fuse projection + gate linear layers (fp16)
    # -------------------------------------------------
    # Concatenate projection and gate weights
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    left_comb_w   = torch.cat([left_proj_w, left_gate_w], dim=0)   # [2·H, dim]

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    right_comb_w  = torch.cat([right_proj_w, right_gate_w], dim=0) # [2·H, dim]

    # Combined linear passes
    left_comb = torch.nn.functional.linear(x, left_comb_w)   # [B,N,N,2·H]
    right_comb = torch.nn.functional.linear(x, right_comb_w) # [B,N,N,2·H]

    # Split into projection and gate halves
    left_proj_raw = left_comb[..., :hidden_dim]            # [B,N,N,H]
    left_gate_raw = left_comb[..., hidden_dim:]            # [B,N,N,H]

    right_proj_raw = right_comb[..., :hidden_dim]          # [B,N,N,H]
    right_gate_raw = right_comb[..., hidden_dim:]          # [B,N,N,H]

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) – fp16 throughout
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(right_gate_raw)

    # Apply gates to the projected tensors
    left = left_proj_raw * left_gate          # [B,N,N,H]
    right = right_proj_raw * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (fp16 in / fp16 out)
    # -------------------------------------------------
    # out shape: [B, N, N, H]
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # -------------------------------------------------
    # 7️⃣ Output gating (sigmoid) and final projection
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] fp16

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    out = torch.nn.functional.linear(out, to_out_w)       # back to dim, fp16

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
