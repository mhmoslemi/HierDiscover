@triton.jit
def fused_mul3_masked_kernel(a_ptr, b_ptr, c_ptr, out_ptr, hidden: tl.constexpr, size: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)
    mask_idx = offs // hidden
    b = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)
    out = a * c * b
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch algorithm but reduces
memory traffic and kernel launch overhead by

1️⃣ Fusing the projection and gate linear layers:
   * left_proj + left_gate  → one `torch.nn.functional.linear`
   * right_proj + right_gate → one `torch.nn.functional.linear`

2️⃣ Fusing the mask, gate and projection multiplication in a tiny Triton
   kernel (`fused_mul3_masked`).  The mask is broadcast on‑the‑fly over the
   hidden dimension, so no extra memory is allocated for `mask[...,None]`.

3️⃣ Performing the core contraction as a batched GEMM over the hidden
   channel (`torch.matmul`).  After permuting to (B, hidden, N, N) the
   operation is equivalent to

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   which cuBLAS executes with Tensor‑cores (FP16).

4️⃣ Keeping the heavy arithmetic in FP16 and only casting the final
   result back to the input dtype (usually FP32).

The kernel only implements the element‑wise triple product; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,            # left_proj   (B,N,N,hidden)
    b_ptr,            # mask        (B,N,N)
    c_ptr,            # gate        (B,N,N,hidden)
    out_ptr,          # destination (B,N,N,hidden)
    hidden: tl.constexpr,
    total:  tl.constexpr,      # total number of elements in a / c
    BLOCK_SIZE: tl.constexpr, # thread‑block size
):
    """Element‑wise multiplication a * c * mask where mask is broadcast over
    the trailing hidden dimension.

    All pointers point to contiguous memory and the tensors are FP16.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load left projection and gate (both have shape (total,))
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    g = tl.load(c_ptr + offs, mask=mask, other=0.0)

    # Compute corresponding mask index (broadcast over hidden)
    mask_idx = offs // hidden
    m = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(a: torch.Tensor,
                      mask: torch.Tensor,
                      gate: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """Convenience wrapper for the Triton kernel.
    a, gate: (B,N,N,hidden) FP16, contiguous
    mask:    (B,N,N)    FP16, contiguous
    Returns a * gate * mask broadcasted over `hidden`.
    """
    assert a.shape == gate.shape, "a and gate must have identical shapes"
    B, N, _, H = a.shape
    assert H == hidden, "hidden dimension mismatch"
    total = a.numel()
    # Choose a block size that fits well on H100 Tensor‑core lanes.
    BLOCK = 32768  # 32 KB of 16‑bit elements ≈ 64 KB per block → good occupancy
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    out = torch.empty_like(a)
    fused_mul3_masked_kernel[grid](
        a, mask, gate, out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operation.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  shape [B, N, N] (bool/float, CUDA)
            - weights_dict: model weights with keys:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight"
                * "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict: must contain "dim", "hidden_dim".
                Optional: "eps" (float, default 1e-5), "nomask" (bool).

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    device = inp.device

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32 → float16 for later ops)
    # ------------------------------------------------------------------
    # NOTE: LayerNorm expects weight/bias with same dtype as input.
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 2) Fused projection + gate linear layers (still FP16)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights for left and right streams.
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)      # (2*hidden, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)      # (2*hidden, dim)

    # Combined linear calls (no bias)
    left_combined = F.linear(x_norm, left_combined_w)   # (B,N,N,2*hidden)
    right_combined = F.linear(x_norm, right_combined_w) # (B,N,N,2*hidden)

    # Split into projection and gate tensors
    left_proj_raw = left_combined[..., :hidden]
    left_gate_raw = left_combined[..., hidden:]

    right_proj_raw = right_combined[..., :hidden]
    right_gate_raw = right_combined[..., hidden:]

    # Output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm, weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    )  # (B,N,N,hidden)

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 3) Mask + gate fusion (mask optional)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise product
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask once to FP16; Triton kernel will broadcast.
        mask_fp16 = mask.to(torch.float16, non_blocking=True)
        left = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # ------------------------------------------------------------------
    # 4) Core contraction: batched GEMM over the hidden channel
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, hidden, N, N) and perform batched matmul.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, k, j)

    # (B*hidden, N, N) @ (B*hidden, N, N) → (B*hidden, N, N)
    out_hidden = torch.matmul(left_perm, right_perm)

    # Restore original layout: (B, N, N, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating, final projection (all FP16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16, non_blocking=True),
        bias=weights["to_out_norm.bias"].to(torch.float16, non_blocking=True),
        eps=eps,
    )
    # Apply the output gate (broadcast over hidden dim)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )  # (B,N,N,dim), bias‑free

    # Cast back to the original dtype (float32) before returning
    return out_final.to(inp.dtype)
