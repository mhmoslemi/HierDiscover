def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # First layernorm (float32)
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"])

    # Cast to FP16 and move to device
    x = x.to(torch.float16)

    # Concatenate projection and gating weights into one matrix
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    # each shape: (hidden_dim, dim)
    weight_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)  # shape (5*hidden, dim)

    # Linear pass
    fused = torch.nn.functional.linear(x, weight_concat)  # [B,N,N,5*hidden]
    # Split
    fused = fused.view(x.shape[0], x.shape[1], x.shape[2], 5, hidden_dim)  # [B,N,N,5,hidden]
    left = fused[..., 0, :]  # [B,N,N,hidden]
    left_gate = fused[..., 1, :]  # [B,N,N,hidden]
    right = fused[..., 2, :]
    right_gate = fused[..., 3, :]
    out_gate = fused[..., 4, :]

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Optional mask
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Heavy contraction: compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    # Reshape to (B*hidden, N, N)
    B, N, _, H = left.shape
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)

    # Use torch.bmm for each batch: (B*H, N, N) @ (B*H, N, N)^T
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1,2))
    out = out_reshape.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # Second LayerNorm
    out = torch.nn.functional.layer_norm(out,
                                        (hidden_dim,),
                                        weight=weights["to_out_norm.weight"].to(torch.float16),
                                        bias=weights["to_out_norm.bias"].to(torch.float16))

    # Apply out_gate gating
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast to float32
    return out.to(torch.float32)

"""
high‑performance outgoing TriMul (AlphaFold‑3) forward pass

Algorithm
---------
1️⃣  Layer‑norm the input (float32) – keep the statistics in high precision.
2️⃣  Cast to FP16 and fuse the four linear projections (left/right) and the three
    gating projections into a *single* `torch.nn.functional.linear`.  This halves
    the number of passes over the massive `[B, N, N, dim]` tensor and saves a
    large amount of memory bandwidth.
3️⃣  Split the fused result, apply σ‑gates and the optional pair‑wise mask.
4️⃣  Compute the O(N³) contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    – for small `N` (≤256) a hand‑written Triton kernel is used;
    – for larger `N` we fall back to batched cuBLAS GEMM (`torch.bmm`) which
    is far faster on H100 for big matrices.
5️⃣  Second Layer‑norm (FP16) → out‑gate (σ) → final linear projection to `dim`.
6️⃣  Cast the result back to float32 (as required by the harness).

The heavy O(N³) contraction is the only part written in Triton; the rest of
the pipeline leverages highly‑optimised PyTorch/cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B,C are fp16, accumulator fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row index (i)
    pid_n = tl.program_id(1)   # tile col index (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # -----------------------------------------------------------------
        # Load A tile: shape (BLOCK_M, BLOCK_K)
        # -----------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # -----------------------------------------------------------------
        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        # -----------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision MMA: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store C tile (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    `left` & `right` are FP16 tensors of shape [B, N, N, H].
    Returns `out` of the same shape using the Triton kernel above.
    """
    B, N, _, H = left.shape
    # Merge batch & hidden into a single leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # strides (element‑wise, not byte)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # TILE configuration – 64×64×64 works well on H100 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor [B, N, N] (float32/bool) – may be unused.
        weights_dict : mapping name → torch.Tensor (float32)
        config_dict  : must contain ``dim`` and ``hidden_dim``.
                       Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ Layer‑norm (float32) – most stable for the huge input
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy work
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fuse the four linear projections + three gates into a single matmul
    #    Order: left_proj, left_gate, right_proj, right_gate, out_gate
    # -------------------------------------------------
    w_left_proj  = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate  = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate   = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension and cast once to fp16
    fused_weight = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden, dim)

    # Single linear pass – massive memory‑bandwidth win
    fused = torch.nn.functional.linear(x, fused_weight)  # [B,N,N,5*hidden]

    # Reshape to isolate the five groups
    fused = fused.view(x.shape[0], x.shape[1], x.shape[2], 5, hidden_dim)
    left       = fused[..., 0, :]  # [B,N,N,hidden]
    left_gate  = fused[..., 1, :]  # pre‑sigmoid
    right      = fused[..., 2, :]
    right_gate = fused[..., 3, :]  # pre‑sigmoid
    out_gate   = fused[..., 4, :]  # pre‑sigmoid

    # Apply the σ activations (in‑place to save memory)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Fuse the gates into the projections (in‑place)
    # -------------------------------------------------
    torch.mul(left, left_gate, out=left)
    torch.mul(right, right_gate, out=right)

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction
    #    Use a Triton kernel for small N, otherwise fall back to cuBLAS batched GEMM
    # -------------------------------------------------
    B, N, _, H = left.shape
    use_triton = N <= 256  # Triton launch overhead dominates for larger N
    if use_triton:
        out = _batched_matmul_triton(left, right)          # [B,N,N,hidden] fp16
    else:
        # Reshape to (B*hidden, N, N) and use torch.bmm (cuBLAS) – highly optimized
        left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
        right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
        out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (in FP16) + output gating
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    # -------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # -------------------------------------------------
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # [B,N,N,dim]  fp16

    # -------------------------------------------------
    # 9️⃣ Cast to float32 (required by the benchmark harness)
    # -------------------------------------------------
    return out.to(torch.float32)
