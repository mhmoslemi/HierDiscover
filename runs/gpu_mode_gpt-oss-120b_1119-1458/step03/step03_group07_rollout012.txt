import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mul_kernel(
    out_ptr,   # fp16 *
    gate_ptr,  # fp16 *
    N,         # i32 total elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    # sigmoid

    out = out * gate_sig
    tl.store(out_ptr + offs, out, mask=mask)

    (left/right proj + gate) into a single matmul, replaces the heavy
    einsum with a batched `torch.matmul`, and uses a tiny Triton kernel

    performed in FP32 for numerical stability.
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device

    )  # [B, N, N, dim] float32

    left_gate = comb[..., hidden_dim:2 * hidden_dim]

    right_gate = torch.sigmoid(right_gate)

    out_flat = out.contiguous().view(-1)

    # -------------------------------------------------

Algorithm (FP16 where possible)

    BLOCK = 1024  # maximum threads per block on H100
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    _gate_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # -------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimension
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (API contract)
    # -------------------------------------------------
    return out.to(torch.float32)
```
