def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # FP32

    # Cast to FP16
    x = x_norm.to(torch.float16)
    B, N, _, _ = x.shape

    # Flatten for fused linear
    x_flat = x.reshape(-1, dim)  # shape (B*N*N, dim)

    # Fuse left/right projections + gates
    # flatten weight stacks
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)

    fused_weight = torch.cat([left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight], dim=0)  # (4*hidden, dim)

    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # (B*N*N, 4*hidden)

    # Split and reshape
    hidden = hidden_dim
    left_proj_flat = fused_out[:, :hidden]
    left_gate_flat = fused_out[:, hidden:2*hidden]
    right_proj_flat = fused_out[:, 2*hidden:3*hidden]
    right_gate_flat = fused_out[:, 3*hidden:]

    # Reshape back to [B,N,N,hidden]
    left_proj = left_proj_flat.view(B, N, N, hidden)
    left_gate = left_gate_flat.view(B, N, N, hidden)
    right_proj = right_proj_flat.view(B, N, N, hidden)
    right_gate = right_gate_flat.view(B, N, N, hidden)

    # optional mask
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply gates
    left_gate_sigmoid = torch.sigmoid(left_gate)
    right_gate_sigmoid = torch.sigmoid(right_gate)
    left = left_proj * left_gate_sigmoid
    right = right_proj * right_gate_sigmoid

    # Compute out_gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate_flat = torch.nn.functional.linear(x_flat, out_gate_weight)  # (B*N*N, hidden)
    out_gate = torch.sigmoid(out_gate_flat.view(B, N, N, hidden))

    # O(N^3) contraction via optimized Triton kernel
    out = _batched_matmul_transpose_fp16(left, right)   # shape [B,N,N,hidden], FP16

    # Second LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Gate and project back
    out = out * out_gate
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim] FP16

    # Cast back to FP32
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast Triton implementation
================================================

This file implements the forward‑pass of the “outgoing” TriMul module
used by AlphaFold‑3 and other protein‑structure models.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (`x`) over the feature axis (`dim`) in FP32.
2️⃣  Cast the normalized tensor to FP16; fuse the four hidden‑dim linear
    projections (`left_proj`, `right_proj`) and their sigmoid gates
    (`left_gate`, `right_gate`) into a *single* GEMM (shape (4·H, dim)).
3️⃣  If a mask is present, apply it to the projected tensors (broadcast
    over the hidden dimension) before gating.
4️⃣  Apply the sigmoid gates (`left_gate`, `right_gate`).
5️⃣  Compute the heavy O(N³) contraction

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    as a *batched* matrix multiplication C = A @ Bᵀ where
    A and B have shape (M, N, K) with M = B·H, N = seq_len, K = seq_len.
    The custom Triton kernel tiles the multiplication with
    128 × 128 × 32 blocks, accumulates in FP32 (tensor‑core MMA) and
    uses double‑buffering (`num_stages=2`) with 8 warps per program.
6️⃣  Reshape the result back to `[B, N, N, H]` (FP16) and apply a second
    LayerNorm (hidden_dim).
7️⃣  Compute the output gate (`out_gate`) from the original normalized
    tensor, apply it element‑wise, and finish with a linear projection
    back to `dim`.
8️⃣  Cast the final tensor to FP32 (required output dtype).

Only the O(N³) contraction is executed in Triton; all other steps use
high‑performance PyTorch primitives.  The implementation works for the
full range of test configurations (B∈{1,2}, N∈{128,256,512,768,1024},
dim∈{128,384}, hidden_dim=128) and supports optional masking.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for many independent (N×N) matrices.
    Each program (identified by pid_z) processes one (M‑index) i.e. one
    (batch, hidden) pair.  The computation is tiled over the two N‑axes
    (i‑ and j‑axes) with blocks of size BLOCK_M × BLOCK_N.
    """
    pid_i = tl.program_id(0)   # tile along the first N dimension (i)
    pid_j = tl.program_id(1)   # tile along the second N dimension (j)
    pid_z = tl.program_id(2)   # index over M (= batch * hidden_dim)

    # -------------------------------------------------
    # Offsets of the current tile
    # -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over the K‑dimension
    # -------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- Load A tile (M‑idx, i‑idx, k‑idx) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ----- Load B tile (M‑idx, k‑idx, j‑idx) -- transposed view -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # ----- Multiply‑accumulate (tensor‑core MMA) -----
        # tl.dot promotes the fp16 multiplication to fp32 accumulation.
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Write back the result tile (cast to fp16)
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.
    right : torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.
    """
    B, N, _, H = left.shape
    M = B * H  # combined batch‑hidden dimension

    # ------------------------------------------------------------------
    # Pack (batch, hidden) into a single leading dimension:
    #   (B, N, N, H) -> (B*H, N, N)   (contiguous)
    # ------------------------------------------------------------------
    A = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    # Output buffer (FP16)
    C = torch.empty_like(A, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides in elements (Triton expects i32)
    # ------------------------------------------------------------------
    stride_am, stride_ai, stride_ak = A.stride()   # (M,N,K)
    stride_bm, stride_bi, stride_bk = B_.stride()  # (M,N,K)
    stride_cm, stride_ci, stride_cj = C.stride()   # (M,N,N)

    # ------------------------------------------------------------------
    # Block sizes – 128×128×32 is a sweet spot on H100 for fp16 MMA.
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid definition
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along the first N dimension (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along the second N dimension (j)
        M,                              # one program for each (batch, hidden) pair
    )

    # Launch kernel
    _trimul_matmul_kernel[grid](
        A, B_, C,
        M, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original layout [B, N, N, H]
    # ------------------------------------------------------------------
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the remaining heavy computations
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)   # [B, N, N, dim]  (fp16)
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four hidden‑dim linear projections + gates
    # ------------------------------------------------------------------
    # Flatten the [B, N, N, dim] tensor to a 2‑D matrix for GEMM
    x_flat = x.reshape(-1, dim)   # shape (B·N·N, dim)

    # Stack the four weight matrices (out_features = 4·hidden_dim)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    fused_hidden_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4·hidden_dim, dim)

    # Perform the fused GEMM
    fused_out = torch.nn.functional.linear(x_flat, fused_hidden_w)  # (B·N·N, 4·H)

    # Split the result back to the four tensors
    H = hidden_dim
    left_proj_flat = fused_out[:, :H]
    left_gate_flat = fused_out[:, H:2 * H]
    right_proj_flat = fused_out[:, 2 * H:3 * H]
    right_gate_flat = fused_out[:, 3 * H:]

    # Reshape to 4‑D tensors
    left_proj = left_proj_flat.view(B, N, N, H)
    left_gate = left_gate_flat.view(B, N, N, H)
    right_proj = right_proj_flat.view(B, N, N, H)
    right_gate = right_gate_flat.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – all in FP16
    # ------------------------------------------------------------------
    left_gate_sigmoid  = torch.sigmoid(left_gate)
    right_gate_sigmoid = torch.sigmoid(right_gate)

    left  = left_proj * left_gate_sigmoid   # [B, N, N, H]
    right = right_proj * right_gate_sigmoid # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Compute the output gate (separate linear+sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_flat = torch.nn.functional.linear(x_flat, out_gate_w)  # (B·N·N, H)
    out_gate = torch.sigmoid(out_gate_flat.view(B, N, N, H))        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via the optimized Triton kernel
    # ------------------------------------------------------------------
    # Result shape: [B, N, N, H] (fp16)
    contracted = _batched_matmul_transpose_fp16(left, right)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden_dim) – still in FP16
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + projection back to the original dimension
    # ------------------------------------------------------------------
    contracted = contracted * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(contracted, to_out_w)  # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required output type)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
