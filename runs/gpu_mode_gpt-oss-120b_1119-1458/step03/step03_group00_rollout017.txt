Implementation details with Triton:

```
@triton.jit
def fused_left_right_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    # total index counts all elements: B*N*N*hidden

    # load left_proj, left_gate, right_proj, right_gate
    left_proj_val = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_proj_val = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # compute mask index
    mask_idx = offs // hidden
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    left = left_proj_val * left_gate_val * mask_val
    right = right_proj_val * right_gate_val * mask_val

    tl.store(left_out_ptr + offs, left, mask=mask)
    tl.store(right_out_ptr + offs, right, mask=mask)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mask_gate_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load values
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Sigmoid approximations
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    left = lp * lg_sig * m
    right = rp * rg_sig * m

    tl.store(left_out_ptr + offs, left, mask=mask)
    tl.store(right_out_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    inp, mask, weights, config = data
    B,N,_,dim = inp.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = config.get("eps", 1e-5)

    # 1. Input layernorm
    x_norm = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)

    # 2. Concatenated linear layers
    concat_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16).contiguous()
    
    # Input flatten: (B*N*N, dim)
    x_flat = x_norm.view(-1, dim)
    # Linear
    raw = F.linear(x_flat, concat_weight) # no bias
    raw = raw.view(B, N, N, 5, hidden)

    left_proj_raw = raw[...,0,:]
    right_proj_raw = raw[...,1,:]
    left_gate_raw = raw[...,2,:]
    right_gate_raw = raw[...,3,:]
    out_gate_raw = raw[...,4,:]

    # 3. Mask & gate fusion for left/right
    if nomask:
        left = left_proj_raw * torch.sigmoid(left_gate_raw)
        right = right_proj_raw * torch.sigmoid(right_gate_raw)
    else:
        # convert mask to fp16
        mask_fp16 = mask.to(torch.float16)
        # allocate output tensors
        left = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        total = left_proj_raw.numel()
        BLOCK = 8192  # adjust block size for occupancy
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        fused_mask_gate_kernel[grid](
            left_proj_raw, left_gate_raw,
            right_proj_raw, right_gate_raw,
            mask_fp16,
            left, right,
            hidden,
            total,
            BLOCK_SIZE=BLOCK,
        )
        # out_gate compute separately
    # out_gate after sigmoid
    out_gate = torch.sigmoid(out_gate_raw)  # shape (B,N,N,hidden)

    # 4. Core contraction
    # reshape to (B*hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # multiply
    # shape (B*hidden, N, N)
    left_reshape = left_perm.view(B*hidden, N, N)
    right_reshape = right_perm.view(B*hidden, N, N)

    out_hidden = torch.bmm(left_reshape, right_reshape)  # (B*hidden, N, N)
    out = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # 5. Output layernorm + out_gate + final linear
    out_norm = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"].to(torch.float16),
                            bias=weights["to_out_norm.bias"].to(torch.float16), eps=eps)
    out_norm = out_norm * out_gate
    # final projection
    out_flat = out_norm.view(-1, hidden)  # (B*N*N, hidden)
    final = torch.nn.functional.linear(out_flat, weights["to_out.weight"].to(torch.float16))
    final = final.view(B, N, N, -1)

    return final.to(inp.dtype)
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Input LayerNorm (FP32) → cast to FP16.
2️⃣ Fuse the five independent Linear layers (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single matrix multiply:
   `[B,N,N,dim] @ (5·hidden, dim)ᵀ -> [B,N,N,5,hidden]`.
   This cuts the number of GEMMs from 5 to 1 and removes the extra
   copies of the input tensor.
3️⃣ Fuse mask handling, the two gate sigmoids and the element‑wise
   multiplication into **one** Triton kernel (`fused_lr_gate_kernel`).
   The kernel loads `left_proj`, `right_proj`, `left_gate`,
   `right_gate` and the broadcasted mask, computes the sigmoids on‑the‑fly
   and writes `left` and `right`.  Only one kernel launch is required.
4️⃣ Core contraction – for every hidden channel we need
   `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`.
   This is a batched matrix multiplication `left @ rightᵀ` with batch
   size `B·hidden`.  Calling `torch.bmm` on the reshaped tensors lets
   cuBLAS handle the large `N³` work efficiently.
5️⃣ Output LayerNorm over the hidden dimension, gating with `out_gate`,
   and a final linear projection back to `dim`.  All heavy arithmetic
   stays in FP16; the result is cast back to the original dtype.

Only step 3 is custom Triton code – it provides the required kernel
while the rest of the pipeline relies on highly‑optimized PyTorch /
cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask + sigmoid(gate) + element‑wise multiplication
# ----------------------------------------------------------------------
@triton.jit
def fused_lr_gate_kernel(
    left_proj_ptr,   # [B,N,N,hidden]  – left projection (pre‑gate)
    left_gate_ptr,   # [B,N,N,hidden]  – left gate (pre‑sigmoid)
    right_proj_ptr,  # [B,N,N,hidden]  – right projection (pre‑gate)
    right_gate_ptr,  # [B,N,N,hidden]  – right gate (pre‑sigmoid)
    mask_ptr,        # [B,N,N] (broadcasted fp16)
    left_out_ptr,    # output tensor (same shape as left_proj)
    right_out_ptr,   # output tensor (same shape as right_proj)
    hidden: tl.constexpr,
    total: tl.constexpr,          # total # elements = B·N·N·hidden
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load the four tensors (contiguous in the hidden dimension)
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (mask is broadcast over hidden)
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Sigmoid (≈ 1/(1+exp(-x))) executed in FP16
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    # Final fused product
    left = lp * lg_sig * m
    right = rp * rg_sig * m

    tl.store(left_out_ptr + offs, left, mask=mask)
    tl.store(right_out_ptr + offs, right, mask=mask)


def fused_lr_gate(left_proj, left_gate, right_proj, right_gate, mask):
    """
    Helper that launches `fused_lr_gate_kernel`.  All tensors must be
    contiguous FP16 and have the same hidden size.
    """
    B, N, _, H = left_proj.shape
    assert H == left_gate.shape[-1] == right_proj.shape[-1] == right_gate.shape[-1]
    total = left_proj.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    fused_lr_gate_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        left_out,
        right_out,
        H,                # hidden (constexpr)
        total,
        BLOCK_SIZE=BLOCK,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32 CUDA)
            - mask_tensor:  [B, N, N] (bool/float CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim"
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32 CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 → FP16 for the rest)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse the five linear layers into one large matmul
    # ------------------------------------------------------------------
    # Concatenate weight matrices: shape (5*hidden, dim)
    cat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()

    # Linear projection (flatten first three dims)
    x_flat = x.view(-1, dim)                     # (B·N·N, dim)
    raw_concat = F.linear(x_flat, cat_weight)    # (B·N·N, 5·hidden)
    raw_concat = raw_concat.view(B, N, N, 5, hidden)  # (B,N,N,5,hidden)

    # Slice the concatenated result
    left_proj_raw = raw_concat[..., 0, :]   # (B,N,N,hidden)
    right_proj_raw = raw_concat[..., 1, :]
    left_gate_raw = raw_concat[..., 2, :]
    right_gate_raw = raw_concat[..., 3, :]
    out_gate_raw = raw_concat[..., 4, :]   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3) Mask + gate fusion (mask optional)
    # ------------------------------------------------------------------
    if nomask:
        # No mask – just apply sigmoid(gate)
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask once to fp16; the Triton kernel will broadcast it.
        mask_fp16 = mask.to(torch.float16)
        left, right = fused_lr_gate(
            left_proj_raw, left_gate_raw,
            right_proj_raw, right_gate_raw,
            mask_fp16,
        )
    # out_gate is needed after the contraction
    out_gate = torch.sigmoid(out_gate_raw)   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) and reshape to (B·hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose k ↔ j

    left_mat = left_perm.view(B * hidden, N, N)
    right_mat = right_perm.view(B * hidden, N, N)

    # Batched matrix multiplication (uses cuBLAS under the hood)
    out_hidden = torch.bmm(left_mat, right_mat)  # (B·hidden, N, N)

    # Restore layout to (B, N, N, hidden)
    out = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                        # apply out_gate

    # Final linear projection back to `dim`
    out_flat = out_norm.view(-1, hidden)                  # (B·N·N, hidden)
    final = F.linear(
        out_flat,
        weights["to_out.weight"].to(torch.float16)         # (dim, hidden)
    )
    final = final.view(B, N, N, -1)

    # Return in the original dtype (normally float32)
    return final.to(inp.dtype)
