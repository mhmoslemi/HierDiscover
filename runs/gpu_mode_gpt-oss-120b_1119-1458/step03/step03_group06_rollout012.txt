def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # extract weights
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]
    device = input_tensor.device
    # Layer norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b) # float32
    x = x_norm.to(torch.float16)
    # Fuse linear layers: create weight for 5 projections
    fused_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)  # shape [5*hidden_dim, dim]
    fused_w = fused_w.to(torch.float16)
    fused_out = torch.nn.functional.linear(x, fused_w) # shape B,N,N,5*hidden_dim
    # slice
    H = hidden_dim
    # Use view for slicing
    left_proj = fused_out[..., 0:H]
    right_proj = fused_out[..., H:2*H]
    left_gate = fused_out[..., 2*H:3*H]
    right_gate = fused_out[..., 3*H:4*H]
    out_gate = fused_out[..., 4*H:5*H]
    # mask if needed
    if not nomask:
        # cast mask to fp16, unsqueeze
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16
    # apply sigmoid gating
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)
    # apply gates to left/right
    left = left_proj * left_gate
    right = right_proj * right_gate
    # contraction: batched matmul
    B, N, _, _ = left.shape
    # pack B*hidden into batch dimension
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # [M,N,N]
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # Compute out via torch.bmm
    out = torch.bmm(left_batched, right_batched.transpose(1, 2))
    # reshape back
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden_dim]
    # second LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w.to(torch.float16), bias=to_out_norm_b.to(torch.float16))
    # apply out gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, to_out_w.to(torch.float16))
    # cast back to float32
    return out.to(torch.float32)

"""
Optimized outgoing TriMul with fused linear projections and a
custom Triton batched matmul (A @ Bᵀ).  Heavy tensors are kept in fp16,
LayerNorms and sigmoid are performed in PyTorch.  The kernel tiles the
i/j dimensions with 128×128 tiles and the reduction k with 64 elements,
using 8 warps and two pipeline stages – well‑tuned for H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)  # i‑tile index
    pid_j = tl.program_id(1)  # j‑tile index
    pid_z = tl.program_id(2)  # batch‑*‑hidden index

    # offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # load A tile (M × K)
        a_ptrs = A_ptr + pid_z * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & (offs_k[None, :] < K)),
                    other=0.0)   # fp16

        # load B tile (N × K), transposed to (K × N)
        b_ptrs = B_ptr + pid_z * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=((offs_k[:, None] < K) & mask_j[None, :]),
                    other=0.0)   # fp16

        # matmul‑accumulate: fp16×fp16 → fp32
        acc += tl.dot(a, b)   # dot auto‑promotes to FP32 when acc is FP32

    # write out
    c_ptrs = C_ptr + pid_z * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = left @ rightᵀ.
    left/right: [B, N, N, H] (fp16, contiguous).
    Returns C of same shape and dtype.
    """
    B, N, _, H = left.shape
    # pack batch and hidden into a single leading dimension
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # strides (elements)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * H)

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : name -> torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional

    torch.Tensor

    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=norm_w,
                                       bias=norm_b)   # [B, N, N, dim], float32

    # ------------------------------------------------------------------

        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # shape [5*hidden_dim, dim]

    fused = torch.nn.functional.linear(x, fused_w)   # [B, N, N, 5*hidden_dim]

    H = hidden_dim
    left_proj = fused[..., 0:H]
    right_proj = fused[..., H:2*H]
    left_gate = fused[..., 2*H:3*H]
    right_gate = fused[..., 3*H:4*H]
    out_gate = fused[..., 4*H:5*H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (applied only to the projected tensors)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates (fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via custom Triton matmul (fp16)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=to_out_norm_w,
                                         bias=to_out_norm_b)    # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Output gating and final projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)    # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)        # fp16, shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
Optimized outgoing TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in float32.
2️⃣  Cast to fp16 for the heavy part.
3️⃣  Fuse the five linear transformations (left/right projections + three
    gates) into a single matrix multiplication – this reduces memory traffic
    by a factor ≈5 because the input tensor is read only once.
4️⃣  Apply the optional pair‑wise mask to the “left” and “right” projections.
5️⃣  Apply sigmoid gates (still in fp16) and multiply gate values into the
    projected tensors.
6️⃣  Compute the O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    with a custom Triton kernel that performs a batched matrix‑multiply
    C = A @ Bᵀ.  The kernel tiles the i/j dimensions with 128×128 tiles and
    the reduction k with 64 elements, uses 8 warps (256 threads) and a 2‑stage
    pipeline – well‑tuned for H100 tensor‑cores.  All tensors stay in fp16
    (accumulation in FP32 for stability).
7️⃣  Second LayerNorm on the hidden dimension (fp16).
8️⃣  Multiply by the output gate, apply the final linear projection back to
    the original channel dimension, and cast the result to float32.

Only the heavy O(N³) contraction is written in Triton; all surrounding
operations use fast PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: batched matmul C = A @ Bᵀ   (fp16 → fp16, acc fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # *fp16  [M, N, K]   (M = B*H, N = seq_len, K = seq_len)
    B_ptr,                     # *fp16  [M, N, K]
    C_ptr,                     # *fp16  [M, N, N]
    M, N, K,                   # i32 scalars (M unused, kept for API compatibility)
    stride_am, stride_ai, stride_ak,   # strides of A
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile index along the first seq‑len axis
    pid_j = tl.program_id(1)   # tile index along the second seq‑len axis
    pid_z = tl.program_id(2)   # batch * hidden index

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (to avoid overflow)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (BLOCK_M × BLOCK_K) – fp16
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak

            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,

            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk

        C_ptr

        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )

        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*hidden_dim, dim]

    fused = torch.nn.functional.linear(x, fused_w)   # [B, N, N, 5*hidden_dim]

    H = hidden_dim
    left_proj = fused[..., 0:H]
    right_proj = fused[..., H:2 * H]
    left_gate = fused[..., 2 * H:3 * H]
    right_gate = fused[..., 3 * H:4 * H]
    out_gate = fused[..., 4 * H:5 * H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (applied to the projected tensors only)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via the Triton kernel
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Output gating and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)         # fp16, shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
