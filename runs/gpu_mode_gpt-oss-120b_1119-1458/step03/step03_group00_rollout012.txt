def fused_proj_gate_mask_kernel(

"""
Highly‑optimized Triton‑accelerated implementation of the *outgoing* TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm the input (FP32 → FP16).
2️⃣  Concatenate each projection weight with its corresponding gate weight and
    perform a *single* dense layer (`F.linear`) to obtain both the raw
    projection and the raw (pre‑sigmoid) gate in one matrix multiplication.
3️⃣  Fuse the sigmoid gate and the optional mask with a custom Triton kernel
    ( `fused_proj_gate_mask_kernel` ).  The kernel evaluates

        out = proj * sigmoid(gate) * mask

    without expanding the mask to the hidden dimension, minimizing memory traffic.
4️⃣  Core contraction (the O(N³) update) is a batched matrix multiplication:

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    After the optional mask/gate step we reshape the tensors to
    (B·hidden, N, N) and call a single `torch.matmul`, which uses cuBLAS’s
    strided‑batched GEMM and stays in FP16.
5️⃣  Output layer‑norm, gating and linear projection back to the original
    channel dimension.  All heavy arithmetic stays in FP16; the final result
    is cast back to the original dtype (typically float32).

Only the elementwise triple product (step 3) is custom‑written in Triton,
fulfilling the “must use a kernel” requirement while keeping the rest of the
pipeline in highly‑optimized PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: proj * sigmoid(gate) * mask  (gate broadcasted over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr,     #  pointer to projection tensor   (B,N,N,hidden)
    gate_ptr,     #  pointer to pre‑sigmoid gate   (B,N,N,hidden)
    mask_ptr,     #  pointer to mask tensor         (B,N,N)
    out_ptr,      #  destination tensor
    hidden: tl.constexpr,
    size:   tl.constexpr,          # total number of elements in proj/gate
    BLOCK_SIZE: tl.constexpr,      # preferred block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load projection and raw gate (both contiguous in hidden dimension)
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate_raw = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Map each hidden element to its mask index (mask is broadcast over hidden)
    mask_idx = offs // hidden                      # integer division
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Sigmoid (logistic) of the gate
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)


def fused_proj_gate_mask(proj: torch.Tensor,
                         gate_raw: torch.Tensor,
                         mask: torch.Tensor,
                         hidden: int) -> torch.Tensor:
    """
    Compute ``proj * sigmoid(gate_raw) * mask`` where ``mask`` is broadcast over
    the hidden dimension.  All tensors must be contiguous and of dtype
    ``torch.float16``.
    """
    assert proj.shape == gate_raw.shape
    B, N, _, H = proj.shape
    assert H == hidden, "hidden dimension mismatch"
    assert mask.shape == (B, N, N)

    proj = proj.contiguous()
    gate_raw = gate_raw.contiguous()
    mask = mask.contiguous()

    out = torch.empty_like(proj)

    total = proj.numel()
    BLOCK = 65536                                   # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_proj_gate_mask_kernel[grid](
        proj,
        gate_raw,
        mask,
        out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  shape [B, N, N]   (bool/float, CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim"
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data               # inp: [B, N, N, dim]
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Concatenated linear projections for left/right + their gates
    # ------------------------------------------------------------------
    # left: [proj | gate]  → shape (2*hidden, dim)
    left_w_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                                   # (2*hidden, dim)

    right_w_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                                   # (2*hidden, dim)

    # Perform the two fused linear layers (still large matmuls)
    left_concat = F.linear(x_norm, left_w_concat)        # (B,N,N,2*hidden)
    right_concat = F.linear(x_norm, right_w_concat)      # (B,N,N,2*hidden)

    # Split projection / gate channels
    left_proj_raw   = left_concat[..., :hidden]          # (B,N,N,hidden)
    left_gate_raw   = left_concat[..., hidden:]          # (B,N,N,hidden)

    right_proj_raw  = right_concat[..., :hidden]         # (B,N,N,hidden)
    right_gate_raw  = right_concat[..., hidden:]         # (B,N,N,hidden)

    # Out‑gate (no mask)
    out_gate_raw = F.linear(x_norm, weights["out_gate.weight"].to(torch.float16))
    out_gate = torch.sigmoid(out_gate_raw)               # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask + sigmoid gate (fused Triton kernel)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise product
        left  = left_proj_raw * torch.sigmoid(left_gate_raw)
        right = right_proj_raw * torch.sigmoid(right_gate_raw)
    else:
        # Convert mask to FP16 once; Triton kernel will broadcast it on‑the‑fly
        mask_fp16 = mask.to(torch.float16)
        left  = fused_proj_gate_mask(left_proj_raw, left_gate_raw, mask_fp16, hidden)
        right = fused_proj_gate_mask(right_proj_raw, right_gate_raw, mask_fp16, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction: batched matrix multiplication per hidden channel
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Permute to (B, hidden, i, k) and (B, hidden, k, j) – both are just views
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, k, j)

    # Fuse the batch and hidden dimensions and call a single matmul
    B_H = B * hidden
    left_flat  = left_perm.view(B_H, N, N)               # (B*hidden, i, k)
    right_flat = right_perm.view(B_H, N, N)              # (B*hidden, k, j)

    out_flat = torch.matmul(left_flat, right_flat)       # (B*hidden, i, j)
    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, gating and final projection back to dim
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # Apply output gate (element‑wise)
    out_norm = out_norm * out_gate

    # Final linear projection (bias‑free)
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),
    )

    # Cast back to the input dtype (usually float32) before returning
    return out_final.to(inp.dtype)
