                         stride_am, stride_ak, stride_ak2,
                         stride_bm, stride_bk, stride_bj,
                         stride_cm, stride_cn, stride_ci,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # 1 block per i and j tile per pid_m, pid_n, pid_z combined batch*hidden as pid_z.
    pid_z = tl.program_id(0)  # combined batch*hidden index
    pid_m = tl.program_id(1)  # i tiles
    pid_n = tl.program_id(2)  # j tiles

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    a_ptrs = A_ptr + pid_z * stride_am + offs_i[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_z * stride_bm + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_j[None, :] * stride_bj

"""
TriMul (outgoing) forward pass – a highly‑optimized fusion of LayerNorm,
Linear + sigmoid gating, a batched matrix‑multiply (C = A @ Bᵀ) computed
with a custom Triton kernel, and the final LayerNorm + linear projection.
All heavy work is performed in FP16 with FP32 accumulation; the result is
returned in FP32 to match the reference implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C[b] = A[b] @ B[b]ᵀ   for each b = batch * hidden
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,          # A strides (M, N, K)
    stride_bm, stride_bk, stride_bj,          # B strides (M, K, N)  (B is already transposed)
    stride_cm, stride_ci, stride_cj,          # C strides (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiply C = A @ Bᵀ for FP16 inputs.
    A : [M, N, K]   (M = B*hidden, N = seq_len_padded, K = seq_len_padded)
    B : [M, K, N]   (= right.permute(...,2,1) → already transposed)
    C : [M, N, N]
    """
    pid_z = tl.program_id(0)   # index over (batch * hidden)
    pid_m = tl.program_id(1)   # tile row (i)
    pid_n = tl.program_id(2)   # tile col (j)

    # Offsets inside the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for cur_k in range(0, K, BLOCK_K):
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i, k]  → shape (BLOCK_M, BLOCK_K)
        a_ptr = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptr)               # fp16

        # Load Bᵀ[k, j]  (= B transposed) → shape (BLOCK_K, BLOCK_N)
        b_ptr = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptr)               # fp16

        # Dot product (fp16 * fp16 → fp32) and accumulate
        acc += tl.dot(a, b)

    # Write out results (cast back to fp16)
    c_ptr = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptr, acc.to(tl.float16))


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Compute out = left @ rightᵀ for each (batch, hidden) pair using Triton.
    Input tensors must be FP16 with shape [B, N, N, H] where N is already
    a multiple of `tile`. Returns a tensor of the same shape (still padded)
    in FP16.
    """
    B, N, _, H = left.shape

    # fuse batch & hidden → [B*H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # transpose the right matrix so the kernel can load Bᵀ directly
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N] (K,N)

    A = left_perm.view(B * H, N, N)
    B_ = right_T.view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A)

    # strides in number of elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        B * H,                         # pid_z
        (N + BLOCK_M - 1) // BLOCK_M,  # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,  # pid_n
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the last dimension `dim`
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim] , float32
    x = x_norm.to(torch.float16)  # switch to fp16 for the heavy work

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2️⃣ Cast all model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Flatten input for the linear layers
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional masking – fuse mask into the projection step
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        # broadcast mask over the feature dimension before the projection
        x_proj = x_flat * mask_fp16.view(-1, 1)   # (B*N*N, dim)
    else:
        x_proj = x_flat

    # ------------------------------------------------------------------
    # 5️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_proj, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_proj, right_proj_w).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – still in FP16, based on the *unmasked* input
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # Fuse gates into the projected tensors
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Pad sequence dimension to a multiple of TILE to eliminate bounds checks
    # ------------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len:
        N_pad = N + pad_len
        left = torch.nn.functional.pad(left, (0, 0, 0, pad_len, 0, pad_len))   # pad dims 1 & 2
        right = torch.nn.functional.pad(right, (0, 0, 0, pad_len, 0, pad_len))
    else:
        N_pad = N

    # ------------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via Triton (FP16 inputs, FP16 output)
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right, tile=TILE)   # [B, N_pad, N_pad, hidden_dim]

    # Remove padding if we added any
    if pad_len:
        out = out[:, :N, :N, :]   # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 🔟 Output gating + final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]   FP16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to FP32 (the model expects FP32 outputs)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
