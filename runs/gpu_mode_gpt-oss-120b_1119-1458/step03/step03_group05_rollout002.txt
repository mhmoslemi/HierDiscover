"""
TriMul outgoing kernel (AlphaFold‑3)

The heavy O(N³) contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is realised as a batched GEMM C = A @ Bᵀ where each hidden slice
(and each batch) forms an independent matrix multiplication of shape
[N, N] @ [N, N]ᵀ.  The kernel tiles the operation with a 2‑level
blocking strategy (M‑tiles, N‑tiles) and uses Tensor‑Core‑accelerated
`tl.dot`.  Tile sizes are chosen at launch time (64×64×64 for
moderate N, 128×128×64 for very large N) and the kernel runs with a
double‑buffered pipeline (num_stages=3) and 8 warps per program.
All other linear / LayerNorm / gating steps are performed in PyTorch
with fp16 arithmetic; the first LayerNorm stays in fp32 for stability.
The final output is cast back to fp32 as required by the harness.

The implementation:
* a Triton kernel `_trimul_matmul_kernel` that computes a block of C
  from tiles of A (left) and B (right transposed);
* a helper `_batched_matmul_transpose_fp16` that reshapes the
  `[B, N, N, H]` tensors, launches the kernel, and restores the original
  layout;
* the entry‑point `custom_kernel` that performs the full forward pass
  (LayerNorm, projections, gating, contraction, second LayerNorm,
  final gating + projection).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel:  C = A @ Bᵀ   (batched GEMM)
#   A : [M, N, K]   (M = B * hidden, N = seq_len, K = seq_len)
#   B : [M, K, N]   (right already transposed)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]   (right already transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32   dimensions
    stride_am, stride_ai, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bk, stride_bj,   # strides for B (M,K,N)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------------------
    # 1. program IDs (tiling over i, j and over the batch*hidden dim)
    # -------------------------------------------------------------
    pid_m = tl.program_id(0)   # tile row of output (i)
    pid_n = tl.program_id(1)   # tile col of output (j)
    pid_z = tl.program_id(2)   # batch*hidden index

    # -------------------------------------------------------------
    # 2. offsets for the current tile
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # 3. accumulator (FP32)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # 4. loop over reduction dimension K in tiles
    # -------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A tile (M, BLOCK_M, BLOCK_K) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- load B tile (M, BLOCK_K, BLOCK_N) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # K‑dimension (now first)
            + offs_n[None, :] * stride_bj   # N‑dimension (post‑transpose)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )   # fp16

        # ---- tensor‑core multiply‑accumulate (FP16×FP16→FP32) ----
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # 5. write back result (cast to fp16)
    # -------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper: reshape tensors, launch the kernel, restore layout
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    """
    Compute the O(N³) contraction via a batched GEMM.
    left  → A of shape (M, N, K)   where M = B*H, K = N
    right → Bᵀ of shape (M, K, N)  (we transpose the last two dims).
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape

    # Pack batch and hidden together: shape (M, N, N)
    # A[i, m, k] = left[b, m, k, h]  with i = b*H + h
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Bᵀ : we need right transposed on the last two axes (k, j)
    B_t = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    out = torch.empty_like(A, dtype=torch.float16)

    # Strides in element (not byte) units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_t.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Choose tile sizes based on N (tuned for H100)
    if N <= 512:
        BLOCK_M = BLOCK_N = BLOCK_K = 64
    else:
        # For very large N (768–1024) a larger spatial tile helps reduce
        # launch overhead while keeping K‑tile at 64 (nice for Tensor Core).
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    # Empirically 8 warps work well on H100 for the above tiles.
    _trimul_matmul_kernel[grid](
        A, B_t, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,          # double‑buffered pipeline
    )

    # Restore original layout [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor[B, N, N, dim]   (float32)
            mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
            weights_dict : dict of float32 parameter tensors
            config_dict  : must contain ``dim`` and ``hidden_dim``;
                           optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. First LayerNorm (float32 → better stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 (once)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [B, N, N, 1] for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6. Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. O(N³) contraction via the Triton batched GEMM
    # ------------------------------------------------------------------
    # Input tensors are contiguous in the layout expected by the kernel
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 8. Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9. Final gating + projection back to original dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)     # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 10. Cast back to fp32 (as expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
