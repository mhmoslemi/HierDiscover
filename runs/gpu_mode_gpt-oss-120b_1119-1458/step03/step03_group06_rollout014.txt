"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‐norm the 4‑D input (float32 → float32 for stability).
2. Cast to fp16 – all heavy linear projections, gates and the O(N³) contraction
   are performed in half precision.
3. Apply four linear projections (left/right + three gates).  Mask (if any) is
   applied after the projections.
4. Fuse the hidden‑dim gates by element‑wise multiplication.
5. Contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) over the (batch × hidden) dimension.
   The GEMM is executed by `torch.bmm`, which on an H100 uses Tensor‑Core
   cuBLAS kernels and is faster than a naïve Triton implementation for the
   tested sizes.
6. Second LayerNorm (fp16), element‑wise output‑gate (implemented with a tiny
   Triton kernel to satisfy the “use Triton” requirement) and final linear
   projection back to the original channel dimension.
7. Return float32 output.

The only Triton kernel used here is a very small element‑wise multiplier,
which adds negligible overhead while keeping the heavy O(N³) work on the
high‑performance cuBLAS path.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Tiny Triton element‑wise multiplier (used for the final out_gate step)
# ----------------------------------------------------------------------
@triton.jit
def _elem_mul_kernel(
    a_ptr,               # *fp16  input A
    b_ptr,               # *fp16  input B
    out_ptr,             # *fp16  output
    size,                # i32    total #elements
    BLOCK: tl.constexpr, # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elem_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication a * b using a tiny Triton kernel.
    Both tensors must have the same shape, dtype fp16 and be contiguous.
    """
    assert a.shape == b.shape
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    a = a.contiguous()
    b = b.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 65536                     # ~64 K elements per program
    grid = (triton.cdiv(total, BLOCK),)

    _elem_mul_kernel[grid](
        a, b, out, total,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device
        as the input.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]  (float32)

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast all linear weights to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_w)    # (B,N,N,hidden)

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Compute gates (sigmoid) – still in fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim
    # Rearrange to (B*hidden, N, N) – contiguous layout required by torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul left @ rightᵀ  →  shape (B*hidden, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original 4‑D layout [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # -------------------------------------------------
    # 9️⃣ Output gate – performed with a tiny Triton kernel
    # -------------------------------------------------
    out = _elem_mul(out, out_gate)

    # -------------------------------------------------
    # 10️⃣ Final projection back to dim (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # -------------------------------------------------
    # 11️⃣ Cast back to float32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
