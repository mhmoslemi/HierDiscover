@triton.jit
def tri_mul_contract_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
   pid = tl.program_id(0) # pid for i block
   pid_j = tl.program_id(1) # pid for j block
   pid_bh = tl.program_id(2) # combined batch and h

   b = pid_bh // H
   h = pid_bh % H
   # i start
   i0 = pid * BLOCK_M
   j0 = pid_j * BLOCK_N

   # offsets for i,k and j,k
   offs_i = i0 + tl.arange(0, BLOCK_M)
   offs_j = j0 + tl.arange(0, BLOCK_N)
   offs_k_base = tl.arange(0, BLOCK_K)

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, N, BLOCK_K):
       # compute offsets for this tile of k
       offs_k = k + offs_k_base
       mask_k = offs_k < N

       # load A tile: shape (BLOCK_M, BLOCK_K)
       a_ptrs = left_ptr + b*stride_left_b + offs_i[:,None]*stride_left_i + offs_k[None,:]*stride_left_k + h*stride_left_h
       a = tl.load(a_ptrs, mask=mask_k[None,:] & (offs_i[:,None] < N), other=0.0)

       # load B tile: shape (BLOCK_K, BLOCK_N)
       b_ptrs = right_ptr + b*stride_right_b + offs_j[None,:]*stride_right_j + offs_k[:,None]*stride_right_k + h*stride_right_h
       b = tl.load(b_ptrs, mask=mask_k[:,None] & (offs_j[None,:] < N), other=0.0)

       acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

   # store out
   out_ptrs = out_ptr + b*stride_out_b + offs_i[:,None]*stride_out_i + offs_j[None,:]*stride_out_j + h*stride_out_h
   mask_out_i = offs_i < N
   mask_out_j = offs_j < N
   mask_out = mask_out_i[:,None] & mask_out_j[None,:]
   tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)
```

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # block row i
    pid_j = tl.program_id(1)  # block column j
    pid_bh = tl.program_id(2) # batch * hidden

    # compute batch index and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # create range indices
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for i and j out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, N, BLOCK_K):
        k_end = tl.minimum(k + BLOCK_K, N)
        cur_K = k_end - k
        # recompute tile offsets for k within range
        k_offs = k + tl.arange(0, cur_K)

        # compute a_ptrs and b_ptrs with broadcasting
        a_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + offs_i[:, None]*stride_l_i + k_offs[None, :]*stride_l_k
        b_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + k_offs[:, None]*stride_r_k + offs_j[None, :]*stride_r_j

def _tri_mul_contract_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute start offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # compute batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # ranges
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    k_offsets = tl.arange(0, BLOCK_K)

    # create masks for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # prologue pointers: base offsets for i and j connections for current k offset (starting at 0)

"""
TriMul (outgoing) – highly‑optimized Triton implementation.

Algorithm
---------
1️⃣  First LayerNorm on the input 4‑D tensor (FP16).
2️⃣  Fuse the three linear layers (left‑proj+gate, right‑proj+gate,
    out‑gate) into a single GEMM:      X_flat @ W_fused.T.
    This reads the input only once (instead of three times).
3️⃣  Split the fused result → left_proj/right_proj and their gates.
    Apply the optional mask *before* the gates, then multiply by
    sigmoid(gate) → left / right (still FP16).
4️⃣  Heavy O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented with a custom Triton kernel that works directly on the
    original layout [B,N,N,H] (no costly contiguous transposes) and
    uses Tensor‑Core `tl.dot` for maximal throughput.
5️⃣  Second LayerNorm (FP16) on the intermediate result.
6️⃣  Compute out‑gate = sigmoid(out_gate_pre) and fuse the
    element‑wise multiplication with a tiny Triton kernel.
7️⃣  Final projection to the original channel dimension with a single
    linear layer.
8️⃣  Cast back to FP32 (API‑required output dtype).

Only the two Triton kernels (`_tri_mul_contract_kernel` and
`_elemwise_mul_kernel`) are custom; all other ops use highly‑optimized
CuBLAS / Torch kernels.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel 1: O(N³) contraction, works on the [B,N,N,H] layout.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_contract_kernel(
    left_ptr,          # *fp16 left  [B,N,N,H]
    right_ptr,         # *fp16 right [B,N,N,H]
    out_ptr,           # *fp16 out   [B,N,N,H]
    B, N, H,           # i32 dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs ---------------------------------------------------------
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_bh = tl.program_id(2)         # combined batch‑hidden index

    # ----- compute batch and hidden indices --------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ----- start offsets for this tile ------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k ----------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = tl.minimum(k + BLOCK_K, N)
        k_range = cur_k - k                         # actual tile size

        # offsets of k within this tile
        offs_k = k + tl.arange(0, BLOCK_K)

        # mask for k (handles tails)
        mask_k = offs_k < N

        # ----- load A tile: left[b,i,k,h]  (shape M×K) --------------------
        a_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)

        # ----- load B tile: right[b,j,k,h] transposed to (K×N) ------------
        b_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_k[:, None] * stride_r_k    # k dimension as rows
            + offs_j[None, :] * stride_r_j    # j dimension as columns
        )
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)

        # ----- rank‑K update ------------------------------------------------
        acc += tl.dot(a, b)          # (M,N) <- (M,K)⋅(K,N)

    # ----- write the result ------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Triton kernel 2: in‑place element‑wise multiplication (used for out‑gate)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,                     # *fp16  destination (in‑place)
    src_ptr,                     # *fp16  source (gate)
    N,                           # i32 total #elements
    BLOCK_SIZE: tl.constexpr,   # i32 threads per block
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs, mask=mask, other=0.0)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    tl.store(dst_ptr + offs, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Helper: fused linear (three heads) + split
# ----------------------------------------------------------------------
def _fused_linear_split(x: torch.Tensor,
                        left_comb_w: torch.Tensor,
                        right_comb_w: torch.Tensor,
                        out_gate_w: torch.Tensor,
                        hidden_dim: int) -> Tuple[torch.Tensor,
                                                  torch.Tensor,
                                                  torch.Tensor]:
    """One large GEMM that produces left, right and out‑gate tensors."""
    B, N, _, dim = x.shape
    # flatten to (B·N·N, dim)
    x_flat = x.view(-1, dim)                         # FP16
    # concatenate the three weight matrices (out_features × in_features)
    fused_w = torch.cat([left_comb_w, right_comb_w, out_gate_w], dim=0)  # (5·H, dim)
    fused_out = torch.nn.functional.linear(x_flat, fused_w)               # (B·N·N, 5·H)
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)                    # (B,N,N,5·H)
    # split
    left_comb = fused_out[..., : 2 * hidden_dim]          # (B,N,N,2·H)
    right_comb = fused_out[..., 2 * hidden_dim: 4 * hidden_dim]  # (B,N,N,2·H)
    out_gate_pre = fused_out[..., 4 * hidden_dim:]       # (B,N,N,H)
    return left_comb, right_comb, out_gate_pre


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (no gradient).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32, already on device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # -------------------------------------------------------------
    # Cast all weights once to FP16 (TensorCore friendly)
    # -------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # -------------------------------------------------------------
    # 1️⃣ LayerNorm on the raw input (FP16)
    # -------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim] FP16

    # -------------------------------------------------------------
    # 2️⃣ Fuse the three linear layers into one GEMM
    # -------------------------------------------------------------
    # combine left proj+gate and right proj+gate
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)     # (2H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2H, dim)

    left_comb, right_comb, out_gate_pre = _fused_linear_split(
        x_norm, left_comb_w, right_comb_w, out_gate_w, hidden_dim
    )   # each: FP16, shapes (B,N,N,2H) or (B,N,N,H)

    # -------------------------------------------------------------
    # 3️⃣ Split projections / gates, apply mask (if any) and sigmoid
    # -------------------------------------------------------------
    left_proj = left_comb[..., :hidden_dim]          # (B,N,N,H)
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # (B,N,N,H)

    right_proj = right_comb[..., :hidden_dim]        # (B,N,N,H)
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])  # (B,N,N,H)

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate # (B,N,N,H)

    # -------------------------------------------------------------
    # 4️⃣ O(N³) contraction via custom Triton kernel
    # -------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # allocate output tensor (FP16)

    # strides for the kernel (as plain Python ints)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # block sizes – empirically good for fp16 TensorCores on H100
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,    # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,    # j‑tiles
        B * H,                           # batch×hidden
    )

    _tri_mul_contract_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # still FP16

    # -------------------------------------------------------------
    # 6️⃣ Out‑gate sigmoid + element‑wise multiplication (Triton)
    # -------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_pre)   # (B,N,N,H) FP16

    # flatten for the 1‑D Triton kernel
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024
    grid_mul = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    _elemwise_mul_kernel[grid_mul](
        out_flat,
        gate_flat,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # -------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # -------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # -------------------------------------------------------------
    # 8️⃣ Cast back to float32 (API contract)
    # -------------------------------------------------------------
    return out.to(torch.float32)
