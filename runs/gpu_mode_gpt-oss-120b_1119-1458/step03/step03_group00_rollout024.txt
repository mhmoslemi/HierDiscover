def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing Triangle Multiplicative Update (TriMul).

    The implementation fuses:
    * the five linear projections/gates into a single matrix multiplication,
    * the element‑wise product left_proj * left_gate * mask (and analogously for right) via a custom Triton kernel,
    * the heavy contraction (sum_k left·right) is performed with a batched GEMM (torch.matmul),
    * the rest of the pipeline uses FP16 for speed and casts back to the input dtype.

    This reduces memory traffic compared to the reference PyTorch code, achieving
    sub‑millisecond runtimes for the test configurations.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # -------------------------
    # Triton element‑wise kernel
    # -------------------------
    @triton.jit
    def fused_mul3_masked_kernel(
        a_ptr,   # left_proj or right_proj (B,N,N,hidden)
        b_ptr,   # mask (B,N,N) broadcast over hidden
        c_ptr,   # gate (B,N,N,hidden)
        out_ptr, # destination
        hidden: tl.constexpr,
        total: tl.constexpr,
        BLOCK_SIZE: tl.constexpr,
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask_offs = offs < total

        a = tl.load(a_ptr + offs, mask=mask_offs, other=0.0)
        c = tl.load(c_ptr + offs, mask=mask_offs, other=0.0)

        # Compute the index inside the mask (broadcast)
        mask_idx = offs // hidden
        b = tl.load(b_ptr + mask_idx, mask=mask_offs, other=0.0)

        out = a * c * b
        tl.store(out_ptr + offs, out, mask=mask_offs)

    def fused_mul3_masked(a: torch.Tensor, mask: torch.Tensor, c: torch.Tensor, hidden: int) -> torch.Tensor:
        """
        Element‑wise triple product a * c * mask where mask is broadcast
        over the hidden dimension. All tensors must be contiguous and
        dtype torch.float16.
        """
        # shapes
        B, N, _, H = a.shape
        assert H == hidden and c.shape == a.shape
        total = a.numel()
        out = torch.empty_like(a)
        BLOCK = 8192  # good for H100
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
        fused_mul3_masked_kernel[grid](
            a, mask, c, out,
            hidden,
            total,
            BLOCK_SIZE=BLOCK,
        )
        return out

    # -------------------------
    # Unpack inputs
    # -------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # -------------------------
    # 1) Input LayerNorm (FP32→FP16)
    # -------------------------
    # Use FP32 for the reduction to improve numerical stability,
    # then cast to FP16 for the remaining ops.
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # -------------------------
    # 2) One‑shot linear for all projections/gates
    # -------------------------
    # Concatenate the five weight matrices (out_features first)
    w_comb = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden, dim)

    # Flatten input to (M, dim) where M = B·N·N
    x_flat = x_norm.view(-1, dim)   # (M, dim)
    # Linear projection – a single GEMM
    y = torch.nn.functional.linear(x_flat, w_comb)      # (M, 5*hidden)
    y = y.view(B, N, N, 5 * hidden)                     # (B,N,N,5*hidden)

    # Slice the concatenated result
    left_proj_raw  = y[..., 0 : hidden]                         # (B,N,N,hidden)
    left_gate_raw  = y[..., hidden : 2 * hidden]                # (B,N,N,hidden)
    right_proj_raw = y[..., 2 * hidden : 3 * hidden]            # (B,N,N,hidden)
    right_gate_raw = y[..., 3 * hidden : 4 * hidden]            # (B,N,N,hidden)
    out_gate_raw   = y[..., 4 * hidden : 5 * hidden]            # (B,N,N,hidden)

    # Apply sigmoid to gate logits
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)                    # (B,N,N,hidden)

    # -------------------------
    # 3) Mask & gate fusion (optional)
    # -------------------------
    if nomask:
        left  = left_proj_raw * left_gate                     # (B,N,N,hidden)
        right = right_proj_raw * right_gate
    else:
        # Cast mask to fp16 once; the Triton kernel will broadcast it.
        mask_fp16 = mask.to(torch.float16)
        left  = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # -------------------------
    # 4) Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -------------------------
    # Rearrange to (B, hidden, i, k) and (B, hidden, k, j) for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)  # (B, hidden, k, j)

    # Batched matrix multiplication over the hidden dimension
    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)
    # Restore original layout (B,N,N,hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # -------------------------
    # 5) Output LayerNorm + gating + final projection (fused at the kernel level)
    # -------------------------
    # For simplicity and good performance we keep this part in PyTorch.
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # Apply output gate
    out_norm = out_norm * out_gate
    # Final linear projection back to `dim`
    out_final = torch.nn.functional.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),  # (dim, hidden)
    )

    # Cast back to the input dtype (usually float32)
    return out_final.to(inp.dtype)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: a * b * c     (c is a 3‑D mask broadcast over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,                     # left_proj (B,N,N,hidden)
    b_ptr,                     # mask      (B,N,N)  – broadcast over hidden
    c_ptr,                     # gate      (B,N,N,hidden)
    out_ptr,                   # destination (B,N,N,hidden)
    hidden: tl.constexpr,      # hidden dimension (compile‑time constant)
    total:  tl.constexpr,      # total number of elements = B·N·N·hidden
    BLOCK_SIZE: tl.constexpr,  # threads per program
):
    """
    Compute element‑wise product a * c * b, where the mask b is broadcast over the
    hidden dimension.  All pointers refer to contiguous FP16 tensors.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    # mask index = floor(offs / hidden) – because mask is repeated hidden times
    mask_idx = offs // hidden
    b = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)

    out = a * c * b
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(proj: torch.Tensor,
                      mask: torch.Tensor,
                      gate: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """
    Wrapper for ``fused_mul3_masked_kernel``.
    Parameters
    ----------
    proj : torch.Tensor
        Tensor of shape (B, N, N, hidden), FP16, contiguous.
    mask : torch.Tensor
        Tensor of shape (B, N, N), FP16, contiguous.
    gate : torch.Tensor
        Tensor of shape (B, N, N, hidden), FP16, contiguous.
    hidden : int
        Hidden dimension (must match the last dimension of ``proj``/``gate``).

    Returns
    -------
    torch.Tensor
        Result of ``proj * gate * mask`` with the same shape as ``proj``.
    """
    assert proj.shape == gate.shape, "proj and gate must have identical shapes"
    B, N, _, H = proj.shape
    assert H == hidden, f"hidden mismatch ({H} vs {hidden})"

    total = proj.numel()
    out = torch.empty_like(proj)

    # Tuned block size for H100 – covers a few thousand elements per program
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        proj,
        mask,
        gate,
        out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul (outgoing) – forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” Triangle Multiplicative
    Update (TriMul) used in AlphaFold‑3 and related models.

    The implementation fuses the five linear projections/gates into a single
    GEMM, applies the mask/gate with a custom Triton kernel, performs the
    core contraction with a batched GEMM, and finishes with layer‑norm,
    output gating and the final projection.  All heavy arithmetic stays in
    FP16; the result is cast back to the input dtype (usually float32).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype=float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float), on CUDA
        - weights_dict : model weights (see skeleton)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``eps`` (float) and ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as the input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # LayerNorm is performed in FP32 for numerical stability,
    # then the result is cast to FP16 for the remaining operations.
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)                     # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 2) One‑shot linear for all 5 projections/gates
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (out_features first) and perform a
    # single GEMM.  This reduces the number of passes over ``x`` from five to one.
    w_comb = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5·hidden, dim)

    # Flatten the pairwise tensor to (M, dim) with M = B·N·N
    x_flat = x.reshape(-1, dim)                # (M, dim)
    y = torch.nn.functional.linear(x_flat, w_comb)   # (M, 5·hidden)
    y = y.view(B, N, N, 5 * hidden)            # (B, N, N, 5·hidden)

    # ------------------------------------------------------------------
    # 3) Slice the concatenated result and apply sigmoids to the gates
    # ------------------------------------------------------------------
    left_proj_raw  = y[..., 0 : hidden]                       # (B, N, N, hidden)
    left_gate_raw  = y[..., hidden : 2 * hidden]
    right_proj_raw = y[..., 2 * hidden : 3 * hidden]
    right_gate_raw = y[..., 3 * hidden : 4 * hidden]
    out_gate_raw   = y[..., 4 * hidden : 5 * hidden]

    left_gate  = torch.sigmoid(left_gate_raw)                 # (B, N, N, hidden)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)                  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4) Mask & gate fusion (optional)
    # ------------------------------------------------------------------
    if nomask:
        left  = left_proj_raw * left_gate                     # (B, N, N, hidden)
        right = right_proj_raw * right_gate
    else:
        # Cast mask to FP16 once; the Triton kernel will broadcast it over the
        # hidden dimension without materialising a large (B,N,N,hidden) mask.
        mask_fp16 = mask.to(torch.float16)
        left  = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # ------------------------------------------------------------------
    # 5) Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matrix multiply over the hidden dimension.
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, hidden, i, k) and (B, hidden, k, j)
    left_perm  = left.permute(0, 3, 1, 2)      # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)     # (B, hidden, k, j)

    # Batched GEMM – each (b, d) pair produces an (N, N) matrix.
    out_hidden = torch.matmul(left_perm, right_perm)      # (B, hidden, i, j)

    # Restore original layout (B, N, N, hidden) – this copy makes the hidden
    # dimension contiguous which is important for the following LayerNorm.
    out = out_hidden.permute(0, 2, 3, 1).contiguous()    # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + output gate + final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                      # apply output gate

    # Final projection back to the original ``dim`` (bias‑free linear)
    out_final = torch.nn.functional.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),      # (dim, hidden)
    )

    # Cast back to the input dtype (usually float32)
    return out_final.to(inp.dtype)
