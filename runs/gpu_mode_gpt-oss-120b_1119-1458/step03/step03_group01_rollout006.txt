"""
Optimized “outgoing” TriMul forward pass (AlphaFold‑3 style).

The kernel fuses the three dominant phases of the TriMul operator:

1️⃣ **Linear projection + gating** – performed with torch.nn.functional
   (FP16) and a tiny Triton kernel that multiplies the projected tensors
   by their sigmoid‑gates in‑place.

2️⃣ **Core N³ tensor product** – implemented as a custom batched GEMM in
   Triton (`tri_mul_kernel`).  The 4‑D tensors `[B, N, N, H]` are reshaped
   to `[B·H, N, N]` and the kernel computes
   `out = left @ rightᵀ` for every head in parallel, using FP16
   dot‑products with FP32 accumulation (Tensor‑Core friendly).  This
   replaces the PyTorch `torch.bmm` used in the reference implementation.

3️⃣ **Output post‑processing** – LayerNorm, output‑gate multiplication and
   final projection are kept in FP16 with a second Triton kernel for the
   element‑wise gate.  The result is returned in FP32 to match the
   reference signature.

All heavy arithmetic is performed in half‑precision, minimizing memory
traffic while preserving numerical stability (LayerNorms use FP32
accumulation internally).  The implementation works for any batch size
`B∈{1,2}`, sequence length `N∈{128,256,512,1024}` and hidden dimensions
`C=dim∈{128,384}` and `H=hidden_dim∈{128,384,768}`.  The `nomask`
flag controls whether the optional `[B,N,N]` mask is applied.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,        # fp16*, flattened left tensor
    left_gate_ptr,   # fp16*, flattened left gate tensor
    right_ptr,       # fp16*, flattened right tensor
    right_gate_ptr,  # fp16*, flattened right gate tensor
    numel: tl.int64,          # total number of elements per tensor
    BLOCK: tl.constexpr,      # block size (tiles per program)
):
    """
    In‑place multiplication:
        left  = left  * left_gate
        right = right * right_gate

    The tensors are contiguous 1‑D views of the original 4‑D data.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # left side
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    # right side
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,         # fp16*, output tensor that will be overwritten
    b_ptr,         # fp16*, multiplicative operand (same shape as a_ptr)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        a = a * b
    Both tensors are flattened contiguous 1‑D arrays.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


@triton.jit
def tri_mul_kernel(
    left_ptr,      # fp16* (total_heads, N, N)
    right_ptr,     # fp16* (total_heads, N, N) – already transposed (K,J)
    out_ptr,       # fp16* (total_heads, N, N)
    N: tl.int64,   # sequence length (both M and N in GEMM)
    # strides for left (head, i, k)
    stride_l_head: tl.int64,
    stride_l_i:    tl.int64,
    stride_l_k:    tl.int64,
    # strides for right (head, k, j) – transposed layout
    stride_r_head: tl.int64,
    stride_r_k:    tl.int64,
    stride_r_j:    tl.int64,
    # strides for output (head, i, j)
    stride_o_head: tl.int64,
    stride_o_i:    tl.int64,
    stride_o_j:    tl.int64,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul core:
        out[head, i, j] = Σ_k left[head, i, k] * right[head, k, j]
    Implements (total_heads, N, N) @ (total_heads, N, N)ᵀ using
    FP16 inputs with FP32 accumulation.
    """
    pid_head = tl.program_id(0)
    pid_m    = tl.program_id(1)   # tile over output rows (i)
    pid_n    = tl.program_id(2)   # tile over output columns (j)

    # Offsets for the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this head (already incorporate head stride)
    base_left  = left_ptr  + pid_head * stride_l_head
    base_right = right_ptr + pid_head * stride_r_head
    base_out   = out_ptr   + pid_head * stride_o_head

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_left + offs_m[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask = mask_m[:, None] & mask_k[None, :],
                    other = 0.0)

        # Load a tile of right (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = base_right + cur_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask = mask_k[:, None] & mask_n[None, :],
                    other = 0.0)

        # FP16 dot‑product accumulated in FP32
        acc += tl.dot(a, b)

    # Write the result back (cast to FP16)
    c_ptrs = base_out + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask = mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 variant).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask         : torch.Tensor      [B, N, N] (bool/float, optional)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   containing:
                         * 'dim' (int) – input / output channel size
                         * 'hidden_dim' (int) – inner projection size
                         * optional 'nomask' (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 mean/var, FP16 data)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)          # cast once for the remaining ops

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – all in FP16
    # ------------------------------------------------------------------
    left  = F.linear(x, w_left_proj)    # [B, N, N, hidden_dim]
    right = F.linear(x, w_right_proj)

    # ------------------------------------------------------------------
    # 4️⃣ Gate pre‑activations (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x, w_out_gate))

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Fuse left/right with their sigmoid‑gates (Triton)
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK_ELEM = 65536                     # 64 k elements per program (tuned)
    grid = ((numel + BLOCK_ELEM - 1) // BLOCK_ELEM,)

    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK_ELEM,
    )
    # Gates are no longer needed – free memory early
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ multiplication: batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                # H == hidden_dim
    total_heads = B * H

    # Reshape to (total_heads, N, N) – layout: (head, i, k)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Transpose the second operand once (K×J → K×J) and make it contiguous
    right_t = right_mat.transpose(1, 2).contiguous()

    # Output buffer
    out_mat = torch.empty_like(left_mat)    # (total_heads, N, N), fp16

    # Strides for the three tensors (they are contiguous => row‑major)
    stride_l_head, stride_l_i, stride_l_k = left_mat.stride()
    stride_r_head, stride_r_k, stride_r_j = right_t.stride()   # (head, k, j)
    stride_o_head, stride_o_i, stride_o_j = out_mat.stride()

    # Choose block sizes adaptively (must be compile‑time constants per launch)
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 128

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_t,
        out_mat,
        N,
        stride_l_head,
        stride_l_i,
        stride_l_k,
        stride_r_head,
        stride_r_k,
        stride_r_j,
        stride_o_head,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Free the transposed right matrix early
    del right_t

    # Reshape back to the original 4‑D layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing LayerNorm (FP16 weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )   # still FP16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (Triton – in‑place)
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK_ELEM - 1) // BLOCK_ELEM,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK_ELEM,
    )
    del out_gate   # free

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim], FP16

    # Return float32 as required by the reference module
    return final.to(torch.float32)
