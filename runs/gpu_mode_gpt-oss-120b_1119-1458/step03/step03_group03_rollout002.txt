@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile along first N dimension (i)
    pid_n = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A[i, k] tile ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---- load Bᵀ[k, j] tile (i.e. B[k, j]) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # dot (fp16 * fp16 -> fp32) and accumulate
        acc += tl.dot(a, b)

    # write result (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )
```

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input in FP32 (the reference implementation does the same).
2. Convert to FP16 and fuse the two linear projections (proj + gate) for the
   left and right branches by concatenating the weight matrices.  This halves
   the number of GEMM’s over the input and reduces memory traffic.
3. Apply the (optional) mask only to the projected tensors (as in the paper).
   Compute the sigmoid gates in FP16 and fuse them with the projected values.
4. Pad the sequence dimension to a multiple of 128 (tile size) to eliminate
   bound‑checks inside the Triton kernel.
5. Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM C = A @ Bᵀ where
       A = left   (shape B·H × N × N)
       B = right  (shape B·H × N × N)
   The Triton kernel uses 128×128×32 tiles, FP16 → FP32 accumulation,
   8 (or 16) warps and a 3‑stage pipeline.  The tile size is chosen to
   match the tensor‑core shape (16×16 per warp) on an H100.
6. Remove the padding, apply the second LayerNorm (FP16), the output gate
   and the final linear projection back to the original `dim`.
7. Return the result as FP32.

Only the O(N³) part is written in Triton; all surrounding ops are
implemented with high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: Batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each batch*hidden slice.
    A, B, C are in FP16 (A,B) / FP16 (C) with FP32 accumulation.
    """
    pid_m = tl.program_id(0)   # tile along i dimension
    pid_n = tl.program_id(1)   # tile along j dimension
    pid_z = tl.program_id(2)   # batch * hidden index

    # Offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # FP32 accumulator for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load a tile of A (shape BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
            cache_policy='stream',
        )  # fp16

        # Load a tile of Bᵀ (i.e. B[k, j]) -> shape BLOCK_K × BLOCK_N
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
            cache_policy='stream',
        )  # fp16

        # Tensor‑core dot (fp16×fp16 → fp32) + accumulate
        acc += tl.dot(a, b)

    # Write result back (cast to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 128,
) -> torch.Tensor:
    """
    Batched GEMM (C = left @ rightᵀ) for shape [B, N, N, H] (FP16).
    Returns a tensor of the same shape (still padded) in FP16.
    """
    B, N, _, H = left.shape

    # Collapse batch and hidden dimensions: [B·H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    C = torch.empty_like(A)   # output in FP16

    # Strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile // 4   # 32 for tile==128, matches tensor‑core tile shape

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps = 256 threads per block (good occupancy)
        num_stages=3,         # 3‑stage pipeline for H100
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)   # fp16 for the heavy work

    # ------------------------------------------------------------------
    # 2️⃣ Convert all linear weights to FP16 once
    # ------------------------------------------------------------------
    # Projection + gate weights are fused (proj || gate) to halve the number of GEMM's
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    left_and_gate_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    right_and_gate_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections + gates (all in FP16, fused where possible)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    # fused left projection + left gate
    left_gate_cat = torch.nn.functional.linear(x_flat, left_and_gate_w)   # (B*N*N, 2*H)
    left_proj = left_gate_cat[:, :hidden_dim]               # (B*N*N, H)
    left_gate = left_gate_cat[:, hidden_dim:]               # (B*N*N, H)

    # fused right projection + right gate
    right_gate_cat = torch.nn.functional.linear(x_flat, right_and_gate_w)  # (B*N*N, 2*H)
    right_proj = right_gate_cat[:, :hidden_dim]            # (B*N*N, H)
    right_gate = right_gate_cat[:, hidden_dim:]            # (B*N*N, H)

    # out gate (separate linear)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    )   # (B*N*N, H)

    # reshape back to [B, N, N, H]
    left_proj = left_proj.view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_gate.view(B, N, N, hidden_dim))
    right_proj = right_proj.view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate.view(B, N, N, hidden_dim))
    out_gate = out_gate.view(B, N, N, hidden_dim)   # already sigmoid‑ed

    # ------------------------------------------------------------------
    # 4️⃣ Apply optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; convert to fp16 and broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gates into the projected tensors
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # (B, N, N, H) FP16
    right = right_proj * right_gate       # (B, N, N, H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Pad to a multiple of the tile size (128) to drop bound checks
    # ------------------------------------------------------------------
    TILE = 128
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (FP16 inputs → FP16 output)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, H] FP16

    # Remove padding (if any)
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Output gate + final projection (both in FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to FP32 (the external API expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
