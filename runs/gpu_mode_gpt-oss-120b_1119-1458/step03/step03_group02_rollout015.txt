"""
TriMul (outgoing) – fast Triton implementation
================================================

This module implements the forward‑pass of the “outgoing” TriMul block
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣  Apply a first LayerNorm over the feature dimension (float32 for
    numerical stability).

2️⃣  Compute the two linear projections (left/right) and their three
    hidden‑dim sigmoid gates in float32.  If a mask is provided it is
    applied in float32 as well.

3️⃣  Fuse the hidden‑dim gates with the projected tensors and cast the
    result to float16.

4️⃣  Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a *batched* matrix multiplication
    `C = A @ Bᵀ` where `A` and `B` have shape (M,N,K) with
    `M = batch * hidden_dim`, `N = seq_len`, `K = seq_len`.
    The custom Triton kernel tiles the multiplication with 64×64×64
    blocks, accumulates in FP32 via `tl.dot` (tensor‑core MMA), and
    uses double‑buffering (`num_stages=2`) and 8 warps per program.

5️⃣  Reshape the result back to `[B,N,N,hidden]` (float16) and apply a
    second LayerNorm (still in FP16).

6️⃣  Apply the output gate (cast to FP16) and a final linear projection
    back to the original dimension.

7️⃣  Cast the final tensor to float32 (the required output type).

Only the O(N³) contraction is executed in Triton; all other steps use
high‑performance PyTorch primitives.  The implementation works for the
full range of test configurations (batch ≤ 2, seq_len up to 1024,
`dim` ∈ {128,384}, hidden_dim = 128, optional masking).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batched matrix product for many independent (N×N) matrices.

    Each program (pid_z) processes one (M‑index) i.e. one hidden‑batch pair.
    The computation is tiled over the two N‑axes (i‑ and j‑axes) with
    blocks of size BLOCK_M × BLOCK_N.
    """
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # index over M (= batch * hidden_dim)

    # ------------------------------------------------------------------
    # Row/column offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over the K dimension
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- Load A tile (M‑idx, i‑idx, k‑idx) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----- Load B tile (M‑idx, k‑idx, j‑idx) -----
        # This loads the transposed view of B (K × N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ----- Multiply‑accumulate (tensor‑core MMA) -----
        # tl.dot promotes fp16 multiplication to fp32 accumulation.
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.
    right : torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.
    """
    B, N, _, H = left.shape

    # -------------------------------------------------
    # Pack (batch, hidden) into a single leading dimension:
    # (B, N, N, H) -> (B*H, N, N)
    # -------------------------------------------------
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP16)
    C = torch.empty_like(A, dtype=torch.float16)

    # -------------------------------------------------
    # Strides in number of elements (Triton expects i32)
    # -------------------------------------------------
    stride_am, stride_ai, stride_ak = A.stride()       # (M,N,K)
    stride_bm, stride_bi, stride_bk = B_.stride()      # (M,N,K)
    stride_cm, stride_ci, stride_cj = C.stride()       # (M,N,N)

    # -------------------------------------------------
    # Tile sizes – 64×64×64 is a sweet spot on H100 for fp16 MMA.
    # -------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # one program per (batch, hidden)
    )

    # Launch the kernel
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

        norm_bias = weights["norm.bias"].to(device)
        x_norm = torch.nn.functional.layer_norm(
            input_tensor,
            (dim,),
            weight=norm_weight,
            bias=norm_bias,
        )  # [B, N, N, dim]  float32

        # ------------------------------------------------------------------
        # 2️⃣ Linear projections (float32)
        # ------------------------------------------------------------------
        left_proj_w = weights["left_proj.weight"].to(device)   # [hidden, dim]
        right_proj_w = weights["right_proj.weight"].to(device)

        left_fp32 = torch.nn.functional.linear(x_norm, left_proj_w)   # [B,N,N,hidden]
        right_fp32 = torch.nn.functional.linear(x_norm, right_proj_w)

        # ------------------------------------------------------------------
        # 3️⃣ Optional mask (float32, broadcast over hidden dim)
        # ------------------------------------------------------------------
        if not nomask:
            mask_fp32 = mask_tensor.to(torch.float32).unsqueeze(-1)   # [B,N,N,1]
            left_fp32 = left_fp32 * mask_fp32
            right_fp32 = right_fp32 * mask_fp32

        # ------------------------------------------------------------------
        # 4️⃣ Gating (float32) – sigmoid on the original features
        # ------------------------------------------------------------------
        left_gate_w = weights["left_gate.weight"].to(device)
        right_gate_w = weights["right_gate.weight"].to(device)
        out_gate_w = weights["out_gate.weight"].to(device)

        left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_w))
        right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_w))
        out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_w))

        # Apply hidden‑dim gates
        left_fp32 = left_fp32 * left_gate
        right_fp32 = right_fp32 * right_gate

        # ------------------------------------------------------------------
        # 5️⃣ Cast to FP16 for the heavy O(N³) contraction
        # ------------------------------------------------------------------
        left_fp16 = left_fp32.half()
        right_fp16 = right_fp32.half()

        # ------------------------------------------------------------------
        # 6️⃣ Heavy contraction via Triton (produces [B,N,N,hidden] fp16)
        # ------------------------------------------------------------------
        out_fp16 = _batched_matmul_transpose_fp16(left_fp16, right_fp16)

        # ------------------------------------------------------------------
        # 7️⃣ Second LayerNorm (FP16)
        # ------------------------------------------------------------------
        to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
        to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
        out_fp16 = torch.nn.functional.layer_norm(
            out_fp16,
            (hidden_dim,),
            weight=to_out_norm_w,
            bias=to_out_norm_b,
        )

        # ------------------------------------------------------------------
        # 8️⃣ Output gate + final projection back to original dim
        # ------------------------------------------------------------------
        out_fp16 = out_fp16 * out_gate.half()
        to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden]
        out_fp16 = torch.nn.functional.linear(out_fp16, to_out_w)   # [B,N,N,dim] fp16

        # ------------------------------------------------------------------
        # 9️⃣ Cast back to float32 (required output dtype)
        # ------------------------------------------------------------------
        return out_fp16.float()
```
