"""
Optimised outgoing TriMul (AlphaFold‑3) forward pass.

The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched GEMM C = A @ Bᵀ with Triton.
All linear projections and gate‑computations are fused into a single
matmul (x @ W_concat) to avoid repeated reads of the input tensor.
Mask handling and all element‑wise gates are kept in FP16 to keep the
memory traffic low.  The second LayerNorm and the final projection are
performed with PyTorch (they are negligible compared with the cubic term).

The Triton kernel uses a 128×128×64 tiling, FP32 accumulation, three
pipeline stages (num_stages=4) and 8 warps per block – an empirically
fast configuration on H100 for the tested sequence lengths.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (batched matmul)
#   A : [M, N, K]    (M = B*hidden, N = seq_len, K = seq_len)
#   B : [M, K, N]    (right already transposed)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]   (right already transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32   scalar sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M,i,k)
    stride_bm, stride_bk, stride_bj,   # strides for B (M,k,j)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # 1. Tile indices
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)   # i‑tile index
    pid_n = tl.program_id(1)   # j‑tile index
    pid_z = tl.program_id(2)   # batch*hidden index

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # 2. Accumulator (FP32 for numeric stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3. Loop over reduction dimension K
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- A tile (M x K) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---- B tile (K x N) ----  (right already transposed)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # K‑dimension stride
            + offs_j[None, :] * stride_bj   # N‑dimension stride
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # ---- Multiply‑accumulate (tensor‑core friendly) ----
        acc += tl.dot(a, b)   # FP16×FP16 → FP32

    # --------------------------------------------------------------
    # 4. Write result (casting back to fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper for the batched GEMM (A @ Bᵀ) – takes tensors [B,N,N,H] (fp16)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    B, N, _, H = left.shape

    # Pack batch & hidden together → shape [M, N, K] with M = B*H
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Right needs to be transposed on the last two dims (k, j)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Output buffer (same layout as the packed inputs)
    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – empirically good for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # Restore the original layout [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,         # input tensor [B, N, N, dim] (float32)
        torch.Tensor,         # mask tensor  [B, N, N] (float32/bool) – may be None
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool) – may be None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections (proj, gates, out_gate)
    # ------------------------------------------------------------------
    # Convert each weight to fp16 once
    left_proj_w   = weights["left_proj.weight"].to(dtype_fp16)      # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(dtype_fp16)     # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(dtype_fp16)      # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(dtype_fp16)     # (H, dim)
    out_gate_w    = weights["out_gate.weight"].to(dtype_fp16)       # (H, dim)

    # Concatenate along the output‑feature dimension: (5*H, dim)
    concat_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # shape [5*H, dim]

    # One big matmul: [B,N,N,5*H] = [B,N,N,dim] @ concat_weight.T
    proj_all = torch.nn.functional.linear(x, concat_weight)  # fp16

    # Split the result back into the five components
    H = hidden_dim
    left = proj_all[..., 0:H]                # [B,N,N,H]
    left_gate = proj_all[..., H:2 * H]
    right = proj_all[..., 2 * H:3 * H]
    right_gate = proj_all[..., 3 * H:4 * H]
    out_gate = proj_all[..., 4 * H:5 * H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse gates into the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via the Triton batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Final gating & linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(dtype_fp16)   # (dim, hidden) for Linear(dim←hidden)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(dtype_fp32)
