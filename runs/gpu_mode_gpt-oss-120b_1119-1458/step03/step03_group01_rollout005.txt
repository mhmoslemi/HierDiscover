Optimized outgoing TriMul forward (AlphaFold3) using Triton.

Algorithm
---------
1️⃣ Layer‑normalize the 4‑D input (FP32) and cast to FP16.
2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate,

8️⃣ Linear projection back to the original channel dimension (FP16 → FP32).

All heavy arithmetic runs in FP16, only the two LayerNorms use FP32 weights for

# ----------------------------------------------------------------------
@triton.jit
def left_right_gate_mask_kernel(
    left_ptr,            # fp16*   left tensor (flattened)
    left_gate_ptr,       # fp16*   left‑gate tensor (flattened)
    right_ptr,           # fp16*   right tensor (flattened)
    right_gate_ptr,      # fp16*   right‑gate tensor (flattened)
    mask_ptr,            # fp16*   mask flattened (shape B·N·N)
    hidden_dim: tl.constexpr,
    numel: tl.int64,     # total number of elements in left/right tensors
    BLOCK: tl.constexpr,
):
    """
    In‑place multiplication:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask

    The mask is broadcast over the hidden dimension, therefore
    mask_idx = idx // hidden_dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # broadcast mask index
    mask_idx = offs // hidden_dim

    l  = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,      mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_ptr + offs,  l * lg * m, mask=mask)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,               # fp16*   target tensor (flattened)
    b_ptr,               # fp16*   multiplier tensor (flattened, same shape)
    numel: tl.int64,     # total number of elements
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication   a = a * b
    Both tensors are flattened beforehand.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      shape [B, N, N, D]   (float32)
        - mask         : torch.Tensor      shape [B, N, N]      (bool/float)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict with keys:
                         * 'dim'          – input channel dimension D
                         * 'hidden_dim'   – hidden dimension H
                         * 'nomask' (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – kept in full precision for stability
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for all dense work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Prepare combined weight for the five linear projections
    #    (left, right, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)

    # (5·H, D) weight matrix – one GEMM instead of five
    w_combined = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )  # shape (5*H, D), FP16

    # --------------------------------------------------------------
    # 4️⃣ Fused linear projection for all five streams
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    M = B * N * N                               # number of “pairs”
    x_h_flat = x_h.view(M, dim)                # (M, D)
    proj_all = F.linear(x_h_flat, w_combined)  # (M, 5*H), FP16
    proj_all = proj_all.view(B, N, N, 5 * hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Split and activate the gating tensors
    # --------------------------------------------------------------
    offset = hidden_dim
    left       = proj_all[..., 0:offset]                     # (B,N,N,H)
    right      = proj_all[..., offset:2 * offset]            # (B,N,N,H)
    left_gate  = torch.sigmoid(proj_all[..., 2 * offset:3 * offset])
    right_gate = torch.sigmoid(proj_all[..., 3 * offset:4 * offset])
    out_gate   = torch.sigmoid(proj_all[..., 4 * offset:5 * offset])

    # --------------------------------------------------------------
    # 6️⃣ Apply mask (if any) together with the left/right gates.
    #    For the no‑mask case we simply multiply with the gates using
    #    torch's element‑wise kernel (fast enough because it is cheap).
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1] → flatten to (B·N·N,)
        mask_h = mask.to(torch.float16).unsqueeze(-1)        # (B,N,N,1)
        mask_h_flat = mask_h.reshape(-1)                     # (B·N·N,)

        # Ensure tensors are contiguous for the Triton kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        numel = left.numel()
        BLOCK = 65536
        grid = ((numel + BLOCK - 1) // BLOCK,)

        left_right_gate_mask_kernel[grid](
            left,
            left_gate,
            right,
            right_gate,
            mask_h_flat,
            hidden_dim=hidden_dim,
            numel=numel,
            BLOCK=BLOCK,
        )
        # gate tensors are no longer needed
        del left_gate, right_gate, mask_h_flat, mask_h
    else:
        # No mask → only gate multiplication (torch kernels)
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # --------------------------------------------------------------
    # 7️⃣ Core N³ multiplication (batched GEMM)
    #     out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Transform to (B, H, N, N) for torch.matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, N, N)

    # Back to original layout
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 8️⃣ Post‑processing LayerNorm (half‑precision weights)
    # --------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # still FP16

    # --------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (in‑place) via Triton
    # --------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()       # shape (B,N,N,H), FP16

    numel_out = out.numel()
    BLOCK = 65536
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out,
        out_gate,
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # release memory

    # --------------------------------------------------------------
    # 10️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    final = F.linear(out, w_to_out)                        # (B, N, N, dim), FP16

    # --------------------------------------------------------------
    # 11️⃣ Cast back to float32 (the format expected by the caller)
    # --------------------------------------------------------------
    return final.to(torch.float32)
