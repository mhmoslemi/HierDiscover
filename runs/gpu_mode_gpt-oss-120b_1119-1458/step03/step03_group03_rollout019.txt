def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # LayerNorm: Normalize input
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # Cast to fp16
    x = x_norm.half()   # .half()
    
    B, N, _, _ = x.shape

    # Flatten positions to (B*N*N, dim)
    x_flat = x.reshape(-1, dim)

    # Concatenate all linear weight matrices
    w_all = torch.cat([weights['left_proj.weight'],
                       weights['right_proj.weight'],
                       weights['left_gate.weight'],
                       weights['right_gate.weight'],
                       weights['out_gate.weight']], dim=0)  # shape (5*hidden_dim, dim)
    w_all = w_all.half()
    # Apply all linear in one matmul
    # shape (B*N*N, 5*hidden_dim)
    lin_out = torch.nn.functional.linear(x_flat, w_all)   # yields fp16 (since w_all fp16 leads to fp16 compute)
    # reshape to (B,N,N,5, hidden_dim)
    lin_out = lin_out.view(B, N, N, 5, hidden_dim)

    left_proj = lin_out[...,0,:]   # shape B,N,N,hidden
    right_proj = lin_out[...,1,:]
    left_gate = lin_out[...,2,:]
    right_gate = lin_out[...,3,:]
    out_gate = lin_out[...,4,:]

    # apply mask (if any) to left_proj and right_proj
    if not nomask:
        mask_f = mask_tensor.to(torch.float16).unsqueeze(-1)    # B,N,N,1
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # fuse gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Pad to multiple of tile
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:

    right_perm = right.permute(0,3,2,1).contiguous()

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣  Fuse the five linear projections (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single GEMM:   X_flat @ W_all.T
   where `W_all` is the concatenation of the five weight matrices.
   This eliminates repeated reads of the large input tensor.
3️⃣  Apply the optional mask to the two projected tensors; compute the
    sigmoid gates; fuse the gates into the projected tensors.
4️⃣  Pad the sequence dimension to a multiple of 64 so the Triton kernel
    can drop all boundary checks.
5️⃣  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a batched matrix multiplication C = A @ Bᵀ, where A = left and
    B = right with the two sequence axes swapped.  The kernel operates
    on FP16 data, accumulates in FP32 and uses 64×64×64 tiles, 8 warps
    and a 3‑stage pipeline.
6️⃣  Slice away padding, apply the second LayerNorm, the output‑gate,
    and the final linear projection back to `dim`.
7️⃣  Return the result in FP32.

All operations except the O(N³) contraction are performed with
high‑performance PyTorch primitives; the contraction is the only
custom Triton kernel (required by the problem statement).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (no bounds checks – inputs are padded)
#   A : [M, N, K]   (M = B * hidden, N = seq_len (padded), K = seq_len (padded))
#   B : [M, K, J]   (here K = J = N; B is the right tensor with its first two
#                    sequence axes swapped)
#   C : [M, N, J]   = [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel_padded(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                         # not used but kept for compatibility
    stride_am, stride_ai, stride_ak,  # strides for A (M,i,k)
    stride_bm, stride_bk, stride_bj,  # strides for B (M,k,j)
    stride_cm, stride_ci, stride_cj,  # strides for C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication without boundary checks.
    Each program instance computes a (BLOCK_M × BLOCK_N) tile of C for a
    given batch+hidden index `pid_z`.
    """
    pid_m = tl.program_id(0)   # tile along i dimension
    pid_n = tl.program_id(1)   # tile along j dimension
    pid_z = tl.program_id(2)   # combined batch * hidden index

    # Tile offsets (guaranteed to be inside the padded range)
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i, k]  (shape BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs)                     # fp16

        # Load Bᵀ[k, j] i.e. B[k, j] (shape BLOCK_K × BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs)                     # fp16

        # FP16×FP16 → FP32 accumulate via tensor cores
        acc += tl.dot(a, b)

    # Write the result (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16))


def _batched_matmul_left_right(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Compute out = Σ_k left[...,i,k,:] * right[...,j,k,:] for every hidden channel.
    This is a batched matmul where `right` is first transposed over its two
    sequence axes so that a regular GEMM (A @ Bᵀ) can be used.
    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H]  (fp16).  `right` is *not* transposed a priori.
    tile : int
        Tile size used inside the Triton kernel (must divide the padded length).
    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (fp16), same padding as `left`/`right`.
    """
    B, N, _, H = left.shape

    # A = left  → [B, H, N, N] (i,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # B = rightᵀ → swap the first two sequence axes, then bring H forward
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # [B, H, k, i]

    A = left_perm.view(B * H, N, N)          # (M,N,K)
    B_ = right_perm.view(B * H, N, N)        # (M,K,J)  (note K==J==N)

    C = torch.empty_like(A)                  # output buffer (M,N,J)

    # Strides in element units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK = tile
    grid = (
        (N + BLOCK - 1) // BLOCK,
        (N + BLOCK - 1) // BLOCK,
        B * H,
    )
    _trimul_matmul_kernel_padded[grid](
        A,
        B_,
        C,
        B * H,                     # M (unused)
        N,                         # N (padded)
        N,                         # K (padded)
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=BLOCK,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to original layout [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16 for the heavy work
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32
    x = x_norm.half()  # cast to FP16

    B, N, _, _ = x.shape
    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices: order = [left_proj, right_proj,
    # left_gate, right_gate, out_gate] → shape (5*hidden, dim)
    w_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # (5*hidden_dim, dim)

    # Flatten spatial dimensions and apply the fused linear
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)
    lin_out = F.linear(x_flat, w_all)               # (B*N*N, 5*hidden_dim)  FP16

    # Reshape and split
    lin_out = lin_out.view(B, N, N, 5, hidden_dim)   # [B, N, N, 5, H]
    left_proj   = lin_out[..., 0, :]   # [B, N, N, H]
    right_proj  = lin_out[..., 1, :]
    left_gate   = lin_out[..., 2, :]
    right_gate  = lin_out[..., 3, :]
    out_gate    = lin_out[..., 4, :]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates (FP16) and fuse with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Pad sequence dimension to a multiple of the tile size (64)
    # ------------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via the custom Triton kernel
    # ------------------------------------------------------------------
    out = _batched_matmul_left_right(left_pad, right_pad, tile=TILE)   # [B, N_pad, N_pad, H]

    # Remove padding if we added any
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gating + final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].half())   # → [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (the caller expects float32)
    # ------------------------------------------------------------------
    return out.float()
