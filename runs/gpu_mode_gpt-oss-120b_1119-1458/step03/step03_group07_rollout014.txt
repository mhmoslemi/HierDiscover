```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _elemwise_mul_kernel(
    dst_ptr, src_ptr, N,
    dst_stride: tl.constexpr, src_stride: tl.constexpr,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)
    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)

def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left, right shape [B, N, N, H] fp16
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    High‑performance Triton‑accelerated forward pass of the outgoing TriMul module.
    The implementation fuses the four linear projections (proj+gate, left+right)
    into a single GEMM, reduces kernel launches, and uses a tiny Triton kernel
    for the final element‑wise multiplication with the output gate.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Compute all linear projections (proj & gate) in a single GEMM
    # ------------------------------------------------------------------
    # Gather weight tensors (all stored in float32, cast once)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # concatenate to a (4*hidden_dim, dim) weight matrix
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )   # shape (4*H, dim)

    # Linear: x [B,N,N,dim] -> combined [B,N,N,4*H]
    combined = torch.nn.functional.linear(x, combined_weight)

    # split back
    h = hidden_dim
    left_proj = combined[..., :h]          # [B,N,N,H]
    left_gate = combined[..., h:2*h]
    right_proj = combined[..., 2*h:3*h]
    right_gate = combined[..., 3*h:4*h]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied after projection, before gate)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise product
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate    # [B,N,N,H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Compute out_gate (sigmoid) from the normalized input
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by out_gate (in‑place) via a tiny Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,  # dst stride
        1,  # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    #  🔚 Cast back to float32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

"""
TriMul (outgoing) – fast forward implementation

Algorithm (GPU, FP16)
---------------------
1️⃣ Layer‑Norm the 4‑D input (float32 for numerical stability) → FP16.
2️⃣ Fuse the four linear layers (proj + gate for *left* and *right*) into a
   single GEMM by concatenating their weight matrices:
        [proj_L , gate_L , proj_R , gate_R]   ∈ ℝ^{4·H × C}
   The result is sliced back into the four tensors, the mask (if any) is
   applied, the sigmoid gates are computed and the gated projections are
   performed *in‑place*.
3️⃣ Compute the output‑gate (sigmoid of a linear on the normalized input).
4️⃣ O(N³) contraction is a batched matrix‑multiply:
        out_{b,i,j,:} = Σ_k left_{b,i,k,:} · right_{b,j,k,:}
   This is realized as a strided‑batched GEMM (torch.bmm) after a cheap
   permute+reshape of the tensors.
5️⃣ Layer‑Norm over the hidden dimension, followed by an element‑wise
   multiplication with the output‑gate.  The multiplication is performed by a
   tiny Triton kernel (fulfilling the “use a kernel” requirement).
6️⃣ Final linear projection back to the original channel dimension.
7️⃣ Cast back to float32 (the API expects float32).

Only the forward pass is needed – no gradients are computed.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Tiny Triton kernel: dst *= src (both FP16, 1‑D flattened view)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,   # i32 – stride for dst (normally 1)
    src_stride: tl.constexpr,   # i32 – stride for src (normally 1)
    BLOCK_SIZE: tl.constexpr,   # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Batched matrix‑multiply for the O(N³) contraction.
# Computes C = A @ Bᵀ where A,B: [B·H, N, N] → out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(
    left: torch.Tensor,   # [B, N, N, H]   fp16
    right: torch.Tensor   # [B, N, N, H]   fp16
) -> torch.Tensor:
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] for a strided‑batched GEMM
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ   (torch uses the fast Tensor‑Core implementation)
    C = torch.bmm(A, B_.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Layer‑norm in float32 for stability, then cast to fp16
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)   # [B, N, N, dim]  fp16

    # -------------------------------------------------
    # 2️⃣ Fuse left/right projection + gate into a single GEMM
    # -------------------------------------------------
    # Cast weights once to fp16
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    # (4·H, dim) weight matrix
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )

    # Linear: produces [B, N, N, 4·H]
    combined = torch.nn.functional.linear(x, combined_weight)

    # ------------------------------------------------------------------
    # Slice back into the four components (views, no copy)
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj = combined[..., :H]          # [B,N,N,H]
    left_gate = combined[..., H:2*H]
    right_proj = combined[..., 2*H:3*H]
    right_gate = combined[..., 3*H:4*H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied after projection, before gate)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] (fp16)
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj.mul_(mask_fp16)
        right_proj.mul_(mask_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and in‑place gated projection
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj          # reuse storage
    left.mul_(left_gate)      # left = left_proj * left_gate   (in‑place)

    right = right_proj
    right.mul_(right_gate)    # right = right_proj * right_gate (in‑place)

    # ------------------------------------------------------------------
    # 5️⃣ Output‑gate (sigmoid) computed from the normalized input
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via a strided‑batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # still fp16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by out_gate (in‑place) via a tiny Triton kernel
    # ------------------------------------------------------------------
    out_flat  = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024                      # threads per block
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,               # dst stride
        1,               # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 🔚 Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
