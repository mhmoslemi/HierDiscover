"""
TriMul (outgoing) – AlphaFold‑3 style forward pass
-------------------------------------------------
The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and produces an output
with the same shape.

Algorithm (mixed‑precision, GPU‑only, no gradients)
1. Layer‑norm on the last dimension (float32) → fp16.
2. Two concatenated linear projections (X·W_proj) produce left/right tensors
   of shape (B,N,N,H) in a single GEMM (H = hidden_dim).
3. Two concatenated gating projections (X·W_gate) with a sigmoid produce
   left‑gate/right‑gate (same shape).  Both gate and proj are fused by a
   simple split+elementwise‑mul, reducing memory traffic.
4. Optionally mask left/right (mask shape B×N×N broadcast on H).
5. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed as a batched GEMM C = A @ Bᵀ with a high‑performance Triton
   kernel (fp16→fp32 accumulation, tensor‑core MMA, 64×64×64 tiles,
   8 warps, 3 stages).
6. Second Layer‑norm (fp16) on the hidden dimension.
7. Gating with out_gate (sigmoid of a linear projection of X) and final
   linear projection back to C.
8. Cast result back to float32 for the caller.

Only the O(N³) contraction is written in Triton; all other steps use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B, C : [M, N, K]   (M = B * hidden, N = seq_len, K = seq_len)
#   A, B, C are fp16, accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]   (will be read transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides   (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides   (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides   (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile along first N dimension (i)
    pid_n = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden dimension (M)

    # ------------------------------------------------------------------
    # Compute tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction (k) dimension
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load a tile of A  (shape BLOCK_M × BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # --------------------------------------------------------------
        # Load a tile of Bᵀ  (shape BLOCK_K × BLOCK_N)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (convert back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the custom Triton kernel above.
    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H] (fp16, contiguous)
    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Pack batch & hidden together → [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile size – 64×64×64 works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,                # extra stage for better pre‑fetching
    )

    # Reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``, optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → fp16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast all linear weights to fp16 once
    # -------------------------------------------------
    # Projection weights (left & right) – concatenate
    proj_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                     # shape (2*hidden, dim)

    # Gate weights (left_gate & right_gate) – concatenate
    gate_concat_w = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # shape (2*hidden, dim)

    # Out‑gate weight
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Final projection back to dim
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # LayerNorm parameters for the hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Joint linear projections: left+right (fp16)
    # -------------------------------------------------
    proj_concat = torch.nn.functional.linear(x, proj_concat_w)          # [B,N,N,2*H] fp16
    left, right = torch.split(proj_concat, hidden_dim, dim=-1)         # each [B,N,N,H]

    # -------------------------------------------------
    # 5️⃣ Joint gating projections (sigmoid) (fp16)
    # -------------------------------------------------
    gate_concat = torch.nn.functional.linear(x, gate_concat_w)          # [B,N,N,2*H] fp16
    gate_concat = torch.sigmoid(gate_concat)
    left_gate, right_gate = torch.split(gate_concat, hidden_dim, dim=-1)

    # -------------------------------------------------
    # 6️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 7️⃣ Fuse gate into projection (in‑place)
    # -------------------------------------------------
    torch.mul(left, left_gate, out=left)   # left := left * left_gate
    torch.mul(right, right_gate, out=right)   # right := right * right_gate

    # -------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via Triton (fp16 I/O)
    # -------------------------------------------------
    # Ensure tensors are contiguous before reshaping for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 1️⃣0️⃣ Out‑gate (computed from the original X) and final linear
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] fp16
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)                       # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 1️⃣1️⃣ Cast back to float32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
