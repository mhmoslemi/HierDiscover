"""
Highly‑optimized forward pass for the “outgoing” TriMul module (AlphaFold‑3).

Algorithm
---------
1.  Layer‑norm the 4‑D input (FP32 → FP16 for the heavy work).
2.  Fuse the five linear projections/gates into three calls:
        * left_proj + left_gate
        * right_proj + right_gate
        * out_gate
   (all in FP32 for numerical stability).
3.  Apply the optional mask, Sigmoid gates and fuse the gates into the
    projected tensors.
4.  Cast the two tensors that participate in the cubic contraction to FP16,
    pad the sequence dimension to a multiple of 64 (no bounds‑checks later).
5.  Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton kernel that performs a batched matrix‑multiply
   (C = A @ Bᵀ) for every (batch, hidden) pair.
   * Tile size 64×64×64, FP32 accumulation, 16 warps, 4 pipeline stages.
   * No runtime mask checks – padding guarantees all loads are in‑bounds.
6.  Strip the padding, second LayerNorm, output‑gate multiplication and final
    linear projection back to the original `dim`.
7.  Return the result in FP32.

The only non‑trivial operation in Triton is the batched mat‑mul; all other
steps are expressed with high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ
#   A, B : [M, N, K]  (M = batch * hidden, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel (FP16×FP16 → FP32)."""
    pid_z = tl.program_id(2)        # batch * hidden index
    pid_m = tl.program_id(0)        # tile along output‑row (i)
    pid_n = tl.program_id(1)        # tile along output‑col (j)

    # tile offsets (no mask – N is padded to a multiple of BLOCK_*)
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current (batch, hidden) slice
    a_base = A_ptr + pid_z * stride_am
    b_base = B_ptr + pid_z * stride_bm

    # pointers that will be offset inside the k‑loop
    a_ptrs = a_base + offs_i[:, None] * stride_ai          # shape (BLOCK_M, 1)
    b_ptrs = b_base + offs_j[None, :] * stride_bj          # shape (1, BLOCK_N)

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A[i, k] tile
        a = tl.load(a_ptrs + offs_k[None, :] * stride_ak)   # (BLOCK_M, BLOCK_K)
        # Bᵀ[k, j] tile (i.e. B[k, j])
        b = tl.load(b_ptrs + offs_k[:, None] * stride_bk)   # (BLOCK_K, BLOCK_N)

        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # write back FP32 result (no mask needed – padding guarantees safety)
    c_ptrs = C_ptr + pid_z * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc)


def _batched_matmul_fp16_to_fp32(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Batched mat‑mul C = left @ rightᵀ for each (batch, hidden) pair.
    left/right : FP16 tensors of shape [B, N, N, H] (already padded to a multiple of `tile`).
    Returns a FP32 tensor of the same shape (still padded).
    """
    B, N, _, H = left.shape
    # pack batch and hidden dimensions → [B*H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    A = left.view(B * H, N, N)                     # [M, N, K]
    B_ = right.view(B * H, N, N)                  # [M, N, K]

    # output tensor (FP32)
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # strides (in element units)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # launch configuration
    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_gemm_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,          # more warps exploit H100 SM resources
        num_stages=4,          # deeper pipeline
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (gradient‑free).

    Arguments
    ---------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – keep high‑precision statistics
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into three (still in float32)
    # ------------------------------------------------------------------
    # combine left_proj & left_gate
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )  # [2*hidden_dim, dim]

    # combine right_proj & right_gate
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )  # [2*hidden_dim, dim]

    out_gate_weight = weights["out_gate.weight"]  # [hidden_dim, dim]

    # flatten the 4‑D tensor for a single GEMM
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # [B*N*N, dim] float32

    # linear passes (batched GEMM, highly optimized)
    left_comb = torch.nn.functional.linear(x_flat, left_comb_weight)   # [B*N*N, 2*hidden_dim]
    right_comb = torch.nn.functional.linear(x_flat, right_comb_weight) # [B*N*N, 2*hidden_dim]
    out_gate_lin = torch.nn.functional.linear(x_flat, out_gate_weight) # [B*N*N, hidden_dim]

    # split the concatenated tensors
    left_proj, left_gate_pre = left_comb.split(hidden_dim, dim=1)
    right_proj, right_gate_pre = right_comb.split(hidden_dim, dim=1)

    # sigmoid gates (still float32)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_lin)

    # reshape back to [B, N, N, hidden_dim]
    left_proj = left_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp32 = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp32
        right_proj = right_proj * mask_fp32

    # ------------------------------------------------------------------
    # 4️⃣ Fuse gates into the projected tensors
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # float32
    right = right_proj * right_gate       # float32

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 for the O(N³) contraction and pad to a multiple of TILE
    # ------------------------------------------------------------------
    TILE = 64
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left_fp16.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right_fp16.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left_fp16
        right_pad[:, :N, :N, :] = right_fp16
    else:
        left_pad = left_fp16
        right_pad = right_fp16
        N_pad = N

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via the optimized Triton GEMM (FP16→FP32)
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16_to_fp32(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, hidden_dim] FP32

    # Strip padding back to the original sequence length
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (float32), output gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                # element‑wise output gate (float32)

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 8️⃣ Return result in FP32
    # ------------------------------------------------------------------
    return out
