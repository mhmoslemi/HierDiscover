"""
Outgoing TriMul (AlphaFold‑3) forward pass – high‑performance Triton implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣  Linear projections (left / right) and gating projections are computed with
    PyTorch fp16 primitives.  An optional pair‑wise mask is applied before gating.
3️⃣  The core O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is evaluated as a batched matrix‑multiply C = A @ Bᵀ with a custom Triton
    kernel.  The tensors are reshaped to [B·H, N, N] (H = hidden_dim) and the
    kernel performs a mixed‑precision (fp16×fp16→fp32) MMA using a 128×128×32
    tiling.  The accumulator resides in fp32 for numerical stability.
4️⃣  Second LayerNorm (fp16) → element‑wise out‑gate (sigmoid) → final linear
    projection back to `dim`.
5️⃣  Cast the result to float32 (the harness expects float32).

Only the heavy contraction is written in Triton; the rest of the pipeline
leverages highly‑optimised PyTorch kernels.  This satisfies the “must use a
kernel” requirement while keeping the overall runtime as low as possible.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16, accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides of A (M, N, K)
    stride_bm, stride_bj, stride_bk,   # strides of B (M, N, K)
    stride_cm, stride_ci, stride_cj,   # strides of C (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile index along the first N dimension (i)
    pid_n = tl.program_id(1)   # tile index along the second N dimension (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile (transposed on the fly): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot: fp16×fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result tile (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel defined above.

    Arguments
    ---------
    left, right : torch.Tensor of shape [B, N, N, H]   (fp16, contiguous)

    Returns
    -------
    out : torch.Tensor of shape [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Pack batch & hidden dimension → [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides expressed in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()     # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()    # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()     # (N*N, N, 1)

    # Tile sizes – 128×128×32 works well on H100 for fp16 MMA
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_gemm_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # 8 warps provide good occupancy on H100
        num_stages=2,     # double‑buffering of loads
    )

    # Reshape back to the original layout [B, N, N, H]
    return C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # 0️⃣ Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_w) # [B, N, N, hidden]

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        # mask may be bool or float; ensure fp16 and correct shape
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gating (sigmoid) – all in fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gate into projection in‑place (saves a temporary)
    torch.mul(left, left_gate, out=left)
    torch.mul(right, right_gate, out=right)

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 in → fp16 out)
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    contracted = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden] fp16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over the hidden dimension
    # -------------------------------------------------
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Gating + final projection back to `dim`
    # -------------------------------------------------
    contracted = contracted * out_gate
    out = torch.nn.functional.linear(contracted, to_out_w)   # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
