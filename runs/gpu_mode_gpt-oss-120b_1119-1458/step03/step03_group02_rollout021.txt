"""
TriMul (outgoing) – fast Triton‑accelerated implementation
==========================================================

The forward pass consists of three stages:

1️⃣ **Layer‑norm + linear projections + gating** – performed entirely in
   PyTorch (FP16) because they are cheap compared to the O(N³) core.

2️⃣ **Heavy contraction**
```
out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
```
   This is a batched matrix multiplication C = A @ Bᵀ where
   A,B ∈ (B·H , N , N).  The kernel tiles the multiplication with
   128×128×32 blocks (chosen dynamically from the sequence length) and
   uses `tl.dot` (tensor‑core MMA) with FP16 inputs and FP32 accumulation.
   Each program processes one “hidden‑batch’’ (M‑index = batch·hidden) and
   one (i‑tile, j‑tile) pair.

3️⃣ **Second Layer‑norm, output gating and projection** – again done in
   PyTorch.  The final tensor is returned in float32.

All heavy memory traffic is confined to the Triton kernel; the remaining
operations use highly‑optimised PyTorch kernels.  The implementation
supports optional masking (``nomask=False``) and works for all test
configurations listed in the statement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batched matrix product for many independent (N×N) matrices.
    Each program (identified by pid_z) processes one hidden‑batch (M‑index).
    The computation is tiled over the two N‑axes with blocks BLOCK_M×BLOCK_N.
    """
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # hidden‑batch index (M)

    # -----------------------------------------------------------------
    # Offsets for the current tile
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # -----------------------------------------------------------------
    # Loop over K dimension
    # -----------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- Load A tile (M‑idx, i‑idx, k‑idx) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----- Load B tile (M‑idx, k‑idx, j‑idx) -----
        # This loads the transposed view of B (K × N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ----- Multiply‑accumulate (tensor‑core MMA) -----
        # tl.dot promotes the fp16 multiplication to fp32 accumulation.
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Store the result tile (cast back to fp16)
    # -----------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.
    right : torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] in FP16.
    """
    B, N, _, H = left.shape

    # -----------------------------------------------------------------
    # Pack (batch, hidden) into a single leading dimension
    # -----------------------------------------------------------------
    #   (B, N, N, H) -> (B*H, N, N)   (contiguous)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP16)
    C = torch.empty_like(A, dtype=torch.float16)

    # ---------------------------------------------------------------
    # Strides in number‑of‑elements (Triton expects i32)
    # ---------------------------------------------------------------
    stride_am, stride_ai, stride_ak = A.stride()      # (M,N,K)
    stride_bm, stride_bi, stride_bk = B_.stride()     # (M,N,K)
    stride_cm, stride_ci, stride_cj = C.stride()      # (M,N,N)

    # ---------------------------------------------------------------
    # Choose block sizes based on N (empirically good for H100)
    # ---------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:   # N up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32   # 128×128 tiles give good L2 reuse

    # Grid definition
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # M‑tiles (batch * hidden)
    )

    # ------------------------------------------------------------------
    # Launch kernel
    # ------------------------------------------------------------------
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # Reshape back to original layout [B, N, N, H]
    # ------------------------------------------------------------------
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast to FP16 for the rest of the pipeline (tensor‑core friendly)
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Convert all linear weights to FP16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left  = torch.nn.functional.linear(x, left_proj_w)
    right = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (FP16 inputs, FP16 output)
    # -------------------------------------------------
    # Result shape: [B, N, N, hidden_dim] (FP16)
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # Release memory that is no longer needed early
    # -------------------------------------------------
    del left, right, x_norm, x

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (still FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection back to original dim
    # -------------------------------------------------
    out = out * out_gate                     # element‑wise hidden‑dim gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (required output type)
    # -------------------------------------------------
    return out.to(torch.float32)
