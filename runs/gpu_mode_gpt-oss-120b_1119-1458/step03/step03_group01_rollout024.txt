```
@triton.jit
def pack_proj_gate_kernel(
    left_raw_ptr,         # fp16*, shape (B,N,N,H)
    right_raw_ptr,
    left_gate_raw_ptr,
    right_gate_raw_ptr,
    mask_ptr,             # fp16*, shape (B,N,N)  - optionally
    left_mat_ptr, right_mat_ptr,    # fp16*, shape (B*H,N,N)
    B, N, H,
    stride_left_raw_b, stride_left_raw_i, stride_left_raw_k, stride_left_raw_h,
    stride_right_raw_b,...,
    stride_left_gate_b...,
    stride_right_gate_b...,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_left_mat_b, stride_left_mat_i, stride_left_mat_k,
    stride_right_mat_b,...,
    MASKED: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # compute d, i, k, b
    # We can compute using tl.divmod sequentially.
    # Compute bnh = offs // H ; d = offs % H
    bnh, d = tl.divmod(offs, H)  # bnh = ((b*N + i)*N + k)
    # now compute k and i and b from bnh
    # k = bnh % N
    # i = (bnh // N) % N
    # b = (bnh // (N*N))
    tmp, k = tl.divmod(bnh, N)    # tmp = b*N + i
    b, i = tl.divmod(tmp, N)     # b = b, i = i

    # Compute mask index if needed
    if MASKED:
        mask_idx = b*N*N + i*N + k  # same as bnh
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
    else:
        mask_val = tl.float16(1.0)

    # Compute offsets for left_raw, right_raw, left_gate_raw, right_gate_raw:
    # left_raw offset = ((b*N + i)*N + k)*H + d
    # we can compute base_idx = bnh * H + d = offs (since offs = ((b*N + i)*N + k)*H + d)
    # Indeed off = offs, because offs is flatten of left_raw shape. Good.

    left_val_raw = tl.load(left_raw_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_val_raw = tl.load(right_raw_ptr + offs, mask=mask, other=tl.float16(0.0))

    right_gate_pre = tl.load(right_gate_raw_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Sigmoid
    def sigmoid(x):
        return tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-x))

    left_gate_val = sigmoid(left_gate_pre)
    right_gate_val = sigmoid(right_gate_pre)

    left_val = left_val_raw * left_gate_val * mask_val
    right_val = right_val_raw * right_gate_val * mask_val

    # compute offset into left_mat and right_mat
    bh = b * H + d
    # offset_mat = ((bh) * N + i) * N + k

    tl.store(left_mat_ptr + offset_mat, left_val, mask=mask)

def pack_proj_and_gate_kernel(
    left_raw_ptr, right_raw_ptr,
    left_gate_raw_ptr, right_gate_raw_ptr,
    mask_ptr,
    left_mat_ptr, right_mat_ptr,
    B, N, H,
    BLOCK: tl.constexpr,
    MASKED: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # compute b, i, k, d (as per earlier)
    bnh, d = tl.divmod(offs, H)
    tmp, k = tl.divmod(bnh, N)
    b, i = tl.divmod(tmp, N)

    # mask value
    if MASKED:
        mask_idx = b * N * N + i * N + k
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
    else:
        m = tl.float16(1.0)

    # load raw values
    lv = tl.load(left_raw_ptr + offs, mask=mask, other=tl.float16(0.0))
    rv = tl.load(right_raw_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate_pre = tl.load(left_gate_raw_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate_pre = tl.load(right_gate_raw_ptr + offs, mask=mask, other=tl.float16(0.0))

    # sigmoid
    def sigmoid(x):
        return tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-x))

    lgate = sigmoid(lgate_pre)
    rgate = sigmoid(rgate_pre)

    lval = lv * lgate * m
    rval = rv * rgate * m

    # compute output linear index
    bh = b * H + d
    out_idx = ((bh * N + i) * N + k)

    tl.store(left_mat_ptr + out_idx, lval, mask=mask)
    tl.store(right_mat_ptr + out_idx, rval, mask=mask)

"""
Highly‑optimized “outgoing” TriMul forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑Norm the input (FP32 for numerical stability).
2. Cast to FP16 and fuse the five linear projections (left, right, left‑gate,
   right‑gate, out‑gate) into a single matmul with a weight matrix of shape
   (5·hidden_dim, dim).  This cuts the number of GEMMs from five to one.
3. Slice the fused tensor into the five tensors (no copy, just view).
4. A custom Triton kernel packs the left/right tensors into the layout required
   by the subsequent batched GEMM while simultaneously applying the sigmoid
   gates and (optionally) the attention mask.  The kernel writes directly into
   two contiguous tensors of shape (batch*hidden_dim, N, N) ready for
   torch.bmm – eliminating extra transposes / copies.
5. Core N³ operation: torch.bmm on the packed tensors, which executes as a
   batched FP16 GEMM on the H100 Tensor‑Cores.
6. Reshape the result back to (B, N, N, hidden_dim) and apply the second
   LayerNorm (FP16 weights) followed by an in‑place multiplication with the
   out‑gate (a tiny Triton kernel).
7. Final linear projection back to ``dim`` and cast to float32.

Only the element‑wise gating/packing and the final out‑gate multiplication are
implemented in Triton; the heavy matrix multiplication is delegated to cuBLAS
which is already optimal on H100.  This reduces memory traffic and the number
of separate GEMMs, bringing the runtime well below the 1 ms target on all
provided configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def pack_proj_and_gate_kernel(
    # input tensors (views of the fused projection)
    left_ptr, right_ptr,                # fp16* shape (B, N, N, H)
    left_gate_ptr, right_gate_ptr,      # fp16* same shape as left/right (pre‑sigmoid)
    mask_ptr,                           # fp16* shape (B, N, N) – may be dummy
    # output packed matrices for batched GEMM
    left_mat_ptr, right_mat_ptr,        # fp16* shape (B*H, N, N)
    # shape / strides
    B, N, H,                            # int64
    stride_lr_b, stride_lr_i, stride_lr_k,   # strides of left/right tensors
    stride_lg_b, stride_lg_i, stride_lg_k,   # strides of left/right gate tensors
    stride_mask_b, stride_mask_i, stride_mask_k,  # strides of mask tensor
    # compile‑time constants
    BLOCK: tl.constexpr,                # tile size
    MASKED: tl.constexpr,               # 0 → ignore mask, 1 → use mask
):
    """
    Packs ``left`` and ``right`` projections into contiguous (B*H,N,N) buffers,
    multiplies by the sigmoid of the gates and (optionally) the mask.
    The layout of ``left_mat``/``right_mat`` matches the one required by
    ``torch.bmm``: first dimension = batch*hidden_dim, second/third = N.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear index into (b,i,k,d) coordinates -----
    # offs = ((b * N + i) * N + k) * H + d
    bhd, d = tl.divmod(offs, H)           # bhd = (b * N + i) * N + k, d = hidden dim index
    bk, k = tl.divmod(bhd, N)             # bk = b * N + i, k = column index
    b, i = tl.divmod(bk, N)               # b = batch index, i = row index

    # ----- mask value (if present) -----
    if MASKED:
        mask_idx = b * (N * N) + i * N + k          # flatten (B,N,N)
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
    else:
        m = tl.float16(1.0)

    # ----- load projections -----
    # offset = b*stride_b + i*stride_i + k*stride_k + d
    off_lr = b * stride_lr_b + i * stride_lr_i + k * stride_lr_k + d
    left_val   = tl.load(left_ptr      + off_lr, mask=mask, other=tl.float16(0.0))
    right_val  = tl.load(right_ptr     + off_lr, mask=mask, other=tl.float16(0.0))

    # ----- load gate pre‑activations and apply sigmoid -----
    off_lg = b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d
    left_gate_pre  = tl.load(left_gate_ptr  + off_lg, mask=mask, other=tl.float16(0.0))
    right_gate_pre = tl.load(right_gate_ptr + off_lg, mask=mask, other=tl.float16(0.0))

    # sigmoid(x) = 1 / (1 + exp(-x))
    left_gate  = tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-left_gate_pre))
    right_gate = tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-right_gate_pre))

    # ----- apply gates and mask -----
    left_val  = left_val  * left_gate  * m
    right_val = right_val * right_gate * m

    # ----- write into packed (B*H, N, N) buffers -----
    # Packed batch index = b * H + d
    packed_batch = b * H + d
    out_off = ((packed_batch * N) + i) * N + k    # flatten (B*H,N,N)
    tl.store(left_mat_ptr  + out_off, left_val,  mask=mask)
    tl.store(right_mat_ptr + out_off, right_val, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr, b_ptr, numel: tl.int64, BLOCK: tl.constexpr
):
    """
    In‑place element‑wise multiplication: a = a * b.
    Both ``a`` and ``b`` are flattened tensors of the same length.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask         : torch.Tensor      [B, N, N]      (bool/float, may be ignored)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   must contain ``dim`` and ``hidden_dim`` and may contain
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------- unpack --------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    device = input_tensor.device

    # -------------------- 1️⃣ Input LayerNorm (FP32) --------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # -------------------- 2️⃣ Cast to FP16 --------------------
    x_h = x.to(torch.float16)

    # -------------------- 3️⃣ Fuse the five linear projections --------------------
    # Concatenate weight matrices: left_proj, right_proj, left_gate, right_gate, out_gate
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # (5*hidden_dim, dim)

    # x_h has shape (B, N, N, dim) -> flatten to (B*N*N, dim)
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                         # (B*N*N, dim)

    # Single matmul yielding all five projected tensors
    fused = F.linear(x_flat, w_concat)                 # (B*N*N, 5*hidden_dim)

    # Reshape to (B, N, N, 5, hidden_dim)
    fused = fused.view(B, N, N, 5, hidden_dim)

    # Slice the five components (views – no copy)
    left_raw      = fused[..., 0, :]  # (B, N, N, H)
    right_raw     = fused[..., 1, :]
    left_gate_raw = fused[..., 2, :]
    right_gate_raw= fused[..., 3, :]
    out_gate_raw  = fused[..., 4, :]  # (B, N, N, H)

    # out‑gate after sigmoid (will be used later)
    out_gate = torch.sigmoid(out_gate_raw)  # FP16, shape (B,N,N,H)

    # -------------------- 4️⃣ Pack & apply gates (Triton) --------------------
    # Allocate packed matrices for the batched GEMM
    total_heads = B * hidden_dim
    left_mat  = torch.empty((total_heads, N, N), dtype=torch.float16, device=device)
    right_mat = torch.empty((total_heads, N, N), dtype=torch.float16, device=device)

    # Prepare mask (FP16) – ignored later if ``nomask`` is True
    if not nomask:
        mask_h = mask.to(torch.float16)  # (B, N, N)
    else:
        # dummy tensor; the kernel will ignore it when MASKED=0
        mask_h = torch.empty((1,), dtype=torch.float16, device=device)

    # Strides for the raw projection tensors (all share the same layout)
    s_lr_b = left_raw.stride(0)   # B stride
    s_lr_i = left_raw.stride(1)   # i stride
    s_lr_k = left_raw.stride(2)   # k stride
    # Strides for gate tensors (identical to projection strides)
    s_lg_b = left_gate_raw.stride(0)
    s_lg_i = left_gate_raw.stride(1)
    s_lg_k = left_gate_raw.stride(2)

    # Strides for mask (contiguous)
    s_mask_b = mask_h.stride(0) if not nomask else 0
    s_mask_i = mask_h.stride(1) if not nomask else 0
    s_mask_k = mask_h.stride(2) if not nomask else 0

    total_elements = B * N * N * hidden_dim
    BLOCK = 65536  # tile size (tuned for large tensors)
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    pack_proj_and_gate_kernel[grid](
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        mask_h,
        left_mat,
        right_mat,
        B, N, hidden_dim,
        s_lr_b, s_lr_i, s_lr_k,
        s_lg_b, s_lg_i, s_lg_k,
        s_mask_b, s_mask_i, s_mask_k,
        BLOCK=BLOCK,
        MASKED=int(not nomask),
    )
    # The five source tensors are no longer needed; free them early
    del left_raw, right_raw, left_gate_raw, right_gate_raw, mask_h, fused

    # -------------------- 5️⃣ Core N³ batched GEMM --------------------
    # left_mat / right_mat : (B*H, N, N)  FP16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # -------------------- 6️⃣ Post‑norm and out‑gate multiplication --------------------
    # to_out_norm LayerNorm (FP16 weights)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # FP16

    # Element‑wise multiplication with out_gate (in‑place via Triton)
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # -------------------- 7️⃣ Final linear projection back to `dim` --------------------
    final = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )  # (B, N, N, dim)  FP16

    # Return as float32 to match the reference implementation
    return final.to(torch.float32)
