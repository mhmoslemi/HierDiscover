"""
Optimized outgoing TriMul forward pass (AlphaFold‑3) using Triton
for the O(N³) contraction. The kernel performs a batched GEMM
C = A @ Bᵀ where A and B have shape (batch*hidden, N, N). The
kernel uses 64×64×64 tiling, FP16→FP32 accumulation and a
3‑stage pipeline. All other operations (LayerNorm, Linear,
Sigmoid) are kept in PyTorch with FP16 arithmetic; the first
LayerNorm stays in FP32 for numerical stability. A combined
Linear is used to fuse the five hidden‑dimensional linear
projections/gates into a single matmul, reducing kernel launch
overhead.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i tile
    pid_n = tl.program_id(1)   # j tile
    pid_z = tl.program_id(2)   # batch*hidden

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # load A tile (M, i, k)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # load B tile (M, k, j) – note B is stored as [M, K, N] (k, j)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        acc += tl.dot(a, b)   # fp16×fp16 -> fp32 accumulation

    # write back
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ for tensors
    left/right of shape [B, N, N, H] (FP16).

    Returns:
        out: torch.Tensor of shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # pack batch and hidden into a single leading dimension
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # transpose k ↔ j

    out = torch.empty_like(left_, dtype=torch.float16)

    # strides in elements
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to original layout
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict : must contain ``dim`` and ``hidden_dim``;

    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # combine the five hidden‑dim linear layers into a single matmul
    combined_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ One fused linear projection that yields proj, gates, out_gate
    # ------------------------------------------------------------------
    # proj shape: [B, N, N, 5*hidden_dim]
    proj = torch.nn.functional.linear(x, combined_weight)  # fp16

    # split
    split = torch.split(proj, hidden_dim, dim=-1)
    left_proj = split[0]    # [B,N,N,H]
    right_proj = split[1]   # [B,N,N,H]
    left_gate_pre = split[2]
    right_gate_pre = split[3]
    out_gate_pre = split[4]

    # gate activations
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates (and optional mask) to the projected tensors
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM (TriMul)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float16),
        bias=to_out_norm_b.to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 8️⃣ Gating and final projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w.to(torch.float16))

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)

The heavy O(N³) contraction

arithmetic while the first LayerNorm stays in FP32 for numerical

projections/gates into a single matmul, reducing kernel launch

Both mask‑aware and mask‑free variants are supported.

def _trimul_matmul_kernel(

    stride_bm, stride_bk, stride_bj,   # strides for B (M, k, j)
    stride_cm, stride_ci, stride_cj,   # strides for C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,

    # ------------------------------------------------------------------

    mask_n = offs_n < N

def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)

    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides expressed in element counts (not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

        BLOCK_K=BLOCK_K,

    )

    # ---------------------------------------------------------------
    # 7️⃣ O(N³) contraction via the Triton batched GEMM (outgoing TriMul)
    # ---------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # ---------------------------------------------------------------
    # 8️⃣ Second LayerNorm on the hidden dimension
    # ---------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float16),
        bias=to_out_norm_b.to(torch.float16),
    )

    # ---------------------------------------------------------------
    # 9️⃣ Final gating + projection back to original dim
    # ---------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w.to(torch.float16))

    # ---------------------------------------------------------------
    # 10️⃣ Cast back to float32 as required by the harness
    # ---------------------------------------------------------------
    return out.to(torch.float32)
```
