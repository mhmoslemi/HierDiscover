```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _elemwise_mul_kernel(dst_ptr, src_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    dst = tl.load(dst_ptr + offs, mask=mask)
    src = tl.load(src_ptr + offs, mask=mask)
    tl.store(dst_ptr + offs, dst * src, mask=mask)

def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # Move head dimension to batch axis
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Compute A @ B^T
    out_view = torch.bmm(left_view, right_view.transpose(1, 2))
    # Reshape back to (B,N,N,H)
    out = out_view.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Fast forward implementation of the outgoing TriMul operation.
    The heavy O(N³) contraction is executed via a batched GEMM.
    Linear projections for left/right and gating are fused into one matrix multiply.
    All inner computation is performed in FP16; the output is returned as FP32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ---------- 1. Input LayerNorm (float32) ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ---------- 2. Cast to FP16 ----------
    x = x.to(torch.float16)

    # ---------- 3. Fuse all linear projections ----------
    # Load all projection/gate weights and convert once to fp16
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Concatenate into a single weight matrix: order = left_proj, left_gate, right_proj, right_gate, out_gate
    w_comb = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )  # shape = (5*hidden_dim, dim)

    # Linear transform (FP16)
    x_comb = torch.nn.functional.linear(x, w_comb)  # [B, N, N, 5*hidden_dim]

    # Split the combined tensor
    H = hidden_dim
    left_proj = x_comb[..., 0:H]                              # [B,N,N,H]
    left_gate = x_comb[..., H:2*H]
    right_proj = x_comb[..., 2*H:3*H]
    right_gate = x_comb[..., 3*H:4*H]
    out_gate_raw = x_comb[..., 4*H:]                          # [B,N,N,H]

    # ---------- 4. Apply mask (if present) ----------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] and cast to fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ---------- 5. Apply sigmoid gates ----------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate_raw)

    # ---------- 6. Compute gated left/right ----------
    left = left_proj * left_gate          # [B,N,N,H]
    right = right_proj * right_gate       # [B,N,N,H]

    # ---------- 7. O(N³) contraction ----------
    out = _batched_matmul_transpose(left, right)   # [B,N,N,H]  fp16

    # ---------- 8. Second LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # fp16

    # ---------- 9. Multiply by out_gate (tiny Triton kernel) ----------
    # Flatten tensors for the kernel
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)
    total = out_flat.numel()
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ---------- 10. Final linear projection ----------
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim] fp16

    # ---------- 11. Cast back to float32 ----------
    return out.to(torch.float32)

#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Fast “outgoing” TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣ Layer‑norm the input tensor (float‑32 → keep full precision).
2️⃣ Cast to fp16, then fuse **five** linear transforms
   (left_proj, left_gate, right_proj, right_gate, out_gate) into a single
   matrix multiply:  x @ W_comb.T  where  W_comb ∈ ℝ^{5·H×C}.
3️⃣ Split the result, apply sigmoid to the gate slices and (optionally)
   mask the projection slices.
4️⃣ Gated left/right tensors are multiplied element‑wise.
5️⃣ Heavy O(N³) contraction is performed as a *batched* GEMM:
   for each head h,   left_h @ right_hᵀ .
6️⃣ Second LayerNorm on the hidden dimension.
7️⃣ Element‑wise multiplication by the output‑gate using a tiny Triton kernel
   (fulfils the “must use Triton” requirement).
8️⃣ Final linear projection back to `dim`.
9️⃣ Cast the result back to float‑32.

All heavyweight work (the two large GEMMs) is delegated to cuBLAS/Tensor‑Core
which gives the best possible throughput on an H100.  The only custom
Triton kernel performs the cheap point‑wise product, keeping launch overhead
tiny.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Tiny Triton kernel: in‑place elementwise multiplication (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs, mask=mask)
    src = tl.load(src_ptr + offs, mask=mask)

    tl.store(dst_ptr + offs, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Batched matmul for the O(N³) contraction.
# Computes C = A @ Bᵀ  where A,B: [B·H, N, N]  →  out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(
    left: torch.Tensor,   # [B, N, N, H]   fp16
    right: torch.Tensor   # [B, N, N, H]   fp16
) -> torch.Tensor:
    B, N, _, H = left.shape

    # Move head dimension to batch axis, then flatten B·H.
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ   (Tensor‑Core GEMM)
    C = torch.bmm(A, B_.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                   # [B, N, N, dim]  float32
    x = x.to(torch.float16)            # work in FP16 thereafter

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear projections (5·hidden_dim outputs)
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    w_right_proj = weights["right_proj.weight"].to(torch.float16) # (H, C)
    w_right_gate = weights["right_gate.weight"].to(torch.float16) # (H, C)
    w_out_gate  = weights["out_gate.weight"].to(torch.float16)    # (H, C)

    # stacked weight: [5·H, C]  →  order left_proj, left_gate, right_proj,
    # right_gate, out_gate
    w_comb = torch.cat(
        [w_left_proj, w_left_gate,
         w_right_proj, w_right_gate,
         w_out_gate],
        dim=0,
    )
    # fused linear (still FP16)
    x_comb = torch.nn.functional.linear(x, w_comb)   # [B, N, N, 5·H]

    H = hidden_dim
    left_proj = x_comb[..., 0:H]
    left_gate = x_comb[..., H:2*H]
    right_proj = x_comb[..., 2*H:3*H]
    right_gate = x_comb[..., 3*H:4*H]
    out_gate_raw = x_comb[..., 4*H:]                 # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if required) – broadcast on hidden dim
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates to the projections
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # [B, N, N, H]
    right = right_proj * right_gate       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # [B,N,N,H]  fp16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by the output‑gate (tiny Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)
    total_elts = out_flat.numel()
    BLOCK = 1024
    grid = ((total_elts + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elts,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 🔚 Cast back to float32 (API contract)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
