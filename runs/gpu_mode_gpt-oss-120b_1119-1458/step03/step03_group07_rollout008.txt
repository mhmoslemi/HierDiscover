"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input tensor (float32 → fp16 for speed).
2️⃣ Fuse each projection+gate pair into a single linear call:
       [proj, gate] = Linear(x, [W_proj; W_gate])   → split after.
3️⃣ (Optional) Apply the pair‑wise mask after the projections.
4️⃣ Apply sigmoid to the gates and multiply (proj * gate).
5️⃣ Heavy O(N³) contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] · right[b,j,k,d]
   performed by a single `torch.einsum` which avoids the costly
   permute‑contiguous copy required by a naïve batched‑gemm.
6️⃣ Layer‑norm the intermediate result.
7️⃣ Compute the output‑gate (sigmoid of a linear on the normalized input).
8️⃣ Fuse the out‑gate multiplication with a tiny Triton kernel
   (in‑place element‑wise `out *= out_gate`).
9️⃣ Final linear projection back to the original channel dimension.
🔚 Cast the result to float32 (the API’s expected dtype).

Only step 8 uses a custom Triton kernel – the rest relies on
high‑performance PyTorch kernels (especially the Tensor‑Core‐enabled
einsum for the O(N³) term).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Tiny Triton kernel: dst *= src  (both fp16, 1‑D flattened view)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination tensor (in‑place)
    src_ptr,            # *fp16 – source tensor (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # torch.nn.functional.layer_norm works with fp16 weights if input is fp16
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  float32

    x_half = x_norm.to(torch.float16)  # work in half precision from now on

    # ------------------------------------------------------------------
    # 2️⃣ Cast linear weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (hidden_dim, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (hidden_dim,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate (2 linear calls instead of 4)
    # ------------------------------------------------------------------
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*hidden_dim, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden_dim, dim)

    # shape after linear: [B, N, N, 2*hidden_dim]
    left_comb  = torch.nn.functional.linear(x_half, left_comb_w)   # FP16
    right_comb = torch.nn.functional.linear(x_half, right_comb_w)  # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Split back into proj / gate and (optional) mask
    # ------------------------------------------------------------------
    left_proj = left_comb[..., :hidden_dim]          # [B,N,N,hidden_dim]
    left_gate = left_comb[..., hidden_dim:]          # [B,N,N,hidden_dim]

    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:]

    if not nomask:
        # mask shape [B,N,N] → broadcast to [B,N,N,1] → FP16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate       # [B,N,N,hidden_dim]
    right = right_proj * right_gate    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via einsum (no large permute copy)
    # ------------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate (sigmoid) and fused element‑wise multiplication
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_half, out_gate_w))  # [B,N,N,hidden_dim] FP16

    # flatten for Triton kernel (in‑place dst ← dst * src)
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024                # max threads per block on H100
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,          # dst stride
        1,          # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 🔚 Cast back to float32 (expected output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
