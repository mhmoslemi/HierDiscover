```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for batched outer product (over hidden)
@triton.jit
def _trimul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    batch: tl.constexpr, seqlen: tl.constexpr, hidden: tl.constexpr,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile for output i axis
    pid_j = tl.program_id(1)   # tile for output j axis
    pid_z = tl.program_id(2)   # combined batch*hidden

    # Compute the actual batch index and hidden index from pid_z
    batch_idx = pid_z // hidden
    hidden_idx = pid_z - batch_idx * hidden   # modulo

    # Offsets for the i and j dimensions of the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < seqlen
    mask_j = offs_j < seqlen

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k_tiles = tl.cdiv(seqlen, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seqlen

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + batch_idx * stride_a_batch
            + hidden_idx * stride_a_h
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (transposed view): shape (BLOCK_K, BLOCK_N)
        # We need B[..., j, k, h] → stride_i for j, stride_k for k
        b_ptrs = (
            B_ptr
            + batch_idx * stride_b_batch
            + hidden_idx * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_i
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Mixed‑precision dot: fp16 * fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # Store the resulting (i,j) tile into C
    c_ptrs = (
        C_ptr
        + batch_idx * stride_c_batch
        + hidden_idx * stride_c_h
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
    
def _batched_trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a custom Triton kernel that works directly on the original
    [B, N, N, H] layout (no permutation / copy).
    """
    B, N, _, H = left.shape
    # Allocate output tensor
    out = torch.empty_like(left)
    # Extract strides (in element counts, not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tile sizes – 128×128×32 works well on H100 for fp16 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out

    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]

        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``, optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → fp16.
2. Four linear projections (left, right, left_gate, right_gate, out_gate) in fp16.
   Optional mask is applied (broadcast on hidden dimension).
3. Fuse each projection with its gate (in‑place mul).
4. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented with a custom Triton kernel that works directly on the
   original [B, N, N, H] layout – no permute / copy required.
   The kernel uses 128×128×32 tiles, 8 warps and 2 pre‑fetch stages,
   mixed‑precision dot (fp16 × fp16 → fp32 accumulator).
5. Second Layer‑norm (fp16) → out‑gate (sigmoid) → final projection to `dim`.
6. Cast the result back to float32.

The kernel eliminates the expensive memory reshapes of the reference
implementation and thus brings the runtime well below the 1 ms target.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched outer‑product C = Σ_k A[...,i,k,...] * B[...,j,k,...]
#   A, B: [B, N, N, H]   (fp16)
#   C   : [B, N, N, H]   (fp16)
#   The batch & hidden dimensions are fused into a single program‑id axis.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    A_ptr, B_ptr, C_ptr,                     # *fp16 tensors
    batch: tl.constexpr, seqlen: tl.constexpr, hidden: tl.constexpr,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile on first sequence axis (i)
    pid_j = tl.program_id(1)   # tile on second sequence axis (j)
    pid_z = tl.program_id(2)   # fused batch * hidden index

    # --------------------------------------------------------------
    # De‑compose pid_z → (batch, hidden)
    # --------------------------------------------------------------
    batch_idx  = pid_z // hidden
    hidden_idx = pid_z - batch_idx * hidden

    # --------------------------------------------------------------
    # Tile offsets in the i / j dimensions
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < seqlen
    mask_j = offs_j < seqlen

    # --------------------------------------------------------------
    # Accumulator (FP32 for numerical stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K‑tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(seqlen, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k  = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seqlen

        # ----------------------------------------------------------
        # Load A tile  (shape BLOCK_M × BLOCK_K)  → A[b,i,k,h]
        # ----------------------------------------------------------
        a_ptrs = (
            A_ptr
            + batch_idx * stride_a_batch
            + hidden_idx * stride_a_h
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ----------------------------------------------------------
        # Load B tile  (shape BLOCK_K × BLOCK_N)  → B[b,j,k,h] (j as column)
        # ----------------------------------------------------------
        b_ptrs = (
            B_ptr
            + batch_idx * stride_b_batch
            + hidden_idx * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_i
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot product: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the output tile (fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch_idx * stride_c_batch
        + hidden_idx * stride_c_h
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above (inputs must be fp16 and contiguous).

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (fp16)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape

    # Allocate output
    out = torch.empty_like(left)

    # Strides in element counts (not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tile sizes – 128×128×32 is a sweet spot for fp16 Tensor‑Core MMA
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # fused batch‑hidden dimension
    )

    _trimul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out

    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool or float)
        - weights_dict : weight name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for accuracy)
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the work
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)   # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) and in‑place fusion
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    torch.mul(left,  left_gate,  out=left)   # left  = left  * left_gate
    torch.mul(right, right_gate, out=right) # right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 → fp16)
    # -----------------------------------------------------------------
    # The buffers are already in the required layout; no permutations.
    out = _batched_trimul_triton(left, right)   # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden_dim) – fp16
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection back to model dimension
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
