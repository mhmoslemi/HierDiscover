"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold3 style).

The implementation follows the reference PyTorch module but moves almost all
dense work to FP16 and reduces the number of heavy linear layers:

* left_proj and right_proj are computed together with a single
  `torch.nn.functional.linear` (2 × hidden_dim output).
* left_gate and right_gate are also fused into one linear (2 × hidden_dim).
* The mask is applied *before* the projections, which is mathematically
  equivalent but saves an extra element‑wise pass.
* Element‑wise multiplications of projections with their gates and the final
  out‑gate are performed by tiny Triton kernels (`apply_gates_kernel` and
  `mul_inplace_kernel`).
* The N³ core (einsum over the k‑dimension) is carried out with a batched
  matrix multiplication via `torch.einsum`, which internally exploits fast
  GEMM kernels without any extra memory copies.
* All intermediate tensors are kept in FP16; the two LayerNorms run in FP32
  for numerical stability.  The final output is returned in float32.

The function signature matches the required `custom_kernel` entry point.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels for element‑wise multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,        # fp16* , shape (numel,)
    left_gate_ptr,   # fp16*
    right_ptr,       # fp16*
    right_gate_ptr,  # fp16*
    numel: tl.int64,    # total number of elements per tensor
    BLOCK: tl.constexpr,
):
    """
    In‑place:
        left  = left  * left_gate
        right = right * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l  = tl.load(left_ptr + offs,   mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs,  l * lg, mask=mask)

    r  = tl.load(right_ptr + offs,   mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,           # fp16*  (output tensor, will be overwritten)
    b_ptr,           # fp16*  (operand)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        a = a * b
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)

        - mask         : torch.Tensor      [B, N, N]       (bool/float) – may be dummy
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict with keys
                         * "dim" (int) – input feature dimension
                         * "hidden_dim" (int) – hidden dimension used internally
                         * "nomask" (bool, optional) – True if mask is all‑ones

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]   FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the rest of the heavy work runs in half precision
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # [B, N, N, dim]   fp16

    # ------------------------------------------------------------------
    # 3️⃣ Prepare mask (if any). Mask is broadcast over the hidden dim.
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – convert to fp16
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        proj_input = x_h * mask_h                       # apply mask before projection
    else:
        proj_input = x_h

    # ------------------------------------------------------------------
    # 4️⃣ Fuse left/right projections (2 * hidden_dim output)
    # ------------------------------------------------------------------
    left_right_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                               # [(2*H), dim]

    proj = F.linear(proj_input, left_right_weight)   # [B, N, N, 2*H]  fp16
    left, right = torch.split(proj, hidden_dim, dim=-1)  # each [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse left/right gates (2 * hidden_dim output) – gates use *unmasked* x
    # ------------------------------------------------------------------
    gate_weight = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                               # [(2*H), dim]

    gate_pre = F.linear(x_h, gate_weight)            # [B,N,N,2*H]  fp16
    left_gate_pre, right_gate_pre = torch.split(gate_pre, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)         # fp16
    right_gate = torch.sigmoid(right_gate_pre)       # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (separate linear, hidden_dim output)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )                                                 # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right tensors with their gates via a Triton kernel
    # ------------------------------------------------------------------
    # All tensors are already contiguous after F.linear, so we can flatten safely.
    numel = left.numel()               # B * N * N * H
    BLOCK = 65536                      # ~64 k elements per program
    grid = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Gate tensors are no longer needed – free them early
    del left_gate, right_gate, gate_pre, gate_weight

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ reduction: einsum over the k‑dimension
    #    This is equivalent to a batch of (H × B) matrix multiplications of
    #    size (N×N) @ (N×N)ᵀ.
    # ------------------------------------------------------------------
    # Using einsum avoids an explicit copy of the tensors.
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm over `hidden_dim` (still FP16 weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 🔟 Multiply by the output gate (in‑place) via a Triton kernel
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate, left, right   # release memory early

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # [B, N, N, dim]   fp16

    # Return in the expected FP32 precision
    return final.to(torch.float32)
