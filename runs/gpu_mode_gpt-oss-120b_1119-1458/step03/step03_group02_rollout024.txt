def _batched_matmul_transpose_fp16(left, right):
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    C = torch.empty_like(A, dtype=torch.float16)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bi, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128
    grid = ((N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N, B*H)
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B*H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C
```

"""
TriMul (outgoing) – fast Triton‑accelerated implementation
==========================================================

The forward pass consists of:
1️⃣ Layer‑norm the input (FP32 → FP16).
2️⃣ Fuse the 5 linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single FP16 matmul – this eliminates
   repeated loads of the large 4‑D tensor.
3️⃣ Apply the sigmoid gates and optional mask in‑place.
4️⃣ Heavy O(N³) contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   realised as a batched matrix multiplication (A @ Bᵀ) using a
   custom Triton kernel (64 × 64 tiles, K‑tile = 128 for fewer loop
   iterations, FP16 inputs, FP32 accumulation).
5️⃣ 2nd Layer‑norm over the hidden dimension, gating, and final
   linear projection back to the original feature dimension.
The result is returned as FP32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 dimensions
    stride_am, stride_ai, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bi, stride_bk,   # strides for B (M,N,K)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul contraction.
    The kernel processes one (M‑index) – i.e. (batch, hidden) pair –
    with a 2‑D tiling over the two sequence axes.
    """
    pid_i = tl.program_id(0)   # tile along output‑row (i)
    pid_j = tl.program_id(1)   # tile along output‑col (j)
    pid_z = tl.program_id(2)   # index over M (= batch * hidden_dim)

    # -------------------------------------------------
    # Offsets for the current tile (row/col of the output)
    # -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles (ceil division)
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # -------------------------------------------------
    # Main K‑loop
    # -------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile  (M‑idx, i‑idx, k‑idx)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile  (M‑idx, k‑idx, j‑idx) – this is the transposed view
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Tensor‑core MMA (fp16 × fp16 → fp32 accumulate)
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Store the partial result (cast back to fp16)
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that performs the batched matmul (A @ Bᵀ) using the kernel above
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ----------
    left : torch.Tensor   [B, N, N, H]  (fp16)
    right: torch.Tensor   [B, N, N, H]  (fp16)

    Returns
    -------
    torch.Tensor  [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # pack batch and hidden dimension into a single leading axis
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # strides in elements (Triton expects integer strides, not byte strides)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bi, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # tile sizes – 64×64 tiles with a 128‑element K‑tile give very good
    # occupancy on H100 (few K‑loop iterations, good register reuse).
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along first N dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along second N dimension
        B * H,                          # M‑tiles (batch * hidden)
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps per program gives good latency hiding
        num_stages=3,       # double‑buffering + one extra stage
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point – the forward pass of the “outgoing” TriMul module
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = x.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for numerical stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast to FP16 for the rest of the pipeline
    # -------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Fuse the five linear projections (proj + gates) into one matmul
    # -------------------------------------------------
    # Gather the weight matrices and cast to FP16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    # Concatenate along the output‑feature axis (out_features = H)
    # Resulting shape: (5*H, D)
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )   # fp16

    # Single fused linear: [B, N, N, 5*H] = x_fp16 @ fused_weightᵀ
    fused_out = torch.nn.functional.linear(x_fp16, fused_weight)   # fp16

    # Split back into the individual tensors
    split_tensors = torch.split(fused_out, hidden_dim, dim=-1)
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = split_tensors

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)   # [B, N, N, H]

    # -------------------------------------------------
    # 3️⃣ Element‑wise mask & gate fusion
    # -------------------------------------------------
    # mask: [B, N, N] → [B, N, N, 1]  (broadcast over hidden)
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        # left = left_proj * left_gate * mask
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction (batched matmul)
    # -------------------------------------------------
    # Results in shape [B, N, N, H] (fp16)
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 6️⃣ Final gating + projection back to the original dimension
    # -------------------------------------------------
    out = out * out_gate                                          # element‑wise gate
    to_out_w = weights["to_out.weight"].to(torch.float16)        # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)             # [B, N, N, dim] (fp16)

    # -------------------------------------------------
    # 7️⃣ Cast back to float32 (required output dtype)
    # -------------------------------------------------
    return out.to(torch.float32)
