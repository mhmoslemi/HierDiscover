"""
Outgoing TriMul (AlphaFold‑3) – forward only.

The kernel computes, for every hidden channel h,
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]        (i.e. left @ rightᵀ)

The surrounding operations (layer‑norms, linear projections, gating,
optional masking) are performed with high‑performance PyTorch primitives.
All heavy arithmetic is carried out in fp16 with fp32 accumulation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len)
#   C    : [M, N, N]
# --------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # fp16 *
    B_ptr,               # fp16 *
    C_ptr,               # fp16 *
    M, N, K,             # problem size
    stride_am, stride_ai, stride_ak,   # strides for A (M,i,k)
    stride_bm, stride_bj, stride_bk,   # strides for B (M,j,k) – note the order!!!
    stride_cm, stride_ci, stride_cj,   # strides for C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # (batch, hidden) index

    # tile offsets within the N dimension
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (higher precision for the long reduction)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ----- K‑loop --------------------------------------------------------------
    for k_iter in range(num_k_tiles):
        cur_k = k_iter * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A[i,k] tile -----------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Bᵀ[k,j] tile (i.e. reading B[j,k] as a (K,N) matrix) -----------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk      # rows  = k
            + offs_j[None, :] * stride_bj      # cols  = j
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # fused dot product (fp16×fp16 → fp32) ----------------------------------
        acc += tl.dot(a, b)

    # write back (cast to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Compute out = left @ rightᵀ for every (batch, hidden) pair.
    `left` and `right` must be fp16 tensors of shape [B, N, N, H].
    Returns a tensor of the same shape (still fp16).
    """
    B, N, _, H = left.shape

    # Bring hidden in front so each (b,h) slice becomes a contiguous N×N matrix.
    left_cont = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_cont = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden into a single leading dimension.
    A = left_cont.view(B * H, N, N)   # [M, N, N]
    B_ = right_cont.view(B * H, N, N)   # [M, N, N]

    # Output buffer
    C = torch.empty_like(A)

    # stride in element units (torch uses element strides)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : ``dim`` and ``hidden_dim`` required.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    )                     # [B, N, N, dim]  (fp32)
    x = x_norm.to(torch.float16)  # keep everything fp16 from now on

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)   # (hidden,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (hidden,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (dim, hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections & gating (all in fp16)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                 # (B*N*N, dim)

    left  = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # Optional mask – broadcast over hidden dimension
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # Gating (sigmoid)
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(x_flat, left_gate_w)
    ).view(B, N, N, hidden_dim)

    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, right_gate_w)
    ).view(B, N, N, hidden_dim)

    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    ).view(B, N, N, hidden_dim)

    # Fuse gates into the projected tensors
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via Triton
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right, tile=64)   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (still fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output gating + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 (the model expects fp32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
