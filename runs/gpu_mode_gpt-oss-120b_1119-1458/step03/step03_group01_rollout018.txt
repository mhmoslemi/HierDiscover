    left_ptr, right_ptr, out_ptr,
    N: tl.int64,
    stride_head: tl.int64,
    stride_i: tl.int64,
    stride_k: tl.int64,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul outgoing forward (AlphaFold3 style) – Triton‑accelerated.

Algorithm
---------
1️⃣ Normalize the input pair representation (LayerNorm over `dim`).
2️⃣ Cast to FP16 for all heavy arithmetic.
3️⃣ Linear projections → `left`, `right` (shape [B,N,N,H]) and three
   gates (`left_gate`, `right_gate`, `out_gate`) = sigmoid(Linear(x)).
4️⃣ Optional pair mask (broadcast over the hidden dimension).
5️⃣ Fuse gate application (left←left*left_gate, right←right*right_gate) with
   a tiny Triton kernel (`apply_gates_kernel`).
6️⃣ Core N³ operation:
   For every head (head = batch * hidden_dim) we need
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is exactly a batched matrix multiplication `A @ Bᵀ`.  A custom
   Triton kernel (`trimul_outgoing_kernel`) performs *all* heads in a
   single launch, tile‑wise over the output (i,j) and the reduction dim k.
   The kernel works on FP16 values and accumulates in FP32 for numerical
   stability.
7️⃣ Layer‑norm over the hidden dimension (`to_out_norm`).
8️⃣ Multiply by the output gate with an in‑place Triton kernel
   (`mul_inplace_kernel`).
9️⃣ Final linear projection back to the original channel dimension.
🔟 Return a FP32 tensor of shape [B, N, N, dim].

The implementation preserves the exact functional behaviour of the
reference PyTorch module while moving the dominant O(B·H·N³) work to a
single highly‑parallel Triton kernel, reducing kernel‑launch overhead
and improving arithmetic density on H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny element‑wise kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    a_ptr,      # fp16* – left tensor (flattened)
    a_gate_ptr, # fp16* – left gate (flattened)
    b_ptr,      # fp16* – right tensor (flattened)
    b_gate_ptr, # fp16* – right gate (flattened)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """Fuse left/right tensors with their respective gates (in‑place)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # left side
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    ag = tl.load(a_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * ag, mask=mask)

    # right side
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    bg = tl.load(b_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(b_ptr + offs, b * bg, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,   # fp16* – output tensor (flattened)
    b_ptr,   # fp16* – operand (flattened)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place element‑wise multiplication a = a * b."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM kernel for the N³ “einsum”
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,   # fp16*  shape (Htotal, N, N)
    right_ptr,  # fp16*  same layout as left_ptr
    out_ptr,    # fp16*  output (Htotal, N, N)
    N: tl.int64,               # sequence length
    stride_head: tl.int64,     # stride between heads   (= N * N)
    stride_i: tl.int64,        # stride between rows   (= N)
    stride_k: tl.int64,        # stride between cols   (= 1)
    BLOCK_M: tl.constexpr,     # tile size in M dimension
    BLOCK_N: tl.constexpr,     # tile size in N dimension
    BLOCK_K: tl.constexpr,     # tile size in reduction K
):
    """
    Compute, for every head h:
        out[h, i, j] = Σ_k left[h, i, k] * right[h, j, k]
    The tensors are stored in row‑major order (i, k) / (j, k).
    """
    pid_head = tl.program_id(0)          # which head
    pid_m = tl.program_id(1)             # tile row index (i)
    pid_n = tl.program_id(2)             # tile col index (j)

    # Offsets for the current output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointer for this head
    base = pid_head * stride_head

    # Accumulator in FP32 for better precision on Tensor‑Cores
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left matrix tile: (BLOCK_M, BLOCK_K) ----
        a_ptrs = left_ptr + base \
                 + offs_i[:, None] * stride_i \
                 + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ---- right matrix transposed tile: (BLOCK_K, BLOCK_N) ----
        # right is stored as (j, k); we need (k, j) for Bᵀ.
        b_ptrs = right_ptr + base \
                 + offs_k[:, None] * stride_k \
                 + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product – TL.dot returns FP32 when inputs are FP16
        acc += tl.dot(a, b)

    # Store the result tile (cast back to FP16)
    c_ptrs = out_ptr + base \
             + offs_i[:, None] * stride_i \
             + offs_j[None, :] * stride_k
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        - mask         : torch.Tensor  [B, N, N]        (bool/float)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict containing:
              * dim          – input channel dimension
              * hidden_dim   – hidden dimension (proj/gate size)
              * nomask (bool, optional) – if True, no mask is applied

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the bulk of the work runs in half precision
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)    # [B, N, N, H]
    right = F.linear(x_h, w_right_proj)   # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gate logits → sigmoid (still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Optional pair mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates (in‑place)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536                       # ~64 k elements per kernel launch
    grid_gates = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid_gates](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Gates no longer needed → free memory
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ reduction via a single batched GEMM Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    total_heads = B * H

    # reshape to (total_heads, N, N) – contiguous layout (head, i, k)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)   # fp16

    # strides for the contiguous (head, N, N) layout
    stride_head = left_mat.stride(0)   # N * N
    stride_i = left_mat.stride(1)      # N
    stride_k = left_mat.stride(2)      # 1

    # Tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    n_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    n_tiles_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (total_heads, n_tiles_m, n_tiles_n)

    trimul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_head,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # free the now‑unused tensors
    del left, right, left_mat, right_mat

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 🔟 Multiply by the output gate (in‑place) via Triton
    # ------------------------------------------------------------------
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate   # free

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to the original `dim`
    # ------------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim]  fp16

    # Return as float32 (exact type expected by downstream code)
    return final.to(torch.float32)
