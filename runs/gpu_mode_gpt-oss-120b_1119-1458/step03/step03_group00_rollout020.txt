def apply_proj_gate_mask_kernel(
    combined_ptr, mask_ptr, left_ptr, right_ptr, out_gate_ptr,
    hidden: tl.constexpr, total_hidden: tl.constexpr, num_elems: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < num_elems

    # Load mask if needed
    if HAS_MASK:
        mask_vals = tl.load(mask_ptr + offs, mask=mask, other=0.0)  # shape (BLOCK_SIZE)
    else:
        mask_vals = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # Compute base offset for each row in combined
    base_offset = offs * total_hidden

    # load left_proj
    left_proj = tl.load(combined_ptr + base_offset[:,None] + tl.arange(0, hidden)[None,:], mask=mask[:,None], other=0.0)
    left_gate_raw = tl.load(combined_ptr + (base_offset + hidden)[:,None] + tl.arange(0, hidden)[None,:], mask=mask[:,None], other=0.0)
    right_proj = tl.load(combined_ptr + (base_offset + 2*hidden)[:,None] + tl.arange(0, hidden)[None,:], mask=mask[:,None], other=0.0)
    right_gate_raw = tl.load(combined_ptr + (base_offset + 3*hidden)[:,None] + tl.arange(0, hidden)[None,:], mask=mask[:,None], other=0.0)
    out_gate_raw = tl.load(combined_ptr + (base_offset + 4*hidden)[:,None] + tl.arange(0, hidden)[None,:], mask=mask[:,None], other=0.0)

    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

@triton.jit
def apply_proj_gate_mask_kernel(
    combined_ptr, mask_ptr, left_ptr, right_ptr, out_gate_ptr,
    hidden: tl.constexpr,
    total: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load mask values
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs, mask=mask, other=0.0)  # shape (BLOCK_SIZE,)
    else:
        mask_val = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # Compute per-element base pointer for combined
    # hidden5 = hidden * 5
    hidden5 = hidden * 5
    base = offs * hidden5

    # Create hidden-range arange for inner loops
    hid = tl.arange(0, hidden)

    # Load slices
    # left_proj
    left_proj = tl.load(combined_ptr + base[:, None] + hid[None, :], mask=mask[:, None], other=0.0)
    left_gate_raw = tl.load(combined_ptr + (base + hidden)[:, None] + hid[None, :], mask=mask[:, None], other=0.0)
    right_proj = tl.load(combined_ptr + (base + 2*hidden)[:, None] + hid[None, :], mask=mask[:, None], other=0.0)
    right_gate_raw = tl.load(combined_ptr + (base + 3*hidden)[:, None] + hid[None, :], mask=mask[:, None], other=0.0)
    out_gate_raw = tl.load(combined_ptr + (base + 4*hidden)[:, None] + hid[None, :], mask=mask[:, None], other=0.0)

    # Sigmoid
    one = tl.full((1,), 1.0, dtype=tl.float16)
    left_gate = one / (one + tl.exp(-left_gate_raw))
    right_gate = one / (one + tl.exp(-right_gate_raw))
    out_gate = one / (one + tl.exp(-out_gate_raw))

    # Apply mask broadcast

    left = left_proj * left_gate * mask_broadcast
    right = right_proj * right_gate * mask_broadcast

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The original PyTorch implementation expands the input tensor four times:
* 5 independent linear projections (left/right and their gates + out_gate)
* 2 elementwise mask‑gate fusions
* 1 layer‑norm + gating
* 2 matrix‑multiplications (the N³ contraction and the final projection)

The heavy memory traffic comes from repeatedly reading the large tensor
`[B,N,N,dim]` for every linear projection and from storing many intermediate
tensors (`left_proj`, `left_gate`, …).  The kernels below reduce this traffic
by:

1️⃣ **Combined linear projection** – all five weight matrices are concatenated
   into a single `(5·hidden, dim)` matrix and multiplied once with the
   normalized input.  This reads the input only once.

2️⃣ **Fused projection‑gate‑mask kernel** – from the concatenated result we
   compute `left`, `right` and `out_gate` in a single Triton kernel,
   applying the sigmoid gate and the (optional) mask without an extra
   memory copy.

3️⃣ **Core contraction** – unchanged, performed with a batched GEMM
   (`torch.matmul`) which provides the best possible performance on H100.

4️⃣ **Fused layer‑norm‑gate kernel** – normalises the contracted tensor,
   applies the learned output‑norm weight/bias and multiplies by `out_gate`
   in a single pass.

5️⃣ **Final projection** – a standard FP16 GEMM (`F.linear`) from the
   normalised, gated tensor to the original `dim`.

All arithmetic after the first layer‑norm is kept in FP16; the final result
is cast back to the input dtype (usually `float32`).  The two custom Triton
kernels are the only non‑trivial operations, satisfying the problem
requirement while dramatically reducing memory traffic and runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: combine the 5 linear projections, apply sigmoid gates and mask
# ----------------------------------------------------------------------
@triton.jit
def apply_proj_gate_mask_kernel(
    combined_ptr,       # [M, 5*H]   – result of the single large linear layer
    mask_ptr,           # [M]        – broadcast mask (float16)
    left_ptr,           # [M, H]     – output left tensor
    right_ptr,          # [M, H]     – output right tensor
    out_gate_ptr,       # [M, H]     – output out_gate tensor
    H: tl.constexpr,    # hidden dimension
    M: tl.constexpr,    # total number of (b,i,j) entries = B*N*N
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_M
    offs = start + tl.arange(0, BLOCK_M)
    valid = offs < M

    # pointer stride for the concatenated matrix (contiguous)
    stride_hidden5 = 5 * H

    # --------------------------------------------------------------
    # Load the 5 slices (proj, gate, proj, gate, gate) for each entry
    # --------------------------------------------------------------
    hid = tl.arange(0, H)

    base = offs * stride_hidden5

    left_proj = tl.load(combined_ptr + base[:, None] + hid[None, :],
                        mask=valid[:, None], other=0.0)
    left_gate_raw = tl.load(combined_ptr + (base + H)[:, None] + hid[None, :],
                            mask=valid[:, None], other=0.0)
    right_proj = tl.load(combined_ptr + (base + 2 * H)[:, None] + hid[None, :],
                         mask=valid[:, None], other=0.0)
    right_gate_raw = tl.load(combined_ptr + (base + 3 * H)[:, None] + hid[None, :],
                             mask=valid[:, None], other=0.0)
    out_gate_raw = tl.load(combined_ptr + (base + 4 * H)[:, None] + hid[None, :],
                           mask=valid[:, None], other=0.0)

    # --------------------------------------------------------------
    # Sigmoid ( 1 / (1 + exp(-x)) ) in FP16
    # --------------------------------------------------------------
    one = tl.full([1], 1.0, dtype=tl.float16)
    left_gate  = one / (one + tl.exp(-left_gate_raw))
    right_gate = one / (one + tl.exp(-right_gate_raw))
    out_gate   = one / (one + tl.exp(-out_gate_raw))

    # --------------------------------------------------------------
    # Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs, mask=valid, other=0.0)  # (BLOCK_M,)
        mask_vec = mask_val[:, None]                                 # (BLOCK_M,1)
        left  = left_proj  * left_gate  * mask_vec
        right = right_proj * right_gate * mask_vec
    else:
        left  = left_proj  * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # Store results
    # --------------------------------------------------------------
    tl.store(left_ptr     + offs[:, None] * H + hid[None, :],
             left,  mask=valid[:, None])
    tl.store(right_ptr    + offs[:, None] * H + hid[None, :],
             right, mask=valid[:, None])
    tl.store(out_gate_ptr + offs[:, None] * H + hid[None, :],
             out_gate, mask=valid[:, None])


# ----------------------------------------------------------------------
# Kernel 2: layer‑norm + out‑gate (single pass)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_outgate_kernel(
    inp_ptr,            # [M, H]   – contracted tensor (out after matmul)
    gate_ptr,           # [M, H]   – out_gate from kernel 1
    out_ptr,            # [M, H]   – normalised & gated tensor
    norm_w_ptr,         # [H]      – LayerNorm weight
    norm_b_ptr,         # [H]      – LayerNorm bias
    H: tl.constexpr,    # hidden dimension
    EPS: tl.constexpr,  # epsilon (float16 constant)
    M: tl.constexpr,    # total number of (b,i,j) entries
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_M
    offs = start + tl.arange(0, BLOCK_M)
    valid = offs < M

    hid = tl.arange(0, H)
    base = offs * H                         # (BLOCK_M,)

    # --------------------------------------------------------------
    # Load data and gate
    # --------------------------------------------------------------
    x = tl.load(inp_ptr  + base[:, None] + hid[None, :],
                mask=valid[:, None], other=0.0)   # (BLOCK_M, H)
    g = tl.load(gate_ptr + base[:, None] + hid[None, :],
                mask=valid[:, None], other=0.0)

    # --------------------------------------------------------------
    # Compute mean and variance (per row)
    # --------------------------------------------------------------
    sum_x = tl.sum(x, axis=1)                     # (BLOCK_M,)
    mean  = sum_x * (1.0 / H)

    diff = x - mean[:, None]                      # (BLOCK_M, H)
    var = tl.sum(diff * diff, axis=1) * (1.0 / H) # (BLOCK_M,)

    inv_std = 1.0 / tl.sqrt(var + EPS)            # (BLOCK_M,)

    # --------------------------------------------------------------
    # Apply learned LayerNorm weight/bias
    # --------------------------------------------------------------
    w = tl.load(norm_w_ptr + hid)                 # (H,)
    b = tl.load(norm_b_ptr + hid)                 # (H,)

    normed = diff * inv_std[:, None] * w[None, :] + b[None, :]

    # --------------------------------------------------------------
    # Multiply by out‑gate
    # --------------------------------------------------------------
    out = normed * g

    # --------------------------------------------------------------
    # Store
    # --------------------------------------------------------------
    tl.store(out_ptr + base[:, None] + hid[None, :],
             out, mask=valid[:, None])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  shape [B, N, N] (bool/float, CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains "dim", "hidden_dim",
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 -> float16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)                      # work in FP16 from now on

    # --------------------------------------------------------------
    # 2) Single large linear layer that produces all 5·hidden outputs
    # --------------------------------------------------------------
    # Concatenate the 5 weight matrices (shape: (5*hidden, dim))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    M = B * N * N                               # number of (b,i,j) entries
    x_flat = x.view(M, dim)                     # (M, dim)
    combined = F.linear(x_flat, combined_weight)   # (M, 5*hidden)
    combined = combined.view(B, N, N, 5 * hidden).contiguous()

    # --------------------------------------------------------------
    # 3) Fuse projection, gate and mask into left/right/out_gate tensors
    # --------------------------------------------------------------
    if nomask:
        # a tensor of ones to keep the kernel simple (the mask branch will be disabled)
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=inp.device)
        has_mask = 0
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()
        has_mask = 1

    left = torch.empty((B, N, N, hidden), dtype=torch.float16, device=inp.device)
    right = torch.empty((B, N, N, hidden), dtype=torch.float16, device=inp.device)
    out_gate = torch.empty((B, N, N, hidden), dtype=torch.float16, device=inp.device)

    total = B * N * N
    BLOCK = 256                                    # fits comfortably in one thread block

    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_M"]),)

    apply_proj_gate_mask_kernel[grid](
        combined,
        mask_fp16,
        left,
        right,
        out_gate,
        H=hidden,
        M=total,
        HAS_MASK=has_mask,
        BLOCK_M=BLOCK,
    )

    # --------------------------------------------------------------
    # 4) Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # rearrange for batched GEMM (each hidden channel is a separate batch)
    left_perm = left.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1) # (B, hidden, k, j)

    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)
    out_hidden = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B, i, j, hidden)

    # --------------------------------------------------------------
    # 5) LayerNorm + out_gate (single Triton kernel)
    # --------------------------------------------------------------
    norm_w = weights["to_out_norm.weight"].to(torch.float16)
    norm_b = weights["to_out_norm.bias"].to(torch.float16)

    normed = torch.empty_like(out_hidden)  # (B, N, N, hidden)

    total = B * N * N
    eps_f16 = torch.tensor(eps, dtype=torch.float16).item()

    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_M"]),)

    layernorm_outgate_kernel[grid](
        out_hidden,
        out_gate,
        normed,
        norm_w,
        norm_b,
        H=hidden,
        EPS=eps_f16,
        M=total,
        BLOCK_M=BLOCK,
    )

    # --------------------------------------------------------------
    # 6) Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    norm_flat = normed.view(M, hidden)                         # (M, hidden)
    out_final = F.linear(norm_flat, to_out_weight)             # (M, dim)
    out_final = out_final.view(B, N, N, dim).contiguous()

    # --------------------------------------------------------------
    # Cast back to the original dtype (usually float32) and return
    # --------------------------------------------------------------
    return out_final.to(inp.dtype)
