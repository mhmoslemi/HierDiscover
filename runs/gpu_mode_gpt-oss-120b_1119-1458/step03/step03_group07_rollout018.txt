def trimul_outgoing_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute BH (batch_hidden) => (b,d)
    num_hidden = H # pass hidden_dim to kernel
    b = batch_id // H
    d = batch_id % H

    # base pointers for this batch_hidden
    # left offset base + d (since d is innermost)
    base_l = left_ptr + b * stride_l_b + d * stride_l_h
    base_r = right_ptr + b * stride_r_b + d * stride_r_h
    base_o = out_ptr + b * stride_o_b + d * stride_o_h

    # compute i offsets and j offsets 
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_k):
        # k base offset
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers for left tile [M,K]
        # left offset: base_l + i*stride_i + k*stride_k
        a_ptrs = base_l + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        # Compute pointers for right tile [K,N] (but we need B_T[range k rows, range j cols])
        # B_T[k, j] = right[b, j, k, d]
        b_ptrs = base_r + offs_k[:, None] * stride_r_i + offs_n[None, :] * stride_r_k
        # Load tiles (if needed maybe we need to broadcast d not change)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)
        # Compute dot product across K dimension
        # Use dot product across K dimension on each pair of rows? For each M,N, we compute sum over k: left[i,k] * right[j,k]
        # Since we loaded blocks of shape (BLOCK_M,BLOCK_K) and (BLOCK_K,BLOCK_N), we can compute product using `tl.dot`.
        # We'll compute partial accumulator using matmul: acc += tl.dot(a, b)
        acc += tl.dot(a, b)

    # after loop, write result to output
    # store as fp16 (or fp32? cast)
    out = acc.to(tl.float16)  # cast to fp16

    # store
    out_ptrs = base_o + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_triton_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)   # batch * hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    hidden_dim = H

    b = bh // hidden_dim
    d = bh % hidden_dim

    # base offset for this (b,d)
    base_l = b * stride_l_b + d * stride_l_h
    base_r = b * stride_r_b + d * stride_r_h
    base_o = b * stride_o_b + d * stride_o_h

    # compute block start indices
    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    # coordinates
    offs_i = row_start + tl.arange(0, BLOCK_M)
    offs_j = col_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute number of K blocks
    num_k = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for A (left) tile shape [BLOCK_M, BLOCK_K]
        # left[b,i,k,d]

"""
TriMul (outgoing) – fast forward implementation for AlphaFold‑3.
================================================================

We implement the forward pass of the “outgoing” TriMul module, fusing as many
operations as possible into a few high‑throughput kernels:

1️⃣ Layer‑norm the input (float32 → fp16) – the most expensive pass over the
   [B, N, N, C] tensor.

2️⃣ Fuse the five linear layers (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a **single** GEMM:
   ```
   combined = x_norm @ [left_proj; left_gate; right_proj; right_gate; out_gate]
   ```
   This reads the normalized tensor only once (≈ 0.8 GB for N=1024, C=384).

3️⃣ Split the combined output, optionally apply the pair‑wise mask, and
   apply the sigmoid gates.

4️⃣ Compute the heavy O(N³) contraction
   `out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]`
   with a custom Triton kernel that treats the hidden dimension as the batch
   axis, thus **avoiding** the expensive `permute(...).contiguous()` copies
   required by cuBLAS.

5️⃣ Second LayerNorm (fp16), element‑wise multiplication with the output‑gate
   (tiny Triton kernel), and final linear projection back to the original
   channel dimension.

All heavy arithmetic runs in fp16 (Tensor‑Core friendly).  The final tensor
is cast back to float32, as required by the API.

The implementation respects the `nomask` flag, works for both masked and
unmasked inputs, and supports all shapes listed in the test matrix.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# 1️⃣ Element‑wise multiplication kernel (dst *= src)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# 2️⃣ Batched GEMM kernel for the O(N³) contraction.
#    out = left @ rightᵀ  where left, right : [B, N, N, H] (fp16)
#    The hidden dimension H is flattened into the batch axis (BH = B*H).
# ----------------------------------------------------------------------
@triton.jit
def _batched_triu_gemm_kernel(
    left_ptr,     # *fp16 – left tensor (src)
    right_ptr,    # *fp16 – right tensor (src)
    out_ptr,      # *fp16 – output tensor
    B, N, H,      # i32 – problem dimensions
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (identical layout)
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out (identical layout)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # each program works on one (b, h) pair → batch index = bh
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # recover original batch and hidden index
    b = bh // H
    h = bh % H

    # base pointers for this (b, h)
    base_l = b * stride_l_b + h * stride_l_h
    base_r = b * stride_r_b + h * stride_r_h
    base_o = b * stride_o_b + h * stride_o_h

    # tile offsets in the i‑ and j‑dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  A[i,k]  (shape BLOCK_M × BLOCK_K)
        a_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # rightᵀ tile Bᵀ[k,j] = right[b, j, k, h]
        # note the swap of the second and third axes
        b_ptrs = base_r + offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b)

    # write the result back (cast to fp16)
    out_ptrs = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[:, None])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out = left @ rightᵀ  with a Triton kernel.
    left / right : [B, N, N, H] (fp16)
    Returns out of the same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for left/right/out (they are identical)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – tuned for H100 fp16 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_triu_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps → 128 threads per block
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32, already on device)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``.
    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # float32
    x = x.to(torch.float16)  # work in fp16 from now on

    # -------------------------------------------------
    # 2️⃣ Fuse the five linear layers (left/right proj & gates, out_gate)
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, dim)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, dim)

    # concatenate along the output‑feature axis
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0  # (5*H, dim)
    )
    # single GEMM: x [B,N,N,dim] @ fused_weightᵀ → [B,N,N,5*H]
    combined = torch.nn.functional.linear(x, fused_weight)  # fp16

    H = hidden_dim
    left_proj = combined[..., 0:H]
    left_gate = combined[..., H:2*H]
    right_proj = combined[..., 2*H:3*H]
    right_gate = combined[..., 3*H:4*H]
    out_gate = combined[..., 4*H:5*H]

    # -------------------------------------------------
    # 3️⃣ Optional mask (applied after projection, before gating)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left  = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate # [B,N,N,H]

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via custom Triton batched GEMM
    # -------------------------------------------------
    out = _batched_matmul_triton(left, right)  # [B,N,N,H] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 7️⃣ Out‑gate (sigmoid) and fused element‑wise multiplication
    # -------------------------------------------------
    out_gate = torch.sigmoid(out_gate)  # fp16

    # Flatten for the tiny Triton kernel (in‑place dst *= src)
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)
    total = out_flat.numel()
    BLOCK_SIZE = 1024
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total,
        1, 1,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # -------------------------------------------------
    # 8️⃣ Final linear projection back to ``dim``
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (API contract)
    # -------------------------------------------------
    return out.to(torch.float32)
