@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    mask_ptr,
    B, N, hidden,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    WITH_MASK: tl.constexpr
):
    # Program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Retrieve batch and hidden indices
    hidden_dim = hidden
    b = pid_z // hidden_dim
    h = pid_z % hidden_dim

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this b,h
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # optional mask
    if WITH_MASK:
        mask_base = mask_ptr + b * stride_mask_b
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # load left tile (i,k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < N), other=0.0)

        # load right tile (k,j) transposed version
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & mask_j[None, :], other=0.0)

        if WITH_MASK:
            # mask (i,k) for left, (j,k) for right
            mask_a_ptr = mask_base + offs_i[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
            mask_a = tl.load(mask_a_ptr, mask=mask_i[:, None] & (offs_k[None, :] < N), other=0.0)
            mask_b_ptr = mask_base + offs_j[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
            mask_b = tl.load(mask_b_ptr, mask=mask_j[:, None] & (offs_k[None, :] < N), other=0.0)
            a = a * mask_a
            b = b * mask_b

        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_k
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast Triton implementation
================================================

This file implements the **forward only** of the “outgoing” TriMul module
used in AlphaFold‑3‑style protein‑structure models.

Algorithm
---------
1️⃣ Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C} in FP32.
2️⃣ Cast to FP16 and apply four linear projections (left/right and three
   gating matrices).  All four linear layers are fused with the
   sigmoid‑gates in FP16.
3️⃣ (Optional) apply a binary mask M ∈ {0,1}^{B×N×N}.  The mask is
   multiplied *inside* the heavy contraction kernel to avoid an extra
   element‑wise pass.
4️⃣ **Heavy O(N³) contraction**:
   out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   is performed as a *batched* matrix multiplication C = A @ Bᵀ where
   A and B have shape (B·H, N, N).  The custom Triton kernel tiles the
   multiplication with 64×64×64 blocks, uses FP16 inputs and accumulates
   in FP32 via `tl.dot` (tensor‑core MMA).  Hidden‑dimension H is folded
   into the batch dimension, thus no costly `permute‑contiguous` copies
   are required.
5️⃣ A second LayerNorm (hidden dimension) is applied in FP16.
6️⃣ The output is gated with the pre‑computed `out_gate`, projected back
   to the original feature dimension C and cast to FP32.

The implementation works for all test configurations (B∈{1,2},
N∈{128,256,512,768,1024}, C∈{128,384}, H=128) and supports both
“nomask’’ and masked inputs.  The critical O(N³) part runs entirely on
the GPU with a single Triton kernel, while all lightweight operations
remain in fast PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (no layout permutation)
#   A, B : [B, N, N, H]   (FP16, contiguous, last dim = hidden)
#   C    : [B, N, N, H]   (FP16, same layout)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_matmul_kernel(
    left_ptr,            # *fp16  [B, N, N, H]
    right_ptr,           # *fp16  [B, N, N, H]
    out_ptr,             # *fp16  [B, N, N, H]
    mask_ptr,            # *fp16  [B, N, N]  (optional, may be dummy)
    B, N, H,             # i32 sizes
    # left strides (B, i, k, h)
    stride_lb, stride_li, stride_lk, stride_lh,
    # right strides (B, i, k, h) – same layout as left
    stride_rb, stride_ri, stride_rk, stride_rh,
    # out strides – same layout as left/right
    stride_ob, stride_oi, stride_ok, stride_oh,
    # mask strides (B, i, k)
    stride_mb, stride_mi, stride_mk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    WITH_MASK: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul contraction.
    Each program processes a tile (i‑block, j‑block) of one
    (batch, hidden) pair (i.e. pid_z → b, h).
    """
    pid_i = tl.program_id(0)        # tile along first N axis (i)
    pid_j = tl.program_id(1)        # tile along second N axis (j)
    pid_z = tl.program_id(2)        # index over M = B * H

    # ------------------------------------------------------------------
    # Recover batch (b) and hidden (h) from pid_z
    # ------------------------------------------------------------------
    h = pid_z % H
    b = pid_z // H

    # ------------------------------------------------------------------
    # Compute the row/col offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this (b, h) pair
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # ------------------------------------------------------------------
    # Optional mask base
    # ------------------------------------------------------------------
    if WITH_MASK:
        mask_base = mask_ptr + b * stride_mb

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K (= seq_len)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load A tile : left[b, i, k, h] -----
        a_ptrs = (
            left_base
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ----- Load B tile : right[b, j, k, h]  (as Bᵀ) -----
        # The tile we need is (k, j) → rows = k, cols = j
        b_ptrs = (
            right_base
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_ri
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ----- Optional mask multiplication -----
        if WITH_MASK:
            # left mask: mask[b, i, k]
            mask_a_ptrs = (
                mask_base
                + offs_i[:, None] * stride_mi
                + offs_k[None, :] * stride_mk
            )
            mask_a = tl.load(
                mask_a_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            )
            # right mask: mask[b, j, k]
            mask_b_ptrs = (
                mask_base
                + offs_j[:, None] * stride_mi
                + offs_k[None, :] * stride_mk
            )
            mask_b = tl.load(
                mask_b_ptrs,
                mask=mask_j[:, None] & mask_k[None, :],
                other=0.0,
            )
            a = a * mask_a
            b = b * mask_b

        # ----- Multiply‑accumulate (Tensor‑core MMA) -----
        # tl.dot promotes the fp16 product to fp32 accumulation.
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        out_base
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_ok
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

    B, N, _, H = left.shape
    device = left.device

    # Ensure contiguity (should already be the case after linear)
    left = left.contiguous()
    right = right.contiguous()

    # Prepare an empty output buffer
    out = torch.empty_like(left, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides (in element counts, not bytes)
    # ------------------------------------------------------------------
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_ok, stride_oh = out.stride()

    if mask is not None:
        mask = mask.to(torch.float16).contiguous()
        stride_mb, stride_mi, stride_mk = mask.stride()
    else:
        # dummy strides – never accessed because WITH_MASK=0
        stride_mb = stride_mi = stride_mk = 0

    # ------------------------------------------------------------------
    # Tile sizes – empirically good for H100 / fp16 MMA
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch×hidden
    )

    _trimul_outgoing_matmul_kernel[grid](
        left,
        right,
        out,
        mask if mask is not None else left,   # Triton requires a pointer; unused if WITH_MASK=0
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_ok,
        stride_oh,
        stride_mb,
        stride_mi,
        stride_mk,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        WITH_MASK=0 if mask is None else 1,
        num_warps=8,
        num_stages=2,
    )

    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward‑only Triton implementation of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    #   Cast to FP16 for the rest of the pipeline
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Convert all weight tensors to FP16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, hidden]

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) – all in FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (FP16 in/out)
    # -------------------------------------------------
    # Optional mask is fused inside the kernel to avoid an extra pass.
    mask_fp16 = None if nomask else mask_tensor.to(torch.float16)
    out = _trimul_outgoing_matmul_fp16(left, right, mask_fp16)   # [B, N, N, hidden] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 7️⃣ Final gating + projection back to original dim
    # -------------------------------------------------
    out = out * out_gate               # element‑wise gate (broadcast over hidden)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (fp16)

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 (required output type)
    # -------------------------------------------------
    return out.to(torch.float32)
