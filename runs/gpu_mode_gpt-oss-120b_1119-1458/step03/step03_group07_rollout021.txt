def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # weights
    norm_weight_fp16 = weights["norm.weight"].to(torch.float16)
    norm_bias_fp16 = weights["norm.bias"].to(torch.float16)
    # combined projection+gate weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    # combine
    combined_proj_gate_w = torch.cat([left_proj_w,
                                      left_gate_w,
                                      right_proj_w,
                                      right_gate_w], dim=0)  # shape [4*hidden_dim, dim]
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 1. first layer norm (fp32->fp16)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
    )  # fp32 output 
    x = x_norm.to(torch.float16)

    # 2. combined linear to get all proj+gate values
    comb = torch.nn.functional.linear(x, combined_proj_gate_w) # shape [B,N,N,4*H] fp16

    # 3. fused projection+gate (and mask)
    B, N, _, _ = input_tensor.shape
    total_positions = B * N * N
    total_elements = total_positions * hidden_dim

    left_out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right_out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    comb_flat = comb.view(-1)  # flatten to 1D
    left_out_flat = left_out.view(-1)
    right_out_flat = right_out.view(-1)

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # [total_positions]
    else:
        # dummy mask pointer: can use left_out's pointer, will not be read if MASKED=0

        comb_flat,
        mask_flat,
        left_out_flat,
        right_out_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        MASKED=0 if nomask else 1,
        num_warps=8,
    )

    out = _batched_matmul_transpose(left_out, right_out)

        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    comb_ptr, mask_ptr, left_out_ptr, right_out_ptr,
    total_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    MASKED: tl.constexpr

    gate_left_ptr = comb_ptr + pos * stride_comb + hidden_dim + hid

    gate_left = tl.load(gate_left_ptr, mask=mask, other=0.0)

"""
TriMul (outgoing) – fast forward implementation for AlphaFold‑3.

Key ideas
---------
1️⃣ Fuse the four linear layers (proj + gate for left & right) into a single
   matrix multiply by stacking their weight matrices.
2️⃣ Fuse the projection, optional mask, sigmoid‑gate and final left/right
   tensor creation into one Triton kernel.  This halves the work of the
   four original Linear calls and removes three separate element‑wise passes.
3️⃣ Perform the heavy O(N³) contraction with a single batched GEMM
   (torch.bmm), which on H100 maps to a Tensor‑Core kernel.
4️⃣ Apply the second LayerNorm in FP16, then fuse the out‑gate multiplication
   into an element‑wise Triton kernel.
5️⃣ Finish with the output Linear (hidden → dim) and cast back to FP32.

The only explicit Triton kernels are:
   • `_proj_gate_fused_kernel` – creates the left/right tensors from the
       combined projection‑gate output, handling mask and sigmoid.
   • `_elemwise_mul_kernel` – out *= gate (tiny element‑wise kernel).

All heavy linear algebra (GEMM, batch‑matmul, final Linear) uses PyTorch’s
highly‑optimized cuBLAS/Tensor‑Core implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1: fused projection + gate (+ optional mask) for left & right.
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_fused_kernel(
    comb_ptr,          # *fp16 – packed [projL, gateL, projR, gateR] per position
    mask_ptr,          # *fp16 – mask per position (only used when MASKED==1)
    left_out_ptr,      # *fp16 – resulting left tensor  (B,N,N,H)
    right_out_ptr,     # *fp16 – resulting right tensor (B,N,N,H)
    total_elements,    # i32   – total number of scalar elements (=B*N*N*H)
    hidden_dim: tl.constexpr,   # i32 – H
    BLOCK_SIZE: tl.constexpr,   # i32 – threads per block
    MASKED: tl.constexpr,        # i32 – 0: no mask, 1: apply mask
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # 1‑D index → (position, hidden_index)
    pos = offs // hidden_dim                # position = b*nn + i*nn + j
    hid = offs % hidden_dim                 # hidden index 0 … H‑1

    # each position stores 4*H consecutive values in `comb`
    stride_comb = hidden_dim * 4            # distance between successive positions

    # offsets for the four components
    projL_ptr = comb_ptr + pos * stride_comb + hid
    gateL_ptr = comb_ptr + pos * stride_comb + hidden_dim + hid
    projR_ptr = comb_ptr + pos * stride_comb + 2 * hidden_dim + hid
    gateR_ptr = comb_ptr + pos * stride_comb + 3 * hidden_dim + hid

    # load the raw values
    projL = tl.load(projL_ptr, mask=mask, other=tl.float16(0.0))
    gateL = tl.load(gateL_ptr, mask=mask, other=tl.float16(0.0))
    projR = tl.load(projR_ptr, mask=mask, other=tl.float16(0.0))
    gateR = tl.load(gateR_ptr, mask=mask, other=tl.float16(0.0))

    # sigmoid gate = 1/(1+exp(-x))
    gateL = tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-gateL))
    gateR = tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-gateR))

    if MASKED:
        # mask is a single fp16 value per position (0 or 1)
        m = tl.load(mask_ptr + pos, mask=mask, other=tl.float16(1.0))
        projL = projL * m
        projR = projR * m

    left_val  = projL * gateL
    right_val = projR * gateR

    # store the results (contiguous layout: stride = H)
    left_out_ptr  = left_out_ptr  + pos * hidden_dim + hid
    right_out_ptr = right_out_ptr + pos * hidden_dim + hid
    tl.store(left_out_ptr,  left_val,  mask=mask)
    tl.store(right_out_ptr, right_val, mask=mask)


# ----------------------------------------------------------------------
# Tiny element‑wise kernel: dst *= src  (both FP16, 1‑D flattened view)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Batched matrix‑multiply for the O(N³) contraction.
# Computes C = A @ Bᵀ  where A,B: [B*H, N, N]  →  out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(
    left: torch.Tensor,   # [B, N, N, H]   fp16
    right: torch.Tensor   # [B, N, N, H]   fp16
) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – a strided batched GEMM
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights_dict : name → torch.Tensor (float32, same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # Cast weights to FP16 once (except layer‑norm which we keep FP32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    # projection + gate weights (stacked later)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, dim)

    # out‑gate and final output weights
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)          # (H, dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)      # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)        # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)           # (dim, H)

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    # Using FP16 weight/bias here is fine for inference; the input is float32.
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    x = x_norm.to(torch.float16)          # work in FP16 from now on

    # -------------------------------------------------
    # 2️⃣ One fused Linear that produces all proj+gate values
    # -------------------------------------------------
    # stack the four (proj,gates) weight matrices -> shape [4*H, dim]
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*H, dim)

    # Result: [B, N, N, 4*H]  FP16
    comb = torch.nn.functional.linear(x, combined_weight)

    # -------------------------------------------------
    # 3️⃣ Fuse projection, mask and sigmoid‑gate for left/right
    # -------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_positions = B * N * N
    total_elements = total_positions * hidden_dim

    left_out  = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right_out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    comb_flat = comb.view(-1)               # 1‑D view, length = total_positions*4*H
    left_flat = left_out.view(-1)
    right_flat = right_out.view(-1)

    if not nomask:
        # mask is broadcast over the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)      # length = total_positions
    else:
        # dummy pointer (won’t be read because MASKED==0)
        mask_flat = left_flat

    BLOCK_SIZE = 1024
    grid = ((total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    _proj_gate_fused_kernel[grid](
        comb_flat,
        mask_flat,
        left_flat,
        right_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        MASKED=0 if nomask else 1,
        num_warps=8,
    )

    # -------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via a strided‑batched GEMM
    # -------------------------------------------------
    out = _batched_matmul_transpose(left_out, right_out)   # [B,N,N,H]  FP16

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 6️⃣ Out‑gate (sigmoid) and element‑wise multiplication (Triton)
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] FP16

    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)
    total_out_elems = out_flat.numel()
    grid2 = ((total_out_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    _elemwise_mul_kernel[grid2](
        out_flat,
        gate_flat,
        total_out_elems,
        1,                     # dst stride
        1,                     # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # -------------------------------------------------
    # 7️⃣ Final Linear projection back to `dim`
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim]  FP16

    # -------------------------------------------------
    # 🔚 Cast back to float32 (API expects float32)
    # -------------------------------------------------
    return out.to(torch.float32)
