"""
TriMul (outgoing) – fast forward kernel for AlphaFold‑3 (no grads).

Algorithm (all in FP16 except the first layer‑norm which is kept in FP32 for
numerical stability):
1️⃣ Layer‑norm the input [B,N,N,C] tensor (C = `dim`).
2️⃣ Fuse the two projection linear layers (`left_proj`/`left_gate`,
   `right_proj`/`right_gate`) by concatenating their weights → one GEMM per
   side.  This halves the number of matrix‑multiplications.
3️⃣ (Optional) Apply the pair‑wise mask *after* the projection but before
   the gating (exactly as in the PyTorch reference).
4️⃣ Apply the sigmoid gates and multiply: `left = left_proj * sigmoid(left_gate)`,
   `right = right_proj * sigmoid(right_gate)`.
5️⃣ Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   by reshaping the tensors to [B·H, N, N] and using a single
   `torch.bmm` (batched GEMM).  This maps to an optimal Tensor‑Core kernel on
   H100.
6️⃣ Second layer‑norm on the hidden dimension (`hidden_dim`).
7️⃣ Compute the output gate, apply `sigmoid`, and fuse the in‑place
   multiplication `out *= out_gate` with a tiny Triton kernel
   (`_elemwise_mul_kernel`).  This satisfies the “use a Triton kernel”
   requirement.
8️⃣ Final linear projection back to the original channel dimension (`dim`).
9️⃣ Cast the result back to float32 (the API expects float32).

The implementation is deliberately memory‑efficient: only the tensors that
are needed for the next step are kept, and all heavy compute is performed in
FP16.  The Triton kernel is 1‑D, processes `BLOCK_SIZE` elements per block,
and runs with 4 warps (the H100 can execute this kernel in a single warp
cycle for typical tensor sizes).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton element‑wise in‑place multiplication: dst *= src (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,               # *fp16 – destination (in‑place)
    src_ptr,               # *fp16 – source (gate)
    N,                     # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    # Load one element per thread
    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    # Store the multiplied result back in place
    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Batched matrix‑multiply for the O(N³) contraction.
# Computes C = A @ Bᵀ  where A,B: [B, N, N, H] (FP16) → out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(
    left: torch.Tensor,   # shape [B, N, N, H] in FP16
    right: torch.Tensor   # shape [B, N, N, H] in FP16
) -> torch.Tensor:
    """
    Reshape to [B·H, N, N] and perform a batched GEMM.
    """
    B, N, _, H = left.shape

    # [B, H, N, N] → contiguous → [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ   (torch.bmm uses fast Tensor‑Core kernels)
    C = torch.bmm(A, B_.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out  # FP16

    # -------------------------------------------------
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)

    # -------------------------------------------------

    left  = left_proj * left_gate          # (B,N,N,H)

    )   # (B,N,N,H) FP16

    # 8️⃣ Output gate (sigmoid) and in‑place multiplication via Triton
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # (B,N,N,H) FP16

    # Flatten tensors to 1‑D for the Triton kernel
    out_flat = out.contiguous().view(-1)        # ensures contiguous layout
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024                     # max threads per block on H100
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    # Triton launch – in‑place out *= out_gate
    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,          # dst stride (contiguous)
        1,          # src stride (contiguous)
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,  # 4 warps is a good default for FP16 loads/stores
    )

    # -------------------------------------------------
    # 9️⃣ Final projection back to `dim`
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # -------------------------------------------------
    # 🔚 Cast back to float32 (API contract)
    # -------------------------------------------------
    return out.to(torch.float32)
```
