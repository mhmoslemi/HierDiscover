@triton.jit
def _trimul_matmul_kernel_no_copy(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)  # batch*hidden index
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile

    # decode b and h
    b = pid_z // H
    h = pid_z % H

    # base pointers: offset to start of the specific batch and hidden slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Offsets within N dimension (i, j, k)
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile (left) [i,k]
        # left[i,k] pointer: left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile (right) [k,j] (transposed)
        # right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        # Because right is shape [B,N,N,H] (b, i, k, h); we need right at (b, j, k, h) i.e., i dimension for right is j we want to treat as column index. So use stride for i axis (original axis i) which is stride_right_i.
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)  # accumulate
    # Write out C tile
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (batched matmul)
#   A : [B, H, N, N] (as view [M, N, N] with M = B*H)
#   B : [B, H, N, N] (accessed transposed on the last two axes)
#   C : [B, H, N, N]
# Strides are given explicitly so we can keep the original layout
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # *fp16  [B, N, N, H]
    B_ptr,                     # *fp16  [B, N, N, H]
    C_ptr,                     # *fp16  [B, N, N, H]
    batch, hidden, N,          # i32 scalars
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,   # strides for A
    stride_b_b, stride_b_i, stride_b_k, stride_b_h,   # strides for B
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # 1️⃣  Program IDs and decoded batch/hidden indices
    # ---------------------------------------------------------------
    pid_m = tl.program_id(0)    # tile along i (output rows)
    pid_n = tl.program_id(1)    # tile along j (output cols)
    pid_z = tl.program_id(2)    # flattened batch*hidden index

    # decode batch (b) and hidden (h) from pid_z
    b = pid_z // hidden
    h = pid_z % hidden

    # base pointers for this (b, h) slice
    a_base = A_ptr + b * stride_a_b + h * stride_a_h
    b_base = B_ptr + b * stride_b_b + h * stride_b_h
    c_base = C_ptr + b * stride_c_b + h * stride_c_h

    # ---------------------------------------------------------------
    # 2️⃣  Tile offsets for i, j and k dimensions
    # ---------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑indices
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑indices
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------------------------------------------------
        #   Load A tile (shape BLOCK_M × BLOCK_K)  ← left[b,i,k,h]
        # -----------------------------------------------------------
        a_ptrs = (
            a_base
            + offs_i[:, None] * stride_a_i   # i‑stride
            + offs_k[None, :] * stride_a_k   # k‑stride
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16 → up‑casted to fp32 inside dot

        # -----------------------------------------------------------
        #   Load B tile (shape BLOCK_K × BLOCK_N)  ← right[b,j,k,h]ᵀ
        #   We read the transposed view by swapping the i‑ and k‑strides.
        # -----------------------------------------------------------
        b_ptrs = (
            b_base
            + offs_k[:, None] * stride_b_k   # k‑stride (rows)
            + offs_j[None, :] * stride_b_i   # j‑stride (cols) – note the swap
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # -----------------------------------------------------------
        #   FP16 × FP16 → FP32 accumulation via Tensor‑cores
        # -----------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N)

    # ---------------------------------------------------------------
    # 3️⃣  Write back the result (cast back to fp16)
    # ---------------------------------------------------------------
    c_ptrs = (
        c_base
        + offs_i[:, None] * stride_c_i   # i‑stride
        + offs_j[None, :] * stride_c_j   # j‑stride
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel on tensors in their native
# [B, N, N, H] layout (fp16).
# ----------------------------------------------------------------------
def _batched_matmul_outgoing_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for tensors of shape [B, N, N, H] (fp16).

    The function keeps the original layout to avoid an extra permute+copy.
    """
    B, N, _, H = left.shape
    # Output buffer
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides (in element counts, not bytes)
    stride_a_b, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_b, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_b, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tunable tile sizes – empirically good on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        stride_a_b,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_b,
        stride_b_i,
        stride_b_k,
        stride_b_h,
        stride_c_b,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,         # 8 warps give good occupancy on H100
        num_stages=3,        # double‑buffered pipeline
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   (float32)

    # ------------------------------------------------------------------
    # Cast to fp16 for the expensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast model weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Fuse gates (in‑place to lower memory traffic)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # O(N³) contraction via the Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_outgoing_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # Final gating + projection back to original dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
