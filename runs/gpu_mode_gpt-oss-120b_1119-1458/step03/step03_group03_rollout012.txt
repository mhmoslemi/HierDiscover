@triton.jit
def _proj_gate_kernel(
    X_ptr, proj_w_ptr, gate_w_ptr,
    out_ptr,
    M, K, H,
    stride_xm, stride_xk,
    stride_pw_h, stride_pw_k,
    stride_gw_h, stride_gw_k,
    stride_om, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Each program computes a tile of output (M rows, H cols)
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    # coordinate for row offset
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load X tile
        x_ptrs = X_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)  # fp16

        # load proj weight tile: shape (K, H)
        pw_ptrs = proj_w_ptr + offs_k[:, None] * stride_pw_k + offs_h[None, :] * stride_pw_h
        pw = tl.load(pw_ptrs,
                    mask=mask_k[:, None] & mask_h[None, :],
                    other=0.0)  # fp16

        # load gate weight tile: shape (K, H)
        gw_ptrs = gate_w_ptr + offs_k[:, None] * stride_gw_k + offs_h[None, :] * stride_gw_h
        gw = tl.load(gw_ptrs,
                    mask=mask_k[:, None] & mask_h[None, :],
                    other=0.0)  # fp16

        # dot products
        acc_proj += tl.dot(x, pw)  # (M,H)
        acc_gate += tl.dot(x, gw)

    # Apply sigmoid to gate and multiply
    gate_sig = 1.0 / (1.0 + tl.exp(-acc_gate))
    out = acc_proj * gate_sig

    # store
    out_ptrs = out_ptr + offs_m[:, None] * stride_om + offs_h[None, :] * stride_oh
    tl.store(out_ptrs,
             out.to(tl.float16),
             mask=mask_m[:, None] & mask_h[None, :])
```

        stride_pw_k_r,
        stride_gw_h_r,
        stride_gw_k_r,
        stride_om_r,
        stride_oh_r,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # Optional mask
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # 5. Compute out_gate (sigmoid)
    # out_gate = sigmoid(x @ out_gate_w.T)
    out_gate_pre = torch.nn.functional.linear(x_flat, out_gate_w)  # shape [M, hidden_dim]
    out_gate = torch.sigmoid(out_gate_pre).view(B, N, N, hidden_dim)

    # 6. Heavy O(N^3) batch matmul using torch.bmm
    # permute to [B, hidden, N, N] then flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*hidden, N, N]

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # 7. Second LayerNorm (over hidden)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Multiply by out_gate
    out = out * out_gate

    # 9. Final linear to original dim (fp16)
    out = torch.nn.functional.linear(out, to_out_w)  # shape [B,N,N,dim]

    # 10. Cast back to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32) and cast it to fp16.
2.  Fuse the two linear projections (proj & gate) for the “left” and “right”
    tensors into a single Triton kernel:
        left  = (x·W_left_proj) * sigmoid(x·W_left_gate)
        right = (x·W_right_proj) * sigmoid(x·W_right_gate)
    The kernel works on the flattened tensor X∈[M,dim] (M = B·N·N) and
    directly produces the gated tensors in fp16.
3.  (Optional) apply the pairwise mask (broadcast over the hidden dim).
4.  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a batched matrix multiply C = A @ Bᵀ using cuBLAS (`torch.bmm`),
    which is far faster on an H100 than a pure‑Triton implementation.
5.  Second Layer‑norm (fp16) on the hidden dimension.
6.  Compute the output‑gate = sigmoid(x·W_out_gate) with a regular PyTorch
    linear (the cost is negligible).
7.  Multiply the contracted tensor by the output‑gate.
8.  Final linear projection to the original channel dimension.
9.  Cast the result back to float32 and return.

Only the projection+gate step uses a custom Triton kernel; all other
operations rely on high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _proj_gate_kernel(
    X_ptr,            # [M, K]  (flattened input)
    proj_w_ptr,       # [H, K]  (projection weights)
    gate_w_ptr,       # [H, K]  (gate weights)
    OUT_ptr,          # [M, H]  (output after gating)
    M, K, H,          # sizes
    stride_xm, stride_xk,                     # strides for X
    stride_pw_h, stride_pw_k,                 # strides for proj weight
    stride_gw_h, stride_gw_k,                 # strides for gate weight
    stride_om, stride_oh,                     # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Fused projection + sigmoid gate.
    Computes:
        out[i, h] = (Σ_k X[i,k] * W_proj[h,k]) * σ( Σ_k X[i,k] * W_gate[h,k] )
    """
    pid_m = tl.program_id(0)   # tile along M (rows)
    pid_h = tl.program_id(1)   # tile along H (hidden)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # Accumulators (FP32)
    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load X[i, k] ----
        x_ptrs = X_ptr + (offs_m[:, None] * stride_xm) + (offs_k[None, :] * stride_xk)
        x = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ---- Load projection weight tile (K, H) ----
        pw_ptrs = (
            proj_w_ptr
            + (offs_k[:, None] * stride_pw_k)
            + (offs_h[None, :] * stride_pw_h)
        )
        pw = tl.load(
            pw_ptrs,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )  # fp16

        # ---- Load gate weight tile (K, H) ----
        gw_ptrs = (
            gate_w_ptr
            + (offs_k[:, None] * stride_gw_k)
            + (offs_h[None, :] * stride_gw_h)
        )
        gw = tl.load(
            gw_ptrs,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )  # fp16

        # ---- Dot products (fp16 * fp16 -> fp32) ----
        acc_proj += tl.dot(x, pw)   # [M_tile, H_tile]
        acc_gate += tl.dot(x, gw)   # [M_tile, H_tile]

    # Sigmoid on gate accumulator (FP32)
    gate_sig = 1.0 / (1.0 + tl.exp(-acc_gate))

    # Apply gate
    out = acc_proj * gate_sig  # still FP32

    # Store result in FP16
    out_ptrs = OUT_ptr + (offs_m[:, None] * stride_om) + (offs_h[None, :] * stride_oh)
    tl.store(
        out_ptrs,
        out.to(tl.float16),
        mask=mask_m[:, None] & mask_h[None, :],
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)  # cast to fp16 for the heavy work

    B, N, _, _ = x.shape
    M = B * N * N  # total number of (i,j) positions per batch

    # ------------------------------------------------------------------
    # 2️⃣ Prepare flattened input and weight tensors (fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(M, dim).contiguous()               # [M, dim]
    # Linear weights (all are stored as [out_features, in_features])
    left_proj_w  = weights["left_proj.weight"].to(torch.float16).contiguous()
    left_gate_w  = weights["left_gate.weight"].to(torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w   = weights["out_gate.weight"].to(torch.float16).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).contiguous()
    to_out_w      = weights["to_out.weight"].to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gating (Triton kernel)
    # ------------------------------------------------------------------
    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 128          # rows per block (M dimension)
    BLOCK_H = 64           # hidden columns per block
    BLOCK_K = 64           # reduction (feature) block

    # ---------- left ----------
    left = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    # strides (in element units)
    stride_xm, stride_xk = x_flat.stride()
    stride_lp_h, stride_lp_k = left_proj_w.stride()
    stride_lg_h, stride_lg_k = left_gate_w.stride()
    stride_om, stride_oh = left.stride()
    grid = ((M + BLOCK_M - 1) // BLOCK_M, (hidden_dim + BLOCK_H - 1) // BLOCK_H)
    _proj_gate_kernel[grid](
        x_flat,
        left_proj_w,
        left_gate_w,
        left,
        M,
        dim,
        hidden_dim,
        stride_xm,
        stride_xk,
        stride_lp_h,
        stride_lp_k,
        stride_lg_h,
        stride_lg_k,
        stride_om,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # ---------- right ----------
    right = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    stride_rp_h, stride_rp_k = right_proj_w.stride()
    stride_rg_h, stride_rg_k = right_gate_w.stride()
    stride_om_r, stride_oh_r = right.stride()
    _proj_gate_kernel[grid](
        x_flat,
        right_proj_w,
        right_gate_w,
        right,
        M,
        dim,
        hidden_dim,
        stride_xm,
        stride_xk,
        stride_rp_h,
        stride_rp_k,
        stride_rg_h,
        stride_rg_k,
        stride_om_r,
        stride_oh_r,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # Reshape back to [B, N, N, hidden]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Apply optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Compute out‑gate (sigmoid) using regular PyTorch linear (cheap)
    # ------------------------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(x_flat, out_gate_w)      # [M, hidden_dim]
    out_gate = torch.sigmoid(out_gate_pre).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # reshape to [B, hidden, N, N] → flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    left_mat = left_perm.view(B * hidden_dim, N, N)      # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]

    # Batched matrix multiply: C = left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Restore to [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
