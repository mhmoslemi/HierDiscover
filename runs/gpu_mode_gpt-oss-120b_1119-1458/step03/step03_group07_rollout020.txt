    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    # get program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    # compute b, h
    h_per_batch = H
    b = pid_batch // h_per_batch
    h = pid_batch % h_per_batch

    # compute offsets for output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base offsets
    # compute address offsets in bytes? Actually in triton pointer arithmetic it's in number of elements not bytes.
    # pointers are passed as 1D arrays with appropriate strides.

    # compute offset for each element: (b,i,k,h) using strides

    # right: (b*stride_right_b + j*stride_right_i + k*stride_right_k + h*stride_right_h)

import torch
import triton
import triton.language as tl

# Define Triton kernel for batched matmul

@triton.jit
def trimul_batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # tile over rows (i)
    pid_j = tl.program_id(1)  # tile over columns (j)
    pid_batch = tl.program_id(2)  # combined batch*head

    # Derive batch and head index
    h = pid_batch % H
    b = pid_batch // H

    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base offset for this (b, h)
    offset_b = b * stride_l_b          # same for left, right, out
    offset_h_l = h * stride_l_h
    offset_h_r = h * stride_r_h
    offset_h_o = h * stride_o_h

    # Create accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_base in range(0, N, BLOCK_K):
        # k axis indices for this block
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            offset_b + \
            (offs_i[:, None] * stride_l_i) + \
            (offs_k[None, :] * stride_l_k) + \
            offset_h_l
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # Load right tile transposed (k, j) to match matmul: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            offset_b + \
            (offs_j[None, :] * stride_r_i) + \
            (offs_k[:, None] * stride_r_k) + \
            offset_h_r
        right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)
        right = right.to(tl.float32)

        # Accumulate block product
        acc += tl.dot(left, right)

    # Write out result
    out_ptrs = out_ptr + \
        offset_b + \
        (offs_i[:, None] * stride_o_i) + \
        (offs_j[None, :] * stride_o_j) + \
        offset_h_o

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

TriMul (outgoing) – fused Triton implementation.

The forward pass consists of:
1️⃣ LayerNorm the input (FP32 → FP16).
2️⃣ Fuse the two projection linear layers (proj + gate) per side into a single
   mat‑mul, split the result and apply the sigmoid‑gate.
3️⃣ (Optional) apply the pair‑wise mask to the projected features.
4️⃣ Batch a set of independent matrix multiplications
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   (h = hidden_dim)
   using a custom Triton kernel – this removes the expensive
   permute‑+‑contiguous copy required by the naïve torch.bmm approach.
5️⃣ Second LayerNorm (over the hidden dimension), element‑wise gating,
   and final projection back to the original channel dimension.
All heavy arithmetic is performed in FP16 (accumulated in FP32) and the
result is cast back to FP32 as required by the API.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# left / right / out : [B, N, N, H] (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for the three tensors (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    # 0️⃣ program IDs
    # -----------------------------------------------------------------------------

    pid_j = tl.program_id(1)          # tile over cols  (j‑dimension)
    pid_batch = tl.program_id(2)      # combined batch*head index

    # 1️⃣ decode combined batch*head index
    # -------------------------------------------------------------------------
    h = pid_batch % H                 # hidden‑dim index
    b = pid_batch // H                 # batch index

    # -------------------------------------------------------------------------
    # 2️⃣ tile offsets inside the N×N matrix
    # -------------------------------------------------------------------------

    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # 3️⃣ offsets that are constant for the whole tile (batch & head)
    # -------------------------------------------------------------------------
    base_off_b = b * stride_l_b               # same for left/right/out
    base_off_h_l = h * stride_l_h
    base_off_h_r = h * stride_r_h
    base_off_h_o = h * stride_o_h

    # accumulator in FP32 (better numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # 4️⃣ loop over the contracted dimension k
    # -------------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile (i, k) -------------------------------------------------
        left_ptrs = left_ptr + \
            base_off_b + \
            (offs_i[:, None] * stride_l_i) + \
            (offs_k[None, :] * stride_l_k) + \
            base_off_h_l
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # ----- right tile transposed (k, j)  (so we can use tl.dot) -------------
        right_ptrs = right_ptr + \
            base_off_b + \
            (offs_j[None, :] * stride_r_i) + \
            (offs_k[:, None] * stride_r_k) + \
            base_off_h_r
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[None, :] & mask_k[:, None],
                             other=0.0).to(tl.float32)

        # ----- accumulate the block product ------------------------------------
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # -------------------------------------------------------------------------
    # 5️⃣ write back the result (cast to FP16)
    # -------------------------------------------------------------------------
    out_ptrs = out_ptr + \
        base_off_b + \
        (offs_i[:, None] * stride_o_i) + \
        (offs_j[None, :] * stride_o_j) + \
        base_off_h_o
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, on GPU)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool, on GPU)
        - weights_dict : name → torch.Tensor (float32, on GPU)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   FP32
    x = x.to(torch.float16)             # work in FP16 from now on

    # -------------------------------------------------
    # 2️⃣ Fuse projection + gate (2 linear calls instead of 4)
    # -------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                 # (2*H, dim)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                 # (2*H, dim)

    left_comb   = torch.nn.functional.linear(x, left_comb_w)   # [B,N,N,2*H]
    right_comb  = torch.nn.functional.linear(x, right_comb_w) # [B,N,N,2*H]

    # split back
    left_proj   = left_comb[..., :hidden_dim]   # [B,N,N,H]
    left_gate   = left_comb[..., hidden_dim:]   # [B,N,N,H]
    right_proj  = right_comb[..., :hidden_dim]  # [B,N,N,H]
    right_gate  = right_comb[..., hidden_dim:]  # [B,N,N,H]

    # -------------------------------------------------
    # 3️⃣ Optional masking (applied to the projected features)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Apply sigmoid gates and combine
    # -------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left  = left_proj * left_gate      # [B,N,N,H]
    right = right_proj * right_gate    # [B,N,N,H]

    # -------------------------------------------------
    # 5️⃣ Compute out_gate (still needed after the second LN)
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H]

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via the custom Triton GEMM
    # -------------------------------------------------
    out = torch.empty_like(left)   # allocate FP16 output tensor

    # strides (in elements) for the three tensors
    stride_l = left.stride()       # (B_stride, i_stride, k_stride, h_stride)
    stride_r = right.stride()
    stride_o = out.stride()

    # block sizes – tuned for H100 FP16 Tensor‑Core GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid dimensions: one program per (i‑tile, j‑tile, batch*head)
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_bh = B * hidden_dim
    grid = (grid_i, grid_j, grid_bh)

    _trimul_batched_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # H100 performs best with 8 warps per block for FP16
    )

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (FP16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Apply the output gate (element‑wise) – cheap, stay in PyTorch
    # -------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  FP16

    # -------------------------------------------------
    # 🔚 Cast back to float32 (API expects FP32)
    # -------------------------------------------------
    return out.to(torch.float32)
