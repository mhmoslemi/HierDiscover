    M, N, K,
    stride_am, stride_ak,

    stride_cm, stride_cn,

TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

The core O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed as a batched GEMM C = A @ Bᵀ where each

pipeline (num_stages=4) is used to maximise H100 tensor‑core
throughput.  All other operations (LayerNorm, Linear, Sigmoid)
stay in PyTorch with FP16 arithmetic.  The final result is
returned in float32 as required by the harness.

import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   B : [M, K, N]   (right already transposed)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit

    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]   (right transposed)

    stride_bm, stride_bk, stride_bj,   # strides for B (M,K,N)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,N,N)

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_z = tl.program_id(2)        # batch*hidden index

    # --------------------------------------------------------------

    mask_n = offs_n < N

    # --------------------------------------------------------------

    # Loop over the reduction dimension K
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- Load A tile (shape BLOCK_M×BLOCK_K) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---- Load B tile (shape BLOCK_K×BLOCK_N) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ---- FP16 × FP16 → FP32 accumulation ----
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write out the result (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors, launches the Triton kernel and restores
# the original layout [B, N, N, hidden_dim].
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] (fp16)
    right: torch.Tensor,  # [B, N, N, H] (fp16)
) -> torch.Tensor:
    """
    Compute left @ rightᵀ for each (batch, hidden) pair using Triton.
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape

    # Pack batch & hidden together: shape (M, N, K) with M = B*H
    #   left  → (M, N, K)  where K corresponds to the contracted dim
    #   right → (M, K, N)  (already transposed)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # K‑j order

    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,          # 4‑stage pipeline hides latency
    )

    # Reshape back to [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no grads).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------- unpack -------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ---------------------------- 1️⃣ LayerNorm ---------------------------
    # keep this in fp32 for best numerical stability
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # --------------------------- 2️⃣ Cast to fp16 -------------------------
    x = x.to(torch.float16)

    # ------------ 3️⃣ Cast model weights to fp16 (one‑time conversion) -----
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------------------- 4️⃣ Linear projections (fp16) -----------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # --------------------------- 5️⃣ Mask (if any) -----------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ---------------------- 6️⃣ Gating (sigmoid, fp16) -------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------- 7️⃣ O(N³) contraction via Triton -----------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # -------------------- 8️⃣ Second LayerNorm (fp16) --------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------ 9️⃣ Gating + final projection (fp16) ------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------- 10️⃣ Cast back to float32 -----------------------
    return out.to(torch.float32)
