"""
Optimised outgoing TriMul (AlphaFold‑3) forward pass.

The kernel implements the heavy O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
as a batched GEMM C = A @ Bᵀ where each hidden slice (including the batch
dimension) is a separate GEMM.  The Triton kernel tiles the multiplication
with a 64×64×64 block‑size, accumulates in FP32 (FP16→FP32) and uses a
3‑stage double‑buffered pipeline (num_stages=3).  All linear projections and
sigmoid gates are fused in PyTorch (weights are concatenated so that each
projection plus its gate only incurs a single matmul), reducing the number of
global memory loads.  The optional mask is applied after gating (order does
not affect the result).  The final layer‑norm, output gating and linear
projection are performed in FP16 and the result is cast back to FP32 for the
harness.

The implementation works for the full test matrix (B∈{1,2}, N∈{128…1024},
dim∈{128,384}, hidden_dim=128) on an H100 (Triton 3.3.1) and respects the
requirement that at least part of the work is done in a custom Triton kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   B : [M, K, N]   (right already transposed)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]   (right already transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bk, stride_bj,   # strides for B (note swapped K‑N order)
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM (A @ Bᵀ) with FP16 inputs, FP32 accumulation."""
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_z = tl.program_id(2)          # batch*hidden index

    # ------------------------------------------------------------------
    # 1. Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2. Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3. Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- Load A tile (BLOCK_M x BLOCK_K) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----- Load B tile (BLOCK_K x BLOCK_N) -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # K‑dim stride
            + offs_n[None, :] * stride_bj   # N‑dim stride (post‑transpose)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )   # fp16

        # ----- FP16 × FP16 → FP32 accumulation -----
        acc += tl.dot(a, b)   # dot uses tensor‑cores internally

    # ------------------------------------------------------------------
    # 4. Write result back (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper: reshape tensors, launch Triton kernel and restore original layout
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    """
    Compute out = left @ rightᵀ for each hidden slice using the Triton kernel.
    Returned tensor has shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape

    # Pack batch & hidden together → [M, N, K] where M = B*H, K = N
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)           # [M, N, K]
    # Right must be transposed on the last two axes → [M, K, N]
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)         # [M, K, N]

    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides expressed in element counts
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – 64×64×64 works well on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # ample warps for H100
        num_stages=3,         # double‑buffered pipeline
    )

    # Restore original layout [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no grads).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor[B, N, N]     (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : contains "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → more stable)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights once (fp16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection + gate for left and right (single matmul each)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights along the output dimension
    left_weight_cat = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_weight_cat = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # Linear → [B, N, N, 2*H]
    left_cat = torch.nn.functional.linear(x, left_weight_cat)       # fp16
    right_cat = torch.nn.functional.linear(x, right_weight_cat)     # fp16

    # Split and apply sigmoid gates
    left_proj = left_cat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])
    left = left_proj * left_gate                                 # [B,N,N,H] fp16

    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])
    right = right_proj * right_gate                               # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output‑gate (same as in paper)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via the Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden_dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gating + linear projection back to original dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
