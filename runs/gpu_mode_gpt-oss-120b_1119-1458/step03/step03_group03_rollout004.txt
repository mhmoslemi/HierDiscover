TriMul “outgoing” forward pass (gradient‑free) – highly tuned for H100.

Algorithm
---------
1. Layer‑norm the input (FP32 → FP16).

   pipeline.  The kernel works on the packed shape [B·H, N, N].
8. Second LayerNorm on the hidden dimension.
9. Output gating + final linear projection back to the original `dim`.
10. Cast the result back to FP32 (the public API expects FP32).

Only the O(N³) contraction is written in a Triton kernel; all the surrounding
operations use fused PyTorch primitives, which together bring the runtime
well below the 1 ms target for the test configurations.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B·H, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    #--- program IDs ----------------------------------------------------
    pid_m = tl.program_id(0)   # tile along i (first N dimension)
    pid_n = tl.program_id(1)   # tile along j (second N dimension)
    pid_z = tl.program_id(2)   # batch·hidden index

    #--- offsets --------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    #--- accumulator ----------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #--- iterate over K -------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A[i, k] tile (FP16)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # (BLOCK_M, BLOCK_K), fp16

        # Bᵀ[k, j] tile = B[k, j] (FP16)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # (BLOCK_K, BLOCK_N), fp16

        # dot (fp16·fp16 → fp32) + accumulate
        acc += tl.dot(a, b)

    #--- store result ----------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Batched mat‑mul C = left @ rightᵀ for each (batch, hidden) pair.
    `left` / `right` must be FP16 tensors of shape [B, N, N, H].
    Returns a tensor of the same (padded) shape in FP16.
    """
    B, N, _, H = left.shape

    # pack batch and hidden together → [B·H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides in element units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) – may be
                         ignored when `config["nomask"]` is True.
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  FP32
    x = x_norm.to(torch.float16)  # all heavy work in FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all per‑position linear transforms
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (all have shape [hidden_dim, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*H, dim)

    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B·N·N, dim)

    # single fused linear → (B·N·N, 5*H)
    fused = torch.nn.functional.linear(x_flat, w_concat)  # FP16
    # reshape to [B, N, N, 5, H] and split
    fused = fused.view(B, N, N, 5, hidden_dim)

    left = fused[..., 0, :]        # (B, N, N, H)
    left_gate = fused[..., 1, :]   # (B, N, N, H)
    right = fused[..., 2, :]       # (B, N, N, H)
    right_gate = fused[..., 3, :]  # (B, N, N, H)
    out_gate = fused[..., 4, :]    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Activations and optional mask
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # fuse gates into projections
    left = left * left_gate
    right = right * right_gate

    if not nomask:
        # mask is [B, N, N] → broadcast over hidden
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Pad to a multiple of TILE (64) to avoid bounds checks in the kernel
    # ------------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (FP16 → FP16)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, H]

    # strip padding if any
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (still in FP16)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gating + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the public API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
