Optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3). The pipeline is

  1. LayerNorm (float32 for stability).
  2. Cast to fp16 and apply four linear projections (left/right + three gates).
  3. Optional input mask (broadcast over the hidden dimension).
  4. Fuse gating with the projected tensors.
  5. Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
     Implemented as a batched mat‑mul with a Triton kernel that is
     auto‑tuned for the H100 Tensor‑Core pipeline.
  6. Second LayerNorm (fp16), output gate and final linear projection.
  7. Cast back to float32 (required by the test harness).

Only the O(N³) part runs on Triton; all other operations use fast
torch.nn.functional primitives.  Autotuning explores several tile
sizes (64×64×32, 128×128×64, 256×256×64) and chooses the best
configuration for each input shape.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: batched mat‑mul C = A @ Bᵀ  (fp16 in, fp16 out)

#   C    : [B*H, N, N]  (fp16)

            num_stages=3,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,  # *fp16  [M, N, K] (M = B*H, N = seq_len, K = seq_len)
    B_ptr,  # *fp16  same layout as A
    C_ptr,  # *fp16  output buffer
    M, N, K,
    stride_am, stride_ai, stride_ak,   # strides of A (in elements)
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_j = tl.program_id(1)  # tile index along j  (cols)
    pid_z = tl.program_id(2)  # batch*head index

    mask_j = offs_j < N

    # Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------

        offs_k = cur_k + tl.arange(0, BLOCK_K)

# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16, contiguous)
    right: torch.Tensor,  # [B, N, N, H]  (fp16, contiguous)
) -> torch.Tensor:
    """
    Computes  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using the tiled Triton kernel above.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension:
    #   [B, N, N, H] → [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in *elements* (Triton works with element strides)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Autotuned launch grid – the kernel itself decides the tile size
    grid = lambda meta: (
        triton.cdiv(N, meta["BLOCK_M"]),
        triton.cdiv(N, meta["BLOCK_N"]),
        B * H,
    )

        grid

        A,
        B_,
        C,
        B * H,   # dummy M (not used inside the kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        # BLOCK_M, BLOCK_N, BLOCK_K are supplied by the autotuner
    )

    # Reshape back to the original layout: [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool);
                         may be ignored when ``nomask=True``.
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   float32

    # Cast to fp16 for the heavy work
    x16 = x.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Load all linear weights in fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x16, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x16, right_proj_w)  # same shape

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x16, out_gate_w))

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 in / fp16 out)
    # -------------------------------------------------
    # Result: [B, N, N, hidden_dim]
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to dim

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
