def fused_proj_gate_kernel(
    a_ptr,                         # (M, K)
    proj_weight_ptr,               # (N, K) ??? shape (hidden, K) but stored row major: each hidden row has K columns.
    gate_weight_ptr,               # (N, K)
    mask_ptr,                      # (M)
    out_ptr,                       # (M, N)
    M, N, K,
    stride_a_m, stride_a_k,
    stride_proj_n, stride_proj_k,
    stride_gate_n, stride_gate_k,
    stride_mask_m,
    stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    # Grid: total number of tiles = (ceil(M/BLOCK_M), ceil(N/BLOCK_N)).
    # We'll compute pid mapping
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bound rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for a and weight:
    # a: (M, K) -> a_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    # weight_proj: (N, K) -> but we want (K, N) matrix for dot (like matmul a (M,K) * proj_weight^T (K,N)).
    # Usually weight stored as (N,K) we can treat b tiling as B = weight_proj, we'll read weight_proj as (K,N):
    # So offset for weight_proj: weight_ptr + offs_k[:, None] * stride_proj_k + offs_n[None, :] * stride_proj_n
    # where stride_proj_k = 1 (contiguous over dim?), stride_proj_n = K.
    # Because weight_proj shape (N, K) row-major layout: element weight_proj[i,k] is at pointer + i* stride_n + k* stride_k
    # We'll set stride_proj_n = K, stride_proj_k = 1. Confirm: if weight_proj is of shape (hidden, dim) and stored contiguous row-major, stride for hidden dimension (row) is K (dim), stride for dim dimension (col) is 1.
    # So reading slices this way yields weight_proj rows and columns.
    
    # We'll create accumulators for proj and gate: float32.
    acc_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(a_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        w_gate = tl.load(gate_weight_ptr + offs_k[:, None] * stride_gate_k + offs_n[None, :] * stride_gate_n,
                          mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot: a (M,BLOCK_K) * w (BLOCK_K, N)
        # tl.dot works for shape (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) => (BLOCK_M, BLOCK_N)

Pseudo-code:

```
@triton.jit
def fused_proj_gate_kernel(
    a_ptr, proj_weight_ptr, gate_weight_ptr,
    mask_ptr,
    out_ptr,
    M, N, K,
    stride_ak, stride_am,
    stride_proj_k, stride_proj_n,
    stride_gate_k, stride_gate_n,
    stride_mask_m,
    stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_offs_m = offs_m < M
    mask_offs_n = offs_n < N

    acc_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A block
        a = tl.load(a_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak),
                    mask=mask_offs_m[:, None] & mask_k[None, :], other=0.0)

        # load proj weight block
        w_proj = tl.load(proj_weight_ptr + (offs_k[:, None] * stride_proj_k) + (offs_n[None, :] * stride_proj_n),
                         mask=mask_k[:, None] & mask_offs_n[None, :], other=0.0)

        # load gate weight block
        w_gate = tl.load(gate_weight_ptr + (offs_k[:, None] * stride_gate_k) + (offs_n[None, :] * stride_gate_n),
                         mask=mask_k[:, None] & mask_offs_n[None, :], other=0.0)

        acc_proj += tl.dot(a, w_proj)
        acc_gate += tl.dot(a, w_gate)

    # Apply sigmoid to gate accumulator
    gate = 1.0 / (1.0 + tl.exp(-acc_gate))

    # Multiply proj * gate
    out = acc_proj * gate

    if HAS_MASK:
        # broadcast mask (scalar per row)
        mask_val = tl.load(mask_ptr + offs_m * stride_mask_m, mask=mask_offs_m, other=1.0)
        mask_val = mask_val.to(tl.float32)
        out = out * mask_val[:, None]

    # store output in fp16
    tl.store(out_ptr + (offs_m[:, None] * stride_out_m) + (offs_n[None, :] * stride_out_n),
             out.to(tl.float16), mask=mask_offs_m[:, None] & mask_offs_n[None, :])
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation fuses the five linear layers in the original
PyTorch code into a **single** matrix‑multiply (by concatenating the
weight matrices).  This reduces memory traffic dramatically because the
input tensor `x` (shape B·N·N×dim) is read only once.  The three gating
operations are applied with a tiny Triton kernel that also fuses the
optional mask without ever expanding it to the hidden dimension.

The expensive pairwise contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is carried out as a batched GEMM on the (B·hidden) matrices
(N×N) using `torch.matmul`, which on an H100 uses Tensor‑cores.
All heavy arithmetic stays in fp16; only the final result is cast back
to the input dtype (normally float32).

A Triton kernel is required – the `fused_mul3_masked` kernel implements
the elementwise product `proj * gate * mask` while broadcasting the
mask on‑the‑fly.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel that computes a * c * mask  (mask broadcasted over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,               # left_proj_raw  (B,N,N,hidden)
    b_ptr,               # mask           (B,N,N)
    c_ptr,               # gate           (B,N,N,hidden)
    out_ptr,             # destination
    hidden: tl.constexpr,
    size:   tl.constexpr,          # total number of elements (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < size

    # load left projection and gate (contiguous over hidden)
    a = tl.load(a_ptr + offs, mask=active, other=0.0)
    c = tl.load(c_ptr + offs, mask=active, other=0.0)

    # compute the corresponding mask index (broadcast over hidden)
    mask_idx = offs // hidden                      # integer division
    b = tl.load(b_ptr + mask_idx, mask=active, other=0.0)

    out = a * c * b
    tl.store(out_ptr + offs, out, mask=active)


def fused_mul3_masked(proj: torch.Tensor,
                      mask: torch.Tensor,
                      gate: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """
    Element‑wise product proj * gate * mask where mask is broadcast over the
    last (hidden) dimension.  All tensors must be contiguous and dtype
    torch.float16.
    """
    B, N, _, H = proj.shape
    assert H == hidden, "hidden dimension mismatch"
    assert proj.shape == gate.shape, "proj and gate must share shape"
    assert mask.shape == (B, N, N), "mask shape mismatch"

    proj = proj.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()

    out = torch.empty_like(proj)
    total = proj.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        proj,
        mask,
        gate,
        out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out

# ----------------------------------------------------------------------

    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)

            - mask_tensor:  shape [B, N, N] (bool/float, CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim"

    """

    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    device = inp.device

    # 1) Input LayerNorm (FP32 -> FP16)
    # ------------------------------------------------------------------

        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)

    # 2) Fuse the five linear layers into a single matmul
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # ------------------------------------------------------------------
    M = B * N * N                     # number of rows for the tall‑skinny matmul

        ],

    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N)  (k‑dim transposed)

    bh = B * hidden
    left_mat = left_perm.view(bh, N, N)
    right_mat = right_perm.view(bh, N, N)

    out_hidden = torch.matmul(left_mat, right_mat)       # (B*hidden, N, N)

    # restore original layout (B, N, N, hidden)
    out = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # apply the out‑gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate

    # final linear back to original `dim`
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),
    )   # bias‑free, fp16

    # Cast back to the input dtype (usually float32) before returning
    return out_final.to(inp.dtype)
```
