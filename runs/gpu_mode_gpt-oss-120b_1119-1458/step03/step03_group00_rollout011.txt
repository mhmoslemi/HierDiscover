"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16).  The tensor is normalized over the
   channel dimension *dim* and cast to half‑precision for the heavy
   arithmetic that follows.

2️⃣ **Fused projection & gating** – Five independent Linear layers
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`,
   `out_gate`) are concatenated into a *single* GEMM:
        [left_proj, left_gate, right_proj, right_gate, out_gate]
   =  Linear(x_norm, W_concat)
   This removes the four extra kernel launches and the repeated reads
   of the same input tensor, cutting the memory‑bandwidth cost of the
   projection phase dramatically.

3️⃣ Element‑wise gating (sigmoid) and optional mask application.
   When a mask is present we use a tiny Triton kernel
   `fused_mul3_masked` that multiplies the projection, sigmoid‑gate and
   mask on‑the‑fly without broadcasting the mask, thus saving an extra
   read of the mask per hidden channel.

4️⃣ **Core contraction** – The outgoing update
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply (B·hidden independent GEMMs of size
   N×N).  By permuting the tensors to (B, hidden, N, N) we can call the
   highly‑optimized cuBLAS strided‑batched GEMM via `torch.matmul`.

5️⃣ Output LayerNorm, out‑gate multiplication and final projection.
   The hidden result is layer‑normalized over the hidden dimension,
   multiplied by the previously computed `out_gate`, and finally linearly
   projected back to the original `dim`.  All heavy arithmetic stays in
   FP16; the final tensor is cast back to the input dtype (usually
   float32).

Only the mask‑gate multiplication is written in Triton, satisfying the
requirement to provide a custom kernel while keeping the rest of the
pipeline in highly‑optimized PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: a * sigmoid(b) * mask   (mask broadcasted over hidden)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    proj_ptr,            # left/right projection (B,N,N,hidden)
    mask_ptr,            # mask (B,N,N)
    gate_ptr,            # pre‑sigmoid gate (B,N,N,hidden)
    out_ptr,             # destination (B,N,N,hidden)
    hidden: tl.constexpr,
    total: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # load projection and gate (contiguous over hidden)
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate_raw = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # sigmoid on‑the‑fly
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    # compute mask index (broadcast hidden)
    mask_idx = offs // hidden
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(proj: torch.Tensor,
                      mask: torch.Tensor,
                      gate_raw: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """
    Element‑wise product proj * sigmoid(gate_raw) * mask
    where ``mask`` is broadcast over the hidden dimension.
    All tensors must be contiguous, ``proj``/``gate_raw`` are
    (B,N,N,hidden), ``mask`` is (B,N,N), and dtype is float16.
    """
    B, N, _, H = proj.shape
    assert H == hidden, "hidden dimension mismatch"
    total = proj.numel()
    BLOCK = 8192  # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    out = torch.empty_like(proj)
    fused_mul3_masked_kernel[grid](
        proj,
        mask,
        gate_raw,
        out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out

    )  # (5*hidden, dim)

    # Flatten input to (M, dim) where M = B·N·N
    M = B * N * N
    x_flat = x_norm.view(M, dim)                     # (M, dim)

    # Single linear call – this replaces five separate Linear layers.
    # No bias in the original model.
    proj_all = torch.nn.functional.linear(x_flat, w_concat)  # (M, 5*hidden)

    # Reshape back to [B, N, N, 5*hidden] and split.
    proj_all = proj_all.view(B, N, N, 5 * hidden)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_all, hidden, dim=3
    )  # each (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3) Apply sigmoid to gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,hidden)
    right_gate = torch.sigmoid(right_gate_raw) # (B,N,N,hidden)
    out_gate = torch.sigmoid(out_gate_raw)     # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Apply mask and gate (mask optional)
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask.to(torch.float16)               # (B,N,N)
        # Fuse mask, gate and projection with a tiny Triton kernel.
        left = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate_raw, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate_raw, hidden)

    # ------------------------------------------------------------------
    # 5) Core contraction: batched GEMM per hidden channel
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) so we can use torch.matmul.
    left_perm = left.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1) # (B, hidden, k, j)

    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # Back to (B, i, j, hidden)
    out_hidden = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate           # gating

    # ------------------------------------------------------------------
    # 7) Final projection back to original dimension
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)

    # Flatten, apply linear, and reshape back.
    out_flat = out_norm.view(M, hidden)                 # (M, hidden)
    out_proj = torch.nn.functional.linear(out_flat, to_out_weight)  # (M, dim)
    out = out_proj.view(B, N, N, dim)

    # Cast back to the original dtype (usually float32)
    return out.to(inp.dtype)
