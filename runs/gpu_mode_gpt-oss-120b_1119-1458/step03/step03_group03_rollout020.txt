"""
Optimized outgoing TriMul kernel (AlphaFold‑3)

Algorithm:
1. Layer‑norm input in FP32 → FP16.
2. Fuse the five linear projections (left_proj, right_proj, left_gate, right_gate, out_gate) into a single GEMM.
3. Apply sigmoid to gates, multiply gates into projections, and optionally mask.
4. Pad the sequence dimension to a multiple of TILE (64) to drop bounds checks.
5. Compute ``out = left @ rightᵀ`` for each (batch, hidden) pair with a custom Triton kernel.
   The kernel uses 64×64×64 tiles, FP16 → FP32 accumulation, 8 warps, 3‑stage pipeline.
6. Remove padding, apply second LayerNorm, out‑gate, and final linear projection.
7. Cast the result back to FP32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel for batched matrix multiplication C = A @ Bᵀ
# Input shape:  A, B  -> [M, N, K]  (M = batch * hidden, N = seq_len_padded, K = seq_len_padded)
# Output shape:       C  -> [M, N, N]
# --------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids (tile indices)
    pid_m = tl.program_id(0)   # tile over rows  (i)
    pid_n = tl.program_id(1)   # tile over cols  (j)
    pid_z = tl.program_id(2)   # batch*hidden index

    # start offsets of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointer to the start of the output tile
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k):
        # current K offset
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load a tile of A (shape BLOCK_M×BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs)          # fp16

        # Load a tile of Bᵀ i.e. B[k, j] (shape BLOCK_K×BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs)          # fp16

        # fused multiply‑add (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # store result (cast back to fp16)
    tl.store(c_ptrs, acc.to(tl.float16))


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Batched matmul C = left @ rightᵀ using the Triton kernel above.
    ``left`` and ``right`` must be FP16 tensors of shape [B, N, N, H].
    ``tile`` must divide the padded sequence length.
    Returns a tensor of shape [B, N, N, H] (still padded).
    """
    B, N, _, H = left.shape

    # pack batch and hidden -> [B*H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    # allocate output
    C = torch.empty_like(A)   # fp16

    # stride in element counts
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)  # work in fp16 from now on

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into one GEMM
    # ------------------------------------------------------------------
    # gather projection / gate weights and cast to fp16 once
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    # flatten input for the fused linear layer
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                         # (B*N*N, dim)

    # one large GEMM → (B*N*N, 5*hidden_dim)
    combined = torch.nn.functional.linear(x_flat, combined_w).view(
        B, N, N, 5 * hidden_dim
    )  # [B, N, N, 5*hidden_dim] fp16

    # split back to the individual tensors
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        combined, hidden_dim, dim=3
    )  # each -> [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # mix gate into projections
    left = left_raw * left_gate
    right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Pad sequence dimension to a multiple of TILE (remove bounds checks)
    # ------------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = torch.zeros(
            (B, N_pad, N_pad, hidden_dim), dtype=torch.float16, device=left.device
        )
        right_pad = torch.zeros_like(left_pad)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via the custom Triton GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, hidden_dim]

    # strip padding if it was added
    if pad_len != 0:
        out = out[:, :N, :N, :]  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (still fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected by downstream code)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

"""
Optimized outgoing TriMul kernel (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 → FP16.
2️⃣  Fuse the five linear transforms (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single GEMM.
3️⃣  Apply the three sigmoid gates, multiply them into the
    corresponding projections and (optionally) mask the results.
4️⃣  Pad the sequence dimension to a multiple of TILE (64) so the
    Triton kernel can omit bounds checks.
5️⃣  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for each (batch, hidden) pair with a custom Triton GEMM
    (C = A @ Bᵀ).  The kernel uses 64×64×64 tiles, FP16 inputs,
    FP32 accumulation, 8 warps and a 3‑stage pipeline.
6️⃣  Remove padding, apply a second LayerNorm, the output gate,
    and a final linear projection back to `dim`.
7️⃣  Return the result in FP32.

Only the O(N³) contraction is executed in Triton; the surrounding
operations use high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]  (M = batch * hidden, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile over output rows (i)
    pid_n = tl.program_id(1)   # tile over output columns (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # tile offsets (guaranteed in‑bounds because N is padded)
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i, k] tile  (BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs)          # fp16

        # Load Bᵀ[k, j] tile  (BLOCK_K × BLOCK_N)  i.e. B[k, j]
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs)          # fp16

        # fused multiply‑add: fp16 × fp16 → fp32
        acc += tl.dot(a, b)

    # write result (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16))


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Batched matmul C = left @ rightᵀ using the Triton kernel above.
    ``left`` and ``right`` must be FP16 tensors of shape [B, N, N, H].
    Returns a tensor of the same (padded) shape.
    """
    B, N, _, H = left.shape

    # pack batch and hidden → [B*H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    C = torch.empty_like(A)   # fp16 output

    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                                   # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)         # work in FP16 from now on

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single GEMM
    # ------------------------------------------------------------------
    # collect weights and cast once
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    combined_w = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(torch.float16)                 # (5*hidden_dim, dim)

    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)          # (B·N·N, dim)

    # one large linear projection → (B·N·N, 5·hidden_dim)
    combined = torch.nn.functional.linear(x_flat, combined_w).view(
        B, N, N, 5 * hidden_dim
    )                                   # FP16

    # split the fused result
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        combined, hidden_dim, dim=3
    )                                   # each [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # incorporate gates
    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Pad to a multiple of TILE to drop bounds checks in Triton
    # ------------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad  = torch.zeros(
            (B, N_pad, N_pad, hidden_dim), dtype=torch.float16, device=left.device
        )
        right_pad = torch.zeros_like(left_pad)
        left_pad[:, :N, :N, :]  = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (FP16 I/O, FP32 accumulation)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, hidden_dim]

    # remove padding if it was added
    if pad_len != 0:
        out = out[:, :N, :N, :]        # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + final projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (expected by downstream code)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
