"""
TriMul (outgoing) – fast Triton implementation
================================================

This module implements the forward‑pass of the “outgoing” TriMul operation
used by AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input tensor over the feature axis (float32).
2️⃣ Cast to FP16 and apply all linear projections and sigmoid gates.
3️⃣ (Optional) Apply a pair‑wise mask to the projected tensors.
4️⃣ Fuse the hidden‑dim gates with the projected tensors.
5️⃣ Heavy O(N³) contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realized as a batched matrix multiplication `C = A @ Bᵀ`
   where `A` and `B` have shape (M, N, K) with
       M = batch * hidden_dim, N = seq_len, K = seq_len.
   The custom Triton kernel tiles the multiplication with 64×64×64
   blocks, uses FP16 inputs, accumulates in FP32 via `tl.dot`,
   double‑buffers (`num_stages=2`) and runs 8 warps per program.
6️⃣ Second LayerNorm (hidden_dim) in FP16.
7️⃣ Apply the output gate, a final linear projection back to the original
   dimension and cast the result to float32.

Only the O(N³) contraction is executed in Triton; all other steps use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32: sizes (M unused, N is dim, K is dim)
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix product for many independent (N×N) matrices.

    Each program processes a single (M‑index) i.e. a (batch, hidden) pair.
    The i/j dimensions are tiled with blocks of size BLOCK_M × BLOCK_N.
    """
    pid_i = tl.program_id(0)   # tile along the first N dimension (i)
    pid_j = tl.program_id(1)   # tile along the second N dimension (j)
    pid_z = tl.program_id(2)   # index over M (= batch * hidden_dim)

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: (M, i, k)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load B tile transposed on‑the‑fly: we need Bᵀ (k, j)
        # Original B layout is (M, j, k); we read B[m, j, k] → (k, j) after swap.
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Multiply‑accumulate using tensor‑core MMA (fp16 → fp32)
        acc += tl.dot(a, b)

    # Store the result tile, casting back to fp16
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Perform the tensor contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Both ``left`` and ``right`` are FP16 tensors of shape [B, N, N, H].
    Returns an FP16 tensor of the same shape.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension.
    # After permute & contiguous we have shape (B*H, N, N) with stride (N*N, N, 1).
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in number of elements (Triton expects i32).
    stride_am, stride_ai, stride_ak = A.stride()      # (M, N, K)
    stride_bm, stride_bi, stride_bk = B_.stride()     # (M, N, K)
    stride_cm, stride_ci, stride_cj = C.stride()      # (M, N, N)

    # Tile sizes – 64×64×64 gives a good balance on H100 for fp16 MMA.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: i‑tiles × j‑tiles × (batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the remainder of the pipeline
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ FP16 linear weights (no bias in the original model)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (native PyTorch, FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask has shape [B,N,N,1] and dtype fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates into projections
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (FP16 in/out)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + final projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                 # element‑wise gate (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
