"""
Outgoing TriMul (AlphaFold‑3) – high‑performance forward pass.

Algorithm Overview
------------------
1. Layer‑norm the 4‑D input (float32) → cast to fp16.
2. Four linear projections (left/right) and three gating projections (fp16).
   If a mask is supplied it is applied (broadcast on the hidden dimension).
3. Fuse each projection with its gate (in‑place mul).
4. Heavy O(N³) contraction computed as a batched matrix‑multiply
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly `C = A @ Bᵀ` for each hidden dimension `h`.  The
   contraction is performed by a custom Triton kernel that uses
   fp16 Tensor‑Core MMA with fp32 accumulation.
5. Second Layer‑norm (fp16) → out‑gate (sigmoid) → final linear proj. back
   to `dim`.
6. Cast the result back to float32 (the harness expects float32).

Only the O(N³) contraction is written in Triton; the remaining
operations use high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# --------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # fp16*   [M, N, K]
    B_ptr,               # fp16*   [M, N, K]
    C_ptr,               # fp16*   [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k) – note j/k order!
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile over output i dimension
    pid_n = tl.program_id(1)   # tile over output j dimension
    pid_z = tl.program_id(2)   # batch*hidden index

    # --------------------------------------------------
    # Tile offsets and boundary masks
    # --------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------
        # Load A tile: shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # --------------------------------------------------
        # Load B tile (transposed) : shape (BLOCK_K, BLOCK_N)
        #   B is accessed as B[pid_z, k, j] → this is Bᵀ on the last two axes.
        # --------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision MMA: fp16 * fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------
    # Store C tile (fp16)
    # --------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    using the Triton kernel above.
    Arguments
    ---------
    left, right : torch.Tensor  [B, N, N, H]   (fp16, contiguous)
    Returns
    -------
    out : torch.Tensor [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Pack batch & hidden:   [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile sizes – 64×64×64 works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates in‑place to reduce peak memory
    torch.mul(left, left_gate, out=left)
    torch.mul(right, right_gate, out=right)

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 in / fp16 out)
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate                     # element‑wise gate (fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
```

**Explanation of the Triton kernel**

- The kernel computes a batched matrix multiplication with the second operand transposed on the last two axes, exactly matching `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
- Tile sizes `BLOCK_M = BLOCK_N = BLOCK_K = 64` give 64 × 64 × 64 MMA work‑groups which map efficiently onto H100 Tensor‑Cores.
- Inputs are fp16; the accumulator is fp32 to keep numerical stability.
- Boundary handling is performed with masks (`mask_i`, `mask_j`, `offs_k < K`).
- The Python helper reshapes the 4‑D tensors to a 3‑D view `[B*H, N, N]`, launches the kernel, then reshapes back to `[B, N, N, H]`.

All remaining operations (layer‑norms, linear projections, gating, and final projection) are done with highly‑optimized PyTorch primitives, while the only computational bottleneck—the O(N³) contraction—is fully fused inside the Triton kernel. The implementation works for any of the test configurations listed in the prompt, supports optional masking, and returns a float‑32 tensor as required.
