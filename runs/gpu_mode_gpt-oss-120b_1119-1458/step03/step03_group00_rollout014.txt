def custom_kernel(data):
    x, mask, weights, cfg = data
    B,N,_,dim = x.shape
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps",1e-5)
    nomask = cfg.get("nomask",False)

    # Normalize input x using torch.nn.functional.layer_norm float32 weight biases and then cast to fp16
    # For speed, we can also implement using Triton but easier to keep F.layer_norm.

    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)

    # allocate left, right, out_gate
    left = torch.empty((B,N,N,hidden), dtype=torch.float16, device=x.device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # compute strides
    stride_x = x_norm.stride()  # (B,N,N,dim)
    stride_x_b, stride_x_n1, stride_x_n2, stride_x_dim = stride_x

    mask_fp16 = mask.to(torch.float16) if not nomask else None
    if not nomask:
        mask_stride = mask_fp16.stride()
        stride_mask_b, stride_mask_n1, stride_mask_n2 = mask_stride
    else:
        # dummy values not used
        stride_mask_b = stride_mask_n1 = stride_mask_n2 = 0

    # weight strides: they are contiguous
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    # Ensure weight layout: shape (hidden,dim)
    # No bias.

    # check that weight is contiguous and row-major (out_features, in_features).
    # Strides:
    w_stride0 = left_proj_weight.stride(0)  # dim
    w_stride1 = left_proj_weight.stride(1)  # 1

    # As they are contiguous inner (dim) index stride = 1, outer hidden stride = dim.

    # Define block sizes
    BLOCK_H = 32
    BLOCK_D = 32 # ensure <dim

    # calculate total elements B*N*N
    total = B * N * N
    grid = (total,)

    # Launch kernel
    tri_proj_gate_kernel[grid](
        x_norm,
        mask_fp16 if not nomask else torch.tensor([], dtype=torch.float16, device=x.device),
        left_proj_weight,
        left_gate_weight,
        right_proj_weight,
        right_gate_weight,
        out_gate_weight,
        left,
        right,
        out_gate,
        # strides and dims
        stride_x_b, stride_x_n1, stride_x_n2, stride_x_dim,
        stride_mask_b, stride_mask_n1, stride_mask_n2,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        dim, hidden,
        N,  # we need seq_len for computing i,k given idx
        nomask,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )
    # Step 4: contraction
    # left shape (B,N,N,hidden) => (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # perform batched matmul
    out_hidden = torch.matmul(left_perm, right_perm) # shape (B, hidden, N, N)

    # reshape back:
    out = out_hidden.permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden)

    # Output LayerNorm
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # gating
    out_norm = out_norm * out_gate   # out_gate is already sigmoid

    # final linear projection (to dim)
    out_final_fp16 = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # cast back to original dtype
    return out_final_fp16.to(x.dtype)

"""
TriMul (outgoing) – Triton‑fused projection + gating + mask
==========================================================

We implement the heavy part of the TriMul forward pass (the five linear layers,
their sigmoid gates and the optional pair‑wise mask) in a single Triton kernel.
For each element (b,i,k) of the normalized input tensor:

    * Compute 5 matrix‑vector products:
        left_proj   = x[b,i,k,:] @ W_left_proj   (H x D)
        left_gate   = sigmoid(x[b,i,k,:] @ W_left_gate)
        right_proj  = x[b,i,k,:] @ W_right_proj
        right_gate  = sigmoid(x[b,i,k,:] @ W_right_gate)
        out_gate    = sigmoid(x[b,i,k,:] @ W_out_gate)

    * Apply the mask (broadcast on the hidden dimension) and the gates:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask

The kernel tiles the hidden dimension (BLOCK_H) and the input feature dimension
(BLOCK_D) to keep all intermediate accumulation in FP32 while the stored result
remains FP16.  After this kernel the remaining work (the O(N³) batched matrix‑
multiply, output layer‑norm, output gating and final projection) is performed
with highly‑optimized PyTorch/cuBLAS kernels.

This design removes the repeated reads of the input tensor and the five weight
matrices that a naïve implementation would incur, dramatically reducing memory
traffic and allowing the contracted GEMM to dominate the runtime.
"""

    else:
        mask_idx = b * stride_mask_b + i * stride_mask_n1 + k * stride_mask_n2
        # mask is FP16, we cast to FP32 for arithmetic
        mask_val = tl.load(mask_ptr + mask_idx, mask=1, other=0.0).to(tl.float32)

    # ------------------------------------------------------------------
    # loop over hidden dimension in tiles of BLOCK_H
    # ------------------------------------------------------------------
    for ho in range(0, H, BLOCK_H):
        # initialise FP32 accumulators for this tile
        acc_lproj   = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_lgate   = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_rproj   = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_rgate   = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_outgate = tl.zeros([BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        # dot over the input feature dimension in tiles of BLOCK_D
        # ------------------------------------------------------------------
        for do in range(0, D, BLOCK_D):
            # ------- load a slice of the input vector -------
            dim_off = do
            dim_idx = dim_off + tl.arange(0, BLOCK_D)
            dim_mask = dim_idx < D
            x_vec = tl.load(x_base + dim_off * stride_x_d,
                            mask=dim_mask,
                            other=0.0).to(tl.float32)               # (BLOCK_D,)

            # ------- prepare a mask for the weight sub‑matrix -------
            h_off = ho
            hidden_idx = h_off + tl.arange(0, BLOCK_H)[:, None]   # (BLOCK_H,1)
            dim_idx2   = dim_off + tl.arange(0, BLOCK_D)[None, :] # (1,BLOCK_D)
            w_mask = (hidden_idx < H) & (dim_idx2 < D)            # (BLOCK_H,BLOCK_D)

            # ------- load weight tiles -------
            w_lproj   = tl.load(w_lproj_ptr   + h_off * D + dim_off,
                                 mask=w_mask, other=0.0,
                                 shape=(BLOCK_H, BLOCK_D)).to(tl.float32)
            w_lgate   = tl.load(w_lgate_ptr   + h_off * D + dim_off,
                                 mask=w_mask, other=0.0,
                                 shape=(BLOCK_H, BLOCK_D)).to(tl.float32)
            w_rproj   = tl.load(w_rproj_ptr   + h_off * D + dim_off,
                                 mask=w_mask, other=0.0,
                                 shape=(BLOCK_H, BLOCK_D)).to(tl.float32)
            w_rgate   = tl.load(w_rgate_ptr   + h_off * D + dim_off,
                                 mask=w_mask, other=0.0,
                                 shape=(BLOCK_H, BLOCK_D)).to(tl.float32)
            w_outgate = tl.load(w_outgate_ptr + h_off * D + dim_off,
                                 mask=w_mask, other=0.0,
                                 shape=(BLOCK_H, BLOCK_D)).to(tl.float32)

            # ------- accumulate dot products (matrix‑vector) -------
            #   (BLOCK_H, BLOCK_D) @ (BLOCK_D,) -> (BLOCK_H,)
            acc_lproj   += tl.sum(w_lproj   * x_vec, axis=1)
            acc_lgate   += tl.sum(w_lgate   * x_vec, axis=1)
            acc_rproj   += tl.sum(w_rproj   * x_vec, axis=1)
            acc_rgate   += tl.sum(w_rgate   * x_vec, axis=1)
            acc_outgate += tl.sum(w_outgate * x_vec, axis=1)

        # ------------------------------------------------------------------
        # apply sigmoid to gate accumulators
        # ------------------------------------------------------------------
        sigmoid = lambda a: 1.0 / (1.0 + tl.exp(-a))
        left_gate   = sigmoid(acc_lgate)
        right_gate  = sigmoid(acc_rgate)
        out_gate    = sigmoid(acc_outgate)

        # ------------------------------------------------------------------
        # apply mask and gates to the projected values
        # ------------------------------------------------------------------
        left_val  = acc_lproj * left_gate * mask_val
        right_val = acc_rproj * right_gate * mask_val

        # ------------------------------------------------------------------
        # store results – hidden tail handling via a mask
        # ------------------------------------------------------------------
        h_idx = ho + tl.arange(0, BLOCK_H)
        h_mask = h_idx < H

        # pointers to the destination tiles
        left_ptr   = left_out_ptr   + b * stride_lo_b   + i * stride_lo_n1   + k * stride_lo_n2   + ho * stride_lo_h
        right_ptr  = right_out_ptr  + b * stride_ro_b   + i * stride_ro_n1   + k * stride_ro_n2   + ho * stride_ro_h
        og_ptr     = outgate_out_ptr+ b * stride_og_b   + i * stride_og_n1   + k * stride_og_n2   + ho * stride_og_h

        tl.store(left_ptr,   left_val.to(tl.float16),   mask=h_mask)
        tl.store(right_ptr,  right_val.to(tl.float16),  mask=h_mask)
        tl.store(og_ptr,     out_gate.to(tl.float16),   mask=h_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  shape [B, N, N] (bool/float, CUDA) – may be omitted
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim"
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", False)          # True → no mask is applied

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep result in FP16)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)          # all heavy arithmetic in FP16

    # --------------------------------------------------------------
    # 2) Fuse linear projections, sigmoid gates and optional mask
    # --------------------------------------------------------------
    # allocate output buffers
    left  = torch.empty((B, N, N, hidden), dtype=torch.float16,
                        device=inp.device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # make sure all weight tensors are contiguous FP16
    w_lproj = weights["left_proj.weight"].contiguous().to(torch.float16)
    w_rproj = weights["right_proj.weight"].contiguous().to(torch.float16)
    w_lgate = weights["left_gate.weight"].contiguous().to(torch.float16)
    w_rgate = weights["right_gate.weight"].contiguous().to(torch.float16)
    w_outgate = weights["out_gate.weight"].contiguous().to(torch.float16)

    # mask handling – cast once to FP16 if present
    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        # dummy tensor – never dereferenced because NOMASK=1
        mask_fp16 = torch.empty((1,), dtype=torch.float16, device=inp.device)

    # strides (in number of elements, not bytes)
    stride_x = x_norm.stride()                     # (B,N,N,dim)
    stride_mask = mask_fp16.stride() if not nomask else (0, 0, 0)
    stride_lo = left.stride()
    stride_ro = right.stride()
    stride_og = out_gate.stride()

    # tile sizes – tune for the H100 (these work for all listed configs)
    BLOCK_H = 32   # hidden‑dim tile
    BLOCK_D = 32   # input‑feature tile (dim ≤ 384 → 32 works)

    total_elements = B * N * N
    grid = (total_elements,)

    fused_proj_gate_mask_kernel[grid](
        # inputs
        x_norm,
        mask_fp16,
        w_lproj, w_lgate,
        w_rproj, w_rgate,
        w_outgate,
        # outputs
        left, right, out_gate,
        # strides
        stride_x[0], stride_x[1], stride_x[2], stride_x[3],
        stride_mask[0] if not nomask else 0,
        stride_mask[1] if not nomask else 0,
        stride_mask[2] if not nomask else 0,
        stride_lo[0], stride_lo[1], stride_lo[2], stride_lo[3],
        stride_ro[0], stride_ro[1], stride_ro[2], stride_ro[3],
        stride_og[0], stride_og[1], stride_og[2], stride_og[3],
        # compile‑time constants
        N, dim, hidden,
        BLOCK_H, BLOCK_D,
        int(nomask),           # NOMASK = 1 → no mask, 0 → mask present
    )

    # --------------------------------------------------------------
    # 3) Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM for each hidden channel.
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, N) – keep strides compatible with cuBLAS
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, N)
    out_h = torch.matmul(left_perm, right_perm)         # (B, H, N, N)

    # restore original layout: (B, N, N, H)
    out = out_h.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4) Output LayerNorm → output‑gate multiplication → final projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # apply the previously computed output gate (already sigmoid‑activated)
    out_norm = out_norm * out_gate

    # final linear projection back to the original feature dimension
    out_final_fp16 = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden)
    )

    # cast back to the original dtype (usually float32)
    return out_final_fp16.to(inp.dtype)
