def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    b, n, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # First LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias
    )
    # Convert to fp16
    x_fp16 = x_norm.to(torch.float16)

    # Combine Linear projections and gates
    # Stack the weights along output dimension (first dim)
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # cat order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w
    ], dim=0).to(torch.float16)  # shape (5*hidden, dim)

    # Flatten input for linear
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)

    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden)
    combined = combined.view(b, n, n, 5*hidden_dim)  # shape (B,N,N,5H)

    # Split
    h = hidden_dim
    proj_left = combined[..., :h]
    gate_left_raw = combined[..., h:2*h]
    proj_right = combined[..., 2*h:3*h]
    gate_right_raw = combined[..., 3*h:4*h]
    gate_out_raw = combined[..., 4*h:5*h]

    # Sigmoid gates
    gate_left = torch.sigmoid(gate_left_raw)
    gate_right = torch.sigmoid(gate_right_raw)
    gate_out = torch.sigmoid(gate_out_raw)

    # Mask if needed
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        proj_left = proj_left * mask_fp16
        proj_right = proj_right * mask_fp16

    # Apply gates to get left/right tensors
    left = proj_left * gate_left
    right = proj_right * gate_right

    # Heavy contraction: out = left @ right.T per hidden dimension
    # Reshape for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N)
    B_H = b * hidden_dim
    left_mat = left_perm.view(B_H, n, n)
    right_mat = right_perm.view(B_H, n, n)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    out = out_mat.view(b, hidden_dim, n, n).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Second LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Output gate
    out = out * gate_out

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden)
    out_flat = out.view(-1, hidden_dim)  # shape (B*N*N, hidden)
    out = torch.nn.functional.linear(out_flat, to_out_w)  # (B*N*N, dim)
    out = out.view(b, n, n, dim)

    # Cast back to float32
    return out.to(torch.float32)
```

@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        a_ptrs = A_ptr + pid_z * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + pid_z * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi

        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & (offs_k[None, :] < K), other=0.0, cache_modifier='ca')
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_j[None, :] < N), other=0.0, cache_modifier='ca')

        acc += tl.dot(a, b)

    c_ptrs = C_ptr + pid_z * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – fast Triton‑accelerated implementation
==========================================================

Key ideas
---------

1️⃣ **Layer‑norm** on the input is performed in FP32 for stability.
2️⃣ **Fuse the four linear projections** (left_proj, left_gate, right_proj,
   right_gate) **and the output gate** into a *single* matmul by stacking the
   five weight matrices along the output dimension.  This reduces the number
   of large matmuls from six to four and halves the number of reads of the
   input tensor.
3️⃣ **Masking** is applied only to the two hidden‑projections (left/right)
   before the gates, exactly as in the reference model.
4️⃣ The heavy O(N³) contraction
   ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``
   is realised as a *batched* matrix multiplication
   ``C = A @ Bᵀ`` where ``A`` and ``B`` have shape
   ``(B·H, N, N)``.  A custom Triton kernel tiles the multiplication with
   64×64×64 blocks, loads tiles in FP16, accumulates in FP32 via
   ``tl.dot`` (Tensor‑core MMA), and stores the result back in FP16.
5️⃣ A second LayerNorm (hidden dimension) and the output‑gate are applied,
   followed by the final projection back to the original feature dimension.
6️⃣ All heavy linear work is done in half‑precision; the final result is
   returned in FP32 as required.

The implementation works for all configurations listed in the statement
and satisfies the “must use Triton” requirement (the contraction kernel is
implemented in Triton).  It delivers a sizeable speed‑up over the naïve
reference implementation by fusing the five linear layers and by using an
optimised batched‑gemm kernel.  The code is self‑contained and can be
called as `custom_kernel((x, mask, weights, config))`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A,B: [M,N,K], C: [M,N,N])
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K] (will be accessed transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication for the TriMul contraction.

    Each program (identified by pid_z) processes one batch‑hidden slice
    (i.e. one value of M = batch * hidden_dim).  The two N‑axes are tiled
    with blocks of size BLOCK_M × BLOCK_N.  Accumulation is performed in
    FP32 using Tensor‑core ``tl.dot``.
    """
    pid_i = tl.program_id(0)   # tile along first N axis (i)
    pid_j = tl.program_id(1)   # tile along second N axis (j)
    pid_z = tl.program_id(2)   # index over M (batch × hidden)

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles in the K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # -------------------------------------------------
    # Loop over K dimension
    # -------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile  (M‑idx, i‑idx, k‑idx)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
            cache_modifier='ca',
        )   # fp16

        # Load B tile (M‑idx, k‑idx, j‑idx) – transposed view of B
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
            cache_modifier='ca',
        )   # fp16

        # Multiply‑accumulate using Tensor‑core MMA (fp16→fp32)
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Store the result tile (cast back to fp16)
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that packs tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Arguments
    ---------
    left, right : torch.Tensor
        Shape ``[B, N, N, H]`` in FP16.

    Returns
    -------
    torch.Tensor
        Shape ``[B, N, N, H]`` in FP16.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension:
    #   (B, N, N, H) -> (B·H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides are expressed in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (M,N,K)
    stride_bm, stride_bi, stride_bk = B_.stride()     # (M,N,K)
    stride_cm, stride_ci, stride_cj = C.stride()      # (M,N,N)

    # Block sizes – 64³ is a sweet spot on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # M‑tiles (batch * hidden)
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 / bool) – may be unused
        - weights_dict  : mapping name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``;
                          optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the rest of the forward pass
    # ------------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections (left/right + gates + out_gate)
    # ------------------------------------------------------------------
    # Stack weights along the output dimension:   Lproj | Lgate | Rproj | Rgate | OutGate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*hidden_dim, dim)

    B, N, _, _ = input_tensor.shape

    # Flatten the input for a single large matmul
    x_flat = x_fp16.view(-1, dim)           # (B·N·N, dim)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (B·N·N, 5*H)
    combined = combined.view(B, N, N, 5 * hidden_dim)               # (B,N,N,5H)

    # ------------------------------------------------------------------
    # 4️⃣ Split the concatenated result and apply sigmoids
    # ------------------------------------------------------------------
    H = hidden_dim
    proj_left   = combined[..., :H]                     # (B,N,N,H)
    gate_left   = torch.sigmoid(combined[..., H:2*H])   # (B,N,N,H)
    proj_right  = combined[..., 2*H:3*H]                # (B,N,N,H)
    gate_right  = torch.sigmoid(combined[..., 3*H:4*H]) # (B,N,N,H)
    out_gate    = torch.sigmoid(combined[..., 4*H:5*H]) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        proj_left = proj_left * mask_fp16
        proj_right = proj_right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply hidden‑dim gates
    # ------------------------------------------------------------------
    left  = proj_left * gate_left   # (B,N,N,H)
    right = proj_right * gate_right # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via the Triton kernel
    # ------------------------------------------------------------------
    # out has shape (B,N,N,H) in FP16
    out = _batched_matmul_transpose_fp16(left, right)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out_flat = out.view(-1, hidden_dim)                    # (B·N·N, hidden)
    out = torch.nn.functional.linear(out_flat, to_out_w)  # (B·N·N, dim)
    out = out.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Cast to float32 for the expected output type
    # ------------------------------------------------------------------
    return out.to(torch.float32)
