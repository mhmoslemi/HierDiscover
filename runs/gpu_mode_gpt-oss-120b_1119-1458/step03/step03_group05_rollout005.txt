"""
Highly‑optimized Triton implementation of the outgoing TriMul operator
used in AlphaFold‑3 (and related protein‑folding models).

Algorithm
---------
1. Layer‑Norm the 4‑D input (float32 for numerical stability).
2. Cast to FP16 and apply three learned linear projections:
   - left_proj  : (dim → hidden_dim)
   - right_proj : (dim → hidden_dim)
   - left_gate / right_gate / out_gate : (dim → hidden_dim) followed by sigmoid.
3. Optional mask (broadcast over the hidden dimension) is fused into left/right.
4. The core O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched GEMM C = A @ Bᵀ with:
        A = left   reshaped to [B·hidden, N, N]   (i‑k)
        B = right  reshaped and transposed to [B·hidden, N, N]   (k‑j)
   The Triton kernel tiles the matrices with a 2‑level tiling (BLOCK_M×BLOCK_N)
   and a reduction tile (BLOCK_K).  FP16 values are loaded, multiplied, and
   accumulated in FP32 (Tensor‑Core friendly).  Double‑buffering (num_stages=3)
   hides global‑memory latency.
5. A second LayerNorm (hidden_dim) and the out_gate are applied.
6. Final linear projection back to the original `dim`.
7. Output is cast to float32 (as required by the harness).

The kernel works for any batch size (≤2), any sequence length in
{128,256,512,1024}, any `dim` ∈ {128,384} and any `hidden_dim` (commonly 128).
Mask‑free and mask‑aware variants are both supported.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : [M, N, K]   (M = batch*hidden, N = seq_len, K = seq_len)
#   B : [M, K, N]   (right already transposed)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]   (right transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bk, stride_bj,   # strides for B (M,K,N)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1️⃣ Tile indices (program_id) → tile of output C (i‑j plane) and batch×hidden index
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)                 # tile row index i
    pid_n = tl.program_id(1)                 # tile col index j
    pid_z = tl.program_id(2)                 # batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑offsets within the tile
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑offsets within the tile

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2️⃣ Accumulator (FP32 for accuracy, then cast back to fp16)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3️⃣ Loop over reduction dimension K (tiled)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- Load A tile (shape BLOCK_M × BLOCK_K) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- Load B tile (shape BLOCK_K × BLOCK_N) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # K‑dimension stride
            + offs_n[None, :] * stride_bj   # N‑dimension stride (post‑transpose)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )   # fp16

        # ---- FP16×FP16 → FP32 accumulation using Tensor‑Cores ----
        acc += tl.dot(a, b)   # no explicit cast needed

    # ------------------------------------------------------------------
    # 4️⃣ Write out C (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper: reshape tensors, launch the Triton kernel and restore original layout
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a fused Batched GEMM kernel.
    Returns a fp16 tensor of shape [B, N, N, H] (contiguous).
    """
    B, N, _, H = left.shape

    # Pack batch & hidden together → shape [M, N, K] with M = B*H
    # left  : (i‑k)   → [M, N, K]
    # right : (k‑j)   → [M, K, N]  (transposed)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # k‑j layout

    # Output buffer (same dtype)
    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # Choose tiling parameters.
    # For N ≤ 256 we keep 64×64 tiles (less shared memory, higher occupancy);
    # for larger N we switch to 128×128 tiles to increase arithmetic intensity.
    # BLOCK_K stays 64 (a good trade‑off for H100 Tensor‑Cores).
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 128
        num_warps = 16
    BLOCK_K = 64

    # Grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,                 # double‑buffered pipeline
    )

    # Restore layout [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates in‑place to reduce memory traffic
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden_dim) – fp16
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Final gating + projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gate (fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Return in float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

This implementation keeps all mathematically‑intensive work inside a custom Triton kernel,
uses FP16 for memory bandwidth‑limited stages, FP32 accumulation for numerical safety,
and automatically selects larger tile sizes for long sequences to approach the theoretical
hardware peak on an H100. The surrounding Torch code handles LayerNorm and gating, which
are negligible compared to the O(N³) contraction.
