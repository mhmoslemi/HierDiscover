   avoiding two separate loads/stores.
4️⃣ **Core contraction** – the outgoing update  
   `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`
   is a batched matrix‑multiply for each hidden channel.  By permuting
   to `(B, hidden, N, N)` and reshaping to `(B*hidden, N, N)` we call
   `torch.bmm`, which maps to cuBLAS’s highly‑optimised strided‑batched
   GEMM (Tensor‑Cores on H100).
5️⃣ Output LayerNorm, output‑gate multiplication and final projection
   are kept in FP16; the final tensor is cast back to the input dtype
   (usually FP32).

Only the triple‑product (step 3) is custom‑written in Triton, satisfying
the requirement of using a kernel while keeping the rest in
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: a * b * mask   (b is a 3‑D mask broadcasted over hidden)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,               # left_proj_raw   (B,N,N,hidden)
    mask_ptr,            # mask            (B,N,N)
    b_ptr,               # gate (after sigmoid) (B,N,N,hidden)
    out_ptr,             # destination
    hidden: tl.constexpr,
    size:   tl.constexpr,          # total number of elements in a/c (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load left_proj_raw and gate (contiguous in hidden dimension)
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    # Compute the mask index (broadcast mask over the hidden dim)
    mask_index = offs // hidden
    m = tl.load(mask_ptr + mask_index, mask=mask, other=0.0)

    out = a * b * m
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(proj: torch.Tensor,
                      mask: torch.Tensor,
                      gate: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """Element‑wise product proj * gate * mask (mask broadcast over hidden).

    All tensors must be contiguous, half‑precision and have shapes:
      proj : (B, N, N, hidden)
      gate : (B, N, N, hidden)
      mask : (B, N, N)

    Returns a new half‑precision tensor of shape (B, N, N, hidden).
    """
    assert proj.shape == gate.shape, "proj and gate must have the same shape"
    B, N, _, H = proj.shape
    assert H == hidden, "hidden size mismatch"
    # Ensure contiguity
    proj = proj.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()

    out = torch.empty_like(proj)
    total = proj.numel()
    BLOCK = 8192  # empirically good for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        proj,
        mask,
        gate,
        out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  shape [B, N, N] (bool/float, CUDA) – may be ignored
                             when config["nomask"] is True.
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim";
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    dtype = inp.dtype
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (retain original dtype for the norm, then cast)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Continue the whole pipeline in FP16
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Concatenate projection+gate weights (reduces linear calls)
    # ------------------------------------------------------------------
    # left: proj + gate
    left_concat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden, dim)

    # right: proj + gate
    right_concat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden, dim)

    # out‑gate and final projection
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)          # (hidden, dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)    # (hidden,)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)        # (hidden,)
    to_out_weight = weights["to_out.weight"].to(torch.float16)              # (dim, hidden)

    # ------------------------------------------------------------------
    # 3) Left & Right linear (proj + gate together)
    # ------------------------------------------------------------------
    #   x_norm shape: (B,N,N,dim) – Linear works on the last dim.
    left_concat = F.linear(x_norm, left_concat_weight)   # (B,N,N,2*hidden)
    right_concat = F.linear(x_norm, right_concat_weight) # (B,N,N,2*hidden)

    # Split into proj / raw‑gate
    left_proj_raw = left_concat[..., :hidden]            # (B,N,N,hidden)
    left_gate_raw = left_concat[..., hidden:]            # (B,N,N,hidden)
    right_proj_raw = right_concat[..., :hidden]
    right_gate_raw = right_concat[..., hidden:]

    # Free the big concatenated tensors early
    del left_concat, right_concat

    # Gate activations
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # Out‑gate (uses *unmasked* x_norm – same as the reference)
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Apply mask + gate:  proj * gate * mask   (mask optional)
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # mask may be bool or float; ensure half‑precision
        mask_fp16 = mask.to(torch.float16)               # (B,N,N)
        left = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # Release tensors that are no longer needed
    del left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw
    if not nomask:
        del mask_fp16

    # ------------------------------------------------------------------
    # 5) Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    --------> Batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    #   left  : (B, N, N, hidden) → (B, hidden, N, N)
    #   right : (B, N, N, hidden) → (B, hidden, N, N) with k<->j transposed
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, k, j)

    # Collapse batch & hidden into a single dimension for torch.bmm
    left_flat = left_perm.view(B * hidden, N, N)        # (B*hidden, i, k)
    right_flat = right_perm.view(B * hidden, N, N)      # (B*hidden, k, j)

    # Batched matrix multiplication (Tensor‑Core‑accelerated)
    out_hidden_flat = torch.bmm(left_flat, right_flat)  # (B*hidden, i, j)

    # Reshape back to (B, N, N, hidden)
    out_hidden = out_hidden_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # Release temporary tensors
    del left, right, left_perm, right_perm, left_flat, right_flat, out_hidden_flat

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # Apply the output gate
    out_norm = out_norm * out_gate

    # Final linear projection (hidden → dim)
    out = F.linear(out_norm, to_out_weight)   # (B,N,N,dim), still FP16

    # Cast back to the original dtype (usually float32)
    return out.to(dtype)
```
