"""
TriMul (outgoing) – fast Triton implementation
================================================

The TriMul operator (AlphaFold‑3) updates a pairwise representation
`x ∈ ℝ^{B×N×N×C}` by a “gate‑modulated outer‑product’’ over the
sequence dimension:

    out[b,i,j,:] = Σ_k ( (x·W_left)·σ(x·W_left_gate) )_{b,i,k,:}
                   ⊙
                   ( (x·W_right)·σ(x·W_right_gate) )_{b,j,k,:}

followed by a hidden‑dim layer‑norm, a second sigmoid gate
and a final linear projection back to `C` dimensions.

The original PyTorch implementation performs **six** independent
`nn.Linear` calls (four for projections/gates, one for the final
projection and one more for the output gate) and a naïve
`einsum` for the cubic contraction.  The heavy part is therefore
memory‑bound and the launch overhead of six GEMM kernels dominates
the runtime for the modest sequence lengths used in the tests.

This implementation reduces the number of GEMM launches to **three**:

1. **Projection & projection‑gate fusion** – one GEMM that produces
   `left_proj`, `right_proj`, `left_gate`, `right_gate` and `out_gate`
   in a single matrix‑multiply (`torch.nn.functional.linear` on a
   concatenated weight matrix).  The three sigmoid gates are applied
   afterwards.

2. **Out‑gate+final projection** – a second GEMM for the
   `out_gate` (already fused above) and a third GEMM for the final
   linear projection `to_out`.

3. **Cubic contraction** – an explicit Triton kernel that computes a
   batched matrix‑multiply `C = A @ Bᵀ` for each hidden channel.
   The kernel works on half‑precision tensors, accumulates in FP32
   (to avoid overflow for long reductions) and uses a 128 × 128 × 64
   tile with 8 warps and double‑buffering (`num_stages=2`).

All cheap operations (layer‑norms, masking, element‑wise mul) stay in
FP16 to keep the memory traffic low.  The final result is cast back to
`float32` as required by the evaluation harness.

The function `custom_kernel` follows the exact signature used by the
harness:

    custom_kernel((input, mask, weights, config)) → output

where:
* `input` – `[B, N, N, C]` tensor (float32)
* `mask`  – `[B, N, N]` tensor (float32 or bool) – may be omitted
* `weights` – dict of all model parameters (float32)
* `config` – contains `dim`, `hidden_dim` and optionally `nomask`

"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]   (fp16)   where M = B*H, N = seq_len, K = seq_len
#   C    : [M, N, N]   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32
    stride_am, stride_ai, stride_ak,   # strides of A (elements)
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile index for i (row)
    pid_j = tl.program_id(1)   # tile index for j (col)
    pid_z = tl.program_id(2)   # batch*hidden index

    # Offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Accumulate in FP32 (a·b)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # Write back FP16 result
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H] (fp16)
    right: torch.Tensor,  # [B, N, N, H] (fp16)
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension.
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # Strides expressed in element counts
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,               # dummy M (unused)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device
        as the input.
    """
    # -----------------------------------------------------------------
    # 1️⃣ unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32 – higher precision for mean/var)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – float32

    # -----------------------------------------------------------------
    # 3️⃣ cast to fp16 for the heavy work
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)   # shape [B,N,N,dim]

    # -----------------------------------------------------------------
    # 4️⃣ fuse all linear projections & gates into two GEMMs
    # -----------------------------------------------------------------
    # concat: [left_proj; right_proj] → shape (2*hidden, dim)
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)

    # concat: [left_gate; right_gate; out_gate] → shape (3*hidden, dim)
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # ---- Projection (left & right) ------------------------------------
    proj_out = torch.nn.functional.linear(x, proj_weight)   # [B,N,N,2*hidden]
    left = proj_out[..., :hidden_dim]                      # [B,N,N,hidden]
    right = proj_out[..., hidden_dim : 2 * hidden_dim]

    # ---- Gates ---------------------------------------------------------
    gate_out = torch.nn.functional.linear(x, gate_weight)   # [B,N,N,3*hidden]
    left_gate = torch.sigmoid(gate_out[..., :hidden_dim])
    right_gate = torch.sigmoid(gate_out[..., hidden_dim : 2 * hidden_dim])
    out_gate = torch.sigmoid(gate_out[..., 2 * hidden_dim :])

    # -----------------------------------------------------------------
    # 5️⃣ optional mask (applied after projection, before gating)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left.mul_(mask_fp16)
        right = right.mul_(mask_fp16)

    # -----------------------------------------------------------------
    # 6️⃣ apply hidden‑dim gates (in‑place to avoid extra allocations)
    # -----------------------------------------------------------------
    left = left.mul_(left_gate)   # now contains the gated left tensor
    right = right.mul_(right_gate)

    # -----------------------------------------------------------------
    # 7️⃣ heavy cubic contraction (batched matmul, fp16 in/out)
    # -----------------------------------------------------------------
    # out shape: [B, N, N, hidden]
    out = _batched_matmul_transpose_fp16(left, right)

    # -----------------------------------------------------------------
    # 8️⃣ second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ final gating + projection back to `dim`
    # -----------------------------------------------------------------
    out = out.mul_(out_gate)   # shape stays [B,N,N,hidden]
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden]
    out = torch.nn.functional.linear(out, to_out_w)        # -> [B,N,N,dim]

    # -----------------------------------------------------------------
    # 10️⃣ cast back to float32 (required by harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
