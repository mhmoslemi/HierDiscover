"""
high‑performance outgoing TriMul (AlphaFold‑3) forward pass

Algorithm
---------
1.  Layer‑norm the input (float32) → fp16.
2.  Fuse the five linear projections (left/right + 3 gates) into a *single*
    matrix‑multiply (batched GEMM) to read the 4‑D tensor only once.
    The concatenated weight has shape (5·hidden_dim, dim) and the result is
    split into the five tensors required downstream.
3.  Apply the optional mask *before* the gates (exactly as in the reference).
4.  Fuse each gate by an in‑place element‑wise multiply.
5.  The O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is a batched matrix‑multiply C = A @ Bᵀ with
        A = left.permute(0,3,1,2).view(B·H, N, N)
        B = right.permute(0,3,1,2).view(B·H, N, N)
    It is performed by a custom Triton kernel that uses 64×64 tiles,
    128‑wide K‑tiles, 16 warps and 3 stages.  The kernel works entirely in
    fp16 and accumulates in fp32 for numerical stability.
6.  Second LayerNorm (fp16) → out‑gate (sigmoid) → final linear projection.
7.  Cast the result back to float32 (required by the harness).

The kernel covers the heavy contraction; all other ops use highly‑optimized
PyTorch primitives.  The whole pipeline does **no** gradient computation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : fp16 [M, N, K]   (M = B*H, N = seq_len, K = seq_len)
#   C    : fp16 [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute a fused batched matmul C = A @ B.T."""
    pid_m = tl.program_id(0)   # tile along output row (i)
    pid_n = tl.program_id(1)   # tile along output col (j)
    pid_z = tl.program_id(2)   # batch * hidden index (M)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Main K loop
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile  (M‑tile × K‑tile)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile (K‑tile × N‑tile) – note the transposition
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result tile in fp16
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.
    Arguments
    ---------
    left, right : torch.Tensor   shape [B, N, N, H] (contiguous, fp16)
    Returns
    -------
    out : torch.Tensor   shape [B, N, N, H] (contiguous, fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] (row‑major layout)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # stride in elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # tile configuration – 64×64 tiles, 128‑wide K‑tiles
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,      # more warps → higher occupancy on H100
        num_stages=3,      # double‑buffered shared memory
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  float32
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    total_pairs = B * N * N

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear transformations
    # ------------------------------------------------------------------
    # Concatenate weights: order = left_proj, right_proj, left_gate, right_gate, out_gate
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # stack on out_features axis
    ).to(torch.float16)   # shape [(5*hidden_dim), dim]

    # Flatten input to 2‑D for a single GEMM
    x_flat = x.view(total_pairs, dim)                # [B*N*N, dim]
    y = torch.nn.functional.linear(x_flat, w_concat)  # [B*N*N, 5*hidden_dim]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split the fused output
    # ------------------------------------------------------------------
    # each chunk has shape [B*N*N, hidden_dim]
    h = hidden_dim
    left_proj   = y[:, 0 * h : 1 * h]
    right_proj  = y[:, 1 * h : 2 * h]
    left_gate   = y[:, 2 * h : 3 * h]
    right_gate  = y[:, 3 * h : 4 * h]
    out_gate    = y[:, 4 * h : 5 * h]

    # reshape back to 4‑D tensors
    left_proj   = left_proj.view(B, N, N, h).contiguous()
    right_proj  = right_proj.view(B, N, N, h).contiguous()
    left_gate   = left_gate.view(B, N, N, h).contiguous()
    right_gate  = right_gate.view(B, N, N, h).contiguous()
    out_gate    = out_gate.view(B, N, N, h).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [B, N, N, 1] (broadcast)
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and fusion
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # Fuse gates in‑place to save memory
    torch.mul(left_proj, left_gate, out=left_proj)
    torch.mul(right_proj, right_gate, out=right_proj)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16)
    # ------------------------------------------------------------------
    # left_proj & right_proj are already contiguous and fp16
    out = _batched_matmul_transpose_fp16(left_proj, right_proj)   # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + linear projection (fp16)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate)
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (as required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
