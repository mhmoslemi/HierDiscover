```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def apply_gates_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.int64,
    mask_numel: tl.int64,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # Load values
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index
    mask_idx = offs // hidden_dim
    mask_valid = mask_idx < mask_numel
    m_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    # Apply gates and mask
    tl.store(left_ptr + offs, l * lg * m_val, mask=mask)
    tl.store(right_ptr + offs, r * rg * m_val, mask=mask)

@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)

@triton.jit
def mul_inplace_kernel(
    a_ptr, b_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)

"""
Highly‑optimized forward pass for the “outgoing” TriMul operator (AlphaFold3).
Key ideas
* Fuse the five linear transforms (left_proj, right_proj, left_gate,
  right_gate, out_gate) into a single matrix multiplication to avoid
  repeatedly loading the input tensor.
* Fuse the element‑wise mask, left‑gate and right‑gate multiplication into
  a tiny Triton kernel (apply_gates_mask_kernel).  When no mask is present
  a simpler kernel (apply_gates_kernel) is used.
* The dominant N³ operation is performed with a single batched GEMM
  (torch.bmm) which maps to cuBLAS Tensor‑Core kernels on an H100.
* Fuse the final out‑gate multiplication with a second small Triton kernel
  (mul_inplace_kernel).
* All heavy arithmetic is in FP16; LayerNorms use FP32 statistics for
  stability, the final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,                # fp16*   (flattened mask of shape B·N·N)
    hidden_dim: tl.int64,    # size of the channel dimension
    mask_numel: tl.int64,    # = B·N·N
    numel: tl.int64,        # total elements = B·N·N·hidden_dim
    BLOCK: tl.constexpr,
):
    """
    In‑place fusion of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    mask is broadcast over the channel dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < numel

    # load tensors
    l = tl.load(left_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # mask index = element index // hidden_dim  (broadcasted)
    mask_idx = offs // hidden_dim
    mask_valid = mask_idx < mask_numel
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)   # 1.0 = no‑mask

    # write gated & masked results back in‑place
    tl.store(left_ptr + offs,  l * lg * m, mask=valid)
    tl.store(right_ptr + offs, r * rg * m, mask=valid)


@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place fusion of:
        left  = left  * left_gate
        right = right * right_gate
    Used when no mask is supplied.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(left_ptr + offs,  l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr, b_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        a = a * b
    Used for the post‑norm out‑gate.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask         : torch.Tensor      [B, N, N]      (bool/float, may be None)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   must contain 'dim', 'hidden_dim' and may contain
                           'nomask' (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣  Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → no mask (all‑ones)
    eps = 1e-5

    # --------------------------------------------------------------
    # 2️⃣  Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  fp32

    # --------------------------------------------------------------
    # 3️⃣  Cast to FP16 for all subsequent heavy arithmetic
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)   # [B, N, N, dim]  fp16

    # --------------------------------------------------------------
    # 4️⃣  Fuse the five linear projections into a single GEMM.
    #    The concatenated weight has shape (5*hidden_dim, dim).
    # --------------------------------------------------------------
    w_combined = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # (5*hidden_dim, dim)

    # Single linear pass
    combined = F.linear(x_h, w_combined)          # [B, N, N, 5*hidden_dim]  fp16

    # --------------------------------------------------------------
    # 5️⃣  Slice the concatenated tensor into its components
    # --------------------------------------------------------------
    # NB: slicing returns views – no extra memory copy.
    h = hidden_dim
    left      = combined[..., 0:h]
    right     = combined[..., h:2 * h]
    left_gate = combined[..., 2 * h:3 * h]
    right_gate= combined[..., 3 * h:4 * h]
    out_gate  = combined[..., 4 * h:5 * h]

    # --------------------------------------------------------------
    # 6️⃣  Apply sigmoid to gate tensors (in‑place for speed)
    # --------------------------------------------------------------
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()

    # --------------------------------------------------------------
    # 7️⃣  Fuse left/right gating (and optional mask) via Triton
    # --------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()                     # B·N·N·hidden_dim
    BLOCK = 65536                            # ~64 k elements per program
    grid = ((numel + BLOCK - 1) // BLOCK,)

    if not nomask:
        # mask: [B, N, N] → broadcast over channel dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        mask_flat = mask_h.view(-1)                     # [B·N·N]
        mask_numel = mask_flat.numel()
        apply_gates_mask_kernel[grid](
            left_ptr=left.view(-1),
            left_gate_ptr=left_gate.view(-1),
            right_ptr=right.view(-1),
            right_gate_ptr=right_gate.view(-1),
            mask_ptr=mask_flat,
            hidden_dim=hidden_dim,
            mask_numel=mask_numel,
            numel=numel,
            BLOCK=BLOCK,
        )
    else:
        apply_gates_kernel[grid](
            left_ptr=left.view(-1),
            left_gate_ptr=left_gate.view(-1),
            right_ptr=right.view(-1),
            right_gate_ptr=right_gate.view(-1),
            numel=numel,
            BLOCK=BLOCK,
        )
    # Gate tensors are no longer needed
    del left_gate, right_gate

    # --------------------------------------------------------------
    # 8️⃣  Core N³ operation: batched matrix multiplication.
    #    Each channel (hidden_dim) is treated as an independent head.
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # reshape to (total_heads, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # (total_heads, N, N) @ (total_heads, N, N)^T → (total_heads, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original layout: [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # fp16

    # --------------------------------------------------------------
    # 9️⃣  Post‑processing LayerNorm (half‑precision weights)
    # --------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )   # fp16

    # --------------------------------------------------------------
    # 🔟  Multiply by the output gate (in‑place) via Triton
    # --------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        a_ptr=out.view(-1),
        b_ptr=out_gate.view(-1),
        numel=numel_out,
        BLOCK=BLOCK,
    )
    del out_gate         # free memory

    # --------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = F.linear(out, w_to_out)   # [B, N, N, dim]  fp16

    # Return in the expected dtype (float32)
    return final.to(torch.float32)
