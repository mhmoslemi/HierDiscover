"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated implementation
----------------------------------------------------------------
The module computes for each hidden channel `h`:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left`/`right` are linear projections of the input tensor, multiplied by
per‑position gates and an optional mask.  The heavy O(N³) contraction is a
batched matrix multiplication `C = A @ Bᵀ` for each (batch, hidden) pair.
We fuse this contraction into a single Triton kernel:

* The input is layer‑normed in FP32 and cast to FP16.
* All linear projections and sigmoid‑gates are performed in FP16.
* The sequence dimension `N` is padded to a multiple of the tile size (64) so
  the Triton kernel can run without any out‑of‑bounds checks.
* The kernel processes tiles of size 64 × 64 with a K‑tile of 64,
  using FP16 matrix‐multiply‑accumulate (MMA) and FP32 accumulation.
* After the contraction we apply a second LayerNorm, the output gate,
  and the final linear projection back to the original `dim`.
* The result is returned in FP32.

The implementation is fully self‑contained and works for all shapes listed
in the test suite (batch size 1‑2, `N` up to 1024, `dim` 128/384,
`hidden_dim` 128).  No gradients are needed – the forward pass is
gradient‑free.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A: [M, N, K]  (M = B * hidden_dim)
#   B: [M, N, K]  (same shape as A)
#   C: [M, N, N]
# All tensors are fp16; accumulation is fp32.
# -------------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)   # tile along first N dim (i)
    pid_j = tl.program_id(1)   # tile along second N dim (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # compute tile offsets (no bounds checks – N is already padded)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # -------------------------------------------------
        # Load A[i, k]   shape (BLOCK_M, BLOCK_K)
        # -------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs)          # fp16, no mask needed (padded)

        # -------------------------------------------------
        # Load Bᵀ[k, j] = B[k, j]   shape (BLOCK_K, BLOCK_N)
        # -------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs)          # fp16, no mask needed (padded)

        # accumulate dot product (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # store result (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16))


def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor,
                                   tile: int = 64) -> torch.Tensor:
    """
    Compute `out = left @ rightᵀ` for each (batch, hidden) pair using Triton.
    Args:
        left, right : fp16 tensors of shape [B, N, N, H] (already padded)
        tile        : tile size for i/j/k dimensions (must divide padded N)
    Returns:
        out : fp16 tensor of shape [B, N, N, H] (still padded)
    """
    B, N, _, H = left.shape

    # pack batch and hidden dimensions => [B*H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A)

    # strides (in element units)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,          # M (not used inside kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,          # more warps on H100 for higher occupancy
        num_stages=4,          # 3‑stage pipeline + one pre‑fetch
    )

    # reshape back to original layout [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]      (float32, bool or None)
        - weights_dict : {name: tensor} (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(input_tensor,
                          (dim,),
                          weight=norm_weight,
                          bias=norm_bias)          # [B, N, N, dim] FP32
    x = x_norm.to(torch.float16)                       # switch to fp16 for heavy ops

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Merged linear projections + gates (FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    # Merge projection and its gate for left and right side to cut linear passes
    # Left side: [hidden, dim] + [hidden, dim] → (2*hidden, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    left_combined = F.linear(x_flat, left_combined_w).view(B, N, N, 2 * hidden_dim)
    right_combined = F.linear(x_flat, right_combined_w).view(B, N, N, 2 * hidden_dim)

    left = left_combined[..., :hidden_dim]                       # proj
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])  # gate
    right = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # out_gate is separate
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension) and gating
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Pad sequence dimension to TILE (64) – eliminates bounds checks
    # ------------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    N_pad = N + pad_len
    if pad_len != 0:
        left_pad = left.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (FP16 inputs, FP16 output)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, hidden_dim]

    # Remove padding
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + final linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                     # gating
    out = F.linear(out, to_out_w)            # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 – this is what the surrounding model expects
    # ------------------------------------------------------------------
    return out.to(torch.float32)
