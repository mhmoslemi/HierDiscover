def _trimul_matmul_kernel_no_copy(
    A_ptr, B_ptr, C_ptr,
    B_val, N, H,
    stride_batch_a, stride_i_a, stride_k_a, stride_h_a,
    stride_batch_b, stride_k_b, stride_j_b, stride_h_b,
    stride_batch_c, stride_i_c, stride_j_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # compute batch and hidden indices
    b = pid_z // H
    h = pid_z - b * H

    # base offsets
    a_base = A_ptr + b * stride_batch_a + h * stride_h_a
    b_base = B_ptr + b * stride_batch_b + h * stride_h_b
    c_base = C_ptr + b * stride_batch_c + h

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        # compute masks for k
        mask_k = offs_k < N

        # load A tile

"""
Optimised outgoing TriMul forward pass.

The heavy O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed with a custom Triton kernel that works directly on the
original layout [B, N, N, H] (no expensive permute‑and‑contiguous
re‑ordering).  The kernel computes a batched GEMM C = A @ Bᵀ where the
batch dimension is B·H.  Each program handles a tile (BLOCK_M×BLOCK_N) of
the output for a single hidden slice (identified by a combined
(batch, hidden) ID).  FP16 operands are accumulated in FP32 for
numerical stability and the result is written back as FP16.
All remaining ops (LayerNorms, Linear projections, Sigmoids) are kept in
PyTorch and run in FP16 as well.  The final tensor is returned in FP32.

Key improvements over the reference implementation:
  * Avoids the two large memory copies caused by
    `permute(...).contiguous()` before the GEMM.
  * Uses a 64 × 64 × 64 tiling with 3‑stage double‑buffered pipeline
    (`num_stages=3`, `num_warps=8`) that maps well to H100 tensor‑cores.
  * All heavy arithmetic stays in FP16 → FP32, maximising bandwidth while
    preserving enough precision for the downstream LayerNorm.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   (batched GEMM)
#   left  : [B, N, N, H]   (FP16)
#   right : [B, N, N, H]   (FP16) – accessed as (k, j) tile
#   out   : [B, N, N, H]   (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  left  tensor
    B_ptr,               # *fp16  right tensor
    C_ptr,               # *fp16  output tensor
    N, H,                # int32 : sequence length, hidden dimension
    # strides for A (left)
    stride_batch_a, stride_i_a, stride_k_a, stride_h_a,
    # strides for B (right, k‑j layout)
    stride_batch_b, stride_k_b, stride_j_b, stride_h_b,
    # strides for C (output)
    stride_batch_c, stride_i_c, stride_j_c,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    # hidden dimension as compile‑time constant (enables cheap division)
    H_VAL: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ with A,B ∈ℝ^{B·H × N × N}."""
    pid_i = tl.program_id(0)      # i‑tile
    pid_j = tl.program_id(1)      # j‑tile
    pid_z = tl.program_id(2)      # (batch, hidden) index

    # ---------------------------------------------------------------
    # decode (batch, hidden) from pid_z
    # ---------------------------------------------------------------
    b = pid_z // H_VAL            # batch index
    h = pid_z - b * H_VAL          # hidden index (0 ≤ h < H)

    # base pointer for this (b, h) slice
    a_base = A_ptr + b * stride_batch_a + h * stride_h_a
    b_base = B_ptr + b * stride_batch_b + h * stride_h_b
    c_base = C_ptr + b * stride_batch_c + h          # hidden stride is 1

    # ---------------------------------------------------------------
    # tile offsets
    # ---------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # reduction over k
    # ---------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # ---- load A tile (i,k) ------------------------------------------------
        a_ptrs = a_base + offs_i[:, None] * stride_i_a + offs_k[None, :] * stride_k_a
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ---- load B tile (k,j) ------------------------------------------------
        # B is accessed as (k, j) => stride_k_b for rows, stride_j_b for cols
        b_ptrs = b_base + offs_k[:, None] * stride_k_b + offs_j[None, :] * stride_j_b
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ---- FP16×FP16 → FP32 accumulation -----------------------------------
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # write result back (store as fp16)
    # ---------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_i_c + offs_j[None, :] * stride_j_c
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel and returns the contracted tensor
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Computes the O(N³) TriMul contraction.
    Both ``left`` and ``right`` must be FP16 tensors of shape [B, N, N, H].
    Returns a FP16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    # allocate output
    out = torch.empty_like(left, dtype=torch.float16)

    # strides (in element counts, not bytes)
    stride_batch_a, stride_i_a, stride_k_a, stride_h_a = left.stride()
    stride_batch_b, stride_i_b, stride_k_b, stride_h_b = right.stride()
    stride_batch_c, stride_i_c, stride_k_c, stride_h_c = out.stride()

    # For the right tensor we need the stride for the *j* dimension (the
    # first N after the transpose).  In the original layout this is the
    # stride of dimension 1.
    stride_j_b = stride_i_b

    # Tile sizes – chosen to fit H100 tensor‑core lanes while keeping the
    # register footprint modest.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        N,
        H,
        stride_batch_a,
        stride_i_a,
        stride_k_a,
        stride_h_a,
        stride_batch_b,
        stride_k_b,
        stride_j_b,
        stride_h_b,
        stride_batch_c,
        stride_i_c,
        stride_j_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H_VAL=H,
        num_warps=8,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse the gates (in‑place keeps memory traffic low)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via the batched GEMM kernel (no extra copies)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gate + projection back to original dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
