def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # Cast weights to fp16
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    # Combine projection weights
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    # concatenate along output dim
    combined_proj_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Cast input to fp16
    x = input_tensor.to(torch.float16)

    # LayerNorm on fp16
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # Combined projection
    # x shape [B,N,N,dim]; combined weight shape [4*hidden, dim]
    # using torch.nn.functional.linear which expects weight shape [out_features, in_features]
    proj = torch.nn.functional.linear(x, combined_proj_w)  # shape [B,N,N,4*hidden]

    # Split
    H = hidden_dim
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw = torch.split(proj, H, dim=-1)

    # Gating
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    # apply mask if required
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # Fuse gates
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # Compute out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Heavy contraction: left and right shape (B,N,N,H)
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)  # B,N,N,H

    # Second layer norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate then to_out linear
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    return out.to(torch.float32)

"""
Outgoing TriMul – forward pass in fp16 with a fused projection
and a custom Triton GEMM kernel.  The heavy contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is performed as a batched matmul C = A @ Bᵀ where
A = left, B = right, both reshaped to [B·H, N, N].
The kernel uses 128×128×64 tiles, mixed‑precision Tensor‑Core
dot‑product (fp16·fp16→fp32 accumulator) and 8 warps.
Linear projections for left/right are fused into a single
matmul (four projections concatenated) to cut memory traffic.
All operations except the final cast to float32 run in fp16,
maximising bandwidth and compute on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # tile over output row (i)
    pid_j = tl.program_id(1)  # tile over output col (j)
    pid_z = tl.program_id(2)  # merged batch*hidden index (M)

    # Offsets inside the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k_base in range(0, K, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptr = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptr)  # fp16

        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptr = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptr)  # fp16

        # Mixed‑precision tensor‑core dot
        acc += tl.dot(a, b)

    # Store C tile back in fp16
    c_ptr = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptr, acc.to(tl.float16))


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out = left @ rightᵀ using the Triton kernel above.
    left/right: [B, N, N, H] (fp16, contiguous)
    Returns: [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # Pack batch and hidden channels into a single leading dimension
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – 128×128×64 is a sweet spot on H100 for fp16 MMA
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Cast everything to fp16 once – this halves memory traffic
    # -----------------------------------------------------------------
    # Layer‑norm parameters
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    # Projection / gating weights – fuse 4 linear maps into one
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    combined_proj_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    ).to(torch.float16)  # shape [4*hidden, dim]

    # Out‑gate weight
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Post‑contraction weight‑norm and projection
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 1️⃣ Input layer‑norm (still in fp16 for bandwidth)
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2️⃣ Fused projection + gating
    #    proj = [left_proj, left_gate, right_proj, right_gate] (concatenated)
    # -----------------------------------------------------------------
    proj = torch.nn.functional.linear(x, combined_proj_w)  # [B, N, N, 4*hidden]
    # Split into four tensors of shape [B, N, N, hidden]
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )

    # Optional mask (broadcast on hidden dim)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # Gating (sigmoid) and fuse
    left = left_proj_raw * torch.sigmoid(left_gate_raw)
    right = right_proj_raw * torch.sigmoid(right_gate_raw)

    # -----------------------------------------------------------------
    # 3️⃣ Out‑gate (same shape as hidden)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4️⃣ O(N³) contraction – batch GEMM with custom Triton kernel
    # -----------------------------------------------------------------
    # Ensure contiguous layout before reshaping for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second layer‑norm, final gating, and linear projection back to dim
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 6️⃣ Cast back to float32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul – fast forward‑only implementation (AlphaFold‑3 style)

Algorithm
---------
1. Cast the input from FP32 → FP16 and apply a per‑channel LayerNorm.
2. Fuse the four linear projections (left/right and their gates) into a *single*
   matmul by concatenating the weights (shape [4·hidden, dim]).  After the
   matmul we split the result into the four tensors and apply the sigmoid
   gates, optionally masking the left/right projections.
3. Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM C = A @ Bᵀ with a custom Triton kernel.  The data are
   reshaped to [B·hidden, N, N] and the kernel uses 128×128×64 tiles,
   mixed‑precision Tensor‑Core dot‑product (fp16·fp16 → fp32 accumulator) and
   8 warps.  No masks are needed inside the kernel because the test sizes are
   multiples of the tile sizes.
4. Second LayerNorm on the hidden dimension, apply the out‑gate (sigmoid) and
   a final linear projection back to `dim`.
5. Cast the result back to FP32 (the harness expectation).

All heavy arithmetic runs in FP16, halving memory traffic while keeping
numerical stability via FP32 accumulation in the GEMM.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ (fp16 in, fp16 out, fp32 acc)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # compile‑time scalars
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile index along i (first N)
    pid_j = tl.program_id(1)          # tile index along j (second N)
    pid_z = tl.program_id(2)          # merged batch*hidden index (M)

    # Offsets inside the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator (FP32 for stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles (K is known at compile time)
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
        )  # fp16

        # B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj,
        )  # fp16

        # Mixed‑precision dot product
        acc += tl.dot(a, b)

    # Store FP16 result
    tl.store(
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj,
        acc.to(tl.float16),
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ using the Triton kernel above.
    Arguments
    ---------
    left, right : torch.Tensor [B, N, N, H] (fp16, contiguous)
    Returns
    -------
    out : torch.Tensor [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Pack batch and hidden into one leading dimension: (B·H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes tuned for H100 fp16 Tensor‑Core MMA
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch×hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,   # double‑buffered pipeline
    )

    # Reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # Cast all model weights to FP16 (once)
    # -------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    # Fuse the four linear projections (left/right and their gates)
    combined_proj_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # shape [4·hidden, dim]

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 1️⃣ Input layer‑norm (run in FP16 for bandwidth)
    # -------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # -------------------------------------------------
    # 2️⃣ Fused projection + gating
    #    proj shape → [B, N, N, 4·hidden]
    # -------------------------------------------------
    proj = torch.nn.functional.linear(x, combined_proj_w)  # fp16

    # Split into the four needed tensors (views, no copy)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )

    # Optional mask (broadcast on hidden dim)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # Apply sigmoid gates and fuse
    left = left_proj_raw * torch.sigmoid(left_gate_raw)
    right = right_proj_raw * torch.sigmoid(right_gate_raw)

    # -------------------------------------------------
    # 3️⃣ Compute out‑gate (same hidden shape)
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via Triton (fp16 in/out)
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm, final gate, and projection back to dim
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 6️⃣ Cast back to FP32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
