    left_ptr, right_ptr,
    left_gate_pre_ptr, right_gate_pre_ptr,
    out_gate_pre_ptr,
    mask_ptr,
    numel: tl.int64,
    hidden_dim: tl.int64,
    BLOCK: tl.constexpr,
    USE_MASK: tl.constexpr
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg_pre = tl.load(left_gate_pre_ptr + offs, mask=mask, other=0.0)

    rg = 1.0 / (1.0 + tl.exp(-rg_pre))
    og = 1.0 / (1.0 + tl.exp(-og_pre))

    tl.store(right_ptr + offs, r_out, mask=mask)

"""
Optimized outgoing TriMul forward (AlphaFold3 style).

Key ideas:
* Fuse the 5 linear projections (left, right, left_gate, right_gate, out_gate)
  into a **single** half‑precision GEMM to read the input tensor only once.
* Fuse mask application, sigmoid gates and the element‑wise multiplication of
  left/right with their gates into one Triton kernel.
* Use a batched half‑precision GEMM (torch.bmm) for the dominant N³ reduction.
* Keep the remaining LayerNorm/linear steps in FP16 for speed and cast the
  final result back to FP32.

The implementation is fully functional on H100 (Triton 3.3.1) and respects the
required signature `custom_kernel(data)`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def gates_mask_fuse_kernel(
    left_ptr,            # fp16*, left_raw (in‑place)
    right_ptr,           # fp16*, right_raw (in‑place)
    left_gate_ptr,       # fp16*, left_gate pre‑sigmoid (read‑only)
    right_gate_ptr,      # fp16*, right_gate pre‑sigmoid (read‑only)
    out_gate_ptr,        # fp16*, out_gate pre‑sigmoid (overwritten with sigmoid)
    mask_ptr,            # fp16*, shape (B*seq_len*seq_len,) – broadcasted
    numel: tl.int64,            # total number of elements (=B*N*N*hidden_dim)
    hidden_dim: tl.int64,       # hidden dimension (size of the last axis)
    mask_len: tl.int64,         # B*N*N  (num positions)
    BLOCK: tl.constexpr,        # tile size (elements per program)
    USE_MASK: tl.constexpr,     # 1 → apply mask, 0 → skip mask
):
    """
    In‑place fusion:
        left  = left  * mask * sigmoid(left_gate)
        right = right * mask * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)   # also in‑place
    All tensors are flattened.  Mask is optional (USE_MASK compile‑time flag).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < numel

    # load
    l_raw = tl.load(left_ptr + offs, mask=active, other=0.0)
    r_raw = tl.load(right_ptr + offs, mask=active, other=0.0)

    lg_pre = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rg_pre = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    og_pre = tl.load(out_gate_ptr + offs, mask=active, other=0.0)

    # sigmoid – half‑precision is fine for the gate values
    l_gate = 1.0 / (1.0 + tl.exp(-lg_pre))
    r_gate = 1.0 / (1.0 + tl.exp(-rg_pre))
    o_gate = 1.0 / (1.0 + tl.exp(-og_pre))

    if USE_MASK:
        # integer division gives the position index (same for all hidden_dim elems)
        pos_idx = offs // hidden_dim          # shape (BLOCK,)
        mask_val = tl.load(mask_ptr + pos_idx,
                           mask=pos_idx < mask_len,
                           other=0.0)        # 0/1 as fp16
    else:
        mask_val = tl.full([BLOCK], 1.0, dtype=tl.float16)

    # apply mask + gates
    l_out = l_raw * mask_val * l_gate
    r_out = r_raw * mask_val * r_gate

    # store results in‑place
    tl.store(left_ptr + offs,  l_out, mask=active)
    tl.store(right_ptr + offs, r_out, mask=active)
    tl.store(out_gate_ptr + offs, o_gate, mask=active)


@triton.jit
def mul_inplace_kernel(
    a_ptr,          # fp16*, tensor to be multiplied (in‑place)
    b_ptr,          # fp16*, operand (same shape as a_ptr)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """Element‑wise a = a * b (in‑place)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim]      (float32)
        - mask         : torch.Tensor      [B, N, N]           (bool/float)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict with keys
                         * "dim"          – input last dimension
                         * "hidden_dim"   – hidden dimension for the TriMul heads
                         * "nomask" (opt) – if True, mask is ignored
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    device = inp.device
    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → keep precision here)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (out_features stacked)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*hidden_dim, dim)

    # fused projection: shape -> [B, N, N, 5*hidden_dim]
    proj = torch.nn.functional.linear(x_h, w_concat)  # bias=False

    # ------------------------------------------------------------------
    # 4️⃣ Split fused result + prepare tensors for gating + masking
    # ------------------------------------------------------------------
    splits = torch.split(proj, hidden_dim, dim=-1)
    left_raw, right_raw, left_gate_pre, right_gate_pre, out_gate_pre = (
        s.contiguous() for s in splits
    )
    # Free the large fused tensor (projected) as soon as possible
    del proj

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast) – prepare for Triton kernel
    # ------------------------------------------------------------------
    use_mask = 0 if nomask else 1
    if use_mask:
        # mask is [B, N, N]; convert to fp16 and flatten
        mask_h = mask.to(torch.float16).view(-1).contiguous()
        mask_len = mask_h.numel()
    else:
        # dummy tensor – never read because USE_MASK=0
        mask_h = torch.empty(0, dtype=torch.float16, device=device)
        mask_len = 0

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask + sigmoid(gates) + element‑wise scaling
    # ------------------------------------------------------------------
    # flatten tensors – Triton works on 1‑D pointers
    left_ptr = left_raw.view(-1)
    right_ptr = right_raw.view(-1)
    left_gate_ptr = left_gate_pre.view(-1)
    right_gate_ptr = right_gate_pre.view(-1)
    out_gate_ptr = out_gate_pre.view(-1)   # will be overwritten with sigmoid

    numel = left_ptr.numel()               # = B * N * N * hidden_dim
    BLOCK = 65536                          # tuned tile size (≈64 k elements)
    grid = ((numel + BLOCK - 1) // BLOCK,)

    gates_mask_fuse_kernel[grid](
        left_ptr,
        right_ptr,
        left_gate_ptr,
        right_gate_ptr,
        out_gate_ptr,
        mask_h,
        numel,
        hidden_dim,
        mask_len,
        BLOCK=BLOCK,
        USE_MASK=use_mask,
    )
    # Release intermediate gate pre‑activation tensors
    del left_gate_pre, right_gate_pre, out_gate_pre, mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ reduction (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left_raw.shape  # H == hidden_dim
    total_heads = B * H

    # reshape for batched matmul:
    #  left: (B, N, N, H) → (B, H, N, N) → (total_heads, N, N)
    left_mat = left_raw.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    # right needs to be transposed on its last two axes: (B, N, N, H) → (B, H, N, N) with K ↔ J
    right_mat = right_raw.permute(0, 3, 2, 1).contiguous().view(total_heads, N, N)

    out_mat = torch.bmm(left_mat, right_mat)          # (total_heads, N, N)  fp16

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑norm + output‑gate multiplication
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # fp16

    # In‑place multiply by the (already sigmoid‑ed) output gate
    out_norm = out_norm.contiguous()
    out_gate = out_gate_ptr.view_as(out_norm)   # same shape as out_norm (fp16)

    numel_out = out_norm.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out_norm.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = torch.nn.functional.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),  # (dim, hidden_dim)
        bias=None,
    )  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast to float32 (expected by downstream code) and return
    # ------------------------------------------------------------------
    return final.to(torch.float32)
```
