def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    device = input_tensor.device

    # 1 LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x_norm.to(torch.float16)

    # 2 stack weights for linear transforms
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    stacked_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0) # shape (5*hidden_dim, dim)
    
    # flatten input and compute all linear outputs
    B, N, _, _ = x.shape  # actually shape B,N,N,dim
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)
    proj = torch.nn.functional.linear(x_flat, stacked_w)  # (B*N*N, 5*hidden_dim)
    proj = proj.view(B, N, N, 5*hidden_dim)

    # slice
    o = hidden_dim
    left = proj[..., :o]
    right = proj[..., o:2*o]
    left_gate_pre = proj[..., 2*o:3*o]
    right_gate_pre = proj[..., 3*o:4*o]
    out_gate_pre = proj[..., 4*o:5*o]

    # mask apply to left and right before gating
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # gates sigmoid
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # O(N^3) contraction
    out = _batched_matmul_transpose_fp16(left, right)  # shape [B,N,N,hidden_dim] as fp16

    # second layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )

    # final gating and projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # first layernorm
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            (dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"])
    x = x_norm.to(torch.float16)

    # Pack all projection weights into one big weight matrix
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    stacked_weight = torch.cat([left_proj_w,
                                right_proj_w,
                                left_gate_w,
                                right_gate_w,
                                out_gate_w], dim=0)  # shape (5*hidden_dim, dim)

    # Linear transform for all heads at once
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # (B*N*N, dim)
    proj = torch.nn.functional.linear(x_flat, stacked_weight)  # (B*N*N, 5*hidden_dim)
    proj = proj.view(B, N, N, 5 * hidden_dim)

    h = hidden_dim
    left_proj = proj[..., :h]
    right_proj = proj[..., h:2*h]
    left_gate_pre = proj[..., 2*h:3*h]
    right_gate_pre = proj[..., 3*h:4*h]
    out_gate = torch.sigmoid(proj[..., 4*h:5*h])

    # Gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # Combine projection with gates (mask comes after)
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # Heavy contraction
    out = _batched_matmul_transpose_fp16(left, right)  # (B,N,N,hidden_dim), fp16

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # final gating & projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32
    return out.to(torch.float32)

"""TriMul “outgoing” forward pass (AlphaFold‑3) – highly optimized.

Algorithm
---------
1️⃣  Layer‑norm the [B, N, N, C] input (float32 → float32).
2️⃣  Cast to fp16 and perform **all five** linear projections (left‑proj,
    right‑proj, left‑gate, right‑gate, out‑gate) in a *single* matmul by
    stacking the weight matrices.  This reduces the memory traffic from
    `5 × (B·N²·C)` reads of the input to just one read.
3️⃣  Slice the stacked result, apply sigmoid to the three gate tensors,
    fuse the hidden‑dim gates with the projected tensors and optionally
    multiply by the pair‑wise mask.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply A @ Bᵀ with a custom Triton
   kernel (fp16 inputs, fp32 accumulation).  Tiling 128×128×64 and
   double‑buffering (`num_stages=2`) yields near‑Tensor‑Core utilisation.
5️⃣  Apply the second LayerNorm, the output‑gate and the final linear
    projection (hidden → C).
6️⃣  Return the result in float32 (the harness expects float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]  (fp16)   where M = B*hidden_dim, N = seq_len, K = seq_len
#   C    : [M, N, N]  (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalars (M is unused, kept for interface compatibility)
    stride_am, stride_ai, stride_ak,   # strides of A (in elements)
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile index over rows (i‑axis)
    pid_j = tl.program_id(1)   # tile index over cols (j‑axis)
    pid_z = tl.program_id(2)   # batch * hidden index

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32) – safe for long reductions
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load a tile of A : (BLOCK_M, BLOCK_K)
        a = tl.load(
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load a tile of Bᵀ (i.e. B[j, k]) : (BLOCK_K, BLOCK_N)
        b = tl.load(
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # fp16×fp16 → fp32 dot‑product (Tensor‑core friendly)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back the result (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel for the N³ contraction
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a
    custom Triton kernel.  The input tensors are expected to be on CUDA,
    contiguous, and in fp16.
    """
    B, N, _, H = left.shape

    # Pack batch and head dimension into one leading dimension:
    #   [B, N, N, H] -> [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in *elements* (Triton works with element strides)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes tuned for H100 Tensor‑Core pipelines
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth = batch * hidden
    )

    _trimul_matmul_kernel[
        grid
    ](
        A,
        B_,
        C,
        B * H,               # dummy M (not used inside the kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout: [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32 on the same device
        as the input.
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # Cast to fp16 for the heavy work that follows
    x = x_norm.to(torch.float16)  # [B, N, N, dim], fp16

    # -------------------------------------------------
    # 2️⃣ Fuse *all* linear projections + gating into one matmul
    # -------------------------------------------------
    # Cast weights once to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (hidden, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (hidden, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (hidden, dim)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (hidden, dim)

    # Stack all five matrices → shape (5*hidden, dim)
    stacked_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*hidden_dim, dim)

    B, N, _, _ = x.shape
    # Flatten the first three dimensions: (B·N·N, dim)
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    # One big linear pass – loads x only once
    proj_flat = torch.nn.functional.linear(x_flat, stacked_weight)  # (B*N*N, 5*hidden)
    proj = proj_flat.view(B, N, N, 5 * hidden_dim)  # (B, N, N, 5*hidden)

    h = hidden_dim
    left_proj   = proj[..., :h]                # (B, N, N, h)
    right_proj  = proj[..., h : 2 * h]         # (B, N, N, h)
    left_gate_pre  = proj[..., 2 * h : 3 * h]  # (B, N, N, h)
    right_gate_pre = proj[..., 3 * h : 4 * h]  # (B, N, N, h)
    out_gate_pre   = proj[..., 4 * h : 5 * h]  # (B, N, N, h)

    # -------------------------------------------------
    # 3️⃣ Sigmoids for the three gates
    # -------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)   # used after the second LN

    # -------------------------------------------------
    # 4️⃣ Fuse projection with its gate (mask applied afterwards)
    # -------------------------------------------------
    left  = left_proj * left_gate   # (B, N, N, h)
    right = right_proj * right_gate

    if not nomask:
        # mask shape: [B, N, N, 1]    (broadcast over hidden)
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Heavy N³ contraction (batched matmul A @ Bᵀ)
    # -------------------------------------------------
    # left/right are fp16; the kernel returns fp16 as well
    out = _batched_matmul_transpose_fp16(left, right)   # (B, N, N, h)

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm on the hidden dimension
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # -------------------------------------------------
    # 7️⃣ Final gating + projection back to dim
    # -------------------------------------------------
    out = out * out_gate                      # gating post‑LN
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),   # (dim, hidden)
        bias=None,
    )  # (B, N, N, dim), fp16

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
