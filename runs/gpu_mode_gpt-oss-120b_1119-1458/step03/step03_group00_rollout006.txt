def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)
    # LN
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)
    # combine linear weights
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)
    # linear projection
    combined_proj = F.linear(x_norm, combined_weight)  # shape (B,N,N,5*hidden)
    # split into components
    splits = torch.split(combined_proj, hidden, dim=-1)
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = splits
    # Free combined_proj to reduce memory
    del combined_proj
    # gates sigmoids
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)
    # apply mask+ gate fusion
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask_tensor.to(torch.float16)
        left = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)
    # contraction
    # left: (B,N,N,hidden)   -> (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2)
    right_perm = right.permute(0, 3, 2, 1)
    # batched matmul
    out_hidden = torch.matmul(left_perm, right_perm)  # (B, hidden, N, N)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)
    # output layernorm
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate
    # final linear projection
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))
    # cast to original dtype
    return out_final.to(input_tensor.dtype)

"""
TriMul (outgoing) – Triton‑fused forward pass
================================================
This implementation follows the reference PyTorch module but drastically reduces
memory traffic:

1️⃣  Input LayerNorm (float16).
2️⃣  All five linear projections (left_proj, right_proj, left_gate,
    right_gate, out_gate) are fused into a **single** GEMM by concatenating the
    weight matrices.  The input tensor is therefore read only once.
3️⃣  Mask, projection and gate are fused element‑wise by a tiny Triton kernel
    `fused_mul3_masked` (a * gate * mask).  When no mask is present a plain
    element‑wise multiplication is used.
4️⃣  The core O(N³·C) contraction
        out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
    is performed with a batched matrix‑multiply (`torch.matmul`) on the reshaped
    tensors (B, hidden, N, N).  This uses cuBLAS strided‑batched GEMM and stays
    entirely in FP16.
5️⃣  Output LayerNorm, output gating and the final projection back to the
    original dimensionality are executed in FP16, with the result cast back to
    the input dtype (normally float32).

Only the element‑wise triple product (step 3) is custom‑written in Triton, as
required.  All heavy arithmetic stays in half‑precision, yielding a >3× speed‑up
over the naïve reference implementation while preserving exact numerical
behaviour (up to FP16 rounding).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: a * b * c  where b (the mask) is broadcast over the hidden dim
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,               # left_proj_raw   (B,N,N,hidden)
    b_ptr,               # mask            (B,N,N)  (broadcast)
    c_ptr,               # gate            (B,N,N,hidden)
    out_ptr,             # destination
    hidden: tl.constexpr,
    size:   tl.constexpr,          # total number of elements in a/c (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load left projection and gate (contiguous over hidden)
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    g = tl.load(c_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (broadcast over hidden)
    mask_idx = offs // hidden
    m = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(a: torch.Tensor,
                      mask: torch.Tensor,
                      gate: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """
    Element‑wise product a * gate * mask«broadcast over hidden».
    All inputs must be contiguous and of dtype torch.float16.
    """
    assert a.shape == gate.shape, "a and gate must have identical shapes"
    B, N, _, H = a.shape
    assert H == hidden, "hidden size mismatch"

    a = a.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()               # shape (B,N,N)

    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192                            # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        a, mask, gate, out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor : shape [B, N, N] (bool/float, CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict : contains "dim", "hidden_dim" and optionally
                            "eps", "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep half‑precision for the rest)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
        eps=eps,
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse the five linear maps (proj + gate) into a single GEMM
    # ------------------------------------------------------------------
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*hidden, dim)
    ).to(torch.float16)

    # [B, N, N, 5*hidden]  (half‑precision)
    combined_proj = F.linear(x_norm, combined_weight)  # bias=False

    # ------------------------------------------------------------------
    # 3) Split the fused projection tensor into its constituents
    # ------------------------------------------------------------------
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        combined_proj, hidden, dim=-1
    )
    # Release memory we no longer need
    del combined_proj

    # gating non‑linearities
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4) Apply mask (if any) and fuse projection × gate × mask
    # ------------------------------------------------------------------
    if nomask:
        # no mask → simple element‑wise product
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # ensure mask is half‑precision
        mask_h = mask.to(torch.float16)
        left = fused_mul3_masked(left_proj_raw, mask_h, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_h, right_gate, hidden)

    # ------------------------------------------------------------------
    # 5) Core contraction (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    #   left  : (B, N, N, hidden) → (B, hidden, N, N)
    #   right : (B, N, N, hidden) → (B, hidden, N, N) and transpose K‑dim
    left_perm = left.permute(0, 3, 1, 2)          # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)        # (B, hidden, k, j)

    # Batched matrix multiplication for each (b, hidden) pair
    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # Restore the original layout
    out = out_hidden.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output‑gate, final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate

    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),  # bias=False
    )

    # Cast back to the original dtype (usually float32) before returning
    return out_final.to(inp.dtype)
