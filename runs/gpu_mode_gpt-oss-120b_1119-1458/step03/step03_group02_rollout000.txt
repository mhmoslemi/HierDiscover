"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The *outgoing* TriMul operator implements the following computation
for a 4‑D tensor `x` of shape `[B, N, N, D]`:

    1️⃣   Layer‑norm over the feature axis `D`  (FP32 for stability).
    2️⃣   Linear projections to a hidden dimension `H`:
            left  = x @ W_left
            right = x @ W_right
    3️⃣   (Optional) mask the projected tensors (broadcast over `H`).
    4️⃣   Hidden‑dim gates (sigmoid) and element‑wise gating:
            left  = left  * sigmoid(x @ W_left_gate)
            right = right * sigmoid(x @ W_right_gate)
    5️⃣   Heavy O(N³) contraction
            out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
         which is a batched matrix multiplication C = A @ Bᵀ.
         The contraction is performed by a custom Triton kernel that
         tiles the multiplication with 64×64×64 blocks, uses FP16
         inputs, and accumulates in FP32 via `tl.dot` (tensor‑core MMA).
    6️⃣   Second Layer‑norm over the hidden dimension `H` (FP16).
    7️⃣   Output gate (sigmoid) and final linear projection back to `D`.
    8️⃣   Cast result to float32 (required output type).

Only the O(N³) contraction is executed in Triton; all other steps use
high‑performance PyTorch primitives.  The implementation works for
all test configurations listed in the problem statement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for many independent (N×N) matrices.
    Each program processes one (batch * hidden) slice (pid_z) and a tile
    of the output (pid_i, pid_j).
    """
    pid_i = tl.program_id(0)  # tile along first N dimension (i)
    pid_j = tl.program_id(1)  # tile along second N dimension (j)
    pid_z = tl.program_id(2)  # index over M (= batch * hidden_dim)

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for numerical accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N)
        # B layout is (M, N, K); we need Bᵀ → (M, K, N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Multiply‑accumulate (tensor‑core MMA)
        acc += tl.dot(a, b)  # fp16×fp16 → fp32 accumulation

    # Write the result tile back (cast to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Both `left` and `right` must be FP16 tensors of shape [B, N, N, H].
    The result is returned in FP16 shape [B, N, N, H].
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension:
    # (B, N, N, H) → (B*H, N, N)  (contiguous)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in number of elements (triton expects i32)
    stride_am, stride_ai, stride_ak = A.stride()      # (M, N, K)
    stride_bm, stride_bi, stride_bk = B_.stride()     # (M, N, K)
    stride_cm, stride_ci, stride_cj = C.stride()      # (M, N, N)

    # Tile sizes – 64×64×64 works well on H100 for fp16 MMA.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (i‑tiles, j‑tiles, M‑tiles)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    # Launch Triton kernel
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bi,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Public entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, D] float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Convert all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool, float, or uint8; cast to fp16 and broadcast.
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # Apply hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (FP16 inputs/outputs)
    # ------------------------------------------------------------------
    # Result: [B, N, N, hidden] (FP16)
    out = _batched_matmul_transpose_fp16(left, right)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + final linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
