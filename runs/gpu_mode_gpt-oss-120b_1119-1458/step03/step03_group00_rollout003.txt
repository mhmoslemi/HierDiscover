"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
================================================================
Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – performed with torch.nn.functional.layer_norm.
2️⃣ **Fused linear stage** – all five dim‑→‑hidden projections
   (left_proj, right_proj, left_gate, right_gate, out_gate) are computed in a
   *single* GEMM by concatenating their weight matrices along the output dimension.
   This removes five separate kernel launches and the repeated loading of the same
   input tensor.
3️⃣ Gates (sigmoid) are applied to the three gate tensors.
4️⃣ Mask & gate fusion – when a mask is present a tiny Triton kernel
   `fused_mul3_masked` computes `proj * gate * mask` without broadcasting the
   mask, i.e. each element loads the mask once and re‑uses it for all hidden
   channels.
5️⃣ Core contraction – the outgoing update  
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  
   is a batched matrix‑multiply over the hidden dimension (B·hidden independent
   GEMMs of size N×N).  Using `torch.matmul` leverages cuBLAS’s optimized
   strided‑batched kernels.
6️⃣ Output LayerNorm, multiplication with the output gate, and a final
   dim‑←‑hidden projection (again a single GEMM).  All heavy arithmetic stays in
   FP16; the final result is cast back to the input dtype (normally float32).

Only the element‑wise triple product (step 4) is custom‑written in Triton, as
required by the task.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: a * b * c   where b is a 3‑D mask broadcasted over the hidden dim
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,               # left_proj_raw or right_proj_raw    (B,N,N,hidden)
    b_ptr,               # mask (B,N,N) – broadcasted
    c_ptr,               # gate (B,N,N,hidden)
    out_ptr,             # destination tensor (B,N,N,hidden)
    hidden: tl.constexpr,
    size:   tl.constexpr,          # total number of elements (B·N·N·hidden)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load left/right projection and gate (contiguous in hidden)
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    # Derive mask index (mask is duplicated over hidden)
    mask_idx = offs // hidden
    b = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)

    out = a * c * b
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(a: torch.Tensor,
                      mask: torch.Tensor,
                      c: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """
    Element‑wise product a * c * mask where ``mask`` is broadcast over the last
    (hidden) dimension.  All tensors must be contiguous and of dtype
    torch.float16.
    """
    assert a.shape == c.shape, "Shape mismatch between a and c"
    B, N, _, H = a.shape
    assert H == hidden, "Hidden dimension mismatch"

    a = a.contiguous()
    c = c.contiguous()
    mask = mask.contiguous()          # (B,N,N)

    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192                      # tuned for H100 FP16
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        a_ptr=a,
        b_ptr=mask,
        c_ptr=c,
        out_ptr=out,
        hidden=hidden,
        size=total,
        BLOCK_SIZE=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  shape [B, N, N] (bool/float, CUDA) – may be ignored
                            when config["nomask"] is True.
            - weights_dict: model weights, same names as in the reference module.
            - config_dict: contains keys
                * "dim" (int) – input channel dimension
                * "hidden_dim" (int) – hidden dimension (c_z)
                * optional "eps" (float) – LayerNorm epsilon (default 1e-5)
                * optional "nomask" (bool) – if True, mask is not applied.

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32 for stability, then cast to FP16)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Fuse the five dim→hidden linear projections into a single GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices (each is [hidden, dim])
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"].to(torch.float16),
            weights["right_proj.weight"].to(torch.float16),
            weights["left_gate.weight"].to(torch.float16),
            weights["right_gate.weight"].to(torch.float16),
            weights["out_gate.weight"].to(torch.float16),
        ],
        dim=0,
    )  # shape (5*hidden, dim)

    # Linear: (B·N·N, dim) @ (dim, 5*hidden) → (B·N·N, 5*hidden)
    x_flat = x_norm.view(-1, dim)                     # (M, dim)  with M = B·N·N
    proj_gate = F.linear(x_flat, combined_weight)    # no bias
    proj_gate = proj_gate.view(B, N, N, 5 * hidden)   # (B,N,N,5*hidden)

    # Split the concatenated output
    h = hidden
    left_proj_raw  = proj_gate[..., 0:h]           # (B,N,N,h)
    right_proj_raw = proj_gate[..., h:2 * h]
    left_gate_raw  = proj_gate[..., 2 * h:3 * h]
    right_gate_raw = proj_gate[..., 3 * h:4 * h]
    out_gate_raw   = proj_gate[..., 4 * h:5 * h]

    # --------------------------------------------------------------
    # 3) Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)      # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 4) Mask + gate fusion (optional)
    # --------------------------------------------------------------
    if nomask:
        # No mask: simple element‑wise product
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask.to(torch.float16)       # (B,N,N)
        left  = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # --------------------------------------------------------------
    # 5) Core outgoing contraction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Rearrange tensors so that each hidden channel becomes a separate batch.
    left_perm  = left.permute(0, 3, 1, 2)   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)  # (B, hidden, k, j)

    # Batched matrix multiplication (B*hidden independent GEMMs of size N×N)
    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # Back to original layout: (B, N, N, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                     # apply output gate

    # Final linear projection hidden -> dim (bias‑free)
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),
    )   # (B,N,N,dim)

    # Cast back to the original input dtype (usually float32)
    return out_final.to(inp.dtype)
