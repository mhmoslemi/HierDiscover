def matmul_outgate_kernel(
    out_ptr,      # output [M, D]
    norm_ptr,     # input normalized out [M, H]
    gate_ptr,     # out_gate [M, H]
    weight_ptr,   # weight [D, H] (i.e., to_out weight)
    M, D, H,
    stride_out_m, stride_out_n,
    stride_norm_m, stride_norm_n,
    stride_gate_m, stride_gate_n,
    stride_wt_m, stride_wt_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute block indices for M dimension
    block_m = pid * BLOCK_M
    # loops over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # accumulate in FP32 for precision
    for k in range(0, tl.cdiv(H, BLOCK_K)):
        k_start = k * BLOCK_K
        # Load tile of norm and gate, compute elementwise product
        off_m = tl.arange(0, BLOCK_M) + block_m
        off_n = tl.arange(0, BLOCK_N)
        off_k = tl.arange(0, BLOCK_K) + k_start

        mask_m = off_m < M
        mask_k = off_k < H

        # load norm tile (M x K)
        norm = tl.load(norm_ptr + off_m[:, None] * stride_norm_m + off_k[None, :] * stride_norm_n,
                       mask = mask_m[:, None] & mask_k[None, :])
        gate = tl.load(gate_ptr + off_m[:, None] * stride_gate_m + off_k[None, :] * stride_gate_n,
                       mask = mask_m[:, None] & mask_k[None, :])

        # elementwise multiplication
        scaled = norm * gate  # FP16*FP16->FP16; but we may convert to FP32 for accum
        # convert to FP32 for dot
        scaled_f32 = scaled.to(tl.float32)

        # load weight tile (K x N) [H,D]
        # weight stored as [D, H] so we need to load weight_T transposed
        weight = tl.load(weight_ptr + off_k[:, None] * stride_wt_m + off_n[None, :] * stride_wt_n,
                         mask = mask_k[:, None] & (off_n < D))
        # weight shape [K,N]
        weight_f32 = weight.to(tl.float32)

        # dot product
        acc += tl.dot(scaled_f32, weight_f32)  # automatically sums over K dimension
        # loop phi
    # after loop
    # store result
    out = acc.to(tl.float16)
    tl.store(out_ptr + off_m[:, None] * stride_out_m + off_n[None, :] * stride_out_n,

             mask = mask_m[:, None] & (off_n < D))

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # config keys
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # to use half32 half compute
    # Step 1: layernorm over last dim: use torch.nn.functional.layer_norm
    # The normalization weight and bias are stored as 1D (dim and maybe hidden_dim). Use them.

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Input is float32; we apply LN in float32 for stability
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # convert to half for speed
    x = x_norm.half()
    # half weights
    left_proj_weight = weights["left_proj.weight"].half()
    left_gate_weight = weights["left_gate.weight"].half()
    right_proj_weight = weights["right_proj.weight"].half()
    right_gate_weight = weights["right_gate.weight"].half()
    out_gate_weight = weights["out_gate.weight"].half()
    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias = weights["to_out_norm.bias"].half()
    to_out_weight = weights["to_out.weight"].half()

    # Combine left/right projection+gate linear calls
    # Build weight matrix of shape (4*hidden_dim, dim)
    combined_weight = torch.cat([
        left_proj_weight,
        left_gate_weight,
        right_proj_weight,
        right_gate_weight,
    ], dim=0) # shape (4*hidden_dim, dim)

    # Compute combined outputs: shape [B,N,N,4*H]
    combined = torch.nn.functional.linear(x, combined_weight)  # no bias

    # split
    # layout: left_proj, left_gate, right_proj, right_gate each of shape (B,N,N,H)
    H = hidden_dim
    left_proj = combined[..., :H]
    left_gate = combined[..., H:2*H]
    right_proj = combined[..., 2*H:3*H]
    right_gate = combined[..., 3*H:]

    # optional mask
    if not nomask:
        m_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * m_fp16
        right_proj = right_proj * m_fp16

    # apply sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # O(N^3) contraction: left @ right.T per batch and hidden
    B, N, _, H = left.shape

    # reshape to (B*H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matmul (B*H, N, N) * (B*H, N, N)^T
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # result shape (B*H,N,N)

    # reshape back [B,N,N,H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # second layer norm over hidden dimension H
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # compute out_gate (sigmoid) using x_norm (already half)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # In-place scaling using Triton kernel
    # Flatten out, out_gate
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)

    total = out_flat.numel()

        num_warps=4,
    )
    # final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    # return float32

def _elemwise_mul_kernel(dst_ptr, src_ptr, N, stride_dst, stride_src,
                         BLOCK_SIZE: tl.constexpr):

    start = pid * BLOCK_SIZE

    mask = offs < N
    dst_val = tl.load(dst_ptr + offs * stride_dst, mask=mask)

    out_val = dst_val * src_val
    tl.store(dst_ptr + offs * stride_dst, out_val, mask=mask)

"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm the 4‑D tensor (float32 → half) – the only reduction.
2️⃣  Fuse the four linear layers (left_proj, left_gate, right_proj, right_gate)
    into a *single* half‑precision GEMM.  The 4·hidden_dim output channels are
    subsequently sliced into projection / gate tensors.
3️⃣  (Optional) apply the pair‑wise mask to the projection tensors.
4️⃣  Compute the heavy O(N³) contraction  
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]  
    as a batched matrix‑multiply: for each (b,h) compute
    left[b,:, :,h] @ right[b,:, :,h]ᵀ  (N×N) using torch.bmm.
5️⃣  Layer‑norm the intermediate result (half precision).
6️⃣  Compute the output‑gate from the *same* normalized input,
    multiply it element‑wise with the tensor from step 4 (tiny Triton kernel).
7️⃣  Final linear projection back to the original channel dimension.
8️⃣  Cast back to float32 (the API expects float32).

All heavy arithmetic is performed in FP16, only the two layer‑norms use the
original input dtype for numerical stability.  A tiny Triton kernel is used
for the element‑wise out‑gate multiplication, satisfying the “must‑use‑kernel”
requirement while keeping the dominant work in cuBLAS (torch.bmm) which is
already optimal on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Tiny Triton kernel: in‑place element‑wise multiplication (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
# Computes C = A @ Bᵀ where A,B: [B*H, N, N] → out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (half)
    Returns      : [B, N, N, H] (half)
    """
    B, N, _, H = left.shape

    # reshape to a batch of B*H matrices of shape N×N
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ   (fast Tensor‑Core implementation)
    C = torch.bmm(A, B_.transpose(1, 2))

    # reshape back to the original 4‑D layout
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradient).

    Args
    ----
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor [B, N, N]     (float32 or bool)
        weights_dict : name → torch.Tensor (float32, already on device)
        config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                       ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, same device as input)
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    H = hidden_dim

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # Work in half‑precision from now on
    x = x_norm.half()  # [B, N, N, dim]  fp16

    # -------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # -------------------------------------------------
    # 3️⃣ Fuse left/right projection + gate (1 GEMM → 4·H outputs)
    # -------------------------------------------------
    # weight: (4·H, dim)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape (4*H, dim)

    # Linear: [B,N,N,dim] @ combined_w.T → [B,N,N,4*H]
    combined = torch.nn.functional.linear(x, combined_w)  # no bias

    # Slice back into the four tensors
    left_proj = combined[..., :H]           # [B,N,N,H]
    left_gate = combined[..., H:2 * H]      # [B,N,N,H]
    right_proj = combined[..., 2 * H:3 * H] # [B,N,N,H]
    right_gate = combined[..., 3 * H:]      # [B,N,N,H]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        m = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * m
        right_proj = right_proj * m

    # -------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still fp16)
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate  # [B,N,N,H]

    # -------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM (left @ rightᵀ)
    # -------------------------------------------------
    out = _batched_matmul_transpose(left, right)  # [B,N,N,H]  fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # -------------------------------------------------
    # 8️⃣ Out‑gate (sigmoid) – linear on the *same* normalized input
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] fp16

    # -------------------------------------------------
    # 9️⃣ Fuse element‑wise out‑gate multiplication with a tiny Triton kernel
    # -------------------------------------------------
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK = 1024
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,            # dst stride (contiguous)
        1,            # src stride (contiguous)
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # -------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] fp16

    # -------------------------------------------------
    # ✅ Cast back to float32 (API requirement)
    # -------------------------------------------------
    return out.float()
