"""
Optimized outgoing TriMul kernel (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input (FP32 → FP16) to improve numeric stability.
2. Cast all linear weights to FP16 once.
3. Fuse the projection and gating linear layers by concatenating the weight
   matrices, reducing the number of matrix‑multiplications from four to two.
4. (Optional) apply the pairwise mask after gating – this is mathematically
   equivalent to the reference implementation.
5. Pad the sequence dimension to a multiple of the tile size (64 or 128) so
   the Triton kernel can avoid per‑element bounds checks.
6. Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton kernel that performs a batched matrix multiply
   C = A @ Bᵀ for each (batch, hidden) pair.
   The kernel uses 64×64×64 (or 128×128×128) tiles, FP16 MMA, FP32
   accumulation and a 3‑stage pipeline.
7. Slice away the padding, apply the second LayerNorm, the output gate and
   the final linear projection back to the original `dim`.
8. Return the result in FP32.

The implementation works for all shapes required by the test suite and
runs entirely on the GPU (the contraction is in Triton, the rest in
high‑performance PyTorch primitives).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]  (M = batch * hidden, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ with A,B ∈ (M,N,K).  All dimensions are padded to the tile size,
    so no bounds checks are required inside the kernel."""
    pid_m = tl.program_id(0)   # tile along the first N dimension (i)
    pid_n = tl.program_id(1)   # tile along the second N dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index (M dimension)

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (already padded)
    num_k_tiles = K // BLOCK_K
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A[i, k] tile
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs)          # fp16

        # Bᵀ[k, j] tile  (i.e. B[k, j])
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs)          # fp16

        # dot (fp16 * fp16 → fp32) and accumulate
        acc += tl.dot(a, b)

    # write result (back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16))


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
    num_warps: int = 8,
) -> torch.Tensor:
    """
    Perform C = left @ rightᵀ for each (batch, hidden) pair using Triton.
    `left` / `right` are FP16 tensors of shape [B, N, N, H] where N is already
    padded to a multiple of `tile`.  Returns a tensor of the same shape (still
    padded) in FP16.
    """
    B, N, _, H = left.shape

    # put the hidden dimension before the two N axes for a contiguous layout
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    A = left.view(B * H, N, N)                     # [B*H, N, N]
    B_ = right.view(B * H, N, N)

    C = torch.empty_like(A)  # output in FP16

    # strides in element units (Triton expects strides in elements)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    grid = (
        (N + tile - 1) // tile,
        (N + tile - 1) // tile,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,          # M (unused inside the kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=tile,
        num_warps=num_warps,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass (gradient‑free) of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)
    x = x_norm.to(torch.float16)  # heavy part works in fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast all model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (dim, hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gate (FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    # --- left side: concat proj and gate weights ---
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*hidden, dim)
    left_cat = torch.nn.functional.linear(x_flat, left_cat_w)  # (B*N*N, 2*hidden)

    left_proj = left_cat[:, :hidden_dim]          # view, no copy
    left_gate_raw = left_cat[:, hidden_dim:]      # view, no copy
    left_gate = torch.sigmoid(left_gate_raw)      # (B*N*N, hidden)

    left = left_proj * left_gate                  # in‑place‑compatible multiplication

    # --- right side: concat proj and gate weights ---
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)
    right_cat = torch.nn.functional.linear(x_flat, right_cat_w)

    right_proj = right_cat[:, :hidden_dim]
    right_gate_raw = right_cat[:, hidden_dim:]
    right_gate = torch.sigmoid(right_gate_raw)

    right = right_proj * right_gate

    # free concatenated buffers early (optional but saves peak memory)
    del left_cat, right_cat

    # reshape back to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)   # [B, N, N, hidden]
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Output gate (same shape as left/right)
    # ------------------------------------------------------------------
    out_gate_raw = torch.nn.functional.linear(x_flat, out_gate_w)   # (B*N*N, hidden)
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden_dim)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6️⃣ Pad sequence dimension to a multiple of TILE (to avoid bounds checks)
    # ------------------------------------------------------------------
    TILE = 128 if N >= 512 else 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad  = left.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :]  = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (FP16 → FP16)
    # ------------------------------------------------------------------
    # Choose an appropriate number of warps for the tile size
    num_warps = 16 if TILE == 128 else 8
    out = _batched_matmul_transpose_fp16(
        left_pad, right_pad, tile=TILE, num_warps=num_warps
    )  # [B, N_pad, N_pad, hidden] (FP16)

    # strip padding
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, hidden] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Output gate + final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (the caller expects FP32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
