@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch and hidden index
    b = pid_bh // H
    d = pid_bh % H

    # compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # create range offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointer to base for left and right for this (b,d)
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile [BLOCK_M, BLOCK_K]
        left_ptrs = left_base \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile [BLOCK_K, BLOCK_N] (k rows, j cols)
        right_ptrs = right_base \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # compute dot product
        acc += tl.dot(left_tile, right_tile)

    # convert to fp16
    acc_fp16 = acc.to(tl.float16)

    # store result
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc_fp16, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1️⃣  Layer‑norm the [B, N, N, C] input (float32 for stability) and cast to FP16.
2️⃣  Fuse the four linear layers (proj+gate for left/right) into two FP16 GEMMs:
    * left_comb  = Linear(x, [left_proj; left_gate])
    * right_comb = Linear(x, [right_proj; right_gate])
    Then split each into projection and gate tensors.
3️⃣  (Optional) Apply the boolean/float mask on the projection tensors.
4️⃣  Apply sigmoid gates → `left = left_proj * sigmoid(left_gate)`,
    `right = right_proj * sigmoid(right_gate)`.
5️⃣  Compute the heavy O(N³) contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] · right[b,j,k,d]
    with a custom Triton kernel that works directly on the original
    [B,N,N,D] layout (no permute / copy).  The kernel performs a
    batch‑wise GEMM over the hidden dimension using Tensor‑Core‑friendly
    `tl.dot` and accumulates in FP32.
6️⃣  Layer‑norm the intermediate result (still FP16) and compute the
    output‑gate `out_gate = sigmoid(Linear(x, out_gate_weight))`.
7️⃣  Fuse the element‑wise multiplication `out *= out_gate` with a tiny
    Triton kernel (in‑place) to satisfy the “use a kernel” requirement.
8️⃣  Final linear projection back to `dim` (`to_out`) and cast to float32.

The implementation keeps the memory layout of the tensors (no expensive
permutes) and off‑loads the N³ contraction to a highly‑parallel Triton
kernel, reducing the overall runtime while preserving numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Tiny Triton kernel: in‑place element‑wise multiplication (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32 – total number of elements
    dst_stride: tl.constexpr,  # i32 – usually 1
    src_stride: tl.constexpr,  # i32 – usually 1
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl_load(src_ptr + offs * src_stride, mask=mask)  # alias for readability
    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel for the O(N³) contraction:
# out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def _trilmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    # left strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    # right strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    # out strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)          # tile over i (first N axis)
    pid_j = tl.program_id(1)          # tile over j (second N axis)
    pid_bh = tl.program_id(2)         # (batch * hidden) dimension

    # decode batch `b` and hidden channel `d` from the combined index
    b = pid_bh // D
    d = pid_bh % D

    # offsets for the current tile
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # pointers to the start of the (b,d) slice for left/right
    left_base  = left_ptr  + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d

    # accumulator in fp32 (Tensor‑Core dot will promote automatically)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ----- load left tile: shape (BLOCK_M, BLOCK_K) -----
        left_ptrs = left_base \
            + i_off[:, None] * stride_l_i \
            + k_off[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # ----- load right tile (k rows, j cols): shape (BLOCK_K, BLOCK_N) -----
        right_ptrs = right_base \
            + k_off[:, None] * stride_r_k \
            + j_off[None, :] * stride_r_i
        right_tile = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # ----- dot product (uses Tensor‑Cores) -----
        acc += tl.dot(left_tile, right_tile)        # fp32 accumulator

    # store the result (cast back to fp16)
    out_ptrs = out_ptr \
        + b * stride_o_b + d * stride_o_d \
        + i_off[:, None] * stride_o_i \
        + j_off[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask_tensor  : [B, N, N]       (float32 or bool)
        - weights_dict : map of weight tensors
        - config_dict  : must contain "dim" and "hidden_dim";
                         optional "nomask" (bool, default True)
    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------
    # 1️⃣ LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32
    x = x.to(torch.float16)   # work in FP16 from now on

    # -------------------------------------------------
    # 2️⃣ Cast weights to FP16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)

    # -------------------------------------------------
    # 3️⃣ Fuse projection + gate (2 GEMMs instead of 4)
    # -------------------------------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)    # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    left_comb = torch.nn.functional.linear(x, left_comb_w)       # [B,N,N,2H]
    right_comb = torch.nn.functional.linear(x, right_comb_w)    # [B,N,N,2H]

    left_proj = left_comb[..., :hidden_dim]     # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:]     # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]   # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:]   # [B,N,N,H]

    # -------------------------------------------------
    # 4️⃣ Optional mask (applied after projection, before gating)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ O(N³) contraction via custom Triton kernel
    # -------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides for left/right/out (int64 → required by Triton)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Tile sizes (tuned for H100 – change only if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_bh = B * hidden_dim

    _trilmul_kernel[(grid_i, grid_j, grid_bh)](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_d=stride_l_d,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_d=stride_r_d,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_d=stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Output gate (sigmoid) + in‑place elementwise multiplication
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] FP16

    # Flatten tensors for the tiny kernel
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)
    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,   # dst stride
        1,   # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # -------------------------------------------------
    # 9️⃣ Final projection back to `dim`
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 🔚 Cast to float32 (API contract)
    # -------------------------------------------------
    return out.to(torch.float32)
