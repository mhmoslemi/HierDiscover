"""
TriMul (outgoing version) – high‑performance forward pass

Algorithm
---------
1. **First LayerNorm** – FP32 → FP16 (the heavy part operates in FP16).
2. **Linear projections & gates** – performed with one‑shot PyTorch GEMMs.
   * left = x @ W_left        (FP16)
   * right = x @ W_right      (FP16)
   * left_gate  = sigmoid(x @ W_left_gate)   (FP16)
   * right_gate = sigmoid(x @ W_right_gate)  (FP16)
   * out_gate   = sigmoid(x @ W_out_gate)    (FP16)
   * left  *= left_gate;   right *= right_gate
   * optional mask multiplication (element‑wise) is applied here.
3. **Padding** – the sequence dimension is padded to a multiple of `BLOCK`
   (64 for N≤512, 128 otherwise) so the Triton kernel can avoid any
   boundary checks.
4. **Heavy O(N³) contraction** – computed with a fused Triton kernel that
   evaluates a batched matrix multiplication C = A @ Bᵀ for each
   (batch, hidden) pair:
       A : [M, N, N]  where M = B·hidden_dim
       B : [M, N, N]  (transposed on‑the‑fly)
       C : [M, N, N]
   The kernel uses 3‑level tiling (BLOCK_M × BLOCK_N × BLOCK_K),
   FP16 tensor‑core MMA, FP32 accumulation and a 3‑stage pipeline.
   Because the tensors have already been padded, the kernel contains
   **no mask / bounds checks**, which yields a large speed‑up.
5. **Slice away padding**, **second LayerNorm**, **output gate** and
   **final linear projection** back to the original `dim`.
6. Cast the result back to FP32 (the API expects FP32).

Only the O(N³) contraction is written in Triton; all surrounding
operations rely on highly‑optimised PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ
#   A, B : [M, N, N]  (M = batch * hidden_dim)
#   C    : [M, N, N]
#   All tensors are FP16; accumulation is in FP32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                     # M = B·H, N = padded seq‑len, K = padded seq‑len (=N)
    stride_am, stride_ai, stride_ak,   # strides of A (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B (M, j, k) – note B is accessed as Bᵀ
    stride_cm, stride_ci, stride_cj,   # strides of C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_z = tl.program_id(2)          # batch*hidden dimension
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑tiles
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---------- load A[i, k] ----------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs)                # FP16, perfectly aligned (no mask)

        # ---------- load Bᵀ[k, j]  (i.e. B[j, k]) ----------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs)                # FP16, perfectly aligned

        # dot product (FP16 * FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # write result (cast back to FP16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16))


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Perform C = left @ rightᵀ for each (batch, hidden) pair using Triton.
    ``left``/``right`` must be FP16 tensors of shape [B, N, N, H] where N
    is already padded to a multiple of ``tile``. The returned tensor has the
    same shape (still padded) and dtype FP16.
    """
    B, N, _, H = left.shape
    # pack batch and hidden dimensions → [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.empty_like(A)                       # FP16 output

    # strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps gives good occupancy on H100
        num_stages=3,         # 3‑stage pipeline (prefetch → compute → write‑back)
    )

    # reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : dict of weight tensors (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                          # [B, N, N, dim] – FP32
    x = x_norm.to(torch.float16)   # heavy work in FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)   # (B·N·N, dim)

    # ---- projections ----
    left  = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # ---- optional mask (broadcast over hidden) ----
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ---- gates (sigmoid) ----
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(x_flat, left_gate_w)
    ).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, right_gate_w)
    ).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    ).view(B, N, N, hidden_dim)

    # fuse gates into projections
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Pad the sequence dimension to a multiple of TILE to drop all
    #     bound‑checks inside the Triton kernel.
    # ------------------------------------------------------------------
    TILE = 128 if N > 512 else 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        # (0,0) pads the hidden dim, (0,pad_len) pads the two N axes
        left = torch.nn.functional.pad(left, (0, 0, 0, pad_len, 0, pad_len))
        right = torch.nn.functional.pad(right, (0, 0, 0, pad_len, 0, pad_len))
        N_pad = N + pad_len
    else:
        N_pad = N

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (FP16 in / FP16 out)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right, tile=TILE)  # [B, N_pad, N_pad, H]

    # strip the padding (if any)
    if pad_len != 0:
        out = out[:, :N, :N, :]   # back to [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (still FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (the public API expects FP32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
