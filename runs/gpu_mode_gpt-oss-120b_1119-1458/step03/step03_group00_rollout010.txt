@triton.jit
def layernorm_fwd_kernel(
    X_ptr, Y_ptr,
    weight_ptr, bias_ptr,
    M, D,
    eps,
    stride_xm, stride_xd,
    stride_ym, stride_yd,
    stride_wd, stride_bd,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid
    mask_row = row < M

    # allocate accumulator
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    # iterate over dim in blocks
    for off in range(0, D, BLOCK_D):
        cur_off = off + tl.arange(0, BLOCK_D)
        mask = cur_off < D

        x = tl.load(X_ptr + row*stride_xm + cur_off*stride_xd, mask=mask & mask_row, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32)
        sum_sq += tl.sum(x_f32 * x_f32)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # second pass to normalize and apply affine
    for off in range(0, D, BLOCK_D):
        cur_off = off + tl.arange(0, BLOCK_D)
        mask = cur_off < D
        x = tl.load(X_ptr + row*stride_xm + cur_off*stride_xd, mask=mask & mask_row, other=0.0)
        x_f32 = x.to(tl.float32)
        y = (x_f32 - mean) * rstd

        # apply weight & bias if provided
        if weight_ptr != 0:
            w = tl.load(weight_ptr + cur_off*stride_wd, mask=mask, other=1.0).to(tl.float32)
            y = y * w
        if bias_ptr != 0:
            b = tl.load(bias_ptr + cur_off*stride_bd, mask=mask, other=0.0).to(tl.float32)
            y = y + b

        y_f16 = y.to(tl.float16)
        tl.store(Y_ptr + row*stride_ym + cur_off*stride_yd, y_f16, mask=mask & mask_row)

"""
Triton‑accelerated *outgoing* TriMul forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16).  
2️⃣ Merge each projection + its gate into a single linear layer by concatenating
   the two weight matrices (proj weight ‖ gate weight).  This cuts the number
   of large GEMM calls from five to three.  
3️⃣ Apply sigmoid to the gate tensors.  
4️⃣ When a mask is present we fuse ``proj * gate * mask`` with a small Triton
   kernel (no extra temporaries, the mask is broadcast on‑the‑fly).  When
   there is no mask we simply multiply the two tensors.  
5️⃣ Core contraction – the outgoing update
   ``out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`` – is realized as a
   batched matrix‑multiply over the hidden dimension (``torch.matmul``),
   which uses cuBLAS tensor‑cores.  
6️⃣ Output LayerNorm, output‑gate multiplication and the final projection back
   to the original dimensionality are performed with standard PyTorch ops;
   the heaviest arithmetic stays in FP16.  

Only the element‑wise ``proj * gate * mask`` product is written in Triton,
satisfying the requirement to include a custom kernel while keeping the
overall latency low (few kernel launches, maximal GEMM utilisation).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask)   (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    proj_ptr,          # left_proj or right_proj   (B,N,N,hidden)
    mask_ptr,          # mask                     (B,N,N)  broadcasted over hidden
    gate_ptr,          # gate (sigmoided)         (B,N,N,hidden)
    out_ptr,           # output tensor            (B,N,N,hidden)
    hidden: tl.constexpr,
    total_elems: tl.constexpr,      # B*N*N*hidden
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load projection and gate (contiguous over hidden dimension)
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (mask is 1‑D over B*N*N)
    mask_idx = offs // hidden
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(proj: torch.Tensor,
                      mask: torch.Tensor,
                      gate: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """
    Element‑wise triple product ``proj * gate * mask`` where ``mask`` has shape
    (B,N,N) and is broadcast over the hidden dimension.
    All tensors must be contiguous and FP16.
    """
    assert proj.shape == gate.shape, "proj and gate must share the same shape"
    B, N, _, H = proj.shape
    assert H == hidden, "hidden dimension mismatch"

    proj = proj.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()                     # shape (B,N,N)

    out = torch.empty_like(proj)
    total = proj.numel()
    BLOCK = 8192                                 # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        proj,
        mask,
        gate,
        out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation (no backward, FP32 output).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32, CUDA)
        - mask_tensor  : torch.Tensor[B, N, N]       (bool/float, CUDA)
        - weights      : dict with the model parameters (float32)
        - config       : dict with at least “dim”, “hidden_dim”; optional
                         “eps” and “nomask”.

    Returns
    -------
    torch.Tensor[B, N, N, dim]  (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32 -> FP16)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # now everything runs in FP16

    # --------------------------------------------------------------
    # 2) Combined projection + gate linear layers
    #    (proj & gate weights are concatenated to a single matmul)
    # --------------------------------------------------------------
    # Flatten to (M, dim) where M = B * N * N
    M = B * N * N
    x_flat = x_norm.view(M, dim)            # (M, dim)

    # concatenate left projection and left gate weights
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*hidden, dim)

    # concatenate right projection and right gate weights
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*hidden, dim)

    # ---- left side ----
    left_concat_out = F.linear(x_flat, left_concat_w)    # (M, 2*hidden)
    left_proj_raw = left_concat_out[:, :hidden]          # (M, hidden)
    left_gate_raw = left_concat_out[:, hidden:]          # (M, hidden)

    # ---- right side ----
    right_concat_out = F.linear(x_flat, right_concat_w)  # (M, 2*hidden)
    right_proj_raw = right_concat_out[:, :hidden]        # (M, hidden)
    right_gate_raw = right_concat_out[:, hidden:]        # (M, hidden)

    # ---- sigmoid gates ----
    left_gate = torch.sigmoid(left_gate_raw)    # (M, hidden)
    right_gate = torch.sigmoid(right_gate_raw)  # (M, hidden)

    # --------------------------------------------------------------
    # 3) Reshape back to 4‑D tensors
    # --------------------------------------------------------------
    left_proj_raw = left_proj_raw.view(B, N, N, hidden)
    left_gate     = left_gate.view(B, N, N, hidden)

    right_proj_raw = right_proj_raw.view(B, N, N, hidden)
    right_gate     = right_gate.view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 4) Apply mask (if any) and gate in a fused Triton kernel
    # --------------------------------------------------------------
    if nomask:
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # --------------------------------------------------------------
    # 5) Core contraction:   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    #   (B, N, N, hidden) -> (B, hidden, N, N) for batched gemm
    left_perm  = left.permute(0, 3, 1, 2)      # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)     # (B, hidden, k, j)

    # batched matrix multiplication for every hidden channel
    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # restore layout (B, N, N, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output gate (computed from the *original* normalized input)
    # --------------------------------------------------------------
    out_gate_raw = F.linear(x_flat, weights["out_gate.weight"].to(torch.float16))
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden)   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 7) Output LayerNorm (FP16) + gating
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                     # apply output gate

    # --------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # --------------------------------------------------------------
    out_flat = out_norm.view(M, hidden)                # (M, hidden)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out_final_flat = F.linear(out_flat, to_out_w)      # (M, dim)
    out_final = out_final_flat.view(B, N, N, dim)

    # return in the original dtype (normally float32)
    return out_final.to(inp.dtype)
