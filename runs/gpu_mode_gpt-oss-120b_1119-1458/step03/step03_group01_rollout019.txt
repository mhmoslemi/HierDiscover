def fuse_gates_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    numel: tl.int64,   // total number of elements per channel = M*H

    has_mask: tl.constexpr,
    BLOCK: tl.constexpr = 1024,
):

    mask_offs = offs // H   # location index
    # Load values
    lp = tl.load(left_proj_ptr + offs, mask=offs < numel, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=offs < numel, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=offs < numel, other=0.0)

import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Kernel to combine gating and mask
@triton.jit
def fuse_left_right_and_outgate_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,  # optional, may be nullptr if nomask
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    numel: tl.int64,  # total elements = M * hidden_dim
    H: tl.constexpr,  # hidden_dim
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr = 1024,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # Load raw values
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    og = tl.load(out_gate_ptr + offs, mask=mask, other=0.0)

    # Sigmoid (approx)
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    lg_sig = sigmoid(lg)
    rg_sig = sigmoid(rg)
    og_sig = sigmoid(og)

    # Optional mask (broadcast across hidden dim)
    if has_mask:
        # position index within M (mask length)
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask_idx < (numel // H), other=1.0)
        # convert to fp16 if needed, tl loads preserve dtype
        lp = lp * m
        rp = rp * m

    left_res = lp * lg_sig
    right_res = rp * rg_sig

    tl.store(left_out_ptr + offs, left_res, mask=mask)
    tl.store(right_out_ptr + offs, right_res, mask=mask)
    tl.store(out_gate_out_ptr + offs, og_sig, mask=mask)

def gating_and_mask_kernel(
    left_proj_ptr,
    left_gate_ptr,
    right_proj_ptr,
    right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    numel,
    H,
    has_mask,
    BLOCK=1024,
):
   pid = tl.program_id(0)
   offs = pid * BLOCK + tl.arange(0, BLOCK)
   mask = offs < numel

   lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)

   og = tl.load(out_gate_ptr + offs, mask=mask, other=0.0)

   rg_sig = sigmoid(rg)
   og_sig = sigmoid(og)

   tl.store(right_proj_ptr + offs, r_out, mask=mask)
   tl.store(out_gate_ptr + offs, og_sig, mask=mask)

```python
"""
Optimized outgoing TriMul kernel.

- Uses a single fused linear projection for the three projection
  matrices (left_proj, right_proj) and the four gate matrices.
- Combines mask application, sigmoid gating and element‑wise multiply
  in a single Triton kernel.
- Keeps the heavy tensor‑core batched GEMM (left @ rightᵀ) unchanged,
  which remains the dominant N³ step and runs at full FP16 throughput.
- Post‑norm gating is performed by a lightweight Triton kernel.
- All arithmetic after the initial LayerNorm is in FP16; the final
  result is cast back to FP32 to match the reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def fuse_gates_and_mask_kernel(
    left_proj_ptr,   # fp16*, shape (numel,)
    left_gate_ptr,   # fp16*, shape (numel,)
    right_proj_ptr,  # fp16*, shape (numel,)
    right_gate_ptr,  # fp16*, shape (numel,)
    out_gate_ptr,    # fp16*, shape (numel,)
    mask_ptr,        # fp16*, shape (M,) where M = numel // H
    left_out_ptr,    # fp16*, shape (numel,)
    right_out_ptr,   # fp16*, shape (numel,)
    out_gate_out_ptr,# fp16*, shape (numel,)
    numel: tl.int64,
    H: tl.constexpr,          # hidden dimension
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK: tl.constexpr = 65536,
):
    """
    In‑place:
        left_out  = (left_proj * mask) * sigmoid(left_gate)
        right_out = (right_proj * mask) * sigmoid(right_gate)
        out_gate_out = sigmoid(out_gate)

    ``mask`` is broadcast from shape (M,) to (M*H,) where each mask
    element affects a whole hidden‑dim vector.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # load raw tensors
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    og = tl.load(out_gate_ptr + offs, mask=mask, other=0.0)

    # sigmoid helper
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    lg_s = sigmoid(lg)
    rg_s = sigmoid(rg)
    og_s = sigmoid(og)

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        # ``mask_idx`` is the index of the element in the original
        # mask tensor (shape M) that corresponds to this offset.
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask_idx < (numel // H), other=1.0)
        lp = lp * m
        rp = rp * m

    left_res  = lp * lg_s
    right_res = rp * rg_s

    # store results (overwrites the original projection buffers)
    tl.store(left_out_ptr  + offs, left_res, mask=mask)
    tl.store(right_out_ptr + offs, right_res, mask=mask)
    tl.store(out_gate_out_ptr + offs, og_s, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,          # fp16*, target tensor (to be multiplied in‑place)
    b_ptr,          # fp16*, multiplier tensor (same shape as ``a_ptr``)
    numel: tl.int64,
    BLOCK: tl.constexpr = 65536,
):
    """
    a = a * b   (element‑wise in‑place)
    Both tensors are flattened before the launch.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)

    # Unpack
    # ------------------------------------------------------------------

    # 1️⃣ Input LayerNorm (float32) – the reference uses full precision.
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for every subsequent heavy arithmetic.
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj + gate) into a single matmul.
    #    Weight order is: left_proj, left_gate, right_proj, right_gate,
    #    out_gate   – each block has shape (hidden_dim, dim).
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    weight_fused = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(torch.float16).contiguous()  # (5*H, dim)

    # Flatten the normalized input to (B*N*N, dim) and apply the fused linear.
    B, N, _, _ = x_h.shape
    M = B * N * N
    x_flat = x_h.view(M, dim)  # (M, dim)
    fused_out = F.linear(x_flat, weight_fused)  # (M, 5*H)   fp16

    # ------------------------------------------------------------------
    # 4️⃣ Split the fused tensor into the individual projection / gate slices.
    #    The slices are views on the same memory region → zero extra allocation.
    # ------------------------------------------------------------------
    splits = torch.split(fused_out, hidden_dim, dim=1)  # 5 tensors
    left_proj, left_gate, right_proj, right_gate, out_gate = (
        splits[0].contiguous(),
        splits[1].contiguous(),
        splits[2].contiguous(),
        splits[3].contiguous(),
        splits[4].contiguous(),
    )  # each (M, H)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask, sigmoid‑gating and mask‑application in a Triton kernel.
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).view(M)  # (M,)
    else:
        # Dummy mask – the kernel will ignore it at compile time.
        mask_h = torch.empty(0, dtype=torch.float16, device=device)

    # Allocate output buffers (in‑place updates are fine – we overwrite the
    # projection buffers with the gated values).
    left_out = left_proj  # will be overwritten
    right_out = right_proj
    out_gate_out = out_gate  # will hold sigmoid(out_gate)

    numel = left_proj.numel()  # = M * H
    BLOCK = 65536
    grid = ((numel + BLOCK - 1) // BLOCK,)

    HAS_MASK = 0 if nomask else 1

    fuse_gates_and_mask_kernel[grid](
        left_proj_ptr=left_proj.view(-1),
        left_gate_ptr=left_gate.view(-1),
        right_proj_ptr=right_proj.view(-1),
        right_gate_ptr=right_gate.view(-1),
        out_gate_ptr=out_gate.view(-1),
        mask_ptr=mask_h.view(-1) if not nomask else mask_h,
        left_out_ptr=left_out.view(-1),
        right_out_ptr=right_out.view(-1),
        out_gate_out_ptr=out_gate_out.view(-1),
        numel=numel,
        H=hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )

    # Release gate tensors that are no longer needed.
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape the gated tensors back to the 4‑D layout required for the
    #     batched matrix multiply: [B, N, N, H].
    # ------------------------------------------------------------------
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)
    out_gate = out_gate_out.view(B, N, N, hidden_dim)  # sigmoid already applied

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ operation – batched GEMM over the “head” dimension.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) before torch.bmm
    total_heads = B * hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # (total_heads, N, N) @ (total_heads, N, N)ᵀ → (total_heads, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑projection LayerNorm (still in fp16 – the hidden dim is low).
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (in‑place) using a tiny Triton kernel.
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    numel_out = out_flat.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    mul_inplace_kernel[grid_out](
        out_flat,
        out_gate_flat,
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to ``dim``.
    # ------------------------------------------------------------------
    final = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32 to match the reference implementation's dtype.
    return final.to(torch.float32)

"""
Optimized outgoing TriMul kernel (AlphaFold‑3 style).

Key ideas
---------

1. **Fused linear projection** – the five independent linear layers
   (left_proj, right_proj, left_gate, right_gate, out_gate) are merged
   into a single matrix multiply with a weight of shape (5·H, dim).

2. **One‑shot gating & mask** – a custom Triton kernel
   * loads the two projection slices and the three gate slices,
   * applies sigmoid to the gate values,
   * (optionally) multiplies the projections by the binary mask,
   * writes back the gated left/right tensors and the sigmoid‑ed out_gate.
   This replaces three separate kernels (two linear + sigmoid + mask
   + two element‑wise multiplies) with a single kernel.

3. **Heavy N³ step** – stays as a batched GEMM (`torch.bmm`) which
   already runs at full FP16/Tensor‑Core throughput on H100.

4. **Post‑norm & output gating** – the LayerNorm after the N³‑product
   is kept in PyTorch for numerical stability, while the final
   element‑wise multiplication by `out_gate` is performed by a tiny
   Triton kernel.

All arithmetic after the first LayerNorm is in FP16; the final
result is cast back to FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def fuse_gates_and_mask_kernel(
    left_proj_ptr,   # fp16*, shape (numel,)
    left_gate_ptr,   # fp16*, shape (numel,)
    right_proj_ptr,  # fp16*, shape (numel,)
    right_gate_ptr,  # fp16*, shape (numel,)
    out_gate_ptr,    # fp16*, shape (numel,)
    mask_ptr,        # fp16*, shape (M,) where M = numel // H
    left_out_ptr,    # fp16*, shape (numel,)
    right_out_ptr,   # fp16*, shape (numel,)
    out_gate_out_ptr,# fp16*, shape (numel,)
    numel: tl.int64,
    H: tl.constexpr,          # hidden dimension
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK: tl.constexpr = 65536,
):
    """
    In‑place:
        left_out  = (left_proj * mask) * sigmoid(left_gate)
        right_out = (right_proj * mask) * sigmoid(right_gate)
        out_gate_out = sigmoid(out_gate)

    ``mask`` is broadcast from shape (M,) to (M·H,) where each mask
    element touches the entire hidden‑dim vector.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < numel

    # -------------------- load raw tensors --------------------
    lp = tl.load(left_proj_ptr + offs, mask=active, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    og = tl.load(out_gate_ptr + offs, mask=active, other=0.0)

    # -------------------- sigmoid helper --------------------
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    lg_sig = sigmoid(lg)
    rg_sig = sigmoid(rg)
    og_sig = sigmoid(og)

    # -------------------- optional mask (broadcast) --------------------
    if HAS_MASK:
        # mask index = floor(offs / H)
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx,
                    mask=mask_idx < (numel // H),
                    other=1.0)  # default = 1 (no masking)
        lp = lp * m
        rp = rp * m

    # -------------------- final gated values --------------------
    left_res  = lp * lg_sig
    right_res = rp * rg_sig

    # -------------------- store results (in‑place) --------------------
    tl.store(left_out_ptr  + offs, left_res,  mask=active)
    tl.store(right_out_ptr + offs, right_res, mask=active)
    tl.store(out_gate_out_ptr + offs, og_sig, mask=active)


@triton.jit
def mul_inplace_kernel(
    a_ptr,          # fp16*, tensor to be multiplied in‑place
    b_ptr,          # fp16*, multiplier (same shape as ``a_ptr``)
    numel: tl.int64,
    BLOCK: tl.constexpr = 65536,
):
    """
    a = a * b   (element‑wise, in‑place)
    Both tensors are flattened before the launch.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < numel
    a = tl.load(a_ptr + offs, mask=active, other=0.0)
    b = tl.load(b_ptr + offs, mask=active, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=active)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask         : torch.Tensor  [B, N, N] (bool/float) – may be ignored
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict with at least ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and basic constants
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for full precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision – everything after this point runs in fp16
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj + gate) into a single matmul.
    # ------------------------------------------------------------------
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()               # (5·H, dim)

    B, N, _, _ = x_h.shape
    M = B * N * N                                    # B·N·N
    x_flat = x_h.view(M, dim)                        # (M, dim)

    fused_out = F.linear(x_flat, w_fused)            # (M, 5·H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Split fused tensor – views share the same storage, no extra copy.
    # ------------------------------------------------------------------
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused_out, hidden_dim, dim=1
    )
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()                # (M, H)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + sigmoid gating (single Triton kernel).
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).view(M)     # (M,)
    else:
        mask_h = torch.empty(0, dtype=torch.float16, device=device)

    # In‑place write‑back: left_proj/right_proj hold the final gated tensors.
    numel = left_proj.numel()                       # M·H
    BLOCK = 65536
    grid = ((numel + BLOCK - 1) // BLOCK,)
    HAS_MASK = 0 if nomask else 1

    fuse_gates_and_mask_kernel[grid](
        left_proj_ptr=left_proj.view(-1),
        left_gate_ptr=left_gate.view(-1),
        right_proj_ptr=right_proj.view(-1),
        right_gate_ptr=right_gate.view(-1),
        out_gate_ptr=out_gate.view(-1),
        mask_ptr=mask_h.view(-1) if not nomask else mask_h,
        left_out_ptr=left_proj.view(-1),            # overwrite in‑place
        right_out_ptr=right_proj.view(-1),
        out_gate_out_ptr=out_gate.view(-1),
        numel=numel,
        H=hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )

    # Release gate tensors we no longer need.
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to 4‑D tensors required for the batched GEMM.
    # ------------------------------------------------------------------
    left = left_proj.view(B, N, N, hidden_dim)      # [B, N, N, H]
    right = right_proj.view(B, N, N, hidden_dim)    # same shape
    out_gate = out_gate.view(B, N, N, hidden_dim)  # sigmoid already applied

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ operation: batched matrix multiplication across heads.
    # ------------------------------------------------------------------
    total_heads = B * hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # (total_heads, N, N) @ (total_heads, N, N)ᵀ → (total_heads, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑projection LayerNorm (still in fp16 – hidden_dim is modest).
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (in‑place) via a tiny Triton kernel.
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    numel_out = out_flat.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    mul_inplace_kernel[grid_out](
        out_flat,
        out_gate_flat,
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to original ``dim``.
    # ------------------------------------------------------------------
    final = F.linear(out, weights["to_out.weight"].to(torch.float16))  # [B, N, N, dim] fp16

    # Return in the expected float32 precision.
    return final.to(torch.float32)
