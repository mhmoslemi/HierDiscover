"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The module receives a 4‑D pairwise tensor x ∈ ℝ^{B×N×N×C} and a
binary mask of shape (B,N,N).  The forward algorithm is:

1️⃣ Layer‑norm over the feature axis C (float32 for stability).
2️⃣ Cast to FP16 and apply three linear projections:
     left  = x·W_left      (→ H)
     right = x·W_right     (→ H)
   The hidden‑dim H equals ``hidden_dim`` (always 128 in the tests).
3️⃣ Optional mask (broadcast over H) is applied to *both* projections.
4️⃣ Gated by three element‑wise sigmoids:
     left  = left  * sigmoid(x·W_left_gate)
     right = right * sigmoid(x·W_right_gate)
     out_gate = sigmoid(x·W_out_gate)
5️⃣ Heavy O(N³) contraction:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix multiplication C = A @ Bᵀ where
   A and B have shape (M,N,K) with M = B·H, N = seq_len, K = seq_len.
   The custom Triton kernel tiles the multiplication with
   64×64×64 blocks, uses FP16 inputs and accumulates in FP32
   (tensor‑core MMA via ``tl.dot``).  Double‑buffering (num_stages=2)
   and 8 warps hide memory latency.
6️⃣ Layer‑norm over the hidden dimension (FP16) and element‑wise
   multiplication by ``out_gate``.
7️⃣ Final linear projection back to the original feature size C and
   cast to FP32.

Only step 5 runs in Triton; the remaining operations use high‑performance
PyTorch primitives.  The implementation works for all test configurations
( B∈{1,2}, N∈{128,256,512,768,1024}, C∈{128,384}, H=128, optional mask ).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched (M,N,K) × (M,N,K)ᵀ → (M,N,N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,  # *fp16  [M, N, K]
    B_ptr,  # *fp16  [M, N, K]
    C_ptr,  # *fp16  [M, N, N]
    M, N, K,                 # sizes (i32)
    stride_am, stride_ai, stride_ak,  # strides of A
    stride_bm, stride_bi, stride_bk,  # strides of B
    stride_cm, stride_ci, stride_cj,  # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for many independent (N×N) matrices.
    A, B have shape (M, N, K).  Each program instance processes a
    single M‑index (i.e. a “head” of a specific batch) and a tile
    (BLOCK_M×BLOCK_N) of the output matrix.
    """
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # index over M = batch*hidden

    # ------------------------------------------------------------------
    # Compute row/col offsets for this tile, clamp with masks
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- Load A tile : (M‑idx, i, k) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ----- Load B tile transposed : (M‑idx, k, j) -----
        # Real memory layout of B is (M, j, k); we reinterpret as (k, j)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # ----- Multiply‑accumulate -----
        acc += tl.dot(a, b)  # fp16·fp16 → fp32 accumulation

    # ------------------------------------------------------------------
    # Store the result tile (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction executed in Triton:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        [B, N, N, H] in FP16.

    Returns
    -------
    torch.Tensor
        [B, N, N, H] in FP16.
    """
    B, N, _, H = left.shape

    # Pack batch and hidden dimension together: (B*H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in number of elements (Triton expects i32)
    stride_am, stride_ai, stride_ak = A.stride()      # (M,N,K)
    stride_bm, stride_bi, stride_bk = B_.stride()     # (M,N,K)
    stride_cm, stride_ci, stride_cj = C.stride()      # (M,N,N)

    # Tile size – 64×64×64 works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # M‑tiles (batch * hidden)
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bi,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------

    torch.Tensor

        input_tensor,

        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 3️⃣ Projection to hidden dimension (FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------

    # Apply per‑position hidden‑dim gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (FP16 inputs → FP16 output)
    # ------------------------------------------------------------------
    # Result shape: [B, N, N, hidden_dim] (FP16)
    out = _batched_matmul_transpose_fp16(left, right)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension (still FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + final projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 as required by the interface
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
