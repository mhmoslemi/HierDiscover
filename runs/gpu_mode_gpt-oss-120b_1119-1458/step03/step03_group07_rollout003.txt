"""
TriMul (outgoing) – fast forward pass for AlphaFold‑3.

Algorithm
---------
1️⃣ Layer‑norm the input (float32 → fp16 for the rest of the pipeline).  
2️⃣ Fuse the two projection linear layers *and* the two gate linear layers
   into two single mat‑muls (one for the “left” path, one for the “right”).
3️⃣ (Optional) Apply the pair‑wise mask *only* to the projection part,
   exactly as in the reference implementation.
4️⃣ Apply sigmoid to the gate tensors and multiply (projection × gate).
5️⃣ Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   with a single ``torch.einsum`` – this avoids the costly
   ``permute``‑+‑contiguous copy used by a naïve batched‑gemm.
6️⃣ Layer‑norm the intermediate result (still fp16).
7️⃣ Compute the output‑gate (sigmoid) and fuse the final element‑wise
   multiplication with a tiny Triton kernel (in‑place).
8️⃣ Linear projection back to the original channel dimension.
9️⃣ Cast the result back to float32 (the API’s expected dtype).

Only the element‑wise out‑gate multiplication is performed with Triton,
fulfilling the “use a kernel” requirement while the dominant O(N³)
contraction uses PyTorch’s highly‑optimised Tensor‑Core GEMM implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Tiny Triton kernel: dst *= src  (both FP16, 1‑D flattened view)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < N

    dst = tl.load(dst_ptr + offset * dst_stride, mask=mask)
    src = tl.load(src_ptr + offset * src_stride, mask=mask)

    tl.store(dst_ptr + offset * dst_stride, dst * src, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
        - weights_dict : mapping from weight name to torch.Tensor (float32, same device)
        - config_dict  : at least ``dim`` and ``hidden_dim``; optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B,N,N,dim]  float32

    x = x_norm.to(torch.float16)  # rest of the pipeline works in FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    out_gate_w   = weights["out_gate.weight"].to(torch.float16)    # (hidden_dim, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    to_out_w = weights["to_out.weight"].to(torch.float16)         # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left and right paths
    # ------------------------------------------------------------------
    # Concatenate along the output dimension (2*hidden_dim, dim)
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # Linear over the last dim (dim) → produce [B,N,N,2*H]
    left_comb  = torch.nn.functional.linear(x, left_comb_w)   # [B,N,N,2*H]
    right_comb = torch.nn.functional.linear(x, right_comb_w)  # [B,N,N,2*H]

    # Split projection and gate tensors
    left_proj = left_comb[..., :hidden_dim]   # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:]   # [B,N,N,H]

    right_proj = right_comb[..., :hidden_dim]   # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:]   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is 0/1 (float or bool) → cast to FP16 and broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates and multiply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate       # [B,N,N,H]
    right = right_proj * right_gate    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via einsum (avoids costly permute/contiguous)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + fused element‑wise multiplication (Triton)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] FP16

    # Flatten tensors for the Triton kernel (in‑place multiply)
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024                               # threads per block
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,          # dst stride
        1,          # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 🔚 Cast back to float32 (expected output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
