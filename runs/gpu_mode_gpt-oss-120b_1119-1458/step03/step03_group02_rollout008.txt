"""
TriMul (outgoing) – fast Triton implementation
================================================

This file implements the forward‑only “outgoing” TriMul operation that
appears in AlphaFold‑3 and related protein‑structure models.
The implementation follows the reference PyTorch module but delegates the
quadratically‑nested contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

to a custom Triton kernel.  The remaining linear layers, gating, and
LayerNorms are performed with high‑performance PyTorch primitives
(half‑precision wherever possible).  The Triton kernel computes a batched
matrix‑multiply C = A @ Bᵀ where

    A[M,i,k] = left[b,i,k,h]   (M = b·hidden_dim + h)
    B[M,j,k] = right[b,j,k,h]

All arithmetic inside the kernel is FP16‐input / FP32‑accumulation
(tensor‑core MMA).  Tile sizes are chosen at launch time:
``BLOCK = min(128, seq_len)``.  This yields 64×64×64 tiles for small
sequences and 128×128×128 tiles for larger ones, fitting comfortably in
H100 shared‑memory (≤ 128 KB per block).  Double buffering
(`num_stages=2`) and 8 warps hide memory latency.

The kernel is the only fused operation; everything else runs in PyTorch.
The final output is returned as FP32, matching the reference model.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bi, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for many independent (N×N) matrices:
        C[m,i,j] = Σₖ A[m,i,k] * B[m,j,k]

    The kernel tiles the i‑ and j‑axes (output) with blocks of size
    BLOCK_M × BLOCK_N, and iterates over K in chunks of BLOCK_K.
    """
    pid_i = tl.program_id(0)   # tile index along the first N axis (i)
    pid_j = tl.program_id(1)   # tile index along the second N axis (j)
    pid_z = tl.program_id(2)   # index over M (= batch * hidden_dim)

    # -----------------------------------------------------------------
    # Offsets of the current output tile
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension K
    # -----------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- Load A tile (M‑idx, i‑idx, k‑idx) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----- Load B tile transposed (M‑idx, k‑idx, j‑idx) -----
        # This loads the view B[m, k, j] (i.e. Bᵀ in the original layout)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # ----- Multiply‑accumulate (tensor‑core MMA) -----
        # tl.dot promotes the fp16 multiplication to fp32 accumulation.
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Store the output tile (cast back to fp16)
    # -----------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape [B, N, N, H] (dtype=torch.float16)
    right : torch.Tensor
        Tensor of the same shape as ``left`` (dtype=torch.float16)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] (dtype=torch.float16)
    """
    B, N, _, H = left.shape

    # -----------------------------------------------------------------
    # Pack the (batch, hidden) dimensions into a single leading axis.
    #   Resulting shape: (M = B*H, N, N) – contiguous layout.
    # -----------------------------------------------------------------
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP16)
    C = torch.empty_like(A, dtype=torch.float16)

    # -----------------------------------------------------------------
    # Strides in number-of‑elements (Triton expects i32 strides)
    # -----------------------------------------------------------------
    stride_am, stride_ai, stride_ak = A.stride()      # (M,N,K)
    stride_bm, stride_bi, stride_bk = B_.stride()     # (M,N,K)
    stride_cm, stride_ci, stride_cj = C.stride()      # (M,N,N)

    # -----------------------------------------------------------------
    # Choose tile size.
    #   Use a larger 128×128×128 tile for N ≥ 512, otherwise 64×64×64.
    #   The tile dimension never exceeds ``N``.
    # -----------------------------------------------------------------
    BLOCK = 128 if N >= 512 else 64
    BLOCK = min(BLOCK, N)                 # safety for very small N
    BLOCK_M = BLOCK_N = BLOCK_K = BLOCK

    # -----------------------------------------------------------------
    # Grid definition – three‑dimensional launch:
    #   pid_i: tiles along the first N axis (i)
    #   pid_j: tiles along the second N axis (j)
    #   pid_z: tiles over the packed (batch*hidden) dimension
    # -----------------------------------------------------------------
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # M‑tiles
    )

    # Launch Triton kernel
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # -----------------------------------------------------------------
    # Reshape back to original layout [B, N, N, H]
    # -----------------------------------------------------------------
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    # 2️⃣ Convert all linear weights once to FP16 (keeps the same device)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -----------------------------------------------------------------
    # Linear without bias: out = x @ weight.T
    left = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # ensure mask has same device & dtype, then unsqueeze to broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply the per‑hidden gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via the custom Triton kernel
    #    (output shape: [B, N, N, hidden], dtype=fp16)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden_dim) – keep in FP16
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Final gating + projection back to the original dimension
    # -----------------------------------------------------------------
    out = out * out_gate                    # element‑wise hidden‑dim gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim], fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (the model’s public dtype)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
