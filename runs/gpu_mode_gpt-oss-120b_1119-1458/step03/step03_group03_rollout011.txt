```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for batched matrix multiplication (left @ right.T) for the TriMul O(N³) contraction.
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for each (batch*hidden) slice.

    A, B, C: shape [M, N, K] where M = B*hidden,
    N = padded sequence length, K = padded sequence length (same as N).
    Uses FP16 inputs and FP32 accumulation.
    """
    pid_m = tl.program_id(0)   # tile along i dimension (output rows)
    pid_n = tl.program_id(1)   # tile along j dimension (output columns)
    pid_z = tl.program_id(2)   # index over combined batch*hidden

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i,k] tile (BLOCK_M x BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load Bᵀ (i.e. B[k,j]) tile (BLOCK_K x BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # GEMM step (fp16*fp16->fp32)
        acc += tl.dot(a, b)

    # Store to C (output)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                  right: torch.Tensor,
                                  tile: int = 128) -> torch.Tensor:
    """
    Compute C = left @ rightᵀ for each (batch, hidden) pair using Triton.

    Input tensors:
        left, right: FP16, shape [B, N, N, H] (already padded to a multiple of `tile`).

    Returns:
        Tensor of shape [B, N, N, H] (still padded) in FP16.
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    C = torch.empty_like(A)

    # strides in element units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Kernel launch grid
    grid = (
        (N + tile - 1) // tile,
        (N + tile - 1) // tile,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=tile,
        num_warps=8,
        num_stages=3,
    )
    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Implements the heavy O(N³) contraction with a Triton kernel and
    performs the surrounding layer‑norm, linear projections and gates
    with mixed‑precision PyTorch operations.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim`` (int).
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1️⃣ First LayerNorm (FP32 input → FP16 for heavy work)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)  # [B, N, N, dim] fp16

    # 2️⃣ Cast linear weights once to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_w = weights["to_out.weight"].to(torch.float16)

    B, N, _, _ = x.shape
    # Flatten spatial dimensions for linear layers
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    # Linear projections (FP16)
    left = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # Apply mask if present (broadcast over hidden dim)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gating (sigmoid) – compute in FP16
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, left_gate_w)
    ).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, right_gate_w)
    ).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    ).view(B, N, N, hidden_dim)

    # Fuse gates into projected tensors
    left = left * left_gate
    right = right * right_gate

    # 3️⃣ Pad sequence dimension to tile (128) to avoid bounds checks
    TILE = 128
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        # allocate padded tensors
        left_pad = left.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # 4️⃣ Heavy O(N³) contraction via Triton (FP16 input, FP16 output)
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, H]

    # Remove padding if we added any
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # 5️⃣ Second LayerNorm on hidden dimension (FP16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=out_norm_w,
        bias=out_norm_b,
    )

    # 6️⃣ Output gating & final linear projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, out_w)  # FP16 result [B, N, N, dim]

    # 7️⃣ Cast back to float32 for downstream code
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast Triton implementation
================================================

The module processes a 4‑D tensor `[B, N, N, C]` (pair representation) with the
following steps:

1. **Pre‑norm** – LayerNorm over the channel dimension (float32).
2. **Linear projections & gates** – All linear layers are executed in FP16
   (weights are cast once).  Gating uses a sigmoid, also in FP16.
3. **Optional mask** – Broadcasted over the hidden dimension.
4. **Fusion** – Projection outputs are multiplied by their gates.
5. **Padding** – Sequence length `N` is padded to a multiple of 128 to
   eliminate per‑element bounds checks inside the Triton kernel.
6. **Heavy O(N³) contraction** – Implemented as a batched matrix multiply
   `C = A @ Bᵀ` for each `(batch, hidden)` slice using a custom Triton kernel.
   * Input matrices `A, B` are FP16, the accumulator is FP32 and the result is
     stored directly in FP32 (no extra casting).
   * Tile size is 128 × 128 × 128, 8 warps, 3‑stage pipeline – a good fit for
     NVIDIA H100 Tensor Cores.
7. **Second LayerNorm** – Applied in FP32 on the hidden dimension.
8. **Output gate & linear projection** – Gate in FP32, final projection back to
   the original `dim` (also FP32).
9. **Return** – FP32 tensor `[B, N, N, dim]`.

Only the O(N³) part runs in Triton; surrounding operations use highly‑optimized
PyTorch primitives.  The implementation works for all shapes required by the
test suite (batch ≤ 2, `N` ∈ {256, 512, 768, 1024}, `dim` ∈ {128, 384},
`hidden_dim` = 128).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A,B: FP16 → C: FP32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,  # fp16   [M, N, K]   (M = B*H)
    B_ptr,  # fp16   [M, N, K]
    C_ptr,  # fp32   [M, N, N]
    M, N, K,
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.

    Arguments:
        A_ptr, B_ptr, C_ptr – base pointers.
        M – total number of batched matrices (ignored in kernel body, kept for API compatibility).
        N – padded sequence length.
        K – padded sequence length (same as N here).
        stride_* – element‑wise strides of the tensors.
        BLOCK_* – tile sizes (must be multiples of 16 for tensor‑core efficiency).
    """
    pid_m = tl.program_id(0)   # tile row index (i)
    pid_n = tl.program_id(1)   # tile col index (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # ----- offsets for the output tile -----
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A[i, k] (BLOCK_M × BLOCK_K) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---- load Bᵀ[k, j] i.e. B[k, j] (BLOCK_K × BLOCK_N) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # ---- matrix‑multiply‑accumulate (fp16 * fp16 → fp32) ----
        acc += tl.dot(a, b)

    # ----- store result (still FP32) -----
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                  right: torch.Tensor,
                                  tile: int = 128) -> torch.Tensor:
    """
    Perform `C = left @ rightᵀ` for each (batch, hidden) slice using the Triton kernel
    above.

    Parameters
    ----------
    left, right : torch.Tensor
        FP16 tensors with shape `[B, N, N, H]`.  `N` should already be padded
        to a multiple of ``tile`` (the kernel pads internally if needed).

    tile : int
        Tile size for the Triton kernel.  128 is a sweet spot on H100.

    Returns
    -------
    torch.Tensor
        Output tensor `[B, N, N, H]` in **FP32**.
    """
    B, N, _, H = left.shape

    # Pack batch and hidden dimensions → [B*H, N, N] (contiguous)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)                     # fp16
    B_ = right_perm.view(B * H, N, N)                   # fp16

    # Allocate FP32 output buffer
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # Strides in element units
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Launch grid: (tiles_i, tiles_j, batch*hidden)
    grid = (
        (N + tile - 1) // tile,
        (N + tile - 1) // tile,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,          # M (unused inside kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=tile,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to the original layout: [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # 4️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # Fuse gates into the projected tensors
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Pad to TILE (128) to avoid per‑element bounds checks inside Triton
    # ------------------------------------------------------------------
    TILE = 128
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:

        left_pad  = left.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :]  = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 inputs → fp32 output)
    # ------------------------------------------------------------------
    out_fp32 = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, hidden_dim] FP32

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    out = out * out_gate_fp32

    to_out_w = weights["to_out.weight"]            # float32
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 10️⃣ Return result (already FP32)
    # ------------------------------------------------------------------
    return out
```

This implementation keeps the computationally‑heavy O(N³) contraction in a highly‑optimized Triton kernel (128×128×128 tiles, FP16 inputs, FP32 accumulation), while the surrounding linear layers, gating, masks and layer‑norms are performed with efficient PyTorch primitives.  All tensors stay on the provided CUDA device, and the function returns the final result in `float32` as required.
