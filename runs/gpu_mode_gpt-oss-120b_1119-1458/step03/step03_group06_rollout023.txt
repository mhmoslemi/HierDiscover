"""
TriMul OUTGOING kernel (AlphaFold‑3)
-----------------------------------
The forward pass consists of three stages:
1️⃣  Layer‑norm the 4‑D input (float32 → float32).
2️⃣  Fuse the two linear projections (proj + gate) for the left and right
    tensors, apply the optional mask and the sigmoid‑gates, all in fp16.
3️⃣  Batched matrix‑multiply C = A @ Bᵀ with a custom Triton kernel
    (A = left, B = right).  The kernel works on a leading dimension
    M = batch × hidden, tiling the N‑axes with 128/256‑sized blocks.
    The accumulation uses FP32 for numerical stability.
4️⃣  Standard PyTorch layer‑norm, out‑gate and final projection (all in fp16)
    → cast back to float32 for the harness.

The heavy N³ contraction is completely performed in Triton, while the
linear‑layers and the two LayerNorms stay in PyTorch (they are already
highly‑optimised).  By merging the proj+gate linear layers we halve the
memory traffic for the heavyweight tensors and by increasing the mm
tile size for large sequences we cut the kernel‑launch overhead.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, N]   (fp16)   where M = batch * hidden
#   C    : [M, N, N]   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]   (K == N)
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalars (M is unused, kept for API compatibility)
    stride_am, stride_ai, stride_ak,   # strides of A (elements)
    stride_bm, stride_bj, stride_bk,   # strides of B
    stride_cm, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,   # tile size along the first N‑axis (rows)
    BLOCK_N: tl.constexpr,   # tile size along the second N‑axis (cols)
    BLOCK_K: tl.constexpr,   # tile size along the reduction dim
):
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_z = tl.program_id(2)          # batch * hidden index

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (shape: BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile, transposed (shape: BLOCK_K × BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Multiply‑accumulate (fp16 × fp16 → fp32)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]   (fp16)
    right: torch.Tensor,  # [B, N, N, H]   (fp16)
) -> torch.Tensor:
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using Triton.

    Arguments
    ---------
    left, right : contiguous fp16 tensors on the same CUDA device.
                  Shape = [B, N, N, H].

    Returns
    -------
    Tensor of shape [B, N, N, H] (fp16) – the contracted result.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension: [M, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)   # transpose on the last two axes

    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element‑units (required by Triton)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Heuristic tile size – larger tiles for long sequences reduce launch overhead.
    if N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 256
        BLOCK_N = 256
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # depth = batch * hidden
    )

    _trimul_matmul_kernel[
        grid
    ](
        A,
        B_,
        C,
        B * H,               # dummy M (not used inside the kernel)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps = 256 threads per block (fits H100)
        num_stages=2,       # double‑buffering to hide memory latency
    )

    # Reshape back to the original 4‑D layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (float32 or bool)
        - weights_dict : map name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------
    # Unpack and clean inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5   # LayerNorm epsilon (matches PyTorch default)

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B,N,N,dim]  float32

    # -------------------------------------------------
    # Cast to fp16 for the expensive ops
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Fuse projection + gate for the *left* side
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # [2*H, dim]

    left_combined = torch.nn.functional.linear(x, left_combined_w)   # [B,N,N,2*H]
    left_proj, left_gate = torch.split(left_combined, hidden_dim, dim=-1)  # each [B,N,N,H]
    left_gate = torch.sigmoid(left_gate)

    # -------------------------------------------------
    # 3️⃣ Fuse projection + gate for the *right* side
    # -------------------------------------------------
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # [2*H, dim]

    right_combined = torch.nn.functional.linear(x, right_combined_w)   # [B,N,N,2*H]
    right_proj, right_gate = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate)

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)    # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Apply hidden‑dim gates
    # -------------------------------------------------
    left = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Compute out_gate (still needed after the contraction)
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H]

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 in / fp16 out)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H], fp16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm + out‑gate
    # -------------------------------------------------
    # PyTorch implementation (fast enough and numerically identical)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate

    # -------------------------------------------------
    # 9️⃣ Final projection back to `dim`
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden]
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim], fp16

    # -------------------------------------------------
    # 10️⃣ Cast back to float32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
```
