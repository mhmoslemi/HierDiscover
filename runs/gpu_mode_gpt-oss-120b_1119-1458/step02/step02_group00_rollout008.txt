@triton.jit
def batch_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    off_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = off_i < N
    mask_j = off_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute K offsets
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_batch * stride_left_batch + off_i[:, None] * stride_left_i + off_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N) from right (j,k)
        # right has stride i (k), stride j (j)
        # We need B^T: index (k,j) => row = off_k, col = off_j
        b_ptrs = right_ptr + pid_batch * stride_right_batch + off_k[:, None] * stride_right_i + off_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Cast back to half
    out = acc.to(tl.float16)

    # Store
    out_ptrs = out_ptr + pid_batch * stride_out_batch + off_i[:, None] * stride_out_i + off_j[None, :] * stride_out_j
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, N2, dim = input_tensor.shape

    eps = 1e-5

    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # FP16 cast
    x_h = x_norm.to(torch.float16)

    # get weights in fp16
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left_proj)  # shape [bs,N,N,hidden_dim]
    right = F.linear(x_h, w_right_proj)

    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))

"""
Optimized outgoing Triangle‑Multiplicative Update (TriMul) – forward only.

Algorithm
---------
1. Input LayerNorm (FP32) -> cast to FP16.
2. Linear projections (left/right) and gate linear layers are computed in FP16.
   * left/right are masked (if a mask is present).
3. Fuse the element‑wise multiplication of the projected tensors with their
   respective sigmoid‑gates using a tiny Triton kernel (`apply_gates_kernel`).
4. Core N³ operation:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   This is a batched matrix‑multiply (B·H independent (N×N) GEMMs).
   We replace the naïve `torch.bmm` with a custom Triton kernel
   (`batched_gemm_kernel`) that tiles the matrices (BLOCK_M×BLOCK_N)
   and accumulates over K‑tiles in FP32 while the data stays in FP16,
   fully exploiting H100 Tensor‑Cores.
5. Restore the (B,N,N,H) layout, apply a second LayerNorm (FP16) and the
   output gate (again fused by a tiny Triton kernel `elementwise_mul_kernel`).
6. Final linear projection back to the original channel dimension and cast
   the result to FP32.

All heavy compute (steps 3 & 4 & 6) is performed on the GPU with Triton,
while light tensor‑wise ops (LayerNorms, sigmoids, linear‑projections) use
highly‑optimized PyTorch kernels.  The implementation works for any
combination of the test‑case dimensions (B∈{1,2}, N∈{128,256,512,1024},
C∈{128}, H∈{128,384,768}) and gracefully handles the presence or absence
of a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny element‑wise Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,          # fp16   (flattened)
    left_gate_ptr,     # fp16   same shape as left_ptr
    right_ptr,         # fp16   (flattened)
    right_gate_ptr,    # fp16   same shape as right_ptr
    numel: tl.int64,   # total number of elements in each tensor
    BLOCK: tl.constexpr,
):
    """
    In‑place: left  = left  * left_gate
              right = right * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def elementwise_mul_kernel(
    a_ptr,            # fp16 input  (flattened)
    b_ptr,            # fp16 input  (flattened)
    out_ptr,          # fp16 output (flattened)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    out = a * b   (both flattened)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM kernel that implements the N³ “einsum”
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    left_ptr,          # fp16, shape (B*H, N, N)
    right_ptr,         # fp16, same shape
    out_ptr,           # fp16, same shape
    total_batches: tl.int64,
    N: tl.int64,
    # strides for the left matrix
    stride_l_batch: tl.int64,
    stride_l_i: tl.int64,
    stride_l_k: tl.int64,
    # strides for the right matrix (we read it transposed)
    stride_r_batch: tl.int64,
    stride_r_i: tl.int64,
    stride_r_j: tl.int64,
    # strides for the output matrix
    stride_o_batch: tl.int64,
    stride_o_i: tl.int64,
    stride_o_j: tl.int64,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for each batch `b` and head `h`:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
    (i.e. a batched GEMM where the right operand is implicitly transposed).
    The kernel tiles the output (i,j) plane with BLOCK_M×BLOCK_N tiles,
    iterates over the reduction dimension k in BLOCK_K steps,
    and accumulates in FP32 for numerical stability.
    """
    pid_batch = tl.program_id(0)        # 0 … total_batches‑1
    pid_m = tl.program_id(1)           # tile row index
    pid_n = tl.program_id(2)           # tile column index

    # ------------------- compute tile offsets -------------------
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base offset for this batch/head
    batch_offset = pid_batch * stride_l_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- reduction over k ----------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ----- load A tile (left) : shape (BLOCK_M, BLOCK_K) -----
        a_ptrs = left_ptr + batch_offset \
                 + i[:, None] * stride_l_i \
                 + k_idx[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ----- load B tile (right, transposed) : shape (BLOCK_K, BLOCK_N) -----
        # right is stored as (batch, row=i=k, col=j) – we need (k, j)
        b_ptrs = right_ptr + batch_offset \
                 + k_idx[:, None] * stride_r_i \
                 + j[None, :] * stride_r_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # ----- multiply‑accumulate (Tensor‑Core) -----
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N) in FP32

    # ------------------- write results -------------------------
    out_ptrs = out_ptr + batch_offset \
               + i[:, None] * stride_o_i \
               + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
        - weights      : dict of torch.Tensor parameters (see skeleton)
        - config       : dict containing at least:
            * "dim"         – C  (input channel size)
            * "hidden_dim" – hidden dimension H
            * "nomask"      – bool, True if the mask should be ignored

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32
    x_h = x.to(torch.float16)  # half precision for the heavy work

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to half precision (biases are all None)
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, w_left_proj)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, w_right_proj) # same shape

    # ------------------------------------------------------------------
    # 4️⃣ Gating logits (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Fuse left/right with their respective gates (tiny Triton kernel)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536  # ~64 Ki elements per program (tuned for H100)
    grid = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # gates are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ operation – batched GEMM using a custom Triton kernel
    # ------------------------------------------------------------------
    # reshape to (batch, head, N, N) → (total_heads, N, N)
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    total_heads = B * hidden_dim

    left_mat = left.view(total_heads, N, N)   # fp16
    right_mat = right.view(total_heads, N, N) # fp16
    out_mat = torch.empty_like(left_mat)      # fp16, will hold the N³ result

    # stride information (in elements, not bytes)
    stride_l_batch, stride_l_i, stride_l_k = left_mat.stride()
    stride_r_batch, stride_r_i, stride_r_j = right_mat.stride()  # j‑stride == col‑stride
    stride_o_batch, stride_o_i, stride_o_j = out_mat.stride()

    # tile sizes – good defaults for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_gemm = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_gemm_kernel[grid_gemm](
        left_mat,
        right_mat,
        out_mat,
        total_heads,
        N,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_i,
        stride_r_j,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Restore the original layout: [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑projection LayerNorm (half‑precision weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (another tiny Triton kernel)
    # ------------------------------------------------------------------
    out_gate = out_gate.contiguous()
    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (fp16 → fp16)
    # ------------------------------------------------------------------
    final = F.linear(out_mul, w_to_out)  # [B, N, N, dim] fp16

    # Return the expected float32 tensor
    return final.to(torch.float32)
