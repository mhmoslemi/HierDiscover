"""
Highly‑optimized forward pass for the **outgoing** TriMul module (AlphaFold‑3).

Algorithm
---------
1.  Layer‑norm the input (float32 for numeric stability).
2.  Cast to fp16 and apply the four linear projections (left/right, gates)
    – all are dense matmuls performed by cuBLAS (torch.nn.functional.linear).
3.  Optional pair‑wise mask is broadcast and applied in fp16.
4.  Fuse gating: left  = (x·W_left_proj) * sigmoid(x·W_left_gate)
                right = (x·W_right_proj) * sigmoid(x·W_right_gate)
5.  **Core N³ contraction**  
    out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]    
    This is a batched GEMM with batch = B·H, M = N, N = N, K = N.
    We implement it with a custom Triton kernel that:
        * loads tiles of A (left) and Bᵀ (right) in fp16,
        * accumulates in fp32,
        * writes the result in fp16,
    thereby avoiding the launch‑overhead of cuBLAS’s many‑small‑gemm loop.
6.  Layer‑norm over the hidden dimension H.
7.  Compute the output gate (sigmoid of a third linear projection) and
    multiply it element‑wise.
8.  Final linear projection back to the original channel dimension C.
9.  Return the result in float32.

The heavy lifting (the N³ matmul) is performed by `tri_mul_gemm_kernel`.
All other ops use torch’s highly‑optimized kernels, keeping the implementation
compact while still meeting the target runtime (≈ 1 ms for the largest test).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
#   Computes:  C = A @ Bᵀ   where A,B ∈ ℝ^{B·H × N × N}
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                # tensor pointers (fp16)
    batch, M, N, K,                     # int scalars
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ for the TriMul step."""
    pid_batch = tl.program_id(2)            # batch × head dimension
    pid_m = tl.program_id(0)                # output‑row tile
    pid_n = tl.program_id(1)                # output‑col tile

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch, i, k]   – regular layout (row‑major)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Bᵀ[batch, k, j] = B[batch, j, k]   – we read B transposed
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate the dot product over the K‑tile
        acc += tl.dot(a, b)                 # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write C = acc (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : model weights (float32)
        - config_dict  : must contain ``hidden_dim`` (int)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    H = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Utility: convert a weight tensor to fp16 on the target device
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16, non_blocking=True)

    # ------------------------------------------------------------------
    # Load model weights (all linear weights are bias‑free)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_h, left_proj_w)          # [B, N, N, H]
    right_proj = F.linear(x_h, right_proj_w)         # [B, N, N, H]

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction via the Triton GEMM kernel
    #    C[b·H+i, j] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    #   a) reshape to (B·H, N, N)
    left_view  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch_heads = B * H
    left_flat  = left_view.view(batch_heads, N, N)   # (B·H, N, N)
    right_flat = right_view.view(batch_heads, N, N)  # (B·H, N, N)

    #   b) allocate output buffer
    out_flat = torch.empty_like(left_flat)           # fp16

    #   c) launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_heads,               # batch × head
    )
    # Strides for A (left_flat)   – batch, row (M), col (K)
    stride_a_batch = left_flat.stride(0)
    stride_a_row   = left_flat.stride(1)
    stride_a_col   = left_flat.stride(2)

    # Strides for B (right_flat) – we read it transposed (Bᵀ)
    stride_b_batch = right_flat.stride(0)
    # Bᵀ:   K‑dimension uses original column stride (1)
    stride_b_k = right_flat.stride(2)   # = 1
    #   N‑dimension uses original row stride (N)
    stride_b_n = right_flat.stride(1)   # = N

    # Strides for C (out_flat)
    stride_c_batch = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)
    stride_c_col   = out_flat.stride(2)

    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_heads,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    #   d) reshape back to the original 4‑D shape
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(H,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=eps,
    )                                                     # [B, N, N, H] (float32)

    # ------------------------------------------------------------------
    # 8️⃣ Output gate and final projection back to dim
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, H] (fp16)
    gated = out_norm * out_gate                            # broadcast multiplication

    # Final linear projection (hidden → dim)
    final = F.linear(gated, to_out_w)                     # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Return in float32 as required
    # ------------------------------------------------------------------
    return final.to(torch.float32)
```
