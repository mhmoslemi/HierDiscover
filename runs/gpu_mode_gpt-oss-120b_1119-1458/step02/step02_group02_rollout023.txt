We'll define:

```
@triton.jit
def fused_pre_gemm_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, hidden,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * hidden
    mask = offs < total

    # Compute linear indices for positions
    idx = offs

    d = idx % hidden
    idx2 = idx // hidden

    j = idx2 % N
    idx3 = idx2 // N

    i = idx3 % N
    b = idx3 // N

    # Compute mask index (just b,i,j)
    mask_idx = ((b * N) + i) * N + j

    # Compute pointers (since contiguous).
    # Offsets inside left_proj: linear index = idx.
    left_proj = tl.load(left_proj_ptr + idx, mask=mask, other=tl.float16(0))
    left_gate_log = tl.load(left_gate_ptr + idx, mask=mask, other=tl.float16(0))
    right_proj = tl.load(right_proj_ptr + idx, mask=mask, other=tl.float16(0))
    right_gate_log = tl.load(right_gate_ptr + idx, mask=mask, other=tl.float16(0))
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0))

    # Compute sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_log))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_log))

    left_out = left_proj * mask_val * left_gate
    right_out = right_proj * mask_val * right_gate

    tl.store(left_out_ptr + idx, left_out, mask=mask)
    tl.store(right_out_ptr + idx, right_out, mask=mask)
```

@triton.jit
def fused_pre_gemm_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, hidden,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * hidden
    mask = offs < total

    # get indices
    idx = offs
    # compute coords
    d = idx % hidden
    idx_i = idx // hidden
    j = idx_i % N
    idx2 = idx_i // N
    i = idx2 % N
    b = idx2 // N

    # compute offsets
    mask_idx = ((b * N + i) * N + j)

    # load values
    left_proj = tl.load(left_proj_ptr + idx, mask=mask, other=tl.float16(0))
    left_gate_log = tl.load(left_gate_ptr + idx, mask=mask, other=tl.float16(0))
    right_proj = tl.load(right_proj_ptr + idx, mask=mask, other=tl.float16(0))
    right_gate_log = tl.load(right_gate_ptr + idx, mask=mask, other=tl.float16(0))
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0))

    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_log))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_log))

    # compute out
    left_out = left_proj * mask_val * left_gate
    right_out = right_proj * mask_val * right_gate

    tl.store(left_out_ptr + idx, left_out, mask=mask)
    tl.store(right_out_ptr + idx, right_out, mask=mask)

def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, hidden, N,
    stride_batch, stride_i, stride_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b_base = b_ptr + b_idx * stride_batch + h_idx
    c_base = c_ptr + b_idx * stride_batch + h_idx

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) – fast Triton implementation
================================================

Algorithm
---------
1. Layer‑norm on the last dimension of the input (float32 → float16).
2. Single large linear projection that produces the three projections
   (left/right) and the three gates (left/right/out) in one matmul.
3. Fuse mask, projection and sigmoid gate using a custom Triton kernel.
   The kernel writes both left‑scaled and right‑scaled tensors directly
   to contiguous memory, avoiding any extra copies.
4. Batched matrix multiply (out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d])
   performed by a Triton GEMM kernel that treats the
   (batch * hidden) dimension as the batch axis and uses tensor‑core
   dot products (fp16 → fp32 accumulation).  No `permute`+`contiguous`
   shuffling is required.
5. Output layer‑norm + out‑gate + final linear projection (all in fp16).
   The result is cast back to the original dtype (float32).

The implementation uses two Triton kernels:
* `fused_scale_gate_masked` – applies mask, sigmoid gate and projection
  for left and right tensors in a single memory pass.
* `batched_gemm_outgoing` – computes the O(N³) contraction without
  materialising intermediate transposes.

Both kernels are written for half‑precision (fp16) and accumulate in
fp32 to preserve numerical stability on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask, projection, sigmoid gate for left & right tensors
# ----------------------------------------------------------------------
@triton.jit
def fused_scale_gate_masked(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, hidden,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * hidden
    mask = offs < total

    # decode linear index -> (b,i,j,d)
    d = offs % hidden
    idx2 = offs // hidden
    j = idx2 % N
    idx3 = idx2 // N
    i = idx3 % N
    b = idx3 // N

    # mask index (no hidden dimension)
    mask_idx = ((b * N + i) * N + j)

    # load raw tensors
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate_log = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate_log = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    # sigmoid (1/(1+exp(-x))) in fp16
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_log))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_log))

    # fused scaling
    left_out = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def fused_scale_gate_masked_launch(left_proj, left_gate_log,
                                   right_proj, right_gate_log,
                                   mask):
    """Launch the fused scaling kernel (mask is fp16)."""
    B, N, _, hidden = left_proj.shape
    total = B * N * N * hidden
    BLOCK = 65536  # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    fused_scale_gate_masked[grid](
        left_proj, left_gate_log,
        right_proj, right_gate_log,
        mask,
        left_out, right_out,
        B, N, hidden,
        BLOCK_SIZE=BLOCK,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the outgoing contraction
#    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing(
    a_ptr, b_ptr, c_ptr,
    B, hidden, N,
    stride_batch, stride_i, stride_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # combined (batch * hidden) axis
    pid_m = tl.program_id(1)       # i‑tile
    pid_n = tl.program_id(2)       # j‑tile

    # decode batch & hidden indices
    b_idx = pid_batch // hidden
    h_idx = pid_batch % hidden

    # base pointers for this (b, h) slice (hidden dim is innermost, stride=1)
    a_base = a_ptr + b_idx * stride_batch + h_idx
    b_base = b_ptr + b_idx * stride_batch + h_idx
    c_base = c_ptr + b_idx * stride_batch + h_idx

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # Pointers to the current tile
    a_ptrs = a_base + offs_m[:, None] * stride_i + offs_k[None, :] * stride_j
    b_ptrs = b_base + offs_k[:, None] * stride_j + offs_n[None, :] * stride_i  # note transposition (k,j) -> (j,k)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))
        acc += tl.dot(a, b)               # fp16 → fp32 accumulation
        a_ptrs += BLOCK_K * stride_j      # advance K‑tile for A
        b_ptrs += BLOCK_K * stride_j      # advance K‑tile for B_T

    # Write the result tile
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_m[:, None] * stride_i + offs_n[None, :] * stride_j
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_outgoing_launch(left, right):
    """Launch the outgoing GEMM kernel (left/right are fp16 tensors)."""
    B, N, _, hidden = left.shape
    total_batches = B * hidden

    # strides for tensors of shape (B, N, N, hidden) in memory order
    stride_batch = N * N * hidden          # stride to next batch element
    stride_i = N * hidden                  # stride over the first N axis
    stride_j = hidden                      # stride over the second N axis

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    out = torch.empty_like(left)  # same shape [B,N,N,hidden], fp16

    batched_gemm_outgoing[grid](
        left, right, out,
        B, hidden, N,
        stride_batch, stride_i, stride_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor:  shape [B, N, N] (bool/float, on CUDA) or None
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim",
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    device = inp.device

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (keep result in fp16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Single linear pass that produces left/right projections and all gates
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices vertically
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # (5*hidden, dim)

    # No bias in the original module
    combined = F.linear(x_norm, w_concat)          # [B, N, N, 5*hidden]  (fp16)

    # Split the concatenated dimension back into the five parts
    left_proj_raw, left_gate_log, right_proj_raw, right_gate_log, out_gate_log = torch.split(
        combined, hidden, dim=-1
    )                                               # each: [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3) Mask + gate fusion (Triton) – only when a mask is present
    # ------------------------------------------------------------------
    if nomask or mask is None:
        # No mask: simple element‑wise scaling with sigmoid gates
        left_gate = torch.sigmoid(left_gate_log)
        right_gate = torch.sigmoid(right_gate_log)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask to fp16 once, keep it contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = fused_scale_gate_masked_launch(
            left_proj_raw, left_gate_log,
            right_proj_raw, right_gate_log,
            mask_fp16,
        )

    # --------------------------------------------------------------
    # 4) Output gate (separate – used after the contraction)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_log)    # [B, N, N, hidden] (fp16)

    # ------------------------------------------------------------------
    # 5) Core contraction: batched GEMM (out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d])
    # ------------------------------------------------------------------
    out = batched_gemm_outgoing_launch(left, right)   # [B, N, N, hidden] fp16

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gating and final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    out_norm = out_norm * out_gate                     # apply output gate

    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    out_final = F.linear(out_norm, to_out_w)               # [B,N,N,dim] fp16

    # Cast back to the original dtype (usually float32) and return
    return out_final.to(inp.dtype)
