"""
Outgoing TriMul (AlphaFold‑3) forward pass, Triton‑accelerated.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input over the feature axis (float32 for numeric stability).
2️⃣ Cast the normalized tensor to fp16 and apply all linear projections and sigmoid
   gates (all fp16).  When a mask is supplied it is broadcast and applied before
   the gates.
3️⃣ Fuse the per‑channel gates with the projected tensors.
4️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply A @ Bᵀ for each (batch, hidden) pair.
   Implemented with a custom Triton kernel that uses 64×64×64 tiles, Tensor‑Core
   MMA (fp16 × fp16 → fp32) and accumulates in FP32.
5️⃣ Second LayerNorm (fp16), multiply by the output gate and a final linear
   projection back to the original feature dimension.
6️⃣ Cast the result to float32 (required by the harness).

Only the O(N³) part is executed in Triton; the surrounding ops use high‑performance
PyTorch primitives and stay in fp16 to minimise memory traffic.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bj, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where each leading dimension (M) does A @ Bᵀ."""
    pid_m = tl.program_id(0)   # tile index along the first N dimension (i)
    pid_n = tl.program_id(1)   # tile index along the second N dimension (j)
    pid_z = tl.program_id(2)   # index over the combined batch*hidden dimension (M)

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows i
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols j
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles needed to cover the third dimension (seq_len)
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- Load A tile: shape (BLOCK_M, BLOCK_K) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---- Load B tile transposed: shape (BLOCK_K, BLOCK_N) ----
        # we read B as (k, j) so that the dot product yields (i, j)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # fp16 × fp16 → fp32 dot product, accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # -------------------- write output --------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  fp16
    right: torch.Tensor,  # [B, N, N, H]  fp16
) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using the Triton kernel
    above.  The tensors must be contiguous fp16 and reside on the same CUDA device.
    Returns a tensor of the same dtype/device, shape [B,N,N,H].
    """
    B, N, _, H = left.shape

    # Pack batch and hidden into a single leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element (not byte) units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile configuration tuned for H100 Tensor‑Core performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[
        torch.Tensor,           # input tensor   [B, N, N, dim]  (float32)
        torch.Tensor,           # mask tensor    [B, N, N]      (float32 or bool)
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------- unpack --------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------- 1️⃣ First LayerNorm (float32) --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # Cast to fp16 for the rest of the pipeline
    x = x_norm.to(torch.float16)

    # -------------------- 2️⃣ Cast all linear weights to fp16 --------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------- 3️⃣ Linear projections (fp16) --------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w)
    right = torch.nn.functional.linear(x, right_proj_w)

    # -------------------- 4️⃣ Optional mask (broadcast over hidden) --------------------
    if not nomask:
        # mask can be bool or float, we cast to fp16 and expand a trailing dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------- 5️⃣ Gating (sigmoid) --------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # -------------------- 6️⃣ Heavy O(N³) contraction via Triton --------------------
    # Result shape: [B, N, N, hidden_dim] (fp16)
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------- 7️⃣ Second LayerNorm (fp16) --------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------- 8️⃣ Final gating + projection --------------------
    out = out * out_gate                     # element‑wise (fp16)
    out = torch.nn.functional.linear(out, to_out_w)  # -> [B,N,N,dim] (fp16)

    # -------------------- 9️⃣ Cast back to float32 (required output) --------------------
    return out.to(torch.float32)
