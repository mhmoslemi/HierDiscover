"""
TriMul (outgoing) – fast forward pass

Algorithm
---------
1️⃣  Layer‑Norm on the input (float32).
2️⃣  Cast to fp16 and compute *both* the projection and its gate in a single
    matmul by concatenating the two weight matrices (no bias).  This halves the
    number of matmuls.
3️⃣  Fuse mask, projection, and gate with a tiny Triton kernel.  When `nomask`
    is True we simply multiply element‑wise (no kernel launch).
4️⃣  The core N³ operation
        out[i,j] = Σ_k left[i,k] * right[j,k]
    is a batch of independent GEMMs:
        Cₕ = Aₕ @ Bₕᵀ   for each hidden‑dim h .
    It is realised with a single `torch.bmm` on fp16 tensors (Tensor‑Core
    accelerated).
5️⃣  Convert back to float32, apply a second LayerNorm, the output gate and a
    final linear projection back to the original feature dimension.

Only the mask‑gate fusion is done in Triton; all heavy matrix multiplications
use cuBLAS/Tensor‑Cores for maximal throughput on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: mask * gate * projection  (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16*  [B,N,N,H]
    right_ptr,         # fp16*  [B,N,N,H]
    mask_ptr,          # fp16*  [B,N,N]
    left_gate_ptr,     # fp16*  [B,N,N,H]
    right_gate_ptr,    # fp16*  [B,N,N,H]
    # strides
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_m0, stride_m1, stride_m2,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    B, N, H,                     # compile‑time constants
    BLOCK_H: tl.constexpr,       # tile size for the hidden dimension
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets (without hidden dimension)
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2

    # mask scalar (broadcast across H)
    mask_val = tl.load(mask_ptr + off_m)

    # ------------------------------------------------------------------
    # Loop over hidden dimension
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        # load slices (vectorised over hidden dim)
        left       = tl.load(left_ptr      + off_l + offs * stride_l3,
                            mask=mask_h, other=tl.float16(0.0))
        right      = tl.load(right_ptr     + off_r + offs * stride_r3,
                            mask=mask_h, other=tl.float16(0.0))
        left_gate  = tl.load(left_gate_ptr + off_lg + offs * stride_lg3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr+ off_rg + offs * stride_rg3,
                            mask=mask_h, other=tl.float16(0.0))

        # fused op:  proj * gate * mask
        left_out  = left  * left_gate  * mask_val
        right_out = right * right_gate * mask_val

        # store back (in‑place)
        tl.store(left_ptr  + off_l + offs * stride_l3,  left_out,  mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (AlphaFold‑3).

    Args:
        data: (input, mask, weights, config)
            - input : torch.Tensor[B, N, N, dim]   (float32)
            - mask  : torch.Tensor[B, N, N]        (float32, may be dummy)
            - weights: dict of model parameters (float32)
            - config : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)
    eps = 1e-5
    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gate (single matmul per side)
    # ------------------------------------------------------------------
    # left side: projection + gate
    left_w_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, dim)
    left_all = F.linear(x_fp16, left_w_concat)      # [B,N,N,2*H]
    left       = left_all[..., :hidden_dim]         # proj
    left_gate  = torch.sigmoid(left_all[..., hidden_dim:])  # gate

    # right side: projection + gate
    right_w_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_all = F.linear(x_fp16, right_w_concat)   # [B,N,N,2*H]
    right      = right_all[..., :hidden_dim]
    right_gate = torch.sigmoid(right_all[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Mask & gate fusion
    # ------------------------------------------------------------------
    if not nomask:
        # ensure contiguous layout (required for correct stride handling)
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        B, N, _, _ = left.shape
        total_programs = B * N * N

        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left,               # left projection
            right,              # right projection
            mask_fp16,          # mask
            left_gate,          # left gate
            right_gate,         # right gate
            # strides for left
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            # strides for right
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # strides for mask (b,i,j)
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for left_gate
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # strides for right_gate
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            # sizes
            B, N, hidden_dim,
            BLOCK_H=32,          # 32 works for hidden_dim up to 384
            num_warps=4,
        )
        # after kernel launch, `left` and `right` already hold proj*gate*mask
    else:
        # no mask → simple elementwise multiplication
        left  = left  * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM  (Σₖ left[..., i,k,:] * right[..., j,k,:])
    #    For each hidden dimension we do a separate matrix multiply.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 batch GEMM – Tensor Cores handle fp32 accumulation internally
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: 2nd LayerNorm → out‑gate → final linear
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # fp32

    # output gate (already computed in fp16; cast to fp32)
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))
    out_norm = out_norm * out_gate.to(torch.float32)

    # final projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])   # fp32, shape [B,N,N,dim]

    return out_final
