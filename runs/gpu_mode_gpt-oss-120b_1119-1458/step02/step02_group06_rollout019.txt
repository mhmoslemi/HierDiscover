"""
Highly‑optimized outgoing TriMul implementation (AlphaFold‑3 style).

Algorithm
---------
1. **Layer‑norm** the input (float32) – cheap and memory‑bound.
2. **Half‑precision** conversion.
3. **Fused linear projection + gating**:
   * All five weight matrices (left_proj, right_proj,
     left_gate, right_gate, out_gate) are concatenated into a
     single matrix `W_all` of shape `(5*hidden_dim, dim)`.
   * One GEMM (`X @ W_all.T`) produces the three projections
     and the three gates in a single pass, cutting the number
     of reads of `X` from five to one.
   * Projections are masked (if a mask is present) and multiplied
     by their sigmoided gates.
4. **Batched N³ reduction** (`out = left @ rightᵀ`):
   * `left` and `right` are reshaped to `(B*hidden, N, N)` and
     multiplied using a tiny Triton GEMM kernel when `N ≤ 512`;
     otherwise a cuBLAS batched `torch.bmm` is used (both run on
     Tensor‑Cores, accumulate in fp32 and store as fp16).
5. **Output LayerNorm → out‑gate → final Linear** (float32).
6. Return a float32 tensor of shape `[B, N, N, dim]`.

Only the N³ reduction is performed in a custom Triton kernel;
all other heavy work is done with GPU‑accelerated cuBLAS kernels.
The fusion in step 3 reduces global‑memory traffic dramatically
and is the main source of the speed‑up over the reference
implementation.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM `C = A @ Bᵀ`  (A & B have shape (BH, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _triton_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM (A @ Bᵀ) accumulating into fp32."""
    pid_batch = tl.program_id(0)   # batch*head index
    pid_i = tl.program_id(1)        # tile row index
    pid_j = tl.program_id(2)        # tile col index

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over K dimension (the sequence length)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_batch * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            cache_hint=tl.constexpr(0),
        )
        # Bᵀ block: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_batch * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            cache_hint=tl.constexpr(0),
        )
        # Matrix‑multiply (Tensor‑Core) – result in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # Store result as fp16
    tl.store(
        c_ptr
        + pid_batch * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * stride_ck,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[:, None],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim] (float32)
            - mask: torch.Tensor, shape [B, N, N]  (bool/float)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0. unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # --------------------------------------------------------------
    # 2. Cast to FP16 and flatten (B*N*N, dim)
    # --------------------------------------------------------------
    x_h = x.half()
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                 # shape (B*N*N, dim)

    # --------------------------------------------------------------
    # 3. Fuse the five linear layers into a single GEMM
    #      W_all = [left_proj; right_proj; left_gate; right_gate; out_gate]
    # --------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()

    # concatenate – order must match the splits below
    w_all = torch.cat(
        [
            w_left_proj,
            w_right_proj,
            w_left_gate,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    )  # shape (5*hidden_dim, dim) – FP16

    # Single half‑precision GEMM
    all_proj = torch.nn.functional.linear(x_flat, w_all)  # (B*N*N, 5*hidden_dim)
    all_proj = all_proj.view(B, N, N, 5 * hidden_dim)    # [B, N, N, 5*H]

    # --------------------------------------------------------------
    # 4. Split into projections and gates; apply sigmoid to gates
    # --------------------------------------------------------------
    h = hidden_dim
    proj_left = all_proj[..., 0:h]                         # [B, N, N, H]
    proj_right = all_proj[..., h:2 * h]                    # [B, N, N, H]

    gate_left = all_proj[..., 2 * h:3 * h].sigmoid()       # [B, N, N, H]
    gate_right = all_proj[..., 3 * h:4 * h].sigmoid()      # [B, N, N, H]
    out_gate = all_proj[..., 4 * h:5 * h].sigmoid()       # [B, N, N, H]

    # --------------------------------------------------------------
    # 5. Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        proj_left = proj_left * mask_fp16
        proj_right = proj_right * mask_fp16

    # --------------------------------------------------------------
    # 6. Apply the element‑wise gates (still FP16)
    # --------------------------------------------------------------
    left = proj_left * gate_left
    right = proj_right * gate_right

    # --------------------------------------------------------------
    # 7. Prepare tensors for the N³ reduction
    #    (B, N, N, H) --> (B*H, N, N)
    # --------------------------------------------------------------
    # Move hidden dim to axis 1 and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)    # (BH, N, N)  fp16
    right_flat = right.view(B * hidden_dim, N, N)

    # --------------------------------------------------------------
    # 8. Batched Gram‑product: out = left @ rightᵀ
    #    Use a Triton kernel for N ≤ 512, otherwise cuBLAS.
    # --------------------------------------------------------------
    BH = B * hidden_dim
    if N <= 512:  # Triton is usually faster for modest sizes
        out_flat = torch.empty_like(left_flat)  # fp16

        # strides for the three tensors
        stride_a0, stride_a1, stride_a2 = left_flat.stride()   # (BH, N, 1)
        stride_b0, stride_b1, stride_b2 = right_flat.stride()
        stride_c0, stride_c1, stride_c2 = out_flat.stride()

        # Tuned tile sizes – work well on H100
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

        grid = (
            BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )
        _triton_tri_mul_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            N,
            stride_a0,
            stride_a1,
            stride_a2,
            stride_b0,
            stride_b1,
            stride_b2,
            stride_c0,
            stride_c1,
            stride_c2,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # cuBLAS batched GEMM – highly‑optimised on H100
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # --------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H) and cast to FP32
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()  # [B, N, N, H]  – float32

    # --------------------------------------------------------------
    # 10. Output LayerNorm + out‑gate
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()  # broadcasted multiplication

    # --------------------------------------------------------------
    # 11. Final linear projection back to original dim (float32)
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].float()
    out = torch.nn.functional.linear(out, w_to_out)  # [B, N, N, dim] – float32

    return out
