"""
Optimized outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1️⃣  Layer‑norm over the feature dimension (float‑32, FP32 stability).
2️⃣  Cast to FP16 and perform *two* fused linear projections:
    - left_proj  + left_gate  (shape hidden_dim)  →  single matmul with weight
      concatenated as [left_proj; left_gate].
    - right_proj + right_gate (shape hidden_dim) →  same trick.
    This halves the number of passes over the large input tensor.
3️⃣  Fuse element‑wise multiplication of projection, gate and (optional) mask
    with a single Triton kernel that updates both the left and right tensors
    in‑place.  This removes two kernel launches and reduces mask loads.
4️⃣  Core O(N³) operation: for each hidden channel compute
    C = A @ Bᵀ  (A,B ∈ ℝ^{N×N}) using a strided batched GEMM (torch.bmm on
    FP16 → FP32 accumulation).  This matches the original einsum but avoids
    explicit Python loops.
5️⃣  Post‑process: LayerNorm on the hidden dimension, multiply by the
    out‑gate, and a final linear projection back to the original feature
    dimension.  All remaining arithmetic stays in FP32 for numerical safety.

The heavy lifting is done with mixed‑precision Tensor‑Core GEMMs; only the
mask‑gate fusion is written in Triton to satisfy the “use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) for *both* left and right tensors
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    proj_left_ptr,      # fp16  left projection (flattened)
    gate_left_ptr,      # fp16  left gate (flattened)
    proj_right_ptr,     # fp16  right projection (flattened)
    gate_right_ptr,     # fp16  right gate (flattened)
    mask_ptr,           # fp16  mask (flattened, shape B*N*N)
    out_left_ptr,       # fp16  destination for gated left
    out_right_ptr,      # fp16  destination for gated right
    hidden_dim,         # i64   hidden dimension (repeat factor of mask)
    n_elements,         # i64   total number of elements (B*N*N*H)
    BLOCK: tl.constexpr,  # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < n_elements

    # Load the four tensors (FP16 → FP32 for higher‑precision multiply)
    proj_l = tl.load(proj_left_ptr + offs, mask=valid).to(tl.float32)
    gate_l = tl.load(gate_left_ptr  + offs, mask=valid).to(tl.float32)
    proj_r = tl.load(proj_right_ptr + offs, mask=valid).to(tl.float32)
    gate_r = tl.load(gate_right_ptr  + offs, mask=valid).to(tl.float32)

    # Mask is broadcast over the hidden dimension:
    # each mask entry repeats `hidden_dim` times.
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid).to(tl.float32)

    # Fuse the three element‑wise products
    left_out  = proj_l * gate_l * mask_val
    right_out = proj_r * gate_r * mask_val

    # Store back as FP16
    tl.store(out_left_ptr  + offs, left_out .to(tl.float16), mask=valid)
    tl.store(out_right_ptr + offs, right_out.to(tl.float16), mask=valid)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no backward needed).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N]       (float32 or bool)
        - weights      : dict of model weights (already on the correct device)
        - config       : dict containing at least:
            * "dim"        : feature dimension (C)
            * "hidden_dim": hidden dimension (C_z)
            * "nomask"    : bool – True means the mask is all‑ones

    Returns
    -------
    out : torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all subsequent linear / gating ops
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear + gate for left and right sides (two matmuls instead of four)
    # ------------------------------------------------------------------
    # concatenate weights: [proj; gate] → shape (2*hidden, dim)
    left_w_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_w_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # Linear pass (produces proj and gate concatenated)
    left_cat = F.linear(x_h, left_w_concat)   # [B, N, N, 2*hidden]
    right_cat = F.linear(x_h, right_w_concat) # [B, N, N, 2*hidden]

    # Split the concatenated tensors
    left_proj = left_cat[..., :hidden_dim]                               # fp16
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])                # fp16, sigmoid
    right_proj = right_cat[..., :hidden_dim]                             # fp16
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])              # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and fuse the element‑wise product using Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16 → flatten to (B*N*N)
        mask_fp = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)                     # shape B·N·N

        # In‑place fusion: results are written back into left_proj/right_proj
        n_elem = left_proj.numel()                       # B·N·N·H
        BLOCK = 65536
        grid = (triton.cdiv(n_elem, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_proj,                    # proj_left_ptr
            left_gate,                    # gate_left_ptr
            right_proj,                   # proj_right_ptr
            right_gate,                   # gate_right_ptr
            mask_flat,                    # mask_ptr
            left_proj,                    # out_left_ptr (in‑place)
            right_proj,                   # out_right_ptr (in‑place)
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        left = left_proj
        right = right_proj
    else:
        # No mask – simple in‑place element‑wise product
        left = left_proj.mul_(left_gate)
        right = right_proj.mul_(right_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³) operation: batch GEMM (A @ Bᵀ) for every hidden channel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, N, N, H
    # reshape to (B*H, N, N)  – need contiguous layout for cuBLAS
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched FP16×FP16 → FP32 GEMM (torch.bmm uses Tensor‑Cores on H100)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))  # [B*H, N, N] (fp16)

    # Restore original shape [B, N, N, hidden]  and promote to FP32
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, hidden] (FP32)

    # out_gate (computed on the original x_h)
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16))).to(torch.float32)

    out = out * out_gate                         # broadcast over hidden dim

    # Final projection back to the original feature dimension (float32)
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
