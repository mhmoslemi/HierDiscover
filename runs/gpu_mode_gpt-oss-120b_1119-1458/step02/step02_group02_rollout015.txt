```
@triton.jit
def pack_left_kernel(
    left_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, hidden,
    stride_lpb, stride_lpi, stride_lpj, stride_lph,
    stride_gpb, stride_gpi, stride_gpj, stride_gph,
    stride_mp, stride_mi, stride_mj,
    HAVE_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_k = tl.program_id(2)

    bh = pid_bh
    b = bh // hidden
    h = bh - b * hidden

    # Compute offsets for i and k
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_k = pid_k * BLOCK_N + tl.arange(0, BLOCK_N)

    mask = (offs_i < N) & (offs_k < N)

    # Compute linear offsets into left pointer
    # left_ptr offset = b*stride_lpb + offs_i*stride_lpi + offs_k*stride_lpj + h*stride_lph
    left_ptrs = left_ptr + b * stride_lpb + h * stride_lph + offs_i[:, None] * stride_lpi + offs_k[None, :] * stride_lpj
    left = tl.load(left_ptrs, mask=mask, other=0.0)

    gate_ptrs = gate_ptr + b * stride_gpb + h * stride_gph + offs_i[:, None] * stride_gpi + offs_k[None, :] * stride_gpj
    gate = tl.load(gate_ptrs, mask=mask, other=0.0)

    if HAVE_MASK:
        mask_ptrs = mask_ptr + b * stride_mp + offs_i[:, None] * stride_mi + offs_k[None, :] * stride_mj
        mask_val = tl.load(mask_ptrs, mask=mask, other=0.0)
        val = left * gate * mask_val
    else:
        val = left * gate

    # Write to out: out[bh, i, k] = val
    out_ptrs = out_ptr + bh * (N * N) + offs_i[:, None] * N + offs_k[None, :]
    tl.store(out_ptrs, val, mask=mask)
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the PyTorch reference but moves the two
memory‑intensive phases onto custom Triton kernels:

1. **Fused projection + gating** – A single linear layer (weights of the
   five linear layers are concatenated) produces the raw projections and
   the three gate logits.

2. **Mask & gate fusion + layout packing** – Two Triton kernels
   (`pack_left_kernel` and `pack_right_kernel`) read the raw projections,
   the corresponding gate values and the optional mask, compute the
   element‑wise product and at the same time pack the result into the
   contiguous layout required by the batched GEMM that implements the
   einsum `'... i k d, ... j k d -> ... i j d'`.  This eliminates the
   separate `*` passes, the `permute/.contiguous` copies and the extra
   memory allocation for the intermediate `left`/`right` tensors.

3. **Core contraction** – The packed tensors have shape
   `(B*hidden, N, N)`; a single `torch.bmm` (FP16) performs all
   `(B*hidden)` matrix multiplications on Tensor‑cores.

4. **Output layer‑norm, gating and final projection** – The result of the
   contraction is reshaped back to `(B, N, N, hidden)` and processed with
   the remaining normalisation, output‑gate multiplication and linear
   projection.

All heavy arithmetic is performed in FP16; the final result is cast back
to the input dtype (normally FP32).  The kernels use compile‑time
constants for block sizes and for the presence of a mask, so the mask
branch is completely removed when `nomask=True`.

This design reduces memory traffic (no extra copies) and launches only
two small Triton kernels, leaving the dominant `bmm` to the highly
optimised cuBLAS kernels on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels: pack projections + gates (optionally with mask) into the
# layout required by torch.bmm.
# ----------------------------------------------------------------------
@triton.jit
def pack_left_kernel(
    proj_ptr,       # raw left projection                          [B,N,N,hidden]
    gate_ptr,       # left gate values (sigmoid)                   [B,N,N,hidden]
    mask_ptr,       # mask (B,N,N) – may be dummy when HAVE_MASK=0
    out_ptr,        # packed output                               [BH,N,N]
    B, N, hidden,   # problem size
    # strides for the three input tensors
    stride_ppb, stride_ppi, stride_ppj, stride_pph,   # proj
    stride_gpb, stride_gpi, stride_gpj, stride_gph,   # gate
    stride_mp, stride_mi, stride_mj,                 # mask
    HAVE_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    # 0‑th axis -> (batch * hidden) dimension
    pid_bh = tl.program_id(0)
    pid_i  = tl.program_id(1)
    pid_k  = tl.program_id(2)

    # decode batch index and hidden index from the flattened dimension
    b = pid_bh // hidden
    h = pid_bh - b * hidden   # same as pid_bh % hidden

    # tile offsets inside the N×N matrix
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_k = pid_k * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_k = offs_k < N
    valid = mask_i[:, None] & mask_k[None, :]

    # ----- load projection -----
    proj_base = proj_ptr + b * stride_ppb + h * stride_pph
    proj = tl.load(
        proj_base + offs_i[:, None] * stride_ppi + offs_k[None, :] * stride_ppj,
        mask=valid,
        other=0.0,
    )

    # ----- load gate -----
    gate_base = gate_ptr + b * stride_gpb + h * stride_gph
    gate = tl.load(
        gate_base + offs_i[:, None] * stride_gpi + offs_k[None, :] * stride_gpj,
        mask=valid,
        other=0.0,
    )

    # ----- optional mask -----
    if HAVE_MASK:
        mask_base = mask_ptr + b * stride_mp
        mask_val = tl.load(
            mask_base + offs_i[:, None] * stride_mi + offs_k[None, :] * stride_mj,
            mask=valid,
            other=0.0,
        )
        val = proj * gate * mask_val
    else:
        val = proj * gate

    # ----- write packed output (BH, i, k) -----
    out_base = out_ptr + pid_bh * (N * N)
    tl.store(
        out_base + offs_i[:, None] * N + offs_k[None, :],
        val,
        mask=valid,
    )


@triton.jit
def pack_right_kernel(
    proj_ptr,       # raw right projection                         [B,N,N,hidden]
    gate_ptr,       # right gate values (sigmoid)                 [B,N,N,hidden]
    mask_ptr,       # mask (B,N,N) – may be dummy when HAVE_MASK=0
    out_ptr,        # packed output                               [BH,N,N]
    B, N, hidden,   # problem size
    stride_ppb, stride_ppi, stride_ppj, stride_pph,   # proj
    stride_gpb, stride_gpi, stride_gpj, stride_gph,   # gate
    stride_mp, stride_mi, stride_mj,                 # mask
    HAVE_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_k  = tl.program_id(1)   # row of the packed matrix = sum‑axis
    pid_j  = tl.program_id(2)   # column of the packed matrix = output column

    b = pid_bh // hidden
    h = pid_bh - b * hidden

    offs_k = pid_k * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_k = offs_k < N
    mask_j = offs_j < N
    valid = mask_k[:, None] & mask_j[None, :]

    # ----- load projection   (right[b, j, k, h]) -----
    proj_base = proj_ptr + b * stride_ppb + h * stride_pph
    proj = tl.load(
        proj_base + offs_j[:, None] * stride_ppi + offs_k[None, :] * stride_ppj,
        mask=valid,
        other=0.0,
    )

    # ----- load gate   (right_gate[b, j, k, h]) -----
    gate_base = gate_ptr + b * stride_gpb + h * stride_gph
    gate = tl.load(
        gate_base + offs_j[:, None] * stride_gpi + offs_k[None, :] * stride_gpj,
        mask=valid,
        other=0.0,
    )

    # ----- optional mask   (mask[b, j, k]) -----
    if HAVE_MASK:
        mask_base = mask_ptr + b * stride_mp
        mask_val = tl.load(
            mask_base + offs_j[:, None] * stride_mi + offs_k[None, :] * stride_mj,
            mask=valid,
            other=0.0,
        )
        val = proj * gate * mask_val
    else:
        val = proj * gate

    # ----- write packed output (BH, k, j) -----
    out_base = out_ptr + pid_bh * (N * N)
    tl.store(
        out_base + offs_k[:, None] * N + offs_j[None, :],
        val,
        mask=valid,
    )


# ----------------------------------------------------------------------
# Helper wrappers that launch the kernels
# ----------------------------------------------------------------------
def pack_left(proj, gate, mask, B, N, hidden, have_mask):
    """Pack left projection into (B*hidden, N, N) layout."""
    out = torch.empty((B * hidden, N, N), dtype=proj.dtype, device=proj.device)

    BLOCK_M = 64
    BLOCK_N = 64

    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride() if have_mask else (0, 0, 0)

    pack_left_kernel[grid](
        proj_ptr=proj,
        gate_ptr=gate,
        mask_ptr=mask,
        out_ptr=out,
        B=B,
        N=N,
        hidden=hidden,
        stride_ppb=s_proj[0],
        stride_ppi=s_proj[1],
        stride_ppj=s_proj[2],
        stride_pph=s_proj[3],
        stride_gpb=s_gate[0],
        stride_gpi=s_gate[1],
        stride_gpj=s_gate[2],
        stride_gph=s_gate[3],
        stride_mp=s_mask[0],
        stride_mi=s_mask[1],
        stride_mj=s_mask[2],
        HAVE_MASK=have_mask,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return out


def pack_right(proj, gate, mask, B, N, hidden, have_mask):
    """Pack right projection into (B*hidden, N, N) layout with swapped axes."""
    out = torch.empty((B * hidden, N, N), dtype=proj.dtype, device=proj.device)

    BLOCK_M = 64
    BLOCK_N = 64

    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride() if have_mask else (0, 0, 0)

    pack_right_kernel[grid](
        proj_ptr=proj,
        gate_ptr=gate,
        mask_ptr=mask,
        out_ptr=out,
        B=B,
        N=N,
        hidden=hidden,
        stride_ppb=s_proj[0],
        stride_ppi=s_proj[1],
        stride_ppj=s_proj[2],
        stride_pph=s_proj[3],
        stride_gpb=s_gate[0],
        stride_gpi=s_gate[1],
        stride_gpj=s_gate[2],
        stride_gph=s_gate[3],
        stride_mp=s_mask[0],
        stride_mi=s_mask[1],
        stride_mj=s_mask[2],
        HAVE_MASK=have_mask,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor:  shape [B, N, N] (bool or float, on CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim",
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Prepare FP16 tensors / weights
    # ------------------------------------------------------------------
    # cast input once to FP16 – LayerNorm can work in half precision
    inp_fp16 = inp.to(torch.float16)

    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp_fp16,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # x: [B,N,N,dim]  (fp16)

    # ------------------------------------------------------------------
    # 2) Fused projection + all five gates (single matmul)
    # ------------------------------------------------------------------
    # concatenate weight matrices vertically: shape (5*hidden, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*hidden, dim)

    combined = F.linear(x, w_concat)   # (B,N,N,5*hidden) – no bias

    # split back into the five components
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(
        combined, hidden, dim=-1
    )
    left_gate  = torch.sigmoid(left_gate_logit)   # [B,N,N,hidden] fp16
    right_gate = torch.sigmoid(right_gate_logit)  # [B,N,N,hidden] fp16
    out_gate   = torch.sigmoid(out_gate_logit)    # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 3) Pack projections + gates (optionally with mask) into layout required
    #    by the batched GEMM.  This fuses the three element‑wise multiplications
    #    (projection * gate * mask) and the transposition needed for the right side.
    # ------------------------------------------------------------------
    have_mask = 0 if nomask else 1
    if nomask:
        # dummy mask – never accessed inside the kernel when HAVE_MASK=0
        mask_fp16 = torch.empty((0,), dtype=torch.float16, device=inp.device)
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    left_mat  = pack_left(left_proj_raw, left_gate, mask_fp16, B, N, hidden, have_mask)
    right_mat = pack_right(right_proj_raw, right_gate, mask_fp16, B, N, hidden, have_mask)

    # ------------------------------------------------------------------
    # 4) Core contraction: out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]
    #    (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    # left_mat/right_mat: (B*hidden, N, N) in FP16
    out_flat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N)

    # reshape back to (B,N,N,hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5) Output layer‑norm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply the output gate (element‑wise)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out_final = F.linear(out_norm, to_out_weight)   # (B,N,N,dim)

    # Cast back to the original dtype (generally float32)
    return out_final.to(inp.dtype)
