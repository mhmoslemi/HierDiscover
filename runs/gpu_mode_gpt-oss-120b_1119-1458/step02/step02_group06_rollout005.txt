"""
Trion implementation of the outgoing TriMul operator.

The algorithm:
1. Layer‑norm input (float32) → half‑precision.
2. Fuse linear projections for left/right and their gates by concatenating weight
   matrices, thus only three GEMM calls (FP16, Tensor‑Core friendly).
3. Apply mask (if present) and element‑wise sigmoid gates, all in FP16.
4. Compute the heavy N³ reduction  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix‑multiply (left @ rightᵀ).  For modest batch*hidden sizes
   a custom Triton kernel does the GEMM entirely on Tensor‑Cores; otherwise we
   fall back to cuBLAS via `torch.bmm`.
5. Reshape back to (B,N,N,H), cast to FP32 and apply output LayerNorm,
   output gating and the final linear projection (still using FP32 for numerical
   stability).
All heavy arithmetic is performed in half‑precision (fp16) and accumulation
is done in fp32.
"""

import torch
import triton
import triton.language as tl
import math
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for left/right tensors.
# Currently A,B shape: (batch, M, K) where M=N, K=N.
# Accumulation in fp32; result stored fp16.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_am, stride_ak,  # A: (batch, M, K) with K = N
    stride_bm, stride_bn, stride_bk,  # B: (batch, N, K)
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    batch_id = tl.program_id(0)  # batch*hidden index
    pid_i = tl.program_id(1)     # tile row
    pid_j = tl.program_id(2)     # tile col

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_id * stride_am
            + offs_i[:, None] * stride_ak
            + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ block: shape (BLOCK_K, BLOCK_N)
        # B is stored as (batch, N, K). We need Bᵀ, i.e. (K, N).
        b = tl.load(
            b_ptr
            + batch_id * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # FP16 dot, accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # Store result (fp16)
    tl.store(
        c_ptr
        + batch_id * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    # config values
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    # Using torch.nn.functional.layer_norm directly because the tensor is 4‑D.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to half precision (fp16) for all subsequent ops
    # ------------------------------------------------------------------
    x_h = x.half()

    # ------------------------------------------------------------------
    # 3. Fuse linear projections + gating (three GEMMs)
    #    a) left_proj + left_gate -> (2*hidden_dim, dim)
    #    b) right_proj + right_gate -> (2*hidden_dim, dim)
    #    c) out_gate -> (hidden_dim, dim)
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Concatenate for fused projection+gate
    w_left_cat = torch.cat([w_left_proj, w_left_gate], dim=0).half()   # (2*H, dim)
    w_right_cat = torch.cat([w_right_proj, w_right_gate], dim=0).half()  # (2*H, dim)

    # 3a: left_proj + left_gate
    left_cat = F.linear(x_h, w_left_cat)  # [B, N, N, 2*H] fp16
    left_proj = left_cat[..., :hidden_dim]
    left_gate = left_cat[..., hidden_dim:].sigmoid()  # fp16

    # 3b: right_proj + right_gate
    right_cat = F.linear(x_h, w_right_cat)  # [B, N, N, 2*H] fp16
    right_proj = right_cat[..., :hidden_dim]
    right_gate = right_cat[..., hidden_dim:].sigmoid()  # fp16

    # 3c: out_gate
    out_gate = F.linear(x_h, w_out_gate.half()).sigmoid()  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and element‑wise gates (still fp16)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left_proj = left_proj * m
        right_proj = right_proj * m

    left = left_proj * left_gate   # [B,N,N,H] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the N³ reduction:
    #    Collapse batch and hidden dims -> (BH, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Convert to (B, H, N, N) by moving hidden dim before the two N dims.
    # No in‑place copy: .permute creates a view with changed strides.
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)

    # Flatten batch and hidden dim (B*H, N, N). This yields a contiguous view.
    left_flat = left_perm.reshape(B * hidden_dim, N, N)
    right_flat = right_perm.reshape(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 6. Batched matrix multiply: left_flat @ right_flatᵀ
    #    Use custom Triton kernel for modest BH, otherwise cuBLAS.
    # ------------------------------------------------------------------
    BH = B * hidden_dim
    use_triton = (BH <= 64) and (N <= 512)   # heuristic small enough for kernel

    if use_triton:
        out_flat = torch.empty_like(left_flat)  # fp16
        # strides for Triton kernel
        a_strides = left_flat.stride()
        b_strides = right_flat.stride()
        c_strides = out_flat.stride()
        # kernel launch config
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
        grid = (BH, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
        batched_gemm_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            N,
            a_strides[0],
            a_strides[2],  # stride over K (N)
            b_strides[0],
            b_strides[1],
            b_strides[2],
            c_strides[0],
            c_strides[1],
            c_strides[2],
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # torch.bmm will invoke the highly‑optimised cuBLAS batched GEMM.
        # We need to transpose the second operand because the formula uses
        # a dot over the *k* dimension.
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H) and cast to FP32 for the rest.
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N)          # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()       # (B, N, N, H)
    out = out.float()                                 # FP32

    # ------------------------------------------------------------------
    # 8. Output LayerNorm + gating (still FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()  # broadcast multiplication

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim (FP32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out

   We reshape the tensors to (B·H, N, N) and perform a batched
   matrix‑multiply (A @ Bᵀ).  For modest (B·H) we launch a custom
   Triton kernel that runs the GEMM entirely on Tensor‑Cores
   (fp16×fp16→fp32).  Otherwise we fall back to cuBLAS via
   torch.bmm.
5. Reshape back to (B,N,N,H), cast to fp32 and apply the output
   LayerNorm, the output gate and the final linear projection.
All heavy arithmetic is fp16; accumulation is fp32.  The function
returns a float‑32 tensor of shape [B, N, N, C].
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for left/right tensors.
# A, B: (batch, M, K) with M = K = N   (fp16)
# C: (batch, M, N)                     (fp16, accumulates in fp32)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_am, stride_ak,        # A : (batch, M, K)   strides (batch, row, col=1)
    stride_bm, stride_bn,        # B : (batch, N, K)   strides (batch, row(N), col=1)
    stride_cm, stride_cn,        # C : (batch, M, N)   strides (batch, row, col=1)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # batch * hidden index
    pid_i    = tl.program_id(1)   # tile row
    pid_j    = tl.program_id(2)   # tile col

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator (fp32 for stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Reduction over K (the sequence dimension)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_id * stride_am
            + offs_i[:, None] * stride_ak
            + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ block: (BLOCK_K, BLOCK_N).  B is stored as (batch, N, K).
        b = tl.load(
            b_ptr
            + batch_id * stride_bm
            + offs_k[:, None] * 1
            + offs_j[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Store result (fp16)
    # --------------------------------------------------------------
    tl.store(
        c_ptr
        + batch_id * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, C]   (float32)
            mask         : torch.Tensor [B, N, N]      (bool/float)
            weights      : dict of model parameters (float32)
            config       : dict containing "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)

    ##################################################################
    # 1. Input LayerNorm (float32)
    ##################################################################
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # [B, N, N, C] (fp32)

    ##################################################################
    # 2. Cast to fp16 for the rest of the pipeline
    ##################################################################
    x_h = x.half()          # [B, N, N, C] (fp16)

    ##################################################################
    # 3. Fused linear projections + gates (three GEMMs, fp16)
    ##################################################################
    #   left_proj + left_gate
    w_left_cat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()               # (2*H, C)

    #   right_proj + right_gate
    w_right_cat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()               # (2*H, C)

    #   out_gate (single GEMM)
    w_out_gate = weights["out_gate.weight"].half()    # (H, C)

    # fused left projection & gate
    left_cat = F.linear(x_h, w_left_cat)              # [B, N, N, 2*H] (fp16)
    left_proj = left_cat[..., :hidden_dim]
    left_gate = left_cat[..., hidden_dim:].sigmoid()   # (fp16)

    # fused right projection & gate
    right_cat = F.linear(x_h, w_right_cat)            # [B, N, N, 2*H] (fp16)
    right_proj = right_cat[..., :hidden_dim]
    right_gate = right_cat[..., hidden_dim:].sigmoid()  # (fp16)

    # out gate
    out_gate = F.linear(x_h, w_out_gate).sigmoid()    # [B, N, N, H] (fp16)

    ##################################################################
    # 4. Mask (if present) and element‑wise gating
    ##################################################################
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)      # [B, N, N, 1]
        left_proj = left_proj * m
        right_proj = right_proj * m

    left  = left_proj * left_gate                     # [B, N, N, H] (fp16)
    right = right_proj * right_gate                   # [B, N, N, H] (fp16)

    ##################################################################
    # 5. Prepare for the N³ reduction (batched GEMM)
    ##################################################################
    B, N, _, _ = left.shape
    # Move hidden dimension before the two N dimensions (no copy, just a view)
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)  # (B, H, N, N)

    # Collapse batch & hidden => (B*H, N, N) – still contiguous in memory
    left_flat  = left_perm.reshape(B * hidden_dim, N, N)
    right_flat = right_perm.reshape(B * hidden_dim, N, N)

    ##################################################################
    # 6. Batched matrix multiplication: left_flat @ right_flatᵀ
    ##################################################################
    BH = B * hidden_dim
    # Heuristic: for relatively small BH and N we launch the custom kernel;
    # otherwise we rely on cuBLAS (torch.bmm) which is already heavily tuned.
    use_triton = (BH <= 64) and (N <= 512)

    if use_triton:
        out_flat = torch.empty_like(left_flat)   # fp16 result

        # Strides (in elements) required by Triton kernel
        a_batch, a_row, a_col = left_flat.stride()
        b_batch, b_row, b_col = right_flat.stride()
        c_batch, c_row, c_col = out_flat.stride()

        # Tuned tile sizes for NVIDIA H100 (Tensor‑Core friendly)
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )
        batched_gemm_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            N,
            a_batch,               # stride_am
            a_row,                 # stride_ak  (row stride)
            b_batch,               # stride_bm
            b_row,                 # stride_bn  (row stride of B, i.e. N dimension)
            c_batch,               # stride_cm
            c_row,                 # stride_cn  (row stride of C)
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # cuBLAS batched GEMM (fp16×fp16→fp16) – very fast for large BH/N.
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    ##################################################################
    # 7. Reshape back to (B, N, N, H) and cast to fp32
    ##################################################################
    out = out_flat.view(B, hidden_dim, N, N)               # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)
    out = out.float()                                      # fp32

    ##################################################################
    # 8. Output LayerNorm + gating (fp32)
    ##################################################################
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()   # broadcast over hidden dim

    ##################################################################
    # 9. Final linear projection back to the original dimensionality
    ##################################################################
    out = F.linear(out, weights["to_out.weight"])

    return out
