def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    # LayerNorm on dim
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # cast to fp16
    x_fp16 = x.to(torch.float16)

    # build combined weights
    left_combined_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_combined_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # proj+gate linear
    left_combined = F.linear(x_fp16, left_combined_w)  # [B,N,N,2*H]
    right_combined = F.linear(x_fp16, right_combined_w)

    # split
    left_proj = left_combined[..., :hidden_dim]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])

    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # apply gate
    left = left_proj * left_gate
    right = right_proj * right_gate

    # optional mask; multiply elementwise
    if not nomask:
        mask_fp16 = mask.to(torch.float16)
        # apply mask using triton kernel
        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]
        _apply_mask_kernel[(total_programs,)](
            left, right, mask_fp16,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            input_tensor.shape[0], input_tensor.shape[1], hidden_dim,
            BLOCK_H=32,
            num_warps=4,
        )
    # else: no mask, left/right already correct

    # compute out_gate
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w)) # [B,N,N,hidden_dim] fp16

    # batched GEMM: reshape left/right as (B*hidden_dim,N,N)
    B, N = input_tensor.shape[0], input_tensor.shape[1]

    # reshape
    left_mat = left.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))

    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # fp16

    # Cast to fp32
    out_fp32 = out.to(torch.float32)

    # second LayerNorm over hidden_dim
    out_norm = F.layer_norm(out_fp32,
                            (hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=eps)

    # gating
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear
    final = F.linear(out_norm, weights["to_out.weight"])

    return final

"""
Optimised forward pass for the **outgoing** Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑norm the input (float‑32) on the last dimension.
2️⃣ Cast to fp16 and fuse the *projection* and *gating* linear layers:
   – concatenate the projection‑ and gate‑weights for left/right branches,
   – a single `F.linear` yields both tensors, we split and apply `sigmoid`
     to the gating part.
3️⃣ Apply the multiplicative mask (if present) with a tiny Triton kernel
   that multiplies the gated tensors **in‑place** (no extra temporaries).
4️⃣ Core O(N³) step: for each head (hidden_dim) compute a batched matrix
   multiplication `C = A @ Bᵀ` where `A` and `B` are the gated left/right
   tensors reshaped to (B·hidden_dim, N, N).  Tensor‑Core accelerated
   `torch.bmm` does the heavy work.
5️⃣ Second Layer‑norm (hidden_dim), element‑wise gating with the
   out‑gate, and final linear projection back to `dim`.
All heavy arithmetic runs in fp16, the final result is returned in
float‑32.  At least one Triton kernel is used (mask‑application) to
satisfy the “custom kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (fp16 tensors)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    left_ptr,                     # fp16 * left tensor  [B,N,N,H]
    right_ptr,                    # fp16 * right tensor [B,N,N,H]
    mask_ptr,                     # fp16 * mask         [B,N,N]
    # strides for left / right (B,N,N,H)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (B,N,N)
    stride_m0, stride_m1, stride_m2,
    B, N, H,                      # compile‑time constants
    BLOCK_H: tl.constexpr,        # tile size along the hidden dimension
):
    pid = tl.program_id(0)                # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # scalar mask value (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # loop over the hidden dimension in BLOCK_H sized chunks
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        valid = offs < H

        # load a slice of left/right for this hidden‑dim tile
        l = tl.load(left_ptr + off_l + offs * stride_l3,
                    mask=valid, other=tl.float16(0.0))
        r = tl.load(right_ptr + off_r + offs * stride_r3,
                    mask=valid, other=tl.float16(0.0))

        # apply mask (broadcasted scalar)
        l = l * mask_val
        r = r * mask_val

        # store back in‑place
        tl.store(left_ptr + off_l + offs * stride_l3, l, mask=valid)
        tl.store(right_ptr + off_r + offs * stride_r3, r, mask=valid)


# ----------------------------------------------------------------------
# Entry‑point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:         [B, N, N]       (float32 or bool; optional)
            - weights: dict of model parameters (float32)
            - config: dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and fuse projection + gating for left/right
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # concatenate projection + gate weight matrices (out_features, in_features)
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)            # (2*H, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)            # (2*H, dim)

    # single linear per side
    left_combined = F.linear(x_fp16, left_combined_w)   # [B, N, N, 2*H] fp16
    right_combined = F.linear(x_fp16, right_combined_w) # [B, N, N, 2*H] fp16

    # split projection / gate parts
    left_proj = left_combined[..., :hidden_dim]                 # fp16
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])  # fp16

    right_proj = right_combined[..., :hidden_dim]                 # fp16
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:]) # fp16

    # free the concatenated tensors early
    del left_combined, right_combined

    # apply gating (element‑wise)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (in‑place multiplication via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dimension
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()

        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]
        _apply_mask_kernel[
            (total_programs,)
        ](
            left,
            right,
            mask_fp16,
            # left strides
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            # right strides
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # mask strides
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # problem sizes
            input_tensor.shape[0],               # B
            input_tensor.shape[1],               # N
            hidden_dim,                          # H
            BLOCK_H=32,                          # hidden‑dim tile size (covers 128‑dim heads)
            num_warps=4,
        )
        # left/right now contain the masked values

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ reduction via batched GEMM (fp16 → Tensor‑Cores)
    # ------------------------------------------------------------------
    B, N = input_tensor.shape[0], input_tensor.shape[1]

    # reshape to (B*H, N, N) where each head is a separate batch entry
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication: C = A @ Bᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore tensor layout: [B, N, N, hidden_dim] (fp16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 5️⃣ Output‑gate, second LayerNorm and final projection
    # ------------------------------------------------------------------
    # compute out‑gate from the **normalized** input (same x as above)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # fp16, shape [B,N,N,hidden_dim]

    # Cast the TriMul result back to float32 for the second LN
    out_f32 = out.to(torch.float32)

    # second LayerNorm over hidden_dim
    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # apply the output‑gate (cast to float32 for the multiplication)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to the original feature dimension (float32)
    final = F.linear(out_norm, weights["to_out.weight"])   # [B, N, N, dim]

    return final
