"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input over the feature axis `dim` (float32).
2. Cast to fp16 and flatten the tensor to [M, dim] where M = B·N·N.
3. Fuse the four linear projections (`left_proj`, `left_gate`,
   `right_proj`, `right_gate`) into two batched GEMMs:
        left_combined  = X @ [W_left_proj ; W_left_gate]^T
        right_combined = X @ [W_right_proj ; W_right_gate]^T
   Split the results, apply `sigmoid` to the gate parts and mask
   (if present) – all in fp16.
4. Compute `out_gate = sigmoid( X @ W_out_gate^T )` (single GEMM).
5. Reshape `left` / `right` to [B, N, N, H] (H = hidden_dim).
6. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed by a custom Triton kernel that executes a batched
   matrix‑multiply `A @ Bᵀ` for each (batch, hidden) pair using
   fp16 tensor‑core MMA and accumulates in FP32.  The kernel uses a
   128×128×64 tile (optimal for H100) and 8 warps.
7. Second LayerNorm on the hidden dimension (fp16).
8. Gating with `out_gate` and final linear projection back to `dim`.
9. Cast the final result to float32.

Only the forward pass is implemented; no gradients are required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batch‑wise matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile index along the first N dimension (i)
    pid_n = tl.program_id(1)   # tile index along the second N dimension (j)
    pid_z = tl.program_id(2)   # batch*hidden index

    # ----------- offsets within the current tile ------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundary handling (only needed for non‑divisible sizes)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 (higher precision for the sum over K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # load A tile   (shape BLOCK_M × BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # load B tile (transposed, shape BLOCK_K × BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ------------ write the result back ----------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]   fp16
    right: torch.Tensor,  # [B, N, N, H]   fp16
) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.  All tensors are expected to be
    contiguous fp16. Returns a tensor of the same dtype and device.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes tuned for H100 tensor‑core performance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as
        ``input_tensor``).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numeric stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, dim]  (float32)

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 and flatten for batched GEMMs
    # -------------------------------------------------
    x = x_norm.to(torch.float16)                     # fp16
    B, N, _, _ = x.shape
    M = B * N * N                                     # total “rows”

    x_flat = x.reshape(M, dim)                       # [M, dim]

    # -------------------------------------------------
    # 3️⃣ Fuse the four linear projections (proj + gate) for left/right
    # -------------------------------------------------
    # combine left_proj & left_gate weights -> [2*H, dim]
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # Batched GEMM (fp16)
    left_combined = torch.nn.functional.linear(x_flat, left_combined_w)   # [M, 2*H]
    right_combined = torch.nn.functional.linear(x_flat, right_combined_w) # [M, 2*H]

    # split into proj / gate streams
    left_proj = left_combined[:, :hidden_dim]          # [M, H]
    left_gate = torch.sigmoid(left_combined[:, hidden_dim:])   # [M, H]

    right_proj = right_combined[:, :hidden_dim]        # [M, H]
    right_gate = torch.sigmoid(right_combined[:, hidden_dim:]) # [M, H]

    # -------------------------------------------------
    # 4️⃣ Apply mask (if any) and gating
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).reshape(M, 1)   # [M,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left = left_proj * left_gate          # [M, H]
    right = right_proj * right_gate       # [M, H]

    # -------------------------------------------------
    # 5️⃣ Out‑gate (single GEMM + sigmoid)
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    )                                   # [M, H]

    # -------------------------------------------------
    # 6️⃣ Reshape tensors back to 4‑D layout
    # -------------------------------------------------
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 in/out)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # -------------------------------------------------
    # 9️⃣ Gating with out_gate and final projection back to dim
    # -------------------------------------------------
    out = out * out_gate                       # fp16
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden]
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 🔟 Cast back to float32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
