"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Key ideas
---------
1️⃣ **Fused linear projections** – left_proj+left_gate and right_proj+right_gate
   are computed with a single matrix‑multiply each (2 × instead of 4 linear
   layers).  This halves the memory traffic for the heavy
   B·N²·dim·hidden work.

2️⃣ **Mask‑gate fusion** – a tiny Triton kernel multiplies
   (projection × gate × mask) in FP16 (intermediate FP32) and writes the
   result back in‑place, avoiding extra Python‑level passes.

3️⃣ **Batched GEMM** – the O(N³) operation is a batch of independent
   matrix multiplications `C = A @ Bᵀ` with shape
   (B·hidden, N, N).  Tensor‑Core GEMM (`torch.bmm`) on fp16 is used,
   which is the fastest primitive on an H100.

4️⃣ **Mixed‑precision flow** – LayerNorm and the final linear are kept in
   FP32 for numerical stability, while the bulk of the computation runs
   in FP16 with FP32 accumulation.

The implementation respects the required signature
`custom_kernel(data) → torch.Tensor` and works for all test configurations
including optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: element‑wise projection × gate × mask   (FP16 → FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,          # fp16  projection (left or right)
    gate_ptr,          # fp16  gate tensor
    mask_ptr,          # fp16  mask (flattened B·N·N)
    out_ptr,           # fp16  destination (can be same as proj_ptr)
    hidden_dim,        # i64   hidden dimension (used for mask broadcasting)
    n_elements,        # i64   total number of elements = B·N·N·hidden_dim
    BLOCK: tl.constexpr,  # compile‑time block size (threads per program)
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load projection and gate (fp16 → fp32)
    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # Broadcast mask: each mask element repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    # Fuse the three multiplications
    val = proj * gate * m
    tl.store(out_ptr + offs, val.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entrypoint (required by the evaluation harness)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]  (float32)
        - mask         : torch.Tensor[B, N, N]       (float32 or bool)
        - weights      : dict of learned parameters
        - config       : dict containing "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 3️⃣ Fused projections + gates (2 linear runs instead of 4)
    # --------------------------------------------------------------
    # Combine left_proj & left_gate weights
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                # (2*hidden, dim)
    # Combine right_proj & right_gate weights
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                # (2*hidden, dim)

    # Linear: shape -> [B,N,N,2*hidden]
    left_comb = F.linear(x_h, left_comb_w)   # no bias
    right_comb = F.linear(x_h, right_comb_w)

    # Slice back to proj / gate
    left_proj = left_comb[..., :hidden]      # [B,N,N,hidden]
    left_gate = torch.sigmoid(left_comb[..., hidden:]).to(torch.float16)
    right_proj = right_comb[..., :hidden]
    right_gate = torch.sigmoid(right_comb[..., hidden:]).to(torch.float16)

    # out_gate (independent linear)
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    ).to(torch.float32)   # stay FP32 for the final scaling

    # --------------------------------------------------------------
    # 4️⃣ Apply mask and fuse projection×gate (Triton) if mask exists
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → FP16 flattened → repeat per hidden dim inside kernel
        mask_flat = mask.to(torch.float16).contiguous().view(-1)  # (B*N*N,)

        # Prepare output tensors – we will overwrite the projection tensors in‑place
        left = left_proj
        right = right_proj

        BLOCK = 1024                     # threads per program (nice warp multiple)
        n_elem = left.numel()            # B·N·N·hidden
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _proj_gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _proj_gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask – elementwise product; reuse the original buffers to save memory.
        left = left_proj
        right = right_proj
        left.mul_(left_gate)          # in‑place fp16 mul
        right.mul_(right_gate)

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³) operation: batch GEMM C = A @ Bᵀ
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # Batched matrix multiplication – uses Tensor Cores (FP16×FP16→FP32)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))  # [B*hidden, N, N] (FP16)

    # Reshape back to [B, N, N, hidden] and promote to FP32 for LayerNorm
    out = (
        out_b.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm, out‑gate scaling, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                 # [B,N,N,hidden] (FP32)

    out = out * out_gate               # broadcast over hidden

    # Final Linear back to original dimension (FP32)
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
