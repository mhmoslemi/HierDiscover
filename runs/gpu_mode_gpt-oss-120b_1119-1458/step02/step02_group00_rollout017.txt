"""
Highly‑optimized outgoing Triangle‑Multiplicative Update (TriMul).

Algorithm
---------
1.  Layer‑norm the input (float32 → float16) and project it to a hidden
    dimension `hidden_dim` with three linear layers (left/right/value
    projections).  No bias is used – the weight tensors are cast to FP16
    once.

2.  Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).

3.  Optional mask multiplication (broadcast over the hidden dim).

4.  Fuse the element‑wise multiplication of the left/right tensors with
    their respective gates using a tiny Triton kernel (`apply_gates_kernel`).

5.  Core O(N³) operation:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a *batched* GEMM in pure Triton
    (`tri_mul_gemm_kernel`).  The hidden dimension is treated as an
    extra batch axis (size = B × hidden_dim) and the kernel works
    directly on the original `[B,N,N,hidden_dim]` layout, thus avoiding
    the costly `permute(...).contiguous()` copy of the reference
    implementation.

6.  Post‑processing Layer‑Norm (`to_out_norm`), element‑wise multiplication
    with `out_gate` (again via a small Triton kernel), and the final
    linear projection back to `dim`.

All heavy arithmetic is performed in FP16 (accumulated in FP32) and the
final result is returned in FP32 as required by the reference model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny element‑wise kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,  # fp16, shape (total_elems,)
    right_ptr, right_gate_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place: left  = left  * left_gate
                 right = right * right_gate
       Operates on flattened tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """out = a * b  (both flattened)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM kernel for the N³ “einsum”  (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                           # problem sizes
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for each head h (0 ≤ h < H) and batch b:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    The hidden dimension is folded into the batch axis (total_heads = B*H).
    No transposes of the original tensor are required.
    """
    head_id = tl.program_id(0)          # 0 … B*H-1
    pid_m   = tl.program_id(1)          # tile row   (i)
    pid_n   = tl.program_id(2)          # tile col   (j)

    # recover batch and hidden‑head indices
    h = head_id % H
    b = head_id // H

    # tile start offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the beginning of the (b,h) slice
    left_base  = left_ptr  + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr + b * stride_r_batch + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_batch + h * stride_o_h

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for tk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (M × K) : left[i,k,h]
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile (K × N) : we need rightᵀ, i.e. right[j,k,h] → (k,j) layout
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        # GEMM accumulation
        acc += tl.dot(a, b_tile)   # (M,N) = (M,K) @ (K,N)

    # write the result back (FP16)
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask          : torch.Tensor      [B, N, N]       (float32 or bool)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   contains 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32
    x_h = x.to(torch.float16)  # fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to half‑precision once
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – all in fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)    # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right_proj)   # same shape

    # ------------------------------------------------------------------
    # 4️⃣ Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Fuse left/right with their gates (tiny Triton kernel)
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK_ELEM = 65536
    grid_gate = ( (numel + BLOCK_ELEM - 1) // BLOCK_ELEM, )
    apply_gates_kernel[grid_gate](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK_ELEM,
    )
    # free temporary gate tensors
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ update – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim
    total_heads = B * H

    # Strides of the original layout (no permute!)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h = right.stride()
    # Output will be allocated with the same layout as the inputs
    out = torch.empty_like(left, dtype=torch.float16, device=device)
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 fp16 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid_gemm = (
        total_heads,                     # batch‑head dimension
        (N + BLOCK_M - 1) // BLOCK_M,    # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,    # tiles over j
    )
    tri_mul_gemm_kernel[grid_gemm](
        left, right, out,
        B, N, H,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Post‑norm (FP16) on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # still fp16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by output gate (tiny Triton kernel)
    # ------------------------------------------------------------------
    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_mul = ( (numel_out + BLOCK_ELEM - 1) // BLOCK_ELEM, )
    elementwise_mul_kernel[grid_mul](
        out.view(-1),
        out_gate.contiguous().view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK_ELEM,
    )
    # free the gate tensor
    del out_gate

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (FP16)
    # ------------------------------------------------------------------
    final = F.linear(out_mul, w_to_out)  # [B, N, N, dim]  fp16

    # Return the result in the expected float32 format
    return final.to(torch.float32)
