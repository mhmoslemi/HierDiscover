def batched_outgoing_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    block_m = tl.program_id(1)
    block_n = tl.program_id(2)

    # compute start offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointers for left and right for each chunk of k
    # base pointers for this batch
    left_batch_ptr = left_ptr + batch_id * stride_l_batch
    right_batch_ptr = right_ptr + batch_id * stride_r_batch
    out_batch_ptr = out_ptr + batch_id * stride_o_batch

    # compute masks for m and n
    mask_m = offs_m < M
    mask_n = offs_n < N
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        cur_k = k * BLOCK_K
        offs_k_cur = cur_k + offs_k
        mask_k = offs_k_cur < K

        # pointers to left tile
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_m) + (offs_k_cur[None, :] * stride_l_k)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. **Input Layer‑Norm** – normalisation over the last dimension of the input.
2. **Fused projection + gating** – All five linear layers (two projections,
   three gates) are merged into a single matrix multiplication by stacking
   their weight matrices vertically.  The result is split back into the five
   components and the three gate tensors are turned into sigmoids.
3. **Mask & gate fusion** – When a mask is present a tiny Triton kernel
   (`fused_mul3`) computes ``out = a * b * c`` for three 1‑D tensors,
   fusing the element‑wise mask, projection and gate into one memory pass.
4. **Core contraction** – The outgoing update
   ``out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]`` is a batched matrix
   multiply for each hidden channel.  A custom Triton kernel
   (`batched_outgoing_matmul`) performs this GEMM for all batches (B×hidden)
   in half‑precision while accumulating in FP32.
5. **Output Layer‑Norm + gating + final linear** – The result is layer‑normed
   over the hidden dimension, multiplied by the output gate and finally
   projected back to the original channel dimension.

All heavy linear‑algebra steps run in FP16 (or BF16) and only the final
tensor is cast back to the input dtype (float32).  The implementation
uses a single custom Triton kernel for the triple element‑wise product
and another kernel for the batched matrix multiplication, giving a large
speed‑up over a naïve PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = a * b * c  (all tensors 1‑D contiguous)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,     # pointers
    size: tl.constexpr,               # total length
    BLOCK_SIZE: tl.constexpr,         # compile‑time constant
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    c = tl.load(c_ptr + offs, mask=mask, other=tl.float16(0.0))

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape, "All inputs must have the same shape"
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
    fused_mul3_kernel[grid](
        a, b, c, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the outgoing update
#   out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]   (right is transposed)
# ----------------------------------------------------------------------
@triton.jit
def batched_outgoing_matmul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    batch, M, N, K,                             # dimensions
    stride_l_batch, stride_l_m, stride_l_k,     # left strides
    stride_r_batch, stride_r_m, stride_r_k,     # right strides (original layout)
    stride_o_batch, stride_o_m, stride_o_n,     # output strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Each program processes a tile (BLOCK_M × BLOCK_N) of the output for a
    single batch (i.e. one (hidden, batch) element of the original tensor).
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # start indices for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    left_batch_ptr = left_ptr + batch_idx * stride_l_batch
    right_batch_ptr = right_ptr + batch_idx * stride_r_batch
    out_batch_ptr = out_ptr + batch_idx * stride_o_batch

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----------- load a tile from left (M × K) ----------
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_m) + (offs_k[None, :] * stride_l_k)
        mask_a = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=tl.float16(0.0))

        # ----------- load a tile from right (K × N) ----------
        # right is stored as (N, K); we need its transpose (K, N)
        b_ptrs = right_batch_ptr + (offs_k[:, None] * stride_r_m) + (offs_n[None, :] * stride_r_k)
        mask_b = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=tl.float16(0.0))

        # ----------- accumulate ----------
        acc += tl.dot(a, b)    # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # ----------- write out ----------
    c_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_m) + (offs_n[None, :] * stride_o_n)
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def batched_outgoing_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the outgoing TriMul contraction.
    ``left``  : [B*hidden, M, K]   (half)
    ``right`` : [B*hidden, N, K]   (half)   (original layout, not transposed)
    Returns ``out`` : [B*hidden, M, N] (half)
    The kernel computes out = left @ right^T.
    """
    # Shapes (all assume contiguous tensors)
    batch, M, K = left.shape
    _, N, _ = right.shape
    assert K == N, "inner dimensions must match (square matrices in TriMul)."

    out = torch.empty((batch, M, N), dtype=left.dtype, device=left.device)

    # Get strides (in elements, not bytes)
    stride_l_batch, stride_l_m, stride_l_k = left.stride()
    stride_r_batch, stride_r_m, stride_r_k = right.stride()
    stride_o_batch, stride_o_m, stride_o_n = out.stride()

    # Tunable block sizes – chosen to fit H100 shared‑memory and give high occupancy
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_outgoing_matmul_kernel[grid](
        left,
        right,
        out,
        batch,
        M,
        N,
        K,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_m,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor:  shape [B, N, N] (bool or float, on CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim",
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (FP32 → FP16 for the rest)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)      # half‑precision for the heavy parts

    # ------------------------------------------------------------------
    # 2) Fused projection + all five gates (single linear)
    # ------------------------------------------------------------------
    # Stack weight matrices vertically (shape: (5*hidden, dim))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Combined linear → shape [B, N, N, 5*hidden]
    combined = F.linear(x_norm, w_concat)   # bias=False

    # Split back into the five components
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(
        combined, hidden, dim=-1
    )

    # Sigmoid gates (still half‑precision)
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # ------------------------------------------------------------------
    # 3) Apply optional mask and gates (fused element‑wise multiplication)
    # ------------------------------------------------------------------
    if nomask:
        # No mask – simple binary element‑wise product
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Broadcast mask to hidden dimension and make it contiguous
        mask_exp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden).contiguous()
        left = fused_mul3(left_proj_raw, mask_exp, left_gate)
        right = fused_mul3(right_proj_raw, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 4) Core contraction (batched GEMM)
    # ------------------------------------------------------------------
    # Permute to shape (B, hidden, N, N) and flatten the batch & hidden dims
    left_mat = left.permute(0, 3, 1, 2).contiguous()     # [B, hidden, N, N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()   # transpose k↔j, still [B, hidden, N, N]

    left_flat = left_mat.view(B * hidden, N, N).contiguous()
    right_flat = right_mat.view(B * hidden, N, N).contiguous()

    # Batched matrix multiplication (out = left @ right^T) in half precision
    out_flat = batched_outgoing_matmul(left_flat, right_flat)    # [B*hidden, N, N]

    # Reshape back to 4‑D layout
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5) Output layer‑norm + gating + final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Return in the original dtype (float32)
    return out_final.to(inp.dtype)
