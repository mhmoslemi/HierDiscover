"""
Optimized outgoing Triangle Multiplicative Update (TriMul) kernel.

Algorithm
---------
1️⃣ Layer‑Norm over the input feature dimension (kept in FP16 after an
   FP32 norm for numerical stability).
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single GEMM – this reduces memory traffic because the
   flattened input is read only once.
3️⃣ Apply the sigmoid non‑linearity to the three gating tensors.
4️⃣ Element‑wise fuse “projection × gate × optional mask” with a tiny
   Triton kernel (different kernels for the masked and non‑masked case).
5️⃣ Core O(N³) operation: for every hidden channel compute a matrix
   multiplication A @ Bᵀ using a batched torch.bmm.  This call maps to an
   H100 Tensor‑Core GEMM (FP16 × FP16 → FP32 accumulation).
6️⃣ Layer‑Norm across the hidden dimension, multiply by the out‑gate,
   and finally project back to the original feature dimension.
All heavy arithmetic is performed in FP16, accumulation in FP32, and the
final result is returned as float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _mul_gate_kernel(
    proj_ptr,               # *fp16  projection (left/right)
    gate_ptr,               # *fp16  corresponding gate
    out_ptr,                # *fp16  destination
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    Computes out = proj * gate  (element‑wise) for n_elements elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements

    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    out = proj * gate
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)


@triton.jit
def _mul_gate_mask_kernel(
    proj_ptr,               # *fp16  projection (left/right)
    gate_ptr,               # *fp16  corresponding gate
    mask_ptr,               # *fp16  mask (flattened, shape B·N·N)
    out_ptr,                # *fp16  destination
    hidden_dim: tl.constexpr,
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    Computes out = proj * gate * mask  (mask is broadcast over the hidden
    dimension).  mask_idx = element_index // hidden_dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements

    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Arguments
    ---------
    data : Tuple[input, mask, weights, config]
        input   – Tensor[bs, N, N, dim] (float32)
        mask    – Tensor[bs, N, N]   (float32) – may be ignored if config["nomask"]=True
        weights – dict of model parameters (float32)
        config  – {"dim": int, "hidden_dim": int, "nomask": bool, ...}
    Returns
    -------
    Tensor[bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 → FP16 for the rest)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    x_h = x.to(torch.float16)  # now fp16 for all heavy work

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into one GEMM
    # ------------------------------------------------------------------
    # Cast all linear weights to fp16 once
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Concatenate the weight matrices (shape: (5*hidden_dim, dim))
    combined_w = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0
    )  # (5*H, dim)

    # One big linear pass → shape (B, N, N, 5*H)
    combined = F.linear(x_h, combined_w)

    # Split back into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each [B,N,N,H]

    # Apply sigmoid to the three gating tensors (still fp16)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse element‑wise projection×gate (×mask) with Triton
    # ------------------------------------------------------------------
    total_elems = left_proj.numel()  # B·N·N·H
    BLOCK = 65536
    grid = (triton.cdiv(total_elems, BLOCK),)

    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → fp16 → flatten
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # shape B·N·N

        _mul_gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
        )
        _mul_gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
        )
    else:
        # No mask – simple proj×gate fusion
        _mul_gate_kernel[grid](
            left_proj,
            left_gate,
            left,
            total_elems,
            BLOCK=BLOCK,
        )
        _mul_gate_kernel[grid](
            right_proj,
            right_gate,
            right,
            total_elems,
            BLOCK=BLOCK,
        )

    # Free large temporaries early
    del left_proj, left_gate, right_proj, right_gate, combined, combined_w, x, x_h

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³) operation – batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape = (B, N, N, H)

    # reshape to (B*H, N, N) where each hidden channel becomes a batch item
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (FP16×FP16 → FP32 accumulation)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))

    # Restore the original 4‑D layout, still in fp16
    out = (
        out_batched.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)  fp16

    # Release the large left/right tensors
    del left, right, left_batched, right_batched

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # fp16

    out = out * out_gate  # fp16 multiplication (broadcast over hidden)

    # Final linear projection back to original dimension (still fp16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)  fp16

    # Cast to float32 for the final output as required by the reference
    return out.to(torch.float32)
