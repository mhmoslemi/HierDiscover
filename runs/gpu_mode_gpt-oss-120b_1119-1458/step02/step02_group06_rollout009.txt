@triton.jit
def triton_batched_gemm(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_az, stride_ai, stride_ak, # a strides: batch*hidden (z?), row i, col k
    stride_bz, stride_bi, stride_bk,
    stride_cz, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)  # batch*hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for bounds
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(M, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < M

        # Load a: shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_ptr + batch * stride_az + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        # Load b: shape (BLOCK_K, BLOCK_N) but we need B[k,j] -> idx row=cur_k, col=offs_j
        b = tl.load(b_ptr + batch * stride_bz + cur_k[:, None] * stride_bi + offs_j[None, :] * stride_bk,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b).to(tl.float32)

    # Store result (fp16)
    tl.store(c_ptr + batch * stride_cz + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_ck,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

"""
Optimised “outgoing” TriMul kernel (AlphaFold‑3 style).

Algorithm
---------
1.  Layer‑norm the 4‑D input in FP32.
2.  Cast to FP16 and fuse the five per‑position linear projections
    (left_proj, left_gate, right_proj, right_gate, out_gate) into a
    single GEMM:   [B·N·N, dim] @ [5·hidden_dim, dim]ᵀ  →  [B·N·N,5·hidden_dim].
    The result is reshaped and split into the individual tensors.
3.  Apply sigmoid to the three gate tensors, fuse the element‑wise mask
    (if present) and multiply the gates into the projected tensors.
4.  Rearrange to (B·hidden, N, N) and compute the costly
    “Gram‑product’’  out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k] .
    For modest batch*hidden sizes a hand‑written Triton kernel is used;
    otherwise we fall back to torch.bmm (cuBLAS Tensor‑Core GEMM).
5.  Reshape back to (B,N,N,hidden) → FP32, Layer‑norm over the hidden dim,
    multiply by the out‑gate, and a final linear projection to the original
    dim (still in FP16, cast to FP32 at the end).
The heavy N³ reduction runs half‑precision → accumulated in FP32,
exploiting H100 Tensor‑Cores.  The fused linear stage reduces memory
traffic by reading the large input only once.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_batched_gemm(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_az, stride_ai, stride_ak,   # A: (batch, M, K) – stored as (BH, N, N)
    stride_bz, stride_bi, stride_bk,   # B: (batch, K, N) – we read Bᵀ (K, N) on‑the‑fly
    stride_cz, stride_ci, stride_ck,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiply C = A @ Bᵀ  (A,B shape (BH,N,N)).
    Accumulation is FP32, stores FP16.
    """
    batch = tl.program_id(0)      # 0‑th axis – batch*hidden dimension
    pid_i = tl.program_id(1)      # row tile
    pid_j = tl.program_id(2)      # col tile

    # ----- tile offsets -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over K (the sequence dimension) --------------------
    for k in range(0, tl.cdiv(M, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < M

        # Load A block (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_ptr
            + batch * stride_az
            + offs_i[:, None] * stride_ai
            + cur_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ block (BLOCK_K x BLOCK_N) → B[batch, cur_k, offs_j]
        b = tl.load(
            b_ptr
            + batch * stride_bz
            + cur_k[:, None] * stride_bi
            + offs_j[None, :] * stride_bk,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b).to(tl.float32)

    # ----- store result -------------------------------------------------
    tl.store(
        c_ptr
        + batch * stride_cz
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_ck,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (FP32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float)
            - weights      : dict of model weights (all tensors on the same GPU)
            - config       : dict containing at least:
                * dim         : input feature dimension
                * hidden_dim  : hidden projection dimension
                * nomask (optional) : if True mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (FP32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 and compute all per‑position linear projections together
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] – FP16
    B, N, _, _ = x_h.shape

    # flatten to (B·N·N, dim) for a single GEMM
    x_flat = x_h.view(B * N * N, dim)

    # half‑precision weight tensors (pre‑convert once)
    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()

    # concatenate into a single weight matrix of shape (5·hidden_dim, dim)
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    )  # (5*hidden_dim, dim)

    # fused linear projection
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B·N·N,5·H)
    combined_out = combined_out.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 3. Split & apply gates
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj_raw = combined_out[..., 0:h]
    left_gate_raw = combined_out[..., h : 2 * h]
    right_proj_raw = combined_out[..., 2 * h : 3 * h]
    right_gate_raw = combined_out[..., 3 * h : 4 * h]
    out_gate_raw = combined_out[..., 4 * h : 5 * h]

    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    # apply gates
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # optional mask (binary mask – 0/1)
    if not nomask:
        # ensure mask is FP16 and broadcastable
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Batched N³ reduction  (left @ rightᵀ)  – produce out of shape [B,N,N,H]
    # ------------------------------------------------------------------
    # move hidden dim to the batch axis: (B, H, N, N) -> (B·H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)   # FP16
    right_flat = right.view(BH, N, N)

    # Heuristic: use Triton for relatively small batch‑hidden sizes
    use_triton = (BH <= 64) and (N <= 512)

    if use_triton:
        out_flat = torch.empty_like(left_flat)  # FP16 output buffer

        # strides in elements (not bytes)
        a_batch, a_row, a_col = left_flat.stride()
        b_batch, b_row, b_col = right_flat.stride()
        c_batch, c_row, c_col = out_flat.stride()

        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,                                 # batch*hidden dimension
            (N + BLOCK_M - 1) // BLOCK_M,       # row tiles
            (N + BLOCK_N - 1) // BLOCK_N,       # col tiles
        )
        triton_batched_gemm[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            N,
            a_batch,
            a_row,
            a_col,
            b_batch,
            b_row,
            b_col,
            c_batch,
            c_row,
            c_col,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # torch.bmm maps to cuBLAS batched GEMM (Tensor‑Cores)
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim) – still FP16
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()  # cast to FP32 for the remaining ops

    # ------------------------------------------------------------------
    # 5. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was computed in FP16 – cast to FP32 for the multiply
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 6. Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].half()
    out = torch.nn.functional.linear(out, w_to_out)   # FP16 result
    out = out.float()  # final FP32 tensor

    return out
