Pseudo:

```
@triton.jit
def _fuse_proj_gate_mask(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_m0, stride_m1, stride_m2,
    B, N, H,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # offsets
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    mask_val = tl.load(mask_ptr + off_m) if HAS_MASK else tl.float16(1.0)

    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H
        # load
        left_raw = tl.load(left_ptr + off_l + offs * stride_l3, mask=mask_h, other=tl.float16(0.0))
        right_raw = tl.load(right_ptr + off_r + offs * stride_r3, mask=mask_h, other=tl.float16(0.0))
        left_gate_raw = tl.load(left_gate_ptr + off_lg + offs * stride_lg3, mask=mask_h, other=tl.float16(0.0))
        right_gate_raw = tl.load(right_gate_ptr + off_rg + offs * stride_rg3, mask=mask_h, other=tl.float16(0.0))

        # sigmoid: 1/(1+exp(-x))
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

        # fused multiplication
        left_out = left_raw * left_gate * mask_val
        right_out = right_raw * right_gate * mask_val

        # store
        tl.store(left_ptr + off_l + offs * stride_l3, left_out, mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)
```

Let's design the kernel signature:

```
@triton.jit
def _left_right_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_m0, stride_m1, stride_m2,
    B, N, H,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    # program id corresponds to each (b,i,j) entry
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # offsets base
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2

    mask_val = tl.full([1], 1.0, tl.float16)[0]  # default if no mask
    if HAS_MASK:
        off_m = b * stride_m0 + i * stride_m1 + j * stride_m2
        mask_val = tl.load(mask_ptr + off_m)

    # loop over hidden dim
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left_raw = tl.load(left_ptr + off_l + offs * stride_l3, mask=mask_h)
        right_raw = tl.load(right_ptr + off_r + offs * stride_r3, mask=mask_h)
        left_gate_raw = tl.load(left_gate_ptr + off_lg + offs * stride_lg3, mask=mask_h)
        right_gate_raw = tl.load(right_gate_ptr + off_rg + offs * stride_rg3, mask=mask_h)

        # sigmoid gate
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

        # fused multiplication
        left_out = left_raw * left_gate * mask_val
        right_out = right_raw * right_gate * mask_val

        # store back
        tl.store(left_ptr + off_l + offs * stride_l3, left_out, mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)
```

"""
Optimized Triton‑augmented forward pass for the *outgoing* Triangle
Multiplicative Update (TriMul) used in AlphaFold‑3.

Algorithm
---------
1️⃣ Layer‑norm the input (float32).
2️⃣ Cast to FP16 and perform **one fused linear** that jointly computes the
   four projections (left, right) and the three gates (left‑gate, right‑gate,
   out‑gate).  This reduces five separate `F.linear` calls to a single GEMM.
3️⃣ If a mask is present, a custom Triton kernel fuses
   `mask * sigmoid(gate) * projection` for *both* left and right tensors,
   performing the sigmoid and the element‑wise scaling in‑place.
   When there is no mask the sigmoid is applied with a cheap PyTorch call.
4️⃣ The core N³ operation  Σₖ left[..., i, k, h]·right[..., j, k, h] is
   realized as a batched half‑precision GEMM (one GEMM per head) using
   `torch.bmm`, which leverages Tensor‑Cores and FP32 accumulation.
5️⃣ Convert the result to float32, apply a second LayerNorm, multiply by the
   *out‑gate* (already sigmoid‑ed), and finish with a linear projection back
   to the original feature dimension.
All heavy work is done in FP16; only the two LayerNorms and the final linear
projection stay in FP32 to preserve numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid gates and projection tensors.
# -------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                     # fp16 * left projection (B,N,N,H)
    right_ptr,                    # fp16 * right projection (B,N,N,H)
    left_gate_ptr,                # fp16 * left gate pre‑sigmoid (B,N,N,H)
    right_gate_ptr,               # fp16 * right gate pre‑sigmoid (B,N,N,H)
    mask_ptr,                     # fp16 * mask (B,N,N) – may be dummy if no mask
    # strides for left/right (B,N,N,H)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for gates (same layout as left/right)
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    # strides for mask (B,N,N)
    stride_m0, stride_m1, stride_m2,
    B, N, H,                     # dimensions
    BLOCK_H: tl.constexpr,       # tile size along hidden dimension
    HAS_MASK: tl.constexpr,      # 1 → use mask, 0 → ignore mask
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2

    # mask value (scalar) – broadcast over H
    mask_val = tl.float16(1.0)
    if HAS_MASK:
        off_m = b * stride_m0 + i * stride_m1 + j * stride_m2
        mask_val = tl.load(mask_ptr + off_m)

    # loop over hidden dim
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left_raw  = tl.load(left_ptr + off_l + offs * stride_l3,
                            mask=mask_h, other=tl.float16(0.0))
        right_raw = tl.load(right_ptr + off_r + offs * stride_r3,
                            mask=mask_h, other=tl.float16(0.0))
        left_gate_raw  = tl.load(left_gate_ptr + off_lg + offs * stride_lg3,
                                 mask=mask_h, other=tl.float16(0.0))
        right_gate_raw = tl.load(right_gate_ptr + off_rg + offs * stride_rg3,
                                 mask=mask_h, other=tl.float16(0.0))

        # sigmoid (in‑place)
        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

        # fused: proj * gate * mask
        left_out  = left_raw  * left_gate  * mask_val
        right_out = right_raw * right_gate * mask_val

        tl.store(left_ptr + off_l + offs * stride_l3,  left_out,  mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)


# -------------------------------------------------------------
# Triton‑augmented TriMul forward (outgoing version)
# -------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:        [B, N, N] (float32) – may be ignored if `nomask` is True
            - weights: dict of model parameters (float32)
            - config: dict with keys `dim`, `hidden_dim`, `nomask`
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------
    # Unpack
    # ---------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ---------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ---------------------------------------------------------
    # 2️⃣ Single fused linear for all projections & gates (FP16)
    # ---------------------------------------------------------
    # Concatenate the five weight matrices (no bias in any of them)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*hidden_dim, dim)
    ).to(torch.float16)

    x_fp16 = x.to(torch.float16)  # cast once
    # Combined linear → shape (B, N, N, 5*hidden_dim)
    combined = F.linear(x_fp16, combined_weight)  # FP16

    # ---------------------------------------------------------
    # 3️⃣ Split combined tensor into its components (views, no copy)
    # ---------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: [B, N, N, hidden_dim] (FP16)

    # ---------------------------------------------------------
    # 4️⃣ Apply mask and sigmoid gates (fused Triton kernel if mask present)
    # ---------------------------------------------------------
    B, N, _, _ = left_raw.shape

    if not nomask:
        # ensure contiguous tensors for correct stride handling
        left_raw = left_raw.contiguous()
        right_raw = right_raw.contiguous()
        left_gate_raw = left_gate_raw.contiguous()
        right_gate_raw = right_gate_raw.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        total_programs = B * N * N
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left_raw,
            right_raw,
            left_gate_raw,
            right_gate_raw,
            mask_fp16,
            # strides for left/right
            left_raw.stride(0), left_raw.stride(1), left_raw.stride(2), left_raw.stride(3),
            right_raw.stride(0), right_raw.stride(1), right_raw.stride(2), right_raw.stride(3),
            # strides for gates
            left_gate_raw.stride(0), left_gate_raw.stride(1), left_gate_raw.stride(2), left_gate_raw.stride(3),
            right_gate_raw.stride(0), right_gate_raw.stride(1), right_gate_raw.stride(2), right_gate_raw.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # sizes
            B,
            N,
            hidden_dim,
            BLOCK_H=64,          # 64 works well for hidden_dim=128
            HAS_MASK=1,
            num_warps=4,
        )
        left = left_raw
        right = right_raw
    else:
        # No mask → just apply sigmoid gates element‑wise
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_raw * left_gate
        right = right_raw * right_gate

    # ---------------------------------------------------------
    # 5️⃣ Core N³ multiplication via batched GEMM (FP16, Tensor‑Core)
    # ---------------------------------------------------------
    # reshape to (B*H, N, N) where each row corresponds to one head
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # GEMM: left @ rightᵀ  (FP16 inputs, FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]  FP16

    # ---------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out‑gate → final projection
    # ---------------------------------------------------------
    out_f32 = out.to(torch.float32)  # to FP32 for LayerNorm
    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]  FP32

    # out‑gate (sigmoid) – also FP32 for the multiplication
    out_gate = torch.sigmoid(out_gate_raw).to(torch.float32)  # [B,N,N,hidden_dim]

    out_norm = out_norm * out_gate

    # final linear projection back to `dim` (FP32)
    out_final = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim]

    return out_final
