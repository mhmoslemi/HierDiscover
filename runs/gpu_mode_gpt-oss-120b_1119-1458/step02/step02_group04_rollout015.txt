Outgoing TriMul (AlphaFold‑3 style) – fused linear projections +
batch‑GEMM contraction.

Algorithm
---------
1. Layer‑norm the input (float32 → stable statistics).
2. Cast to fp16 and compute *four* linear projections (left‑proj,

5. Reshape the tensors to (B·H, N, N) and run a batched GEMM
   (left @ rightᵀ) with cuBLAS – the dominant N³ work.
6. Layer‑norm over the hidden dimension (fp32).
7. Compute the output‑gate (sigmoid of a linear proj) and multiply.
8. Final linear projection back to the original channel dimension.
9. Return the result in float32.

Only the mask‑gate fusion is written in Triton; the heavy N³ contraction
uses cuBLAS which is already optimal on an H100.  By merging the four
projections into one matmul we cut the number of kernel launches in half,
reducing launch overhead and memory traffic while still exploiting Tensor‑Core
FP16 arithmetic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: out = proj * gate * mask   (flat view)
# -------------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    proj_ptr,     # const half *
    gate_ptr,     # const half *
    mask_ptr,     # const half *
    out_ptr,      # half *
    total_elts: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # 1.0 = no‑mask
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_mask_gate(proj, gate, mask, out):
    """Launch ``fuse_mask_gate_kernel`` on a flat view."""
    total = proj.numel()
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    fuse_mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        BLOCK,
        num_warps=4,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or ``None``.
        - weights_dict : mapping from parameter name to torch.Tensor (on the same device).
        - config_dict  : contains at least ``dim`` and ``hidden_dim``.

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: move a tensor to fp16 on the target device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_w = w["norm.weight"]
    norm_b = w["norm.bias"]

    # concatenate four projection weights (left‑proj, left‑gate,
    # right‑proj, right‑gate) – each is (H, dim)
    left_proj_w   = w["left_proj.weight"]
    left_gate_w   = w["left_gate.weight"]
    right_proj_w  = w["right_proj.weight"]
    right_gate_w  = w["right_gate.weight"]
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )                       # (4*H, dim)

    # other linear weights
    out_gate_w    = w["out_gate.weight"]          # (H, dim)
    to_out_norm_w = w["to_out_norm.weight"]       # (H,)
    to_out_norm_b = w["to_out_norm.bias"]         # (H,)
    to_out_w      = w["to_out.weight"]            # (dim, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Combined linear projection (four heads at once)
    # ------------------------------------------------------------------
    comb_out = F.linear(x_half, to_half(combined_weight))  # [B,N,N,4*H] (fp16)

    # Split the concatenated dimension
    # shape -> (B, N, N, 4, H)
    comb_out = comb_out.view(B, N, N, 4, H)
    left_proj, left_gate_logit, right_proj, right_gate_logit = torch.split(
        comb_out, H, dim=3
    )  # each [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_logit)   # fp16
    right_gate = torch.sigmoid(right_gate_logit)  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # broadcast mask to [B,N,N,H]
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)
        mask_fp16 = mask_fp16.expand(-1, -1, -1, H).contiguous()
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)
        _apply_mask_gate(left_proj, left_gate, mask_fp16, left)
        _apply_mask_gate(right_proj, right_gate, mask_fp16, right)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction – batched GEMM via cuBLAS (Tensor‑Core)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for the batched matmul
    # (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_view  = left.permute(0, 3, 1, 2).contiguous()
    right_view = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_view.view(B * H, N, N)            # (B*H, N, N)
    right_flat = right_view.view(B * H, N, N)           # (B*H, N, N)

    # batched matrix multiply: left @ rightᵀ for each head
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(H,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )  # [B, N, N, H] (float32)

    # ------------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid of a linear projection) and gating
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, to_half(out_gate_w)))  # [B,N,N,H] fp16
    gated    = out_norm * out_gate.to(torch.float32)                # fp32

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to the original dim
    # ------------------------------------------------------------------
    final = F.linear(gated.to(torch.float16), to_half(to_out_w))   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
