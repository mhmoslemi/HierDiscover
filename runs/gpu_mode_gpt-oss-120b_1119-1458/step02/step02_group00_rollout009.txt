"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the reference TriMul forward but
* fuses the four linear projections/gates into a single matrix‑multiply,
* uses FP16 everywhere except the (tiny) LayerNorms (kept in FP32 for stability),
* removes several unnecessary Python‑side kernels,
* keeps a tiny Triton kernel that fuses the element‑wise multiplication of the
  left/right tensors with their respective gates (the only required Triton use).

The dominant N³ “einsum” is performed as a batched GEMM
  out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
by reshaping left/right to (B⋅H, N, N) and calling `torch.bmm`,
which on an H100 executes as a Tensor‑Core GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: fuse left/right element‑wise gating (in‑place)
# --------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,       # fp16, flattened tensor
    left_gate_ptr,  # fp16, flattened tensor
    right_ptr,      # fp16, flattened tensor
    right_gate_ptr, # fp16, flattened tensor
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # left side
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    # right side
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


# --------------------------------------------------------------
# Main Triton‑accelerated TriMul forward
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]  (float32)
        - mask         : torch.Tensor   [B, N, N]       (bool/float)
        - weights      : dict of torch.Tensor    model parameters
        - config       : dict   must contain 'dim', 'hidden_dim' and optionally 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------
    # 1️⃣ Unpack & config handling
    # ------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # -------------------------------------------------
    # 2️⃣ Input LayerNorm (float32 → keeps stability)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                           # [B, N, N, dim]  float32
    x_h = x.to(torch.float16)   # cast once for the heavy work

    # -------------------------------------------------
    # 3️⃣ Fuse the four linear layers (proj + gate)
    # -------------------------------------------------
    # cast model weights once
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)

    # concatenate along the out‑feature dimension (4 * hidden_dim, dim)
    w_concat = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate], dim=0
    ).contiguous()

    # single fused linear: shape (B, N, N, 4*hidden_dim)
    proj_all = F.linear(x_h, w_concat)

    # split the result
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        proj_all, hidden_dim, dim=-1
    )  # each => [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # -------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -------------------------------------------------
    # 5️⃣ Gates (sigmoid) and in‑place fusion via Triton
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)   # fp16
    right_gate = torch.sigmoid(right_gate_pre) # fp16

    # ensure contiguous memory for the kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # in‑place element‑wise multiplication: left = left_proj * left_gate, etc.
    numel = left_proj.numel()
    BLOCK = 65536  # ~64 K elements per program (tuned for H100)
    grid = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid](
        left_proj.view(-1),
        left_gate.view(-1),
        right_proj.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    left = left_proj
    right = right_proj

    # -------------------------------------------------
    # 6️⃣ Core N³ contraction via batched GEMM
    #     out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    total_heads = B * H

    # reshape to (total_heads, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(total_heads, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(total_heads, N, N) # (B*H, N, N)

    # batched matrix multiplication (uses Tensor‑Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))        # (B*H, N, N)

    # restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 7️⃣ Post‑processing LayerNorm (half‑precision weights)
    # -------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # still fp16

    # -------------------------------------------------
    # 8️⃣ Output gate (sigmoid) and element‑wise scaling
    # -------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))  # [B, N, N, hidden_dim] fp16
    out = out * out_gate                                 # cheap element‑wise op

    # -------------------------------------------------
    # 9️⃣ Final linear back to `dim`
    # -------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = F.linear(out, w_to_out)                     # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 10️⃣ Cast back to float32 (expected by downstream code)
    # -------------------------------------------------
    return final.to(torch.float32)
