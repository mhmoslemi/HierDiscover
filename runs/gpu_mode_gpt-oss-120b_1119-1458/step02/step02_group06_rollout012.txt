"""
Optimised outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. **Layer‑Norm (input)** – A custom Triton kernel computes mean/variance in FP32
   while keeping the data in FP16, applying the learned scale/bias.
2. **Fused linear projection** – The 5 linear layers
   (left_proj, right_proj, left_gate, right_gate, out_gate) are merged into a
   single GEMM (half‑precision, Tensor‑Core friendly).  The result is sliced
   into the 5 tensors.
3. **Mask & gating** – Optional binary mask is applied to `left`/`right`,
   then element‑wise sigmoid‑gates are multiplied in‑place.
4. **Batched Gram product** – `out[b,i,j] = Σ_k left[b,i,k]·right[b,j,k]`
   is performed with a single `torch.bmm` on tensors shaped
   `(B*H, N, N)`.  This uses cuBLAS/Tensor‑Cores and is far faster than a
   handcrafted Triton kernel for the large N³ workloads in the benchmark.
5. **Layer‑Norm (output)** – Another Triton kernel normalises the
   `(B,N,N,hidden)` result.
6. **Output gating & projection** – The output gate scales the normalised
   tensor, a final half‑precision linear projection maps back to `dim`,
   and the result is cast to float32.

All heavy matrix multiplications run in FP16 with FP32 accumulation,
halving memory traffic compared to the original float32 implementation
while staying numerically stable thanks to the FP32 reductions inside the
kernels.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (works for any row‑major 2‑D tensor)
# ----------------------------------------------------------------------
@triton.jit
def layer_norm_kernel(
    X_ptr,                 # input pointer (M, D)   – FP16
    Y_ptr,                 # output pointer (M, D)  – FP16
    weight_ptr,            # scale (D,)             – FP16
    bias_ptr,              # bias  (D,)             – FP16
    M,                     # number of rows
    D,                     # row length (dim or hidden_dim)
    stride_xm, stride_xd,  # strides for X (row, col)
    stride_ym, stride_yd,  # strides for Y (row, col)
    eps: tl.constexpr,     # epsilon for variance
    BLOCK_D: tl.constexpr, # tile size along D (compile‑time constant)
):
    pid = tl.program_id(0)                # one program per row
    if pid >= M:
        return

    # ------------------------------------------------------------------
    # 1. First pass: compute mean and variance (fp32 accumulator)
    # ------------------------------------------------------------------
    row_ptr = X_ptr + pid * stride_xm
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    # loop over the D dimension in blocks of BLOCK_D
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs_d < D
        x = tl.load(row_ptr + offs_d * stride_xd,
                    mask=mask, other=0.0)           # FP16
        xf = x.to(tl.float32)                       # promote
        sum_val += tl.sum(xf, axis=0)               # reduction over threads
        sum_sq  += tl.sum(xf * xf, axis=0)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2. Second pass: normalise, apply scale & bias, store (FP16)
    # ------------------------------------------------------------------
    out_ptr = Y_ptr + pid * stride_ym
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs_d < D
        x = tl.load(row_ptr + offs_d * stride_xd,
                    mask=mask, other=0.0)
        xf = x.to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (xf - mean) * inv_std

        # affine transformation
        w = tl.load(weight_ptr + offs_d,
                    mask=mask, other=1.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_d,
                    mask=mask, other=0.0).to(tl.float32)

        y_fp32 = norm * w + b
        y = y_fp32.to(tl.float16)

        tl.store(out_ptr + offs_d * stride_yd, y, mask=mask)

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N]   (float32 or bool)
            - weights: dict of tensors (float32)
            - config: dict containing,
                * dim          (int)
                * hidden_dim   (int)
                * nomask       (bool, optional, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = torch.float16

    # ------------------------------------------------------------------
    # 0. convenience helpers for weight conversion
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(dtype)

    # ------------------------------------------------------------------
    # 1. INPUT LAYER‑NORM (custom Triton, stays in FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    M_in = B * N * N                     # rows = B·N²
    # flatten to (M_in, dim) – keep memory layout contiguous
    x_flat = input_tensor.view(M_in, dim).contiguous()
    x_norm_flat = torch.empty_like(x_flat, dtype=dtype, device=device)

    # weight/bias for the first LayerNorm
    norm_weight = to_half(weights["norm.weight"])
    norm_bias   = to_half(weights["norm.bias"])

    # launch kernel (BLOCK_D = 128 works for dim up to 384)
    grid = (M_in,)
    layer_norm_kernel[grid](
        x_flat,
        x_norm_flat,
        norm_weight,
        norm_bias,
        M_in,
        dim,
        x_flat.stride(0), x_flat.stride(1),
        x_norm_flat.stride(0), x_norm_flat.stride(1),
        eps=1e-5,
        BLOCK_D=128,          # compile‑time tile size
        num_warps=4,
        num_stages=2,
    )
    # reshape back to (B, N, N, dim) – still FP16
    x = x_norm_flat.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. FUSED linear projection (5 heads merged into one gemm)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices: shape (5*hidden_dim, dim)
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    combined_weight = torch.cat(
        [
            w_left_proj,
            w_right_proj,
            w_left_gate,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(dtype)                                   # (5*H, dim)

    # single half‑precision GEMM
    proj = torch.nn.functional.linear(x, combined_weight)  # (B, N, N, 5*H)

    # ------------------------------------------------------------------
    # 3. Slice the fused projection & apply mask / gating
    # ------------------------------------------------------------------
    H = hidden_dim
    # slicing helpers – views share storage, no extra copy
    left      = proj[..., :H]            # (B,N,N,H)
    right     = proj[..., H:2*H]         # (B,N,N,H)
    left_gate = proj[..., 2*H:3*H].sigmoid()
    right_gate= proj[..., 3*H:4*H].sigmoid()
    out_gate  = proj[..., 4*H:5*H].sigmoid()   # (B,N,N,H)

    if not nomask:
        # mask broadcasted over the hidden dimension
        m = mask.unsqueeze(-1).to(dtype)    # (B,N,N,1)
        left = left * m
        right = right * m

    # apply element‑wise gates (in‑place to reduce memory traffic)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. N³ reduction (batched matrix multiplication)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for a single bmm
    left = left.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left.view(BH, N, N)                  # (BH, N, N)
    right_flat = right.view(BH, N, N)

    # cuBLAS batched GEMM (FP16×FP16→FP32 accumulate, output FP16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H) – still FP16
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. OUTPUT LAYER‑NORM (custom Triton, stays in FP16)
    # ------------------------------------------------------------------
    M_out = B * N * N
    out_flat = out.view(M_out, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=dtype, device=device)

    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_half(weights["to_out_norm.bias"])

    grid = (M_out,)
    layer_norm_kernel[grid](
        out_flat,
        out_norm_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        M_out,
        H,
        out_flat.stride(0), out_flat.stride(1),
        out_norm_flat.stride(0), out_norm_flat.stride(1),
        eps=1e-5,
        BLOCK_D=128,
        num_warps=4,
        num_stages=2,
    )
    out_norm = out_norm_flat.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out_norm * out_gate                     # element‑wise scale (FP16)

    w_to_out = to_half(weights["to_out.weight"])   # (dim, H)
    out = torch.nn.functional.linear(out, w_to_out)   # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 7. Cast to float32 for the final result
    # ------------------------------------------------------------------
    return out.float()
