    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    # tile row idx
    pid_i = tl.program_id(1)
    # tile col idx
    pid_j = tl.program_id(2)

    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

"""
TriMul “outgoing” forward pass – high‑performance Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) – cheap.
2. Cast to FP16 and run all linear projections (including the three gates)
   with half‑precision weights.  The operations stay on Tensor‑Cores.
3. Apply the optional mask and the element‑wise gates (still FP16).
4. Rearrange the tensors from [B, N, N, H] → [B*H, N, N] (contiguous).
5. Compute the costly N³ reduction as a *batched* matrix multiplication

        out[b·h, i, j] = Σₖ left[b·h, i, k] * right[b·h, j, k]

   using a custom Triton kernel.  The kernel performs a blocked GEMM,
   accumulates in FP32 and writes the result back in FP16.  It is used
   for **all** problem sizes, eliminating the extra launch overhead of
   torch.bmm.
6. Reshape the result back to [B, N, N, H] and cast to FP32.
7. Apply the output LayerNorm, the output‑gate and the final linear
   projection (all in FP32) and return the tensor of shape
   [B, N, N, dim] (float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Batched GEMM kernel: out = left @ rightᵀ   (left/right : [BH, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N,                               # both equal to seq_len
    stride_ah, stride_ai, stride_ak,    # left  : (BH, i, k)
    stride_bh, stride_bi, stride_bk,    # right : (BH, k, j) – we load the transposed view
    stride_ch, stride_ci, stride_ck,    # out   : (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    #  Thread‑block identifiers
    # ---------------------------------------------------------------
    batch_id = tl.program_id(0)   # BH dimension
    pid_i    = tl.program_id(1)   # tile row   (i)
    pid_j    = tl.program_id(2)   # tile col   (j)

    # ---------------------------------------------------------------
    #  Offsets for this tile
    # ---------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32 (numerical stability, matches reference)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    #  Loop over the reduction (k) dimension
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block   : (BLOCK_M, BLOCK_K)  = left[batch_id, i, k]
        a = tl.load(
            a_ptr
            + batch_id * stride_ah
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # rightᵀ block : (BLOCK_K, BLOCK_N)  = right[batch_id, k, j]
        b = tl.load(
            b_ptr
            + batch_id * stride_bh
            + offs_k[:, None] * stride_bi
            + offs_j[None, :] * stride_bk,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # block matrix‑multiply, accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # ---------------------------------------------------------------
    #  Write the result (cast back to FP16)
    # ---------------------------------------------------------------
    tl.store(
        c_ptr
        + batch_id * stride_ch
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_ck,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim] (float32)
        - mask         : [B, N, N] (bool or float)
        - weights      : dict of pre‑trained parameters
        - config       : dict with keys   dim, hidden_dim, nomask (optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    #########################################
    # 1. Input LayerNorm (float32)
    #########################################
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    #########################################
    # 2. Cast to fp16 and prepare half‑precision weights
    #########################################
    x_h = x.half()

    w_left   = weights["left_proj.weight"].half()
    w_right  = weights["right_proj.weight"].half()
    w_lgate  = weights["left_gate.weight"].half()
    w_rgate  = weights["right_gate.weight"].half()
    w_ogate  = weights["out_gate.weight"].half()
    w_out    = weights["to_out.weight"].half()               # final linear

    #########################################
    # 3. Linear projections (fp16)
    #########################################
    left  = torch.nn.functional.linear(x_h, w_left)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_h, w_right)  # [B, N, N, H]

    #########################################
    # 4. Optional mask (broadcast on hidden dim)
    #########################################
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)     # [B, N, N, 1]
        left = left * m
        right = right * m

    #########################################
    # 5. Gating networks (fp16)
    #########################################
    left_gate  = torch.nn.functional.linear(x_h, w_lgate).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, w_rgate).sigmoid()
    out_gate   = torch.nn.functional.linear(x_h, w_ogate).sigmoid()  # [B, N, N, H]

    left  = left * left_gate
    right = right * right_gate

    #########################################
    # 6. Reshape to (BH, N, N) for the N³ reduction
    #########################################
    B, N, _, _ = left.shape
    # move hidden dimension to the second axis and make contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left.view(B * hidden_dim, N, N)    # (BH, N, N)  fp16
    right_flat = right.view(B * hidden_dim, N, N)   # (BH, N, N)  fp16

    #########################################
    # 7. Batched matrix multiplication with Triton
    #########################################
    BH = B * hidden_dim
    out_flat = torch.empty_like(left_flat)          # fp16 output buffer

    # extract raw strides (they are int64)
    s_ah, s_ai, s_ak = left_flat.stride()
    s_bh, s_bi, s_bk = right_flat.stride()
    s_ch, s_ci, s_ck = out_flat.stride()

    # Tile sizes – chosen to fit H100 Tensor‑Core registers and to stay
    # under the 256‑thread limit per program (8 warps × 32 threads).
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64 if N > 256 else 32

    grid = (
        BH,                                 # batch*hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # rows
        (N + BLOCK_N - 1) // BLOCK_N,       # cols
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        s_ah, s_ai, s_ak,
        s_bh, s_bi, s_bk,
        s_ch, s_ci, s_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps → 256 threads, good occupancy on H100
        num_stages=3,       # triple buffering of tiles
    )

    #########################################
    # 8. Reshape back to (B, N, N, H) and cast to FP32
    #########################################
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()                               # [B, N, N, H]  FP32

    #########################################
    # 9. Output LayerNorm + output gate
    #########################################
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()                    # broadcast multiplication

    #########################################
    # 10. Final linear projection back to `dim` (float32)
    #########################################
    out = torch.nn.functional.linear(out, w_out.float())  # [B, N, N, dim]

    return out
