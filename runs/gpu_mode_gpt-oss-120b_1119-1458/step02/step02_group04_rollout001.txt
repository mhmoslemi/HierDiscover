def layernorm_kernel(x_ptr, out_ptr, weight_ptr, bias_ptr,
                     N: tl.constexpr, C: tl.constexpr, eps: tl.constexpr):
    pid = tl.program_id(0)  # row index
    row_start = pid * C
    # compute mean and var
    sum = tl.zeros([], dtype=tl.float32)
    sum2 = tl.zeros([], dtype=tl.float32)
    BLOCK = 128
    for offset in range(0, C, BLOCK):
        cur = tl.minimum(C - offset, BLOCK)
        offs = row_start + offset + tl.arange(0, BLOCK)
        mask = offs < row_start + C
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        x_fp32 = x.to(tl.float32)
        sum += tl.sum(x_fp32, axis=0)
        sum2 += tl.sum(x_fp32 * x_fp32, axis=0)
    mean = sum / C
    var = sum2 / C - mean * mean
    rs = 1.0 / tl.sqrt(var + eps)

    # second pass to write normalized values
    for offset in range(0, C, BLOCK):
        cur = tl.minimum(C - offset, BLOCK)
        offs = row_start + offset + tl.arange(0, BLOCK)
        mask = offs < row_start + C
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        x_fp32 = x.to(tl.float32)
        y = ((x_fp32 - mean) * rs) * tl.load(weight_ptr + offset + tl.arange(0, BLOCK), mask=mask, other=0.0).to(tl.float32) + tl.load(bias_ptr + offset + tl.arange(0, BLOCK), mask=mask, other=0.0).to(tl.float32)
        y = y.to(tl.float16)
        tl.store(out_ptr + offs, y, mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_fwd_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    NROWS, C, EPS,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    if pid >= NROWS:
        return
    row_start = pid * C
    # first pass: compute sum and sum_sq
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    # loop blocks
    for off in range(0, C, BLOCK_SIZE):
        cur = tl.minimum(C - off, BLOCK_SIZE)
        offs = row_start + off + tl.arange(0, BLOCK_SIZE)
        mask = offs < row_start + C
        # load x (float32)
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=0)
        sum_sq += tl.sum(x_f * x_f, axis=0)
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rs = 1.0 / tl.sqrt(var + EPS)

    # second pass: write normalized values with scale & bias
    for off in range(0, C, BLOCK_SIZE):
        cur = tl.minimum(C - off, BLOCK_SIZE)
        offs = row_start + off + tl.arange(0, BLOCK_SIZE)
        mask = offs < row_start + C
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        x_f = x.to(tl.float32)
        # load weight and bias
        w = tl.load(weight_ptr + off + tl.arange(0, BLOCK_SIZE), mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + off + tl.arange(0, BLOCK_SIZE), mask=mask, other=0.0).to(tl.float32)
        y = ((x_f - mean) * rs) * w + b
        # write as fp16
        tl.store(out_ptr + offs, y.to(tl.float16), mask=mask)

@triton.jit
def elementwise_fuse_mask_gate_kernel(
    x_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elts: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)

def layernorm_triton(x, weight, bias, eps=1e-5):
    # x: [B,N,N,C] float32
    B, N, _, C = x.shape
    rows = B * N * N
    x_flat = x.reshape(rows, C).contiguous()
    out = torch.empty_like(x_flat, dtype=torch.float16)
    grid = (rows,)
    block_size = 128
    layernorm_fwd_kernel[grid](
        x_flat, out, weight, bias,
        NROWS=rows,
        C=C,
        EPS=eps,
        BLOCK_SIZE=block_size,
        num_warps=4,
    )
    return out.view(B, N, N, C)

def elementwise_fuse_mask_gate(x, gate, mask):
    # x,g,m are half tensors with same shape
    total = x.numel()
    out = torch.empty_like(x)
    # flatten
    x_flat = x.reshape(total)
    gate_flat = gate.reshape(total)
    mask_flat = mask.reshape(total)
    out_flat = out.reshape(total)
    grid = ( (total + 127) // 128,)
    elementwise_fuse_mask_gate_kernel[grid](
        x_flat, gate_flat, mask_flat, out_flat,
        total_elts=total,
        BLOCK_SIZE=128,
        num_warps=4,
    )
    return out

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (AlphaFold‑3 style) with Triton‑fused LayerNorms and
    element‑wise mask/gate multiplication.  All heavy linear projections are performed
    in FP16, while LayerNorms are computed in FP32 for stability.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Utility: move weight to device and dtype
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # Load weights (they are stored as float32)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = to_half(weights["left_proj.weight"])
    right_proj_w = to_half(weights["right_proj.weight"])
    left_gate_w = to_half(weights["left_gate.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w = to_half(weights["to_out.weight"])

    # 1️⃣ Input LayerNorm (FP32 -> FP16)
    x_norm = layernorm_triton(input_tensor, norm_w, norm_b, eps=eps)  # fp16

    # 2️⃣ Combined projection for left/right + gates
    # concatenate weight matrices into a single kernel weight
    # shapes: (dim, hidden_dim) each -> total (dim, hidden_dim*4)
    # All weights are half
    combined_weight = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w
    ], dim=1)  # [dim, hidden_dim*4]

    # Linear projection (no bias)
    proj = F.linear(x_norm, combined_weight)  # [B,N,N, hidden_dim*4]  fp16

    # Split
    hidden = hidden_dim
    left = proj[..., 0:hidden]
    right = proj[..., hidden:2*hidden]
    left_gate_raw = proj[..., 2*hidden:3*hidden]
    right_gate_raw = proj[..., 3*hidden:4*hidden]

    # Gates sigmoid
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # Optional mask
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
    else:
        mask_f = torch.ones_like(left, dtype=torch.float16)

    # Fuse mask and gate multiplication via Triton
    left = elementwise_fuse_mask_gate(left, left_gate, mask_f)
    right = elementwise_fuse_mask_gate(right, right_gate, mask_f)

    # 3️⃣ N³ contraction via batched GEMM (FP16)
    # reshape to (B*H, N, N)
    left_view = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat = left_view.view(B * hidden, N, N)
    right_flat = right_view.view(B * hidden, N, N)

    # batched matmul: left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # 4️⃣ Output LayerNorm over hidden dim
    out = layernorm_triton(out, to_out_norm_w, to_out_norm_b, eps=eps)  # fp16

    # 5️⃣ Output gate
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))  # [B,N,N,hidden] fp16

    # Apply output gate
    out = out * out_gate

    # 6️⃣ Final linear projection back to dim
    final = F.linear(out, to_out_w)  # fp16

    # Return in float32
    return final.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3 style) – Triton‑fused LayerNorms and mask/gate
multiplication.

Algorithm
---------
1️⃣  Input LayerNorm (FP32 → FP16) – custom Triton kernel.
2️⃣  Concatenate the four linear projections (left/right + 2 gates) into a
    single FP16 weight matrix and apply one fused F.linear.
3️⃣  Compute sigmoid gates, fuse ``mask * gate * projection`` via a tiny Triton
    element‑wise kernel (zero extra memory copies).
4️⃣  N³ contraction: batch GEMM on the (B·H, N, N) tensors
    (left @ rightᵀ) using FP16 torch.bmm (cuBLAS Tensor‑Core).
5️⃣  Output LayerNorm over the hidden dimension – same Triton kernel as (1).
6️⃣  Output gate (sigmoid of a linear projection), apply it, and a final
    FP16 linear projection back to the original channel dimension.
7️⃣  Return the result in FP32.

All heavy arithmetic is in FP16; the two LayerNorms are performed in FP32 for
numerical stability.  At least one operation (the LayerNorms) is implemented
in a Triton kernel as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (FP32 compute, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fwd_kernel(
    x_ptr,                # const float *
    out_ptr,              # half *
    weight_ptr,           # const half *
    bias_ptr,             # const half *
    NROWS,                # number of rows   (B·N·N)
    C,                    # channel size    (dim or hidden_dim)
    EPS,                  # layer‑norm epsilon
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    # each program handles one row
    if pid >= NROWS:
        return
    row_start = pid * C

    # ---------- first pass: mean & variance ----------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    for off in range(0, C, BLOCK_SIZE):
        offs = row_start + off + tl.arange(0, BLOCK_SIZE)
        mask = offs < row_start + C
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)              # FP32
        xv = x.to(tl.float32)
        sum_val += tl.sum(xv, axis=0)
        sum_sq += tl.sum(xv * xv, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rs = 1.0 / tl.sqrt(var + EPS)                                    # 1/σ

    # ---------- second pass: write normalized + affine ----------
    for off in range(0, C, BLOCK_SIZE):
        offs = row_start + off + tl.arange(0, BLOCK_SIZE)
        mask = offs < row_start + C
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        xv = x.to(tl.float32)

        # load weight/bias (stored as half)
        w = tl.load(weight_ptr + off + tl.arange(0, BLOCK_SIZE), mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + off + tl.arange(0, BLOCK_SIZE), mask=mask, other=0.0).to(tl.float32)

        y = ((xv - mean) * rs) * w + b                               # FP32
        tl.store(out_ptr + offs, y.to(tl.float16), mask=mask)        # FP16


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm over the last dimension of ``x`` using the Triton kernel above.
    ``x`` must be contiguous and of dtype torch.float32.
    Returns a FP16 tensor with the same shape as ``x``.
    """
    B, N, _, C = x.shape
    rows = B * N * N
    x_flat = x.reshape(rows, C).contiguous()
    out = torch.empty_like(x_flat, dtype=torch.float16)

    grid = (rows,)
    block = 128
    layernorm_fwd_kernel[grid](
        x_flat, out,
        weight, bias,
        NROWS=rows,
        C=C,
        EPS=eps,
        BLOCK_SIZE=block,
        num_warps=4,
    )
    return out.view(B, N, N, C)


# ----------------------------------------------------------------------
# Triton element‑wise kernel: out = x * gate * mask
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr, out_ptr,
    TOTAL_ELTS: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELTS
    x = tl.load(x_ptr + offs,   mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs,mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs,mask=mask, other=0.0)
    tl.store(out_ptr + offs, x * g * m, mask=mask)


def fuse_mask_gate(x: torch.Tensor,
                   gate: torch.Tensor,
                   mask: torch.Tensor) -> torch.Tensor:
    """`x`, `gate`, `mask` must all be FP16 and have the same shape."""
    total = x.numel()
    out = torch.empty_like(x)
    block = 128
    grid = ((total + block - 1) // block,)
    mask_gate_fuse_kernel[grid](
        x.reshape(-1), gate.reshape(-1), mask.reshape(-1), out.reshape(-1),
        TOTAL_ELTS=total,
        BLOCK_SIZE=block,
        num_warps=2,
    )
    return out


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dictionary with the model parameters (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # Helper to move a weight to the target device + FP16 dtype
    # --------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # Load model parameters
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]          # (dim,)
    norm_b = weights["norm.bias"]            # (dim,)

    left_proj_w   = to_half(weights["left_proj.weight"])    # (hidden, dim)
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])

    to_out_norm_w = to_half(weights["to_out_norm.weight"])  # (hidden,)
    to_out_norm_b = to_half(weights["to_out_norm.bias"])

    to_out_w      = to_half(weights["to_out.weight"])       # (dim, hidden)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # This custom kernel does the mean/var in FP32 but writes FP16.
    x_norm = layernorm_triton(input_tensor, norm_w, norm_b, eps=eps)   # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate)
    # --------------------------------------------------------------
    # Stack the four weight matrices vertically → shape (4*hidden, dim)
    combined_weight = torch.cat([left_proj_w,
                                 right_proj_w,
                                 left_gate_w,
                                 right_gate_w], dim=0)          # (4*hidden, dim)

    # Single fused linear projection (no bias)
    proj = F.linear(x_norm, combined_weight)                     # [B,N,N,4*hidden] fp16

    # Split the concatenated tensor
    h = hidden_dim
    left          = proj[..., 0:h]
    right         = proj[..., h:2*h]
    left_gate_raw = proj[..., 2*h:3*h]
    right_gate_raw= proj[..., 3*h:4*h]

    # --------------------------------------------------------------
    # 3️⃣ Sigmoid gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # --------------------------------------------------------------
    # 4️⃣ Optional pairwise mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)   # [B,N,N,1]
    else:
        mask_fp16 = torch.ones_like(left, dtype=torch.float16)   # broadcast mask

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask & gate multiplication via Triton (still FP16)
    # --------------------------------------------------------------
    left  = fuse_mask_gate(left,  left_gate,  mask_fp16)
    right = fuse_mask_gate(right, right_gate, mask_fp16)

    # --------------------------------------------------------------
    # 6️⃣ N³ contraction – batched GEMM (FP16)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) where H = hidden_dim
    left_view  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left_view.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right_view.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # Batched matmul: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # Back to [B,N,N,hidden] layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm over hidden dimension (FP32 → FP16)
    # --------------------------------------------------------------
    out = layernorm_triton(out, to_out_norm_w, to_out_norm_b, eps=eps)   # fp16

    # --------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid) and final linear projection
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))   # [B,N,N,hidden] fp16
    out = out * out_gate

    final = F.linear(out, to_out_w)                         # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to FP32 as required by the spec
    # --------------------------------------------------------------
    return final.to(torch.float32)
