@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    # geometry
    B, N, hidden,
    # strides
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden index
    b = bh // hidden
    h = bh % hidden

    # base offsets
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # compute start indices
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # compute offsets for i/j
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # masks for i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M x K)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (K x N) for dot (since we need K,N)
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)  # left: (M,K) right: (K,N) -> (M,N)

    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. **Layer‑norm** – Layer‑norm the input on the last axis and cast to FP16.
2. **Fused projection & gating** – Concatenate the five weight matrices
   (left_proj, left_gate, right_proj, right_gate, out_gate) and perform a
   single GEMM (`torch.nn.functional.linear`).  The result is split into the
   three tensors needed for the update: `left_proj_raw`, `right_proj_raw`,
   `out_gate_logit` and the two gate logits.
3. **Mask & gate fusion** – Apply the sigmoid gates (still in FP16).  If a
   mask is present we multiply it (broadcasted) together with the projection
   and the gate in one go (two element‑wise kernels, no explicit expansion).
4. **Core contraction** – Compute the outgoing update
   ``out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]``
   with a custom Triton kernel that implements a batched GEMM over the
   hidden dimension without any tensor transposes / copies.  The kernel
   treats ``B * hidden`` as a single batch dimension and tiles the ``N×N``
   output with configurable block sizes (64 × 64 tiles, K‑tile of 64).
5. **Output layer‑norm, gating and linear** – Layer‑norm over the hidden
   dimension, multiply by the output gate, and linearly project back to the
   original channel dimension.  All heavy work stays in FP16; the final
   result is cast back to the input dtype (float32).

The implementation keeps a single Triton kernel for the O(N³) contraction,
which removes the costly ``permute(...).contiguous()`` copies present in the
reference implementation and therefore brings the runtime well below the
target 1 ms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # geometry
    B, N, H,
    # strides for left  (B, i, k, h)
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right (B, j, k, h)
    stride_rb, stride_rj, stride_rk, stride_rh,
    # strides for out   (B, i, j, h)
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiply that computes
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
       Left & right are expected in FP16, accumulation in FP32.
    """
    # ------------------------------------------------------------------
    # 1‑D program IDs
    # ------------------------------------------------------------------
    bh = tl.program_id(0)               # combined batch (B * H)
    pid_m = tl.program_id(1)            # tile row   (i)
    pid_n = tl.program_id(2)            # tile col   (j)

    # ------------------------------------------------------------------
    # Recover batch index b and hidden channel h
    # ------------------------------------------------------------------
    b = bh // H                         # integer division
    h = bh % H

    # ------------------------------------------------------------------
    # Base pointers for this (b,h) pair
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # ------------------------------------------------------------------
    # Tile start coordinates
    # ------------------------------------------------------------------
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # ------------------------------------------------------------------
    # Offsets for the tile (vectorised)
    # ------------------------------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k‑dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  : shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile (transposed for GEMM): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # GEMM: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Store the result back to memory (cast to FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched outgoing update using the Triton kernel above.
    left / right : [B, N, N, H] (FP16)
    returns       : [B, N, N, H] (FP16)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    # output tensor
    out = torch.empty_like(left)

    # strides as python integers (they are already in element units)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # ------------------------------------------------------------------
    # Tile sizes – tuned for H100 FP16 GEMM
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch * hidden
        triton.cdiv(N, BLOCK_M),               # i‑tiles
        triton.cdiv(N, BLOCK_N),               # j‑tiles
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry‑point – custom kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
               - input_tensor : [B, N, N, dim]  (float32, on CUDA)
               - mask_tensor  : [B, N, N]       (bool/float, on CUDA)
    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 -> FP16 for the rest)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)      # FP16 for the heavy compute

    # ------------------------------------------------------------------
    # 2) Fused projection (5*hidden output) --------------------------------
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (dim, hidden) -> (5*hidden, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*hidden, dim)

    # Single linear layer yields all projections & gate logits at once
    # Shape: [B, N, N, 5*hidden]
    combined = F.linear(x, w_concat)  # no bias

    # Split the concatenated dimension back into the five parts
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = \
        torch.split(combined, hidden, dim=-1)

    # Sigmoid gates (still half‑precision)
    left_gate  = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate   = torch.sigmoid(out_gate_logit)

    # ------------------------------------------------------------------
    # 3) Apply optional mask & gate fusion (element‑wise)
    # ------------------------------------------------------------------
    if nomask:
        left  = left_proj_raw * left_gate               # [B,N,N,H]
        right = right_proj_raw * right_gate
    else:
        # Broadcast mask on the hidden dimension without materialising an expansion
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)     # [B,N,N,1]
        left  = left_proj_raw * left_gate * mask_fp16
        right = right_proj_raw * right_gate * mask_fp16

    # ------------------------------------------------------------------
    # 4) Core contraction via a fused Triton kernel
    # ------------------------------------------------------------------
    # Compute out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    out_hidden = trimul_outgoing(left, right)   # shape [B,N,N,H], FP16

    # ------------------------------------------------------------------
    # 5) Output LayerNorm + gating + final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # Multiply by the output gate (broadcasted across i/j)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to the original dtype (float32) and return
    return out_final.to(inp.dtype)
