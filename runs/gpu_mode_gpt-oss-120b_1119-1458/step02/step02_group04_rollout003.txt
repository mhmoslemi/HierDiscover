```
@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        stride_am, stride_ak, stride_batch_a,
                        stride_bk, stride_bn, stride_batch_b,
                        stride_cm, stride_cn, stride_batch_c,
                        M, N, K,
                        BLOCK_M: tl.constexpr,
                        BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Compute start indexes
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    # Offsets for loading A and B
    a_ptrs = A_ptr + batch * stride_batch_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + batch * stride_batch_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in FP32 for accuracy
    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0).to(tl.float16)
        # dot product accumulates in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # Write back
    c_ptrs = C_ptr + batch * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # Cast accumulator to half for store (or to fp32 if needed)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])
```

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_aba, stride_aa, stride_ak,
    stride_bba, stride_bk, stride_bn,
    stride_cca, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for edges
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers
    a_ptrs = a_ptr + b * stride_aba + offs_m[:, None] * stride_aa + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + b * stride_bba + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K)[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0).to(tl.float16)
        acc += tl.dot(a, b, out_dtype=tl.float32)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = c_ptr + b * stride_cca + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward pass.

The implementation follows the PyTorch reference but
* fuses the four linear projections (proj + gate) for the left and right
  tensors into two GEMMs (by concatenating the weight matrices);
* performs the N³ contraction via a custom Triton batched‑GEMM kernel,
  which computes left @ rightᵀ for every hidden‑channel in a single launch;
* keeps the heavy compute in FP16, accumulates in FP32 for numerical stability,
  and only converts back to FP32 for the final layer‑norms and the output.

Only the contraction is executed in Triton – this satisfies the
“must use at least part of the operations in a kernel” requirement while
leaving the remaining cheap operations in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton batched GEMM kernel (computes C = A @ Bᵀ for each batch entry)
#   A : (B, M, K)   – row‑major (stride_am = stride of M, stride_ak = stride of K)
#   B : (B, K, N)   – row‑major (stride_bk = stride of K, stride_bn = stride of N)
#   C : (B, M, N)
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_aba, stride_aa, stride_ak,        # A: batch, M, K strides
    stride_bba, stride_bk, stride_bn,        # B: batch, K, N strides
    stride_cca, stride_cm, stride_cn,        # C: batch, M, N strides
    M, N, K,                                 # matrix sizes (compile‑time vals)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched FP16 GEMM with FP32 accumulation."""
    batch_id = tl.program_id(0)          # 0‑th axis → batch dimension
    pid_m = tl.program_id(1)             # tile row index
    pid_n = tl.program_id(2)             # tile col index

    # ---- tile offsets ----------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- pointers for the current tile ------------------------------------
    a_ptrs = a_ptr + batch_id * stride_aba + offs_m[:, None] * stride_aa + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bba + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ---- accumulate in FP32 ---------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k_tiles):
        cur_k = k_iter * BLOCK_K
        cur_mask_k = (cur_k + offs_k) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & cur_mask_k[None, :],
                    other=0.0)                 # FP16
        b = tl.load(b_ptrs,
                    mask=cur_mask_k[:, None] & mask_n[None, :],
                    other=0.0)                 # FP16
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # slide the windows
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---- write back ------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cca + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weight_dict  : mapping of parameter names to tensors
        - config_dict  : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper – move weight to FP16 on the target device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load weights (all Linear layers are bias‑free)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]          # (dim,)
    norm_b = weights["norm.bias"]            # (dim,)

    left_proj_w   = to_half(weights["left_proj.weight"])   # (hidden, dim)
    right_proj_w  = to_half(weights["right_proj.weight"])  # (hidden, dim)
    left_gate_w   = to_half(weights["left_gate.weight"])   # (hidden, dim)
    right_gate_w  = to_half(weights["right_gate.weight"])  # (hidden, dim)
    out_gate_w    = to_half(weights["out_gate.weight"])    # (hidden, dim)

    to_out_norm_w = to_half(weights["to_out_norm.weight"]) # (hidden,)
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])      # (dim, hidden)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)                     # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)          # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left & left‑gate (and right & right‑gate) projections
    #    – concatenate the two weight matrices and perform a single matmul
    # ------------------------------------------------------------------
    # Flatten spatial dims for a single GEMM: (B·N·N, dim)
    x_flat = x_h.view(-1, dim)               # [B·N·N, dim]

    # Concatenate weights → shape (2*hidden, dim)
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2h, dim)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2h, dim)

    # One linear per side (FP16 matmul)
    left_cat = F.linear(x_flat, left_cat_w)   # [B·N·N, 2h]
    right_cat = F.linear(x_flat, right_cat_w) # [B·N·N, 2h]

    # Restore 4‑D shape
    left_cat   = left_cat.view(B, N, N, 2 * hidden)
    right_cat  = right_cat.view(B, N, N, 2 * hidden)

    # Split projections / gate pre‑activations
    left_proj_raw  = left_cat[..., :hidden]   # (B,N,N,h)
    left_gate_raw  = left_cat[..., hidden:]   # (B,N,N,h)
    right_proj_raw = right_cat[..., :hidden]
    right_gate_raw = right_cat[..., hidden:]

    # ------------------------------------------------------------------
    # 4️⃣ Apply optional mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(dtype=torch.float16).unsqueeze(-1)    # (B,N,N,1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,h)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 6️⃣ Combine projection + gate
    # ------------------------------------------------------------------
    left  = left_proj_raw * left_gate          # (B,N,N,h)
    right = right_proj_raw * right_gate        # (B,N,N,h)

    # ------------------------------------------------------------------
    # 7️⃣ N³ contraction – batched GEMM via Triton
    #    We compute for each hidden channel:
    #         out[b,i,j,h] = Σ_k left[b,i,k,h] · right[b,j,k,h]
    #    reshaped as (B·h, N, N) @ (B·h, N, N)ᵀ
    # ------------------------------------------------------------------
    # Move hidden dim to the batch dimension
    left_view  = left.permute(0, 3, 1, 2).contiguous()   # (B, h, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, h, N, N)

    batch = B * hidden
    left_flat  = left_view.view(batch, N, N)   # (batch, M=N, K=N)
    # right will be interpreted as (batch, K=N, N) → we keep the same tensor
    # and pass swapped strides to the kernel.
    right_flat = right_view.view(batch, N, N)

    # Output tensor for the contraction
    out_flat = torch.empty((batch, N, N), dtype=torch.float16, device=device)

    # Choose tile sizes (tuned for H100 FP16 Tensor Cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),  # B strides (swap K/N)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure the kernel has finished before we use the result
    torch.cuda.synchronize()

    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(out_fp32,
                            normalized_shape=(hidden,),
                            weight=to_out_norm_w.to(torch.float32),
                            bias=to_out_norm_b.to(torch.float32),
                            eps=1e-5)                              # (B,N,N,hidden) fp32

    # ------------------------------------------------------------------
    # 9️⃣ Output gate (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # (B,N,N,hidden) fp16

    # ------------------------------------------------------------------
    # 10️⃣ Apply gate and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm.to(torch.float16) * out_gate
    final = F.linear(gated, to_out_w)                     # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 11️⃣ Return float32 result as required by the spec
    # ------------------------------------------------------------------
    return final.to(torch.float32)
