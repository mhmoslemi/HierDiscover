"""
custom_kernel for the *outgoing* Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑norm the input (float32) over the feature dimension.
2️⃣ Cast to fp16 and apply three linear projections:
   - left_proj   (dim → hidden)
   - right_proj  (dim → hidden)
   - left_gate   (dim → hidden)   → sigmoid
   - right_gate  (dim → hidden)   → sigmoid
   - out_gate    (dim → hidden)   → sigmoid
   (We perform the two “proj+gate” pairs with torch.nn.functional.linear and
    reuse the same weight layout; the heavy work (inner‑product) is handled by
    cuBLAS/Tensor‑Cores.)

3️⃣ If a mask is present we fuse `mask * left_gate` and `mask * right_gate`
   into the projected tensors with a tiny Triton kernel.  This eliminates two
   extra memory passes.

4️⃣ Core N³ operation:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   realized as a batched half‑precision GEMM (torch.bmm) after reshaping
   left/right to (B·hidden, N, N).  Tensor‑Cores perform the accumulation in
   FP32 internally.

5️⃣ Second LayerNorm over the hidden dimension, multiplication by the
   output gate, and a final linear projection back to the original feature
   dimension.  The final tensor is returned in float32.

The only Triton‑based part is the mask‑gate fusion; all large dense
matrix–multiplications are delegated to cuBLAS for maximal throughput on an
H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * left_proj  (B,N,N,hidden)
    right_ptr,         # fp16 * right_proj (B,N,N,hidden)
    mask_ptr,          # fp16 * mask (B,N,N)
    left_gate_ptr,     # fp16 * left_gate (B,N,N,hidden)
    right_gate_ptr,    # fp16 * right_gate (B,N,N,hidden)
    # strides for left/right (batch, i, j, hidden)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (batch, i, j)
    stride_m0, stride_m1, stride_m2,
    # strides for gates (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,               # batch, sequence length, hidden dim
    BLOCK_H: tl.constexpr, # tile size along hidden dimension
):
    """
    Fuse mask, sigmoid gate and projection in‑place:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    One program per (b,i,j) element.
    """
    pid = tl.program_id(0)                     # (B * N * N) programs
    total = B * N * N
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # scalar mask value (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # iterate over the hidden dimension in blocks
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        # load projection, gate and mask‑broadcasted hidden slice
        left = tl.load(left_ptr + off_l + offs * stride_l3,
                       mask=mask_h, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs * stride_r3,
                        mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_g + offs * stride_g3,
                             mask=mask_h, other=tl.float16(0.0))

        # fused element‑wise multiplication
        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        # store back in‑place
        tl.store(left_ptr + off_l + offs * stride_l3, left_out,
                 mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out,
                 mask=mask_h)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (float32) – may be unused
        - weights      : dict of model parameters (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask``
    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N  # number of (b,i,j) positions

    # flatten spatial dimensions to obtain a dense (M, dim) matrix
    x_flat = x_fp16.view(M, dim)  # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias) – fp16 weights
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # left_proj & left_gate (concatenated to reduce memory traffic)
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    left_concat_flat = F.linear(x_flat, left_concat_w)   # (M, 2*H)
    right_concat_flat = F.linear(x_flat, right_concat_w)  # (M, 2*H)

    # split back
    left = left_concat_flat[:, :hidden_dim].view(B, N, N, hidden_dim)   # proj
    left_gate = left_concat_flat[:, hidden_dim:].view(B, N, N, hidden_dim)  # gate pre‑sigmoid
    right = right_concat_flat[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = right_concat_flat[:, hidden_dim:].view(B, N, N, hidden_dim)

    # gate activations (sigmoid)
    left_gate = torch.sigmoid(left_gate)   # fp16
    right_gate = torch.sigmoid(right_gate) # fp16

    # out‑gate (single linear → sigmoid)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask & gates (mask‑gate fusion via Triton if needed)
    # ------------------------------------------------------------------
    if nomask:
        # simple element‑wise multiplication – no Triton launch
        left = left * left_gate
        right = right * right_gate
    else:
        # broadcast mask to fp16 and launch Triton kernel
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = B * N * N
        _mask_gate_fuse_kernel[(total_programs,)](
            left, right, mask_fp16,
            left_gate, right_gate,
            # strides for left
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            # strides for right
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gates (same layout as left/right)
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # problem sizes
            B, N, hidden_dim,
            BLOCK_H=32,          # tile size – 32 works for hidden_dim ≤ 128
            num_warps=4,
        )
        # after the kernel, `left` and `right` already contain mask*gate*proj

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM: Σₖ left[..., i, k, h] * right[..., j, k, h]
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) → each "head" is a GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # half‑precision batched matrix multiplication (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N) fp16

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # multiply by out‑gate (cast to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])  # float32

    return out_final
