def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Load weights and convert to half where needed
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)
    
    # LayerNorm weights
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    
    # All linear weights without bias
    left_proj_w = to_half(weights["left_proj.weight"])
    left_gate_w = to_half(weights["left_gate.weight"])
    right_proj_w = to_half(weights["right_proj.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w = to_half(weights["out_gate.weight"])
    # Output linear
    to_out_w = to_half(weights["to_out.weight"])
    # Output layernorm
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    
    # 1. Input LayerNorm in FP32
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )
    # cast to fp16
    x_half = x_norm.half()
    # Flatten spatial dimensions
    x_flat = x_half.view(-1, dim)  # shape (B*N*N, dim)
    
    # Concatenate weights
    weight_all = torch.cat([left_proj_w,
                            left_gate_w,
                            right_proj_w,
                            right_gate_w,
                            out_gate_w], dim=0)  # shape (5*hidden_dim, dim)
    # Linear projection (single GEMM)
    all_proj = torch.nn.functional.linear(x_flat, weight_all)  # (B*N*N, 5*hidden_dim)
    # Split
    h = hidden_dim
    left_proj_raw = all_proj[:, 0:h]
    left_gate_raw = all_proj[:, h:2*h]
    right_proj_raw = all_proj[:, 2*h:3*h]
    right_gate_raw = all_proj[:, 3*h:4*h]
    out_gate_raw = all_proj[:, 4*h:5*h]
    
    # Gating: sigmoid
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)
    
    # Apply gating
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate
    
    # Apply mask if any
    if mask is not None:
        mask_f = mask.view(-1, 1).to(dtype=torch.float16)  # shape (B*N*N,1)
        left = left * mask_f
        right = right * mask_f
    
    # Reshape left/right to (B, hidden_dim, N, N)
    left = left.view(B, N, N, h).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, h).permute(0, 3, 1, 2).contiguous()
    
    # Flatten batch and head for batched GEMM
    batch = B * hidden_dim
    left_flat = left.view(batch, N, N).contiguous()
    right_flat = right.view(batch, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)  # fp16
    
    # Prepare strides for kernel
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)
    stride_b_batch = right_flat.stride(0)
    stride_b_i = right_flat.stride(1)  # i (row index) corresponds to j in output
    stride_b_k = right_flat.stride(2)
    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)
    
    # Triton kernel launch
    import triton
    import triton.language as tl
    
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    
    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    
    # back to original shape (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16
    
    # Output LayerNorm (float32)
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)
    
    # Apply out_gate (reshaped)
    out_gate = out_gate.view(B, N, N, hidden_dim)  # shape same as out_norm
    
    gated = out_norm * out_gate
    
    # Final projection
    final = torch.nn.functional.linear(gated, to_out_w)  # (B,N,N,dim) half
    
    return final.to(torch.float32)

    a_ptr, b_ptr, c_ptr,

"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward pass.

Algorithm
---------
1. Input LayerNorm (float32) → cast to fp16.
2. Fuse the four 1×1 linear projections (left/right) and the three gate
   projections (left_gate/right_gate/out_gate) into a single large GEMM:
       Z = x_flat @ W_all.T   where W_all ∈ ℝ^{5·H×C}
   Split Z into the five heads and apply sigmoid gates.
3. Apply the optional pair‑wise mask (broadcast over the hidden dimension).
4. Reshape the gated left/right tensors to (B·H, N, N) and compute the
   N³ contraction  out = left @ rightᵀ per head with a custom Triton
   batched‑GEMM kernel (fp16 × fp16 → fp32 accumulation → fp16 output).
5. Reshape the result back to (B, N, N, H) (fp16).
6. Output LayerNorm over the hidden dimension (float32 for stability).
7. Multiply by the out‑gate (sigmoid of the fused projection).
8. Final linear projection (H → C) and return in float32.

The critical N³ contraction is performed by Triton, while the fused
projections/gates reduce the number of separate cuBLAS kernels from five
to one.  All heavy arithmetic stays in fp16 (except the LayerNorms,
which run in fp32 for numerical safety).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton batched GEMM: C[b] = A[b] @ B[b]^T  (A,B ∈ ℝ^{N×N}, fp16)
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                # const half *  (left)
    b_ptr,                # const half *  (right)
    c_ptr,                #       half *  (output)
    M, N, K,              # dimensions (M=N, N=N, K=N)
    stride_a_batch,       # stride for batch dim in A
    stride_a_m,           # stride for row   dim in A
    stride_a_k,           # stride for col   dim in A
    stride_b_batch,       # stride for batch dim in B (original layout)
    stride_b_i,           # stride for row   dim in B (used as column after transpose)
    stride_b_k,           # stride for col   dim in B
    stride_c_batch,       # stride for batch dim in C
    stride_c_m,           # stride for row   dim in C
    stride_c_n,           # stride for col   dim in C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]  (i,j,k ∈ N)

    Grid: (batch, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -----------------------------------------------------------------
    # 1️⃣ Compute global offsets for this block
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # -----------------------------------------------------------------
    # 2️⃣ Accumulate over k dimension
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_idxs = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_idxs < K

        # A block: rows = offs_m, cols = k_idxs
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_a_m
            + k_idxs[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block (transposed on‑fly): rows = k_idxs, cols = offs_n
        b = tl.load(
            b_batch
            + k_idxs[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → accumulate in FP32
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # 3️⃣ Write result (cast back to fp16)
    # -----------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# Custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : model parameters (float32 tensors)
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack and move everything to the correct device
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ Load model weights (all linear layers have bias=False)
    # --------------------------------------------------------------
    # LayerNorm over the input channels
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    # Linear / gate weights (store in fp16 to feed Triton)
    left_proj_w   = to_half(weights["left_proj.weight"])   # (H, C)
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])

    # Final projection and its LayerNorm
    to_out_w      = to_half(weights["to_out.weight"])      # (C, H)
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])

    # --------------------------------------------------------------
    # 2️⃣ Input LayerNorm (float32 for stability) → fp16
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, C]  (float32)
    x_half = x_norm.half()                         # fp16
    x_flat = x_half.view(-1, dim)                  # (B·N·N, C)

    # --------------------------------------------------------------
    # 3️⃣ Fuse all projections + gates into ONE large GEMM
    #    Output dimension = 5·H  (left_proj, left_gate, right_proj,
    #    right_gate, out_gate)
    # --------------------------------------------------------------
    w_all = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    )  # (5·H, C), fp16
    all_proj = F.linear(x_flat, w_all)  # (B·N·N, 5·H)

    H = hidden_dim
    left_proj_raw = all_proj[:, 0:H]
    left_gate_raw = all_proj[:, H:2 * H]
    right_proj_raw = all_proj[:, 2 * H:3 * H]
    right_gate_raw = all_proj[:, 3 * H:4 * H]
    out_gate_raw = all_proj[:, 4 * H:5 * H]

    # Gates → sigmoid (still fp16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)   # (B·N·N, H)

    # Apply gates to the projected tensors
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.view(-1, 1).to(dtype=torch.float16)  # (B·N·N,1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 5️⃣ Reshape for the N³ contraction
    #    left/right : (B, H, N, N)   (contiguous)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # Fold batch & head dimensions → (B·H, N, N)
    batch = B * hidden_dim
    left_flat = left.view(batch, N, N).contiguous()
    right_flat = right.view(batch, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)   # fp16 output buffer

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM via Triton (left @ rightᵀ per head)
    # --------------------------------------------------------------
    # Strides for the flattened tensors
    stride_a_batch = left_flat.stride(0)  # N*N
    stride_a_m = left_flat.stride(1)      # N
    stride_a_k = left_flat.stride(2)      # 1

    stride_b_batch = right_flat.stride(0) # N*N
    stride_b_i = right_flat.stride(1)     # N   (used as column after transpose)
    stride_b_k = right_flat.stride(2)     # 1

    stride_c_batch = out_flat.stride(0)   # N*N
    stride_c_m = out_flat.stride(1)       # N
    stride_c_n = out_flat.stride(2)       # 1

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_gemm_kernel[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 8️⃣ Output LayerNorm (float32) → fp16
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=eps,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 9️⃣ Apply out‑gate (broadcasted) and final projection
    # --------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim)          # (B,N,N,H) fp16
    gated = out_norm * out_gate                           # element‑wise gate
    final = F.linear(gated, to_out_w)                     # (B,N,N,C) fp16

    # --------------------------------------------------------------
    # 10️⃣ Return in float32 as required by the specification
    # --------------------------------------------------------------
    return final.to(torch.float32)
