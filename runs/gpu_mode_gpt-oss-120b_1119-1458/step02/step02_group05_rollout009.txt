import triton.language as tl

@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    hidden_dim: tl.constexpr,
    n_elem: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elem

    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # Mask broadcasting: each mask index corresponds to element // hidden_dim
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)

   mask is present).  When there is no mask the element‑wise product is done
   directly in PyTorch, which is faster for the common “nomask” case.
4️⃣ Core O(N³) step: for each hidden channel compute
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is implemented as a batched GEMM:
   left/right are reshaped to (B·hidden, N, N) and `torch.bmm` (FP16→FP32
   accumulation) is used – a Tensor‑Core‐friendly operation on H100.
5️⃣ Layer‑Norm over the hidden dimension, multiply by the out‑gate
   (broadcasted), and a final FP32 linear projection back to the original
   feature dimension.
6️⃣ The final tensor has shape [B, N, N, dim] and dtype float32.

All heavy arithmetic is performed in FP16 (except the two LayerNorms and
the final linear), the whole forward runs with ``torch.no_grad()`` to
eliminate autograd overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_kernel(
    proj_ptr,                     # fp16: projection (left or right)
    gate_ptr,                     # fp16: gate after sigmoid
    mask_ptr,                     # fp16: mask (flattened B·N·N)
    out_ptr,                      # fp16: destination
    hidden_dim: tl.constexpr,     # hidden dimension (for mask broadcasting)
    n_elements: tl.constexpr,     # total number of elements in proj/gate
    BLOCK: tl.constexpr,          # compile‑time block size
):
    """fused (proj * gate * mask) for the mask‑present case."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < n_elements

    # Load projection & gate in fp32 for higher‑precision multiplication
    proj = tl.load(proj_ptr + offs, mask=valid).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=valid).to(tl.float32)

    # Each mask entry repeats `hidden_dim` times along the hidden axis
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32
            - mask:        [B, N, N]  (float, 0/1) – may be ignored when config["nomask"]=True
            - weights: dict with the 9 weight tensors required by the module
            - config:  {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5  # LayerNorm epsilon
    device = input_tensor.device

    with torch.no_grad():
        # ------------------------------------------------------------------
        # 1️⃣ LayerNorm over the feature dimension (float32)
        # ------------------------------------------------------------------
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps,
        )  # [B, N, N, dim]  fp32

        # ------------------------------------------------------------------
        # 2️⃣ Fuse the five linear ops into one mat‑mul
        # ------------------------------------------------------------------
        x_h = x.to(torch.float16)  # cast once

        # concatenate the five weight matrices (out_features stacked)
        w_concat = torch.cat(
            [
                weights["left_proj.weight"],
                weights["right_proj.weight"],
                weights["left_gate.weight"],
                weights["right_gate.weight"],
                weights["out_gate.weight"],
            ],
            dim=0,
        ).to(torch.float16)  # shape (5*hidden_dim, dim)

        # single linear → shape (B,N,N,5*hidden_dim)
        combined = F.linear(x_h, w_concat)  # no bias (as in original)

        # split the concatenated tensor
        left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
            combined, hidden_dim, dim=-1
        )  # each [B,N,N,hidden]

        # ---------------------------------------------------------------
        # 3️⃣ Gates (sigmoid) – keep gates in FP16, out_gate later promoted
        # ---------------------------------------------------------------
        left_gate = torch.sigmoid(left_gate_raw)          # FP16
        right_gate = torch.sigmoid(right_gate_raw)        # FP16
        # out_gate is needed in FP32 after the second LN
        out_gate = torch.sigmoid(out_gate_raw.float())    # FP32

        # ------------------------------------------------------------------
        # 4️⃣ Apply mask (if any) and fuse with gates
        # ------------------------------------------------------------------
        if not nomask:
            # mask: [B,N,N] → [B,N,N,1] → flatten → FP16
            mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
            mask_flat = mask_f.view(-1)                               # (B·N·N)

            # allocate destination tensors
            left = torch.empty_like(left_proj)
            right = torch.empty_like(right_proj)

            n_elem = left_proj.numel()  # B*N*N*hidden
            BLOCK = 65536
            grid = (triton.cdiv(n_elem, BLOCK),)

            # left = left_proj * left_gate * mask
            _gate_mask_kernel[grid](
                left_proj,
                left_gate,
                mask_flat,
                left,
                hidden_dim,
                n_elem,
                BLOCK=BLOCK,
            )
            # right = right_proj * right_gate * mask
            _gate_mask_kernel[grid](
                right_proj,
                right_gate,
                mask_flat,
                right,
                hidden_dim,
                n_elem,
                BLOCK=BLOCK,
            )
        else:
            # Simple element‑wise product (no mask)
            left = left_proj * left_gate
            right = right_proj * right_gate

        # ------------------------------------------------------------------
        # 5️⃣ Core O(N³) operation – batched GEMM (left @ rightᵀ)
        # ------------------------------------------------------------------
        B, N, _, _ = left.shape

        # reshape to (B*hidden, N, N); make contiguous for cuBLAS
        left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_b = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

        # batched matrix multiply (FP16×FP16 → FP32 accumulation)
        out_b = torch.bmm(left_b, right_b.transpose(1, 2))  # [B*hidden, N, N] (fp16)

        # reshape back to [B, N, N, hidden] and promote to FP32 for the next LN
        out = (
            out_b.view(B, hidden_dim, N, N)
            .permute(0, 2, 3, 1)
            .contiguous()
            .float()
        )  # [B, N, N, hidden] in FP32

        # ------------------------------------------------------------------
        # 6️⃣ Second LayerNorm, out‑gate multiplication and final projection
        # ------------------------------------------------------------------
        out = F.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=eps,
        )  # FP32

        # multiply by out_gate (broadcast over hidden dimension)
        out = out * out_gate

        # final linear projection back to original dim (FP32 weight)
        out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

        return out
```
