    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # LayerNorm over dim
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)  # float32

    # Cast to fp16 for heavy compute
    x_h = x.to(torch.float16)

    # Load weights in fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right_proj = F.linear(x_h, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w)).to(torch.float32)  # keep FP32

    B, N, _, _ = input_tensor.shape

    # Prepare mask (broadcast across hidden)
    if nomask:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=input_tensor.device)
    else:
        mask_flat = mask.unsqueeze(-1).contiguous().view(-1).to(torch.float16)  # [B,N,N]

    # Allocate buffers for gated+masked tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Triton launch
    n_elements = left_proj.numel()
    BLOCK = 65536  # threads per program (tuned)
    grid = (triton.cdiv(n_elements, BLOCK),)

    _gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        left,
        hidden_dim,
        n_elements,
        BLOCK=BLOCK,
    )

    _gate_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        right,
        hidden_dim,
        n_elements,
        BLOCK=BLOCK,
    )

    # Core triangle multiplicative update (batched GEMM across the third dim)
    # Using einsum avoids a costly contiguous+view+batched matmul.
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # fp16

    # Convert to FP32 for subsequent layernorm and gating
    out = out.to(torch.float32)

    # Hidden-dim LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # Apply output gate (broadcast over hidden)
    out = out * out_gate

    out_h = out.to(torch.float16)
    output = F.linear(out_h, to_out_w)  # [B,N,N,dim] (fp16)

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the feature dimension (float32 for numerical stability).
2️⃣  Cast to FP16 and apply the two linear projections (left/right) and the three
    gating projections.  Gating values go through a sigmoid.
3️⃣  Fuse element‑wise multiplication of projection × gate × mask (when present)
    using a tiny Triton kernel.  This avoids three separate memory passes.
4️⃣  Compute the O(N³) “triangular” update with a single einsum:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    This replaces a costly permutation/contiguous/torch.bmm sequence and
    eliminates the extra memory copy.
5️⃣  Layer‑norm on the hidden dimension, multiply by the output gate and finally
    project back to the original feature dimension with a FP16 GEMM.
6️⃣  Return the result as float32.

All heavy arithmetic (linear projections, the O(N³) contraction and the final
projection) is performed in FP16; accumulation is done in FP32 by PyTorch’s
native kernels.  The Triton kernel is used for the projection × gate × mask
fusion, satisfying the requirement to employ Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_kernel(
    proj_ptr,                  # fp16*  – projection tensor (left or right)
    gate_ptr,                  # fp16*  – corresponding gate tensor
    mask_ptr,                  # fp16*  – mask flattened [B*N*N]
    out_ptr,                   # fp16*  – destination
    hidden_dim,                # i64    – hidden dimension (used for broadcasting mask)
    n_elements,                # i64    – total number of elements in proj/gate (B*N*N*hidden)
    BLOCK: tl.constexpr,       # compile‑time block size (threads per program)
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)               # linear offsets
    mask = offs < n_elements

    # Load projection and gate (up‑cast to fp32 for higher‑precision mul)
    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # Broadcast mask: each mask entry repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * m                # fp32 intermediate
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] float32
            - mask_tensor : [B, N, N]   (may be None)
            - weights_dict: model weights (see code for required keys)
            - config_dict : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for all subsequent heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # Load weight tensors in FP16 (except LayerNorm weights which stay FP32)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"]   # FP32
    to_out_norm_b = weights["to_out_norm.bias"]     # FP32
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP16) and gating values (sigmoid)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_h, left_proj_w)          # [B, N, N, hidden]
    right_proj = F.linear(x_h, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w)).to(torch.float32)

    B, N, _, _ = input_tensor.shape
    num_masks = B * N * N

    # ------------------------------------------------------------------

    else:
        # mask : [B, N, N] → [B, N, N, 1] → flatten
        mask_flat = mask.unsqueeze(-1).contiguous().view(-1).to(torch.float16)

    # Allocate output tensors for the gated & masked values
    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    n_elem = left_proj.numel()                      # B*N*N*hidden
    BLOCK = 65536                                   # threads per program (tuned)
    grid = (triton.cdiv(n_elem, BLOCK),)

    # left = left_proj * left_gate * mask
    _gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        left,
        hidden_dim,
        n_elem,
        BLOCK=BLOCK,
    )
    # right = right_proj * right_gate * mask
    _gate_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        right,
        hidden_dim,
        n_elem,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³) contraction using a single einsum (no extra copy)
    # ------------------------------------------------------------------
    # left, right : [B, N, N, hidden] (fp16)
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # fp16
    out = out.to(torch.float32)                                     # fp32 for layernorm

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                                               # fp32

    out = out * out_gate                            # broadcast over hidden

    # Cast to fp16 for the final linear projection
    out_h = out.to(torch.float16)
    output = F.linear(out_h, to_out_w)              # [B, N, N, dim] fp16
    output = output.to(torch.float32)               # final dtype required by the model

    return output
```
