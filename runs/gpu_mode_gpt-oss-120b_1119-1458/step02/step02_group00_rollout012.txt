"""
Optimized outgoing Triangle Multiplicative Update (TriMul) – forward only.

Key ideas
---------
1. **Fuse the 5 linear projections (left/right + three gates) into a single
   matrix multiplication**.  The input tensor `x` (after the first LayerNorm)
   is flattened to `[B·N·N, dim]` and multiplied by a concatenated weight
   matrix of shape `[5·hidden_dim, dim]`.  This reads the input only once and
   reduces kernel launch overhead.

2. **Element‑wise operations** (masking, gating, final out‑gate multiplication)
   are performed with simple PyTorch broadcasting – cheap compared to the
   N³ step.  The only mandatory Triton kernel is a tiny element‑wise multiply
   used after the second LayerNorm.

3. **Core N³ “einsum”** is realized as a batched matrix multiplication
   (B·hidden_dim batched GEMMs of size N×N) using `torch.bmm`.  The matrices
   are laid out as `(B·H, N, N)` and `(B·H, N, N)` with the second operand
   transposed so the operation matches the required `'... i k d, ... j k d -> ... i j d'`
   pattern.  This leverages H100 Tensor‑Core GEMM paths (FP16).

4. **Mixed‑precision** – all heavy work is done in FP16 while the two
   LayerNorms are performed in FP16 with FP32 accumulation (the default
   behaviour of `F.layer_norm`).  The final output is cast back to FP32.

The overall data flow (shapes) is:

    x (fp32) ──► LN ──► fp16
        └─► combined linear (5·hidden_dim) ──► split
            ├─ left_proj, right_proj (fp16)
            ├─ left_gate , right_gate , out_gate (sigmoid, fp16)
            └─ mask (if present) broadcasted onto proj
        └─ left = left_proj * left_gate, right = right_proj * right_gate
    left, right (fp16) ──► batched GEMM (B·H × N × N) ──► out (fp16)
    out ──► LN (hidden_dim) ──► out_norm (fp16)
    out_norm ──► * out_gate (element‑wise Triton kernel) ──► out_mul (fp16)
    out_mul ──► final linear (hidden_dim → dim) ──► result (fp32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny Triton kernel – element‑wise multiply (used for out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # fp16 input  (out_norm)
    b_ptr,          # fp16 input  (out_gate)
    out_ptr,        # fp16 output
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    out = a * b   (both flattened)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights      : dict of model parameters (see keys in the code)
        - config       : dict containing at least
                         * "dim"        – input channel dimension
                         * "hidden_dim" – hidden dimension used in the module
                         * "nomask" (optional) – whether the mask is all‑ones

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half precision for compute
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32
    x_h = x.to(torch.float16)  # FP16 for the heavy work

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    #    (left_proj, right_proj, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    # Concatenate weight matrices along the output‑feature dimension
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]
    # shape after cat: (5*hidden_dim, dim)
    weight_concat = torch.cat(
        [
            w_left_proj,
            w_right_proj,
            w_left_gate,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(torch.float16)

    B, N, _, _ = x_h.shape
    # flatten the (B,N,N) positions
    x_flat = x_h.view(-1, dim)                         # [B·N·N, dim]
    y = F.linear(x_flat, weight_concat)               # [B·N·N, 5·hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Split & apply sigmoid to gate logits
    # --------------------------------------------------------------
    H = hidden_dim
    # reshape to make slicing easier: [B·N·N, 5, H]
    y = y.view(-1, 5, H)

    left_proj   = y[:, 0, :]               # [B·N·N, H]
    right_proj  = y[:, 1, :]               # [B·N·N, H]
    left_gate   = torch.sigmoid(y[:, 2, :])
    right_gate  = torch.sigmoid(y[:, 3, :])
    out_gate    = torch.sigmoid(y[:, 4, :])

    # --------------------------------------------------------------
    # 4️⃣ Reshape back to 4‑D tensors
    # --------------------------------------------------------------
    left_proj   = left_proj.view(B, N, N, H)
    right_proj  = right_proj.view(B, N, N, H)
    left_gate   = left_gate.view(B, N, N, H)
    right_gate  = right_gate.view(B, N, N, H)
    out_gate    = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 6️⃣ Apply gates (element‑wise)
    # --------------------------------------------------------------
    left  = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Core N³ operation via batched GEMM:
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # --------------------------------------------------------------
    total_heads = B * H

    # left: (B, H, i, k) → view as (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    # right: (B, H, k, j) → view as (B·H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(total_heads, N, N)

    # Batched matrix multiply (Tensor‑Core FP16 GEMM)
    out_mat = torch.bmm(left_mat, right_mat)            # [B·H, N, N]

    # Restore the original layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ Post‑processing LayerNorm (hidden_dim)
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # FP16

    # --------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (Triton element‑wise kernel)
    # --------------------------------------------------------------
    out_mul = torch.empty_like(out_norm)
    numel = out_norm.numel()
    BLOCK = 65536
    grid = ((numel + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out_norm.view(-1),
        out_gate.contiguous().view(-1),
        out_mul.view(-1),
        numel,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # --------------------------------------------------------------
    final = F.linear(
        out_mul,
        weights["to_out.weight"].to(torch.float16),  # [dim, hidden_dim]
    )  # [B, N, N, dim]  FP16

    # Convert to the expected float32 output
    return final.to(torch.float32)
