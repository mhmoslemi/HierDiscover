def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    eps = 1e-5

    # layernorm on dim
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps)

    # cast to fp16 for linear ops
    x_h = x.to(torch.float16)

    # flatten to (M, dim)
    B, N, _, _ = x_h.shape
    M = B * N * N
    x_flat = x_h.view(M, dim)

    # combine projection and gate weight matrices
    # vertical stacking order: left_proj, left_gate, right_proj, right_gate, out_gate
    weight_combined = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]], dim=0).to(torch.float16)

    # compute all linear outputs together
    combined = torch.nn.functional.linear(x_flat, weight_combined)  # shape (M, 5*hidden)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # split
    left_proj = combined[..., :hidden_dim]
    left_gate = combined[..., hidden_dim:2 * hidden_dim]
    right_proj = combined[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = combined[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate = combined[..., 4 * hidden_dim:]

    # apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # apply mask and fuse with gate via Triton kernel
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)  # shape B*N*N
        # allocate output
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # kernel launch
        BLOCK = 65536
        n_elem = left_proj.numel()
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core O(N^3) operation using batched GEMM
    # reshape for batched matmul: (B*H, N, N)
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))  # -> (B*H, N, N)
    out = out_batched.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B, N, N, H) in fp32? It's fp16 currently
    out = out.to(torch.float32)  # convert to FP32 for layernorm

    # hidden dim LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Multiply by out_gate (broadcast)
    out = out * out_gate

    # Final projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16)).to(torch.float32)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    hidden_dim, n_elements,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # mask per position: repeat each mask hidden_dim times
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Algorithm
    ---------
    1. LayerNorm over ``dim`` (FP32 for numerical stability).
    2. Fuse the five Linear projections (left/right value and gate, plus out_gate)
       into a single GEMM to minimise passes over the input.
    3. Apply sigmoid to the three gate tensors.
    4. Fuse element‑wise ``proj * gate * mask`` (if a mask is present) with a tiny
       Triton kernel.
    5. Core O(N³) operation: for each hidden channel compute a matrix product
       ``C = A @ Bᵀ`` where ``A = left`` and ``B = right``. Implemented as a
       batched GEMM (B*H matrices of size N×N) using ``torch.bmm``.
    6. LayerNorm on the hidden dimension, element‑wise multiplication with
       ``out_gate`` and final linear projection back to ``dim``.
    All heavy arithmetic is carried out in FP16 (except the two LayerNorms,
    which stay in FP32).  The final output is float32, matching the reference.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  FP32

    # 2. Cast to fp16 and flatten
    x_h = x.to(torch.float16)
    B, N, _, _ = x_h.shape
    M = B * N * N
    x_flat = x_h.view(M, dim)   # (B*N*N, dim)

    # 3. Combined linear for projections and gates (no bias)
    # Stack weights in the order: left_proj, left_gate, right_proj, right_gate, out_gate
    weight_combined = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*hidden, dim)

    combined = torch.nn.functional.linear(x_flat, weight_combined)  # (M, 5*hidden)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # 4. Split combined tensor
    left_proj = combined[..., :hidden_dim]
    left_gate = combined[..., hidden_dim:2 * hidden_dim]
    right_proj = combined[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = combined[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate = combined[..., 4 * hidden_dim:]

    # 5. Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # 6. Apply mask (if any) and fuse with gate using Triton
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] -> flatten
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)   # (B*N*N)

        # allocate fused tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        BLOCK = 65536
        n_elem = left_proj.numel()
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7. Core O(N³) – batched matrix multiplication across hidden channels
    # reshape to (B*hidden, N, N) for torch.bmm
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))

    # reshape back to [B, N, N, hidden]  (FP16)
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )   # [B, N, N, hidden]

    # 8. Hidden‑dim LayerNorm (FP32)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 9. Apply out‑gate (element‑wise)
    out = out * out_gate   # broadcast over hidden dim

    # 10. Final linear projection back to original dim
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16)
    ).to(torch.float32)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – optimized forward pass.

Algorithm
---------
1️⃣ Layer‑norm over the feature dimension (float32 for stability).
2️⃣ Fuse *all* linear projections (left/right values, left/right gates,
   out‑gate) into a single GEMM to read the input only once.
3️⃣ Apply sigmoid to the three gate tensors.
4️⃣ If a mask is present, fuse `proj * gate * mask` with a small Triton kernel.
5️⃣ Core O(N³) operation: for each hidden channel compute a matrix product
   C = A @ Bᵀ where A/B are the gated tensors.
   Implemented as a batched GEMM (B·H matrices of size N×N) via torch.bmm.
6️⃣ Hidden‑dim layer‑norm (float32), element‑wise multiplication with out‑gate,
   and final linear projection back to the original feature dimension.

All heavy arithmetic is performed in FP16 (Tensor‑Core GEMMs).  The two
LayerNorms stay in FP32, and the final output is cast back to float32 to
match the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask (mask optional)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16*  input projection (left or right)
    gate_ptr,          # fp16*  corresponding gate
    mask_ptr,          # fp16*  mask values (flattened, length = B·N·N)
    out_ptr,           # fp16*  destination
    hidden_dim: tl.int64,   # hidden dimension (repeat count for mask)
    n_elements: tl.int64,   # total number of elements (B·N·N·H)
    BLOCK: tl.constexpr,    # block size (compile‑time)
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < n_elements

    # load projection & gate (fp16 → fp32 for higher‑precision multiply)
    proj = tl.load(proj_ptr + offs, mask=valid).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=valid).to(tl.float32)

    # broadcast mask: each mask entry repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=valid)


# ----------------------------------------------------------------------
# Triton‑backed TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor – shape [B, N, N, dim] (float32)
            mask         – shape [B, N, N] (bool/int, optional)
            weights      – dictionary with the model’s parameters
            config       – dict containing ``dim`` and ``hidden_dim`` and a
                           ``nomask`` flag.

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                     # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and flatten for a single GEMM
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # FP16
    B, N, _, _ = x_h.shape
    M = B * N * N                                 # number of (b,i,j) positions
    x_flat = x_h.view(M, dim)                    # (B·N·N, dim)

    # ------------------------------------------------------------------
    # 3️⃣ Combine all five linear layers (proj‑gate‑out_gate) into one
    # ------------------------------------------------------------------
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    weight_combined = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                          # (5·hidden, dim)

    combined = torch.nn.functional.linear(x_flat, weight_combined)  # (M, 5·hidden)
    combined = combined.view(B, N, N, 5 * hidden_dim)                # [B,N,N,5·H]

    # ------------------------------------------------------------------
    # 4️⃣ Split and apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    left_proj = combined[..., :hidden_dim]
    left_gate = combined[..., hidden_dim:2 * hidden_dim]
    right_proj = combined[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = combined[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate = combined[..., 4 * hidden_dim:]

    left_gate = torch.sigmoid(left_gate)      # FP16
    right_gate = torch.sigmoid(right_gate)    # FP16
    out_gate = torch.sigmoid(out_gate)        # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask (if any) with projection & gate via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → flatten
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                     # (B·N·N,)

        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        BLOCK = 65536
        n_elem = left_proj.numel()                      # B·N·N·H
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # no mask → simple element‑wise product
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³) – batched GEMM across hidden channels
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # (B·H, N, N) @ (B·H, N, N).T → (B·H, N, N)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # back to [B, N, N, H] (float16) → float32 for the following LayerNorm
    out = (
        out_b.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )                                            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate (broadcast over hidden) and final projection
    # ------------------------------------------------------------------
    out = out * out_gate                         # broadcast product

    # final linear: hidden → dim (use FP16 weight for speed)
    out = torch.nn.functional.linear(
        out.to(torch.float16),                    # cast to fp16 for the GEMM
        weights["to_out.weight"].to(torch.float16),
    ).to(torch.float32)                         # cast back to fp32

    return out
