@triton.jit
def triton_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_head, N,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bi, stride_bk,
    stride_ch, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # batch_head

    # compute tile indices for output matrix (i,j)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K) # will be used inside loop

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # Load A (left) element: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + pid * stride_ah + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None,:],
            other=0.0
        )
        # Load B (right) transposed element: shape (BLOCK_K, BLOCK_N)
        # B is originally (batch_head, N, N) as right; we need B_T means we index at (k, j)
        b = tl.load(
            b_ptr + pid * stride_bh + cur_k[:, None] * stride_bi + offs_j[None, :] * stride_bk,
            mask=mask_k[:, None] & mask_j[None,:],
            other=0.0
        )
        # FP16 dot product, accumulate as FP32
        acc += tl.dot(a, b).to(tl.float32)

    # store result
    tl.store(
        c_ptr + pid * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_ck,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None,:]
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_h = x.half()
    # Concatenate weights for projections and gates
    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()
    stacked_weights = torch.cat([w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0) # shape (5*hidden_dim, dim)
    # Linear
    stacked_out = torch.nn.functional.linear(x_h, stacked_weights) # shape (B,N,N,5*hidden_dim)
    # split
    H = hidden_dim
    left = stacked_out[..., 0:H]
    right = stacked_out[..., H:2*H]
    left_gate = stacked_out[..., 2*H:3*H].sigmoid()
    right_gate = stacked_out[..., 3*H:4*H].sigmoid()
    out_gate = stacked_out[..., 4*H:].sigmoid()

    if not nomask:

        left = left * m
        right = right * m
    # gates

    right = right * right_gate

    left = left.permute(0,3,1,2).contiguous()

@triton.jit
def triton_outgoing_matmul(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_ah, stride_ai, stride_aj,
    stride_bh, stride_bi, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program id for batch-head dimension
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    start_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    start_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = start_i < N
    mask_j = start_j < N
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        start_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = start_k < N
        # Load A block
        a = tl.load(
            a_ptr + bh * stride_ah + start_i[:, None] * stride_ai + start_k[None, :] * stride_aj,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B block (transposed)
        b = tl.load(
            b_ptr + bh * stride_bh + start_k[:, None] * stride_bi + start_j[None, :] * stride_bj,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # Dot product
        acc += tl.dot(a, b).to(tl.float32)
    # Store result
    tl.store(
        c_ptr + bh * stride_ch + start_i[:, None] * stride_ci + start_j[None, :] * stride_cj,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]

"""
Optimized outgoing TriMul (AlphaFold3) forward.
Algorithm:
1. Layer‑norm the [B,N,N,C] input in fp32.
2. Cast to fp16 and fuse the five linear projections (left, right,
   left‑gate, right‑gate, out‑gate) into a single matmul using a stacked
   weight matrix.
3. Apply optional mask to left/right, multiply by the two gates.
4. Rearrange to (B*H, N, N) and compute the expensive N³ reduction
   `out = left @ rightᵀ` with a custom Triton batched‑GEMM kernel,
   accumulating in fp32 and storing in fp16.
   For very large problems we fall back to torch.bmm (cuBLAS).
5. Reshape to (B,N,N,H), convert to fp32, apply the output LayerNorm,
   multiply by the out‑gate and finish with a linear projection back to
   C dimensions.
All heavy compute runs in half‑precision on Tensor‑Cores; the final
result is float32.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def triton_outgoing_matmul(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_ah, stride_ai, stride_aj,
    stride_bh, stride_bi, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)          # batch*head index
    pid_i = tl.program_id(1)      # tile row
    pid_j = tl.program_id(2)      # tile column

    # tile coordinates
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over K -------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: (BLOCK_M, BLOCK_K) = left[bh, i, k]
        a = tl.load(
            a_ptr + bh * stride_ah
                 + offs_i[:, None] * stride_ai
                 + offs_k[None, :] * stride_aj,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ block: (BLOCK_K, BLOCK_N) = right[bh, k, j]
        b = tl.load(
            b_ptr + bh * stride_bh
                 + offs_k[:, None] * stride_bi
                 + offs_j[None, :] * stride_bj,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b).to(tl.float32)

    # ---- store ------------------------------------------------------------
    tl.store(
        c_ptr + bh * stride_ch
              + offs_i[:, None] * stride_ci
              + offs_j[None, :] * stride_cj,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and fuse the five linear projections
    # ------------------------------------------------------------------
    x_h = x.half()                                 # [B, N, N, dim] fp16

    # Gather half‑precision weights
    w_left_proj   = weights["left_proj.weight"].half()
    w_right_proj  = weights["right_proj.weight"].half()
    w_left_gate   = weights["left_gate.weight"].half()
    w_right_gate  = weights["right_gate.weight"].half()
    w_out_gate    = weights["out_gate.weight"].half()

    # Stack weights along the output dimension: (5*H, dim)
    stacked_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )  # [5*H, dim], fp16

    # Single fused matmul: result shape [B, N, N, 5*H]
    fused = torch.nn.functional.linear(x_h, stacked_weight)  # no bias

    # ------------------------------------------------------------------
    # 3. Split into the five tensors and apply activation / mask
    # ------------------------------------------------------------------
    H = hidden_dim
    left      = fused[..., 0:H]
    right     = fused[..., H:2*H]
    left_gate = fused[..., 2*H:3*H].sigmoid()
    right_gate= fused[..., 3*H:4*H].sigmoid()
    out_gate  = fused[..., 4*H:5*H].sigmoid()   # shape [B,N,N,H]

    if not nomask:
        # mask broadcasting: (B,N,N,1)
        m = mask.unsqueeze(-1).to(torch.float16)
        left = left * m
        right = right * m

    # Apply gate to projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Batched N³ reduction (left @ rightᵀ)
    #    Rearrange to (B*H, N, N) – contiguous for the kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, H, N, N) -> (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * H, N, N)   # fp16
    right_flat = right.view(B * H, N, N)

    BH = B * H
    # Heuristic: use Triton for modest batch‑head sizes; otherwise cuBLAS.
    # Triton is fast for up to a few hundred batch‑head matrices.
    use_triton = (BH <= 256) and (N <= 1024)

    if use_triton:
        out_flat = torch.empty_like(left_flat)  # fp16 output buffer

        # strides in element units (not bytes)
        stride_a_h, stride_a_i, stride_a_j = left_flat.stride()
        stride_b_h, stride_b_i, stride_b_j = right_flat.stride()
        stride_c_h, stride_c_i, stride_c_j = out_flat.stride()

        # Tuned block sizes – work well on H100 (Tensor‑Cores)
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )
        triton_outgoing_matmul[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            stride_a_h,
            stride_a_i,
            stride_a_j,
            stride_b_h,
            stride_b_i,
            stride_b_j,
            stride_c_h,
            stride_c_i,
            stride_c_j,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # torch.bmm maps to cuBLAS batched GEMM (Tensor‑Cores)
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
        out_flat = out_flat.half()

    # ------------------------------------------------------------------
    # 5. Reshape back to (B, N, N, H) and cast to fp32 for the rest
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out = out.float()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"]  # float32
    out = torch.nn.functional.linear(out, w_to_out)

    return out

   For very large problems we fall back to torch.bmm (cuBLAS) which also
   uses Tensor‑Cores.
5. Reshape back to (B, N, N, H), cast to FP32 and apply the output
   LayerNorm + out‑gate.
6. Final linear projection (H → C) in FP32 to produce the result tensor
   of shape [B, N, N, C] (float32).

All heavy arithmetic runs in half‑precision on Tensor‑Cores; the final
output is FP32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the N³ reduction.
# Computes   C = A @ Bᵀ   where A,B have shape (batch_head, N, N).
# Accumulation is performed in fp32 and the result is written as fp16.
# ----------------------------------------------------------------------
@triton.jit
def triton_outgoing_matmul(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_ah, stride_ai, stride_aj,
    stride_bh, stride_bi, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh   = tl.program_id(0)          # batch * head index
    pid_i = tl.program_id(1)        # tile row
    pid_j = tl.program_id(2)        # tile col

    # Tile offsets (row i, column j)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over the K (sequence) dimension --------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: left[bh, i, k]  shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + bh * stride_ah
                 + offs_i[:, None] * stride_ai
                 + offs_k[None, :] * stride_aj,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block (transposed): right[bh, k, j]  shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr + bh * stride_bh
                 + offs_k[:, None] * stride_bi
                 + offs_j[None, :] * stride_bj,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # FP16 dot, accumulate as FP32
        acc += tl.dot(a, b).to(tl.float32)

    # ---- store result -------------------------------------------------
    tl.store(
        c_ptr + bh * stride_ch
              + offs_i[:, None] * stride_ci
              + offs_j[None, :] * stride_cj,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 and fuse the five linear projections
    # ------------------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim] (fp16)

    # half‑precision weights
    w_left_proj   = weights["left_proj.weight"].half()
    w_right_proj  = weights["right_proj.weight"].half()
    w_left_gate   = weights["left_gate.weight"].half()
    w_right_gate  = weights["right_gate.weight"].half()
    w_out_gate    = weights["out_gate.weight"].half()

    # stack along the output dimension: (5*H, dim)
    stacked_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0
    )  # [5*H, dim] (fp16)

    # single fused matmul -> [B, N, N, 5*H]
    fused = torch.nn.functional.linear(x_h, stacked_weight)  # bias=None

    # ------------------------------------------------------------------
    # 3. Split results and apply mask / gates
    # ------------------------------------------------------------------
    H = hidden_dim
    left       = fused[..., 0:H]
    right      = fused[..., H:2*H]
    left_gate  = fused[..., 2*H:3*H].sigmoid()
    right_gate = fused[..., 3*H:4*H].sigmoid()
    out_gate   = fused[..., 4*H:5*H].sigmoid()   # [B, N, N, H]

    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] and cast to fp16
        m = mask.unsqueeze(-1).to(torch.float16)
        left  = left * m
        right = right * m

    # apply element‑wise gates (still fp16)
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Batched N³ reduction  (left @ rightᵀ)
    #    Rearrange to (B*H, N, N) – contiguous for Triton.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, H, N, N) → (B*H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left.view(B * H, N, N)   # fp16
    right_flat = right.view(B * H, N, N)  # fp16

    BH = B * H
    # Heuristic: use Triton for modest batch‑head counts; otherwise delegate
    # to cuBLAS via torch.bmm (highly tuned for large problems).
    use_triton = (BH <= 256) and (N <= 1024)

    if use_triton:
        out_flat = torch.empty_like(left_flat)  # fp16 output buffer

        # strides (in element units)
        stride_a_h, stride_a_i, stride_a_j = left_flat.stride()
        stride_b_h, stride_b_i, stride_b_j = right_flat.stride()
        stride_c_h, stride_c_i, stride_c_j = out_flat.stride()

        # Tuned block sizes for H100 Tensor‑Cores
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )
        triton_outgoing_matmul[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            stride_a_h,
            stride_a_i,
            stride_a_j,
            stride_b_h,
            stride_b_i,
            stride_b_j,
            stride_c_h,
            stride_c_i,
            stride_c_j,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # torch.bmm uses cuBLAS and Tensor‑Cores automatically
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
        out_flat = out_flat.half()

    # ------------------------------------------------------------------
    # 5. Reshape back to (B, N, N, H) and convert to FP32
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out = out.float()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()  # broadcast multiplication

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dimensionality (float32)
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"]  # shape (dim, hidden_dim), float32
    out = torch.nn.functional.linear(out, w_to_out)

    return out
