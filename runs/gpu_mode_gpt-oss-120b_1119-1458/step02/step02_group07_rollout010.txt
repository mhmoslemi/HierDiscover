import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_m0, stride_m1, stride_m2,
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    mask_val = tl.load(mask_ptr + off_m)

    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left = tl.load(left_ptr + off_l + offs * stride_l3,
                       mask=mask_h, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs * stride_r3,
                        mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_g + offs * stride_g3,
                             mask=mask_h, other=tl.float16(0.0))

        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_ptr + off_l + offs * stride_l3, left_out, mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    The implementation fuses:
      * layer‑norm of the input,
      * a single fused linear projection that yields left/right projections and their gates,
      * optional mask‑gate fusion via a Triton kernel,
      * a batched half‑precision GEMM that implements the N³ sum,
      * a second layer‑norm, output‑gate and final linear projection.
    All heavy compute is done in fp16, the final result is returned in fp32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # 1. Input layer‑norm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # 2. Cast to fp16 for all dense work
    x_fp16 = x.half()  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3. Single fused linear that produces left/right projections and gates
    # ------------------------------------------------------------------
    # concatenate weights: left_proj, left_gate, right_proj, right_gate
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [4*hidden_dim, dim]

    # Linear projection
    proj = F.linear(x_fp16, w_concat)  # [B, N, N, 4*hidden_dim]

    B, N, _, _ = proj.shape
    # reshape to separate the four heads
    proj = proj.view(B, N, N, 4, hidden_dim)  # [B, N, N, 4, hidden_dim]

    left = proj[..., 0, :]               # [B, N, N, hidden_dim]
    left_gate = torch.sigmoid(proj[..., 1, :])   # [B, N, N, hidden_dim]
    right = proj[..., 2, :]              # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(proj[..., 3, :])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Apply mask & gates (Triton fusion if a mask is present)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        # ensure contiguous for kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = B * N * N
        BLOCK_H = 32  # works for hidden_dim = 128, 384, 768
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left, right, mask_fp16,
            left_gate, right_gate,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            B, N, hidden_dim,
            BLOCK_H=BLOCK_H,
            num_warps=4,
        )
        # left/right now already contain mask * gate * projection
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ operation as a batched GEMM in fp16
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # BMM: C = A @ Bᵀ   (fp16 → fp32 accumulation inside Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to original layout
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]   fp16

    # ------------------------------------------------------------------
    # 6. Post‑processing: second LayerNorm, out‑gate, final proj
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),

"""
Optimised outgoing Triangle Multiplicative Update (TriMul) – forward pass only.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32).
2️⃣  Fuse the *four* linear maps (left‑proj, left‑gate, right‑proj, right‑gate)
    into a single half‑precision GEMM by concatenating their weight matrices.
3️⃣  Apply the sigmoid non‑linearity to the two gate slices.
4️⃣  If a mask is present, a tiny Triton kernel fuses
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    in‑place, saving two extra memory passes.
   When no mask is supplied the multiplication is performed with a simple
   element‑wise product.
5️⃣  The core N³ reduction Σₖ left[...,i,k,d]·right[...,j,k,d] is realised as a
    batch of independent fp16 GEMMs:
        Cₕ = Aₕ @ Bₕᵀ      (h = batch*hidden_dim)
    using torch.bmm (Tensor‑Core accelerated).
6️⃣  Convert the result back to fp32, apply a second LayerNorm, the output gate,
    and a final linear projection to the original feature dimension.

All heavy arithmetic is done in fp16; the final tensor is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: mask × gate × projection fusion (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    # strides for left/right (B,N,N,hidden)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (B,N,N)
    stride_m0, stride_m1, stride_m2,
    # strides for gates (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,                       # batch, seq_len, hidden_dim
    BLOCK_H: tl.constexpr,        # tile size for the hidden dimension
):
    pid = tl.program_id(0)
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    mask_val = tl.load(mask_ptr + off_m)                     # scalar fp16

    # loop over hidden dimension
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left      = tl.load(left_ptr      + off_l + offs * stride_l3,
                            mask=mask_h, other=tl.float16(0.0))
        right     = tl.load(right_ptr     + off_r + offs * stride_r3,
                            mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate= tl.load(right_gate_ptr+ off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))

        left_out  = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_ptr  + off_l + offs * stride_l3, left_out,  mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor – [B, N, N, dim] (float32)
            mask         – [B, N, N] (float32) or an empty tensor when nomask=True
            weights      – dict containing all model parameters (float32)
            config       – dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the huge linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Single fused linear for left/right projections and their gates
    # ------------------------------------------------------------------
    # concatenate the four weight matrices -> shape [4*hidden_dim, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # unified projection → [B, N, N, 4*hidden_dim]
    proj = F.linear(x_fp16, w_concat)

    B, N, _, _ = proj.shape
    proj = proj.view(B, N, N, 4, hidden_dim)           # [B, N, N, 4, hidden_dim]

    left       = proj[..., 0, :]                      # [B, N, N, hidden_dim]
    left_gate  = torch.sigmoid(proj[..., 1, :])       # [B, N, N, hidden_dim]
    right      = proj[..., 2, :]                      # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(proj[..., 3, :])       # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Mask & gate fusion (Triton) – only when a mask is provided
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()

        # make sure everything is contiguous for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = B * N * N                     # one program per (b,i,j)
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left, right, mask_fp16,
            left_gate, right_gate,
            # strides for left/right tensors
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # strides for mask tensor
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gate tensors (identical layout to left/right)
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            B, N, hidden_dim,
            BLOCK_H=32,               # 32 works for 128/384/768 hidden dims
            num_warps=4,
        )
        # after the kernel, `left` and `right` already contain
        # proj * gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ reduction as a batched fp16 GEMM
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # compute C = A @ Bᵀ (Tensor‑Core fp16 matmul, internal fp32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: to‑out‑norm → out‑gate → final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # output gate (computed on the same x used earlier)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # fp16
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to the original dimensionality
    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B, N, N, dim] float32

    return out_final
