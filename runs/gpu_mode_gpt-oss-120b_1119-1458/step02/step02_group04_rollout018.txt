```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elts: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * sigmoid(gate) * mask   (mask broadcast over hidden dim)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elts

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    pos = offs // HIDDEN_DIM            # index of the (b,i,j) element
    m = tl.load(mask_ptr + pos, mask=valid, other=1.0)
    # sigmoid(gate) using builtin exp
    sig = 1.0 / (1.0 + tl.exp(-gate))
    out = proj * sig * m
    tl.store(out_ptr + offs, out, mask=valid)
    
@triton.jit
def elem_mul_kernel(a_ptr, b_ptr, out_ptr,
                    total_elts: tl.constexpr,
                    BLOCK_SIZE: tl.constexpr):
    """out = a * b (elementwise)"""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (AlphaFold‑3 style) optimized with
    – a single fused linear layer that produces both projections and gates,
    – a Triton kernel that fuses mask, sigmoid(gate) and element‑wise multiply,
    – batched GEMM for the N³ contraction,
    – half‑precision computation everywhere except the final cast to fp32.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 0. cast weights to fp16 on the device (no bias in linear layers)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    norm_w = to_half(weights["norm.weight"])
    norm_b = to_half(weights["norm.bias"])
    # combine all linear weights that take the same input (dim) -> (5*hidden, dim)
    left_proj_w   = to_half(weights["left_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])

    # concatenate in the order used later for easy slicing
    combined_linear_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    )   # shape (5*hidden, dim)

    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (half‑precision)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    ).to(torch.float16).contiguous()   # [B,N,N,dim] half

    # ------------------------------------------------------------------
    # 2. One fused linear that yields all projections & gates
    # ------------------------------------------------------------------
    x_flat = x_norm.view(-1, dim)               # (B*N*N, dim)
    combined = F.linear(x_flat, combined_linear_w)  # (B*N*N, 5*hidden)

    # split views
    H = hidden_dim
    left_proj_raw  = combined[:, 0:H]          # (B*N*N, H)
    left_gate_raw  = combined[:, H:2*H]
    right_proj_raw = combined[:, 2*H:3*H]
    right_gate_raw = combined[:, 3*H:4*H]
    out_gate_raw   = combined[:, 4*H:5*H]

    # reshape back to [B,N,N,H]
    left_proj_raw  = left_proj_raw.view(B, N, N, H).contiguous()
    left_gate_raw  = left_gate_raw.view(B, N, N, H).contiguous()
    right_proj_raw = right_proj_raw.view(B, N, N, H).contiguous()
    right_gate_raw = right_gate_raw.view(B, N, N, H).contiguous()
    out_gate_pre   = out_gate_raw.view(B, N, N, H).contiguous()

    # ------------------------------------------------------------------
    # 3. Apply sigmoid to the gate tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 4. Fuse mask, sigmoid(gate) and elementwise multiplication with a kernel
    # ------------------------------------------------------------------
    if mask is not None:
        mask_half = mask.to(dtype=torch.float16, device=device)  # (B,N,N)
        total = left_proj_raw.numel()   # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        left  = torch.empty_like(left_proj_raw, dtype=torch.float16, device=device)
        right = torch.empty_like(right_proj_raw, dtype=torch.float16, device=device)

        # left
        mask_gate_fuse_kernel[grid](
            left_proj_raw.view(-1).data_ptr(),
            left_gate.view(-1).data_ptr(),
            mask_half.view(-1).data_ptr(),
            left.view(-1).data_ptr(),
            total,
            H,
            BLOCK_SIZE=BLOCK,
        )
        # right
        mask_gate_fuse_kernel[grid](
            right_proj_raw.view(-1).data_ptr(),
            right_gate.view(-1).data_ptr(),
            mask_half.view(-1).data_ptr(),
            right.view(-1).data_ptr(),
            total,
            H,
            BLOCK_SIZE=BLOCK,
        )
    else:
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction – batched GEMM over (i,k) x (j,k)
    # ------------------------------------------------------------------
    # re‑order to (B, H, N, N) then reshape to (B*H, N, N)
    left_view  = left.permute(0, 3, 1, 2).contiguous()
    right_view = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_view.view(B * H, N, N)
    right_flat = right_view.view(B * H, N, N)

    # batched matrix multiply: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension (still fp16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate   # element‑wise (fp16)

    # final projection hidden -> dim
    final = F.linear(gated.view(-1, H), to_out_w)   # (B*N*N, dim)
    final = final.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 8. Return in float32 as required by the spec
    # ------------------------------------------------------------------
    return final.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input (half‑precision for speed).
2️⃣ One fused linear layer produces *all* projections and gate pre‑activations:
   left_proj, left_gate, right_proj, right_gate and out_gate.
3️⃣ A tiny Triton kernel fuses the mask (if present), the sigmoid of the
   gate and the element‑wise multiplication:
        out = proj * sigmoid(gate) * mask
   This eliminates separate mask‑*‑gate kernels and extra memory traffic.
4️⃣ The N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   is carried out as a batched GEMM (cuBLAS/Tensor‑Core) on the reshaped
   tensors (B·H, N, N).
5️⃣ Layer‑norm over the hidden dimension, an output gate (sigmoid) and the
   final hidden‑→‑dim linear projection are performed in half‑precision.
6️⃣ The result is returned as float32.

Only the mask‑gate fusion is written in Triton; the heavyweight batched GEMM
uses PyTorch’s highly‑optimised cuBLAS implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # const half *
    gate_ptr,          # const half *
    mask_ptr,          # const half *
    out_ptr,           # half *
    total_elts: tl.constexpr,   # B·N·N·H
    HIDDEN_DIM: tl.constexpr,   # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * sigmoid(gate) * mask
    (mask is broadcast over the hidden dimension).
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    guard = offs < total_elts

    # load projected value and gate pre‑activation
    proj = tl.load(proj_ptr + offs, mask=guard, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=guard, other=0.0)

    # compute index of the (b,i,j) element for broadcasting the mask
    pos = offs // HIDDEN_DIM
    mask = tl.load(mask_ptr + pos, mask=guard, other=1.0)

    # sigmoid(gate) = 1 / (1 + exp(-gate))
    sig = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * sig * mask
    tl.store(out_ptr + offs, out, mask=guard)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : model parameters as torch.Tensors
        - config_dict  : contains ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]       # hidden dimension
    # ------------------------------------------------------------------
    # Helper for half‑precision conversion (keeps device)
    # ------------------------------------------------------------------
    to_half = lambda t: t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model weights – everything is stored as FP32 → cast to FP16
    # ------------------------------------------------------------------
    norm_w = to_half(weights["norm.weight"])
    norm_b = to_half(weights["norm.bias"])

    # Linear weights that share the *same* input dimension (dim) are stacked
    # to a single matrix of shape (5*H, dim):
    #   [left_proj; left_gate; right_proj; right_gate; out_gate]
    left_proj_w   = to_half(weights["left_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])

    combined_linear_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    )  # shape (5*H, dim)

    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (still in FP16 for speed)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    ).to(torch.float16).contiguous()                 # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 2️⃣ One fused linear that yields all projections & gating logits
    # ------------------------------------------------------------------
    x_flat = x_norm.view(-1, dim)                    # (B·N·N, dim)
    combined = F.linear(x_flat, combined_linear_w)   # (B·N·N, 5*H)

    # ------------------------------------------------------------------
    # 3️⃣ Split the combined result into the five tensors we need
    # ------------------------------------------------------------------
    left_proj_raw  = combined[:, 0:H].view(B, N, N, H).contiguous()
    left_gate_raw  = combined[:, H:2*H].view(B, N, N, H).contiguous()
    right_proj_raw = combined[:, 2*H:3*H].view(B, N, N, H).contiguous()
    right_gate_raw = combined[:, 3*H:4*H].view(B, N, N, H).contiguous()
    out_gate_raw   = combined[:, 4*H:5*H].view(B, N, N, H).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask + sigmoid(gate) + element‑wise multiplication.
    #    When a mask is provided we fuse everything in a Triton kernel
    #    (this also computes the sigmoid, saving a separate kernel launch).
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16, device=device)  # (B, N, N)
        total = left_proj_raw.numel()          # B·N·N·H
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        # allocate output tensors
        left  = torch.empty_like(left_proj_raw, dtype=torch.float16, device=device)
        right = torch.empty_like(right_proj_raw, dtype=torch.float16, device=device)

        # left side
        mask_gate_fuse_kernel[grid](
            left_proj_raw.view(-1).data_ptr(),
            left_gate_raw.view(-1).data_ptr(),
            mask_f.view(-1).data_ptr(),
            left.view(-1).data_ptr(),
            total,
            H,
            BLOCK_SIZE=BLOCK,
        )
        # right side
        mask_gate_fuse_kernel[grid](
            right_proj_raw.view(-1).data_ptr(),
            right_gate_raw.view(-1).data_ptr(),
            mask_f.view(-1).data_ptr(),
            right.view(-1).data_ptr(),
            total,
            H,
            BLOCK_SIZE=BLOCK,
        )
    else:
        # No mask – just multiply by sigmoid(gate)
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction – batched GEMM (left @ rightᵀ) over the K dimension
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) → (B·H, N, N)
    left_view  = left.permute(0, 3, 1, 2).contiguous()
    right_view = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_view.view(B * H, N, N)
    right_flat = right_view.view(B * H, N, N)

    # batched matrix multiplication (Tensor‑Core fp16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B·H, N, N)

    # back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm over the hidden dimension (still fp16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate (sigmoid) and final linear projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)   # (B, N, N, H)
    gated = out_norm * out_gate               # element‑wise (fp16)

    # final projection back to the original channel dimension
    final = F.linear(gated.view(-1, H), to_out_w)      # (B·N·N, dim)
    final = final.view(B, N, N, dim)                   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8️⃣ Cast to fp32 as required by the interface
    # ------------------------------------------------------------------
    return final.to(torch.float32)
