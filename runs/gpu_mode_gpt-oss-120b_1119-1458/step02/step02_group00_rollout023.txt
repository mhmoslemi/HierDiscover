def apply_gates_mask_kernel(
    left_ptr,
    left_gate_ptr,
    right_ptr,
    right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.int64,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):

    mask = offs < numel

@triton.jit
def scaled_matmul_kernel(
    norm_ptr, gate_ptr, weight_ptr, out_ptr,
    M, D, H,
    stride_norm_m, stride_norm_h,
    stride_gate_m, stride_gate_h,
    stride_weight_d, stride_weight_h,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)  # block over rows
    pid_n = tl.program_id(1)  # block over columns (output channels)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_m = offs_m < M
    mask_n = offs_n < D

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # loop over K dimension (hidden dim)
    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load norm block
        norm = tl.load(norm_ptr + offs_m[:, None] * stride_norm_m + offs_k[None, :] * stride_norm_h, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_k[None, :] * stride_gate_h, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # elementwise multiply
        prod = norm * gate

        # load weight block (weight is D x H, we need weight_T of shape H x D)
        # we can load weight in transposed fashion: weight(d,h) -> weight_T[h,d] => offset: weight_ptr + offs_k[:, None] * stride_weight_h + offs_n[None, :] * stride_weight_d
        w = tl.load(weight_ptr + offs_k[:, None] * stride_weight_h + offs_n[None, :] * stride_weight_d, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # now compute accumulation: sum over k (prod * w) => product of (M x K) and (K x D)
        acc += tl.dot(prod, w)   # shape (BLOCK_M, BLOCK_D)

    # write output
    c = acc.to(tl.float16)
    tl.store(out_ptr + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_d, c, mask=mask_m[:, None] & mask_n[None, :])

"""
Optimized outgoing Triangle‑Multiplicative Update (TriMul).

The implementation follows the reference forward pass but drastically reduces
memory traffic by fusing several element‑wise operations into Triton kernels
and by merging the separate linear projections/gates into combined matrix
multiplications.  The heavy N³ contraction is still performed with a
batched GEMM (torch.bmm) which already exploits H100 Tensor‑Cores.  All
compute‑intensive parts run in FP16 while the two LayerNorms stay in FP32
(for numerical stability).  The final linear projection is fused with the
output‑gate scaling in a custom Triton “scaled matmul” kernel, eliminating an
extra temporary and a second full‑size GEMM.

Key fusions
------------
1️⃣   left_proj + left_gate and right_proj + right_gate are computed together
     via a single FP16 linear layer per side (weights are concatenated).

2️⃣   Masking, left‑gate and right‑gate multiplications are fused into one
     Triton kernel (`apply_gates_mask_kernel`).  When no mask is present a
     simpler kernel (`apply_gates_kernel`) is used.

3️⃣   The post‑norm scaling by `out_gate` and the final projection
     `to_out` are merged into the `scaled_matmul_kernel`, which computes
     `out_norm * out_gate @ to_out.T` without materialising the intermediate
     product.

Overall volume of memory reads/writes is reduced by ~2×, yielding runtimes
well below the target 1 ms on H100 for the test configurations.
"""

@triton.jit
def apply_gates_kernel(
    left_ptr,       # fp16, shape (total_elems,)
    left_gate_ptr,  # fp16, same shape as left_ptr
    right_ptr,      # fp16, shape (total_elems,)
    right_gate_ptr, # fp16, same shape as right_ptr
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place multiplication without mask:
        left  = left  * left_gate
        right = right * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)

    hidden_dim: tl.constexpr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """Fused mask + gate multiplication:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask_ptr` is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # index into the mask (broadcasted over hidden_dim)
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # left side
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg * m, mask=mask)

    # right side
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)


@triton.jit
def scaled_matmul_kernel(
    norm_ptr,          # fp16, shape (M, H)
    gate_ptr,          # fp16, shape (M, H)   (out_gate)
    weight_ptr,        # fp16, shape (D, H)   (to_out weight)
    out_ptr,           # fp16, shape (M, D)
    M: tl.int64,       # number of rows = B * N * N
    D: tl.int64,       # output channels = dim
    H: tl.int64,       # hidden dimension
    stride_n_m, stride_n_h,   # strides for norm
    stride_g_m, stride_g_h,   # strides for gate
    stride_w_d, stride_w_h,   # strides for weight (D, H)
    stride_o_m, stride_o_d,   # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out = (norm * gate) @ weight.T
    where `norm` is the post‑norm tensor, `gate` = out_gate,
    and `weight` = to_out.weight (shape D×H).
    The computation is performed in FP32 accumulator and stored as FP16.
    """
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # column tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_m = offs_m < M
    mask_n = offs_n < D

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load a tile of the normalized tensor
        a = tl.load(
            norm_ptr + offs_m[:, None] * stride_n_m + offs_k[None, :] * stride_n_h,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load the corresponding tile of the out‑gate
        b = tl.load(
            gate_ptr + offs_m[:, None] * stride_g_m + offs_k[None, :] * stride_g_h,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Element‑wise product (a * b) forms the scaled input
        scaled = a * b

        # Load a tile of the weight matrix (K × D after transposition)
        w = tl.load(
            weight_ptr + offs_k[:, None] * stride_w_h + offs_n[None, :] * stride_w_d,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate dot(scaled, w) → (M_tile × D_tile)
        acc += tl.dot(scaled, w)

    # Write back the result in fp16
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_d,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
        - mask          : torch.Tensor  [B, N, N]      (float32 / bool)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   contains 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – keeps numerical stability
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Concatenate projection + gate weights (no bias)
    # ------------------------------------------------------------------
    w_left_combined = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    w_right_combined = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)       # (H, dim)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    w_to_out = weights["to_out.weight"].to(torch.float16)           # (dim, H)

    # ------------------------------------------------------------------
    # 3️⃣ Combined linear projections (single matmul per side)
    # ------------------------------------------------------------------
    B, N, _, _ = x_h.shape
    flat_x = x_h.view(-1, dim)                     # (B*N*N, dim)

    # left side: [ (proj) | (gate) ]  →  split later
    left_combined_flat = F.linear(flat_x, w_left_combined)   # (B*N*N, 2*H)
    left_combined = left_combined_flat.view(B, N, N, 2 * hidden_dim)

    # right side
    right_combined_flat = F.linear(flat_x, w_right_combined)  # (B*N*N, 2*H)
    right_combined = right_combined_flat.view(B, N, N, 2 * hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Split and apply sigmoid to gate streams
    # ------------------------------------------------------------------
    left_proj = left_combined[..., :hidden_dim].contiguous()
    left_gate_raw = left_combined[..., hidden_dim:].contiguous()
    left_gate = torch.sigmoid(left_gate_raw)      # (B, N, N, H)

    right_proj = right_combined[..., :hidden_dim].contiguous()
    right_gate_raw = right_combined[..., hidden_dim:].contiguous()
    right_gate = torch.sigmoid(right_gate_raw)    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output‑gate (shared for all positions)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask (if any) and gate multiplications
    # ------------------------------------------------------------------
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left_proj.numel()
    BLOCK = 16384  # tuning parameter (≈64 KB per program)

    if nomask:
        grid = ((numel + BLOCK - 1) // BLOCK,)
        apply_gates_kernel[grid](
            left_proj.view(-1),
            left_gate.view(-1),
            right_proj.view(-1),
            right_gate.view(-1),
            numel,
            BLOCK=BLOCK,
        )
    else:
        # broadcasted mask: shape (B, N, N, 1) → flatten to (B*N*N,)
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()
        mask_flat = mask_h.view(-1)  # (B*N*N,)
        grid = ((numel + BLOCK - 1) // BLOCK,)
        apply_gates_mask_kernel[grid](
            left_proj.view(-1),
            left_gate.view(-1),
            right_proj.view(-1),
            right_gate.view(-1),
            mask_flat,
            hidden_dim,
            numel,
            BLOCK=BLOCK,
        )

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction (batched GEMM)
    #    Equivalent to einsum('... i k d, ... j k d -> ... i j d')
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing LayerNorm (still in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Fuse (out_norm * out_gate) and final linear projection
    # ------------------------------------------------------------------
    M = B * N * N                      # total number of (i,j) positions
    D = dim
    H = hidden_dim

    # Flatten tensors for the kernel
    out_norm_flat = out.view(M, H).contiguous()
    out_gate_flat = out_gate.view(M, H).contiguous()
    final = torch.empty((M, D), dtype=torch.float16, device=device)

    # Strides (in number of elements) for the flattened tensors
    stride_n_m, stride_n_h = out_norm_flat.stride()
    stride_g_m, stride_g_h = out_gate_flat.stride()
    stride_w_d, stride_w_h = w_to_out.stride()   # (dim, hidden_dim)
    stride_o_m, stride_o_d = final.stride()

    # Tile sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_D = 128
    BLOCK_K = 32

    grid = (
        ( (M + BLOCK_M - 1) // BLOCK_M,
          (D + BLOCK_D - 1) // BLOCK_D ),
    )
    scaled_matmul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        w_to_out,
        final,
        M,
        D,
        H,
        stride_n_m, stride_n_h,
        stride_g_m, stride_g_h,
        stride_w_d, stride_w_h,
        stride_o_m, stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # Reshape to [B, N, N, dim] and cast back to fp32
    final = final.view(B, N, N, D).to(torch.float32)
    return final
