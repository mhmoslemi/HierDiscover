def proj_gate_mask_kernel(
    x_ptr,  # input
    mask_ptr,
    w_proj_ptr,
    w_gate_ptr,
    out_ptr,
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_m_b, stride_m_i, stride_m_j,
    stride_w_proj_r, stride_w_proj_c,  # weight dims hidden, dim
    stride_w_gate_r, stride_w_gate_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    B, N, D, H,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # load mask value if needed
    if MASK:
        mask_val = tl.load(mask_ptr + b*stride_m_b + i*stride_m_i + j*stride_m_j)
    else
        mask_val = 1.0

    # load x vector
    x_offset = b*stride_x_b + i*stride_x_i + j*stride_x_j
    # We'll accumulate for hidden groups
    for h_start in range(0, H, BLOCK_H):
        h_off = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_off < H

        # compute projection: dot(x, w_proj[h,:])
        # We'll compute dot across dim in chunks sized BLOCK_D
        sum_proj = tl.zeros([BLOCK_H], dtype=tl.float32)
        sum_gate = tl.zeros([BLOCK_H], dtype=tl.float32)
        for d_start in range(0, D, BLOCK_D):
            d_off = d_start + tl.arange(0, BLOCK_D)
            mask_d = d_off < D
            x_vec = tl.load(x_ptr + x_offset + d_off*stride_x_c, mask=mask_d, other=0.0).to(tl.float32)
            w_proj_vec = tl.load(w_proj_ptr + h_off*stride_w_proj_r + d_off*stride_w_proj_c,
                                 mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            w_gate_vec = tl.load(w_gate_ptr + h_off*stride_w_gate_r + d_off*stride_w_gate_c,
                                 mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            # compute dot using matmul?
            # we can compute tl.dot(x_vec, w_proj_vec) for each hidden? Use broadcast: tl.dot across last dimension but here we need 2D multiply: hidden x dim times dim vector.
            sum_proj += tl.dot(w_proj_vec, x_vec)      # shape BLOCK_H
            sum_gate += tl.dot(w_gate_vec, x_vec)

        proj_fp16 = sum_proj.to(tl.float16)

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated.

Algorithm (all heavy work stays in fp16):
1️⃣ LayerNorm on the input tensor (float32).
2️⃣ Concatenate the projection‑and‑gate weight matrices for the *left* and
   *right* branches (shape (2·hidden_dim, dim)).  A single `F.linear`
   per side therefore yields both the projection and the pre‑sigmoid gate.
3️⃣ Apply `sigmoid` to the gate part.
4️⃣ If a mask is supplied, launch a tiny Triton kernel that fuses
   `proj * gate * mask` (in‑place) for the left and right tensors.
   When no mask is present the multiplication is performed with a simple
   element‑wise Python expression.
5️⃣ Compute the N³ reduction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM (`torch.bmm`).  Each hidden dimension is treated as
   an independent batch entry, i.e. shape (B·hidden, N, N) × (B·hidden, N, N)ᵀ.
6️⃣ Second LayerNorm, element‑wise multiplication with the output‑gate,
   and a final linear projection back to `dim`.
   The final tensor is returned in float32.

Only the mask‑gate fusion is implemented in Triton, satisfying the
requirement that “part of the operations” be a Triton kernel while the
rest stays in highly‑optimized PyTorch calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into
# the projected tensors (in‑place).
# ------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    # 4‑D tensors (B,N,N,H) in fp16
    left_ptr,                # fp16 * left projection
    right_ptr,               # fp16 * right projection
    mask_ptr,                # fp16 * mask (B,N,N)
    left_gate_ptr,           # fp16 * left gate (sigmoid output)
    right_gate_ptr,          # fp16 * right gate (sigmoid output)

    # strides for left/right (B,N,N,H)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,

    # strides for mask (B,N,N)
    stride_m0, stride_m1, stride_m2,

    # strides for gates (same layout as left/right)
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,

    # sizes
    B, N, H,                 # batch‑size, sequence‑length, hidden‑dim
    BLOCK_H: tl.constexpr,   # tile size on the hidden dimension
):
    pid = tl.program_id(0)                         # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # mask value for this (b,i,j) – broadcast over H
    mask_val = tl.load(mask_ptr + off_m)            # fp16 scalar

    # loop over hidden dimension in blocks
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        # load left/right projections and gates
        left   = tl.load(left_ptr   + off_l   + offs * stride_l3,
                         mask=mask_h, other=tl.float16(0.0))
        right  = tl.load(right_ptr  + off_r   + offs * stride_r3,
                         mask=mask_h, other=tl.float16(0.0))
        left_g = tl.load(left_gate_ptr  + off_lg + offs * stride_lg3,
                         mask=mask_h, other=tl.float16(0.0))
        right_g = tl.load(right_gate_ptr + off_rg + offs * stride_rg3,
                         mask=mask_h, other=tl.float16(0.0))

        # fused operation: proj * gate * mask
        left_out  = left * left_g * mask_val
        right_out = right * right_g * mask_val

        # store back (in‑place)
        tl.store(left_ptr  + off_l   + offs * stride_l3, left_out,
                 mask=mask_h)
        tl.store(right_ptr + off_r   + offs * stride_r3, right_out,
                 mask=mask_h)


# ------------------------------------------------------------
# Triton‑augmented forward pass of the outgoing TriMul.
# ------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (float32) – optional
            - weights:     dict of model parameters (all float32)
            - config:      dict with keys
                * "dim" (int) – input feature dimension
                * "hidden_dim" (int) – inner projection dimension
                * "nomask" (bool, optional) – True ⇒ mask is ignored

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection+gate linear layers (2 calls instead of 4)
    #    left: [proj | gate]   right: [proj | gate]
    # ------------------------------------------------------------------
    # concatenate left‑proj and left‑gate weights
    left_concat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # shape (2*hidden, dim)

    right_concat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # shape (2*hidden, dim)

    # linear → [B,N,N,2*hidden]  (still fp16)
    left_concat = F.linear(x_fp16, left_concat_weight)   # proj + gate
    right_concat = F.linear(x_fp16, right_concat_weight)

    # split projection and gate parts
    left_proj = left_concat[..., :hidden_dim]            # [B,N,N,hidden]
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])   # fp16

    right_proj = right_concat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:]) # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) and fuse projection‑gate multiplication
    # ------------------------------------------------------------------
    if nomask:
        # simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask is present – fuse with a tiny Triton kernel
        mask_fp16 = mask.to(torch.float16).contiguous()

        # ensure all tensors are contiguous for correct stride semantics
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]  # B * N * N
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            # tensor pointers
            left_proj, right_proj,
            mask_fp16,
            left_gate, right_gate,
            # strides for left/right
            left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
            right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gates
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            # sizes
            input_tensor.shape[0],                # B
            input_tensor.shape[1],                # N
            hidden_dim,                           # H
            BLOCK_H=32,                           # hidden‑tile size (covers up to 384)
            num_warps=4,
        )
        # after the kernel, `left_proj` and `right_proj` already contain
        # mask * gate * projection
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ contraction: batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape from (B,N,N,H) → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # GEMM: out_mat[b*h, i, j] = Σₖ left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B,N,N,hidden_dim] (still fp16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: second LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # output gate (computed from the *original* normalized input)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))   # fp16 → broadcast later
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])  # [B,N,N,dim] float32

    return out_final
