def apply_mask_kernel(input_ptr, mask_ptr, output_ptr, hidden_dim: tl.constexpr, n_elements: tl.constexpr, BLOCK: tl.constexpr):

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣ Layer‑norm over the feature dimension (FP32 for numeric stability).
2️⃣ Cast the normalised tensor to FP16 and flatten it to [B·N·N, dim].
3️⃣ Fuse the two linear projections (left/right) into a single matmul:
   *proj* = X @ [W_left_proj ; W_right_proj]ᵀ → shape (..., 2·hidden).
   Slice the result into *left_proj* and *right_proj*.
4️⃣ Fuse the three gating linear layers into a second matmul:
   *gate* = X @ [W_left_gate ; W_right_gate ; W_out_gate]ᵀ → shape (..., 3·hidden).
   Apply sigmoid and split to *left_gate*, *right_gate*, *out_gate*.
5️⃣ Apply element‑wise product of projection, gate and optional mask.
   This step is performed by a tiny Triton kernel to avoid several
   memory passes.
6️⃣ Core O(N³) operation: for each hidden channel compute a matrix‑multiply
   C = A @ Aᵀ, i.e. a batched GEMM of shape (B·hidden, N, N) using
   `torch.bmm` (FP16×FP16 → FP32 accumulation on H100 Tensor‑Cores).
7️⃣ Hidden‑dim LayerNorm → multiply by *out_gate* → final linear projection
   back to the original feature dimension.
All heavy arithmetic is in FP16, final output is FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply projection × gate × mask (if present)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,               # fp16  projection (left or right)
    gate_ptr,               # fp16  corresponding gate
    mask_ptr,               # fp16  flattened mask (B·N·N)
    out_ptr,                # fp16  destination
    hidden_dim,             # i64   hidden dimension (used for mask broadcasting)
    n_elements,             # i64   total number of elements in proj/gate (= B·N·N·hidden)
    BLOCK: tl.constexpr,    # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load projection and gate (promote to fp32 for higher‑precision mul)
    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # Each mask entry repeats `hidden_dim` times along the hidden axis
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * m                     # fp32 intermediate
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Args:
        data: Tuple containing
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights      : dict of model weight tensors
            - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy linear work and flatten
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)
    B, N, _, _ = x_h.shape
    flat = x_h.view(-1, dim)                     # [B·N·N, dim]  (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Fused projection (left & right) -> single matmul
    # --------------------------------------------------------------
    # weights are stored as (out_features, in_features); concat along out_features
    proj_w = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                           # (2·hidden, dim)
    proj = F.linear(flat, proj_w)                  # (B·N·N, 2·hidden)

    left_proj_raw  = proj[:, :hidden]              # (B·N·N, hidden)
    right_proj_raw = proj[:, hidden:]              # (B·N·N, hidden)

    # --------------------------------------------------------------
    # 4️⃣ Fused gating (left_gate, right_gate, out_gate) -> single matmul
    # --------------------------------------------------------------
    gate_w = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # (3·hidden, dim)
    gate = F.linear(flat, gate_w)                  # (B·N·N, 3·hidden)

    left_gate_raw  = torch.sigmoid(gate[:, :hidden]).to(torch.float16)
    right_gate_raw = torch.sigmoid(gate[:, hidden : 2 * hidden]).to(torch.float16)
    out_gate       = torch.sigmoid(gate[:, 2 * hidden : 3 * hidden]).to(torch.float32)

    # --------------------------------------------------------------
    # 5️⃣ Fuse (proj × gate) and optional mask using Triton
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B·N·N] (fp16, broadcast over hidden)
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)               # (B·N·N)

        # Allocate output tensors
        left  = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        # Triton launch configuration
        BLOCK = 65536
        n_elem = left_proj_raw.numel()            # B·N·N·hidden
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj_raw * left_gate_raw * mask
        _gate_mask_kernel[grid](
            left_proj_raw,
            left_gate_raw,
            mask_flat,
            left,
            hidden,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj_raw * right_gate_raw * mask
        _gate_mask_kernel[grid](
            right_proj_raw,
            right_gate_raw,
            mask_flat,
            right,
            hidden,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask: simple element‑wise multiplication (still FP16)
        left  = left_proj_raw * left_gate_raw
        right = right_proj_raw * right_gate_raw

    # Reshape back to the 4‑D layout
    left  = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 6️⃣ Core O(N³) operation: batched GEMM (left @ rightᵀ) per hidden
    # --------------------------------------------------------------
    # Bring hidden to the batch dimension: [B·hidden, N, N]
    left_b  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # FP16×FP16 → FP32 accumulation on H100 Tensor‑Cores
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))   # (B·hidden, N, N)  FP16

    # Restore the original layout and promote to FP32 for the next layers
    out = (
        out_b.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )                                                   # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm, out‑gate, final projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                   # FP32

    # broadcast out_gate over the hidden dimension and multiply
    out = out * out_gate.view(B, N, N, hidden)

    # Final linear projection (hidden → dim)
    out = F.linear(out, weights["to_out.weight"])       # FP32, shape [B, N, N, dim]

    return out
```
