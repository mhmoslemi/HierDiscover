"""
TriMul (outgoing) forward pass – heavily fused implementation.

Key ideas
---------
1️⃣  LayerNorm on the input (float‑32 for numeric stability).
2️⃣  Cast the normalized tensor to FP16 and compute the four linear
    projections/gates in **only three** GEMM calls:
       • left_proj + left_gate   → 2·hidden columns
       • right_proj + right_gate → 2·hidden columns
       • out_gate                → hidden columns
   This halves the number of times we have to read the huge
   (B·N·N·dim) tensor from global memory.
3️⃣  A tiny Triton kernel fuses *mask* (if present), *gate* and
    *projection* element‑wise multiplication, avoiding several
    memory passes.
4️⃣  The core O(N³) update is a batch of independent matrix
    multiplies `C = A @ Bᵀ` where A,B ∈ ℝ^{B·H × N × N}.  The
    tensors are reshaped to (B·H,N,N) and evaluated with
    `torch.bmm`, which on an H100 executes a mixed‑precision
    Tensor‑Core GEMM (FP16×FP16 → FP32) at peak throughput.
5️⃣  Post‑processing: LayerNorm over the hidden dimension,
    element‑wise multiplication with the out‑gate and a final
    linear projection back to the original feature dimension.
   The final result is returned as float‑32.

Only the mask–gate fusion is performed in Triton (as required);
all other heavy arithmetic stays in PyTorch to benefit from the
highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply proj × gate × mask (if mask exists)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    proj_ptr,          # fp16*  projection (left or right)
    gate_ptr,          # fp16*  corresponding gate (after sigmoid)
    mask_ptr,          # fp16*  mask (flattened, shape B·N·N)
    out_ptr,           # fp16*  destination tensor
    hidden_dim,        # i64   number of hidden channels (repeated for mask)
    n_elements,        # i64   total #elements = B·N·N·hidden
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load projection and gate (convert to fp32 for higher‑precision mul)
    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # Mask broadcasting: each mask entry repeats `hidden_dim` times.
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    # Fused element‑wise product
    out = proj * gate * mask_val

    # Store back as fp16
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point – the custom kernel expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            • input_tensor – [B, N, N, dim] float32
            • mask         – [B, N, N]  (float32/float16) or None
            • weights: dict with the following keys (all float32):
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight',
                'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            • config: dict with at least
                'dim' (int) and 'hidden_dim' (int).
                May also contain 'nomask' (bool).

    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – all heavy linear algebra will be in FP16
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right projections with their gates (single GEMM each)
    # ------------------------------------------------------------------
    # Concatenate the projection & gate weight matrices for the left side
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # shape (2*hidden, dim)

    # Same for the right side
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # shape (2*hidden, dim)

    # Compute both projection and gate in one go
    left_big = F.linear(x_h, left_concat_w)   # [B, N, N, 2*hidden]
    right_big = F.linear(x_h, right_concat_w) # [B, N, N, 2*hidden]

    # Split into projection & gate tensors
    left_proj = left_big[..., :hidden]                      # fp16
    left_gate = torch.sigmoid(left_big[..., hidden:])       # fp16

    right_proj = right_big[..., :hidden]                    # fp16
    right_gate = torch.sigmoid(right_big[..., hidden:])     # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate (separate linear)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )  # [B, N, N, hidden]  fp16
    # Keep out_gate in fp32 for later fp32 LayerNorm (convert later)
    out_gate_fp32 = out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask (if any) and fuse mask * gate * proj with a Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16 → flattened
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)  # shape = B·N·N

        # Output tensors that will hold the gated+masked values
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Triton launch configuration
        BLOCK = 65536  # processing granularity
        n_elem = left_proj.numel()  # = B·N·N·hidden
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _apply_mask_gate_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _apply_mask_gate_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask – simple element‑wise product
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³) step: batch GEMM (A @ Bᵀ)
    #    Shapes after permutation: (B·hidden, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B·hidden, N, N) without an extra copy of the data
    # (permute changes stride; view flattens the batch dimensions)
    left_b = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_b = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # Batched matrix multiplication – Tensor‑Core GEMM (FP16×FP16→FP32)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))  # [B*hidden, N, N] fp32

    # Reshape back to [B, N, N, hidden] (still fp32)
    out = out_b.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,hidden]  fp32

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate (broadcast over hidden) and final projection
    # ------------------------------------------------------------------
    out = out * out_gate_fp32  # element‑wise product, fp32

    # Final linear projection back to original dim (float32)
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim] (float32)

    return out
