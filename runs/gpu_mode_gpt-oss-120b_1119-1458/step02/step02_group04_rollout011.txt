import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, mean_ptr, rstd_ptr,
    stride_xb, stride_xn, stride_xn2, stride_xc,
    stride_ob, stride_on, stride_on2, stride_oc,
    N, C, eps,
    BLOCK_X: tl.constexpr, BLOCK_C: tl.constexpr):
    # x shape: (B, N, N, C)
    pid = tl.program_id(axis=0)
    # compute block start positions
    b = pid // (tl.cdiv(N, BLOCK_X) * tl.cdiv(N, BLOCK_X))
    rem = pid % (tl.cdiv(N, BLOCK_X) * tl.cdiv(N, BLOCK_X))
    i = (rem // tl.cdiv(N, BLOCK_X)) * BLOCK_X
    j = (rem % tl.cdiv(N, BLOCK_X)) * BLOCK_X

    # compute offsets for the C dimension: we iterate over C dimension in inner loop
    c_offs = tl.arange(0, BLOCK_C)
    mask_c = c_offs < C

    # compute base pointer offsets for x
    base_x = x_ptr + b * stride_xb + i * stride_xn + j * stride_xn2
    # iterate over C in blocks
    # Load a TILE of C values for each (b,i,j)
    x = tl.load(base_x + c_offs * stride_xc, mask=mask_c, other=0.0)  # shape (BLOCK_C)

    # treat each (b,i,j) as independent; need to reduce across C (i.e., compute mean & var)
    # But we want to compute normalization across C dimension for each position. We'll produce
    # out for each C element after computing mean and rstd.
    # Instead of computing per-block, we need to compute sum and sumsq across C dimension.
    # We can do a reduction per element (i.e., each (b,i,j) has a reduction across C).
    # So need an additional loop across C blocks. Use a for loop.
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)
    
    # Loop over C dimension
    for start_c in range(0, C, BLOCK_C):
        offs_c = start_c + c_offs
        mask = offs_c < C
        xb = tl.load(x_ptr + b*stride_xb + i*stride_xn + j*stride_xn2 + offs_c*stride_xc, mask=mask, other=0.0)
        xb_f32 = xb.to(tl.float32)
        sum_val += tl.sum(xb_f32, axis=0)
        sum_sq += tl.sum(xb_f32 * xb_f32, axis=0)

    # compute mean & rstd
    # number of elements = C
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Store mean and rstd if needed (optional)
    # Emit normalized values
    for start_c in range(0, C, BLOCK_C):
        offs_c = start_c + c_offs
        mask = offs_c < C
        xb = tl.load(x_ptr + b*stride_xb + i*stride_xn + j*stride_xn2 + offs_c*stride_xc, mask=mask, other=0.0)
        # normalize
        out = (xb.to(tl.float32) - mean) * rstd
        # store as half
        tl.store(out_ptr + b*stride_ob + i*stride_on + j*stride_on2 + offs_c*stride_oc,
                 out.to(tl.float16), mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_three_kernel(a_ptr, b_ptr, c_ptr, out_ptr,
                      total_elts: tl.constexpr,
                      BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)

@triton.jit
def mul_two_kernel(a_ptr, b_ptr, out_ptr,
                   total_elts: tl.constexpr,
                   BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Implements the TriMul operation using fused element‑wise Triton kernels
    for mask/gate application, while keeping the N³ contraction as a batched
    GEMM (torch.bmm) which is highly optimized on H100 tensor cores.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper for half precision weights
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # shape [B,N,N,dim] float32
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – half precision
    # ------------------------------------------------------------------
    left_proj = F.linear(x_half, left_proj_w)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_half, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_half, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask handling – broadcast across hidden_dim
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] -> expand to hidden_dim
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        mask_f_exp = mask_f.expand(-1, -1, -1, hidden_dim).contiguous()
    else:
        # all‑ones mask – no allocation for the backward pass
        mask_f_exp = torch.ones((B, N, N, hidden_dim),
                                dtype=torch.float16,
                                device=device)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse (proj * gate * mask) with a Triton kernel
    # ------------------------------------------------------------------
    total = left_proj.numel()
    BLOCK_SIZE = 65536  # 64k elements per program
    grid = (triton.cdiv(total, BLOCK_SIZE),)

    # left: out = left_proj * left_gate * mask
    left_out = torch.empty_like(left_proj)
    fuse_three_kernel[grid](
        left_proj.view(-1), left_gate.view(-1), mask_f_exp.view(-1),
        left_out.view(-1),
        total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # right: out = right_proj * right_gate * mask
    right_out = torch.empty_like(right_proj)
    fuse_three_kernel[grid](
        right_proj.view(-1), right_gate.view(-1), mask_f_exp.view(-1),
        right_out.view(-1),
        total,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction – batched GEMM via torch.bmm
    # ------------------------------------------------------------------
    # reshape to (B * hidden_dim, N, N)
    left_view = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)

    # ------------------------------------------------------------------

    )

    # 9️⃣ Return float32 tensor as required
    # ------------------------------------------------------------------

"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (float32) – identical to the reference nn.LayerNorm.
2️⃣ Cast to fp16 for the heavy‑lifting.
3️⃣ Fuse the two linear projections (proj & gate) for *left* and *right* by
   concatenating the weight matrices and performing a single torch.nn.functional
   linear.  The gate logits are passed through a sigmoid.
4️⃣ Apply the optional pair‑wise mask together with the gate by a custom
   Triton kernel (proj * gate * mask).  When no mask is provided we simply
   multiply proj and gate in PyTorch.
5️⃣ Compute the N³ contraction as a batched GEMM (left @ rightᵀ) – this uses
   the highly‑optimized cuBLAS tensor‑core kernels on H100.
6️⃣ Layer‑norm over the hidden dimension of the contraction result.
7️⃣ Compute the output gate (sigmoid(linear(x))) and fuse the gate with the
   normalized output via another Triton kernel (element‑wise multiplication).
8️⃣ Final linear projection back to the original channel dimension.
9️⃣ Return the result as float32.

Only the element‑wise mask‑gate multiplications are performed in Triton;
the expensive matrix‑multiply is left to cuBLAS which is already optimal
on the target hardware.  The use of concatenated weight matrices reduces
the number of separate GEMM launches, cutting launch overhead and memory
traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def fuse_three_kernel(
    a_ptr,  # left_proj or right_proj   (half *)
    b_ptr,  # gate (sigmoid)            (half *)
    c_ptr,  # mask (broadcasted)        (half *)
    out_ptr,  # result                    (half *)
    total_elts: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * b * c   (flat view)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def mul_two_kernel(
    a_ptr,  # tensor A (half *)
    b_ptr,  # tensor B (half *)
    out_ptr,  # result (half *)
    total_elts: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * b   (flat view)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (torch.Tensor)
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to cast weights to fp16 on the correct device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ Load model weights (half‑precision where possible)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])   # (H, dim)
    right_proj_w  = to_half(weights["right_proj.weight"])  # (H, dim)
    left_gate_w   = to_half(weights["left_gate.weight"])   # (H, dim)
    right_gate_w  = to_half(weights["right_gate.weight"])  # (H, dim)
    out_gate_w    = to_half(weights["out_gate.weight"])    # (H, dim)
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim] (float32)
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for *left* and *right* by concatenating
    #    the weight matrices (single GEMM each)
    # ------------------------------------------------------------------
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    left_combined = F.linear(x_half, left_combined_w)   # [B, N, N, 2*H]
    right_combined = F.linear(x_half, right_combined_w)  # [B, N, N, 2*H]

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    left_gate = torch.sigmoid(left_gate_pre)   # (B,N,N,H) half
    right_gate = torch.sigmoid(right_gate_pre)  # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – broadcast across the hidden dimension
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] -> [B,N,N,1] -> expand to hidden_dim
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)          # (B,N,N,1)
        mask_f_exp = mask_f.expand(-1, -1, -1, hidden_dim).contiguous()
        use_mask = True
    else:
        use_mask = False

    # ------------------------------------------------------------------
    # 5️⃣ Apply (proj * gate * mask) with a Triton kernel.
    #    When there is no mask we fall back to a simple pointwise multiply.
    # ------------------------------------------------------------------
    if use_mask:
        BLOCK_SIZE = 65536
        total_elts = left_proj.numel()                     # B*N*N*H
        grid = (triton.cdiv(total_elts, BLOCK_SIZE),)

        left = torch.empty_like(left_proj)
        fuse_three_kernel[grid](
            left_proj.view(-1), left_gate.view(-1), mask_f_exp.view(-1),
            left.view(-1),
            total_elts,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        right = torch.empty_like(right_proj)
        fuse_three_kernel[grid](
            right_proj.view(-1), right_gate.view(-1), mask_f_exp.view(-1),
            right.view(-1),
            total_elts,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask – just element‑wise multiply proj and gate
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction – batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten the batch*head dimension.
    left_view = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat = left_view.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right_view.view(B * hidden_dim, N, N)  # (B*H, N, N)

    # Batched matrix multiply – uses cuBLAS tensor‑core kernels.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape [B, N, N, H] (still fp16)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension (float32 internals)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 8️⃣ Output gate and final projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))   # (B,N,N,H) half

    # Fuse out_norm * out_gate with a Triton kernel
    BLOCK_SIZE = 65536
    total_out = out_norm.numel()
    grid_out = (triton.cdiv(total_out, BLOCK_SIZE),)
    gated = torch.empty_like(out_norm)
    mul_two_kernel[grid_out](
        out_norm.view(-1), out_gate.view(-1), gated.view(-1),
        total_out,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Final linear projection back to the original dim
    final = F.linear(gated, to_out_w)      # (B,N,N,dim) half

    # ------------------------------------------------------------------
    # 9️⃣ Return in float32 as required by the API
    # ------------------------------------------------------------------
    return final.to(torch.float32)
```
