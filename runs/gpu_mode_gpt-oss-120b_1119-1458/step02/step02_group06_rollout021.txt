def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm over last dim
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half for remaining ops
    x_h = x.half()
    B, N, _, _ = x_h.shape
    # Flatten to shape (B*N*N, dim)
    x_flat = x_h.view(-1, dim)  # (M, dim)

    # 2. Combine all linear weights (5 * hidden_dim total)
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Cast & concatenate
    w_all = torch.cat([
        w_left_proj,
        w_left_gate,
        w_right_proj,
        w_right_gate,
        w_out_gate
    ], dim=0).half()  # shape (5*H, dim)

    # 3. Single linear projection
    combined = torch.nn.functional.linear(x_flat, w_all)  # (M, 5*H) half

    # 4. Slice and reshape
    offset = 0
    left_proj = combined[:, offset: offset+hidden_dim].view(B, N, N, hidden_dim)
    offset += hidden_dim
    left_gate = combined[:, offset: offset+hidden_dim].view(B, N, N, hidden_dim)
    offset += hidden_dim
    right_proj = combined[:, offset: offset+hidden_dim].view(B, N, N, hidden_dim)
    offset += hidden_dim
    right_gate = combined[:, offset: offset+hidden_dim].view(B, N, N, hidden_dim)
    offset += hidden_dim
    out_gate = combined[:, offset: offset+hidden_dim].view(B, N, N, hidden_dim)

    # 5. Apply mask (if any) and gates
    # mask shape: [B,N,N]; broadcast to hidden dim
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)  # (B,N,N,1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left_proj * left_gate  # (B,N,N,H) half
    right = right_proj * right_gate

    # 6. Compute the N^3 reduction (batched GEMM)
    # Rearrange to (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N).transpose(1, 2).contiguous()

    BH = B * hidden_dim
    # Heuristic to select Triton kernel
    use_triton = (BH <= 64) and (N <= 512)
    if use_triton:
        # allocate output
        out_flat = torch.empty_like(left_flat)
        # strides
        stride_a_m, stride_a_n, stride_a_k = left_flat.stride()
        stride_b_m, stride_b_n, stride_b_k = right_flat.stride()
        stride_c_m, stride_c_n, stride_c_k = out_flat.stride()
        # Block sizes tuning
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
        grid = (BH,
                (N + BLOCK_M - 1)//BLOCK_M,
                (N + BLOCK_N - 1)//BLOCK_N)
        batched_gemm_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,  # M = N (square)
            N,
            stride_a_m,
            stride_a_n,
            stride_b_m,
            stride_b_n,
            stride_b_k,
            stride_c_m,
            stride_c_n,
            stride_c_k,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        out_flat = torch.bmm(left_flat, right_flat)

    # 7. Reshape back to (B,N,N,H) and cast to float32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()  # FP32

    # 8. Output LayerNorm + out_gate gating
    out = torch.nn.functional.layer_norm(

        weight=weights["to_out_norm.weight"],

    )
    out = out * out_gate.float()

    # 9. Final projection to original dim
    w_to_out = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, w_to_out)

"""
Highly‑optimized “outgoing” TriMul kernel.

Algorithm
---------
1. Layer‑norm the input (FP32).
2. Cast to FP16 and flatten the [B,N,N,D] tensor → (B·N·N, D).
3. Fuse *all* linear projections (left/right projections, their gates,
   and the output gate) into a *single* matmul by concatenating the
   weight matrices along the output‐feature axis.  This reduces the
   number of kernel launches and memory traffic from five matmuls to
   one.
4. Slice the result back into the five tensors, reshape to [B,N,N,H],
   apply the mask (if any) and the sigmoid gates (still FP16).
5. Compute the costly N³ reduction
        out[b,i,j] = Σ_k left[b,i,k]·right[b,j,k]
   as a batched GEMM.  For small problems (BH≤64 and N≤512) a custom
   Triton kernel is used; otherwise we fall back to `torch.bmm`,
   which maps to cuBLAS’s Tensor‑Core path.
6. Reshape the result to [B,N,N,H] (FP32), apply the output
   LayerNorm, the output gate, and the final linear projection back to
   the original feature dimension.
7. Return a float32 tensor of shape [B,N,N,D].

The implementation follows the reference TriMul forward pass while
keeping the heavy linear work in a single fused matmul and the
N³ gram‑product on Tensor‑Cores, achieving the required speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for the N³ reduction.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,                      # square dimensions (both = seq_len)
    stride_am, stride_ak,      # A: (batch, M, K)   K == N, strides for batch & row
    stride_bm, stride_bn, stride_bk,   # B: (batch, K, N) strides for batch, col, row
    stride_cm, stride_cn, stride_ck,   # C: (batch, M, N) strides for batch & row
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)          # batch * hidden
    tile_i   = tl.program_id(1)          # row tile
    tile_j   = tl.program_id(2)          # col tile

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32 (Tensor‑Core promotion)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K (=seq_len)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a  BLOCK_M × BLOCK_K  tile from A
        a = tl.load(
            a_ptr + batch_id * stride_am
                 + offs_i[:, None] * stride_ak
                 + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load a  BLOCK_K × BLOCK_N  tile from B
        b = tl.load(
            b_ptr + batch_id * stride_bm
                 + offs_k[:, None] * stride_bk
                 + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # Tensor‑Core dot: FP16 × FP16 → FP32
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write the result (FP16) back to C
    # ------------------------------------------------------------------
    tl.store(
        c_ptr + batch_id * stride_cm
              + offs_i[:, None] * stride_cn
              + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # [B, N, N, D] – FP32

    # --------------------------------------------------------------
    # 2. Cast to FP16 and flatten the spatial dimensions
    # --------------------------------------------------------------
    x_h   = x.half()                         # FP16
    B, N, _, _ = x_h.shape
    M = B * N * N                             # flattened batch‑spatial size
    x_flat = x_h.view(M, dim)                 # (M, D) – contiguous

    # --------------------------------------------------------------
    # 3. Fuse all linear projections (left/right + three gates)
    #    into a single matmul to cut launches & memory traffic.
    # --------------------------------------------------------------
    # concatenate weights along the output‑feature dimension
    w_all = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).half()                                 # (5*H, D)

    # single fused linear
    comb = F.linear(x_flat, w_all)           # (M, 5*H) – FP16

    # --------------------------------------------------------------
    # 4. Slice & reshape the five tensors
    # --------------------------------------------------------------
    off = 0
    left_proj = comb[:, off:off + hidden_dim].view(B, N, N, hidden_dim)
    off += hidden_dim
    left_gate = comb[:, off:off + hidden_dim].view(B, N, N, hidden_dim)
    off += hidden_dim
    right_proj = comb[:, off:off + hidden_dim].view(B, N, N, hidden_dim)
    off += hidden_dim
    right_gate = comb[:, off:off + hidden_dim].view(B, N, N, hidden_dim)
    off += hidden_dim
    out_gate = comb[:, off:off + hidden_dim].view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5. Apply optional mask and the sigmoid gates (still FP16)
    # --------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)   # (B,N,N,1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    left_gate  = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate   = out_gate.sigmoid()

    left  = left_proj * left_gate
    right = right_proj * right_gate                     # (B,N,N,H) – FP16

    # --------------------------------------------------------------
    # 6. N³ reduction: batched GEMM L @ Rᵀ   (B*H, N, N) × (B*H, N, N)
    # --------------------------------------------------------------
    # move the hidden dimension into the batch axis
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left.view(B * hidden_dim, N, N)               # (BH, N, N)
    # transpose the second matrix so that we compute L @ Rᵀ
    right_flat = right.view(B * hidden_dim, N, N).transpose(1, 2).contiguous()

    BH = B * hidden_dim
    use_triton = (BH <= 64) and (N <= 512)   # heuristic: small batch * large N → Triton

    if use_triton:
        out_flat = torch.empty_like(left_flat)   # FP16 buffer

        # strides for the three tensors
        stride_a_m, stride_a_n, stride_a_k = left_flat.stride()
        stride_b_m, stride_b_n, stride_b_k = right_flat.stride()
        stride_c_m, stride_c_n, stride_c_k = out_flat.stride()

        # tuning parameters for H100 (good Tensor‑Core occupancy)
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )
        batched_gemm_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,                     # M
            N,                     # N
            stride_a_m,
            stride_a_n,
            stride_b_m,
            stride_b_n,
            stride_b_k,
            stride_c_m,
            stride_c_n,
            stride_c_k,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # cuBLAS batched GEMM (Tensor‑Core path)
        out_flat = torch.bmm(left_flat, right_flat)   # (BH, N, N) – FP16

    # --------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H) and cast to FP32
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()                                 # (B, N, N, H) – FP32

    # --------------------------------------------------------------
    # 8. Output LayerNorm + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()                      # broadcasted multiply

    # --------------------------------------------------------------
    # 9. Final linear projection back to the original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])     # (B, N, N, D) – FP32

    return out
