                            stride_br, stride_bj, stride_bk_r, stride_brh,  # right strides (maybe same)

def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    B,
    N,
    hidden,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    b = bh // hidden
    h = bh % hidden

    # compute base offset
    left_base = left_ptr + b * stride_l_batch + h * 1  # hidden stride = 1
    right_base = right_ptr + b * stride_r_batch + h * 1
    out_base = out_ptr + b * stride_o_batch + h * 1

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The kernel performs the following steps

1️⃣  Input LayerNorm (torch, half‑precision after LayerNorm).
2️⃣  All five linear layers (left_proj, right_proj, left_gate,
    right_gate, out_gate) are fused into a single matrix‑multiply by
    concatenating the weight matrices.  This reduces the number of
    passes over the huge (B·N·N) tensor from five to one.
3️⃣  Optional mask‑gate fusion – a tiny Triton kernel multiplies the
    projection, the mask and the gate in a single memory pass.
4️⃣  Core contraction  
    `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`  
    is a batched GEMM for each hidden channel.  A custom Triton
    kernel computes this directly on the original memory layout,
    avoiding the costly `permute(...).contiguous()` copies required by
    `torch.bmm`.  The kernel tiles the (i,j)‑plane (64×64) and loops
    over k‑tiles (32) using Tensor‑Core `tl.dot`.
5️⃣  Output LayerNorm, gating and final projection (torch kernels).

All heavy arithmetic stays in float16; accumulations are performed in
float32 for numerical stability.  The final result is cast back to the
input dtype (usually float32).

The implementation uses two Triton kernels:
* `fused_mul3_kernel` – element‑wise product of three tensors.
* `tri_mul_outgoing_kernel` – the batched matrix multiplication that
  implements the outgoing update without any extra copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Element‑wise product of three tensors (mask * proj * gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,      # pointers
    size: tl.constexpr,                # total number of elements
    BLOCK_SIZE: tl.constexpr,          # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused a * b * c using Triton."""
    assert a.shape == b.shape == c.shape
    a, b, c = (x.contiguous() for x in (a, b, c))

    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192                     # good for large tensors on H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# 2) Outgoing contraction kernel (batched GEMM over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # stride information (same layout for left/right/out)
    stride_batch, stride_i, stride_k,
    # dimensions (compile‑time constants)
    N: tl.constexpr,               # sequence length
    HIDDEN: tl.constexpr,          # hidden dimension (always 128 in our tests)
    # tiling parameters (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs:
    #   pid0 – combined batch (b) and hidden channel (h) index
    #   pid1 – tile row (i)
    #   pid2 – tile col (j)
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch (b) and hidden (h) from pid0
    b = pid0 // HIDDEN
    h = pid0 % HIDDEN

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_batch + h
    right_base = right_ptr + b * stride_batch + h
    out_base   = out_ptr   + b * stride_batch + h

    # tile offsets within the N×N matrix
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    num_k_tiles = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(0, num_k_tiles):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (i, k)   shape: BLOCK_M × BLOCK_K
        a_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile (k, j) – note the transpose: we need
        # right[j,k] => treat as (k, j) for matmul.
        b_ptrs = right_base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N) accumulation

    # write the result back (cast to FP16)
    c_ptrs = out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def outgoing_contraction(left: torch.Tensor, right: torch.Tensor,
                         BLOCK_M=64, BLOCK_N=64, BLOCK_K=32) -> torch.Tensor:
    """
    Compute `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
    using a Triton kernel that works directly on the
    [B, N, N, hidden] layout (no extra copies).
    """
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    out = torch.empty_like(left)   # FP16 output
    # strides for the contiguous layout [B, N, i, k, hidden]
    stride_batch = left.stride(0)  # N*N*H
    stride_i = left.stride(1)      # N*H
    stride_k = left.stride(2)      # H

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        stride_batch,
        stride_i,
        stride_k,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# 3) Full “outgoing” TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict of model weights (torch.Tensor)
        - config_dict  : dict with keys ``dim``, ``hidden_dim``,
                        optional ``eps`` and ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (kept in float32, then cast to FP16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)          # half‑precision for the rest

    # ------------------------------------------------------------------
    # 2) Fuse five linear layers into a single matmul
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                       # (5*hidden, dim)
    ).to(torch.float16)

    # result shape: [B, N, N, 5*hidden]
    combined = F.linear(x, w_concat)    # bias is None

    # Split the concatenated dimension back into the five parts
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(
        combined, hidden, dim=-1
    )

    # Sigmoid gates (still FP16)
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present) and gates – fused element‑wise product
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # expand mask to hidden dimension (contiguous for Triton)
        mask_exp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden).contiguous()
        left = fused_mul3(left_proj_raw, mask_exp, left_gate)
        right = fused_mul3(right_proj_raw, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 4) Core contraction (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    out = outgoing_contraction(left, right,
                               BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)   # shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                         # element‑wise gating

    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to the original dtype (usually float32) and return
    return out_final.to(inp.dtype)
