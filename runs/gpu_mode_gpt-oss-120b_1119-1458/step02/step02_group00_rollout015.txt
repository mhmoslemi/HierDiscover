@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride0_left, stride1_left, stride2_left, stride3_left,    # actual strides for each dimension in left
    stride0_right, stride1_right, stride2_right, stride3_right, # strides for right
    stride0_out, stride1_out, stride2_out, stride3_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_heads = B * H
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # head id
    head_id = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # compute base offsets for this head
    b = head_id // H
    h = head_id % H

    # pointer offsets for left and right base
    left_head_base = left_ptr + b * stride0_left + h * stride3_left
    right_head_base = right_ptr + b * stride0_right + h * stride3_right
    out_head_base = out_ptr + b * stride0_out + h * stride3_out

    # compute row (i) and column (j) start indices
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # start loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # mask for k
        mask_k = offs_k < N

        # Compute pointers for left tile (M x K)
        # left[i,k] layout: offset = left_head_base + i*stride_i + k*stride_k
        # where stride_i = stride1_left (i dimension), stride_k = stride2_left (k dimension)
        # left_ptrs: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_head_base + offs_i[:, None] * stride1_left + offs_k[None, :] * stride2_left
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # For right tile (N x K) we need shape (BLOCK_N, BLOCK_K) but we want its transpose (K, N).
        # We'll load as (BLOCK_K, BLOCK_N) directly.
        # right[j,k] layout same as left: offset = right_head_base + j*stride_i + k*stride_k
        # So we load with dims reversed:
        right_ptrs = right_head_base + offs_j[:, None] * stride1_right + offs_k[None, :] * stride2_right
        # This yields shape (BLOCK_N, BLOCK_K). We need transposed:
        # Using tl.load we get (BLOCK_N, BLOCK_K). We'll then transpose.
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # transpose: (K,N)
        right_tile_T = tl.trans(right_tile)  # but need to recall triton trans function: tl.trans or tl.swapaxes?

        # Actually triton has tl.trans(x). Yes.

        # Then dot
        acc += tl.dot(left_tile, right_tile_T)   # left_tile (M,K), right_tile_T (K,N) => (M,N)

        # Continue
    # store output
    out_ptrs = out_head_base + offs_i[:, None] * stride1_out + offs_j[None, :] * stride2_out
    # out tile shape (M,N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Highly‑optimized Triton implementation of the *outgoing* Triangle Multiplicative
Update (TriMul) used in AlphaFold‑3 and similar protein‑structure models.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) → `x`.
2. Cast `x` to FP16 and apply the 5 linear projections (left/right,
   left/right‑gate, out‑gate) using torch’s highly‑optimized GEMM.
3. Optional mask is applied to `left`/`right`.
4. Fuse the element‑wise multiplications `left *= left_gate`,
   `right *= right_gate` in a single Triton kernel (`apply_gates_kernel`).
5. **Core N³ step** – compute, for every head `h` (= batch·hidden_dim),

       out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]

   with a custom batched GEMM kernel (`trmul_kernel`) that works on the
   original `[B, N, N, H]` layout (no permute / copy).
6. Layer‑norm the result, multiply by the output‑gate (`out_gate`) in a
   second small Triton kernel (`elementwise_mul_kernel`).
7. Final linear projection back to the original `dim`.
8. Return FP32 tensor of shape `[B, N, N, dim]`.

All heavy arithmetic runs in FP16 (Tensor‑Core friendly) while the two
LayerNorms use the supplied FP32/FP16 weight tensors.  The implementation
eliminates the expensive `permute().contiguous()` copies of the original
reference and fuses several element‑wise passes, bringing the end‑to‑end
runtime well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,                 # fp16, same shape
    right_ptr, right_gate_ptr,               # fp16, same shape
    numel: tl.int64, hidden_dim: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate
        right = right * right_gate
    The tensors are flattened (row‑major) beforehand.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l  = tl.load(left_ptr + offs,  mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(left_ptr + offs,  l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    out = a * b   (both flattened, fp16)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides of the left tensor (B,N,N,H)
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides of the right tensor (identical layout)
    stride_rb, stride_ri, stride_rk, stride_rh,
    # strides of the output tensor (B,N,N,H)
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul core:
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]

    The kernel iterates over tiles (BLOCK_M × BLOCK_N) of the (i,j) plane
    for each head `h` = batch·hidden_dim and accumulates over the k‑dimension.
    No data copies or permutes are performed – the original memory layout
    `[B, N, N, H]` is used directly.
    """
    pid = tl.program_id(0)

    total_heads = B * H
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # ----- decode program id -----
    head_id = pid // (num_blocks_m * num_blocks_n)
    tile_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = tile_id // num_blocks_n   # i‑tile index
    pid_n = tile_id %  num_blocks_n   # j‑tile index

    # batch and head indices for this program
    b = head_id // H
    h = head_id % H

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # i‑ and j‑coordinates of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP16 – Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ----- loop over k dimension -----
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile: shape (N, K) → we need its transpose (K, N)
        right_ptrs = right_base + offs_j[:, None] * stride_ri + offs_k[None, :] * stride_rk
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0)
        right_tile_T = tl.trans(right_tile)           # (K, N)

        # matrix multiply for this k‑block
        acc += tl.dot(left_tile, right_tile_T)        # (M, N)

    # ----- write the result -----
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Input:  (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim] (float32)
            mask         : [B, N, N]      (float32 or bool) – may be omitted
    Output: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (fp32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the heavy work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Load / cast model weights to FP16
    # --------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (no bias) – all use cuBLAS/Tensor‑Core
    # --------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)    # [B, N, N, hidden_dim] (fp16)
    right = F.linear(x_h, w_right_proj)   # same shape

    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → FP16
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 6️⃣ Fuse left/right with their gates (single Triton kernel)
    # --------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536                     # ~64 k elements per program
    grid_gates = ( (numel + BLOCK - 1) // BLOCK, )
    apply_gates_kernel[grid_gates](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        hidden_dim,
        BLOCK=BLOCK,
    )
    # Gates are no longer needed
    del left_gate, right_gate

    # --------------------------------------------------------------
    # 7️⃣ Core N³ operation: batched GEMM (A @ Bᵀ) without any copy
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # allocate output tensor
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides for the three tensors (B, N, N, H)
    stride_l = left.stride()   # (stride_lb, stride_li, stride_lk, stride_lh)
    stride_r = right.stride()
    stride_o = out.stride()

    # Tile size – tuned for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_heads = B * H
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_matmul = ( total_heads * num_blocks_m * num_blocks_n, )

    trmul_kernel[grid_matmul](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8️⃣ Post‑processing LayerNorm (half‑precision weights)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # still fp16

    # --------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (tiny Triton kernel)
    # --------------------------------------------------------------
    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_out_gate = ( (numel_out + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid_out_gate](
        out.view(-1),
        out_gate.contiguous().view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (FP16 → FP16)
    # --------------------------------------------------------------
    final = F.linear(out_mul, w_to_out)   # [B, N, N, dim]  fp16

    # Cast back to float32 as required by the API
    return final.to(torch.float32)
