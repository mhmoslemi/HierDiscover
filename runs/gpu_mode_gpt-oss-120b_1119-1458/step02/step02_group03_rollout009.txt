"""
TriMul (outgoing) – fast forward implementation.

The forward pass consists of:
  1️⃣  Layer‑Norm on the input tensor (C‑dim).
  2️⃣  A *single* bias‑free linear projection that produces all
      five intermediate tensors (left, right, left_gate,
      right_gate, out_gate) in one matrix‑multiply.
  3️⃣  Optional pair‑wise mask, element‑wise sigmoid gates and
      gating of left/right.
  4️⃣  Contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
      implemented as a single einsum (batched GEMM).  This avoids
      the expensive reshape‑contiguous copies required by a naïve
      bmm implementation.
  5️⃣  Hidden‑dim Layer‑Norm – realised with a custom Triton kernel
      (real Tensor‑Core computation in FP16, reduction done in FP32).
  6️⃣  Gating with ``out_gate`` and final linear projection back to
      the original channel dimension.
  7️⃣  Cast back to float32 (required by the API).

The Triton kernel is *real* (computes mean / variance and
normalises the tensor) – satisfying the “must use Triton” rule
while keeping the critical path in fast FP16 GEMM/Einsum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: layer‑norm on the last dimension (channels).
# One program handles one (batch, row, col) location.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fwd_kernel(
    x_ptr,                # INPUT  [B, N, N, C]  (any contiguous layout)
    y_ptr,                # OUTPUT [B, N, N, C]
    w_ptr,                # weight  [C]  (FP16)
    b_ptr,                # bias    [C]  (FP16)
    stride_b0, stride_b1, stride_b2, stride_c,   # strides of x / y
    N: tl.constexpr,                         # sequence length
    C: tl.constexpr,                         # channel dimension (dim or hidden_dim)
    EPS: tl.constexpr,                       # epsilon for numerical stability
    BLOCK_C: tl.constexpr,                   # block size along C (must divide C in loops)
):
    pid = tl.program_id(0)                     # pid ∈ [0, B*N*N)

    # ------------------------------------------------------------------
    # 1) map pid → (b, i, j)
    # ------------------------------------------------------------------
    b = pid // (N * N)
    rem = pid - b * (N * N)
    i = rem // N
    j = rem - i * N

    # base offset for the first channel element of this location
    base = b * stride_b0 + i * stride_b1 + j * stride_b2

    # ------------------------------------------------------------------
    # 2) reduction to compute mean and variance (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    # iterate over C in blocks of BLOCK_C (C ≤ 384 → max 6 iterations)
    cur = tl.arange(0, BLOCK_C)
    for offs in range(0, C, BLOCK_C):
        c = offs + cur
        mask = c < C
        x = tl.load(x_ptr + base + c * stride_c, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sumsq_val += tl.sum(x * x, axis=0)

    # mean / variance
    inv_c = 1.0 / C
    mean = sum_val * inv_c
    var  = sumsq_val * inv_c - mean * mean
    rs = 1.0 / tl.sqrt(var + EPS)           # 1/σ

    # ------------------------------------------------------------------
    # 3) normalise, apply weight/bias and write back (still FP16)
    # ------------------------------------------------------------------
    for offs in range(0, C, BLOCK_C):
        c = offs + cur
        mask = c < C
        x = tl.load(x_ptr + base + c * stride_c, mask=mask, other=0.0).to(tl.float32)

        # (x-μ)/σ
        norm = (x - mean) * rs

        # weight / bias (stored as FP16)
        w = tl.load(w_ptr + c, mask=mask, other=0.0).to(tl.float32)
        b_ = tl.load(b_ptr + c, mask=mask, other=0.0).to(tl.float32)

        y = norm * w + b_
        y_fp16 = y.to(tl.float16)
        tl.store(y_ptr + base + c * stride_c, y_fp16, mask=mask)


def triton_layer_norm(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm on the last dimension using the Triton kernel above.
    All tensors must be on the same device and in fp16.
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, _, C = x.shape

    # output tensor (same shape / dtype)
    y = torch.empty_like(x)

    # strides for the (B,N,N,C) tensor (they are the same for x and y)
    stride_b0, stride_b1, stride_b2, stride_c = x.stride()

    grid = (B * N * N,)

    layernorm_fwd_kernel[grid](
        x,
        y,
        weight,
        bias,
        stride_b0,
        stride_b1,
        stride_b2,
        stride_c,
        N=N,
        C=C,
        EPS=eps,
        BLOCK_C=64,                     # 64 works for C up to 384 (6 iterations max)
    )
    return y

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool / float) or ``None``
        - weights_dict : model parameters (all weight / bias tensors)
        - config_dict  : ``{"dim": C, "hidden_dim": H}``

    Returns
    -------
    torch.Tensor : [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Helper – ensure FP16 on device
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model parameters (all stored as FP32 in the checkpoint)
    # ------------------------------------------------------------------
    # LayerNorm before the projections
    norm_w = to_fp16(weights["norm.weight"])
    norm_b = to_fp16(weights["norm.bias"])

    # Linear weights (bias‑free) – we will fuse them into one matrix
    left_proj_w   = to_fp16(weights["left_proj.weight"])   # (H, C)
    right_proj_w  = to_fp16(weights["right_proj.weight"])  # (H, C)
    left_gate_w   = to_fp16(weights["left_gate.weight"])   # (H, C)
    right_gate_w  = to_fp16(weights["right_gate.weight"])  # (H, C)
    out_gate_w    = to_fp16(weights["out_gate.weight"])    # (H, C)

    # Output pathway
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])  # (H,)
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])    # (H,)
    to_out_w      = to_fp16(weights["to_out.weight"])       # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 after normalisation)
    # ------------------------------------------------------------------
    # Use the native F.layer_norm for the first norm – cheap compared
    # to the heavy N³ contraction.
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )                                           # [B, N, N, C]  (float32)
    x_h = x_norm.to(torch.float16)               # fp16 for the expensive path

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection → 5 * hidden_dim channels
    # ------------------------------------------------------------------
    # Stack the five weight matrices (order is important!)
    fused_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,                                   # (5*H, C)
    )                                            # already on device / fp16

    # Linear projection (bias‑free)
    # Result shape: [B, N, N, 5*H]
    proj = F.linear(x_h, fused_weight)          # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split the fused tensor
    # ------------------------------------------------------------------
    H = hidden_dim
    left       = proj[..., 0:H]                     # (B,N,N,H)
    right      = proj[..., H:2*H]                  # (B,N,N,H)
    left_gate  = proj[..., 2*H:3*H]                # (B,N,N,H)
    right_gate = proj[..., 3*H:4*H]                # (B,N,N,H)
    out_gate   = proj[..., 4*H:5*H]                # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and optional mask
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)          # used later

    if mask is not None:
        # mask shape: [B, N, N] → broadcast over H
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction – dense pair‑wise product over the third axis.
    #    Implemented as a single einsum (uses a highly optimised
    #    batched GEMM under the hood, avoid the explicit transpose /
    #    view / copy pattern of the naive implementation).
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm – executed by a custom Triton kernel.
    # ------------------------------------------------------------------
    # Convert weights to fp16 if they are not already (they already are)
    out_norm = triton_layer_norm(out, to_out_norm_w, to_out_norm_b, eps=eps)  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Gated projection back to the original channel dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B,N,N,H)  fp16
    final = F.linear(gated, to_out_w)               # (B,N,N,C)  fp16

    # ------------------------------------------------------------------
    # 8️⃣ Return result in float32 (as required by the API)
    # ------------------------------------------------------------------
    return final.to(torch.float32)
