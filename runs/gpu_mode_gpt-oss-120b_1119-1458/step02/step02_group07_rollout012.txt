def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim, nomask = config["dim"], config["hidden_dim"], config.get("nomask", True)

    eps=1e-5
    # Layer norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps) # float32
    x_fp16 = x.to(torch.float16)

    # Concatenate left_proj and left_gate weights
    left_concat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)  # shape (2*hidden_dim, dim)
    right_concat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    left_concat = F.linear(x_fp16, left_concat_weight) # shape (B,N,N,2*hidden_dim)
    right_concat = F.linear(x_fp16, right_concat_weight)

    left = left_concat[..., :hidden_dim]  # fp16
    left_gate_pre = left_concat[..., hidden_dim:]  # fp16

    right = right_concat[..., :hidden_dim]

        right = right * right_gate

        right = right.contiguous()

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣ Layer‑Norm the input (float32, fp32 accumulation).
2️⃣ Cast to fp16 and fuse each projection + its gate into a single matmul
   by concatenating the weight matrices (left_proj+left_gate, right_proj+right_gate).
   This halves the number of passes over the large (B,N,N,dim) tensor.
3️⃣ Apply sigmoid to the gate slices.
4️⃣ If a mask is present, a single Triton kernel fuses:
       left = left * left_gate * mask
       right = right * right_gate * mask
   All operations are in‑place, avoiding extra memory traffic.
5️⃣ Core O(N³) operation – for every head d we need
       out[...,i,j,d] = Σₖ left[...,i,k,d] * right[...,j,k,d] .
   This is expressed as a batch of independent GEMMs:
       C = left_mat @ right_matᵀ
   where left_mat / right_mat have shape (B*hidden_dim, N, N).
   The batched GEMM runs in fp16 with Tensor‑Core accumulation.
6️⃣ Layer‑Norm on the hidden dimension, element‑wise multiplication by the
   output gate (sigmoid of a third linear projection) and a final linear
   projection back to the original feature dimension.

All heavy work is executed in half‑precision; the final result is returned in
float32. The implementation reduces global‑memory passes and kernel launch
overheads compared to the reference PyTorch version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                       # fp16 * left projection      [B, N, N, H]
    right_ptr,                      # fp16 * right projection     [B, N, N, H]
    mask_ptr,                       # fp16 * mask                 [B, N, N]
    left_gate_ptr,                  # fp16 * left gate            [B, N, N, H]
    right_gate_ptr,                 # fp16 * right gate           [B, N, N, H]
    # strides for left / right (b,i,j,h)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    # strides for gates (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,                       # batch, seq_len, hidden_dim
    BLOCK_H: tl.constexpr,         # tile size along H
):
    pid = tl.program_id(0)                # one program per (b, i, j)
    total = B * N * N
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors (no striding on element dimension)
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # scalar mask (fp16) – broadcast across H
    mask_val = tl.load(mask_ptr + off_m)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in BLOCK_H‑sized chunks
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        # load a tile of each tensor (vectorised across H)
        left = tl.load(left_ptr + off_l + offs * stride_l3,
                       mask=mask_h, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs * stride_r3,
                        mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_g + offs * stride_g3,
                             mask=mask_h, other=tl.float16(0.0))

        # fused operation: proj * gate * mask
        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        # store back in‑place
        tl.store(left_ptr + off_l + offs * stride_l3, left_out, mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)


# ----------------------------------------------------------------------
# Entry‑point expected by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: shape [B, N, N, dim] (float32)
            - mask: shape [B, N, N] (float32) – may be None/ignored if `nomask=True`
            - weights: dict with the eight parameter tensors (float32)
            - config: dict containing keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate (left & right) by concatenating weights
    # ------------------------------------------------------------------
    # concatenate weights along the out_features dimension: [proj | gate]
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                         # (2*H, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                         # (2*H, dim)

    # single matmul per side
    left_concat = F.linear(x_fp16, left_concat_w)   # [B, N, N, 2*H] fp16
    right_concat = F.linear(x_fp16, right_concat_w) # [B, N, N, 2*H] fp16

    # split projection and gate tensors
    left = left_concat[..., :hidden_dim]                     # proj
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:]) # gate
    right = right_concat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask & gates (mask optional)
    # ------------------------------------------------------------------
    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:
        # ensure contiguous tensors for Triton
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]  # B * N * N
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left,
            right,
            mask_fp16,
            left_gate,
            right_gate,
            # strides for left / right
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gates
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # problem sizes
            input_tensor.shape[0],            # B
            input_tensor.shape[1],            # N
            hidden_dim,                       # H
            BLOCK_H=32,                       # tile along hidden dim
            num_warps=4,
        )
        # after kernel, `left` and `right` already contain mask*gate*proj

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: Σₖ left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    batch_heads = B * H

    # reshape to (B*H, N, N) without an extra copy of the data layout.
    # permute moves the hidden dimension before the two sequence dims,
    # then we view it as a batch of independent matrices.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch_heads, N, N)

    # GEMM (fp16 inputs, fp32 accumulation on Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # output gate (shared with the very first linear projection)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_fp16 = torch.sigmoid(F.linear(x_fp16, out_gate_w))
    out_gate_f32 = out_gate_fp16.to(torch.float32)

    out_norm = out_norm * out_gate_f32

    # final linear back to dim (float32 weights)
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
