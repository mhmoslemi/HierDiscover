"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32 for stability).
2️⃣ Cast to fp16 and flatten to (B·N·N, dim).
3️⃣ Fuse the four linear projections (left, left‑gate, right, right‑gate)
   into a single GEMM:   X @ W_all.T   where
   W_all = [left_proj; left_gate; right_proj; right_gate]  (shape 4·H × dim).
4️⃣ Split the result into the four raw tensors.
5️⃣ A tiny Triton kernel fuses:
      * mask broadcasting (if a mask is supplied)
      * sigmoid‑gate computation
      * element‑wise multiplication
   producing the gated left/right tensors in fp16.
6️⃣ Reshape to (B, H, N, N) and execute the N³ contraction as a batched
   matrix‑multiply (left @ rightᵀ) using cuBLAS (`torch.bmm`), which on an H100
   runs at Tensor‑Core speed.
7️⃣ Layer‑norm over the hidden dimension (fp32 for numerical safety) and
   apply the output‑gate (sigmoid of a linear on the original input).
8️⃣ Final linear projection back to the original channel dimension.
9️⃣ Cast the result to float32 and return.

Only the mask‑gate fusion (step 5) is implemented in Triton; the rest
leverages highly‑optimised PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid‑gate and element‑wise multiplication
# -------------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,      # const half *  (raw projection)
    gate_ptr,      # const half *  (raw gate logits)
    mask_ptr,      # const half *  (mask broadcasted over hidden dim)
    out_ptr,       # half *        (output tensor)
    hidden_dim: tl.constexpr,
    total_elts: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """out = proj * mask * sigmoid(gate)   (flattened view)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    # Load raw tensors
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask: one mask value per hidden_dim entries
    mask_idx = offs // hidden_dim            # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1.0 → “no mask”

    # Sigmoid (1 / (1 + exp(-x))) – safe in fp16
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * m * gate_sig
    tl.store(out_ptr + offs, out, mask=mask)

    x_half = x_norm.to(torch.float16)                # fp16
    x_flat = x_half.view(-1, dim)                    # (B·N·N, dim)

    # ------------------------------------------------------------------
    # 3️⃣ One‑shot GEMM for all four linear maps
    # ------------------------------------------------------------------
    # Stack the four weight matrices vertically → shape (4·H, dim)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*hidden_dim, dim)

    # Linear (bias‑free) → (B·N·N, 4·hidden_dim)
    all_proj = F.linear(x_flat, combined_w)          # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Split into the four raw tensors
    # ------------------------------------------------------------------
    hp = hidden_dim
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw = torch.split(
        all_proj, hp, dim=1
    )  # each (B·N·N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & sigmoid‑gate using Triton
    # ------------------------------------------------------------------
    # Prepare mask (flattened, fp16). If there is no mask, use all‑ones.
    if mask is None:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(dtype=torch.float16, device=device).view(-1)

    # Allocate output tensors for gated left/right
    left = torch.empty_like(left_proj_raw)   # (B·N·N, hidden_dim)
    right = torch.empty_like(right_proj_raw)

    # Kernel launch configuration
    total_elems = left_proj_raw.numel()      # B·N·N·hidden_dim
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    # Left branch
    mask_gate_fuse_kernel[grid](
        left_proj_raw,
        left_gate_raw,
        mask_flat,
        left,
        hidden_dim=hidden_dim,
        total_elts=total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # Right branch
    mask_gate_fuse_kernel[grid](
        right_proj_raw,
        right_gate_raw,
        mask_flat,
        right,
        hidden_dim=hidden_dim,
        total_elts=total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Reshape for the N³ contraction (batched GEMM)
    # ------------------------------------------------------------------
    # From (B·N·N, hidden_dim) → (B, hidden_dim, N, N)
    left_view = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right_view = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()

    # Batch dimension for torch.bmm: (B·H, N, N)
    left_flat = left_view.view(B * hidden_dim, N, N)
    right_flat = right_view.view(B * hidden_dim, N, N)

    # Batched matmul: left @ rightᵀ  (Tensor‑Core fp16 GEMM)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension (float32 for stability)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid) and final projection back to original dim
    # ------------------------------------------------------------------
    # out_gate = sigmoid(out_gate_weight @ x_half)   – same x used earlier
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))   # (B,N,N,hidden_dim)

    gated = out_norm * out_gate                             # (B,N,N,hidden_dim)

    # Final linear projection (hidden_dim → dim)
    final = F.linear(gated, to_out_w)                       # (B,N,N,dim)  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Return in float32 as required
    # ------------------------------------------------------------------
    return final.to(torch.float32)
```
