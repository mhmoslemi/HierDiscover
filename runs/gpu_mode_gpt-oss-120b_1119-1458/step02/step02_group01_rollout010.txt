"""
Optimized outgoing TriMul kernel.

Implements the heavy N^3 contraction with a Triton kernel that operates on
padded tensors to avoid per‑element masking.  Input tensors are LayerNormed
in FP32, then cast to FP16 for all linear / gating operations.  The kernel
computes C = A @ Bᵀ for each (batch, hidden) pair using 64×64×64 tiles,
FP16 MMA, FP32 accumulation and a 3‑stage pipeline (8 warps).  The output
is sliced back to the original sequence length, LayerNormed again, gated,
and linearly projected to the original dimension.  The final result is
returned as FP32.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]  (M = batch * hidden, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # batch*hidden index

    # compute offsets for the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for kk in range(num_k):
        cur_k = kk * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load tile from A (i, k) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- load tile from B transposed (k, j) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # dot product + accumulate (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # write back (cast to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Compute batched C = left @ rightᵀ with Triton.
    left/right: [B, N, N, H] (fp16).  N may be padded to a multiple of `tile`.
    Returns: tensor of shape [B, N_original, N_original, H] (fp16).
    """
    B, N, _, H = left.shape
    # pack batch and hidden dimension
    # after permute we have shape [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A)

    # strides (in element units)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # Cast to FP16 for the rest of the pipeline
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear weight conversion to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Flattened linear projections and gates (FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    # Project
    left = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # Apply mask if present

        left = left * mask_fp16
        right = right * mask_fp16

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)

    # Fuse gates into projections
    left = left * left_gate

    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
Optimized outgoing TriMul kernel (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the 4‑D input in FP32 → cast to FP16.
2. Flatten the tensor and apply all linear projections / sigmoids in FP16.
3. (Optional) multiply projected tensors by the mask.
4. Fuse the per‑position gates into the projections.
5. Pad the sequence dimension to a multiple of 64 so the Triton kernel can
   avoid per‑element bounds checks.
6. Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton kernel that performs a batched matrix multiply
   C = A @ Bᵀ for each (batch, hidden) pair.  The kernel uses 64×64×64
   tiles, FP16 MMA, FP32 accumulation, 8 warps and a 3‑stage pipeline.
7. Slice away the padding, apply the second LayerNorm, the output gate,
   and the final linear projection back to the original `dim`.
8. Return the result in FP32.

Only the O(N³) part is written in Triton; all surrounding ops use
high‑performance PyTorch primitives.  The implementation works for the
full range of shapes required in the test suite.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]  (M = batch * hidden, N = seq_len (padded), K = seq_len (padded))
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile along first N dimension (i)
    pid_n = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A[i, k] tile ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---- load Bᵀ[k, j] tile (i.e. B[k, j]) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # dot (fp16 * fp16 -> fp32) and accumulate
        acc += tl.dot(a, b)

    # write result (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
) -> torch.Tensor:
    """
    Perform C = left @ rightᵀ for each (batch, hidden) pair using Triton.
    `left` / `right` must be FP16 tensors of shape [B, N, N, H] where N is
    already padded to a multiple of `tile`.  Returns a tensor of the same
    shape (still padded) in FP16.
    """
    B, N, _, H = left.shape

    # pack batch and hidden dimensions -> [B*H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    C = torch.empty_like(A)  # FP16 output

    # strides in element units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = tile
    BLOCK_N = tile
    BLOCK_K = tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)  # switch to fp16 for the heavy work

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Flattened linear projections & gates (FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                       # (B*N*N, dim)

    left  = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # Fuse gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Pad to a multiple of the tile size to drop bound‑checks in Triton
    # ------------------------------------------------------------------
    TILE = 64
    pad_len = (TILE - (N % TILE)) % TILE
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad  = left.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, hidden_dim), dtype=torch.float16)
        left_pad[:, :N, :N, :]  = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (FP16 inputs, FP16 output)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_pad, right_pad, tile=TILE)  # [B, N_pad, N_pad, hidden_dim]

    # Strip padding
    if pad_len != 0:
        out = out[:, :N, :N, :]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Output gating + final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (harness expects FP32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
