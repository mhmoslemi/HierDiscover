Thus left_tile: shape (BM, BK) from left pointer with i_offset * stride_i + k_offset*stride_k + h*stride_h.

Right_tile: shape (BK, BN) loaded similarly but we need to use stride_k for rows and stride_i for columns.

Thus dot product: tl.dot(left_tile, right_tile) yields shape (BM, BN). Good.

Thus we can compute per hidden channel.

Thus the overhead of copy is eliminated. Performance might approach cublas but possibly slower. But it's plausible can achieve runtime < baseline.

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)   # combined batch and hidden

    # Compute batch index and hidden channel
    hidden_per_batch = hidden_dim
    batch_idx = pid_hb // hidden_per_batch
    hidden_idx = pid_hb % hidden_per_batch

    # Compute tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Offsets for hidden dimension
    hidden_off = hidden_idx

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        cur_K = tl.minimum(BLOCK_K, N - k)
        # Offsets for k
        offs_k = k + tl.arange(0, BLOCK_K)

        # Load left tile (BM, cur_K)
        left_ptrs = left_ptr \
            + batch_idx * stride_left_b \
            + (offs_i[:, None] * stride_left_i) \
            + (offs_k[None, :] * stride_left_k) \
            + hidden_off * stride_left_h

        left = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < N), other=0.0).to(tl.float32)  # shape (BM, BK)

        # Load right tile (BK, BN) (as transposed)
        right_ptrs = right_ptr \
            + batch_idx * stride_right_b \
            + (offs_k[:, None] * stride_right_k) \
            + (offs_j[None, :] * stride_right_i) \
            + hidden_off * stride_right_h

        right = tl.load(right_ptrs, mask=(offs_k[:, None] < N) & mask_j[None, :], other=0.0).to(tl.float32)  # shape (BK, BN)

        # Compute block multiplication
        acc += tl.dot(left, right)  # MxN result
    # End K loop

    # Store result
    out_ptrs = out_ptr \
        + batch_idx * stride_out_b \
        + (offs_i[:, None] * stride_out_i) \
        + (offs_j[None, :] * stride_out_j) \
        + hidden_off * stride_out_h

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul – “outgoing” update (AlphaFold3) – Triton‑accelerated version
------------------------------------------------------------------

Algorithm
---------
1️⃣  Layer‑Norm over the feature dimension (float32 for numerical stability).
2️⃣  Cast to FP16 and apply the five linear projections (left/right, three gates).
    The gated values are fused with the optional mask by a small Triton kernel
    to avoid extra Python / CUDA calls.
3️⃣  Core O(N³) operation:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]   (hidden‐dim d ∈ [0,…,H))
    This is a batch of independent matrix multiplies A @ Bᵀ, one for each
    (batch, hidden) pair.  We implement it with a custom Triton kernel that
    works directly on the original (non‑contiguous) layout, thus removing the
    costly ``permute(...).contiguous()`` copy required by a naïve torch.bmm.
    The kernel tiles the (i,j) plane (BLOCK_M × BLOCK_N) and iterates over k
    in blocks (BLOCK_K).  Inside each k‑tile we load a sub‑matrix from
    ``left`` (M×K) and a sub‑matrix from ``right`` (K×N) (the latter is loaded
    transposed by swapping the stride order) and accumulate with ``tl.dot``.
    All arithmetic is performed in FP16; accumulation uses FP32.
4️⃣  Convert the result back to FP32, apply the hidden‑dim LayerNorm,
    multiply by the output gate and finally project back to the original
    feature dimension.

The heavy O(N³) work stays on the GPU, all tensor‑core friendly and
memory‑copy free.  The only Triton kernels used are the mask/gate fusion
(kernel ``_gate_mask_kernel``) and the batched mat‑mul kernel
(``_tri_mul_outgoing_kernel``).  The forward pass returns a float‑32
tensor of shape ``[B, N, N, dim]`` matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1 – fuse (projection * gate) and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,        # fp16 – projection (left or right)
    gate_ptr,        # fp16 – corresponding gate
    mask_ptr,        # fp16 – mask (flattened B×N×N)
    out_ptr,         # fp16 – destination
    hidden_dim,      # i64   hidden dimension (used for broadcasting the mask)
    n_elements,      # i64   total number of elements in proj/gate (B·N·N·H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < n_elements

    # Load projection and gate (fp16 → fp32 for precision)
    proj = tl.load(proj_ptr + offs, mask=valid).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=valid).to(tl.float32)

    # Mask broadcasting: each mask entry repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=valid)


# ----------------------------------------------------------------------
# Kernel 2 – batched “outgoing” triangle multiplication.
# Computes: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,          # fp16 – [B,N,N,H] (already gated & masked)
    right_ptr,         # fp16 – same layout as left
    out_ptr,           # fp16 – result tensor
    B, N, H,           # runtime constants
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Grid layout
    #   program_id(0) → i‑tile
    #   program_id(1) → j‑tile
    #   program_id(2) → (batch, hidden) pair
    # --------------------------------------------------------------
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)             # 0 … B*H-1

    batch_idx = pid_hb // H               # which batch (B ≤ 2)
    hidden_idx = pid_hb % H                # which hidden channel (0 … H-1)

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Accumulator in FP32
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, N - k)               # length of this k‑tile
        offs_k = k + tl.arange(0, BLOCK_K)               # [0, BLOCK_K)

        # ----- left tile (M × K) -------------------------------------------------
        # layout: left[batch, i, k, hidden]
        left_ptrs = (left_ptr
                     + batch_idx * stride_left_b
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k
                     + hidden_idx * stride_left_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & (offs_k[None, :] < N),
                       other=0.0).to(tl.float32)          # (M, K)

        # ----- right tile (K × N)  – we load the transposed view ---------------
        # layout: right[batch, j, k, hidden]  → we need rightᵀ[k, j]
        right_ptrs = (right_ptr
                      + batch_idx * stride_right_b
                      + offs_k[:, None] * stride_right_k        # rows = k
                      + offs_j[None, :] * stride_right_i        # cols = j
                      + hidden_idx * stride_right_h)
        right = tl.load(right_ptrs,
                        mask=(offs_k[:, None] < N) & mask_j[None, :],
                        other=0.0).to(tl.float32)          # (K, N)

        # Block matrix multiplication (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)                       # FP32 accumulation
    # end for k

    # -----------------------------------------------------------------
    # Store the result (FP16, with masking)
    # -----------------------------------------------------------------
    out_ptrs = (out_ptr
                + batch_idx * stride_out_b
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j
                + hidden_idx * stride_out_h)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Triton‑powered forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   (float32)
            mask         : [B, N, N]        (bool / float32)
            weights      : dict of model weights (float32)
            config       : {"dim": dim, "hidden_dim": hidden_dim}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    eps = 1e-5                         # LayerNorm epsilon

    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                               # [B,N,N,dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for all heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear / gating weights (cast once to FP16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].t().to(torch.float16)   # (dim, hidden)
    right_proj_w  = weights["right_proj.weight"].t().to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].t().to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].t().to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].t().to(torch.float16)

    # ------------------------------------------------------------------
    # Projections (FP16, bias‑free)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right_proj = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # Gates (sigmoid, stay in FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w)).to(torch.float32)  # keep FP32

    # ------------------------------------------------------------------
    # Apply mask & fuse projection × gate × mask using a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:        # mask shape: [B,N,N] → broadcast over hidden
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                     # [B·N·N]
        # allocate the gated tensors (same shape as projections)
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # launch config (chosen large enough for the whole tensor)
        BLOCK = 65536
        n_elem = left_proj.numel()                       # B·N·N·H
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask → simple element‑wise product (still FP16)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Core O(N³) multiplication – Triton implementation
    # ------------------------------------------------------------------
    # Allocate output (FP16) – shape [B,N,N,hidden]
    out = torch.empty_like(left)

    # Block sizes – tuned for H100 tensor‑core usage
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_hb = B * hidden_dim                     # (batch, hidden) dimension
    grid = (grid_i, grid_j, grid_hb)

    # Strides (int64) from the tensors – required for the kernel
    s_left = left.stride()       # (batch, i, k, hidden)
    s_right = right.stride()
    s_out = out.stride()

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                 # uses 4 warps (adjustable)
    )
    # out now holds the raw triangle‑multiplication result (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Post‑processing (float32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)                     # [B,N,N,hidden] FP32

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                # hidden‑dim LayerNorm

    out = out * out_gate                             # broadcast over hidden

    # Final projection back to original feature dimension (float32)
    to_out_weight = weights["to_out.weight"]          # shape (dim, hidden)
    out = F.linear(out, to_out_weight)                # [B,N,N,dim] (float32)

    return out
