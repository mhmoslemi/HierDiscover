TriMul (outgoing) – fast forward implementation for AlphaFold‑3.

Algorithm
---------
1️⃣ Layer‑norm the [B, N, N, C] tensor (float32 for stability).
2️⃣ Cast to FP16 and fuse the *projection* + *gate* linear layers:

@triton.jit
def _elemwise_mul_kernel(
    dst_ptr,            # *fp16 – destination (in‑place)
    src_ptr,            # *fp16 – source (gate)
    N,                  # i32   – total number of elements
    dst_stride: tl.constexpr,  # i32 – stride for dst (usually 1)
    src_stride: tl.constexpr,  # i32 – stride for src (usually 1)
    BLOCK_SIZE: tl.constexpr,  # i32 – threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    dst = tl.load(dst_ptr + offs * dst_stride, mask=mask)
    src = tl.load(src_ptr + offs * src_stride, mask=mask)

    tl.store(dst_ptr + offs * dst_stride, dst * src, mask=mask)


# ----------------------------------------------------------------------
# Batched matrix‑multiply for the O(N³) contraction.
# Computes C = A @ Bᵀ  where A,B: [B*H, N, N]  →  out: [B, N, N, H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose(
    left: torch.Tensor,   # [B, N, N, H]   FP16
    right: torch.Tensor   # [B, N, N, H]   FP16
) -> torch.Tensor:
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] for a strided‑batched GEMM
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ   (torch uses the fast Tensor‑Core implementation)
    C = torch.bmm(A, B_.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32, on the same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    x = x_norm.to(torch.float16)          # work in FP16 from now on

    # -------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # -------------------------------------------------
    # projection + gate weights are fused later
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fuse projection + gate (2 × Linear instead of 4)
    # -------------------------------------------------
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    left_comb   = torch.nn.functional.linear(x, left_comb_w)    # [B,N,N,2*H]
    right_comb  = torch.nn.functional.linear(x, right_comb_w)   # [B,N,N,2*H]

    # split back into projection & gate tensors
    left_proj = left_comb[..., :hidden_dim]          # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:]          # [B,N,N,H]

    right_proj = right_comb[..., :hidden_dim]        # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:]        # [B,N,N,H]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        # mask is 0/1 (float or bool) → convert to FP16 and broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still FP16)
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left  = left_proj * left_gate      # [B,N,N,H]
    right = right_proj * right_gate    # [B,N,N,H]

    # -------------------------------------------------
    # 6️⃣ O(N³) contraction via a strided‑batched GEMM (torch.bmm)
    # -------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B,N,N,H]  FP16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Out‑gate (sigmoid) and fused element‑wise multiplication
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] FP16

    # flatten tensors for the Triton kernel (in‑place dst ← dst * src)
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    total_elems = out_flat.numel()
    BLOCK_SIZE = 1024                         # ≤ max threads per block
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elems,
        1,          # dst stride
        1,          # src stride
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # -------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 🔚 Cast back to float32 (required output dtype)
    # -------------------------------------------------
    return out.to(torch.float32)
