@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr, 
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # row tile
    pid_n = tl.program_id(1)  # col tile
    pid_batch_head = tl.program_id(2)  # batch_head index

    # Compute batch index b and head index h from pid_batch_head
    # Assume H maybe >1, but we treat head dimension as H; we need both
    # batched = B*H
    total_bh = B * H
    bh = pid_batch_head
    b = bh // H
    h = bh % H

    # Compute offsets for row and column start
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask out-of-bounds
    i_mask = offs_i < N
    j_mask = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Offsets of K block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute pointer address for left: L[b,i,k,h]
        # left_ptr: shape [B, N, N, H]
        left_offset = (b * stride_left_b) + (offs_i[:, None] * stride_left_i) \
                      + (offs_k[None, :] * stride_left_k) + (h * stride_left_h)
        left = tl.load(left_ptr + left_offset, mask=(i_mask[:, None] & (offs_k[None, :] < N)), other=0.0)

        # Compute pointer for right: R[b,j,k,h] -> we need transposed: right_T[k,j] = R[b,j,k,h]
        # So we'll load as: (k dimension as rows, j dimension as columns): Shape (BLOCK_K, BLOCK_N)
        # offsets: k rows index (offs_k) as first dim; j columns as offs_j as second dim. So compute offset accordingly.
        right_offset = (b * stride_right_b) + (offs_j[None, :] * stride_right_j) \
                       + (offs_k[:, None] * stride_right_k) + (h * stride_right_h)
        right_T = tl.load(right_ptr + right_offset, mask=(offs_k[:, None] < N) & (j_mask[None, :]), other=0.0)
        # now left shape (M,K) and right_T shape (K,N)
        # cast to float32 for accumulation
        acc += tl.dot(left, right_T)

    # Store result to out: out[b,i,j,h]
    # out offsets:
    out_offset = (b * stride_out_b) + (offs_i[:, None] * stride_out_i) \
                 + (offs_j[None, :] * stride_out_j) + (h * stride_out_h)
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

    # left and right shape (B, N, N, H) dtype fp16
    B, N, _, H = left.shape

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_bh = tl.program_id(2) # batch*head

    total_heads = H
    # map pid_bh to batch b and head h
    b = pid_bh // total_heads
    h = pid_bh % total_heads

    # Compute global row/col start indices
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Offsets for M and N dimensions
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Boundary masks
    i_valid = i < N
    j_valid = j < N

    # Initialize accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute base offsets for this batch and head
    # left base = b*stride_l_b + h*stride_l_h
    left_base = b * stride_l_b + h * stride_l_h
    right_base = b * stride_r_b + h * stride_r_h
    out_base = b * stride_o_b + h * stride_o_h

    # Loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        k_valid = k < N

        # Load left tile [M, K]
        # offsets = left_base + i[:,None]*stride_l_i + k[None,:]*stride_l_k
        left_ptrs = left_ptr + left_base + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)

        # Load right tile transposed [K, N] (i.e., right[j, k] -> k rows, j cols)
        right_ptrs = right_ptr + right_base + j[None, :] * stride_r_i + k[:, None] * stride_r_k
        right = tl.load(right_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)

        # Dot product accumulation. Both left and right are fp16; accumulator fp32.
        acc += tl.dot(left, right)

    # Store output ([M,N])
    out_ptrs = out_ptr + out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])
```

```python
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # Ensure contiguous and half dtype
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # same shape and dtype

    # Determine block sizes (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides as ints
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # Launch grid
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )
    # Variation: use num_warps=8 for fp16 performance
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32) → produces `x_norm`.
2. Cast to FP16 for the compute‑heavy path.
3. Two bias‑free linear projections generate tensors `left` and `right`
   of shape [B, N, N, H] (H = hidden_dim, half‑precision).
4. Optional pair‑wise mask is broadcast and applied in‑place.
5. Gating: sigmoid(linear(x)) is multiplied into `left` and `right`,
   and a third gate `out_gate` is stored for later use.
6. Contraction over the third dimension (k):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This dense N³ operation is performed by a custom Triton kernel that
   treats each (b, h) pair as an independent GEMM and tiles the
   multiplication (M‑tile × N‑tile × K‑tile).  Accumulation is done in
   FP32 and the result is stored as FP16.
7. Layer‑norm on the hidden dimension (`to_out_norm`).
8. Apply the output gate and a final linear projection back to the
   original channel dimension (`dim`).
9. Return a float32 tensor as required.

The heavy N³ contraction dominates runtime; the tiled Triton GEMM
exploits Tensor‑Core instructions on H100 and removes the overhead of
batched cuBLAS calls.  All other lightweight ops are kept in PyTorch
for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM that implements the
# out‑going TriMul contraction:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Tiled batched matrix multiplication for TriMul."""
    pid_m = tl.program_id(0)          # output row block
    pid_n = tl.program_id(1)          # output column block
    pid_bh = tl.program_id(2)         # batch*head identifier

    # Resolve batch and hidden‑head indices
    b = pid_bh // H
    h = pid_bh % H

    # Global start indices for this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Tile coordinates
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Bounds checks
    i_mask = i < N
    j_mask = j < N

    # Base pointers for the current (b, h) slice
    left_base  = b * stride_l_b + h * stride_l_h
    right_base = b * stride_r_b + h * stride_r_h
    out_base   = b * stride_o_b + h * stride_o_h

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(0, num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # ------------------------------------------------------------------
        # Load left tile  [M, K] → left[i, k, h]
        # ------------------------------------------------------------------
        left_offsets = left_base \
                     + i[:, None] * stride_l_i \
                     + k[None, :] * stride_l_k
        left = tl.load(left_ptr + left_offsets,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # ------------------------------------------------------------------
        # Load right tile transposed  [K, N] → right_T[k, j, h] = right[j, k, h]
        # ------------------------------------------------------------------
        right_offsets = right_base \
                      + j[None, :] * stride_r_i \
                      + k[:, None] * stride_r_k
        right_T = tl.load(right_ptr + right_offsets,
                          mask=k_mask[:, None] & j_mask[None, :],
                          other=0.0)

        # ------------------------------------------------------------------
        # Accumulate dot‑product over K
        #   left : (M, K)  FP16
        #   right_T : (K, N) FP16
        #   acc : (M, N)  FP32
        # ------------------------------------------------------------------
        acc += tl.dot(left, right_T)

    # ----------------------------------------------------------------------
    # Store the result as FP16
    # ----------------------------------------------------------------------
    out_offsets = out_base \
                + i[:, None] * stride_o_i \
                + j[None, :] * stride_o_k
    tl.store(out_ptr + out_offsets, acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Wrapper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
    `left` and `right` must be half‑precision tensors of shape
    [B, N, N, H].
    Returns a half‑precision tensor of the same shape.
    """
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Tile sizes – tuned for H100 Tensor‑Core FP16 matmul
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides (as Python integers)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # Grid: (row tiles, col tiles, batch*head)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_k,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # enough for fp16 tensor cores on H100
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the Linear / LayerNorm weights
        - config_dict  : dict with at least ``dim`` and ``hidden_dim`` keys

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]                 # C in the paper
    hidden_dim = config["hidden_dim"]   # H

    # ------------------------------------------------------------------
    # Helper to move a weight to FP16 on the correct device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C] – still float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy path
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on H)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction – highly‑optimized Triton GEMM
    # ------------------------------------------------------------------
    out_hidden = tri_mul_triton(left, right)          # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    out_fp32 = out_hidden.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)        # back to fp16

    # ------------------------------------------------------------------
    # 8️⃣ Gate + final linear projection to original dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate          # [B, N, N, H]  fp16
    final = F.linear(gated, to_out_w)    # [B, N, N, C]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (the specification requires FP32 output)
    # ------------------------------------------------------------------
    return final.to(torch.float32)
