"""
TriMul (outgoing) forward kernel – highly optimized for H100.

Algorithm
---------
1. Layer‑norm the input (FP32) → cast to FP16.
2. Fuse the four linear projections (left_proj, right_proj,
   left_gate, right_gate) into a single mat‑mul by stacking their
   weights:   W = [W_left, W_right, W_left_gate, W_right_gate]  (4·H×D).
   The result (B,N,N,4·H) is sliced into the four tensors.
3. Apply the optional mask (broadcast over the hidden dimension) on
   the projected left/right tensors.
4. Fuse the element‑wise multiplication of the projected tensors with
   their sigmoid‑gates using a tiny Triton kernel (`apply_gates_kernel`).
5. Core N³ operation – batched matrix multiplication:
   for each (batch, head) pair compute    left @ rightᵀ .
   This is expressed as a single torch.bmm on tensors of shape
   (B·H, N, N) and (B·H, N, N)ᵀ, fully exploiting H100 Tensor‑Cores.
6. Post‑processing LayerNorm (FP16) → sigmoid out‑gate.
7. Fuse the out‑gate multiplication with a second Triton kernel
   (`elementwise_mul_kernel`).
8. Final linear projection back to `dim` and cast to FP32.

All heavy arithmetic runs in FP16, only the two LayerNorms stay in FP32
to preserve numerical stability.  Two small Triton kernels are used
to eliminate extra PyTorch launches and to keep the data on‑chip for
the element‑wise steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,       # fp16,   shape (total_elems,)
    left_gate_ptr,  # fp16,   same shape as left_ptr
    right_ptr,      # fp16,   shape (total_elems,)
    right_gate_ptr, # fp16,   same shape as right_ptr
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place Fuse the sigmoid gates:
        left  = left  * left_gate
        right = right * right_gate
    Tensors are flattened beforehand.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # left side    (l * g)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    # right side   (r * g)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # fp16 input (flattened)
    b_ptr,          # fp16 input (flattened)
    out_ptr,        # fp16 output (flattened)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    out = a * b   element‑wise multiplication of two flattened tensors.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (bool) – if True the mask
                         is ignored.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------- unpack -------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True → ignore the mask
    eps = 1e-5
    device = input_tensor.device

    # -------------------------- 1️⃣ Input LayerNorm -----------------------
    # FP32 LayerNorm (stable) → keep result in FP32 for now.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------------
    # Cast everything needed for the heavy arithmetic to FP16.
    # --------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # [B, N, N, dim]  fp16

    # -------------------------- 2️⃣ Stack linear weights ------------------
    #   left_proj  : (hidden_dim, dim)
    #   right_proj : (hidden_dim, dim)
    #   left_gate  : (hidden_dim, dim)
    #   right_gate : (hidden_dim, dim)
    # Stack them into a single weight matrix of shape (4*hidden_dim, dim).
    w_combined = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (4*H, D) fp16

    # -------------------------- 3️⃣ Fused projection ---------------------
    # This single matmul yields four tensors at once.
    # Result shape: [B, N, N, 4*hidden_dim]
    all_proj = F.linear(x_h, w_combined)         # fp16

    # Slice the combined tensor.
    left      = all_proj[..., 0:hidden_dim]                        # [B,N,N,H]
    right     = all_proj[..., hidden_dim:2 * hidden_dim]          # [B,N,N,H]
    left_gate = torch.sigmoid(all_proj[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(all_proj[..., 3 * hidden_dim:4 * hidden_dim])

    # --------------------------- 4️⃣ Optional mask ------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------- 5️⃣ Gate fusion -------------------------
    # Ensure contiguous tensors for the Triton kernel.
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536                     # 64 K elements per program (tuned)
    grid = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Gates are no longer needed.
    del left_gate, right_gate

    # --------------------------- 6️⃣ Core N³ (batched GEMM) --------------
    # Rearrange tensors to shape (B*H, N, N) → batch dimension = B·hidden_dim
    B, N, _, H = left.shape                      # H == hidden_dim
    total_heads = B * H

    # (B, H, N, N) → (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Batched matmul: left @ rightᵀ   →   [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout [B, N, N, hidden_dim].
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)    # (B, N, N, H)
        .contiguous()
    )  # fp16

    # --------------------------- 7️⃣ Post‑norm ---------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # --------------------------- 8️⃣ Output gate -------------------------
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    # Fuse the multiplication with a tiny Triton kernel.
    out_gate = out_gate.contiguous()
    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    # No longer need out_gate nor out.
    del out_gate, out

    # --------------------------- 9️⃣ Final projection -------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final_fp16 = F.linear(out_mul, w_to_out)   # [B, N, N, dim]  fp16

    # Return result in FP32 as required by the spec.
    return final_fp16.to(torch.float32)
