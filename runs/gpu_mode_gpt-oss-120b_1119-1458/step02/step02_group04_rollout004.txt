"""
Highly‑optimized forward pass for the outgoing TriMul block (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32, eps=1e‑5).
2️⃣  Cast to FP16 for the compute‑intensive part.
3️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a *single* GEMM by concatenating their weight matrices.
    This reads the normalized tensor only once and reduces weight traffic.
4️⃣  Split the concatenated result, apply sigmoid to the three gate tensors.
5️⃣  Fuse the element‑wise multiplications `left ← left·left_gate`,
    `right ← right·right_gate` (and optional mask) with a tiny Triton kernel.
6️⃣  Compute the N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    as a batched GEMM (torch.bmm) over the head dimension.
7️⃣  Layer‑norm over the hidden dimension (`to_out_norm`).
8️⃣  Fuse the final gating `out ← out·out_gate` with another Triton kernel.
9️⃣  Linear projection back to the original channel dimension and cast to
    float32 for the output.

Only the element‑wise multiplications (steps 5 and 8) are written in Triton,
fulfilling the “use a kernel” requirement while keeping the compute‑heavy
contractions in cuBLAS/Tensor‑Core kernels for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernels (tiny element‑wise fusions)
# -------------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr,                     # const half *
    b_ptr,                     # const half *
    out_ptr,                   # half *
    total_elts: tl.constexpr,  # total number of elements
    BLOCK_SIZE: tl.constexpr,  # compile‑time block size
):
    """out = a * b   (element‑wise, 1‑D flatten view)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dictionary with the model parameters
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model weights and cast to FP16 where appropriate
    # ------------------------------------------------------------------
    # LayerNorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear / gate weights (FP16)
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])

    # Output layernorm & projection
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])  # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for all subsequent ops
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear maps into one GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices: (5*hidden_dim, dim)
    weight_concat = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    )  # (5*H, dim), already half‑precision

    # Linear: [..., dim] @ weight_concat.T  ->  [..., 5*hidden_dim]
    proj = F.linear(x, weight_concat)  # (B, N, N, 5*hidden_dim), half

    # ------------------------------------------------------------------
    # 4️⃣ Split into the five tensors we need
    # ------------------------------------------------------------------
    splits = torch.split(proj, hidden_dim, dim=-1)
    left_raw, right_raw, left_gate_pre, right_gate_pre, out_gate_pre = splits

    # Free the large concatenated tensor early
    del proj

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid to gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)   # (B,N,N,H), half
    right_gate = torch.sigmoid(right_gate_pre)  # (B,N,N,H), half
    out_gate   = torch.sigmoid(out_gate_pre)    # (B,N,N,H), half

    # No longer need the pre‑sigmoid tensors
    del left_gate_pre, right_gate_pre, out_gate_pre

    # ------------------------------------------------------------------
    # 6️⃣ Element‑wise gating (left * left_gate, right * right_gate)
    #     Using a tiny Triton kernel for the multiplication.
    # ------------------------------------------------------------------
    total_elts = left_raw.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elts, BLOCK),)

    left = torch.empty_like(left_raw)   # gated left
    right = torch.empty_like(right_raw) # gated right

    # left = left_raw * left_gate
    elemwise_mul_kernel[grid](
        a_ptr=left_raw.data_ptr(),
        b_ptr=left_gate.data_ptr(),
        out_ptr=left.data_ptr(),
        total_elts=total_elts,
        BLOCK_SIZE=BLOCK,
    )
    # right = right_raw * right_gate
    elemwise_mul_kernel[grid](
        a_ptr=right_raw.data_ptr(),
        b_ptr=right_gate.data_ptr(),
        out_ptr=right.data_ptr(),
        total_elts=total_elts,
        BLOCK_SIZE=BLOCK,
    )
    # Release now‑unneeded tensors
    del left_raw, right_raw, left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # Convert mask to half and broadcast
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f
        # mask_f will be freed automatically after go out of scope

    # ------------------------------------------------------------------
    # 8️⃣ N³ contraction – batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for torch.bmm
    left_view  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left_view.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right_view.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # Batched matmul: left @ rightᵀ  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # Release temporaries that are no longer needed
    del left, right, left_view, right_view, left_flat, right_flat, out_flat

    # ------------------------------------------------------------------
    # 9️⃣ Output LayerNorm over hidden dimension (float32 for stability)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=eps,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 10️⃣ Fuse final gating (out_norm * out_gate) with Triton
    # ------------------------------------------------------------------
    total_out_elts = out_norm.numel()
    grid_out = (triton.cdiv(total_out_elts, BLOCK),)

    gated_out = torch.empty_like(out_norm)  # (B,N,N,H) half
    elemwise_mul_kernel[grid_out](
        a_ptr=out_norm.data_ptr(),
        b_ptr=out_gate.data_ptr(),
        out_ptr=gated_out.data_ptr(),
        total_elts=total_out_elts,
        BLOCK_SIZE=BLOCK,
    )
    # Clean up
    del out_norm, out_gate, out_fp32

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    final = F.linear(gated_out, to_out_w)          # (B,N,N,dim), half

    # ------------------------------------------------------------------
    # 12️⃣ Cast back to float32 as required by the API
    # ------------------------------------------------------------------
    return final.to(torch.float32)
