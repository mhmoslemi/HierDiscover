def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Input in float32
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    eps = 1e-5
    device = input_tensor.device

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps) # float32

    # cast to fp16 and flatten
    x_fp16 = x.to(torch.float16)
    B, N, _, _ = x.shape
    M = B * N * N
    x_fp16_flat = x_fp16.view(M, dim)  # [M, dim]

    # Concatenate weights
    w_left = weights["left_proj.weight"].to(torch.float16) # [hidden_dim, dim]
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Cat along out_features dimension:
    w_cat = torch.cat([w_left, w_right, w_left_gate, w_right_gate, w_out_gate], dim=0) # [5*hidden_dim, dim]
    # Linear projection
    all_proj = torch.nn.functional.linear(x_fp16_flat, w_cat) # shape (M, 5*hidden_dim)

    # Split
    split = torch.split(all_proj, hidden_dim, dim=1)
    left = split[0] # (M, hidden_dim)
    right = split[1]
    left_gate_pre = split[2]
    right_gate_pre = split[3]
    out_gate_pre = split[4]

    # gate sigmoids
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)  # (M, hidden_dim)

    # Apply mask and gates to left/right
    if not nomask:
        mask_fp16 = mask.to(torch.float16).view(M, 1)  # broadcast
        # Use triton kernel to fuse mask.mul gates * left/right
        # We'll write a kernel that loads left, right, left_gate, right_gate, mask and writes back left mask*gate*mask etc.
        # We'll allocate left and right in fp16, and then call the kernel.
        # but we already have left and right as tensors; we can launch kernel but need to ensure contiguous and correct shape.
        # We'll do:
        total_threads = M
        _apply_mask_gate_kernel[ (total_threads,) ](
            left, right, mask_fp16, left_gate, right_gate,
            left.stride(0), left.stride(1),
            right.stride(0), right.stride(1),
            mask_fp16.stride(0), mask_fp16.stride(1),
            left_gate.stride(0), left_gate.stride(1),
            right_gate.stride(0), right_gate.stride(1),
            M, hidden_dim, BLOCK_H=32, num_warps=4, num_stages=2
        )
        # The kernel will write left and right in-place.
    else:
        # elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # For the no mask case we still need left = left * left_gate, etc. Already done.

    # At this point left,right are (M, hidden_dim)
    # Reshape to original
    left = left.view(B, N, N, hidden_dim) # float16
    right = right.view(B, N, N, hidden_dim)

    # Transform for batched GEMM: (B, hidden_dim, N, N)
    left_mat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)

    # Compute batched GEMM
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # shape (B*hidden_dim, N, N)

    # Reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous() # fp16

    # Cast to float32
    out_fp32 = out.to(torch.float32)

    # Second layernorm
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply out_gate (need to reshape out_gate to [B,N,N,hidden_dim] )
    out_gate = out_gate.view(B,N,N,hidden_dim).to(torch.float32)
    out_norm_fp32 = out_norm_fp32 * out_gate

    # final linear projection
    out_final = F.linear(out_norm_fp32, weights["to_out.weight"])

    return out_final

@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    stride_l0, stride_l1,
    stride_r0, stride_r1,
    stride_m0, stride_m1,
    stride_lg0, stride_lg1,
    stride_rg0, stride_rg1,
    M, H,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    # each pid corresponds to a row index = pid
    if pid >= M:
        return
    # compute base offsets
    # left shape: (M, H) -> stride_l0 stride_l1 (row, col)
    # mask shape: (M, 1) but broadcast across H
    # left_gate shape: (M, H)
    # right_gate shape: same

    off_l = pid * stride_l0
    off_r = pid * stride_r0
    off_m = pid * stride_m0  # mask row offset
    off_lg = pid * stride_lg0
    off_rg = pid * stride_rg0

    # load mask scalar
    mask_val = tl.load(mask_ptr + off_m)  # scalar fp16

    # now loop over hidden dims
    for start in range(0, H, BLOCK_H):
        cur = start + tl.arange(0, BLOCK_H)
        mask = cur < H
        # load
        left = tl.load(left_ptr + off_l + cur * stride_l1, mask=mask)
        right = tl.load(right_ptr + off_r + cur * stride_r1, mask=mask)
        left_gate = tl.load(left_gate_ptr + off_lg + cur * stride_lg1, mask=mask)
        right_gate = tl.load(right_gate_ptr + off_rg + cur * stride_rg1, mask=mask)
        # compute
        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val
        # store
        tl.store(left_ptr + off_l + cur * stride_l1, left_out, mask=mask)
        tl.store(right_ptr + off_r + cur * stride_r1, right_out, mask=mask)

"""
Highly‑optimized forward pass for the *outgoing* Triangle Multiplicative Update
(TriMul) used in AlphaFold‑3.  The implementation fuses:

* 5 linear projections (left, right, left_gate, right_gate, out_gate) into a
  single low‑precision GEMM (FP16) by concatenating the weight matrices.
* Mask‑gate fusion for the left/right branches (executed by a small Triton
  kernel) when a pairwise mask is present.
* The core N³ operation Σₖ left[i,k,d]·right[j,k,d] is performed as a batch of
  independent half‑precision GEMMs (torch.bmm) which fully exploits Tensor‑Core
  performance on an H100.
* The remaining LayerNorms and the final projection are carried out in
  float32 for numerical stability.

The function `custom_kernel` follows the required signature:
    data = (input_tensor, mask, weights, config)
and returns the processed tensor of shape [B, N, N, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: fuse element‑wise mask and gate onto left/right tensors.
# --------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                 # fp16 * left  (M, H)
    right_ptr,                # fp16 * right (M, H)
    mask_ptr,                 # fp16 * mask  (M, 1)
    left_gate_ptr,            # fp16 * left_gate  (M, H)
    right_gate_ptr,           # fp16 * right_gate (M, H)
    stride_l0, stride_l1,     # strides for left
    stride_r0, stride_r1,     # strides for right
    stride_m0, stride_m1,     # strides for mask (m1 is unused)
    stride_lg0, stride_lg1,   # strides for left_gate
    stride_rg0, stride_rg1,   # strides for right_gate
    M, H,                     # total rows, hidden dim
    BLOCK_H: tl.constexpr,    # tile size along H
):
    pid = tl.program_id(0)               # one program per row (M rows)
    # Load the scalar mask for this row
    mask_val = tl.load(mask_ptr + pid * stride_m0)

    # Base offsets for the 2‑D tensors
    off_l = pid * stride_l0
    off_r = pid * stride_r0
    off_lg = pid * stride_lg0
    off_rg = pid * stride_rg0

    # Loop over the hidden dimension
    for start in range(0, H, BLOCK_H):
        cur = start + tl.arange(0, BLOCK_H)
        mask_h = cur < H

        left      = tl.load(left_ptr      + off_l  + cur * stride_l1,
                           mask=mask_h, other=tl.float16(0.0))
        right     = tl.load(right_ptr     + off_r  + cur * stride_r1,
                           mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_lg + cur * stride_lg1,
                           mask=mask_h, other=tl.float16(0.0))
        right_gate= tl.load(right_gate_ptr+ off_rg + cur * stride_rg1,
                           mask=mask_h, other=tl.float16(0.0))

        # fused: proj * gate * mask
        left_out  = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_ptr  + off_l  + cur * stride_l1, left_out,
                 mask=mask_h)
        tl.store(right_ptr + off_r  + cur * stride_r1, right_out,
                 mask=mask_h)


# --------------------------------------------------------------
# Entry point required by the harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:        [B, N, N]      (float32, optional)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ----------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ----------------------------------------------------------
    # 2. Cast to fp16 and flatten for a single GEMM that produces
    #    left, right, left_gate, right_gate and out_gate together.
    # ----------------------------------------------------------
    x_fp16 = x.to(torch.float16)                         # [B,N,N,dim] fp16
    B, N, _, _ = x_fp16.shape
    M = B * N * N                                          # rows for the fused GEMM
    x_flat = x_fp16.view(M, dim)                          # (M, dim)

    # Concatenate weight matrices (all are (hidden_dim, dim))
    w_left      = weights["left_proj.weight"].to(torch.float16)   # (h, dim)
    w_right     = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate= weights["right_gate.weight"].to(torch.float16)
    w_out_gate  = weights["out_gate.weight"].to(torch.float16)

    w_concat = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate], dim=0
    )  # shape (5*h, dim)

    # Single low‑precision GEMM: (M, dim) @ (dim, 5*h).T -> (M, 5*h)
    all_proj = torch.nn.functional.linear(x_flat, w_concat)     # fp16

    # ----------------------------------------------------------
    # 3. Split the concatenated output
    # ----------------------------------------------------------
    left, right, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        all_proj, hidden_dim, dim=1
    )  # each (M, hidden_dim)

    # Apply sigmoid to the gating vectors (still fp16)
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)          # (M, hidden_dim)

    # ----------------------------------------------------------
    # 4. Apply mask and gates to left/right
    #    * If a mask is present we fuse mask·gate·proj via Triton.
    #    * When no mask is used a simple element‑wise multiply suffices.
    # ----------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> (M, 1) and cast to fp16
        mask_fp16 = mask.to(torch.float16).contiguous().view(M, 1)

        # Ensure contiguous layout for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = M                           # one program per row
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left, right, mask_fp16,
            left_gate, right_gate,
            # strides (row, col) for each tensor
            left.stride(0),  left.stride(1),
            right.stride(0), right.stride(1),
            mask_fp16.stride(0), mask_fp16.stride(1),
            left_gate.stride(0), left_gate.stride(1),
            right_gate.stride(0), right_gate.stride(1),
            M, hidden_dim,
            BLOCK_H=32,          # tile size over hidden dim
            num_warps=4,
        )
        # after the kernel, `left` and `right` already contain
        #   proj * gate * mask
    else:
        # No mask → just multiply by gates
        left  = left * left_gate
        right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core N³ operation: batch GEMM over the hidden dimension
    # ----------------------------------------------------------
    # reshape to (B, N, N, hidden_dim)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # (B, hidden_dim, N, N) → (B*hidden_dim, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched half‑precision GEMM (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*hidden_dim, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ----------------------------------------------------------
    # 6. Second LayerNorm, out‑gate scaling and final projection
    # ----------------------------------------------------------
    out_fp32 = out.to(torch.float32)   # back to float32 for LayerNorm

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # reshape out_gate to (B, N, N, hidden_dim) and cast to fp32
    out_gate = out_gate.view(B, N, N, hidden_dim).to(torch.float32)

    out_norm = out_norm * out_gate

    # final linear projection to original feature dimension (float32)
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
