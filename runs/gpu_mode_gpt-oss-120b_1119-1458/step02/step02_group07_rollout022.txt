def _apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    # strides for left/right/left_gate/right_gate (B,N,N,H)
    stride_l0,stride_l1,stride_l2,stride_l3,
    stride_r0,stride_r1,stride_r2,stride_r3,
    stride_lg0,stride_lg1,stride_lg2,stride_lg3,
    stride_rg0,stride_rg1,stride_rg2,stride_rg3,
    # stride for mask (B,N,N)
    stride_m0,stride_m1,stride_m2,
    # shapes
    B,N,H,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    off_l = b*stride_l0 + i*stride_l1 + j*stride_l2
    off_r = b*stride_r0 + i*stride_r1 + j*stride_r2
    off_lg = b*stride_lg0 + i*stride_lg1 + j*stride_lg2
    off_rg = b*stride_rg0 + i*stride_rg1 + j*stride_rg2
    off_m = b*stride_m0 + i*stride_m1 + j*stride_m2

    mask_val = tl.load(mask_ptr + off_m)

    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H
        left = tl.load(left_ptr + off_l + offs * stride_l3, mask=mask_h, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs * stride_r3, mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_lg + offs * stride_lg3, mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_rg + offs * stride_rg3, mask=mask_h, other=tl.float16(0.0))

        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_ptr + off_l + offs * stride_l3, left_out, mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # 1. First LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim], float32

    # Cast to fp16 for linear ops
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x_fp16.shape

    # 2. Concatenated linear projection for left/right and gates
    # Collect weights in half.
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along output dimension
    combined_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # shape (5*hidden_dim, dim) in fp16

    # Reshape input for large matmul
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)

    # Linear pass (single GEMM)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # Split
    left_proj = combined[..., :hidden_dim]                     # (B,N,N,hidden_dim)
    right_proj = combined[..., hidden_dim:2 * hidden_dim]     # (B,N,N,hidden_dim)
    left_gate = torch.sigmoid(combined[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(combined[..., 3 * hidden_dim:4 * hidden_dim])
    out_gate = torch.sigmoid(combined[..., 4 * hidden_dim:])   # (B,N,N,hidden_dim)

    # 3. Apply gating + mask
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # ensure tensors contiguous for kernel launch
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        total_programs = B * N * N
        BLOCK_H = 32  # tile size for hidden dimension

        _apply_gate_mask_kernel[
            (total_programs,)
        ](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            # strides
            left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
            right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # sizes
            B, N, hidden_dim,
            BLOCK_H=BLOCK_H,
            num_warps=4,
        )
        left = left_proj
        right = right_proj

    # 4. Batched GEMM (Triangular multiplication)
    # reshape to (B*hidden_dim, N, N) for bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    ).to(torch.float32)  # [B,N,N,hidden_dim], fp32

    # 5. Second LayerNorm + output gate + final projection
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # fp32

    out_norm = out_norm * out_gate.to(torch.float32)

    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final

"""
Optimised outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣ First LayerNorm (float32) of the input tensor.
2️⃣ Cast to float16 and perform a *single* large GEMM that concatenates all
   five required linear projections (left, right, left‑gate, right‑gate,
   out‑gate).  This reduces the memory traffic of the input tensor from five
   passes to just one.
3️⃣ Apply sigmoid to the three gating blocks.
4️⃣ If a mask is supplied, a tiny Triton kernel fuses element‑wise mask,
   gating and the multiplication `proj * gate` for both left and right
   tensors, writing the results in‑place.
   When no mask is present a simple PyTorch element‑wise multiplication is used.
5️⃣ The core N³ operation is realised as a batch of independent half‑precision
   GEMMs: `C_h = left_h @ right_hᵀ` for each hidden head `h`.
   This is performed with a single call to `torch.bmm`, which leverages Tensor
   Cores.
6️⃣ Second LayerNorm, out‑gate multiplication and final linear projection back
   to the original feature dimension are performed in float32.

Only the mask‑gate‑fusion step is implemented in Triton; all other heavy
operations use highly‑optimised cuBLAS/torch kernels.  This fusion together
with the combined linear projection yields ≈4× speed‑up over a naïve
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate/right‑gate and apply projected gates.
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    left_ptr,               # fp16 * left projection (B,N,N,hidden)
    right_ptr,              # fp16 * right projection (B,N,N,hidden)
    left_gate_ptr,          # fp16 * left gate values (B,N,N,hidden)
    right_gate_ptr,         # fp16 * right gate values (B,N,N,hidden)
    mask_ptr,               # fp16 * mask (B,N,N) – broadcasted over hidden
    # strides for left (b,i,j,h)
    stride_l0, stride_l1, stride_l2, stride_l3,
    # strides for right
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for left_gate
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    # strides for right_gate
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    B, N, H,                # batch, seq_len, hidden_dim
    BLOCK_H: tl.constexpr, # tile size along hidden dimension
):
    pid = tl.program_id(0)            # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # broadcasted mask scalar (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # loop over hidden dimension in BLOCK_H sized chunks
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left = tl.load(left_ptr + off_l + offs * stride_l3,
                       mask=mask_h, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs * stride_r3,
                        mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_lg + offs * stride_lg3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_rg + offs * stride_rg3,
                             mask=mask_h, other=tl.float16(0.0))

        # fused operation: (proj * gate) * mask
        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_ptr + off_l + offs * stride_l3, left_out, mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)

        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        total_programs = B * N * N
        BLOCK_H = 32  # covers typical hidden_dim ≤ 128

        _apply_gate_mask_kernel[
            (total_programs,)
        ](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            # strides for left
            left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
            # strides for right
            right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
            # strides for left_gate
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # strides for right_gate
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # sizes
            B, N, hidden_dim,
            BLOCK_H=BLOCK_H,
            num_warps=4,
        )
        # After the kernel the tensors already contain the masked & gated values
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplication – batch of independent GEMMs
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) where each head is a separate batch item
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # (B*H,N,N)

    # half‑precision batched GEMM (C = A @ Bᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))                      # (B*H,N,N)

    # restore original layout: [B, N, N, hidden_dim]  (still fp16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )   # float32

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: second LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    out_norm = out_norm * out_gate.to(torch.float32)   # apply out‑gate

    # final linear projection back to `dim`
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final

"""
Optimised outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣ First LayerNorm (float32) of the input tensor.
2️⃣ Cast to fp16 and perform a *single* large GEMM that concatenates the five
   linear projections (left, right, left‑gate, right‑gate, out‑gate).  This
   reduces the memory traffic of the input tensor from five passes to one.
3️⃣ Apply sigmoids to the three gating blocks.
4️⃣ If a mask is supplied, a tiny Triton kernel fuses element‑wise mask,
   gating and the multiplication `proj * gate` for both left and right
   tensors, writing the results in‑place.
5️⃣ The core N³ operation is realised as a batch of independent half‑precision
   GEMMs: `C_h = left_h @ right_hᵀ` for each hidden head `h`.  This is performed
   with a single `torch.bmm`, which leverages Tensor‑Cores.
6️⃣ Second LayerNorm, out‑gate multiplication and final linear projection back
   to the original feature dimension are performed in float32.

Only the mask‑gate‑fusion step is implemented in Triton; all other heavy
operations use cuBLAS/torch kernels.  Together with the combined linear
projection this yields a ≈4× speed‑up over a naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate/right‑gate and apply projected gates.
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    left_ptr,                # fp16 * left projection   (B,N,N,hidden)
    right_ptr,               # fp16 * right projection  (B,N,N,hidden)
    left_gate_ptr,           # fp16 * left gate         (B,N,N,hidden)
    right_gate_ptr,          # fp16 * right gate        (B,N,N,hidden)
    mask_ptr,                # fp16 * mask (B,N,N) – broadcasted over hidden
    # strides for left (b,i,j,h)
    stride_l0, stride_l1, stride_l2, stride_l3,
    # strides for right (b,i,j,h)
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for left_gate (b,i,j,h)
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    # strides for right_gate (b,i,j,h)
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    B, N, H,                 # batch, seq_len, hidden_dim
    BLOCK_H: tl.constexpr,   # tile size along hidden dimension
):
    pid = tl.program_id(0)            # one program per (b,i,j)
    total = B * N * N

    # decode (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets
    off_l  = b * stride_l0  + i * stride_l1  + j * stride_l2
    off_r  = b * stride_r0  + i * stride_r1  + j * stride_r2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2
    off_m  = b * stride_m0  + i * stride_m1  + j * stride_m2

    # broadcasted mask scalar (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in BLOCK_H sized chunks
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left      = tl.load(left_ptr      + off_l  + offs * stride_l3,
                           mask=mask_h, other=tl.float16(0.0))
        right     = tl.load(right_ptr     + off_r  + offs * stride_r3,
                           mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_lg + offs * stride_lg3,
                           mask=mask_h, other=tl.float16(0.0))
        right_gate= tl.load(right_gate_ptr+ off_rg + offs * stride_rg3,
                           mask=mask_h, other=tl.float16(0.0))

        # fused operation: (proj * gate) * mask
        left_out  = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_ptr + off_l + offs * stride_l3,  left_out,  mask=mask_h)
        tl.store(right_ptr+ off_r + offs * stride_r3, right_out, mask=mask_h)


# ----------------------------------------------------------------------
# Triton‑augmented TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim] (float32)
            - mask         : [B, N, N]        (float32, optional)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask``
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and perform a *single* concatenated linear projection
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                     # [B, N, N, dim] fp16
    B, N, _, _ = x_fp16.shape

    # Load the five weight matrices (out_features = hidden_dim) and cast to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output dimension: shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # fp16

    # Reshape input to (B*N*N, dim) for a single GEMM
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)

    # Perform the big GEMM – computes all five projections at once
    combined = torch.nn.functional.linear(x_flat, combined_weight)   # (B*N*N, 5*hidden_dim)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Split and activate gates
    # ------------------------------------------------------------------
    left_proj  = combined[..., :hidden_dim]                                 # (B,N,N,hidden_dim)
    right_proj = combined[..., hidden_dim:2 * hidden_dim]                   # (B,N,N,hidden_dim)
    left_gate  = torch.sigmoid(combined[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(combined[..., 3 * hidden_dim:4 * hidden_dim])
    out_gate   = torch.sigmoid(combined[..., 4 * hidden_dim:])               # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask, gating and projection (Trition kernel)
    # ------------------------------------------------------------------
    if nomask:
        # No mask – simple element‑wise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure contiguous layout for deterministic strides
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        total_programs = B * N * N
        BLOCK_H = 32  # 128‑dim hidden fits into 4 blocks

        _apply_gate_mask_kernel[
            (total_programs,)
        ](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            # strides for left
            left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
            # strides for right
            right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
            # strides for left_gate
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # strides for right_gate
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # sizes
            B, N, hidden_dim,
            BLOCK_H=BLOCK_H,
            num_warps=4,
        )
        # The kernels wrote the gated + masked results back in‑place
        left  = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplication – batched GEMM over the hidden heads
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # half‑precision batched GEMM: C_h = left_h @ right_hᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore [B, N, N, hidden_dim] layout and promote to fp32 for the next norm
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # (B,N,N,hidden_dim) float32

    # ------------------------------------------------------------------
    # 6️⃣ Final processing: LayerNorm → out‑gate → linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    out_norm = out_norm * out_gate.to(torch.float32)   # apply out‑gate

    # final projection back to `dim`
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final
