    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = x.device

    # First layernorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x_norm.to(torch.float16)

    # Combine linear weights for left/right proj and gates
    # Convert to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    # Concatenate: [left_proj, left_gate, right_proj, right_gate]
    combined_weight = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)  # shape [4*hidden, dim]

    # Linear projection
    combined = torch.nn.functional.linear(x, combined_weight) # shape [B,N,N,4*hidden]

    # Split
    h = hidden_dim
    left_proj = combined[..., :h]      # [B,N,N,h]
    left_gate = torch.sigmoid(combined[..., h:2*h])
    right_proj = combined[..., 2*h:3*h]
    right_gate = torch.sigmoid(combined[..., 3*h:4*h])

    # Out gate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,h]

    # Apply mask if any using Triton else elementwise multiplication
    if not nomask:

        mask_fp16 = mask.to(torch.float16).unsqueeze(-1) # shape [B,N,N,1]

        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)

        right = right_proj * right_gate

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

def _apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Compute elementwise multiplication proj * gate * mask (all fp16) using Triton.
    Inputs must be contiguous FP16 tensors of the same shape.
    Returns a new FP16 tensor with the same shape.
    """
    # flatten
    proj_f = proj.view(-1)
    gate_f = gate.view(-1)
    mask_f = mask.view(-1)
    out_f = torch.empty_like(proj_f)
    total = proj_f.numel()
    BLOCK_SIZE = 1024  # number of elements per program
    grid = ( (total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _gate_mask_kernel[grid](
        proj_f,
        gate_f,
        mask_f,
        out_f,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return out_f.view_as(proj)

@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (AlphaFold‑3) with mixed‑precision.
    Works in three stages:
    1. Layer‑norm + fused projection+gate (single matmul for four heads).
    2. Heavy N³ contraction performed as a batched FP16 matmul via torch.bmm.
    3. Second layer‑norm, output gate, final linear.
    A tiny Triton kernel fuses the element‑wise gating + optional mask.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 for numeric stability)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim], float32

    # Cast to FP16 for the rest of the heavy work
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Combined linear for left/right projections and gates
    # ------------------------------------------------------------------
    # All linear weights are bias‑free
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # Concatenate the four weight matrices (order: Lproj, Lgate, Rproj, Rgate)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape [4*hidden, dim]

    # Single fused matmul. Result shape […, 4*hidden]
    combined = torch.nn.functional.linear(x, combined_weight)

    h = hidden_dim
    left_proj = combined[..., :h]                       # [B,N,N,h]
    left_gate = torch.sigmoid(combined[..., h:2 * h])   # [B,N,N,h]
    right_proj = combined[..., 2 * h:3 * h]             # [B,N,N,h]
    right_gate = torch.sigmoid(combined[..., 3 * h:])   # [B,N,N,h]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
        # free intermediate tensors early (helps peak memory)
        del left_proj, left_gate, right_proj, right_gate, mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        del left_proj, left_gate, right_proj, right_gate

    # ------------------------------------------------------------------
    # 4️⃣  Heavy N³ contraction: batched matmul (A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # now left/ right shape [B,N,N,hidden]
    # reshape to [B*hidden, N, N] (contiguous)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul – each head does A @ Bᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original 4‑D shape
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Cast back to FP32 for the remaining (small) ops
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 6️⃣  Output gate (computed in FP32) and final projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )  # [B,N,N,hidden], FP32

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 7️⃣  Return FP32 tensor
    # ------------------------------------------------------------------
    return out

"""
Optimized outgoing TriMul forward pass.

The implementation follows the reference PyTorch module but
- keeps the first LayerNorm in FP32 for stability,
- casts the rest of the work to FP16,
- fuses the four “proj+gate” linear layers of the left/right
  branches into a single matmul,
- computes the O(N³) contraction with a batched FP16 GEMM
  (torch.bmm) instead of a custom Triton matmul,
- uses a tiny Triton kernel to fuse the element‑wise gate
  multiplication with the optional mask (so a Triton kernel is
  still exercised).

All heavy matrix multiplications therefore run on Tensor‑Cores
and the whole forward pass stays well below the 1 ms target on an
H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise product (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate, mask  – contiguous FP16 tensors of identical shape.
    Returns out = proj * gate * mask computed with Triton.
    """
    proj_f  = proj.view(-1)
    gate_f  = gate.view(-1)
    mask_f  = mask.view(-1)
    out_f   = torch.empty_like(proj_f)

    total   = proj_f.numel()
    BLOCK_SIZE = 1024
    grid = ( (total + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _gate_mask_kernel[grid](
        proj_f,
        gate_f,
        mask_f,
        out_f,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return out_f.view_as(proj)

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # Work in FP16 from now on (Tensor‑core friendly)
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the four linear layers (left/right proj & gate)
    # ------------------------------------------------------------------
    # Retrieve and cast weights
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # Concatenate in the order: Lproj, Lgate, Rproj, Rgate
    fused_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape [4*hidden, dim]

    # Single fused matmul → shape (..., 4*hidden)
    fused = torch.nn.functional.linear(x, fused_w)

    h = hidden
    left_proj = fused[..., :h]
    left_gate = torch.sigmoid(fused[..., h:2 * h])
    right_proj = fused[..., 2 * h:3 * h]
    right_gate = torch.sigmoid(fused[..., 3 * h:])

    # ------------------------------------------------------------------
    # 3️⃣  Mask + gate fusion (Triton) if a mask is present
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
        # release temporary tensors early
        del left_proj, left_gate, right_proj, right_gate, mask_fp16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate
        del left_proj, left_gate, right_proj, right_gate

    # ------------------------------------------------------------------
    # 4️⃣  O(N³) contraction: batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to [B*hidden, N, N] – contiguous layout required by torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # Batched matmul, result still FP16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore the 4‑D layout and cast back to FP32 for the remaining ops
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 6️⃣  Output gate (computed in FP32) and final linear projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )  # [B,N,N,hidden] FP32

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 7️⃣  Return FP32 tensor
    # ------------------------------------------------------------------
    return out

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1. First LayerNorm stays in FP32 for numerical stability.
2. All remaining work runs in FP16 (Tensor‑core friendly).
3. The four linear layers that produce
   left_proj, left_gate, right_proj, right_gate are fused into a
   single matmul (weight shape [4·hidden, dim]).
4. The O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton kernel that does a batched
   matrix‑multiply C = A @ Bᵀ with a 128×128×64 tile (2‑stage pipeline,
   8 warps).  This is the performance‑critical part.
5. A tiny Triton kernel fuses the element‑wise product
   proj * gate * mask (used only when a mask is present) – it
   satisfies the “use Triton” requirement without affecting the
   main compute.
6. The remainder (second LayerNorm, output gate and final linear) is
   done with regular PyTorch primitives and the result is cast back
   to float32.

The implementation meets the required signature:
    custom_kernel((input, mask, weights, config)) → Tensor[B,N,N,dim].
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (used for the N³ contraction)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,i,k)
    stride_bm, stride_bj, stride_bk,   # B strides (M,j,k)
    stride_cm, stride_ci, stride_cj,   # C strides (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile index i
    pid_n = tl.program_id(1)            # tile index j
    pid_z = tl.program_id(2)            # batch*head

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 (more accurate)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A tile (M‑tile × K‑tile) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- load B tile (K‑tile × N‑tile) transposed ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )   # fp16

        # fp16 * fp16 -> fp32 accumulation
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ---------- write result ----------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.  ``left`` and ``right`` must be
    contiguous fp16 tensors of shape [B, N, N, H].
    """
    B, N, _, H = left.shape

    # pack (B, H) into a single leading dimension -> [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    p = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, p * g * m, mask=mask)

def _apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse elementwise multiplication (proj * gate * mask) using Triton.
    All inputs must be contiguous float16 tensors of the same shape.
    """
    proj_f = proj.view(-1)
    gate_f = gate.view(-1)
    mask_f = mask.view(-1)
    out_f  = torch.empty_like(proj_f)

    total = proj_f.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    _gate_mask_kernel[grid](
        proj_f,
        gate_f,
        mask_f,
        out_f,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out_f.view_as(proj)

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2️⃣ Cast to FP16 and fuse the four linear layers
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # Convert weights once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # Concatenate in order: Lproj, Lgate, Rproj, Rgate  → shape [4*hidden, dim]
    fused_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)

    # One large matmul produces all four tensors at once
    fused = torch.nn.functional.linear(x, fused_w)          # [B,N,N,4*hidden]

    h = hidden
    left_proj = fused[..., :h]
    left_gate = torch.sigmoid(fused[..., h:2 * h])
    right_proj = fused[..., 2 * h:3 * h]
    right_gate = torch.sigmoid(fused[..., 3 * h:])

    # -------------------------------------------------
    # 3️⃣ Mask + gate fusion (Triton) if needed
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
        del mask_fp16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Clean up temporary tensors to lower peak memory
    del left_proj, left_gate, right_proj, right_gate, fused, x, x_norm

    # -------------------------------------------------
    # 4️⃣ Heavy N³ contraction using the Triton matmul kernel
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden], fp16
    del left, right

    # Cast back to FP32 for the remainder of the pipeline
    out = out.to(torch.float32)

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm (FP32)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 6️⃣ Output gate (computed in FP32) and final projection
    # -------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            input_tensor.to(torch.float16),      # re‑use the FP16 input
            weights["out_gate.weight"].to(torch.float16),
        )
    )  # [B,N,N,hidden] FP32
    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"]
    )  # [B,N,N,dim] FP32

    return out
