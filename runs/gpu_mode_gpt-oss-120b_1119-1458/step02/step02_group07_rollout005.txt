"""
Optimized Triton‑augmented implementation of the *outgoing* Triangle
Multiplicative Update (TriMul) from AlphaFold‑3.

Algorithm
---------
1️⃣ Input Layer‑Norm (float32) – unchanged, using the highly‑optimised
   PyTorch kernel.

2️⃣ FP16 casting and **fused** linear projections:
   * Concatenate left‑ and right‑projection weights → a single matmul.
   * Concatenate the three gating weight matrices (left‑gate, right‑gate,
     out‑gate) → a single matmul followed by a sigmoid.
   This halves the number of memory‑intensive GEMM passes.

3️⃣ Fuse mask, left‑gate and right‑gate into the projected tensors with a
   tiny Triton kernel (`_mask_gate_fuse_kernel`).  The kernel works on
   the hidden dimension in 32‑element tiles, broadcasting the scalar mask
   to all channels and writing the results back in‑place.

4️⃣ Core N³ multiplication:
   The hidden dimension is independent → we reshape the tensors to
   (B·H, N, N) and invoke a single batched half‑precision GEMM
   (`torch.bmm`).  Tensor‑cores on an H100 execute this step in ~µs.

5️⃣ Post‑processing:
   * Layer‑Norm over the hidden dimension (float32).
   * Multiply by the *out*‑gate (sigmoid‑activated).
   * Final linear projection back to the original feature dimension.
   The final projection is also performed in FP16 and cast back to FP32.

The only custom Triton kernel is the mask‑gate‑fusion; all heavy linear
algebra is delegated to cuBLAS/Tensor‑Cores, which gives the best
throughput on modern NVIDIA GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                  # fp16 * left projection (B,N,N,H)
    right_ptr,                 # fp16 * right projection (B,N,N,H)
    mask_ptr,                  # fp16 * mask (B,N,N)
    left_gate_ptr,             # fp16 * left gate (B,N,N,H)
    right_gate_ptr,            # fp16 * right gate (B,N,N,H)
    # strides for left (b,i,j,h)
    stride_l0, stride_l1, stride_l2, stride_l3,
    # strides for right (b,i,j,h)
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    # strides for left‑gate (b,i,j,h)
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    # strides for right‑gate (b,i,j,h)
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    B, N, H,                   # batch, seq_len, hidden_dim
    BLOCK_H: tl.constexpr,    # tile size along hidden dimension
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2

    # broadcasted mask scalar (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in BLOCK_H‑sized chunks
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)                 # (BLOCK_H,)
        mask_h = offs < H

        # load projection tensors (vectorised across hidden dim)
        left   = tl.load(left_ptr  + off_l  + offs * stride_l3,
                         mask=mask_h, other=tl.float16(0.0))
        right  = tl.load(right_ptr + off_r  + offs * stride_r3,
                         mask=mask_h, other=tl.float16(0.0))
        left_g = tl.load(left_gate_ptr  + off_lg + offs * stride_lg3,
                         mask=mask_h, other=tl.float16(0.0))
        right_g = tl.load(right_gate_ptr + off_rg + offs * stride_rg3,
                         mask=mask_h, other=tl.float16(0.0))

        # fused operation:  proj * gate * mask
        left_out  = left * left_g * mask_val
        right_out = right * right_g * mask_val

        # store back (in‑place)
        tl.store(left_ptr  + off_l  + offs * stride_l3, left_out,  mask=mask_h)
        tl.store(right_ptr + off_r  + offs * stride_r3, right_out, mask=mask_h)

            # strides for right

            # strides for mask

            # strides for left_gate

            # strides for right_gate

            # sizes
            input_tensor.shape[0],          # B
            input_tensor.shape[1],          # N
            hidden_dim,                     # H
            BLOCK_H=32,                     # tile size for the hidden dimension
            num_warps=4,
        )
        # After the kernel, `left` and `right` already contain mask * gate * proj.
    else:
        # No mask → simple element‑wise multiplication with the two gates.
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM over the hidden dimension:
    #     For each head d, compute C_d = left_d @ right_dᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) where each hidden channel is an independent batch entry
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # fp16 GEMM – Tensor Cores perform fp32 accumulation internally
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    # second LayerNorm over the hidden dimension
    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # multiply by the out‑gate (cast gate to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim) – float32
    out_final = F.linear(out_norm, to_out_w)  # [B, N, N, dim] float32

    return out_final
```
