```
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0,BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    
    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
        l = l * m
        r = r * m

    tl.store(left_ptr + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
        left = left * m
        right = right * m

    tl.store(left_ptr + offs, left * left_gate, mask=mask)
    tl.store(right_ptr + offs, right * right_gate, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated outgoing Triangle Multiplicative (TriMul) update.

    The heavy N³ reduction ``einsum('...ikd,...jkd->...ijd')`` is performed
    by an optimized batched GEMM (`torch.einsum`/`torch.bmm`).  Per‑element
    masking and gating are fused into a tiny Triton kernel to avoid extra
    memory passes.  All linear layers and the final LayerNorm are run in
    half‑precision (FP16) on H100 Tensor Cores; the input LayerNorm stays
    in FP32 for numerical stability.  The function returns a float‑32
    tensor of shape ``[B, N, N, dim]``.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)  FP32

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for all downstream ops
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Cast model weights to FP16
    # -----------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections (bias‑free) and gate pre‑activations
    # -----------------------------------------------------------------
    left = F.linear(x_h, w_left_proj)          # (B, N, N, H)  FP16
    right = F.linear(x_h, w_right_proj)        # same

    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    # -----------------------------------------------------------------
    # 5. Fuse mask (if any) and gating using a Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # (B, N, N, 1)
        mask_ptr = mask_h.view(-1)
    else:
        # dummy scalar mask (value “1.0”) – never read when HAS_MASK=0
        mask_ptr = torch.tensor(1., dtype=torch.float16, device=device)

    left = left.view(-1)
    right = right.view(-1)
    left_gate = left_gate.view(-1)
    right_gate = right_gate.view(-1)

    numel = left.numel()
    BLOCK = 65536  # 64 Ki elements per program – fits H100 L2 cache well
    grid = ( (numel + BLOCK - 1) // BLOCK, )

    apply_gates_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        mask_ptr,
        numel,
        BLOCK=BLOCK,
        HAS_MASK=int(not nomask),
    )

    # reshape tensors back to 4‑D
    B, N, _, _ = input_tensor.shape  # dim of original input
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 6. Core triple‑multiplicative update: einsum over the sequence axis
    #    This is mathematically equivalent to a batched GEMM with shape
    #    (B*H, N, N) @ (B*H, N, N)ᵀ.
    # -----------------------------------------------------------------
    # Using torch.einsum; internally it falls back to a strided‑batched
    # GEMM with FP16 Tensor‑Cores on H100.
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # -----------------------------------------------------------------
    # 7. Post‑projection LayerNorm (FP16)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # (B, N, N, H)  FP16

    # -----------------------------------------------------------------
    # 8. Apply output gate (in‑place) and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate  # element‑wise FP16 multiply, in‑place style works
    final = F.linear(out, w_to_out)  # (B, N, N, dim)  FP16

    # -----------------------------------------------------------------
    # 9. Cast back to FP32 (the API expects float32)
    # -----------------------------------------------------------------
    return final.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def apply_gates_kernel(
    left_ptr,
    left_gate_ptr,
    right_ptr,
    right_gate_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)

    tl.store(left_ptr + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)

        numel,
        BLOCK=BLOCK,

        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )

    out = out * out_gate

    # Final linear projection
    final = F.linear(out, w_to_out)

    # Cast to FP32 output
    return final.to(torch.float32)

"""
Optimized outgoing Triangle‑Multiplicative update (TriMul).

The N³ reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is carried out with a batched GEMM (FP16 Tensor‑Cores) after flattening the
hidden dimension into the batch axis.  Element‑wise masking and the two
gates are fused in a tiny Triton kernel (≈1 µs launch) to avoid extra
memory passes.  All linear layers and the post‑norm are performed in FP16;
the input‑norm stays in FP32 for numerical stability.  The function returns
a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny Triton kernel: fuse per‑head gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,        # fp16, flattened left  tensor  (B*N*N*H,)
    left_gate_ptr,   # fp16, flattened left‑gate tensor (same shape)
    right_ptr,       # fp16, flattened right tensor (same shape)
    right_gate_ptr,  # fp16, flattened right‑gate tensor (same shape)
    numel: tl.int64,        # total number of elements
    BLOCK: tl.constexpr,    # block size (elements processed per program)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # load
    l  = tl.load(left_ptr + offs,        mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,   mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,       mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs,  mask=mask, other=0.0)

    # write gated values back in‑place
    tl.store(left_ptr  + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask          : torch.Tensor      [B, N, N]    (bool/float)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   must contain ``dim`` and ``hidden_dim``,
                          optional ``nomask`` (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack & precision handling
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 2️⃣ input LayerNorm (float32 – stable)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # --------------------------------------------------------------
    # 3️⃣ cast everything to FP16 for the bulk of the work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ cast model weights to FP16
    # --------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 5️⃣ linear projections (bias‑free) and gate pre‑activations
    # --------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)    # (B, N, N, H)   fp16
    right = F.linear(x_h, w_right_proj)   # same shape

    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # --------------------------------------------------------------
    # 6️⃣ optional mask (broadcast over the hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)       # (B, N, N, 1)
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 7️⃣ fuse left/right with their gates (single Triton kernel)
    # --------------------------------------------------------------
    # flatten tensors – they are already contiguous after the linear ops
    left_flat      = left.view(-1)
    right_flat     = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    numel = left_flat.numel()
    BLOCK = 65536                                   # ~128 KB per program
    grid = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid](
        left_flat,
        left_gate_flat,
        right_flat,
        right_gate_flat,
        numel,
        BLOCK=BLOCK,
    )
    # recover the 4‑D layout
    B, N, _, _ = input_tensor.shape                # (B, N, N, dim)
    left  = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 8️⃣ core contraction:   Σₖ left[..., i, k, d] * right[..., j, k, d]
    #    using a batched GEMM (FP16 Tensor‑Cores)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) -> batch GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication (equivalent to the einsum)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # re‑assemble shape [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # --------------------------------------------------------------
    # 9️⃣ post‑norm (FP16) and output‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    out = out * out_gate                     # in‑place style element‑wise mul

    # --------------------------------------------------------------
    # 🔟 final linear projection back to `dim` (FP16 → FP16)
    # --------------------------------------------------------------
    final = F.linear(out, w_to_out)          # (B, N, N, dim)  fp16

    # --------------------------------------------------------------
    # 11️⃣ cast back to the API‑expected float32
    # --------------------------------------------------------------
    return final.to(torch.float32)
