def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, N_, dim = input_tensor.shape

    # default for mask, maybe None
    nomask = config.get("nomask", True)
    hidden_dim = config["hidden_dim"]
    # LN
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)
    # Cast to half
    x_fp16 = x.to(torch.float16)
    # Flatten for linear
    x_flat = x_fp16.view(B * N * N, dim)
    # Prepare left cat weight
    left_cat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_cat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    left_cat = F.linear(x_flat, left_cat_weight)  # shape (B*N*N, 2*hidden_dim)
    # split
    left_proj_flat = left_cat[:, :hidden_dim]
    left_gate_pre_flat = left_cat[:, hidden_dim:]
    left_gate = torch.sigmoid(left_gate_pre_flat)
    left_proj = left_proj_flat.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)

    # right similarly
    right_cat = F.linear(x_flat, right_cat_weight)
    right_proj_flat = right_cat[:, :hidden_dim]
    right_gate_pre_flat = right_cat[:, hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre_flat)
    right_proj = right_proj_flat.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)

    # out_gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight)).view(B,N,N,hidden_dim)
    # apply mask
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # convert mask to half
        mask_fp16 = mask.to(torch.float16).contiguous()
        # ensure contiguous for left and right and gates
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        # total programs
        total = B * N * N
        _mask_gate_fuse_kernel[(total,)](
            left, right, mask_fp16, left_gate, right_gate,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),

            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            B, N, hidden_dim,
            BLOCK_H=32,
            num_warps=4,
        )
        # After kernel, left and right have masked and gated values
    # BMM
    # keep shape [B,N,N,hidden_dim]
    # unify batch dimension
    # left_mat shape (B*hidden_dim, N, N)
    B, N, _, hidden_dim = left.shape
    left_mat = left.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # cast

    # final linear
    out_final = F.linear(out_norm, weights["to_out.weight"])
    return out_final

import torch, torch.nn.functional as F
import numpy as np

class TriMulRef(nn.Module):
    def __init__(self, dim, hidden_dim):
        super().__init__()
        self.norm = nn.LayerNorm(dim)
        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)
    def forward(self,x,mask):
        x = self.norm(x)
        left = self.left_proj(x)
        right = self.right_proj(x)
        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask
        left_gate = torch.sigmoid(self.left_gate(x))
        right_gate = torch.sigmoid(self.right_gate(x))
        out_gate = torch.sigmoid(self.out_gate(x))
        left = left * left_gate
        right = right * right_gate
        out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)
        out = self.to_out_norm(out)
        out = out * out_gate
        out = self.to_out(out)
        return out

"""
custom_kernel  –  Outgoing Triangle Multiplicative Update (TriMul)

Algorithm (all tensors are on the same CUDA device)
-----------------------------------------------------------------
1️⃣ Layer‑norm the 4‑D input (float32).
2️⃣ Cast to fp16 and flatten to (B·N·N, dim) for fast GEMM.
3️⃣ Fuse the left/right linear projections **and** the corresponding
   gating linear layers by concatenating the two weight matrices
   (proj + gate) → one GEMM per side.
   Sigmoid is applied to the gate part afterwards.
4️⃣ If a mask is present, launch a tiny Triton kernel that fuses
   *mask × gate × projection* for both left and right tensors,
   writing the result back in‑place.
   If no mask, a simple element‑wise multiplication is used.
5️⃣ The core N³ operation
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   is carried out as a batch of independent GEMMs:
        C_{b·d} = left_{b·d} @ right_{b·d}ᵀ
   using `torch.bmm` (Tensor‑Core accelerated fp16 MatMul).
6️⃣ Convert the result to float32, apply a second LayerNorm,
   multiply by the output‑gate (sigmoid of a linear projection) and
   a final linear projection back to the original feature dimension.
   The final tensor shape is `[B, N, N, dim]` (float32).

The Triton kernel only touches the mask‑gate fusion step, which is
memory‑bound and runs in a few microseconds.  All heavy linear algebra
leverages cuBLAS/Tensor‑Cores for maximal throughput on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 * left projection      [B,N,N,H]
    right_ptr,              # fp16 * right projection     [B,N,N,H]
    mask_ptr,               # fp16 * mask                 [B,N,N]
    left_gate_ptr,          # fp16 * left gate            [B,N,N,H]
    right_gate_ptr,         # fp16 * right gate           [B,N,N,H]
    # strides for the 4‑D tensors
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    # strides for the gate tensors (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    # problem size
    B, N, H,
    BLOCK_H: tl.constexpr,   # tile size on the hidden dimension
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base pointers for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # broadcast scalar mask element (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in BLOCK_H sized chunks
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        # load (vectorised across hidden dim)
        left      = tl.load(left_ptr      + off_l + offs * stride_l3,
                           mask=mask_h, other=tl.float16(0.0))
        right     = tl.load(right_ptr     + off_r + offs * stride_r3,
                           mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                           mask=mask_h, other=tl.float16(0.0))
        right_gate= tl.load(right_gate_ptr+ off_g + offs * stride_g3,
                           mask=mask_h, other=tl.float16(0.0))

        # fused operation:  proj * gate * mask
        left_out  = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        # store back in‑place
        tl.store(left_ptr  + off_l + offs * stride_l3, left_out,
                 mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out,
                 mask=mask_h)


# ----------------------------------------------------------------------
# Entry point – the harness calls this function
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation.

    Args:
        data: (input, mask, weights, config)
            - input : torch.Tensor[B, N, N, dim]   (float32)
            - mask  : torch.Tensor[B, N, N]      (float32, optional)
            - weights: dict of model parameters (float32)
                Expected keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with "dim", "hidden_dim", "nomask" (bool, default True)

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and flatten for the fused linear projections
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    B, N, _, _ = x_fp16.shape
    x_flat = x_fp16.reshape(B * N * N, dim)   # (B·N·N, dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fused left/right projection + gating (single GEMM per side)
    # ------------------------------------------------------------------
    # concatenate proj + gate weights for the left side
    left_cat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, dim)

    # same for the right side
    right_cat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, dim)

    # Linear → (B·N·N, 2*H)
    left_cat = F.linear(x_flat, left_cat_weight)          # fp16
    right_cat = F.linear(x_flat, right_cat_weight)

    # split into proj / gate and reshape back to 4‑D
    left_proj = left_cat[:, :hidden_dim].reshape(B, N, N, hidden_dim)
    left_gate_pre = left_cat[:, hidden_dim:].reshape(B, N, N, hidden_dim)
    right_proj = right_cat[:, :hidden_dim].reshape(B, N, N, hidden_dim)
    right_gate_pre = right_cat[:, hidden_dim:].reshape(B, N, N, hidden_dim)

    # sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Output gate (separate linear)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight)).reshape(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask & element‑wise gating
    # ------------------------------------------------------------------
    if nomask or mask is None:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure all tensors are contiguous for correct stride handling
        mask_fp16 = mask.to(torch.float16).contiguous()
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = B * N * N  # one Triton program per (b,i,j)
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left_proj,
            right_proj,
            mask_fp16,
            left_gate,
            right_gate,
            # strides of left/right tensors
            left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
            right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gates (same layout as left/right)
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # problem size
            B, N, hidden_dim,
            BLOCK_H=32,          # 32 works well for hidden_dim ≤ 384
            num_warps=4,
        )
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM (core N³ operation)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) required by torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # C = A @ Bᵀ  (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original 4‑D layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing (LayerNorm → out‑gate → final projection)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B, N, N, dim] float32

    return out_final
