def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,

    pid_m = tl.program_id(0)

"""
Implementation of the outgoing TriMul operator using Triton for the
N^3 pairwise contraction. The kernel computes a batched matrix
multiplication for each (batch, head) pair:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where left/right are the projected and gated pairwise features.
The rest of the module (layer‑norms, linear projections, gating)
are performed with torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_b = tl.program_id(2)  # batch index (combined batch*head)
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column

    # compute offsets within the matrices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate in fp32 for numerical quality
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # load b tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_b * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product: accumulator updated in fp32
        # a and b are fp16, but tl.dot automatically does fp32 accumulation on Ampere+ tensor cores.
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def batched_gemm_triton(a: torch.Tensor, b: torch.Tensor, out: torch.Tensor,
                        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    """
    Batched GEMM kernel using Triton.
    a: (B, M, K)    fp16
    b: (B, K, N)    fp16 (already transposed for the contraction)
    out: (B, M, N)  fp16
    """
    assert a.is_contiguous() and b.is_contiguous() and out.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Incompatible inner dimensions"
    grid = (
        (tl.cdiv(M, BLOCK_M), tl.cdiv(N, BLOCK_N), batch)
    )
    batched_gemm_kernel[grid](
        a, b, out,
        batch, M, N, K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        * mask_tensor   : torch.Tensor of shape [B, N, N] (bool/float) or None
        * weights_dict  : dictionary with all linear / layer‑norm weights
        * config_dict   : contains ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move weights to the target dtype/device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load weights (fp32 for layernorm, fp16 for linear ops)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 → fp16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C] – float32

    # Convert to half‑precision for the heavy steps.
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float, ensure fp16 and broadcast
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid‑shaped linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction – batched GEMM implemented in Triton
    # ------------------------------------------------------------------
    # Move hidden dimension to the batch axis: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right so that we can do a regular matmul.
    right_T = right_perm.transpose(-2, -1).contiguous()

    # Collapse (B, H) → B*H for the kernel.
    B_H = B * hidden_dim
    left_flat  = left_perm.view(B_H, N, N)
    right_flat = right_T.view(B_H, N, N)
    out_flat   = torch.empty_like(left_flat)  # fp16, shape (B*H, N, N)

    # Run the Triton kernel
    batched_gemm_triton(left_flat, right_flat, out_flat,
                        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32 ↔ fp16 conversion around it)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7️⃣ Gated projection back to the original channel dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate          # shape (B, N, N, hidden_dim)
    final = F.linear(gated, to_out_w)    # shape (B, N, N, dim) – fp16

    # ------------------------------------------------------------------

"""
Outgoing TriMul (AlphaFold3) – Triton implementation.

The heavy N³ contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is turned into a batched GEMM (B·H independent matrix multiplies)
that is executed with a custom Triton kernel.  All other
operations (layer‑norms, linear projections, gating) are kept in
PyTorch but the five linear maps are fused into a single
matrix‑multiply to reduce memory traffic.  The kernel works on
FP16 tensors, accumulates in FP32 and stores the result back in FP16.
The final output is returned in FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers to the three tensors
    batch, M, N, K,                     # logical dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B  where
        A : (batch, M, K)   (row‑major, fp16)
        B : (batch, K, N)   (row‑major, fp16)   – already transposed
        C : (batch, M, N)   (fp16, written out)
    The kernel is tiled with blocks (BLOCK_M, BLOCK_N, BLOCK_K) and
    accumulates in FP32 for numerical quality.
    """
    pid_b = tl.program_id(2)                # batch index (combined B·H)
    pid_m = tl.program_id(0)                # block‑row index
    pid_n = tl.program_id(1)                # block‑col index

    # Offsets for the (M,N) tile handled by this program
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_b * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_b * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # fp16×fp16 → fp32 accumulation (Tensor‑Core on Ampere+)
        acc += tl.dot(a, b)

    # Write the result back to C
    c_ptrs = c_ptr + pid_b * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_triton(a: torch.Tensor, b: torch.Tensor, out: torch.Tensor,
                        BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32):
    """
    Launch the Triton GEMM kernel for a batch of matrices.

    Parameters
    ----------
    a : torch.Tensor
        Shape (B, M, K), dtype torch.float16, contiguous.
    b : torch.Tensor
        Shape (B, K, N), dtype torch.float16, contiguous (already transposed).
    out : torch.Tensor
        Shape (B, M, N), dtype torch.float16, contiguous (output buffer).

    The function returns ``out`` for convenience.
    """
    assert a.is_contiguous() and b.is_contiguous() and out.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match."

    grid = (
        (tl.cdiv(M, BLOCK_M), tl.cdiv(N, BLOCK_N), B)  # (grid_x, grid_y, grid_z)
    )
    batched_gemm_kernel[grid](
        a, b, out,
        B, M, N, K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Public entry‑point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dictionary with the following keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config_dict  : must contain ``dim`` (C) and ``hidden_dim`` (H)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move a weight to fp16 on the right device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]           # float32, shape (C,)
    norm_b = weights["norm.bias"]             # float32, shape (C,)

    # Concatenate all five linear maps (left_proj, right_proj,
    # left_gate, right_gate, out_gate) into a single weight matrix.
    # Shape after concatenation: (5·H, C)
    all_proj_w = torch.cat([
        to_half(weights["left_proj.weight"]),
        to_half(weights["right_proj.weight"]),
        to_half(weights["left_gate.weight"]),
        to_half(weights["right_gate.weight"]),
        to_half(weights["out_gate.weight"]),
    ], dim=0)  # (5*hidden_dim, dim)

    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )                     # [B, N, N, C] – float32

    # Convert to FP16 for the expensive parts
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ One fused linear pass producing all projections and gates
    # ------------------------------------------------------------------
    # (B, N, N, 5·H)
    proj = F.linear(x_h, all_proj_w)          # bias‑free
    # Slice the concatenated tensor
    hidden = hidden_dim
    left      = proj[..., 0 * hidden : 1 * hidden]   # (B,N,N,H)
    right     = proj[..., 1 * hidden : 2 * hidden]
    left_gate = proj[..., 2 * hidden : 3 * hidden]
    right_gate= proj[..., 3 * hidden : 4 * hidden]
    out_gate  = proj[..., 4 * hidden : 5 * hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (broadcast on H)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and apply to left/right tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    # Move hidden dimension to the batch axis -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose the last two axes of right so that we can use a plain GEMM
    right_T = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    # Collapse the first two axes (B·H) → batch dimension for the kernel
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # (B·H, N, N)
    right_flat = right_T.view(BH, N, N)    # (B·H, N, N)  (already transposed)
    out_flat = torch.empty_like(left_flat)  # fp16 output buffer

    # Run the Triton GEMM
    batched_gemm_triton(left_flat, right_flat, out_flat,
                        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32 ↔ fp16 conversion around it)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7️⃣ Gated projection back to the original channel dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                # (B, N, N, H)
    final = F.linear(gated, to_out_w)          # (B, N, N, C) – fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 as required by the API
    # ------------------------------------------------------------------
    return final.to(torch.float32)
