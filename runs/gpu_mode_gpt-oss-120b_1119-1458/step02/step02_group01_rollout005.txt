@triton.jit
def trimul_kernel(
    A_ptr, B_ptr, C_ptr,
    N, H,
    stride_batch, stride_i, stride_k, stride_h,
    stride_batch_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H_TILE: tl.constexpr, # number of hidden channels processed per block (maybe 1)
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2) # combined batch*hidden index

    # compute batch index and hidden index
    hidden_dim = H
    b_idx = pid_z // hidden_dim
    h_idx = pid_z % hidden_dim

    # offsets for i, j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        a_ptrs = (
            A_ptr
            + b_idx * stride_batch
            + h_idx * stride_h
            + offs_i[:, None] * stride_i
            + offs_k[None, :] * stride_k
        )
        b_ptrs = (
            B_ptr
            + b_idx * stride_batch
            + h_idx * stride_h
            + offs_j[:, None] * stride_i
            + offs_k[None, :] * stride_k
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None,:], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None,:], other=0.0)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # store result
    c_ptrs = (
        C_ptr
        + b_idx * stride_batch_out
        + h_idx * stride_h_out
        + offs_i[:, None] * stride_i_out
        + offs_j[None, :] * stride_j_out
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

The implementation follows the reference PyTorch version but
* runs the O(N³) contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
  with a custom Triton kernel,
* avoids the costly `permute(...).contiguous()` copy by addressing the
  original [B, N, N, H] layout directly,
* performs all linear projections, gates and layer‑norms in fp16 (the first
  norm is kept in fp32 for stability) and accumulates the mat‑mul in fp32,
* finally returns a float‑32 tensor as required by the harness.

The kernel tiles the two input matrices with a 64×64×64 tile, uses two
pipeline stages and 8 warps – a good trade‑off for the H100 tensor cores.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batch‑wise mat‑mul C = A @ Bᵀ  (A,B,C are [B,N,N,H])
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    A_ptr,                     # *fp16  left  [B,N,N,H]
    B_ptr,                     # *fp16  right [B,N,N,H]
    C_ptr,                     # *fp16  out   [B,N,N,H]
    N,                         # i32   sequence length
    H,                         # i32   hidden dimension
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides of A
    stride_cb, stride_ci, stride_cj, stride_ch,   # strides of C
    BLOCK_M: tl.constexpr,     # tile size in i‑dimension
    BLOCK_N: tl.constexpr,     # tile size in j‑dimension
    BLOCK_K: tl.constexpr,     # tile size in k‑dimension
):
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # combined batch*hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden indices from pid_z
    # ------------------------------------------------------------------
    hidden = H
    b_idx = pid_z // hidden            # batch index
    h_idx = pid_z % hidden             # hidden‑channel index

    # ------------------------------------------------------------------
    # Offsets for the current i/j tiles
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (fp32 for precision)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k‑tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load a tile from left  (i × k) -----
        a_ptrs = (
            A_ptr
            + b_idx * stride_ab                     # batch offset
            + h_idx * stride_ah                     # hidden offset
            + offs_i[:, None] * stride_ai           # i offset
            + offs_k[None, :] * stride_ak           # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ----- load a tile from right (j × k) -----
        b_ptrs = (
            B_ptr
            + b_idx * stride_ab                     # batch offset (same stride as A)
            + h_idx * stride_ah                     # hidden offset
            + offs_j[:, None] * stride_ai           # j offset (shares stride with i)
            + offs_k[None, :] * stride_ak           # k offset
        )
        b = tl.load(
            b_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ----- fp16×fp16 -> fp32 dot product -----
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ------------------------------------------------------------------
    # Write the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + b_idx * stride_cb                     # batch offset
        + h_idx * stride_ch                     # hidden offset
        + offs_i[:, None] * stride_ci           # i offset
        + offs_j[None, :] * stride_cj           # j offset
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _trimul_outgoing_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for fp16 tensors ``left`` and ``right`` of shape [B,N,N,H].
    Returns a fp16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    # Output buffer
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # Strides in element (not byte) units
    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out.stride()

    # Tile size – 64×64×64 works well on an H100 (fits in shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * H,                                  # batch * hidden
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        H,
        stride_ab, stride_ai, stride_ak, stride_ah,
        stride_cb, stride_ci, stride_cj, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------

    """

          - weights_dict : mapping name → torch.Tensor (float32)
          - config_dict  : must contain ``dim`` and ``hidden_dim``;
                           optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # Cast to fp16 for the remaining heavy work
    # -------------------------------------------------
    x = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w) # [B,N,N,hidden]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 in, fp16 out)
    # -------------------------------------------------
    out = _trimul_outgoing_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
