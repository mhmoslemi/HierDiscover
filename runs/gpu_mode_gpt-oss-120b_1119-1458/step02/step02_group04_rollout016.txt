    x_ptr, # float16 input after LN, shape [M, dim]

    pid_pos = tl.program_id(0)
    pid_head = tl.program_id(1)

"""
TriMul (outgoing) – fused projection/gate kernel + batched GEMM.

Algorithm
---------
1. Layer‑norm the input (float32 for numerical stability).
2. Cast to fp16 and flatten the pairwise tensor (shape B·N·N × C).
3. One Triton kernel (`proj_gate_fused_kernel`) simultaneously:
   • Computes the four linear projections (left/right × proj/gate) on the
     flattened input.
   • Applies the sigmoid‑gate to the two gate branches.
   • Applies the optional pair‑wise mask (broadcast over the hidden dim).
   • Multiplies the projection by its gate (and mask) producing the gated
     tensors `left` and `right` of shape (B·N·N, H).
4. Reshape `left` / `right` to (B, H, N, N) and run a batched GEMM
   (`torch.bmm`) to evaluate the N³ contraction:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
5. Layer‑norm over the hidden dimension, apply the output gate
   (linear + sigmoid) and a final linear projection back to the original
   channel dimension.
6. Return the result in float32.

Only the projection/gate part is written in Triton (the heavy contraction
still uses cuBLAS‑accelerated `torch.bmm`).  This reduces the number of
kernel launches and the total memory traffic compared with the naïve
PyTorch implementation while still satisfying the requirement to use a
custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: fused projection + gating + optional mask.
# -------------------------------------------------------------------------
@triton.jit
def proj_gate_fused_kernel(
    x_ptr,                 # const half*   [M, C]   flattened input after LN
    left_proj_w_ptr,       # const half*   [H, C]   left projection weights
    left_gate_w_ptr,       # const half*   [H, C]   left gate weights
    right_proj_w_ptr,      # const half*   [H, C]   right projection weights
    right_gate_w_ptr,      # const half*   [H, C]   right gate weights
    mask_ptr,              # const half*   [M]      broadcasted mask (1.0 if no mask)
    left_out_ptr,          # half*         [M, H]   gated left tensor
    right_out_ptr,         # half*         [M, H]   gated right tensor
    M,                     # total number of (b,i,k) positions
    C,                     # input channel dimension (dim)
    H,                     # hidden dimension (hidden_dim)
    BLOCK_POS: tl.constexpr,   # tile size along the flattened position axis
    BLOCK_H: tl.constexpr,     # tile size along the hidden‑dim axis
):
    """Compute left/right projections, gates and mask in a single pass.

    Output shapes:
        left_out, right_out : (M, H)   (flattened B·N·N × hidden_dim)
    """
    pid_pos = tl.program_id(0)      # position tile id
    pid_h   = tl.program_id(1)      # hidden‑dim tile id

    # -----------------------------------------------------------------
    # Offsets for the current block
    # -----------------------------------------------------------------
    pos_start = pid_pos * BLOCK_POS
    h_start   = pid_h   * BLOCK_H

    offs_pos = pos_start + tl.arange(0, BLOCK_POS)
    offs_h   = h_start   + tl.arange(0, BLOCK_H)

    mask_pos = offs_pos < M
    mask_h   = offs_h   < H

    # -----------------------------------------------------------------
    # Load a tile of the flattened input (shape BLOCK_POS × C)
    # -----------------------------------------------------------------
    offs_c = tl.arange(0, C)
    # x[offs_pos, :]  ->  pointer arithmetic
    x_ptrs = x_ptr + (offs_pos[:, None] * C + offs_c[None, :])
    x = tl.load(x_ptrs, mask=mask_pos[:, None], other=0.0)          # [BP, C]

    # -----------------------------------------------------------------
    # Load mask values (scalar per position) and broadcast
    # -----------------------------------------------------------------
    mask_val = tl.load(mask_ptr + offs_pos, mask=mask_pos, other=1.0)   # [BP]
    mask_val = mask_val[:, None]                                      # [BP, 1]

    # -----------------------------------------------------------------
    # Load weight sub‑matrices for this hidden‑dim tile
    # -----------------------------------------------------------------
    # shape of each weight tile: [BH, C]
    lp_ptrs = left_proj_w_ptr  + (offs_h[:, None] * C + offs_c[None, :])
    lg_ptrs = left_gate_w_ptr   + (offs_h[:, None] * C + offs_c[None, :])
    rp_ptrs = right_proj_w_ptr + (offs_h[:, None] * C + offs_c[None, :])
    rg_ptrs = right_gate_w_ptr + (offs_h[:, None] * C + offs_c[None, :])

    lp = tl.load(lp_ptrs, mask=mask_h[:, None], other=0.0)   # [BH, C]
    lg = tl.load(lg_ptrs, mask=mask_h[:, None], other=0.0)   # [BH, C]
    rp = tl.load(rp_ptrs, mask=mask_h[:, None], other=0.0)   # [BH, C]
    rg = tl.load(rg_ptrs, mask=mask_h[:, None], other=0.0)   # [BH, C]

    # -----------------------------------------------------------------
    # Perform the four dot products:
    #   left_proj = x @ lp.T
    #   left_gate = x @ lg.T
    #   right_proj = x @ rp.T
    #   right_gate = x @ rg.T
    # tl.dot expects (M,K) and (N,K) and reduces over K
    # -----------------------------------------------------------------
    left_proj  = tl.dot(x, lp)   # [BP, BH]
    left_gate  = tl.dot(x, lg)   # [BP, BH]
    right_proj = tl.dot(x, rp)   # [BP, BH]
    right_gate = tl.dot(x, rg)   # [BP, BH]

    # -----------------------------------------------------------------
    # Sigmoid for gates (in fp16)
    # -----------------------------------------------------------------
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

    # -----------------------------------------------------------------
    # Apply mask and gates
    # -----------------------------------------------------------------
    left  = left_proj * left_gate * mask_val    # [BP, BH]
    right = right_proj * right_gate * mask_val  # [BP, BH]

    # -----------------------------------------------------------------
    # Store results (flattened tensors)
    # -----------------------------------------------------------------
    out_ptrs = left_out_ptr + (offs_pos[:, None] * H + offs_h[None, :])
    tl.store(out_ptrs, left,
             mask=(mask_pos[:, None] & mask_h[None, :]))

    out_ptrs = right_out_ptr + (offs_pos[:, None] * H + offs_h[None, :])
    tl.store(out_ptrs, right,
             mask=(mask_pos[:, None] & mask_h[None, :]))

# User‑facing entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack inputs & basic shapes
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    M = B * N * N                               # flattened pairwise dimension

    # --------------------------------------------------------------
    # 2️⃣ Helper: move a weight to fp16 on the target device (no copy if already)
    # --------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16, non_blocking=True)

    # --------------------------------------------------------------
    # 3️⃣ Load / reshape all model weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]                     # (dim,)
    norm_bias   = weights["norm.bias"]                       # (dim,)

    left_proj_w   = to_half(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = to_half(weights["right_proj.weight"])   # (hidden_dim, dim)

    left_gate_w   = to_half(weights["left_gate.weight"])    # (hidden_dim, dim)
    right_gate_w  = to_half(weights["right_gate.weight"])   # (hidden_dim, dim)

    out_gate_w    = to_half(weights["out_gate.weight"])     # (hidden_dim, dim)

    to_out_norm_w = to_half(weights["to_out_norm.weight"])  # (hidden_dim,)
    to_out_norm_b = to_half(weights["to_out_norm.bias"])    # (hidden_dim,)

    to_out_w      = to_half(weights["to_out.weight"])       # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Input LayerNorm (float32 for stability) + cast to fp16
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    x_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 5️⃣ Flatten for the Triton kernel
    # --------------------------------------------------------------
    x_flat = x_fp16.reshape(M, dim).contiguous()   # (M, C)
    # Triton works with contiguous tensors; obtain raw pointers later.

    # --------------------------------------------------------------
    # 6️⃣ Prepare mask (broadcasted to the flattened view)
    # --------------------------------------------------------------
    if mask_tensor is None:
        # all‑ones mask – allocate a small buffer of ones
        mask_flat = torch.ones(M, dtype=torch.float16, device=device)
    else:
        mask_flat = mask_tensor.to(dtype=torch.float16, device=device).reshape(M)

    # --------------------------------------------------------------
    # 7️⃣ Allocate output buffers for the gated left / right tensors
    # --------------------------------------------------------------
    left_flat  = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right_flat = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 8️⃣ Triton launch config (tuned for typical H100 fp16 workloads)
    # --------------------------------------------------------------
    BLOCK_POS = 128   # rows of the flattened (B·N·N) dimension per program
    BLOCK_H   = 64    # hidden‑dim tiles per program

    grid = (
        triton.cdiv(M, BLOCK_POS),                     # along position axis
        triton.cdiv(hidden_dim, BLOCK_H),              # along hidden axis
    )

    # -----------------------------------------------------------------
    # 9️⃣ Launch fused projection/gate kernel
    # -----------------------------------------------------------------
    proj_gate_fused_kernel[grid](
        x_ptr=x_flat,
        left_proj_w_ptr=left_proj_w,
        left_gate_w_ptr=left_gate_w,
        right_proj_w_ptr=right_proj_w,
        right_gate_w_ptr=right_gate_w,
        mask_ptr=mask_flat,
        left_out_ptr=left_flat,
        right_out_ptr=right_flat,
        M=M,
        C=dim,
        H=hidden_dim,
        BLOCK_POS=BLOCK_POS,
        BLOCK_H=BLOCK_H,
        num_warps=8,                # enough warps to hide latency
    )

    # --------------------------------------------------------------
    # 10️⃣ Reshape to (B, N, N, H) – ready for the N³ contraction
    # --------------------------------------------------------------
    left = left_flat.view(B, N, N, hidden_dim)     # [B, N, N, H]
    right = right_flat.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 11️⃣ Batched GEMM: out[b, h, i, j] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # move hidden dim to the batch axis to reuse torch.bmm (cuBLAS)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat_gemm  = left_perm.view(B * hidden_dim, N, N)   # (B·H, N, N)
    right_flat_gemm = right_perm.view(B * hidden_dim, N, N)  # (B·H, N, N)

    # torch.bmm performs (B·H, N, N) @ (B·H, N, N)ᵀ efficiently
    out_flat = torch.bmm(left_flat_gemm, right_flat_gemm.transpose(1, 2))  # (B·H, N, N)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 12️⃣ Output LayerNorm over the hidden dimension (float32 for stability)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    ).to(torch.float16)

    # --------------------------------------------------------------
    # 13️⃣ Output gate (sigmoid of a linear projection) + final projection
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # [B, N, N, H] fp16
    gated = out_norm * out_gate                               # element‑wise gate

    # final linear (hidden_dim → dim)
    final = F.linear(gated, to_out_w)                         # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 14️⃣ Return float32 as required by the spec
    # --------------------------------------------------------------
    return final.to(torch.float32)
