@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_head = tl.program_id(0)  # index for BH
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute base offsets for this batch*head
    # Compute offsets for row and col
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Get start pointers for this batch*head
    left_batch_ptr = left_ptr + batch_head * stride_l_batch
    right_batch_ptr = right_ptr + batch_head * stride_r_batch
    out_batch_ptr = out_ptr + batch_head * stride_o_batch

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute K range
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # mask for k range
        k_mask = offs_k < N

        # pointer for left A tile (M x K)
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_m) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < N) & k_mask[None, :], other=0.0).to(tl.float32)

        # pointer for right B tile, but transposed: B has rows = k, cols = n using stride reversed
        b_ptrs = right_batch_ptr + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_m)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & (offs_n[None, :] < N), other=0.0).to(tl.float32)

        # multiply accumulate
        acc += tl.dot(a, b)

    # End for

    # Write output
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    # apply mask for output boundaries
    out_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ak,  # hmm fix naming
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # calculate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for rows and cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_row) + (offs_k[None, :] * stride_a_col)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_row) + (offs_n[None, :] * stride_b_col)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left, right: torch tensors with shape [B, N, N, H] dtype torch.float16
    B, N, _, H = left.shape

    # Permute and flatten
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right transposed
    right_trans = right.permute(0, 3, 2, 1).contiguous()
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_trans.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # compute strides
    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input (float32 → float16).
2️⃣ Two bias‑free linear projects produce *left* and *right* tensors
   of shape [B, N, N, H] (H = hidden_dim) in FP16.
3️⃣ Optional mask is applied (broadcast on the hidden dimension).
4️⃣ Gating (sigmoid ∘ linear) multiplies into *left* and *right*.
5️⃣ Contraction ∑ₖ left[b,i,k,h]·right[b,j,k,h] is performed by a custom
   Triton batched GEMM kernel – the heavy N³ work is carried out
   completely on‑chip using tensor‑cores.  The hidden‑dim is fused into the
   batch axis (B·H independent multiplications of size N×N).
6️⃣ Hidden‑dim layer‑norm, gated by *out_gate*.
7️⃣ Final linear projection back to the original channel dimension C.
8️⃣ Return float32 output.

The Triton kernel implements a classic tiled GEMM (`A @ Bᵀ`) with
row‑major layout for *A* (left) and column‑major layout for *B*
(right‑transposed).  By flattening (batch, hidden) we launch a single
kernel with a three‑dimensional grid: (batch·head, tile‑row, tile‑col).
All heavy arithmetic uses fp16 inputs and fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched matmul for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix dims (all = seq_len)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C[b,i,j] = Σₖ A[b,i,k]·B[b,k,j] for a batch of independent GEMMs.
    A has shape (B·H, M, K)   – rows = i, cols = k
    B has shape (B·H, K, N)   – rows = k, cols = j (already transposed)
    C has shape (B·H, M, N).
    """
    pid_batch = tl.program_id(0)               # B·H
    pid_m = tl.program_id(1)                   # tile row index
    pid_n = tl.program_id(2)                   # tile col index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (blocked)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_row) + (offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_row) + (offs_n[None, :] * stride_b_col)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back C tile
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: invoke the Triton kernel
# ----------------------------------------------------------------------
def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h].

    Parameters
    ----------
    left, right : torch.Tensor
        FP16 tensors of shape [B, N, N, H] (contiguous).

    Returns
    -------
    out : torch.Tensor
        FP16 tensor with shape [B, N, N, H].
    """
    B, N, _, H = left.shape

    # --------------------------------------------------------------
    # 1) Rearrange tensors so that the hidden dimension becomes part of the batch.
    #    left  → (B·H, N, N)   rows=i, cols=k
    #    right → (B·H, N, N)   rows=k, cols=j  (transpose last two dims)
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_t   = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) with (k, i) layout
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # --------------------------------------------------------------
    # 2) Allocate output, compute strides
    # --------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)

    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat.stride()

    # ------------------------------------------------------------------
    # 3) Kernel launch parameters (tuned for H100 fp16 tensor cores)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch·head dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles in M direction
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles in N direction
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                              # M, N, K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4) Restore the original layout [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Public entry point demanded by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]  (bool/float) or None
        - weights_dict : dict with all linear / layer‑norm parameters
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper to move a weight to fp16 on the correct device
    # --------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # Load model parameters (keep LayerNorm weights in float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C] – float32

    # Switch to FP16 for the heavy path
    x_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid after a bias‑free linear)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ N³ contraction using the Triton kernel
    # --------------------------------------------------------------
    out = _tri_mul_triton(left, right)                     # [B, N, N, H] – fp16

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32 ↔ fp16 conversion around it)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 7️⃣ Gated projection back to the original channel dimension
    # --------------------------------------------------------------
    gated = out_norm * out_gate                # [B, N, N, H]
    final = F.linear(gated, to_out_w)          # [B, N, N, C] – fp16

    # --------------------------------------------------------------
    # 8️⃣ Return in float32 as required by the specification
    # --------------------------------------------------------------
    return final.to(torch.float32)
