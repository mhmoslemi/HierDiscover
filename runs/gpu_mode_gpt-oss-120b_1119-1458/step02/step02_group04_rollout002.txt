@triton.jit
def fuse_proj_gate_mask(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, hidden, mask_is_none: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * hidden
    mask = offs < total

    # compute mask index for broadcast
    if not mask_is_none:
        mask_idx = offs // hidden
        m = tl.load(mask_ptr + mask_idx, mask=mask_idx < (B*N*N), other=0.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (float32 for numerical stability).
2️⃣ Cast to FP16 and flatten to (M, dim) where M = B·N·N.
3️⃣ Fuse the two linear projections (proj & gate) of the “left” and “right”
   paths by concatenating their weight matrices and performing a single
   mat‑mul per side (M × dim) × (2·H × dim) → (M, 2·H).
4️⃣ Split the concatenated result, apply a sigmoid to the gate half and,
   if a mask is present, fuse mask‑multiplication, gate‑multiplication and
   projection‑multiplication with a tiny Triton kernel
   (`fuse_proj_gate_mask_kernel`).
5️⃣ Reshape the gated tensors to (B, H, N, N) and compute the N³ contraction
   with a batched GEMM: out[b,h] = left[b,h] @ right[b,h].T .
6️⃣ Apply a second LayerNorm over the hidden dimension H.
7️⃣ Compute the output gate (sigmoid of a third linear projection) and apply
   it element‑wise.
8️⃣ Final linear projection back to the original channel dimension C.
9️⃣ Return the result in float32.

Only the element‑wise “proj × gate × mask” step is implemented in Triton;
the heavy N³ contraction uses torch.bmm (cuBLAS Tensor‑Core kernels),
which is already optimal on an H100.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: out = proj * gate * mask   (mask is broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def fuse_proj_gate_mask_kernel(
    proj_ptr,          # const half *
    gate_ptr,          # const half *
    mask_ptr,          # const half *
    out_ptr,           # half *
    total_elts: tl.constexpr,      # M * H
    hidden_dim: tl.constexpr,      # H
    BLOCK_SIZE: tl.constexpr,      # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)          # (BLOCK_SIZE,)
    mask = offs < total_elts                         # guard

    # Load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask across hidden dimension:
    #   mask_idx = floor(idx / H)
    mask_idx = offs // hidden_dim
    # mask length = total_elts // hidden_dim = M
    mask_val = tl.load(mask_ptr + mask_idx,
                       mask=mask_idx < (total_elts // hidden_dim),
                       other=0.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model weights (torch.Tensors)
        - config_dict  : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]          # H
    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B,N,N,dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 and flatten to (M, dim) where M = B·N·N
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)
    M = B * N * N
    x_flat = x_h.view(M, dim)               # (M, dim)

    # ------------------------------------------------------------------
    # Helper to convert a weight tensor to FP16 on the correct device
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Concatenate projection and gate weights (left side)
    # ------------------------------------------------------------------
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )
    left_concat_w = to_fp16(left_concat_w)          # (2H, dim)

    # ------------------------------------------------------------------
    # Concatenate projection and gate weights (right side)
    # ------------------------------------------------------------------
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )
    right_concat_w = to_fp16(right_concat_w)        # (2H, dim)

    # ------------------------------------------------------------------
    # Linear: (M, dim) @ (2H, dim).T  -> (M, 2H)
    # ------------------------------------------------------------------
    left_cat = F.linear(x_flat, left_concat_w)      # (M, 2H)
    right_cat = F.linear(x_flat, right_concat_w)    # (M, 2H)

    # ------------------------------------------------------------------
    # Split projection / gate halves and apply sigmoid to the gate halves
    # ------------------------------------------------------------------
    left_proj = left_cat[:, :hidden]                 # (M, H)
    left_gate_raw = left_cat[:, hidden:]             # (M, H)
    left_gate = torch.sigmoid(left_gate_raw)         # (M, H)

    right_proj = right_cat[:, :hidden]               # (M, H)
    right_gate_raw = right_cat[:, hidden:]           # (M, H)
    right_gate = torch.sigmoid(right_gate_raw)       # (M, H)

    # ------------------------------------------------------------------
    # Optional mask handling – fused elementwise kernel
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B, N, N]  -> flatten to (M,)
        mask_fp16 = mask.to(dtype=torch.float16).view(M)

        # Allocate output buffers
        left = torch.empty_like(left_proj)           # (M, H)
        right = torch.empty_like(right_proj)         # (M, H)

        total_elts = M * hidden
        BLOCK_SIZE = 1024
        num_blocks = (total_elts + BLOCK_SIZE - 1) // BLOCK_SIZE

        # left = left_proj * left_gate * mask
        fuse_proj_gate_mask_kernel[(num_blocks,)](
            left_proj, left_gate, mask_fp16, left,
            total_elts, hidden, BLOCK_SIZE,
        )
        # right = right_proj * right_gate * mask
        fuse_proj_gate_mask_kernel[(num_blocks,)](
            right_proj, right_gate, mask_fp16, right,
            total_elts, hidden, BLOCK_SIZE,
        )
        # Release intermediate tensors early
        del left_proj, left_gate, right_proj, right_gate, left_cat, right_cat, mask_fp16
    else:
        # No mask – simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        del left_proj, left_gate, right_proj, right_gate, left_cat, right_cat

    # ------------------------------------------------------------------
    # Reshape to (B, H, N, N) – ready for batched GEMM
    # ------------------------------------------------------------------
    left_view = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # ------------------------------------------------------------------
    # Batched GEMM: out[b,h] = left[b,h] @ right[b,h].T
    # ------------------------------------------------------------------
    left_flat = left_view.view(B * hidden, N, N)               # (B*H, N, N)
    right_flat = right_view.view(B * hidden, N, N)             # (B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # Output LayerNorm (float32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)

    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out_norm_h = out_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # Output gate
    # ------------------------------------------------------------------
    out_gate_w = to_fp16(weights["out_gate.weight"])
    out_gate_raw = F.linear(x_flat, out_gate_w)                # (M, H)
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden).to(torch.float16)

    # ------------------------------------------------------------------
    # Apply output gate
    # ------------------------------------------------------------------
    gated = out_norm_h * out_gate

    # ------------------------------------------------------------------
    # Final projection back to dim
    # ------------------------------------------------------------------
    to_out_w = to_fp16(weights["to_out.weight"])
    final = F.linear(gated.view(M, hidden), to_out_w)          # (M, dim)
    final = final.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # Return float32 as required
    # ------------------------------------------------------------------
    return final.to(torch.float32)
