# Triton kernel to fuse mask and gates

def fuse_gate_mask_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 * (optional)
    H,                 # hidden_dim int64
    apply_mask: tl.constexpr,   # 0 or 1
    N_ELEM: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    # read left, right, gates
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # compute mask factor if needed
    if apply_mask:
        # compute index into mask, each H consecutive elements share same mask element
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # fp16 mask broadcast
        l = l * lg * m
        r = r * rg * m
    else:
        l = l * lg
        r = r * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Input LayerNorm in FP32 (numerical stability).
2. Cast to FP16 for all heavy linear work.
3. Reduce the number of matmuls:
   * Concatenate the projection and gate weight matrices for the
     “left” and “right” paths and perform a single linear per side.
   * Split the result into projection and gate tensors, apply sigmoid
     to the gate part.
4. Fuse mask‑application and element‑wise multiplication with the
   gates in a tiny Triton kernel (`fuse_gate_mask_kernel`).  This
   loads each element of `left_proj`, `right_proj`, `left_gate`,
   `right_gate` (and, optionally, the mask) only once.
5. Core N³ operation (einsum) is executed as a batched matrix
   multiplication (`torch.bmm`) on tensors of shape
   (B·hidden_dim, seq_len, seq_len).  This leverages H100 Tensor‑Cores.
6. Post‑processing LayerNorm (FP16 weights/bias) on the result.
7. Fuse the output‑gate multiplication in‑place with another
   Triton kernel (`apply_out_gate_kernel`).
8. Final linear projection back to the original channel dimension
   (`F.linear`) – a single matmul over the hidden dimension.
9. Cast the output back to FP32 (the API expects float32).

Only the element‑wise gating operations are performed with Triton;
the rest uses highly‑optimized PyTorch kernels.  All compute after
the first LayerNorm runs in FP16, giving a large speed‑up while
preserving the required numerical accuracy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernels
# --------------------------------------------------------------

@triton.jit
def fuse_gate_mask_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 * (optional)
    hidden_dim,        # int64
    APPLY_MASK: tl.constexpr,      # 0 or 1
    N_ELEM: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place multiplication:
        left  = left_proj  * left_gate  * (mask?)
        right = right_proj * right_gate * (mask?)

    Tensors are flattened, hidden_dim elements belong to the same
    (b,i,k) entry, therefore mask index = offset // hidden_dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    # Load the four main tensors
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if APPLY_MASK:
        # broadcast mask over hidden_dim
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        l = l * lg * m
        r = r * rg * m
    else:
        l = l * lg
        r = r * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


@triton.jit
def apply_out_gate_kernel(
    out_ptr,           # fp16 *
    out_gate_ptr,      # fp16 *
    N_ELEM: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place multiplication: out = out * out_gate
    Both tensors are flattened.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    o = tl.load(out_ptr + offs, mask=mask, other=0.0)
    g = tl.load(out_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, o * g, mask=mask)


# --------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32/bool)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
            * "dim"          – input/output channel dimension
            * "hidden_dim"   – internal hidden dimension
            * "nomask" (opt) – if True, the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)  # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Prepare concatenated projection+gate weights (FP16)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)

    w_left_concat  = torch.cat([w_left_proj, w_left_gate], dim=0)   # (2*H, dim)
    w_right_concat = torch.cat([w_right_proj, w_right_gate], dim=0) # (2*H, dim)

    # ------------------------------------------------------------------
    # 3️⃣ Single matmul per side, then split into projection & gate
    # ------------------------------------------------------------------
    # left side
    left_concat = F.linear(x_h, w_left_concat)   # [B, N, N, 2*H]  FP16
    left_proj   = left_concat[..., :hidden_dim]   # [B, N, N, H]
    left_gate_pre = left_concat[..., hidden_dim:] # [B, N, N, H]
    left_gate   = torch.sigmoid(left_gate_pre)    # FP16

    # right side
    right_concat = F.linear(x_h, w_right_concat)  # [B, N, N, 2*H]  FP16
    right_proj   = right_concat[..., :hidden_dim]  # [B, N, N, H]
    right_gate_pre = right_concat[..., hidden_dim:]# [B, N, N, H]
    right_gate   = torch.sigmoid(right_gate_pre)   # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Output‑gate (separate linear)
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))  # [B, N, N, H]  FP16

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask handling (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if nomask:
        # No mask – compile‑time flag will skip loading it.
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)
        apply_mask_flag = 0
    else:
        # Expand mask to fp16; we keep it as (B, N, N) and let the kernel
        # broadcast over the hidden dimension.
        mask_fp16 = mask.to(torch.float16)
        apply_mask_flag = 1

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask + gate multiplication with Triton
    # ------------------------------------------------------------------
    # Ensure contiguity for the flattening operation.
    left_proj   = left_proj.contiguous()
    right_proj  = right_proj.contiguous()
    left_gate   = left_gate.contiguous()
    right_gate  = right_gate.contiguous()

    numel = left_proj.numel()          # B * N * N * H
    BLOCK = 65536                       # tuneable; ≈64 Ki elements per program
    grid = ((numel + BLOCK - 1) // BLOCK,)

    fuse_gate_mask_kernel[grid](
        left_proj.view(-1),
        right_proj.view(-1),
        left_gate.view(-1),
        right_gate.view(-1),
        mask_fp16.view(-1),          # dummy if nomask
        hidden_dim,
        APPLY_MASK=apply_mask_flag,
        N_ELEM=numel,
        BLOCK=BLOCK,
    )
    # At this point `left_proj` and `right_proj` hold the masked &
    # gated tensors that will be used in the N³ multiplication.

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ operation: batched matmul (equivalent to einsum)
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape  # H == hidden_dim

    # Rearrange to (B*H, N, N) layout required by torch.bmm
    left_mat  = left_proj.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched GEMM – FP16, H100 Tensor‑Core fast path
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Restore to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing LayerNorm (FP16 weights)
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output gate (in‑place) via Triton
    # ------------------------------------------------------------------
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    apply_out_gate_kernel[grid_out](
        out.view(-1),
        out_gate.contiguous().view(-1),
        N_ELEM=numel_out,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (FP16 → FP16)
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    final = F.linear(out, w_to_out)                        # [B, N, N, dim] FP16

    # Return in the expected float32 format
    return final.to(torch.float32)
