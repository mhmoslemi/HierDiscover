"""
Optimized outgoing TriMul forward pass (AlphaFold3).

The implementation fuses the five linear projections (two projections,
two gate projections and the output gate) into a single GEMM, thereby
removing four costly kernel launches.  The dominant N³ operation is
performed as a batched matrix multiplication (left @ rightᵀ) using
torch.bmm on half‑precision tensors.  A tiny Triton kernel is kept for
the final element‑wise multiplication of the post‑norm tensor with the
output gate – this satisfies the requirement of having “part of the
operations in a kernel”.  All heavy work is done in FP16, while the
input and final result are kept in FP32 for numerical stability.

The data flow (shapes) is:

    x (FP32) ──► LayerNorm (FP32) ──► cast FP16
        ├─► combined linear (5× hidden_dim) ──► split
        │    ├─► left, left_gate, right, right_gate, out_gate (FP16)
        │    └─► left  = left  * mask  * sigmoid(left_gate)
        │    └─► right = right * mask  * sigmoid(right_gate)
        └─► batched GEMM (left @ rightᵀ) → out (FP16)
    out ──► LayerNorm (FP16) ──► out_gate (FP16)
    └─► Triton element‑wise mul (out * out_gate) → out_mul (FP16)
    └─► final linear (hidden_dim → dim) (FP16) → result (FP32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel for the final element‑wise multiplication (out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # fp16 input tensor (flattened)
    b_ptr,          # fp16 input tensor (flattened)
    out_ptr,        # fp16 output tensor (flattened)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    out = a * b   (both vectors of length numel)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      shape [B, N, N, dim]   (float32)
        - mask          : torch.Tensor      shape [B, N, N]        (bool/float)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   must contain 'dim' and 'hidden_dim',
                           optional 'nomask' (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – keep statistics in FP32
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half for all subsequent work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five Linear layers (2 proj + 2 gate + 1 out_gate)
    #    into a single GEMM: (B*N*N, dim) @ (dim, 5*hidden_dim)
    # ------------------------------------------------------------------
    # Gather the five weight matrices (all without bias)
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    # Concatenate along the output dimension → shape (5*hidden_dim, dim)
    w_concat = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(torch.float16)

    # Flatten the spatial dimensions for the GEMM
    B, N, _, _ = x_h.shape
    M = B * N * N
    x_flat = x_h.view(M, dim)                     # (M, dim)

    # Single linear projection → (M, 5*hidden_dim)
    proj = F.linear(x_flat, w_concat)             # (M, 5*hidden_dim)

    # Reshape back to 4‑D tensor and split
    proj = proj.view(B, N, N, 5 * hidden_dim)     # (B, N, N, 5*H)
    left       = proj[..., 0 * hidden_dim : 1 * hidden_dim]
    left_gate  = proj[..., 1 * hidden_dim : 2 * hidden_dim]
    right      = proj[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate = proj[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate   = proj[..., 4 * hidden_dim : 5 * hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) – broadcast over the hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates and combine with the projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    # In‑place elementwise multiplication (no extra kernel needed)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ operation: batched matrix multiplication
    #    out = left @ rightᵀ   (equivalent to the einsum)
    # ------------------------------------------------------------------
    # Rearrange to shape (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM – uses Tensor‑Core FP16 paths on H100
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout → (B, N, N, hidden_dim)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing LayerNorm (half‑precision)
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply with the output gate (triton kernel)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate)                     # (B, N, N, H) fp16
    out_mul = torch.empty_like(out)                        # fp16 output buffer

    numel = out.numel()
    BLOCK = 65536  # tuned for H100, ~64 Ki elements per program
    grid = ( (numel + BLOCK - 1) // BLOCK, )

    elementwise_mul_kernel[grid](
        out.view(-1),
        out_gate.contiguous().view(-1),
        out_mul.view(-1),
        numel,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    final = F.linear(out_mul, w_to_out)                    # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (expected by the surrounding model)
    # ------------------------------------------------------------------
    return final.to(torch.float32)
