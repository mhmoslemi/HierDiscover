def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batch-wise matmul C = left @ right^T.
    All tensors are [B,N,N,H] fp16 and contiguous.
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.matmul(A, B_.transpose(-2, -1))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

"""
TriMul (outgoing) – optimized forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32, stable).
2. Cast to fp16 and perform **one** large linear projection that yields the
   five tensors required by TriMul:
   left_proj, right_proj, left_gate, right_gate, out_gate.
3. Fuse masking, sigmoid gating (left/right) and sigmoid of out_gate in a
   single Triton kernel.  This removes three separate element‑wise passes.
4. Compute the heavy O(N³) contraction with a batched GEMM:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using torch.matmul (cuBLAS/H100 tensor‑cores).
5. Apply the second LayerNorm, multiply by the already‑sigmoided out_gate,
   project back to the original feature dimension and cast to float32.

Only the element‑wise mask/gate fusion is implemented in Triton; the bulk
matrix multiplication is delegated to cuBLAS which is faster than a custom
kernel for the typical shapes in AlphaFold‑3.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid gates and compute out_gate sigmoid.
# ----------------------------------------------------------------------
@triton.jit
def _apply_gates_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    B, N, H,
    # strides of the five input tensors (4‑D: B,N,N,H)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_j, stride_og_h,
    # stride of mask (3‑D: B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_j,
    # strides of the three output tensors (same layout as the inputs)
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    stride_ogout_b, stride_ogout_i, stride_ogout_j, stride_ogout_h,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)      # batch index
    pid_i = tl.program_id(1)      # seq‑len i
    pid_j = tl.program_id(2)      # seq‑len j

    # -------------------------------------------------
    # Load mask scalar for this (b,i,j) location.
    # The mask has no hidden dimension – it is broadcast.
    # -------------------------------------------------
    mask_offset = pid_b * stride_mask_b + pid_i * stride_mask_i + pid_j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_offset)          # fp16 scalar (0 or 1)

    # -------------------------------------------------
    # Base offsets (common for all hidden‑dim loads/stores)
    # -------------------------------------------------
    off_lp  = pid_b * stride_lp_b  + pid_i * stride_lp_i  + pid_j * stride_lp_j
    off_lg  = pid_b * stride_lg_b  + pid_i * stride_lg_i  + pid_j * stride_lg_j
    off_rp  = pid_b * stride_rp_b  + pid_i * stride_rp_i  + pid_j * stride_rp_j
    off_rg  = pid_b * stride_rg_b  + pid_i * stride_rg_i  + pid_j * stride_rg_j
    off_og  = pid_b * stride_og_b  + pid_i * stride_og_i  + pid_j * stride_og_j

    off_lo  = pid_b * stride_lo_b  + pid_i * stride_lo_i  + pid_j * stride_lo_j
    off_ro  = pid_b * stride_ro_b  + pid_i * stride_ro_i  + pid_j * stride_ro_j
    off_ogout = pid_b * stride_ogout_b + pid_i * stride_ogout_i + pid_j * stride_ogout_j

    # -------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs_h = start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H                         # bounds mask

        # Load tensors (fp16)
        lp = tl.load(left_proj_ptr + off_lp + offs_h * stride_lp_h,
                     mask=mask_h, other=0.0)
        lg = tl.load(left_gate_ptr + off_lg + offs_h * stride_lg_h,
                     mask=mask_h, other=0.0)
        rp = tl.load(right_proj_ptr + off_rp + offs_h * stride_rp_h,
                     mask=mask_h, other=0.0)
        rg = tl.load(right_gate_ptr + off_rg + offs_h * stride_rg_h,
                     mask=mask_h, other=0.0)
        og = tl.load(out_gate_ptr + off_og + offs_h * stride_og_h,
                     mask=mask_h, other=0.0)

        # Sigmoid (1/(1+exp(-x))) in fp16
        lg_sig = 1.0 / (1.0 + tl.exp(-lg))
        rg_sig = 1.0 / (1.0 + tl.exp(-rg))
        og_sig = 1.0 / (1.0 + tl.exp(-og))

        # Apply mask (broadcasted) and gates
        left_out  = lp * lg_sig * mask_val
        right_out = rp * rg_sig * mask_val

        # Store results
        tl.store(left_out_ptr  + off_lo  + offs_h * stride_lo_h,
                 left_out,  mask=mask_h)
        tl.store(right_out_ptr + off_ro  + offs_h * stride_ro_h,
                 right_out, mask=mask_h)
        tl.store(out_gate_out_ptr + off_ogout + offs_h * stride_ogout_h,
                 og_sig,    mask=mask_h)


# ----------------------------------------------------------------------
# Helper: batched matmul C = A @ Bᵀ  (A,B : [B*H,N,N] fp16)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via a single
    batched GEMM.  All tensors are expected to be contiguous fp16.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # batched matmul, using cuBLAS (tensor‑cores on H100)
    C = torch.matmul(A, B_.transpose(-2, -1))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)          # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 3️⃣ One‑shot linear projection producing all five tensors
    # -------------------------------------------------
    B, N, _, _ = x.shape
    flat_x = x.reshape(B * N * N, dim)                     # (B·N·N, dim)

    # concatenate weights in the order: left_proj, right_proj,
    # left_gate, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"].to(torch.float16),
            weights["right_proj.weight"].to(torch.float16),
            weights["left_gate.weight"].to(torch.float16),
            weights["right_gate.weight"].to(torch.float16),
            weights["out_gate.weight"].to(torch.float16),
        ],
        dim=0,
    )   # (5·hidden_dim, dim)

    combined = torch.nn.functional.linear(flat_x, combined_weight)   # (B·N·N, 5·hidden_dim)
    combined = combined.view(B, N, N, 5, hidden_dim)                # [B,N,N,5,hidden]

    # split (contiguous views)
    left_proj  = combined[..., 0, :].contiguous()
    right_proj = combined[..., 1, :].contiguous()
    left_gate  = combined[..., 2, :].contiguous()
    right_gate = combined[..., 3, :].contiguous()
    out_gate   = combined[..., 4, :].contiguous()

    # -------------------------------------------------
    # 4️⃣ Fuse mask, sigmoid gating (left/right) and out_gate sigmoid
    # -------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16)

    # allocate output buffers
    left_out      = torch.empty_like(left_proj)
    right_out     = torch.empty_like(right_proj)
    out_gate_sig  = torch.empty_like(out_gate)

    # stride information (in element units)
    lp_b, lp_i, lp_j, lp_h = left_proj.stride()
    lg_b, lg_i, lg_j, lg_h = left_gate.stride()
    rp_b, rp_i, rp_j, rp_h = right_proj.stride()
    rg_b, rg_i, rg_j, rg_h = right_gate.stride()
    og_b, og_i, og_j, og_h = out_gate.stride()

    lo_b, lo_i, lo_j, lo_h = left_out.stride()
    ro_b, ro_i, ro_j, ro_h = right_out.stride()
    ogout_b, ogout_i, ogout_j, ogout_h = out_gate_sig.stride()

    mask_b, mask_i, mask_j = mask_fp16.stride()

    # launch Triton kernel (grid = B × N × N)
    grid = (B, N, N)
    BLOCK_H = 64                 # 64 threads → two warps, fits hidden ≤ 384

    _apply_gates_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        out_gate,
        mask_fp16,
        left_out, right_out, out_gate_sig,
        B, N, hidden_dim,
        lp_b, lp_i, lp_j, lp_h,
        lg_b, lg_i, lg_j, lg_h,
        rp_b, rp_i, rp_j, rp_h,
        rg_b, rg_i, rg_j, rg_h,
        og_b, og_i, og_j, og_h,
        mask_b, mask_i, mask_j,
        lo_b, lo_i, lo_j, lo_h,
        ro_b, ro_i, ro_j, ro_h,
        ogout_b, ogout_i, ogout_j, ogout_h,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_out, right_out)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 7️⃣ Final gating (out_gate already sigmoid‑ed) and projection
    # -------------------------------------------------
    out = out * out_gate_sig
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
```
