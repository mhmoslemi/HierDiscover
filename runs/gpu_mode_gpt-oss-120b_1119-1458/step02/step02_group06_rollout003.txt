```python
import torch
import triton
import triton.language as tl

@triton.jit
def gram_kernel(
    a_ptr, b_ptr, c_ptr,
    N, BH,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs: batch/hidden (bh), tile row, tile col
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a block A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + bh * stride_a_batch
                 + offs_i[:, None] * stride_a_i
                 + offs_k[None, :] * stride_a_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load a block B: shape (BLOCK_K, BLOCK_N)
        # B is stored as (BH,N,N) but we need b[k,j]
        b = tl.load(
            b_ptr + bh * stride_b_batch
                 + offs_k[:, None] * stride_b_k
                 + offs_j[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # dot product accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store result into C (FP16)
    tl.store(
        c_ptr + bh * stride_c_batch
                + offs_i[:, None] * stride_c_i
                + offs_j[None, :] * stride_c_j,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )
```

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half for compute-intensive parts
    x_h = x.half()  # [B,N,N,dim]

    B, N, _, _ = x_h.shape

    # ---- Fuse projection linear layers ----
    # Weights to half
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*hidden_dim, dim)

    # Linear projection (half)
    proj = torch.nn.functional.linear(x_h, proj_weight)  # [B,N,N,2*hidden]
    left, right = proj.chunk(2, dim=-1)  # each [B,N,N,hidden]

    # Optional mask multiply (broadcast over hidden)
    if not nomask and mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1]
        m = mask.unsqueeze(-1).to(torch.float16)
        left = left * m
        right = right * m

    # ---- Fuse gating linear layers (left_gate, right_gate, out_gate) ----
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    gates_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*hidden_dim, dim)

    gates = torch.nn.functional.linear(x_h, gates_weight).sigmoid()
    left_gate, right_gate, out_gate = gates.chunk(3, dim=-1)

    # Apply gates (still half)
    left = left * left_gate
    right = right * right_gate

    # ---- Prepare tensors for batched Gram product ----
    # permute to (B, hidden, N, N) and flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # Output buffer
    BH = B * hidden_dim
    # Choose kernel vs torch.bmm
    # Use Triton for modest BH (e.g., <=64) or for large N where bmm might be slower due to memory.
    # Empirical heuristic: try Triton for all sizes, but fallback to bmm if BH>128 or N>512
    use_triton = (BH <= 128) and (N <= 512)

    if use_triton:
        out_flat = torch.empty_like(left_flat)  # fp16

        # strides
        stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
        stride_b_batch, stride_b_i, stride_b_k = right_flat.stride()
        stride_c_batch, stride_c_i, stride_c_k = out_flat.stride()

        # In our layout, stride_a_k == stride_b_i == stride_c_k == 1 (contiguous over last axis)
        # stride_a_i == stride_a_k * N == N

        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (BH,
                (N + BLOCK_M - 1) // BLOCK_M,
                (N + BLOCK_N - 1) // BLOCK_N)

        gram_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            BH,
            stride_a_batch,
            stride_a_i,
            stride_a_k,
            stride_b_batch,
            stride_b_i,
            stride_c_batch,
            stride_c_i,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # torch.bmm for the batched matmul (equivalent to left @ right^T)
        # right needs to be transposed on the last two dims
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B,N,N,hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()  # cast to fp32 for layernorm

    # Output layernorm (float32)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Multiply output gate (convert to fp32)
    out = out * out_gate.float()

    # Final linear projection to dim (float32)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
Optimized outgoing TriMul kernel (AlphaFold3 “outgoing” version).

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32).
2. Cast to FP16 and fuse the two projection Linear layers into a single
   MatMul (weights concatenated).  The same trick is applied to the three
   gating Linear layers (left_gate, right_gate, out_gate) → one MatMul.
3. Apply the optional mask and the element‑wise gates while still in FP16.
4. Perform the N³ reduction:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
   This is a batched matrix‑multiply (left @ rightᵀ).  For modest
   batch‑*‑hidden sizes and sequence lengths ≤ 512 we launch a custom
   Triton kernel that accumulates in FP32 and writes FP16.  For larger
   problems we fall back to `torch.bmm` (cuBLAS Tensor‑Core path).
5. Cast the result back to FP32, apply the output LayerNorm, multiply by
   the output gate and a final Linear projection to the original hidden
   dimension.
6. Return a FP32 tensor of shape [B, N, N, dim].

All heavy arithmetic is done in half precision on Tensor‑Cores; only the
LayerNorms and the final projection use FP32 for numerical stability.
The Triton kernel implements a classic blocked GEMM (M×K)·(K×N) with
FP32 accumulation, using `tl.dot(..., out_dtype=tl.float32)`.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the Gram‑product  (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def gram_kernel(
    a_ptr, b_ptr, c_ptr,
    N, BH,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k,
    stride_c_batch, stride_c_i,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch*hidden index `bh`:

        C[bh, i, j] = Σ_k A[bh, i, k] * B[bh, j, k]

    where A and B are stored as (BH, N, N) in row‑major order.
    Accumulation is performed in FP32, result is stored as FP16.
    """
    bh = tl.program_id(0)         # batch*hidden dimension
    pid_i = tl.program_id(1)      # tile row
    pid_j = tl.program_id(2)      # tile column

    # ------------------------------------------------------------------
    # Tile offsets & masks
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over K dimension (the sequence dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a block of A (shape BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + bh * stride_a_batch
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a block of Bᵀ (shape BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + bh * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * 1,          # column stride == 1 (contiguous)
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # FP16 dot, accumulate as FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Store result (cast back to FP16)
    # ------------------------------------------------------------------
    tl.store(
        c_ptr
        + bh * stride_c_batch
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:        [B, N, N]      (float32 or torch.bool)
            - weights:     dict of model tensors (float32)
            - config:      dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0. unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, dim] FP32

    # --------------------------------------------------------------
    # 2. Cast to FP16 for all heavy arithmetic
    # --------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim] FP16
    B, N, _, _ = x_h.shape

    # --------------------------------------------------------------
    # 3. Fuse projections: left_proj & right_proj → one Linear
    # --------------------------------------------------------------
    #   weight shape: (hidden_dim, dim)
    left_w = weights["left_proj.weight"].half()
    right_w = weights["right_proj.weight"].half()
    proj_weight = torch.cat([left_w, right_w], dim=0)          # (2*H, dim)

    proj = torch.nn.functional.linear(x_h, proj_weight)      # [B,N,N,2*H] FP16
    left, right = proj.chunk(2, dim=-1)                       # each [B,N,N,H]

    # --------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.unsqueeze(-1).to(torch.float16)              # [B,N,N,1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 5. Fuse gating Linear layers (left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    gates_weight = torch.cat(
        [left_gate_w, right_gate_w, out_gate_w], dim=0
    )                                                          # (3*H, dim)

    gates = torch.nn.functional.linear(x_h, gates_weight).sigmoid()
    left_gate, right_gate, out_gate = gates.chunk(3, dim=-1)   # each [B,N,N,H] FP16

    # --------------------------------------------------------------
    # 6. Apply element‑wise gates (still half precision)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7. Prepare tensors for the N³ reduction
    #    Transform from [B, N, N, H] → [B*H, N, N] (contiguous)
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)    # FP16
    right_flat = right.view(B * hidden_dim, N, N)

    # --------------------------------------------------------------
    # 8. Batched Gram product (left @ rightᵀ)
    #    Use custom Triton kernel for moderate problem sizes,
    #    otherwise fall back to torch.bmm (cuBLAS Tensor‑Cores).
    # --------------------------------------------------------------
    BH = B * hidden_dim
    use_triton = (BH <= 256) and (N <= 512)   # heuristic

    if use_triton:
        out_flat = torch.empty_like(left_flat)   # FP16 buffer

        # strides of the flattened tensors (contiguous layout)
        stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
        stride_b_batch, stride_b_k, _ = right_flat.stride()
        stride_c_batch, stride_c_i, _ = out_flat.stride()

        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )
        gram_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            BH,
            stride_a_batch,
            stride_a_i,
            stride_a_k,
            stride_b_batch,
            stride_b_k,
            stride_c_batch,
            stride_c_i,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # torch.bmm uses the highly‑optimised cuBLAS batched GEMM.
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # --------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H) and cast to FP32
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()                                 # [B, N, N, H] FP32

    # --------------------------------------------------------------
    # 10. Output LayerNorm + output gate
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()                      # broadcast multiply

    # --------------------------------------------------------------
    # 11. Final Linear projection back to `dim` (FP32)
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
