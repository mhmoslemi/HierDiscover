"""
Outgoing Triangular Multiplicative Update (TriMul) – Triton implementation.

Algorithm
---------
1️⃣  Layer‑Norm over the feature dimension (float32, stable).
2️⃣  Cast to FP16 and apply linear projections (left/right) and gating
    projections (left_gate/right_gate/out_gate).  Gating values are passed
    through a sigmoid; the output‑gate stays in FP32.
3️⃣  Optional mask is broadcast and multiplied element‑wise with the
    projected tensors (still in FP16).
4️⃣  The cubic O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom Triton kernel that treats each (batch, hidden)
    slice as an independent N×N matrix‑multiply (A @ Bᵀ).  The kernel is a
    classic tiled GEMM (BLOCK_M×BLOCK_N tiles, BLOCK_K reduction) using
    Tensor‑Core‑friendly FP16 × FP16 → FP32 accumulation.
5️⃣  The result (still FP16) is promoted to FP32, Layer‑Norm is applied over
    the hidden dimension, multiplied by the out‑gate, and finally projected
    back to the original feature dimension.

All heavy arithmetic is in FP16 (except the final LayerNorms and output‑gate,
which are FP32).  The Triton kernel is the required “use Triton” part while
the rest leverages fast PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul cubic term
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16  left tensor (B,N,N,H)
    right_ptr,         # fp16  right tensor transposed on the N‑axes
    out_ptr,           # fp16  destination tensor (B,N,N,H)
    N,                 # i64   sequence length
    # strides for the left tensor
    stride_left_batch,  # i64   stride for batch dimension (B)
    stride_left_i,      # i64   stride for i (row) dimension
    stride_left_k,      # i64   stride for k (col) dimension
    # strides for the right (already transposed) tensor
    stride_right_batch, # i64   stride for batch dimension (B)
    stride_right_k,     # i64   stride for k (row) dimension after transpose
    stride_right_j,     # i64   stride for j (col) dimension after transpose
    # strides for the output tensor
    stride_out_batch,   # i64   stride for batch dimension (B)
    stride_out_i,       # i64   stride for i (row) dimension
    stride_out_j,       # i64   stride for j (col) dimension
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    H: tl.constexpr,    # hidden dimension (used to split program_id(2) into batch & head)
):
    """Batched matrix multiply C = A @ Bᵀ for each (batch, head) pair."""
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_z = tl.program_id(2)          # combined batch*head index (0 … B*H‑1)

    # ----------- decode combined batch / head index -------------
    batch_idx = pid_z // H            # actual batch (0 … B‑1)
    head_idx  = pid_z % H             # hidden head (0 … H‑1)

    # Base pointers that already include the head offset (hidden stride = 1)
    left_base  = left_ptr  + batch_idx * stride_left_batch + head_idx
    right_base = right_ptr + batch_idx * stride_right_batch + head_idx
    out_base   = out_ptr   + batch_idx * stride_out_batch + head_idx

    # ----------- tile coordinates ---------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- K‑loop (reduction over the shared dimension) ----
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for the current K‑tile
        # A: left[batch, i, k, head]
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i
                              + offs_k[None, :] * stride_left_k)
        # B: rightᵀ[batch, k, j, head]  (right already transposed)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k
                               + offs_j[None, :] * stride_right_j)

        # Load with masking, promote to fp32 for accumulation
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N)

    # ----------- write back ---------------------------------------
    c_ptrs = out_base + (offs_i[:, None] * stride_out_i
                         + offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H in the paper (always 128 in tests)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for projections / gating
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear weight tensors (cast once)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"]      # keep float32 for out‑gate

    # ------------------------------------------------------------------
    # 3️⃣ Projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right_proj = F.linear(x_h, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Gating values
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).to(torch.float16)

    # out‑gate stays in FP32 (computed from the original FP32 tensor)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # [B, N, N, H] (float32)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is float16 and has a trailing dim for broadcasting
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gating (still in fp16)
    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³) contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Transpose right on the two N axes (no copy, just a view)
    right_t = right.transpose(1, 2)  # shape [B, N, N, H] with swapped strides

    # Allocate output buffer
    out = torch.empty_like(left)  # fp16, same shape

    # Tile sizes – chosen to fit comfortably in shared memory on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute launch grid
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over cols (j)
        B * H,                           # combined batch × head dimension
    )

    # Strides (in elements) for each tensor
    stride_left_batch = left.stride(0)
    stride_left_i     = left.stride(1)
    stride_left_k     = left.stride(2)

    stride_right_batch = right_t.stride(0)
    stride_right_k     = right_t.stride(1)   # corresponds to original k after transpose
    stride_right_j     = right_t.stride(2)   # corresponds to original j after transpose

    stride_out_batch = out.stride(0)
    stride_out_i     = out.stride(1)
    stride_out_j     = out.stride(2)

    # Launch Triton kernel
    _tri_mul_kernel[grid](
        left,
        right_t,
        out,
        N,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_right_batch,
        stride_right_k,
        stride_right_j,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing (LayerNorm over hidden, out‑gate, final proj)
    # ------------------------------------------------------------------
    # Convert to float32 for the remaining layers
    out_fp32 = out.to(torch.float32)  # [B, N, N, H]

    # Hidden‑dim LayerNorm
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H] (float32)

    # Apply out‑gate (broadcast over the hidden dimension)
    out_gated = out_norm * out_gate

    # Final linear projection back to original dim
    final_out = F.linear(out_gated, weights["to_out.weight"])  # [B, N, N, dim]

    return final_out
