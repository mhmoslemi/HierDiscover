def _proj_gate_mask_reorder_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,

"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ LayerNorm on the input feature dimension (float32 for numerical stability).
2️⃣ Cast to FP16 and compute linear projections (left/right) and gating
   (left/right/out) using the supplied weights.
3️⃣ Fuse element‑wise multiplication of projection, gate and *optional* mask
   while simultaneously re‑ordering the data layout from (B, N, N, H) → (B, H, N, N)
   with a single Triton kernel.  This eliminates the costly
   `permute(...).contiguous()` copies present in the naïve PyTorch version.
4️⃣ Perform the core O(N³·H) operation as a batch of independent
   matrix‑multiplications:  C = A @ Bᵀ where A,B ∈ ℝ^{B·H × N × N}.
   We use `torch.bmm`, which on H100 runs as a mixed‑precision Tensor‑Core
   GEMM (FP16×FP16 → FP32 accumulation).
5️⃣ Reshape the result back to (B, N, N, H) and apply the hidden‑dim LayerNorm,
   multiply by the out‑gate and finally project back to the original feature
   dimension with a linear layer.  The final output is float32.
All heavy arithmetic is carried out in FP16; only the two LayerNorms and the
final linear projection use FP32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fused projection·gate·mask multiplication + layout reorder.
# Input layout  : (B, N, N, H)   – contiguous in the last (H) dimension.
# Output layout : (B, H, N, N)   – contiguous (fastest index = H).
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_reorder_kernel(
    proj_ptr,                 # fp16 [] projection (left or right)
    gate_ptr,                 # fp16 [] corresponding gate
    mask_ptr,                 # fp16 [] mask (flattened, size B·N·N)
    out_ptr,                  # fp16 [] destination (B,H,N,N) flattened
    B, N, H,                  # runtime sizes
    TOTAL,                    # total number of output elements = B*H*N*N
    BLOCK: tl.constexpr,      # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices of output
    mask = offs < TOTAL                               # bounds check

    # ---- decode linear index into (b, h, i, j) ----
    b = offs // (H * N * N)
    tmp = offs % (H * N * N)
    h = tmp // (N * N)
    tmp2 = tmp % (N * N)
    i = tmp2 // N
    j = tmp2 % N

    # ---- compute flat offsets for the input tensors (proj & gate) ----
    # original layout: (b, i, j, h) → offset = ((b*N + i)*N + j)*H + h
    proj_offset = ((b * N + i) * N + j) * H + h

    # ---- compute flat offset for the mask (broadcast over H) ----
    # mask layout: (b, i, j) → offset = (b*N + i)*N + j
    mask_offset = (b * N + i) * N + j

    # ---- load values (fp16 → fp32 for higher‑precision multiply) ----
    proj_val = tl.load(proj_ptr + proj_offset, mask=mask).to(tl.float32)
    gate_val = tl.load(gate_ptr + proj_offset, mask=mask).to(tl.float32)
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask).to(tl.float32)

    # ---- fused multiply (proj * gate * mask) ----
    out_val = proj_val * gate_val * mask_val

    # ---- store result in the transposed layout (fp16) ----
    tl.store(out_ptr + offs, out_val.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → no mask (all ones)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                           # [B, N, N, dim], FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and compute linear projections / gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                    # FP16 for speed

    # Linear weight tensors (cast once)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Projections (B,N,N,hidden_dim) – FP16
    left_proj  = F.linear(x_h, left_proj_w)   # -> [B,N,N,hidden]
    right_proj = F.linear(x_h, right_proj_w)  # -> [B,N,N,hidden]

    # Gates (sigmoid, keep FP16)
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w)).float()  # FP32 later

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection·gate·mask and reshape (B, hidden, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape

    if nomask:
        # All‑ones mask – allocate a tiny tensor of ones and reuse it
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype=torch.float16, device=device)

    # Flatten mask for the kernel (size = B·N·N)
    mask_flat = mask_fp16.view(-1).contiguous()

    # Allocate output buffers in the transposed layout
    left_out  = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right_out = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)

    total_elements = B * hidden_dim * N * N
    BLOCK = 65536                                 # ~64 k elements per program
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Left side kernel launch
    _proj_gate_mask_reorder_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        left_out,
        B, N, hidden_dim,
        total_elements,
        BLOCK=BLOCK,
    )
    # Right side kernel launch (identical logic)
    _proj_gate_mask_reorder_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        right_out,
        B, N, hidden_dim,
        total_elements,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) batch GEMM: C = A @ Bᵀ  (FP16×FP16 → FP32 accumulation)
    # ------------------------------------------------------------------
    # Collapse batch and hidden dimensions for torch.bmm
    left_batched  = left_out.view(B * hidden_dim, N, N)                     # (B·H, N, N)
    right_batched = right_out.view(B * hidden_dim, N, N)                    # (B·H, N, N)

    # Batched matrix multiplication (uses Tensor Cores on H100)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))   # (B·H, N, N), FP16

    # Restore the (B, N, N, hidden) shape required by the following layers
    out = (
        out_batched
        .view(B, hidden_dim, N, N)                # (B, H, N, N)
        .permute(0, 2, 3, 1)                     # (B, N, N, H)
        .contiguous()
        .float()                                 # promote to FP32 for LayerNorm
    )                                            # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate, final projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                            # [B, N, N, hidden] FP32

    # Apply the out‑gate (broadcast over the hidden dimension)
    out = out * out_gate

    # Final linear projection to original feature dimension (float32)
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
