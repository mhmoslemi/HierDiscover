                   stride_right_b, stride_right_i, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    b = pid_batch // H
    h = pid_batch % H

    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h

    pid_m = tl.program_id(0)

    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = off_n < N

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
1. Layer‑norm the input (FP16‑compatible) – small tensor, cheap.
2. Compute the five linear projections (left, right, left_gate,
   right_gate, out_gate) + optional mask.  All ops stay in half‑precision.
3. Fuse left/right gating:  left = (x·W_left) * sigmoid(x·W_left_gate) *
   mask   ;   right = (x·W_right) * sigmoid(x·W_right_gate) * mask
4. Batched tensor‑core GEMM across the sequence dimension:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that treats the hidden
   dimension as part of the batch axis (batch = B * hidden_dim).  No
   intermediate transpose/copy is required.
5. Output LayerNorm (FP16) → output gate → final linear projection
   back to the original channel dimension.
6. Return FP32 tensor (the model downstream expects float32).

The kernel processes each (batch, head) pair as an independent
matrix‑multiply using block‑wise tiles (64×64) and a 32‑element K‑tile.
All accumulations are performed in FP32 for numerical stability and are
cast back to FP16 only when stored.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *float16
    right_ptr,           # *float16
    out_ptr,             # *float16
    B: tl.int32, N: tl.int32, H: tl.int32,
    stride_l_b: tl.int32, stride_l_i: tl.int32, stride_l_k: tl.int32, stride_l_h: tl.int32,
    stride_r_b: tl.int32, stride_r_i: tl.int32, stride_r_k: tl.int32, stride_r_h: tl.int32,
    stride_o_b: tl.int32, stride_o_i: tl.int32, stride_o_j: tl.int32, stride_o_h: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # 0‑dim: tile over rows (i), 1‑dim: tile over cols (j), 2‑dim: batch*head
    # ------------------------------------------------------------------
    pid_h = tl.program_id(2)                     # 0 .. B*H-1
    batch_id = pid_h // H
    head_id  = pid_h % H

    # base pointers for this (batch, head) pair
    left_base  = left_ptr + batch_id * stride_l_b + head_id * stride_l_h
    right_base = right_ptr + batch_id * stride_r_b + head_id * stride_r_h
    out_base   = out_ptr + batch_id * stride_o_b + head_id * stride_o_h

    pid_m = tl.program_id(0)                     # row tile index
    pid_n = tl.program_id(1)                     # col tile index

    # offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
            + offs_m[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        a = tl.load(left_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)      # fp16

        # right block (transposed): shape (BLOCK_K, BLOCK_N)
        # we load right[j,k] as (k,j) to match tl.dot(A,B) which expects (M,K)*(K,N)
        right_ptrs = right_base \
            + offs_k[:, None] * stride_r_k \
            + offs_n[None, :] * stride_r_i
        b = tl.load(right_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)      # fp16

        # Accumulate into FP32
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_N) FP32
    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_base \
        + offs_m[:, None] * stride_o_i \
        + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel and returns the contracted tensor.
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Both `left` and `right` must be half‑precision tensors with shape
    [B, N, N, H] (contiguous).  The result is a half‑precision tensor of
    the same shape.
    """
    B, N, _, H = left.shape
    # ensure contiguity (no extra copy of the 4‑D view)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float16)

    # strides in *elements* (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile size – tuned for fp16 tensor cores on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # give the kernel enough warps for H100
    )
    return out


# ----------------------------------------------------------------------
# Public entry point – matches the required signature.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of model weights
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0. Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Helper to cast a weight to FP16 on the correct device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 2. Load model parameters (convert to half‑precision where possible)
    # ------------------------------------------------------------------
    norm_w = to_half(weights["norm.weight"])
    norm_b = to_half(weights["norm.bias"])

    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])

    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])

    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 2. Input layer‑norm (keep in FP16 for the rest of the forward)
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16)
    x_norm = F.layer_norm(
        x_fp16,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C]  fp16

    # ------------------------------------------------------------------
    # 3. Linear projections + optional mask
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_w) # [B, N, N, hidden_dim]

    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating paths (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via the Triton batched GEMM
    # ------------------------------------------------------------------
    out = tri_mul_triton(left, right)   # [B, N, N, hidden_dim]  fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (still in FP16) + output gate
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    gated = out_norm * out_gate  # element‑wise gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    final_fp16 = F.linear(gated, to_out_w)   # [B, N, N, C]  fp16

    # ------------------------------------------------------------------
    # 8. Return float32 tensor as required by the spec
    # ------------------------------------------------------------------
    return final_fp16.to(torch.float32)
