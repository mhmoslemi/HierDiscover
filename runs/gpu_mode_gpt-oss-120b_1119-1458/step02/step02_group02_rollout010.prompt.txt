import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
=======================================================

Algorithm
---------
1. **Input layer‑norm** – Layer‑norm on the last axis of the input tensor.
2. **Fused projection & gating** – All five linear layers
   (left_proj, left_gate, right_proj, right_gate, out_gate) are merged
   into a *single* matrix multiplication by concatenating their weight
   matrices vertically.  This reduces the number of passes over the
   huge (B·N·N) input from five to one and keeps the data in FP16.
3. **Mask & gate fusion** – If a mask is present, a tiny Triton kernel
   computes `out = a * b * c` for three 1‑D tensors, fusing the three
   element‑wise multiplications (mask, projection, gate) into one memory
   pass.
4. **Core contraction** – The outgoing update
   `out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]`
   is a batched matrix‑multiply for each hidden channel.  We reshape the
   tensors to `(B*hidden, N, N)` and invoke a single `torch.bmm`
   (FP16) which uses Tensor‑cores on the H100.
5. **Output layer‑norm + gating + final linear** – The result is
   layer‑normed over the hidden dimension, multiplied by the output
   gate, and linearly projected back to the original channel dimension.
   All operations stay in FP16 until the final cast back to the input
   dtype.

The implementation uses only one custom Triton kernel (`fused_mul3`)
for the triple element‑wise product; all heavy linear algebra is performed
with highly‑optimized torch kernels in half‑precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = a * b * c  (all tensors 1‑D contiguous)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,     # pointers
    size: tl.constexpr,               # total length
    BLOCK_SIZE: tl.constexpr,         # compile‑time constant
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape, "All inputs must have the same shape"
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()

    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192                     # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor:  shape [B, N, N] (bool or float, on CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim",
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (FP32 → FP16 for the rest)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)      # half‑precision for the rest

    # ------------------------------------------------------------------
    # 2) Fused projection + all five gates
    # ------------------------------------------------------------------
    # Stack weight matrices vertically: each weight has shape (hidden, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                               # (5*hidden, dim)
    ).to(torch.float16)

    # Single linear layer yields all projections and gate logits at once
    # output shape: [B, N, N, 5*hidden]
    combined = F.linear(x_norm, w_concat)   # no bias

    # Split the concatenated dimension back into the five parts
    split = torch.split(combined, hidden, dim=-1)
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = split

    # Sigmoid gates (still in half)
    left_gate  = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate   = torch.sigmoid(out_gate_logit)

    # ------------------------------------------------------------------
    # 3) Apply optional mask and gates (fused element‑wise multiplication)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple two‑operand product
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Broadcast mask to the hidden dimension and make it contiguous
        mask_exp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden).contiguous()
        left  = fused_mul3(left_proj_raw, mask_exp, left_gate)
        right = fused_mul3(right_proj_raw, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 4) Core contraction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    # Shapes → (B, hidden, N, N) for both tensors
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # swap k and j → [B, hidden, N, N]

    # Collapse batch & hidden into a single batch dimension for torch.bmm
    left_flat  = left_mat.view(B * hidden, N, N)       # (B*hidden, N, N)
    right_flat = right_mat.view(B * hidden, N, N)      # (B*hidden, N, N)

    # Batched matrix multiplication in FP16 (Tensor‑cores)
    out_flat = torch.bmm(left_flat, right_flat)        # (B*hidden, N, N)

    # Reshape back to the original 4‑D layout
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5) Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    # Layer‑norm over the hidden dimension (still half‑precision)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # Apply the output gate (pair‑wise multiplication)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to the original dtype (float32) and return
    return out_final.to(inp.dtype)
