   start_j = block_j * BLOCK_N

   offs_j = start_j + tl.arange(0, BLOCK_N)

def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BH, N,
    stride_A_b, stride_A_i, stride_A_k,
    stride_B_b, stride_B_j, stride_B_k,
    stride_C_b, stride_C_i, stride_C_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch dimension
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # mask for valid indices
        mask_i = offs_i < N
        mask_j = offs_j < N

    # store

    tl.store(C_ptr + pid_b * stride_C_b + offs_i[:, None] * stride_C_i + offs_j[None, :] * stride_C_j,

@triton.jit
def proj_gate_fused_kernel(
    x_ptr,          # (M, D) half
    left_proj_T_ptr, left_gate_T_ptr,
    right_proj_T_ptr, right_gate_T_ptr,
    mask_ptr,        # (M) half
    left_out_ptr, right_out_ptr,   # (M, H) half
    M, D, H,
    stride_xm, stride_xd,
    stride_lp_Td, stride_lp_Th,
    stride_lg_Td, stride_lg_Th,
    stride_rp_Td, stride_rp_Th,
    stride_rg_Td, stride_rg_Th,
    stride_left_out_m, stride_left_out_h,
    stride_right_out_m, stride_right_out_h,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)

    row_start = pid * BLOCK_M
    col_start = pid_h * BLOCK_H

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_h = col_start + tl.arange(0, BLOCK_H)

    mask_row = offs_m < M
    mask_col = offs_h < H

    # Init accumulators
    left_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    left_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    right_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    right_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Loop over D dimension
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        d_start = d * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # Load left_proj_T block (BLOCK_D, BLOCK_H)
        w_lp = tl.load(left_proj_T_ptr + (offs_d[:, None] * stride_lp_Td + offs_h[None, :] * stride_lp_Th),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_lg = tl.load(left_gate_T_ptr + (offs_d[:, None] * stride_lg_Td + offs_h[None, :] * stride_lg_Th),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_rp = tl.load(right_proj_T_ptr + (offs_d[:, None] * stride_rp_Td + offs_h[None, :] * stride_rp_Th),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_rg = tl.load(right_gate_T_ptr + (offs_d[:, None] * stride_rg_Td + offs_h[None, :] * stride_rg_Th),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)

        # Load X tile (BLOCK_M, BLOCK_D)
        x_tile = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd),
                         mask=mask_row[:, None] & mask_d[None, :], other=0.0)

        # Accumulate
        left_proj_acc += tl.dot(x_tile, w_lp)   # shape (BLOCK_M, BLOCK_H)
        left_gate_acc += tl.dot(x_tile, w_lg)
        right_proj_acc += tl.dot(x_tile, w_rp)
        right_gate_acc += tl.dot(x_tile, w_rg)

    # Sigmoid for gates
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_acc))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_acc))

    # Load mask (broadcast over hidden)
    if mask_ptr != 0:
        mask_vals = tl.load(mask_ptr + offs_m, mask=mask_row, other=0.0)  # half, upcast to float32
        mask_f = mask_vals.to(tl.float32)[:, None]  # shape (BLOCK_M,1)
    else:
        mask_f = tl.full((BLOCK_M, 1), 1.0, dtype=tl.float32)

    # Apply mask to proj before gating
    left = left_proj_acc * mask_f * left_gate
    right = right_proj_acc * mask_f * right_gate

    # Store results as half
    tl.store(left_out_ptr + offs_m[:, None] * stride_left_out_m + offs_h[None, :] * stride_left_out_h,
             left.to(tl.float16), mask=mask_row[:, None] & mask_col[None, :])
    tl.store(right_out_ptr + offs_m[:, None] * stride_right_out_m + offs_h[None, :] * stride_right_out_h,
             right.to(tl.float16), mask=mask_row[:, None] & mask_col[None, :])
```

@triton.jit
def proj_gate_fused_kernel(
    x_ptr, left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    M, D, H,
    stride_xm, stride_xd,
    stride_lp_d, stride_lp_h,
    stride_lg_d, stride_lg_h,
    stride_rp_d, stride_rp_h,
    stride_rg_d, stride_rg_h,
    stride_mask_m,
    stride_left_out_m, stride_left_out_h,
    stride_right_out_m, stride_right_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)

    row_start = pid * BLOCK_M
    col_start = pid_h * BLOCK_H

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_h = col_start + tl.arange(0, BLOCK_H)

    mask_row = offs_m < M
    mask_col = offs_h < H

    # accumulator
    left_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    left_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    right_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    right_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for d in range(0, tl.cdiv(D, BLOCK_D)):
        d_start = d * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # Load weight tiles
        w_lp = tl.load(left_proj_ptr + (offs_d[:, None] * stride_lp_d + offs_h[None, :] * stride_lp_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_lg = tl.load(left_gate_ptr + (offs_d[:, None] * stride_lg_d + offs_h[None, :] * stride_lg_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_rp = tl.load(right_proj_ptr + (offs_d[:, None] * stride_rp_d + offs_h[None, :] * stride_rp_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_rg = tl.load(right_gate_ptr + (offs_d[:, None] * stride_rg_d + offs_h[None, :] * stride_rg_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)

        # Load x tile
        x = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd),
                    mask=mask_row[:, None] & mask_d[None, :], other=0.0)

        # Accumulate matmul contributions
        left_proj_acc += tl.dot(x, w_lp)
        left_gate_acc += tl.dot(x, w_lg)
        right_proj_acc += tl.dot(x, w_rp)
        right_gate_acc += tl.dot(x, w_rg)

    # sigmoid gates
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_acc))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_acc))

    # Load mask values (broadcast over hidden)
    mask_vals = tl.load(mask_ptr + offs_m, mask=mask_row, other=0.0)
    mask_f = mask_vals.to(tl.float32)[:, None]   # shape (BLOCK_M,1)

    # Apply mask before gating (as in reference)
    left = left_proj_acc * mask_f * left_gate
    right = right_proj_acc * mask_f * right_gate

    # Store results as fp16
    tl.store(left_out_ptr + offs_m[:, None] * stride_left_out_m + offs_h[None, :] * stride_left_out_h,
             left.to(tl.float16), mask=mask_row[:, None] & mask_col[None, :])
    tl.store(right_out_ptr + offs_m[:, None] * stride_right_out_m + offs_h[None, :] * stride_right_out_h,
             right.to(tl.float16), mask=mask_row[:, None] & mask_col[None, :])
```

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module optimized with fused Triton kernels.
    Steps:
      1. LayerNorm over the input (float32 for stability).
      2. Cast to fp16.
      3. Fuse projection, gating, and mask for both left and right tensors using a single Triton kernel.
      4. Compute the N³ contraction out = left @ rightᵀ for each batch‑head slice using a custom Triton batched GEMM.
      5. Apply a second LayerNorm, an output gate, and the final linear projection (all in fp16).
      6. Return result in float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # ----------- LayerNorm (float32) ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=1e-5)  # (B,N,N,dim) float32
    
    # Cast to half for heavy ops
    x_half = x.to(torch.float16)
    
    # Flatten for projection kernel
    M = B * N * N
    x_flat = x_half.reshape(M, dim).contiguous()
    
    # Prepare mask
    if mask is None:
        mask_flat = torch.ones((M,), dtype=torch.float16, device=device)
    else:
        mask_flat = mask.reshape(M).to(torch.float16)
    # Ensure contiguous
    mask_flat = mask_flat.contiguous()
    
    # ----------- Load and transpose weights (half) ----------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)
    left_proj_T = to_half(weights["left_proj.weight"]).t().contiguous()
    right_proj_T = to_half(weights["right_proj.weight"]).t().contiguous()
    left_gate_T = to_half(weights["left_gate.weight"]).t().contiguous()
    right_gate_T = to_half(weights["right_gate.weight"]).t().contiguous()
    out_gate_w = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w = to_half(weights["to_out.weight"])
    
    # Output buffers for left/right
    left_out = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right_out = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    
    # ----------- Launch fused projection + gate kernel ----------
    BLOCK_M = 128
    BLOCK_H = 64   # hidden dimension tiling
    BLOCK_D = 32   # input feature tiling
    
    grid = ((M + BLOCK_M - 1) // BLOCK_M,
            (hidden_dim + BLOCK_H - 1) // BLOCK_H)
    
    proj_gate_fused_kernel[grid](
        x_flat,                         # x_ptr
        left_proj_T, left_gate_T,       # left proj & gate
        right_proj_T, right_gate_T,     # right proj & gate
        mask_flat,                      # mask_ptr
        left_out, right_out,            # out ptrs
        M, dim, hidden_dim,
        x_flat.stride(0), x_flat.stride(1),
        left_proj_T.stride(0), left_proj_T.stride(1),
        left_gate_T.stride(0), left_gate_T.stride(1),
        right_proj_T.stride(0), right_proj_T.stride(1),
        right_gate_T.stride(0), right_gate_T.stride(1),
        mask_flat.stride(0),
        left_out.stride(0), left_out.stride(1),
        right_out.stride(0), right_out.stride(1),
        BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H, BLOCK_D=BLOCK_D
    )
    
    # Reshape left/right back to [B, N, N, hidden]
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)
    
    # ----------- N³ Contraction via batched GEMM ----------
    # Rearrange to (B, hidden, N, N) and flatten batch*head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)
    
    BLOCK_MG = 128
    BLOCK_NG = 128
    BLOCK_KG = 32
    
    grid_gemm = (BH,
                 (N + BLOCK_MG - 1) // BLOCK_MG,
                 (N + BLOCK_NG - 1) // BLOCK_NG)
    
    batched_gemm_kernel[grid_gemm](
        left_flat, right_flat, out_flat,
        BH, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_MG, BLOCK_N=BLOCK_NG, BLOCK_K=BLOCK_KG
    )
    
    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    
    # ----------- Output LayerNorm (float32) ----------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w.to(torch.float32),
                                 bias=to_out_norm_b.to(torch.float32),
                                 eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)
    
    # ----------- Output gate ----------
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))   # shape (B,N,N,hidden)
    gated = out_norm * out_gate
    
    # ----------- Final linear projection ----------
    final = F.linear(gated, to_out_w)   # (B,N,N,dim) fp16
    
    return final.to(torch.float32)

    pid_h = tl.program_id(1)

    row_start = pid * BLOCK_M
    col_start = pid_h * BLOCK_H

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_h = col_start + tl.arange(0, BLOCK_H)

    mask_row = offs_m < M
    mask_col = offs_h < H

    left_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    right_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for d in range(0, tl.cdiv(D, BLOCK_D)):
        d_start = d * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # Load weight tiles
        w_lp = tl.load(left_proj_ptr + (offs_d[:, None] * stride_lp_d + offs_h[None, :] * stride_lp_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_lg = tl.load(left_gate_ptr + (offs_d[:, None] * stride_lg_d + offs_h[None, :] * stride_lg_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_rp = tl.load(right_proj_ptr + (offs_d[:, None] * stride_rp_d + offs_h[None, :] * stride_rp_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)

                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)

        # Load X tile
        x = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd),
                    mask=mask_row[:, None] & mask_d[None, :], other=0.0)

        left_proj_acc += tl.dot(x, w_lp)
        left_gate_acc += tl.dot(x, w_lg)
        right_proj_acc += tl.dot(x, w_rp)
        right_gate_acc += tl.dot(x, w_rg)

    # Sigmoid gates (float32)
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_acc))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_acc))

    mask_vals = tl.load(mask_ptr + offs_m * stride_mask_m,

    left = left_proj_acc * mask_f * left_gate

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm (all heavy work in Triton):
  1. Layer‑norm the 4‑D input (float32 for stability).
  2. Cast to fp16 and flatten to (M, dim) where M = B·N·N.
  3. A single Triton kernel fuses:
       – left & right linear projections (weight^T)
       – left & right sigmoid gates
       – optional pair‑wise mask (broadcast over the hidden dim)
     producing two tensors L,R of shape (M, hidden) in fp16.
  4. Reshape L,R → (B, hidden, N, N) and run a batched GEMM kernel that
     computes   out_{b,h,i,j} = Σ_k L_{b,h,i,k}·R_{b,h,j,k}
     i.e. a per‑head matrix‑multiply with the transpose of R.
  5. Reshape the result back to (B, N, N, hidden) (fp16).
  6. Apply a second LayerNorm over the hidden dimension (float32),
     an output gate (sigmoid of a linear projection) and a final linear
     projection back to the original channel dimension.
  7. Return the result as float32.

Both Triton kernels are tiled (BLOCK_M×BLOCK_H×BLOCK_D for the projection/
gate fusion and BLOCK_M×BLOCK_N×BLOCK_K for the batched GEMM) and use
fp16 arithmetic with fp32 accumulation, fully exploiting H100 TensorCores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# 1. Fuse left/right projection, gating and mask in a single kernel.
# -------------------------------------------------------------------------
@triton.jit
def proj_gate_fused_kernel(
    x_ptr,                     # (M, D)          fp16 *
    left_proj_ptr, left_gate_ptr,    # (D, H)      fp16 *
    right_proj_ptr, right_gate_ptr,  # (D, H)      fp16 *
    mask_ptr,                  # (M)             fp16 *
    left_out_ptr, right_out_ptr,    # (M, H)      fp16 *
    M, D, H,
    stride_xm, stride_xd,
    stride_lp_d, stride_lp_h,
    stride_lg_d, stride_lg_h,
    stride_rp_d, stride_rp_h,
    stride_rg_d, stride_rg_h,
    stride_mask_m,
    stride_left_out_m, stride_left_out_h,
    stride_right_out_m, stride_right_out_h,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)          # block over rows   (M)
    pid_h = tl.program_id(1)        # block over hidden (H)

    row_start = pid * BLOCK_M
    col_start = pid_h * BLOCK_H

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_h = col_start + tl.arange(0, BLOCK_H)

    mask_row = offs_m < M
    mask_col = offs_h < H

    # accumulators (float32 for numerical safety)
    left_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    left_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    right_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    right_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # iterate over the input feature dimension (D)
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        d_start = d * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # weight tiles (D‑block × H‑block)
        w_lp = tl.load(left_proj_ptr + (offs_d[:, None] * stride_lp_d + offs_h[None, :] * stride_lp_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_lg = tl.load(left_gate_ptr + (offs_d[:, None] * stride_lg_d + offs_h[None, :] * stride_lg_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_rp = tl.load(right_proj_ptr + (offs_d[:, None] * stride_rp_d + offs_h[None, :] * stride_rp_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)
        w_rg = tl.load(right_gate_ptr + (offs_d[:, None] * stride_rg_d + offs_h[None, :] * stride_rg_h),
                       mask=mask_d[:, None] & mask_col[None, :], other=0.0)

        # input tile (M‑block × D‑block)
        x = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd),
                    mask=mask_row[:, None] & mask_d[None, :], other=0.0)

        # matmul‑accumulate (automatically up‑casts the fp16 product to fp32)
        left_proj_acc += tl.dot(x, w_lp)
        left_gate_acc += tl.dot(x, w_lg)
        right_proj_acc += tl.dot(x, w_rp)
        right_gate_acc += tl.dot(x, w_rg)

    # sigmoid gates (still float32)
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_acc))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_acc))

    # mask (broadcast over hidden)
    mask_vals = tl.load(mask_ptr + offs_m * stride_mask_m, mask=mask_row, other=0.0)
    mask_fp32 = mask_vals.to(tl.float32)[:, None]          # (BLOCK_M, 1)

    # apply mask * gate *
    left = left_proj_acc * mask_fp32 * left_gate
    right = right_proj_acc * mask_fp32 * right_gate

    # write back as fp16
    tl.store(left_out_ptr + offs_m[:, None] * stride_left_out_m + offs_h[None, :] * stride_left_out_h,
             left.to(tl.float16), mask=mask_row[:, None] & mask_col[None, :])
    tl.store(right_out_ptr + offs_m[:, None] * stride_right_out_m + offs_h[None, :] * stride_right_out_h,
             right.to(tl.float16), mask=mask_row[:, None] & mask_col[None, :])


# -------------------------------------------------------------------------
# 2. Batched GEMM: out = L @ Rᵀ  (one GEMM per batch*head)
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BH, N,
    stride_A_b, stride_A_i, stride_A_k,
    stride_B_b, stride_B_i, stride_B_k,
    stride_C_b, stride_C_i, stride_C_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # which (batch,head) pair
    pid_i = tl.program_id(1)     # i‑tile
    pid_j = tl.program_id(2)     # j‑tile

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k (the shared sequence dimension)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[batch, i, k]
        a = tl.load(A_ptr + batch_id * stride_A_b
                    + offs_i[:, None] * stride_A_i
                    + offs_k[None, :] * stride_A_k,
                    mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ block = B[batch, j, k]  (note the swapped strides)
        b = tl.load(B_ptr + batch_id * stride_B_b
                    + offs_k[:, None] * stride_B_k
                    + offs_j[None, :] * stride_B_i,
                    mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result as fp16
    c = acc.to(tl.float16)
    tl.store(C_ptr + batch_id * stride_C_b
             + offs_i[:, None] * stride_C_i
             + offs_j[None, :] * stride_C_j,
             c, mask=mask_i[:, None] & mask_j[None, :])

# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : model parameters as torch.Tensors
        - config_dict  : at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and flatten for the fused projection kernel
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    M = B * N * N
    x_flat = x_fp16.reshape(M, dim).contiguous()

    # ------------------------------------------------------------------
    # 3️⃣ Mask handling (broadcast to hidden dim later)
    # ------------------------------------------------------------------
    if mask is None:
        mask_flat = torch.ones((M,), dtype=torch.float16, device=device)
    else:
        mask_flat = mask.reshape(M).to(torch.float16)
    mask_flat = mask_flat.contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Prepare weight matrices (transpose for matmul, fp16)
    # ------------------------------------------------------------------
    to_fp16 = lambda t: t.to(device=device, dtype=torch.float16)

    left_proj_T = to_fp16(weights["left_proj.weight"]).t().contiguous()
    right_proj_T = to_fp16(weights["right_proj.weight"]).t().contiguous()
    left_gate_T = to_fp16(weights["left_gate.weight"]).t().contiguous()
    right_gate_T = to_fp16(weights["right_gate.weight"]).t().contiguous()
    out_gate_w = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 5️⃣ Allocate output buffers for left/right projections
    # ------------------------------------------------------------------
    left_out = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right_out = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6️⃣ Launch fused projection+gate kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128          # rows of the flattened (B·N²) dimension
    BLOCK_H = 64           # hidden‑dim tiling
    BLOCK_D = 32           # input‑feature tiling (dim)

    grid = ((M + BLOCK_M - 1) // BLOCK_M,
            (hidden_dim + BLOCK_H - 1) // BLOCK_H)

    proj_gate_fused_kernel[grid](
        x_flat,
        left_proj_T, left_gate_T,
        right_proj_T, right_gate_T,
        mask_flat,
        left_out, right_out,
        M, dim, hidden_dim,
        x_flat.stride(0), x_flat.stride(1),
        left_proj_T.stride(0), left_proj_T.stride(1),
        left_gate_T.stride(0), left_gate_T.stride(1),
        right_proj_T.stride(0), right_proj_T.stride(1),
        right_gate_T.stride(0), right_gate_T.stride(1),
        mask_flat.stride(0),
        left_out.stride(0), left_out.stride(1),
        right_out.stride(0), right_out.stride(1),
        BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H, BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape to [B, N, N, hidden]
    # ------------------------------------------------------------------
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8️⃣ N³ contraction via batched GEMM (out = left @ rightᵀ per head)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    BLOCK_MG = 128
    BLOCK_NG = 128
    BLOCK_KG = 32

    grid_gemm = (BH,
                 (N + BLOCK_MG - 1) // BLOCK_MG,
                 (N + BLOCK_NG - 1) // BLOCK_NG)

    batched_gemm_kernel[grid_gemm](
        left_flat, right_flat, out_flat,
        BH, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_MG, BLOCK_N=BLOCK_NG, BLOCK_K=BLOCK_KG,
    )

    # Back to (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Output LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 🔟 Output gate and final projection back to original dim
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B,N,N,hidden) fp16
    gated = out_norm * out_gate

    final = F.linear(gated, to_out_w)   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 11️⃣ Return in float32 as required
    # ------------------------------------------------------------------
    return final.to(torch.float32)
