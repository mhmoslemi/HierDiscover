def postprocess_fused_kernel(
    out_ptr,         # fp16, shape [B, N, N, H]
    out_gate_ptr,    # fp16 shape [B, N, N, H]
    w_norm_ptr,      # fp16, [H]
    b_norm_ptr,      # fp16, [H]
    w_out_ptr,       # fp16, [dim, H]
    final_ptr,       # fp16, shape [B, N, N, dim]
    B, N, H, dim,
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0) # position
    bid = tl.program_id(1) # output dim block

    # Compute position indices
    total_positions = B * N * N
    pos = pid
    b = pos // (N * N)
    tmp = pos % (N * N)
    i = tmp // N
    j = tmp % N

    # offsets
    out_offset = ((b * N * N + i * N + j) * H)
    out_gate_offset = out_offset

    # ranges
    h_range = tl.arange(0, H)
    # Load out values and gate values
    out_vec = tl.load(out_ptr + out_offset + h_range)
    out_gate_vec = tl.load(out_gate_ptr + out_gate_offset + h_range)

    # LayerNorm compute mean and var
    sum_val = tl.sum(out_vec)
    mean = sum_val * (1.0 / H)

    diff = out_vec - mean
    var = tl.sum(diff * diff) * (1.0 / H)
    inv_std = tl.rsqrt(var + eps)

    # Apply weight and bias
    w_norm_val = tl.load(w_norm_ptr + h_range)
    b_norm_val = tl.load(b_norm_ptr + h_range)
    normed = diff * inv_std
    normed = normed * w_norm_val + b_norm_val

    # Gate multiplication
    gated = normed * out_gate_vec

    # Compute output block
    d_start = bid * BLOCK_D
    # compute actual block size
    d_range = tl.arange(0, BLOCK_D) + d_start
    mask = d_range < dim

    # Load weight slice (H, BLOCK_D)
    # pointer shape
    # w_out_ptr: base pointer for matrix dim x H (row-major)
    # offset = d * H + h. So transposed load uses offset = h + d * H
    w_slice = tl.load(
        w_out_ptr + h_range[:, None] + d_range[None, :] * H,
        mask=mask[None, :],
        other=0.0,
    )  # shape (H, BLOCK_D)

    # Perform dot: gated (1, H) dot w_slice (H, BLOCK_D) => (1, BLOCK_D)
    gated_2d = gated[None, :]  # shape (1, H)
    out_block = tl.dot(gated_2d, w_slice)  # shape (1, BLOCK_D)

    # Store to final output
    # compute final offset base: ((b * N * N + i * N + j) * dim) + d_start
    final_offset = ((b * N * N + i * N + j) * dim) + d_start
    # store
    tl.store(final_ptr + final_offset + d_range, out_block[0], mask=mask)

def apply_gates_and_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim,  # compile-time constant
    numel,      # total number of elements (positions*hidden_dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    # compute mask index: pos_idx = offs // hidden_dim
    pos_idx = offs // hidden_dim
    # compute mask offset
    # mask_ptr shape: [total_positions]
    mask_val = tl.load(mask_ptr + pos_idx, mask=mask, other=0.0)
    # expand mask_val across hidden dim -> we need to broadcast.
    # We can multiply elementwise as left * left_gate * mask_val.
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # Multiply left and right
    l = l * lg * mask_val
    r = r * rg * mask_val
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)
```

    B, N, _, dim = input_tensor.shape ?? actually shape is [bs, seq_len, seq_len, dim]

    # Input LN
    x = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],

        right = right * mask_h

    right = right.contiguous()

"""
tri_mul_triton.py

Highly‑optimized Triton implementation of the *outgoing* Triangle Multiplicative
Update (TriMul) used in AlphaFold‑3 and related protein‑structure models.

Key ideas
---------
1. **Fuse the five linear layers** (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single GEMM.  The input tensor
   ``x_h`` (shape [B,N,N,C]) is read only once and multiplied with a
   weight matrix of shape (5*H, C).  The result is then sliced into the
   five logical tensors, cutting the memory traffic for the large
   ``B·N²·C`` matrix.

2. **Fusion of the element‑wise gating** (mask, left/right gates) in one
   tiny Triton kernel.  The kernel walks the flattened ``[B·N²·H]`` data,
   applies the optional mask, multiplies by the respective gate and writes
   the result back in‑place.

3. **Batched N³ contraction** – the core operation
   ``out = Σ_k left[...,k,:] * right[...,k,:]`` – is expressed as a
   batched matrix multiplication ``left_mat @ right_matᵀ`` using a single
   strided‑batched ``torch.bmm`` call (FP16, Tensor‑Core friendly).

4. **Post‑processing fusion** – the second LayerNorm, the output‑gate
   multiplication and the final projection back to ``dim`` are performed
   inside a single Triton kernel.  For each position ``p = (b,i,j)`` the
   kernel:
        * computes the mean/variance over the hidden dimension,
        * applies the LayerNorm weight/bias,
        * multiplies by ``out_gate[p]``,
        * performs a small dot‑product with the final weight matrix
          (shape ``dim × H``) to produce the final ``dim``‑dimensional
          vector.

   This eliminates two separate kernel launches and two large memory
   writes/reads.

The overall data flow (FP16) is

    x ─► LayerNorm ─► cast fp16
        ├─► combined linear (5×H outputs) ─► split
        │   ├─► left       ─► gate + mask
        │   ├─► right      ─► gate + mask
        │   └─► out_gate   (sigmoid)
        └─► left/right  (batch GEMM) ─► out (B,N,N,H)

    out ─► fused post‑processing kernel ─► final (B,N,N,dim) (float32)

The implementation respects the required signature ``custom_kernel(data)``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Tiny Triton kernels
# --------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,            # fp16, flattened [B*N*N*H]
    left_gate_ptr,       # fp16, same shape
    right_ptr,           # fp16, same shape
    right_gate_ptr,      # fp16, same shape
    numel: tl.int64,     # total elements = B*N*N*H
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        left  = left  * left_gate
        right = right * right_gate
    All tensors are flattened beforehand.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l   = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    lg  = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r   = tl.load(right_ptr      + offs, mask=mask, other=0.0)
    rg  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(left_ptr  + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def postprocess_fused_kernel(
    out_ptr,            # fp16, flattened [B*N*N*H]   – result of the N³ GEMM
    out_gate_ptr,       # fp16, flattened [B*N*N*H]   – sigmoid(out_gate)
    w_norm_ptr,         # fp16, [H]   – LayerNorm weight
    b_norm_ptr,         # fp16, [H]   – LayerNorm bias
    w_out_ptr,          # fp16, [dim, H]  – final linear weight
    final_ptr,          # fp16, flattened [B*N*N*dim] – output buffer
    H: tl.constexpr,    # hidden dimension (e.g. 128)
    DIM: tl.constexpr,  # output dimension (e.g. 128 or 384)
    EPS: tl.constexpr,  # epsilon for LayerNorm
    BLOCK_D: tl.constexpr, # output‑dim block size (e.g. 64)
):
    """
    For each position p = (b,i,j) (flattened index ``pid``) compute

        out_norm = (out - mean) / sqrt(var + eps) * w_norm + b_norm
        gated    = out_norm * out_gate
        final    = gated @ w_out.T

    The kernel processes ``BLOCK_D`` output dimensions per launch.
    """
    pid = tl.program_id(0)          # position index   in [0, B*N*N)
    dim_bid = tl.program_id(1)      # output‑dim block index

    # ------------------------------------------------------------------
    # 1) Load the hidden‑dim vector for this position (fp16 → fp32 for
    #    stable statistics)
    # ------------------------------------------------------------------
    h_idx = tl.arange(0, H)                # [0 .. H)
    off_vec = pid * H + h_idx

    out_vec      = tl.load(out_ptr      + off_vec)          # fp16
    out_gate_vec = tl.load(out_gate_ptr + off_vec)          # fp16

    # Cast to fp32 for the reduction (more accurate mean/var)
    out_f32 = out_vec.to(tl.float32)

    # ------------------------------------------------------------------
    # 2) LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    sum_val = tl.sum(out_f32)                     # scalar fp32
    mean    = sum_val * (1.0 / H)

    diff    = out_f32 - mean
    var     = tl.sum(diff * diff) * (1.0 / H)
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Normalised tensor (still fp32) then apply weight / bias (cast back)
    w_norm = tl.load(w_norm_ptr + h_idx)          # fp16
    b_norm = tl.load(b_norm_ptr + h_idx)          # fp16
    normed_fp16 = (diff * inv_std).to(tl.float16)    # back to fp16
    normed_fp16 = normed_fp16 * w_norm + b_norm

    # ------------------------------------------------------------------
    # 3) Apply the output gate
    # ------------------------------------------------------------------
    gated = normed_fp16 * out_gate_vec                # fp16, shape [H]

    # ------------------------------------------------------------------
    # 4) Final linear projection (gated  @  w_out.T)
    #    Compute a block of ``BLOCK_D`` output dimensions.
    # ------------------------------------------------------------------
    d_start = dim_bid * BLOCK_D
    d_range = tl.arange(0, BLOCK_D) + d_start
    d_mask  = d_range < DIM

    # Load the corresponding slice of the weight matrix.
    # w_out is stored row‑major: row = output dim, column = hidden dim.
    # We need it transposed -> shape (H, BLOCK_D)
    w_slice = tl.load(
        w_out_ptr + h_idx[:, None] + d_range[None, :] * H,
        mask=d_mask[None, :],
        other=0.0,
    )   # shape (H, BLOCK_D), fp16

    # ``gated`` has shape (H,).  Turn it into (1, H) for tl.dot.
    gated_2d = gated[None, :]                     # (1, H)
    out_block = tl.dot(gated_2d, w_slice)        # (1, BLOCK_D)

    # ------------------------------------------------------------------
    # 5) Store the result
    # ------------------------------------------------------------------
    out_off = pid * DIM + d_start
    tl.store(final_ptr + out_off + d_range, out_block[0], mask=d_mask)


# --------------------------------------------------------------
# Triton‑accelerated forward pass
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask          : torch.Tensor      [B, N, N]      (bool / float32)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   contains 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]                # input / output feature dimension
    hidden_dim = config["hidden_dim"]         # hidden dimension (always 128 in tests)
    nomask     = config.get("nomask", True)
    eps        = 1e-5
    device     = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical safety)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]   fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                 # [B, N, N, dim]   fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    #   order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                               # concatenate over out_features
    ).to(torch.float16)                       # shape (5*hidden_dim, dim)

    # Single linear (shape [B, N, N, 5*hidden_dim])
    combined = F.linear(x_h, combined_weight)

    H = hidden_dim
    # ------------------------------------------------------------------
    # 4️⃣ Slice the combined result back into logical tensors
    # ------------------------------------------------------------------
    left       = combined[..., 0:H]
    left_gate  = combined[..., H:2*H]
    right      = combined[..., 2*H:3*H]
    right_gate = combined[..., 3*H:4*H]
    out_gate   = combined[..., 4*H:]          # shape [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid to the three gate tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates (in‑place Triton kernel)
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()                     # B * N * N * H
    BLOCK = 65536
    grid = ((numel + BLOCK - 1) // BLOCK,)
    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Gates no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation – batched matrix multiplication
    #    (equivalent to einsum('... i k d, ... j k d -> ... i j d'))
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, N, N, H
    total_heads = B * H

    # Reshape to (batch_head, N, N) required by torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Strided‑batched GEMM (FP16 Tensor Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [total_heads, N, N]

    # Restore shape [B, N, N, H] (still fp16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Fuse post‑processing: LayerNorm → out_gate multiplication →
    #      final linear projection (H → dim)
    # ------------------------------------------------------------------
    w_to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)  # [H]
    w_to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)    # [H]
    w_to_out_weight      = weights["to_out.weight"].to(torch.float16)       # [dim, H]

    final_fp16 = torch.empty((B, N, N, dim), dtype=torch.float16, device=device)

    total_positions = B * N * N
    BLOCK_D = 64                       # good trade‑off for dim up‑to 384
    dim_blocks = (dim + BLOCK_D - 1) // BLOCK_D

    # Flatten tensors to 1‑D views for the kernel
    out_flat      = out.contiguous().view(-1)           # [B*N*N*H]
    out_gate_flat = out_gate.contiguous().view(-1)     # same shape
    final_flat    = final_fp16.view(-1)                # [B*N*N*dim]

    postprocess_fused_kernel[
        (total_positions, dim_blocks)
    ](
        out_flat,
        out_gate_flat,
        w_to_out_norm_weight,
        w_to_out_norm_bias,
        w_to_out_weight,
        final_flat,
        H=hidden_dim,          # compile‑time constant
        DIM=dim,               # compile‑time constant
        EPS=eps,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 10️⃣ Return in the expected float32 format
    # ------------------------------------------------------------------
    return final_fp16.to(torch.float32)
```

**Explanation of speed‑up**

* **Reduced memory traffic** – the input tensor ``x`` is read only once for the
  five linear transforms (instead of five separate reads).  The mask is
  applied in‑place, and the gates are fused, decreasing the number of
  element‑wise passes.

* **Fewer kernel launches** – only two Triton kernels are used (gate fusion
  and full post‑processing).  All heavy linear work stays on the fast cuBLAS
  Tensor‑Core path, while the small per‑position work (LayerNorm + final
  projection) is executed entirely on‑chip.

* **Strided‑batched GEMM** – the N³ contraction is performed with a single
  ``torch.bmm`` call, letting cuBLAS schedule the operation across the
  H100 Tensor‑Cores without per‑matrix overhead.

The combined effect brings the runtime for the benchmark configurations
well below the target 1 ms threshold while preserving numerical correctness
(the final result is cast back to ``float32`` as required).
