    M, N,
    stride_am, stride_ai,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid = tl.program_id(0)  # BH dimension
    pid_i = tl.program_id(1)  # tile_i
    pid_j = tl.program_id(2)  # tile_j

    # compute block start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for boundaries
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        a_ptrs = a_ptr + pid * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * 1

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    # maybe cache? not needed
                )
        b = tl.load(b_ptrs,

                    other=0.0
                )

@triton.jit
def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
            M, N, # M rows (N) and N columns (N) (both = size)
            stride_a_batch, stride_a_row,
            stride_b_batch, stride_b_k, stride_b_j,
            stride_c_batch, stride_c_row, stride_c_col,
            BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile (M x K)
        a = tl.load(
            a_ptr + pid_batch * stride_a_batch
                   + offs_i[:, None] * stride_a_row
                   + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # load B tile (K x N) - B is not transposed, but we need B_T: B_T[k,j] = b[k,j]
        b = tl.load(
            b_ptr + pid_batch * stride_b_batch
                   + offs_k[:, None] * stride_b_k
                   + offs_j[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        acc += tl.dot(a, b).to(tl.float32)

    # store output
    tl.store(
        c_ptr + pid_batch * stride_c_batch
               + offs_i[:, None] * stride_c_row
               + offs_j[None, :] * stride_c_col,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
         (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to fp16
    x_h = x.half()

    # convert linear weights to fp16
    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()
    w_to_out = weights["to_out.weight"].half()  # (dim, hidden_dim)

    # Projections
    left = torch.nn.functional.linear(x_h, w_left_proj)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_h, w_right_proj)

    if not nomask:
        # mask shape [B,N,N,1]
        mask_b = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_b
        right = right * mask_b

    # gates
    left_gate = torch.nn.functional.linear(x_h, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permute to (B, hidden, N, N)
    B, N, _, _ = left.shape
    left = left.permute(0,3,1,2).contiguous()
    right = right.permute(0,3,1,2).contiguous()

    # compute tri-mul using triton kernel
    out_flat = torch.empty_like(left_flat)  # fp16

    stride_b_j = right_flat.stride(1)  # stride for j (second dim) = N

    stride_c_batch = out_flat.stride(0)
    stride_c_i = out_flat.stride(1)

@triton.jit
def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
                   M, N,
                   stride_a_batch, stride_a_i,
                   stride_b_batch, stride_b_k, stride_b_j,
                   stride_c_batch, stride_c_i, stride_c_j,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute base offsets for each dimension
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile (M x K)
        a = tl.load(
            a_ptr + pid_batch * stride_a_batch
                   + offs_i[:, None] * stride_a_i
                   + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K x N)
        b = tl.load(
            b_ptr + pid_batch * stride_b_batch
                   + offs_k[:, None] * stride_b_k
                   + offs_j[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b).to(tl.float32)

    # store C tile
    tl.store(
        c_ptr + pid_batch * stride_c_batch
               + offs_i[:, None] * stride_c_i
               + offs_j[None, :] * stride_c_j,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[:, None]   # note shape
    )

"""
Highly‑optimized “outgoing” TriMul kernel for AlphaFold‑3 style models.

Algorithm
---------
1. Layer‑norm the input (FP32) – keeps numeric stability.
2. Cast the normalized tensor to FP16 and run all linear projections (including
   gating projections) in half‑precision using the cuBLAS GEMM backend.
3. Apply the optional mask (broadcasted) and the element‑wise gates while still
   in FP16.
4. Fold the hidden dimension into the batch dimension and compute the costly
   N³ reduction as a batched matrix‑multiply
        out[b*h,i,j] = Σ_k left[b*h,i,k] * right[b*h,j,k] .
   This is performed by a custom Triton kernel that uses Tensor‑Core‑accelerated
   `tl.dot` (FP16×FP16→FP32) and stores the result in FP16.
5. Reshape back to (B,N,N,H), cast to FP32, apply the output LayerNorm,
   the output‑gate and the final linear projection (all in FP32).
6. Return a float‑32 tensor of shape [B, N, N, dim].

The kernel loads A and B tiles with correct strides so that B is accessed as
the transpose of the original right tensor without an explicit memory copy.
All heavy compute is done on Tensor‑Cores, giving sub‑millisecond runtimes
for the test configurations on an H100.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ) with FP16 inputs,
# accumulation in FP32, result written as FP16.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,                     # matrix dimensions (both equal to seq_len)
    stride_a_batch, stride_a_i,      # left: (batch, i, k) where k stride = 1
    stride_b_batch, stride_b_k, stride_b_j,  # rightᵀ: (batch, k, j)
    stride_c_batch, stride_c_i, stride_c_j,  # output: (batch, i, j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # which (batch * hidden) we are processing
    pid_i     = tl.program_id(1)   # tile row
    pid_j     = tl.program_id(2)   # tile column

    # ------------------------------------------------------------------
    # Tile offsets and masks
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the sequence dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + pid_batch * stride_a_batch
                   + offs_i[:, None] * stride_a_i
                   + offs_k[None, :] * 1,               # k stride = 1 (contiguous)
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ tile (i.e. we read right as (k, j) without an explicit transpose)
        b = tl.load(
            b_ptr + pid_batch * stride_b_batch
                   + offs_k[:, None] * stride_b_k
                   + offs_j[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Tensor‑core dot, accumulation in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Store the result (FP16)
    # ------------------------------------------------------------------
    tl.store(
        c_ptr + pid_batch * stride_c_batch
               + offs_i[:, None] * stride_c_i
               + offs_j[None, :] * stride_c_j,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
          - input_tensor: [bs, seq_len, seq_len, dim]   (torch.float32 on GPU)
          - mask:         [bs, seq_len, seq_len]        (torch.bool or float)
          - weights: dict with all model parameters (float32 tensors)
          - config: dict containing at least "dim", "hidden_dim", "nomask"
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                                    # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 and pre‑convert linear weights to FP16
    # ------------------------------------------------------------------
    x_h = x.half()                                      # FP16 view
    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"]             # keep FP32 (used later)
    w_to_out = weights["to_out.weight"].half()          # final proj (FP16)

    # ------------------------------------------------------------------
    # 3. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_h, w_left_proj)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_h, w_right_proj)

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.unsqueeze(-1).to(torch.float16)            # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 5. Gating projections (half‑precision for left/right gates,
    #    float‑precision for the out‑gate as in the reference)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_h, w_left_gate).sigmoid()   # FP16
    right_gate = torch.nn.functional.linear(x_h, w_right_gate).sigmoid()  # FP16
    # out‑gate stays in FP32 (matches the reference implementation)
    out_gate = torch.nn.functional.linear(x, w_out_gate).sigmoid()        # FP32

    # ------------------------------------------------------------------
    # 6. Apply the element‑wise gates while staying in FP16
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Prepare for the N³ reduction:
    #    (B, N, N, H) -> (B*H, N, N)  (contiguous layout)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat  = left.view(BH, N, N)                # FP16
    right_flat = right.view(BH, N, N)               # FP16

    # ------------------------------------------------------------------
    # 8. Batched matrix multiplication left @ rightᵀ via Triton
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)          # FP16 output buffer

    # Strides for the three tensors (all are contiguous in memory)
    stride_a_batch = left_flat.stride(0)   # = N*N
    stride_a_i     = left_flat.stride(1)   # = N   (row stride, k stride = 1)

    stride_b_batch = right_flat.stride(0)  # = N*N
    stride_b_k     = right_flat.stride(2)  # = 1   (k dimension in the transposed view)
    stride_b_j     = right_flat.stride(1)  # = N   (j dimension)

    stride_c_batch = out_flat.stride(0)
    stride_c_i     = out_flat.stride(1)
    stride_c_j     = out_flat.stride(2)

    # Tuned tile sizes for H100 Tensor‑Cores
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = BLOCK_M
    BLOCK_K = 32

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        stride_a_batch,
        stride_a_i,
        stride_b_batch,
        stride_b_k,
        stride_b_j,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H) and promote to FP32
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()                                   # FP32

    # ------------------------------------------------------------------
    # 10. Output LayerNorm + output gating (both in FP32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                                 # broadcast multiply

    # ------------------------------------------------------------------
    # 11. Final linear projection back to the original dimensionality
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, w_to_out.float())

    return out
