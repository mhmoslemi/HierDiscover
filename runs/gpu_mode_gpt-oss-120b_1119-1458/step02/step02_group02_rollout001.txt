def fused_masked_mul3_kernel(
    a_ptr, b_ptr, mask_ptr, out_ptr,
    total_elems: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK: tl.constexpr
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)  # [BLOCK] indices (flat index)
    mask = offs < total_elems

    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out = a * b * m

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B,N,_,dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # Cast to half
    x = input_tensor.to(torch.float16)

    # LayerNorm in half
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
        eps=eps,
    )
    # Concatenated weights
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)

    # Linear projection
    combined = torch.nn.functional.linear(x_norm, w_concat)  # shape (B,N,N,5*hidden)

    # Split
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(combined, hidden, dim=-1)

    # Gates
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()  # shape (B,N,N)
        left = fused_mul3_masked(left_proj_raw, left_gate, mask_fp16)
        right = fused_mul3_masked(right_proj_raw, right_gate, mask_fp16)

    # Compute batched GEMM
    # Bring hidden axis to batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous()  # B, hidden, N, N
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # B, hidden, N, N

    left_flat = left_mat.view(B * hidden, N, N)
    right_flat = right_mat.view(B * hidden, N, N)

    out_flat = torch.bmm(left_flat, right_flat)  # shape (B*hidden,N,N)

    # Reshape back
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # B,N,N,hidden

    # Output LN
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # Multiply by output gate
    out_norm = out_norm * out_gate

    # Final linear
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    return out_final.to(input_tensor.dtype)

"""
Trition‑accelerated “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1. **Half‑precision layer‑norm** on the last axis of the 4‑D input.
2. **Fused projection + gating** – concatenate the five linear weight
   matrices (left_proj, left_gate, right_proj, right_gate, out_gate) into
   one matrix of shape (5·hidden, dim) and perform a single GEMM
   (the heavy part of the operator) on the half‑precision tensor.
3. **Mask & gate fusion** – when a mask is present we avoid the huge
   (B·N·N·hidden) expansion (`mask[...,None]`).  A tiny Triton kernel
   computes the element‑wise product `a * b * mask` while broadcasting the
   mask implicitly (the mask index is `offset // hidden`).  This removes a
   ~128× memory copy and cuts the runtime dramatically.
4. **Core contraction** – the outgoing update
   `out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]`
   is a batched matrix multiply.  We reshape the tensors to
   (B·hidden, N, N) and call `torch.bmm`, which uses Tensor‑Core‑optimized
   cuBLAS.
5. **Output layer‑norm, output‑gate, final linear** – all remaining
   operations stay in fp16 and a final cast back to the input dtype is
   performed.

The implementation respects the required signature
`custom_kernel((input, mask, weights, config))` and works for all
provided test configurations on an H100 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: triple element‑wise product with broadcasted mask
#   out = a * b * mask      a,b : (B,N,N,hidden)   mask : (B,N,N)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr, b_ptr, mask_ptr, out_ptr,         # pointers
    total: tl.constexpr,                     # total number of elements in a/b (B·N·N·hidden)
    hidden: tl.constexpr,                    # hidden dimension (compile‑time)
    BLOCK_SIZE: tl.constexpr,                # block size (compile‑time)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    msk = offs < total

    a = tl.load(a_ptr + offs, mask=msk, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=msk, other=tl.float16(0.0))

    # Broadcast mask: one mask entry per hidden slice
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=msk, other=tl.float16(0.0))

    out = a * b * m
    tl.store(out_ptr + offs, out, mask=msk)


def fused_mul3_masked(a: torch.Tensor, b: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fused element‑wise multiplication `a * b * mask[...,None]` using Triton.
    All tensors must be contiguous.  `mask` has shape (B,N,N) and is broadcasted
    over the hidden dimension.
    """
    assert a.shape == b.shape
    B, N, _, hidden = a.shape
    # flatten tensors to 1‑D
    a_flat = a.contiguous()
    b_flat = b.contiguous()
    mask_flat = mask.contiguous()
    out = torch.empty_like(a_flat)

    total = a_flat.numel()
    BLOCK = 65536  # tuned for H100, multiple of 128 for good occupancy
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        a_flat,
        b_flat,
        mask_flat,
        out,
        total,
        hidden=hidden,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool/float, CUDA)
            - weights_dict:  mapping from layer names to weight tensors
            - config_dict:   must contain "dim", "hidden_dim";
                            optional "eps" (default 1e-5) and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep everything in fp16)
    # --------------------------------------------------------------
    x = inp.to(torch.float16)  # cast once
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2) Fused projection + all five gates (single GEMM)
    # --------------------------------------------------------------
    # Concatenate weight matrices: each has shape (hidden, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                     # (5*hidden, dim)
    ).to(torch.float16)

    # Linear projection without bias
    combined = F.linear(x_norm, w_concat)                     # (B,N,N,5*hidden)

    # Split back into the five parts
    split_tensors = torch.split(combined, hidden, dim=-1)
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = split_tensors

    # Gates (still fp16)
    left_gate  = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate   = torch.sigmoid(out_gate_logit)

    # --------------------------------------------------------------
    # 3) Apply mask (if any) together with gates – fused Triton kernel
    # --------------------------------------------------------------
    if nomask:
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # mask is [B,N,N] -> broadcast over hidden dimension inside the kernel
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = fused_mul3_masked(left_proj_raw, left_gate, mask_fp16)
        right = fused_mul3_masked(right_proj_raw, right_gate, mask_fp16)

    # --------------------------------------------------------------
    # 4) Core contraction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    Implemented as a batched GEMM over (B·hidden) matrices.
    # --------------------------------------------------------------
    # Bring hidden dimension to batch axis and transpose right for the dot.
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden, N, N]

    left_flat  = left_mat.view(B * hidden, N, N)        # (B*hidden, N, N)
    right_flat = right_mat.view(B * hidden, N, N)       # (B*hidden, N, N)

    out_flat = torch.bmm(left_flat, right_flat)         # (B*hidden, N, N)

    # Reshape back to the original layout
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 5) Output LayerNorm, gated multiplication, final projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                         # apply output gate

    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Cast back to the original dtype (float32) before returning
    return out_final.to(inp.dtype)
