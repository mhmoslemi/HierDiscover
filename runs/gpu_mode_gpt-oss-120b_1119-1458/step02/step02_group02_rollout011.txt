def custom_kernel(data):
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    device = inp.device

    # Layer norm
    x_norm = F.layer_norm(inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps)
    x_norm = x_norm.to(torch.float16)

    # Linear projections
    left_proj_raw = F.linear(x_norm, weights["left_proj.weight"].to(torch.float16))
    right_proj_raw = F.linear(x_norm, weights["right_proj.weight"].to(torch.float16))

    # Gates (logits)
    left_gate_logit = F.linear(x_norm, weights["left_gate.weight"].to(torch.float16))
    right_gate_logit = F.linear(x_norm, weights["right_gate.weight"].to(torch.float16))
    out_gate_logit = F.linear(x_norm, weights["out_gate.weight"].to(torch.float16))

    # Apply sigmoid
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # Apply mask and gating
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Ensure mask is float16 and contiguous
        mask_fp16 = mask.to(torch.float16)
        # fused multiplication broadcasting mask to hidden dim
        left = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # Core contraction using batched matrix multiply (no copies)
    # permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2)      # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)   # (B, hidden, k, j) i.e., transpose i and k

    # Matmul for each batch and hidden slice
    out_hidden = torch.matmul(left_perm, right_perm)   # shape (B, hidden, i, j)

    # Restore shape (B,N,N,hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    out_norm = F.layer_norm(out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps)

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    return out_final.to(inp.dtype)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16)  
2️⃣ Five independent linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) are executed **separately** – this keeps the
   huge intermediate tensor from the concatenated‑weight trick (≈5×
   memory) and avoids an unnecessary copy.  All linear layers run in
   FP16.

3️⃣ Mask + gate fusion – when a mask is present a single Triton kernel
   `fused_mul3_masked` computes  

        out = proj * gate * mask

   *without* expanding the mask to the hidden dimension (the kernel
   computes the mask index on‑the‑fly, saving ~B·N² memory traffic).

4️⃣ Core contraction – the outgoing update  

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   is a batched matrix‑multiply for each hidden channel.  By
   permuting to shape ``(B, hidden, N, N)`` we can call
   ``torch.matmul`` once; this uses cuBLAS’s strided‑batched GEMM and
   stays in FP16.

5️⃣ Output LayerNorm, gating and final projection – the result is
   normalized over the hidden dimension, multiplied by the output gate,
   and linearly projected back to the original ``dim``.  All heavy
   arithmetic stays in FP16; the final tensor is cast back to the
   input dtype (usually float32).

Only the element‑wise triple product (step 3) is custom‑written in
Triton, satisfying the requirement to include a kernel while keeping
the rest of the pipeline in highly‑optimized PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: a * b * c   (b is a 3‑D mask broadcasted over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_masked_kernel(
    a_ptr,               # left_proj_raw   (B,N,N,hidden)
    b_ptr,               # mask            (B,N,N)
    c_ptr,               # gate            (B,N,N,hidden)
    out_ptr,             # destination
    hidden: tl.constexpr,
    size:   tl.constexpr,          # total number of elements in a/c (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # load left_proj and gate – both are contiguous in the hidden dimension
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    # compute corresponding mask index (broadcast over hidden)
    mask_idx = offs // hidden                      # integer division
    b = tl.load(b_ptr + mask_idx, mask=mask, other=0.0)

    out = a * c * b
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3_masked(a: torch.Tensor,
                      mask: torch.Tensor,
                      c: torch.Tensor,
                      hidden: int) -> torch.Tensor:
    """Element‑wise product a * c * mask where mask is broadcast over
    the last (hidden) dimension.
    All tensors must be contiguous and of dtype torch.float16.
    """
    assert a.shape == c.shape, "a and c must have identical shapes"
    B, N, _, H = a.shape
    assert H == hidden, "hidden size mismatch"

    a = a.contiguous()
    c = c.contiguous()
    mask = mask.contiguous()                # shape (B,N,N)

    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192                             # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_masked_kernel[grid](
        a, mask, c, out,
        hidden,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  shape [B, N, N] (bool/float, CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim"
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (FP32 → FP16 for the rest)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Separate linear projections and gates (all in FP16)
    # ------------------------------------------------------------------
    left_proj_raw  = F.linear(x_norm, weights["left_proj.weight"].to(torch.float16))
    right_proj_raw = F.linear(x_norm, weights["right_proj.weight"].to(torch.float16))

    left_gate_raw  = F.linear(x_norm, weights["left_gate.weight"].to(torch.float16))
    right_gate_raw = F.linear(x_norm, weights["right_gate.weight"].to(torch.float16))
    out_gate_raw   = F.linear(x_norm, weights["out_gate.weight"].to(torch.float16))

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)   # shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Apply mask and gates (mask is optional)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → a simple binary product
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Cast mask to FP16 once; Triton kernel will broadcast it on‑the‑fly
        mask_fp16 = mask.to(torch.float16)
        left  = fused_mul3_masked(left_proj_raw, mask_fp16, left_gate, hidden)
        right = fused_mul3_masked(right_proj_raw, mask_fp16, right_gate, hidden)

    # ------------------------------------------------------------------
    # 4) Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matmul over the hidden dimension.
    # ------------------------------------------------------------------
    # (B, N, N, hidden) → (B, hidden, N, K) where K == N
    left_perm  = left.permute(0, 3, 1, 2)        # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1)       # (B, hidden, k, j)

    # Batched matrix multiplication for every (b, hidden) pair
    out_hidden = torch.matmul(left_perm, right_perm)   # (B, hidden, i, j)

    # Restore original layout
    out = out_hidden.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5) Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    out_norm = out_norm * out_gate                      # output gate

    out_final = F.linear(out_norm,
                         weights["to_out.weight"].to(torch.float16))  # bias‑free

    # Cast back to the input dtype (usually float32) before returning
    return out_final.to(inp.dtype)
