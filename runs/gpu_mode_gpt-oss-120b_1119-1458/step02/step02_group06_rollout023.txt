def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm (float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # 2. Cast to bfloat16 for compute
    x_bf16 = x.bfloat16()
    B, N, _, _ = x_bf16.shape
    M = B * N * N  # number of rows for proj

    # 3. Prepare transposed weights (dim, hidden_dim) in bf16
    w_left_proj = weights["left_proj.weight"].to(torch.bfloat16).t().contiguous()
    w_right_proj = weights["right_proj.weight"].to(torch.bfloat16).t().contiguous()
    w_left_gate = weights["left_gate.weight"].to(torch.bfloat16).t().contiguous()
    w_right_gate = weights["right_gate.weight"].to(torch.bfloat16).t().contiguous()
    w_out_gate = weights["out_gate.weight"].to(torch.bfloat16).t().contiguous()

    # 4. Output tensors for fused projection
    left_out = torch.empty((M, hidden_dim), dtype=torch.bfloat16, device=x.device)
    right_out = torch.empty((M, hidden_dim), dtype=torch.bfloat16, device=x.device)
    out_gate_out = torch.empty((M, hidden_dim), dtype=torch.bfloat16, device=x.device)

    # Flatten input to (M, dim)
    x_flat = x_bf16.view(M, dim)

    # Compute strides
    stride_x_m, stride_x_k = x_flat.stride()
    stride_w_row, stride_w_col = w_left_proj.stride()
    stride_out_m, stride_out_d = left_out.stride()

    # Launch fused projection kernel
    BLOCK_M = 32
    BLOCK_K = 32
    BLOCK_D = 64   # tile size for hidden dimension

    grid = ( (M + BLOCK_M - 1) // BLOCK_M, )
    fused_proj_gate_kernel[grid](
        x_flat,
        w_left_proj,
        w_left_gate,
        w_right_proj,
        w_right_gate,
        w_out_gate,
        left_out,
        right_out,
        out_gate_out,
        M, dim, hidden_dim,
        stride_x_m, stride_x_k,
        stride_w_row, stride_w_col,
        stride_out_m, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=3,
    )
    # Reshape outputs back to [B,N,N,hidden_dim]
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)
    out_gate = out_gate_out.view(B, N, N, hidden_dim)

    # 5. Apply optional mask (broadcasted)
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.bfloat16)  # [B,N,N,1]
        left = left * m
        right = right * m

    # 6. N^3 reduction: batch matmul left @ right^T per hidden dim
    # Permute to (B, hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # Use torch.bmm which maps to cuBLAS batched GEMM (Tensor Cores)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B,N,N,hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()   # FP32 for final layernorm

    # 7. Output layernorm + out_gate gating
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()

    # 8. Final linear projection to original dim (float32)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def fused_proj_gate_kernel(
    x_ptr,          # [M, K] input
    w_left_proj_ptr,
    w_left_gate_ptr,
    w_right_proj_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    left_out_ptr,   # [M, H]
    right_out_ptr,  # [M, H]
    out_gate_out_ptr,
    M, K, H,
    stride_x_m, stride_x_k,
    stride_w_row, stride_w_col,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)  # each program handles BLOCK_M rows
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    num_d_tiles = tl.cdiv(H, BLOCK_D)

    # loop over hidden dimension tiles
    for d_tile in range(0, num_d_tiles):
        d_start = d_tile * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # accumulators in fp32
        acc_left = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
        acc_right = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
        acc_right_gate = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
        acc_out_gate = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

        # reduction over K (input dim)
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            # Load X block
            x = tl.load(
                x_ptr + offs_m[:, None] * stride_x_m + offs_k[None, :] * stride_x_k,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            )
            # Load weight blocks (K, D_tile)
            w_l = tl.load(
                w_left_proj_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )
            w_lg = tl.load(
                w_left_gate_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )
            w_r = tl.load(
                w_right_proj_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )
            w_rg = tl.load(
                w_right_gate_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )
            w_og = tl.load(
                w_out_gate_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )

            # Accumulate
            acc_left += tl.dot(x, w_l).to(tl.float32)
            acc_left_gate += tl.dot(x, w_lg).to(tl.float32)
            acc_right += tl.dot(x, w_r).to(tl.float32)
            acc_right_gate += tl.dot(x, w_rg).to(tl.float32)
            acc_out_gate += tl.dot(x, w_og).to(tl.float32)

        right = (acc_right * right_gate).to(tl.float16)

"""
Highly‑optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1.  Layer‑norm the input in FP32.
2.  Cast the normalized tensor to bfloat16 (Tensor‑Core friendly).
3.  Fuse the five linear projections (left‑proj, left‑gate,
    right‑proj, right‑gate, out‑gate) into a single Triton kernel.
    The kernel:
      * loads a tile of the input once,
      * loads the corresponding slices of all five weight matrices,
      * accumulates the five matrix‑products in FP32,
      * applies the sigmoid to the gate accumulators,
      * multiplies the projections with their gates,
      * writes the three results (left, right, out_gate) in bfloat16.
    This reduces the memory traffic of the projection stage by a factor‑5.
4.  Optionally multiply left/right with the binary mask (broadcasted).
5.  Perform the cubic reduction
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    as a batched GEMM (left @ rightᵀ) using torch.bmm – this maps to
    the cuBLAS Tensor‑Core kernel and is optimal for the large batch*
    hidden (BH) sizes encountered in the tests.  The data layout is
    (B, hidden_dim, N, N) → (BH, N, N) to expose the batch dimension.
6.  Convert the result back to FP32, apply LayerNorm on the hidden
    dimension, multiply with the already‑sigmoid‑ed out_gate, and apply
    the final linear projection back to the model dimension.
7.  Return the FP32 tensor of shape [B, N, N, dim].

The implementation uses bfloat16 for all heavy linear work,
FP32 accumulation for numerical stability, and a custom Triton kernel
to fuse the projection‑gate stages, achieving a large speed‑up over a
naïve PyTorch implementation while keeping the result bit‑identical
(up to FP16/BF16 round‑off) to the reference module.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def fused_proj_gate_kernel(
    # pointers
    x_ptr,                     # [M, K]   (flattened normalized input)
    w_left_proj_ptr,           # [K, H]   (left projection weight)
    w_left_gate_ptr,           # [K, H]   (left gate weight)
    w_right_proj_ptr,          # [K, H]   (right projection weight)
    w_right_gate_ptr,          # [K, H]   (right gate weight)
    w_out_gate_ptr,            # [K, H]   (output gate weight)
    left_out_ptr,              # [M, H]   (gated left)
    right_out_ptr,             # [M, H]   (gated right)
    out_gate_out_ptr,          # [M, H]   (output‑gate, sigmoided)
    # sizes
    M, K, H,                   # M = B*N*N, K = dim, H = hidden_dim
    # strides
    stride_x_m, stride_x_k,    # input strides
    stride_w_row, stride_w_col,# weight strides (row = K, col = H)
    stride_out_m, stride_out_d,# output strides
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Fused projection + sigmoidal gating.
    Produces:
        left  = (x @ W_left)   * sigmoid(x @ W_left_gate)
        right = (x @ W_right)  * sigmoid(x @ W_right_gate)
        out_gate = sigmoid(x @ W_out_gate)
    All tensors are in BF16; accumulations are performed in FP32.
    """
    pid = tl.program_id(0)                     # each program works on BLOCK_M rows
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # loop over hidden‑dimension tiles
    num_d_tiles = tl.cdiv(H, BLOCK_D)
    for d_tile in range(0, num_d_tiles):
        d_start = d_tile * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # FP32 accumulators for this (M‑tile, D‑tile)
        acc_left      = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
        acc_right     = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
        acc_right_gate= tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)
        acc_out_gate  = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

        # reduction over the input feature dimension K
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            # ---- load a tile of the input -------------------------------------------------
            x = tl.load(
                x_ptr + offs_m[:, None] * stride_x_m + offs_k[None, :] * stride_x_k,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            )   # shape (BLOCK_M, BLOCK_K)

            # ---- load the corresponding weight tiles --------------------------------------
            w_l  = tl.load(
                w_left_proj_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )
            w_lg = tl.load(
                w_left_gate_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )
            w_r  = tl.load(
                w_right_proj_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )
            w_rg = tl.load(
                w_right_gate_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )
            w_og = tl.load(
                w_out_gate_ptr + offs_k[:, None] * stride_w_row + offs_d[None, :] * stride_w_col,
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            )

            # ---- accumulate the five GEMMs ------------------------------------------------
            acc_left       += tl.dot(x, w_l ).to(tl.float32)
            acc_left_gate  += tl.dot(x, w_lg).to(tl.float32)
            acc_right      += tl.dot(x, w_r ).to(tl.float32)
            acc_right_gate += tl.dot(x, w_rg).to(tl.float32)
            acc_out_gate   += tl.dot(x, w_og).to(tl.float32)

        # ---- post‑reduction: sigmoid and gating -----------------------------------------

        right_gate_sig = 1.0 / (1.0 + tl.exp(-acc_right_gate))
        out_gate_sig   = 1.0 / (1.0 + tl.exp(-acc_out_gate))

        right = (acc_right * right_gate_sig).to(tl.float16)
        out_g = out_gate_sig.to(tl.float16)

        # ---- store the tiles ------------------------------------------------------------

            - input_tensor : [B, N, N, dim]  (float32)
            - mask         : [B, N, N]        (bool/float, optional)
            - weights      : dict of torch.Tensor (model parameters)
            - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------------
    # unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  float32

    # -------------------------------------------------------------
    # 2. Cast to BF16 for the heavy work
    # -------------------------------------------------------------
    x_bf16 = x.bfloat16()
    B, N, _, _ = x_bf16.shape
    M = B * N * N                     # total number of (b,i,k) rows

    # -------------------------------------------------------------
    # 3. Prepare transposed weight matrices (K, H) in BF16
    # -------------------------------------------------------------
    # Original weight shape = (out_features, in_features) = (hidden_dim, dim)
    # We need (dim, hidden_dim) for the GEMM in the kernel.
    w_left_proj = weights["left_proj.weight"].t().contiguous().bfloat16()
    w_right_proj = weights["right_proj.weight"].t().contiguous().bfloat16()
    w_left_gate = weights["left_gate.weight"].t().contiguous().bfloat16()
    w_right_gate = weights["right_gate.weight"].t().contiguous().bfloat16()
    w_out_gate = weights["out_gate.weight"].t().contiguous().bfloat16()

    # -------------------------------------------------------------
    # 4. Output buffers for the fused projection+gate kernel
    # -------------------------------------------------------------
    left_out   = torch.empty((M, hidden_dim), dtype=torch.bfloat16, device=x.device)
    right_out  = torch.empty((M, hidden_dim), dtype=torch.bfloat16, device=x.device)
    out_gate_out = torch.empty((M, hidden_dim), dtype=torch.bfloat16, device=x.device)

    # Flatten input to (M, K) where K = dim
    x_flat = x_bf16.view(M, dim)

    # -------------------------------------------------------------
    # 5. Launch the fused projection/gate Triton kernel
    # -------------------------------------------------------------
    # Strides for the flattened tensors
    stride_x_m, stride_x_k = x_flat.stride()
    stride_w_row, stride_w_col = w_left_proj.stride()   # same for all weight tensors
    stride_out_m, stride_out_d = left_out.stride()      # identical for the three outputs

    # Tile sizes – chosen to respect H100 register budget while giving good occupancy
    BLOCK_M = 32   # rows per program
    BLOCK_K = 32   # reduction over input dimension (dim)
    BLOCK_D = 64   # columns (hidden‑dim) per program tile

    grid = ((M + BLOCK_M - 1) // BLOCK_M, )
    fused_proj_gate_kernel[grid](
        x_flat,
        w_left_proj,
        w_left_gate,
        w_right_proj,
        w_right_gate,
        w_out_gate,
        left_out,
        right_out,
        out_gate_out,
        M, dim, hidden_dim,
        stride_x_m, stride_x_k,
        stride_w_row, stride_w_col,
        stride_out_m, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=3,
    )

    # -------------------------------------------------------------
    # 6. Reshape back to [B, N, N, hidden_dim]
    # -------------------------------------------------------------
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)
    out_gate = out_gate_out.view(B, N, N, hidden_dim)   # already sigmoided

    # -------------------------------------------------------------
    # 7. Optional mask (broadcasted over the hidden dimension)
    # -------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.bfloat16)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # -------------------------------------------------------------
    # 8. Cubic reduction: batched GEMM left @ rightᵀ for every hidden channel
    #    (B*hidden_dim) independent matrix multiplies of shape (N,N)×(N,N).
    # -------------------------------------------------------------
    # Move hidden_dim to the batch axis to expose a large batched GEMM.
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)                # (BH, N, N)  bf16
    right_flat = right.view(BH, N, N)              # (BH, N, N)  bf16

    # torch.bmm maps to cuBLAS batched GEMM (Tensor‑Cores) – excellent performance.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # -------------------------------------------------------------
    # 9. Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # -------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()   # [B, N, N, hidden_dim]  FP32

    # -------------------------------------------------------------
    # 10. Output LayerNorm + gating with out_gate
    # -------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()   # broadcasted element‑wise

    # -------------------------------------------------------------
    # 11. Final linear projection back to the model dimension (float32)
    # -------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
