"""
TriMul (outgoing) – Triton‑accelerated forward with a custom batched GEMM.
==========================================================================

Algorithm
---------
1️⃣ Input Layer‑Norm (FP32 → FP16)  
2️⃣ **Fused linear** – concatenate the five weight matrices (proj + gates) and
   execute a single `F.linear`.  This yields the raw projections and the
   gate‑logits for *left*, *right* and *out* in one pass.
3️⃣ Apply element‑wise sigmoid to the gate logits.  
4️⃣ Apply the (optional) pair‑wise mask and fuse the mask with the projection
   using a tiny Triton kernel (`fused_mul3`).  This saves two memory passes.
5️⃣ **Core contraction** – compute  
   `out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]`  
   with a custom Triton GEMM that works on the
   `(batch*hidden) × N × N` tensors directly, avoiding the extra transposes
   and the overhead of `torch.bmm`.  The kernel uses FP16 loads, FP32
   accumulation and Tensor‑core friendly block sizes.
6️⃣ Output Layer‑Norm + out‑gate + final linear projection back to `dim`.  
   All heavy arithmetic stays in FP16; the result is cast back to the original
   dtype.

The implementation keeps the heavy linear algebra in highly‑optimized
Tensor‑core kernels (standard torch for the large GEMMs, Triton for the tiny
element‑wise fusions, and a custom Triton GEMM for the TriMul contraction).
It fulfills the requirement of using at least one Triton kernel while staying
fully differentiable‑free (forward‑only).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel 1: triple element‑wise product (mask * proj * gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    size: tl.constexpr,            # total number of elements
    BLOCK_SIZE: tl.constexpr,      # compile‑time constant
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fuse mask·proj·gate using Triton (all tensors 1‑D contiguous)."""
    assert a.shape == b.shape == c.shape, "Shape mismatch"
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 8192                     # tuned for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel 2: Batched GEMM for the outgoing TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                # FP16, shape (M, N, N)
    right_ptr,               # FP16, shape (M, N, N)   (right already transposed)
    out_ptr,                 # FP16, shape (M, N, N)
    M,                       # total number of matrices (= batch * hidden)
    N,                       # sequence length
    BLOCK_M: tl.constexpr,   # tile rows   (i‑dimension)
    BLOCK_N: tl.constexpr,   # tile cols   (j‑dimension)
    BLOCK_K: tl.constexpr,   # tile reduction (k‑dimension)
):
    """
    Compute for each m ∈ [0, M):
        out[m, i, j] = Σ_k left[m, i, k] * right[m, k, j]
    The right tensor is already the transposed variant, i.e. its memory layout
    stores (k, j) in row‑major order.
    """
    pid_x = tl.program_id(0)      # tile index in the (i,j) plane
    pid_m = tl.program_id(1)      # index of the matrix (batch*hidden)

    # number of tiles along each output dimension
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # tile coordinates
    ti = pid_x // num_j           # block row   (i)
    tj = pid_x % num_j            # block col   (j)

    # offsets for the output tile
    offs_i = ti * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tj * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offset for the current matrix (contiguous layout)
    # Each matrix occupies N*N elements
    base = pid_m * N * N

    # accumulator in FP32 (to avoid overflow)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left tile   shape (BLOCK_M, BLOCK_K)  ->  left[i,k]
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + base \
                    + offs_i[:, None] * N \
                    + offs_k[None, :] * 1
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Load right tile   shape (BLOCK_K, BLOCK_N)  ->  right[k,j]
        # (right already stores (k, j) in row‑major order)
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + base \
                     + offs_k[:, None] * N \
                     + offs_j[None, :] * 1
        b = tl.load(right_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Rank‑K update
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ----------------------------------------------------------------------
    # Store the result (cast back to FP16)
    # ----------------------------------------------------------------------
    out_ptrs = out_ptr + base \
               + offs_i[:, None] * N \
               + offs_j[None, :] * 1
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# TriMul entry point (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor : shape [B, N, N] (bool or float, on CUDA)
            - weights_dict : model weights (see skeleton)
            - config_dict  : {"dim":int, "hidden_dim":int, "eps":float, "nomask":bool}
    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm   (keep result in FP16 for the rest)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates (single linear)
    # ------------------------------------------------------------------
    # Concatenate all five weight matrices (shape (5*hidden, dim))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # (5*hidden, dim)

    # Single linear gives all five tensors: [B, N, N, 5*hidden]
    combined = F.linear(x_norm, w_concat)    # no bias

    # Split back
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(
        combined, hidden, dim=-1
    )
    # Sigmoid gates (stay in FP16)
    left_gate  = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate   = torch.sigmoid(out_gate_logit)   # shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Mask & gate fusion
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple two‑operand multiplication
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Broadcast mask to the hidden dimension
        mask_exp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden).contiguous()
        left = fused_mul3(left_proj_raw, mask_exp, left_gate)
        right = fused_mul3(right_proj_raw, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – custom batched GEMM (outgoing TriMul)
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) and then collapse B*hidden → M
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transposed k ↔ j → [B, hidden, N, N]

    M = B * hidden
    left_flat = left_perm.view(M, N, N)    # (M, N, N)  FP16
    right_flat = right_perm.view(M, N, N)  # (M, N, N)  FP16 (already k‑j swapped)

    out_flat = torch.empty_like(left_flat)  # will hold the result in FP16

    # Kernel block sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid: (tiles over i·j, matrices)
    num_i_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_j_tiles = (N + BLOCK_N - 1) // BLOCK_N
    grid_x = num_i_tiles * num_j_tiles
    grid = (grid_x, M)

    trimul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm + out‑gate + final linear
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                     # gating (FP16)

    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),  # (dim, hidden)
    )   # result in FP16

    # Cast back to the original dtype of `inp`
    return out_final.to(inp.dtype)
