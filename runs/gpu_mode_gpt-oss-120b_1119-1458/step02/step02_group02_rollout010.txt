@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, M_stride_a, K_stride_a,
    batch_stride_b, N_stride_b, K_stride_b,
    batch_stride_c, M_stride_c, N_stride_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for block start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K loop start
    # init accumulator with zeros
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load A block [BLOCK_M, BLOCK_K]
        k_range = k_offset + tl.arange(0, BLOCK_K)
        # compute mask for out-of-bounds
        a_mask = (offs_m < M) & (k_range < K)
        b_mask = (offs_n < N) & (k_range < K)
        # pointers
        a_ptr = A_ptr + batch_id * batch_stride_a + offs_m[:, None] * M_stride_a + k_range[None, :] * K_stride_a
        b_ptr = B_ptr + batch_id * batch_stride_b + offs_n[:, None] * N_stride_b + k_range[None, :] * K_stride_b
        # load
        a = tl.load(a_ptr, mask=a_mask, other=0.0)
        b = tl.load(b_ptr, mask=b_mask, other=0.0)
        # convert to fp32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        # compute dot
        acc += tl.dot(a, b)  # (M,K)*(N,K) -> (M,N)
    # store result
    c_ptr = C_ptr + batch_id * batch_stride_c + offs_m[:, None] * M_stride_c + offs_n[None, :] * N_stride_c
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptr, acc.to(tl.float16), mask=c_mask)

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    M, N, K,
    HIDDEN: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute b,d
    b = pid_batch // HIDDEN
    d = pid_batch - b * HIDDEN

    # Offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # base pointer address for this batch and hidden
    a_base = A_ptr + b * stride_a_batch + d * stride_a_h
    b_base = B_ptr + b * stride_b_batch + d * stride_b_h
    c_base = C_ptr + b * stride_c_batch + d * stride_c_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_start * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # masks for out-of-bounds
        mask_a = (offs_i[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_j[:, None] < N) & (offs_k[None, :] < K)

        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_j[:, None] * stride_b_j + offs_k[None, :] * stride_b_k

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # cast to fp32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) dot (BLOCK_N, BLOCK_K) -> (BLOCK_M, BLOCK_N)
    # mask for output

    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

This implementation performs:
1. Input layer‑norm.
2. A fused linear that simultaneously computes the two projections and the three
   gates (left_proj, right_proj, left_gate, right_gate, out_gate).  The result is in
   fp16 to keep the data on Tensor‑cores.
3. Optional mask multiplication fused via a tiny Triton kernel.
4. Core “outgoing” update 
      out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   implemented as a custom bat‑ched GEMM kernel that directly works on the
   original [B,N,N,hidden] layout, thus avoiding extra transposes or copies.
5. Output layer‑norm, output‑gate multiplication and the final linear projection
   back to the original channel dimension.

All heavy matrix‑multiplications remain in fp16 with fp32 accumulation; the final
output is cast back to the input dtype (float32)."""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise product of three tensors (used for mask)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,
    size: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)
    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
    fused_mul3_kernel[grid](
        a, b, c, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out

    pair of tensors. The input tensors have shape (B, N, N, H) and are interpreted as
    left[i,k] and right[j,k] for each hidden channel.
    """
    pid_batch = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)              # tile over rows (i)
    pid_n = tl.program_id(2)              # tile over cols (j)

    # Decompose combined index into batch b and hidden index d
    b = pid_batch // HIDDEN
    d = pid_batch - b * HIDDEN

    # Base pointers for this (b,d) pair
    a_base = A_ptr + b * stride_a_batch + d * stride_a_h
    b_base = B_ptr + b * stride_b_batch + d * stride_b_h
    c_base = C_ptr + b * stride_c_batch + d * stride_c_h

    # Offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load tiles of A (left) and B (right) – both are (tile_M, tile_K) / (tile_N, tile_K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_j[:, None] * stride_b_j + offs_k[None, :] * stride_b_k

        mask_a = (offs_i[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_j[:, None] < N) & (offs_k[None, :] < K)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # tl.dot performs Σ_k a[i,k] * b[j,k]  → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    mask_c = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing update using the Triton batched GEMM kernel.
    left / right : shape [B, N, N, hidden] (fp16)
    Returns: tensor of the same shape (fp16)
    """
    B, N, _, hidden = left.shape
    # Output tensor
    out = torch.empty_like(left)

    # Strides (in elements)
    s_a_batch, s_a_i, s_a_k, s_a_h = left.stride()
    s_b_batch, s_b_j, s_b_k, s_b_h = right.stride()
    s_c_batch, s_c_i, s_c_j, s_c_h = out.stride()

    # Tunable block sizes – chosen to fit in shared memory and give good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left,
        right,
        out,
        s_a_batch,
        s_a_i,
        s_a_k,
        s_a_h,
        s_b_batch,
        s_b_j,
        s_b_k,
        s_b_h,
        s_c_batch,
        s_c_i,
        s_c_j,
        s_c_h,
        N,  # M
        N,  # N
        N,  # K
        HIDDEN=hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):

    Forward pass of the “outgoing” TriMul operator, accelerated with Triton.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              input_tensor : torch.Tensor [B, S, S, C] (float32)
              mask_tensor  : torch.Tensor [B, S, S] (bool/float) – may be None
              weights_dict : model weights (see skeleton)
              config_dict  : contains keys "dim", "hidden_dim", optional "eps",
                             "nomask" (bool)

    Returns:
        Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, S, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep FP32 for stats, then down‑cast)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)  # fp16 for the rest

    # ------------------------------------------------------------------
    # 2) Fused projection + gate logits (single matmul)
    # ------------------------------------------------------------------
    # Concatenate weight matrices: shape (5*hidden, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Combined linear (no bias)
    combined = F.linear(x, w_concat)  # [B, S, S, 5*hidden]

    # Split back into the five parts
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(
        combined, hidden, dim=-1
    )
    # Gates – stay in fp16
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # ------------------------------------------------------------------
    # 3) Apply optional mask and element‑wise gating
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj_raw * left_gate      # [B, S, S, hidden]
        right = right_proj_raw * right_gate
    else:
        # mask is [B, S, S] – expand to hidden dimension
        mask_exp = mask.unsqueeze(-1).to(torch.float16)
        left = fused_mul3(left_proj_raw, mask_exp, left_gate)
        right = fused_mul3(right_proj_raw, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 4) Core contraction: outgoing update via custom batched GEMM
    # ------------------------------------------------------------------
    # left/right are already in the layout expected by batched_gemm:
    #   (batch, seq, seq, hidden) where the reduction is over the middle seq.
    out = batched_gemm(left, right)   # shape [B, S, S, hidden] fp16

    # ------------------------------------------------------------------
    # 5) Output layer‑norm, output gate, and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                       # gating
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),
    )   # back to dim

    # Cast back to the original dtype (usually float32)
    return out_final.to(inp.dtype)

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16).  
2️⃣ Fuse the five linear layers (left_proj, right_proj, three gates) into a single
   matmul by concatenating their weight matrices; the result stays in fp16.  
3️⃣ Optional mask‑gate fusion via a tiny Triton kernel that computes a*b*c in one
   pass.  
4️⃣ Core “outgoing” update  
       out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]  
   is performed by a custom batched‑GEMM Triton kernel that works directly on the
   original `[B, N, N, hidden]` layout – no transpose, no extra copies.  
5️⃣ Output LayerNorm, output‑gate multiplication and the final linear projection
   back to `dim`.  The final tensor is cast back to the input dtype (float32).
All heavy matrix‑multiplications stay in fp16 with fp32 accumulation,
leveraging H100 Tensor‑cores.

# Kernel 1: fused element‑wise product of three tensors (mask * proj * gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,
    size: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b * c, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Element‑wise a*b*c using Triton (inputs must be contiguous)."""
    a, b, c = a.contiguous(), b.contiguous(), c.contiguous()

    BLOCK = 8192

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out
