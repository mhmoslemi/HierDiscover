@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_am, stride_ak,  # a: (M, N, K)  – we interpret K=N
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # batch*hidden dimension
    pid_i = tl.program_id(1)   # tile row
    pid_j = tl.program_id(2)   # tile col

    # ------------------------------------------------------------------
    #  Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Reduction over the K dimension (the sequence dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a block of A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + pid_m * stride_am
                 + offs_i[:, None] * stride_ak
                 + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load a block of Bᵀ: shape (BLOCK_K, BLOCK_N)
        # B is stored as (M, N, K) = (batch*H, N, N)
        b = tl.load(
            b_ptr + pid_m * stride_bm
                 + offs_k[:, None] * stride_bk
                 + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # FP16 dot, accumulate as FP32
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    #  Store the result (FP16)
    # ------------------------------------------------------------------
    tl.store(
        c_ptr + pid_m * stride_cm
              + offs_i[:, None] * stride_cn
              + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )

                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

"""
custom_kernel
==============

Forward implementation of the **outgoing** TriMul operator (AlphaFold3 style).
The heavy N³ reduction `out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]`
is performed by a small, fused Triton GEMM kernel (FP16×FP16→FP32).
All linear projections and gating are done in half‑precision to exploit
Tensor‑Cores; the surrounding LayerNorms and the final projection run in
full precision, matching the reference PyTorch implementation.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _triton_gram_kernel(
    a_ptr, b_ptr, c_ptr,
    N,                     # sequence length (rows = cols = N)
    stride_am, stride_ak, # a: (batch, N, N)  → batch‑stride, row‑stride
    stride_bm, stride_bn, stride_bk,   # b: (batch, N, N) (row‑major after transpose)
    stride_cm, stride_cn, stride_ck,   # c: (batch, N, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch ``b``:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]

    A, B, C are half‑precision tensors; accumulation is performed in FP32.
    """
    pid_b = tl.program_id(0)   # batch * hidden_dim index
    pid_i = tl.program_id(1)   # tile row
    pid_j = tl.program_id(2)   # tile col

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k dimension (the "sequence" axis)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i, k] block
        a = tl.load(
            a_ptr
            + pid_b * stride_am
            + offs_i[:, None] * stride_ak
            + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B[k, j] block (B is already transposed, i.e. row‑major)
        b = tl.load(
            b_ptr
            + pid_b * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Store the result (FP16)
    # ------------------------------------------------------------------
    tl.store(
        c_ptr
        + pid_b * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision and prepare half‑precision weights
    # ------------------------------------------------------------------
    x_h = x.half()                                        # [B, N, N, dim]  fp16
    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()
    w_to_out = weights["to_out.weight"].half()             # final linear (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (half‑precision, bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x_h, w_left_proj)   # [B, N, N, H]  fp16
    right = F.linear(x_h, w_right_proj) # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted over the hidden channel)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        m = mask.unsqueeze(-1).to(torch.float16)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 5️⃣ Gating networks (half‑precision, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates (still fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Rearrange tensors for the batched Gram product
    #     [B, N, N, H] → [B, H, N, N] → contiguous → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left.view(B * hidden_dim, N, N)    # (BH, N, N)   fp16
    right_flat = right.view(B * hidden_dim, N, N)  # (BH, N, N)   fp16

    # The Gram product needs right transposed on the last two axes.
    # We create a *contiguous* row‑major version of the transposed matrix.
    right_T = right_flat.permute(0, 2, 1).contiguous()  # (BH, N, N) row‑major

    # ------------------------------------------------------------------
    # 8️⃣ Batched matrix multiply (N³ reduction) via Triton
    # ------------------------------------------------------------------
    BH = B * hidden_dim
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    out_flat = torch.empty_like(left_flat)   # fp16 output buffer

    # strides for the three tensors (batch‑major layout)
    stride_am = left_flat.stride(0)   # batch stride for A
    stride_ak = left_flat.stride(1)   # row stride for A (i)
    stride_bm = right_T.stride(0)     # batch stride for B
    stride_bn = right_T.stride(2)     # column stride for B (unused in kernel)
    stride_bk = right_T.stride(1)     # row stride for B (k)
    stride_cm = out_flat.stride(0)    # batch stride for C
    stride_cn = out_flat.stride(1)    # row stride for C (i)
    stride_ck = out_flat.stride(2)    # column stride for C (j)

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _triton_gram_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N, N,
        stride_am,
        stride_ak,
        stride_bm,
        stride_bn,
        stride_bk,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to (B, N, N, H) and convert to float32 for the rest
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()   # [B, N, N, H]  fp32

    # ------------------------------------------------------------------
    # 🔟 Output layer‑norm + output gate (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()   # broadcast over hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final projection back to original dimension (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out.float())   # bias is None in the original model

    return out
