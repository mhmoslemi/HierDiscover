def gated_mask_fusion_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_proj_ptr, out_gate_ptr,
    hidden_dim, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offsets = start + tl.arange(0, BLOCK)
    mask = offsets < n_elements

    tl.store(out_proj_ptr + offsets, out.to(tl.float16), mask=mask)

def fused_gate_mask_kernel(
    proj_left_ptr, gate_left_ptr, proj_right_ptr, gate_right_ptr,
    mask_ptr, out_left_ptr, out_right_ptr,
    hidden_dim, n_elements, BLOCK: tl.constexpr):

    proj_l = tl.load(proj_left_ptr + offs, mask=mask).to(tl.float32)
    gate_l = tl.load(gate_left_ptr + offs, mask=mask).to(tl.float32)
    proj_r = tl.load(proj_right_ptr + offs, mask=mask).to(tl.float32)

    tl.store(out_right_ptr + offs, out_r.to(tl.float16), mask=mask)

@triton.jit
def _gate_mask_fused_kernel(
    proj_left_ptr,
    gate_left_ptr,
    proj_right_ptr,
    gate_right_ptr,
    mask_ptr,
    out_left_ptr,
    out_right_ptr,
    hidden_dim: tl.constexpr,
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load
    proj_l = tl.load(proj_left_ptr + offs, mask=mask).to(tl.float32)
    gate_l = tl.load(gate_left_ptr + offs, mask=mask).to(tl.float32)
    proj_r = tl.load(proj_right_ptr + offs, mask=mask).to(tl.float32)
    gate_r = tl.load(gate_right_ptr + offs, mask=mask).to(tl.float32)

    # Sigmoid
    gate_l_sig = 1.0 / (1.0 + tl.exp(-gate_l))
    gate_r_sig = 1.0 / (1.0 + tl.exp(-gate_r))

    # mask index
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

@triton.jit
def _proj_gate_mask_fused_kernel(
    combined_ptr,   # fp16 pointer to combined tensor (B,N,N,4*hidden)
    mask_ptr,       # fp16 pointer to mask (B*N*N)
    out_left_ptr,   # fp16 pointer left output (B,N,N,hidden)
    out_right_ptr,  # fp16 pointer right output (B,N,N,hidden)
    hidden_dim: tl.constexpr,
    n_elements: tl.constexpr, # number of elements per output (B*N*N*hidden)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Compute indices
    L = offs // hidden_dim
    d = offs - L * hidden_dim    # remainder

    # Base offset in combined tensor
    base = L * (hidden_dim * 4)

    # Offsets for each component
    proj_l_off = base + d
    gate_l_off = base + hidden_dim + d
    proj_r_off = base + 2 * hidden_dim + d
    gate_r_off = base + 3 * hidden_dim + d

    proj_l = tl.load(combined_ptr + proj_l_off, mask=mask).to(tl.float32)
    gate_l = tl.load(combined_ptr + gate_l_off, mask=mask).to(tl.float32)
    proj_r = tl.load(combined_ptr + proj_r_off, mask=mask).to(tl.float32)
    gate_r = tl.load(combined_ptr + gate_r_off, mask=mask).to(tl.float32)

    gate_l_sig = 1.0 / (1.0 + tl.exp(-gate_l))
    gate_r_sig = 1.0 / (1.0 + tl.exp(-gate_r))

    mask_val = tl.load(mask_ptr + L, mask=mask).to(tl.float32)

    out_l = proj_l * gate_l_sig * mask_val
    out_r = proj_r * gate_r_sig * mask_val

    tl.store(out_left_ptr + offs, out_l.to(tl.float16), mask=mask)
    tl.store(out_right_ptr + offs, out_r.to(tl.float16), mask=mask)
```

* Keep the heavy O(N³) contraction as a batched GEMM (torch.bmm) which
  already uses Tensor‑Cores (FP16 × FP16 → FP32).
* Perform the remaining LayerNorm, out‑gate scaling and final projection in
  PyTorch (the work is tiny compared to the GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, sigmoid‑gate and mask for left & right
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_fused_kernel(
    combined_ptr,            # fp16  pointer to combined tensor (B,N,N,4*H)
    mask_ptr,                # fp16  pointer to flattened mask (B*N*N)
    out_left_ptr,            # fp16  destination for left  tensor (B,N,N,H)
    out_right_ptr,           # fp16  destination for right tensor (B,N,N,H)
    hidden_dim: tl.constexpr,   # int   hidden dimension H
    n_elements: tl.constexpr,   # int   total elements per side (=B*N*N*H)
    BLOCK: tl.constexpr,        # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # ------------------------------------------------------------------
    # Compute indices inside the packed (B,N,N,4*H) tensor
    #   L = floor(offs / H)         – index of the (b,i,k) triple
    #   d = offs % H                – offset inside a hidden block
    # ------------------------------------------------------------------
    L = offs // hidden_dim                 # shape (BLOCK,)
    d = offs - L * hidden_dim              # remainder, also shape (BLOCK,)

    # base offset for the current (b,i,k) entry in the packed tensor
    base = L * (hidden_dim * 4)            # each entry stores 4*H values

    # offsets for the four sub‑tensors inside the packed layout
    proj_l_off = base + d
    gate_l_off = base + hidden_dim + d
    proj_r_off = base + 2 * hidden_dim + d
    gate_r_off = base + 3 * hidden_dim + d

    # ------------------------------------------------------------------
    # Load values (fp16 → fp32 for higher‑precision arithmetic)
    # ------------------------------------------------------------------
    proj_l = tl.load(combined_ptr + proj_l_off, mask=mask).to(tl.float32)
    gate_l = tl.load(combined_ptr + gate_l_off, mask=mask).to(tl.float32)
    proj_r = tl.load(combined_ptr + proj_r_off, mask=mask).to(tl.float32)
    gate_r = tl.load(combined_ptr + gate_r_off, mask=mask).to(tl.float32)

    # sigmoid in fp32
    gate_l_sig = 1.0 / (1.0 + tl.exp(-gate_l))
    gate_r_sig = 1.0 / (1.0 + tl.exp(-gate_r))

    # mask (broadcasted over hidden dim)
    mask_val = tl.load(mask_ptr + L, mask=mask).to(tl.float32)

    # ------------------------------------------------------------------
    # Apply gate, mask and write results back (fp16)
    # ------------------------------------------------------------------
    out_l = proj_l * gate_l_sig * mask_val
    out_r = proj_r * gate_r_sig * mask_val

    tl.store(out_left_ptr + offs, out_l.to(tl.float16), mask=mask)
    tl.store(out_right_ptr + offs, out_r.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Input:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]      (float32)
            mask         : [B, N, N]           (bool/float)
            weights      : dict of model parameters
            config       : dict containing “dim” and “hidden_dim”
    Output:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32, stable)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                 # [B, N, N, dim]  float32
    x_h = x.to(torch.float16)        # FP16 for the heavy linear work

    # --------------------------------------------------------------
    # 2️⃣ Fuse four linear layers (proj + gate for left/right)
    # --------------------------------------------------------------
    # Stack weights: [left_proj, left_gate, right_proj, right_gate]  → (4*H, dim)
    stacked_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)              # FP16

    # Single fused matrix‑multiply
    combined = F.linear(x_h, stacked_weight)          # [B, N, N, 4*H] FP16

    # --------------------------------------------------------------
    # 3️⃣ Apply mask + sigmoid‑gate (fused Triton kernel when mask present)
    # --------------------------------------------------------------
    B, N, _, _ = combined.shape
    if not nomask:                                     # mask is used
        # flatten mask to shape (B*N*N,) and cast to FP16
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                    # [B*N*N]

        # allocate output tensors
        left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
        right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

        # launch Triton kernel (one launch for both sides)
        BLOCK = 65536
        n_elem = B * N * N * hidden_dim               # elements per side
        grid = (triton.cdiv(n_elem, BLOCK),)

        _proj_gate_mask_fused_kernel[grid](
            combined.view(-1),                     # packed pointer
            mask_flat,
            left.view(-1),
            right.view(-1),
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask → simple element‑wise ops (still FP16)
        # reshape combined to (B,N,N,4,hidden) for easy slicing
        combined_5d = combined.view(B, N, N, 4, hidden_dim)
        left_proj = combined_5d[..., 0, :]          # [B,N,N,H]
        left_gate = combined_5d[..., 1, :]
        right_proj = combined_5d[..., 2, :]
        right_gate = combined_5d[..., 3, :]

        left = left_proj * torch.sigmoid(left_gate)   # FP16
        right = right_proj * torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 4️⃣ Core O(N³) contraction  C = left @ rightᵀ   (batched GEMM)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) without materialising a copy
    left_batched = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # FP16
    right_batched = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # FP16

    # Batched matrix multiply – FP16×FP16 → FP32 accumulation
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))    # [B*H,N,N] FP16

    # Reshape back to (B,N,N,H) and promote to FP32 for the following layers
    out = (
        out_batched
        .reshape(B, hidden_dim, N, N)      # (B,H,N,N)
        .permute(0, 2, 3, 1)               # (B,N,N,H)
        .to(torch.float32)                 # FP32
    )                                      # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                      # [B,N,N,H] FP32

    # out‑gate (single linear, sigmoid)
    out_gate_fp16 = F.linear(x_h, weights["out_gate.weight"].to(torch.float16))   # [B,N,N,H] FP16
    out_gate = torch.sigmoid(out_gate_fp16).to(torch.float32)                    # FP32

    out = out * out_gate                       # broadcast over hidden dim

    # final projection back to original feature dim
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim] FP32

    return out
```
