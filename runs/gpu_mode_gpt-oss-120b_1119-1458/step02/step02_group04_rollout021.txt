```
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,                     # [M, K] input flattened
    left_proj_w_ptr,           # [K, H] left proj weight
    left_gate_w_ptr,           # [K, H]
    right_proj_w_ptr,
    right_gate_w_ptr,
    out_gate_w_ptr,
    mask_ptr,                  # [M] optional mask (float16)
    out_left_ptr,              # [M, H]
    out_right_ptr,             # [M, H]
    out_gate_ptr,              # [M, H]
    M, K, H,
    stride_xm, stride_xk,
    stride_lp_k, stride_lp_n,   # for each weight matrix, stride across K and hidden dim N.
    ...,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # loops over hidden dimension tile
    for h_off in range(0, H, BLOCK_N):
        # allocate accumulators for each output set
        acc_left = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_right = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_right_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_out_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            # load A block
            a = tl.load(x_ptr + (offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk), mask=mask_m[:, None] & (offs_k < K), other=0.0)

            # load weight blocks
            w_left = tl.load(left_proj_w_ptr + (offs_k[:, None] * stride_lp_k + (h_off + tl.arange(0, BLOCK_N))[None, :] * stride_lp_n), mask=(offs_k < K) & ((h_off + tl.arange(0, BLOCK_N)) < H), other=0.0)
            w_left_gate = tl.load(left_gate_w_ptr + (offs_k[:, None] * stride_lg_k + (h_off + tl.arange(0, BLOCK_N))[None, :] * stride_lg_n), mask=..., other=0.0)
            w_right = tl.load(right_proj_w_ptr + (offs_k[:, None] * stride_rp_k + (h_off + tl.arange(0, BLOCK_N))[None, :] * stride_rp_n), mask=..., other=0.0)
            w_right_gate = tl.load(right_gate_w_ptr + (offs_k[:, None] * stride_rg_k + (h_off + tl.arange(0, BLOCK_N))[None, :] * stride_rg_n), mask=..., other=0.0)
            w_out_gate = tl.load(out_gate_w_ptr + (offs_k[:, None] * stride_og_k + (h_off + tl.arange(0, BLOCK_N))[None, :] * stride_og_n), mask=..., other=0.0)

            # dot
            acc_left += tl.dot(a, w_left)
            acc_left_gate += tl.dot(a, w_left_gate)
            acc_right += tl.dot(a, w_right)
            acc_right_gate += tl.dot(a, w_right_gate)
            acc_out_gate += tl.dot(a, w_out_gate)

        # after K loop: apply activation/signature
        left_proj = acc_left.to(tl.float16)
        left_gate = 1.0 / (1.0 + tl.exp(-acc_left_gate.to(tl.float16)))
        right_proj = acc_right.to(tl.float16)
        right_gate = 1.0 / (1.0 + tl.exp(-acc_right_gate.to(tl.float16)))
        out_gate = 1.0 / (1.0 + tl.exp(-acc_out_gate.to(tl.float16)))

        # apply mask if present
        if mask_ptr != 0:  # cannot use if mask is None; we will pass a dummy pointer.
            mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)[:, None]   # shape (BLOCK_M, 1)
            left_proj = left_proj * mask_val
            right_proj = right_proj * mask_val

        left_out = left_proj * left_gate
        right_out = right_proj * right_gate

        # store outputs
        off_h = h_off
        # compute offsets for out pointers - each out shape (M, H)
        out_left_off = out_left_ptr + (offs_m[:, None] * stride_ol_m + (off_h + tl.arange(0, BLOCK_N))[None, :] * stride_ol_h)
        tl.store(out_left_off, left_out, mask=mask_m[:, None] & ((off_h + tl.arange(0, BLOCK_N)) < H))
        out_right_off = out_right_ptr + (offs_m[:, None] * stride_or_m + (off_h + tl.arange(0, BLOCK_N))[None, :] * stride_or_h)
        tl.store(out_right_off, right_out, mask=mask_m[:, None] & ((off_h + tl.arange(0, BLOCK_N)) < H))
        out_gate_off = out_gate_ptr + (offs_m[:, None] * stride_og_m + (off_h + tl.arange(0, BLOCK_N))[None, :] * stride_og_h)
        tl.store(out_gate_off, out_gate, mask=mask_m[:, None] & ((off_h + tl.arange(0, BLOCK_N)) < H))
```

Pseudo for kernel:

```
@triton.jit
def fused_proj_gate_kernel(
    x_ptr, mask_ptr,
    lp_ptr, lg_ptr, rp_ptr, rg_ptr,
    out_left_ptr, out_right_ptr,
    M, K, H,
    stride_xm, stride_xk,
    stride_mask_m,
    stride_lp_k, stride_lp_h,
    stride_lg_k, stride_lg_h,
    stride_rp_k, stride_rp_h,
    stride_rg_k, stride_rg_h,
    stride_out_m, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute offsets for rows
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_valid = offs_m < M

    # load mask values for this block
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_valid, other=1.0)[:, None]  # broadcast across hidden dim later
    
    for h_off in range(0, H, BLOCK_N):
        # Compute output column offsets
        offs_n = h_off + tl.arange(0, BLOCK_N)
        # compute column masks
        col_valid = offs_n < H

        # Initialize accumulators
        acc_lp = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_rp = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
            k_valid = k_offset < K

            # load A block (x)
            a = tl.load(x_ptr + offs_m[:, None] * stride_xm + k_offset[None, :] * stride_xk,
                         mask=mask_valid[:, None] & k_valid[None, :], other=0.0)
            # load weight blocks
            w_lp = tl.load(lp_ptr + k_offset[:, None] * stride_lp_k + offs_n[None, :] * stride_lp_h,
                         mask=k_valid[:, None] & col_valid[None, :], other=0.0)
            w_lg = tl.load(lg_ptr + k_offset[:, None] * stride_lg_k + offs_n[None, :] * stride_lg_h,
                         mask=k_valid[:, None] & col_valid[None, :], other=0.0)
            w_rp = tl.load(rp_ptr + k_offset[:, None] * stride_rp_k + offs_n[None, :] * stride_rp_h,
                         mask=k_valid[:, None] & col_valid[None, :], other=0.0)
            w_rg = tl.load(rg_ptr + k_offset[:, None] * stride_rg_k + offs_n[None, :] * stride_rg_h,
                         mask=k_valid[:, None] & col_valid[None, :], other=0.0)

            acc_lp += tl.dot(a, w_lp)
            acc_lg += tl.dot(a, w_lg)
            acc_rp += tl.dot(a, w_rp)
            acc_rg += tl.dot(a, w_rg)

        # Apply activation and mask
        left_proj = acc_lp.to(tl.float16)
        left_gate = 1.0 / (1.0 + tl.exp(-acc_lg.to(tl.float16)))
        right_proj = acc_rp.to(tl.float16)
        right_gate = 1.0 / (1.0 + tl.exp(-acc_rg.to(tl.float16)))

        # Apply mask to left/right proj
        left = left_proj * mask_val * left_gate
        right = right_proj * mask_val * right_gate

    out_ptr,   # [B*H, N, N] (fp16)
    N,               # dimension

    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # right load (batch, k, j) because we want right[pos, j, k], originally right[batch, j, k]

                     mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store

"""
Optimized outgoing TriMul (AlphaFold‑3 style) forward pass.

Algorithm
---------
1. Layer‑norm the input (FP32 for numerical stability).
2. Cast to FP16 and compute **both** left/right linear projections *and*
   their gates in a **single fused Triton kernel**.  The kernel:
   * reads each position x[b,i,k,:] once,
   * multiplies it with four weight matrices (left‑proj, left‑gate,
     right‑proj, right‑gate),
   * applies the optional pair‑wise mask,
   * applies sigmoid to the two gate tensors,
   * writes the gated tensors `left` and `right` (shape [B,N,N,hidden_dim]).
   This reduces memory traffic from five separate matmuls to a single pass.
3. Reshape ``left`` and ``right`` to (B·hidden_dim, N, N) and perform a
   **batched GEMM** (left @ rightᵀ) with a small Triton kernel that
   follows the classic tiled‑matmul pattern.  The contraction implements
   ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]``.
4. Convert the result to FP32 and apply a second Layer‑Norm over the
   hidden dimension.
5. Compute the output‑gate (σ(out_gate(x))) with a regular PyTorch linear
   layer (the cost is negligible compared to the N³ contraction).
6. Apply the output‑gate and a final linear projection back to the original
   channel dimension.
7. Return the result in FP32.

Only the heavy data‑movement parts (projection+gate and the N³ contraction)
are written in Triton, keeping the rest of the pipeline in PyTorch for
readability and to reuse highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# 1️⃣  Fused projection + gating kernel (produces left/right tensors)
# -------------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    # pointers
    x_ptr,              # fp16   (M, K)  input rows
    mask_ptr,           # fp16   (M,)   broadcast mask (1.0 = keep)
    left_proj_ptr,      # fp16   (K, H) left projection weight   (K=dim, H=hidden_dim)
    left_gate_ptr,      # fp16   (K, H) left gate weight
    right_proj_ptr,     # fp16   (K, H) right projection weight
    right_gate_ptr,     # fp16   (K, H) right gate weight
    left_out_ptr,       # fp16   (M, H) gated left output
    right_out_ptr,      # fp16   (M, H) gated right output
    # sizes
    M, K, H,
    # strides (in elements, not bytes)
    stride_xm, stride_xk,
    stride_mask_m,
    stride_lp_k, stride_lp_h,
    stride_lg_k, stride_lg_h,
    stride_rp_k, stride_rp_h,
    stride_rg_k, stride_rg_h,
    stride_out_m, stride_out_h,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute left/right projections + sigmoid gates + optional mask."""
    pid = tl.program_id(0)

    # -----------------------------------------------------------------
    # Row (position) block
    # -----------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    mask_m = offs_m < M
    # Load mask once per row block (scalar per row, broadcast later)
    m_val = tl.load(mask_ptr + offs_m * stride_mask_m,
                    mask=mask_m, other=1.0)[:, None]   # shape (BLOCK_M,1)

    # -----------------------------------------------------------------
    # Loop over hidden dimension in tiles of BLOCK_N
    # -----------------------------------------------------------------
    for h_start in range(0, H, BLOCK_N):
        offs_h = h_start + tl.arange(0, BLOCK_N)        # [BLOCK_N]
        mask_h = offs_h < H

        # accumulators (FP32 for accuracy)
        acc_lp = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_rp = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # -----------------------------------------------------------------
        # K‑dimension loop (dim)
        # -----------------------------------------------------------------
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)        # [BLOCK_K]
            mask_k = offs_k < K

            # Load a tile of the input vector (M×K)
            a = tl.load(
                x_ptr + offs_m[:, None] * stride_xm
                       + offs_k[None, :] * stride_xk,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            )   # (BLOCK_M, BLOCK_K)

            # Load corresponding weight tiles (K×H) for the four linear maps
            w_lp = tl.load(
                left_proj_ptr + offs_k[:, None] * stride_lp_k
                              + offs_h[None, :] * stride_lp_h,
                mask=mask_k[:, None] & mask_h[None, :],
                other=0.0,
            )
            w_lg = tl.load(
                left_gate_ptr + offs_k[:, None] * stride_lg_k
                             + offs_h[None, :] * stride_lg_h,
                mask=mask_k[:, None] & mask_h[None, :],
                other=0.0,
            )
            w_rp = tl.load(
                right_proj_ptr + offs_k[:, None] * stride_rp_k
                               + offs_h[None, :] * stride_rp_h,
                mask=mask_k[:, None] & mask_h[None, :],
                other=0.0,
            )
            w_rg = tl.load(
                right_gate_ptr + offs_k[:, None] * stride_rg_k
                                + offs_h[None, :] * stride_rg_h,
                mask=mask_k[:, None] & mask_h[None, :],
                other=0.0,
            )

            # Accumulate
            acc_lp += tl.dot(a, w_lp)   # (BLOCK_M, BLOCK_N)
            acc_lg += tl.dot(a, w_lg)
            acc_rp += tl.dot(a, w_rp)
            acc_rg += tl.dot(a, w_rg)

        # -------------------------------------------------------------
        # Apply sigmoid to gates (still FP32, then cast back to FP16)
        # -------------------------------------------------------------
        left_gate = (1.0 / (1.0 + tl.exp(-acc_lg))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_rg))).to(tl.float16)

        # Apply mask * gate to the projected tensors
        left_proj = acc_lp.to(tl.float16) * m_val * left_gate
        right_proj = acc_rp.to(tl.float16) * m_val * right_gate

        # -----------------------------------------------------------------
        # Store results (M×H) – same strides for left and right outputs
        # -----------------------------------------------------------------
        out_l_ptr = left_out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h
        out_r_ptr = right_out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h

        tl.store(out_l_ptr,
                 left_proj,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_r_ptr,
                 right_proj,
                 mask=mask_m[:, None] & mask_h[None, :])


# -------------------------------------------------------------------------
# 2️⃣  Batched GEMM kernel for the N³ contraction
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_trimul_kernel(
    left_ptr,   # fp16 (B*H, N, N)    left matrix  (i,k)
    right_ptr,  # fp16 (B*H, N, N)    right matrix (j,k) – read transposed
    out_ptr,    # fp16 (B*H, N, N)    result
    N,                          # sequence length
    # strides for left (batch, i, k)
    stride_l_b, stride_l_i, stride_l_k,
    # strides for right (batch, k, j) – we read it as (k,j)
    stride_r_b, stride_r_j, stride_r_k,
    # strides for output (batch, i, j)
    stride_o_b, stride_o_i, stride_o_j,
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each head (batched GEMM)."""
    batch_id = tl.program_id(0)   # which (batch * hidden_dim) pair
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block col

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: (i,k)
        a = tl.load(
            left_ptr
            + batch_id * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right transposed: we need element right[j,k] → load as (k,j)
        b = tl.load(
            right_ptr
            + batch_id * stride_r_b
            + offs_k[:, None] * stride_r_k      # k as the first axis
            + offs_j[None, :] * stride_r_j,     # j as the second axis
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # store fp16 result
    out = acc.to(tl.float16)
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block, fully compatible with the
    reference PyTorch implementation.  At least one computation (the
    projection+gate) is performed in Triton to minimise memory traffic.
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------
    # Helper – move tensors to FP16 on the target device
    # -------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # Load model weights (transposed to (in_features, out_features))
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    left_proj_w   = to_fp16(weights["left_proj.weight"]).t().contiguous()
    right_proj_w  = to_fp16(weights["right_proj.weight"]).t().contiguous()
    left_gate_w   = to_fp16(weights["left_gate.weight"]).t().contiguous()
    right_gate_w  = to_fp16(weights["right_gate.weight"]).t().contiguous()
    out_gate_w    = to_fp16(weights["out_gate.weight"]).t().contiguous()

    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # -------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for precision)
    # -------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # -------------------------------------------------
    # 2️⃣ Cast to FP16 and flatten to (M, K) where M = B·N·N, K = dim
    # -------------------------------------------------
    x_half = x_norm.to(torch.float16)
    M = B * N * N
    K = dim
    x_flat = x_half.view(M, K).contiguous()          # (M, K)

    # -------------------------------------------------
    # 3️⃣ Prepare mask (flattened).  If no mask, use ones.
    # -------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones(M, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype=torch.float16, device=device).view(M).contiguous()

    # -------------------------------------------------
    # 4️⃣ Allocate output buffers for left/right (FP16)
    # -------------------------------------------------
    left_out = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right_out = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)

    # -------------------------------------------------
    # 5️⃣ Launch fused projection + gating kernel
    # -------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(M, BLOCK_M),)

    fused_proj_gate_kernel[grid](
        # pointers
        x_flat,
        mask_fp16,
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        left_out,
        right_out,
        # sizes
        M, K, hidden_dim,
        # strides (elements, not bytes)
        x_flat.stride(0), x_flat.stride(1),
        mask_fp16.stride(0),
        left_proj_w.stride(0), left_proj_w.stride(1),
        left_gate_w.stride(0), left_gate_w.stride(1),
        right_proj_w.stride(0), right_proj_w.stride(1),
        right_gate_w.stride(0), right_gate_w.stride(1),
        left_out.stride(0), left_out.stride(1),
        # block sizes
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 6️⃣ Reshape left/right back to [B, N, N, H] and permute to
    #    [B·H, N, N] for the batched GEMM.
    # -------------------------------------------------
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)

    # (B, H, N, N) layout expected by the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N).contiguous()
    right_flat = right_perm.view(B * hidden_dim, N, N).contiguous()

    # -------------------------------------------------
    # 7️⃣ Batched GEMM (N³ contraction) via Triton
    # -------------------------------------------------
    out_flat = torch.empty_like(left_flat)  # fp16 tensor (B*H, N, N)

    BLOCK_M_GEMM = 128
    BLOCK_N_GEMM = 128
    BLOCK_K_GEMM = 32
    grid_gemm = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M_GEMM),
        triton.cdiv(N, BLOCK_N_GEMM),
    )
    batched_gemm_trimul_kernel[grid_gemm](
        left_flat,
        right_flat,
        out_flat,
        N,
        # strides for left (batch, i, k)
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for right (batch, k, j) – note the swapped axes
        right_flat.stride(0),
        right_flat.stride(2),      # stride for j (fastest dimension after transpose)
        right_flat.stride(1),      # stride for k
        # output strides (batch, i, j)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        # tile sizes
        BLOCK_M=BLOCK_M_GEMM,
        BLOCK_N=BLOCK_N_GEMM,
        BLOCK_K=BLOCK_K_GEMM,
    )

    # -------------------------------------------------
    # 8️⃣ Bring contraction result back to original layout
    # -------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # -------------------------------------------------
    # 9️⃣ Output LayerNorm over hidden_dim (float32 for stability)
    # -------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # -------------------------------------------------
    # 🔟 Compute the output‑gate (sigmoid of a linear projection)
    # -------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))  # [B,N,N,H] fp16

    # -------------------------------------------------
    # 1️⃣1️⃣ Apply gate and final projection back to dim
    # -------------------------------------------------
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)          # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 12️⃣ Return FP32 tensor as required by the spec
    # -------------------------------------------------
    return final.to(torch.float32)
