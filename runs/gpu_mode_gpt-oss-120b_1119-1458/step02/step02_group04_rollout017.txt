@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # decode indices
    pid_head = tl.program_id(0)  # index for (b, h)
    pid_i = tl.program_id(1)  # tile row
    pid_j = tl.program_id(2)  # tile col

    # compute batch and head
    b = pid_head // H
    h = pid_head % H

    # Compute offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # create ranges
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    # Masks for out-of-bound tail
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + \
            offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block (transposed): shape (BLOCK_K, BLOCK_N)
        # right is stored as (b, i, j, h). To get R_T for multiplication we need right[j,k], i.e., we read right at (b, k, j, h).
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        # This yields shape (BLOCK_K, BLOCK_N)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None,:], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(left, right)   # left: (M K), right: (K N) -> (M N)

    # Store result into out
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + \
        offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None,:])

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_hw = tl.program_id(0)   # batch*head
    pid_i = tl.program_id(1)    # tile row (i)
    pid_j = tl.program_id(2)    # tile col (j)

    # Compute batch and head indices
    b = pid_hw // H
    h = pid_hw % H

    # Compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Range within block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M x K)
        a_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + \
                 offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed (K x N)
        b_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + \
                 offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Store result in half
    c_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + \
             offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3 style) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 for stability).
2. Cast to fp16, apply the four linear projections (left/right and their gates)
   and an optional pair‑wise mask.  All linear layers are executed with
   torch.nn.functional.linear (cuBLAS Tensor‑Core kernels) and the results
   stay in FP16.
3. Fuse the left/right projections with their sigmoid gates (element‑wise)
   – the tensors are now ready for the O(N³) contraction.
4. Perform the contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   with a custom Triton kernel.  The kernel treats each (batch, head) pair
   as an independent GEMM and computes the product with FP16 inputs and FP32
   accumulation (Tensor‑Core friendly).  No extra permutes or copies are
   required.
5. Layer‑norm over the hidden dimension, apply the output gate and a final
   linear projection back to the original channel dimension.
6. Return the result in float32.

The heavy N³ contraction is the only part kept in PyTorch; everything else
remains in FP16 to maximise Tensor‑Core throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel for the batched N³ contraction (left @ rightᵀ per head)
# -------------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr,          # const half *
    right_ptr,         # const half *
    out_ptr,           # half *
    B, N, H,           # python scalars
    # strides for the left tensor (B,N,N,H) ─ contiguous layout
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for the right tensor (B,N,N,H) – same ordering
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for the output tensor (B,N,N,H)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each (batch, head) pair."""
    pid_head = tl.program_id(0)      # batch * head index
    pid_i    = tl.program_id(1)      # tile row index
    pid_j    = tl.program_id(2)      # tile col index

    # -----------------------------------------------------------------
    # Decode batch and head indices
    # -----------------------------------------------------------------
    b = pid_head // H                # batch index
    h = pid_head % H                 # head (hidden) index

    # -----------------------------------------------------------------
    # Compute the start offsets of the tile
    # -----------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension K
    # -----------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (M, K)  (i dimension, k dimension)
        a_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + \
                 offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load right block transposed: we need rightᵀ ⇒ shape (K, N) where
        # rows correspond to k and columns to j.
        b_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + \
                 offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate (Tensor‑Core friendly)
        acc += tl.dot(a, b)          # (M,K)·(K,N) → (M,N)

    # -----------------------------------------------------------------
    # Write the tile back to global memory (cast back to fp16)
    # -----------------------------------------------------------------
    c_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + \
             offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Entry‑point: the custom kernel expected by the benchmark harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dictionary with the model parameters (as torch.Tensor)
        - config_dict  : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 1️⃣ Unpack everything
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]      # H in the paper

    # -----------------------------------------------------------------
    # Helper: move weight to FP16 on the target device
    # -----------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Load model parameters (all are bias‑free Linear + LayerNorm)
    # -----------------------------------------------------------------
    norm_w            = weights["norm.weight"]
    norm_b            = weights["norm.bias"]
    left_proj_w       = to_half(weights["left_proj.weight"])
    right_proj_w      = to_half(weights["right_proj.weight"])
    left_gate_w       = to_half(weights["left_gate.weight"])
    right_gate_w      = to_half(weights["right_gate.weight"])
    out_gate_w        = to_half(weights["out_gate.weight"])
    to_out_norm_w     = to_half(weights["to_out_norm.weight"])
    to_out_norm_b     = to_half(weights["to_out_norm.bias"])
    to_out_w          = to_half(weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)          # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 4️⃣ Cast to FP16 – the heavy lifting lives in half precision
    # -----------------------------------------------------------------
    x_half = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 5️⃣ Linear projections (left/right)
    # -----------------------------------------------------------------
    left  = F.linear(x_half, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_half, right_proj_w)  # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 6️⃣ Optional pair‑wise mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 7️⃣ Gating: sigmoid(linear(x)) → element‑wise scale
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_half, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))  # [B,N,N,H] fp16
    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 8️⃣ N³ contraction – custom Triton kernel (batched GEMM)
    # -----------------------------------------------------------------
    # Allocate output buffer (fp16) that will hold [B, N, N, H]
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Grid dimensions: (batch*head, tile_rows, tile_cols)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides in **elements** (not bytes) – PyTorch already gives us that.
    stride_l = left.stride()   # (B,N,N,H) → (sB, sI, sK, sH)
    stride_r = right.stride()
    stride_o = out.stride()

    trmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # -----------------------------------------------------------------
    # 9️⃣ Output LayerNorm over the hidden dimension (float32 internally)
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w.float(),
                                 bias=to_out_norm_b.float(),
                                 eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)

    # -----------------------------------------------------------------
    # 🔟 Output gate + final projection back to original dim
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))   # [B,N,N,H] fp16
    gated = out_norm * out_gate                               # element‑wise, fp16
    final = F.linear(gated, to_out_w)                        # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 1️⃣1️⃣ Return result in float32 as required by the API
    # -----------------------------------------------------------------
    return final.to(torch.float32)
