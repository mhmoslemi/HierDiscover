"""
Optimized outgoing Triangle‑Multiplicative Update (TriMul)

Algorithm (FP16‑heavy, float32 output)
--------------------------------------------------------------
1️⃣ LayerNorm on the last dimension (float32 – stable).
2️⃣ Fuse all five linear projections (left/right project‑gate & out‑gate)
   into a **single** GEMM:  (B·N·N) × dim  × (5·H)  →  B×N×N×5H.
   This reduces the number of passes over the input tensor from 5→1.
3️⃣ Split the fused tensor into the 5 components.
   * The two gate tensors are turned into sigmoids on‑the‑fly.
4️⃣ When a mask is present we fuse the element‑wise
   `proj * sigmoid(gate) * mask` for the left/right streams
   in a tiny Triton kernel (no extra temporaries).
   If there is no mask a simple point‑wise product suffices.
5️⃣ Core O(N³) operation – batched matrix multiplication:
   for every hidden dimension we compute
   `C = A @ Bᵀ`   (A,B ∈ ℝ^{N×N}) using Torch’s highly‑optimised `torch.bmm`.
6️⃣ LayerNorm on the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original feature dimension.
All heavy arithmetic is performed in FP16 (Tensor‑Core GEMM),
accumulation is in FP32, and the final result is returned as FP32.

A Triton kernel is used for step 4 (gate‑mask fusion) to satisfy the
requirement of “use Triton”. The rest of the work is done with
torch’s fast fused kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: proj * sigmoid(gate) * mask   (mask is optional)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,          # fp16*  : projection (left or right)
    gate_ptr,          # fp16*  : pre‑sigmoid gate values
    mask_ptr,          # fp16*  : mask flattened, shape = B·N·N
    out_ptr,           # fp16*  : destination (left or right)
    hidden_dim,        # i64    : hidden dimension (used for mask broadcast)
    n_elements,        # i64    : total number of elements (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)          # element offsets
    valid = offs < n_elements

    # Load projection and gate (FP16 → FP32 for higher‑precision mul)
    proj = tl.load(proj_ptr + offs, mask=valid).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=valid).to(tl.float32)

    # sigmoid(gate) = 1/(1+exp(-x))
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # Mask broadcasting: each mask entry repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid).to(tl.float32)

    out = proj * gate_sig * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=valid)


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask         : torch.Tensor [B, N, N] (bool/float, ignored if config['nomask'])
            weights      : dict of model parameters (float32)
            config       : dict with keys:
                - "dim"         : input feature dimension
                - "hidden_dim"  : hidden dimension (H)
                - "nomask"      : True → ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over feature dimension (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                            # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast once to fp16 – the rest of the heavy arithmetic runs in fp16
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape
    batch_seq = B * N * N

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear maps (left_proj, left_gate, right_proj,
    #    right_gate, out_gate) into a single GEMM.
    # ------------------------------------------------------------------
    # Concatenate weight matrices (shape: (5*H, dim)) and move to FP16
    w_combined = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # (5*H, dim)

    # Flatten the input to (B·N·N, dim) and perform the batched linear op
    x_flat = x_h.view(batch_seq, dim)        # (B·N·N, dim)
    combined = torch.nn.functional.linear(x_flat, w_combined)  # (B·N·N, 5*H)
    combined = combined.view(B, N, N, 5 * hidden_dim)          # (B, N, N, 5*H)

    # ------------------------------------------------------------------
    # 3️⃣ Slice the fused tensor into its components
    # ------------------------------------------------------------------
    split = torch.split(combined, hidden_dim, dim=3)
    left_proj, left_gate_lin, right_proj, right_gate_lin, out_gate_lin = split
    # out_gate used later (after sigmoid)
    out_gate = torch.sigmoid(out_gate_lin)   # (B, N, N, H)  FP16

    # ------------------------------------------------------------------
    # 4️⃣ Apply gate + optional mask using a tiny Triton kernel.
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] (FP16) → flatten for broadcasting
        mask_f = mask.to(torch.float16).unsqueeze(-1)          # (B,N,N,1)
        mask_flat = mask_f.view(-1)                            # (B·N·N)

        # Allocate output tensors for the gated & masked streams
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Triton launch configuration
        BLOCK = 65536
        n_elem = left_proj.numel()                             # B·N·N·H
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * sigmoid(left_gate_lin) * mask
        _proj_gate_mask_kernel[grid](
            left_proj,
            left_gate_lin,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * sigmoid(right_gate_lin) * mask
        _proj_gate_mask_kernel[grid](
            right_proj,
            right_gate_lin,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask – compute gated tensors with pure PyTorch ops
        left = left_proj * torch.sigmoid(left_gate_lin)
        right = right_proj * torch.sigmoid(right_gate_lin)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³) operation: batch GEMM (A @ Bᵀ) over the hidden dimension.
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) where H is the hidden dimension
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B·H, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # (B·H, N, N)

    # Batched matrix multiplication – FP16×FP16 → FP32 accumulation
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))          # (B·H, N, N), FP16 result

    # Restore original layout and promote to FP32 for the following LayerNorm
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)                # (B, N, N, H)
        .contiguous()
        .to(torch.float32)
    )   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: hidden‑dim LayerNorm → out_gate → final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # (B, N, N, H) – FP32

    # Multiply by out_gate (cast gate to FP32 for the product)
    out = out * out_gate.to(torch.float32)   # (B, N, N, H)  FP32

    # Final linear projection back to the original dimension (FP16 path)
    out = F.linear(
        out.to(torch.float16),
        weights["to_out.weight"].to(torch.float16),
    ).to(torch.float32)                     # (B, N, N, dim)

    return out
