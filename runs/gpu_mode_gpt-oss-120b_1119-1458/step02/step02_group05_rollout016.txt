@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    hidden_dim, n_elements,
    BLOCK: tl.constexpr,
    WITH_MASK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    # Load projection and gate raw
    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate_raw = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)
    # Compute sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))
    if WITH_MASK:
        # compute mask index
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)
        out = proj * gate * mask_val
    else:
        out = proj * gate
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)
```

"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

Algorithm
---------
1️⃣ LayerNorm over the feature dimension (float32 for numerical stability).
2️⃣ Cast to FP16 and perform **one** fused linear projection that produces all
   required intermediate tensors (left/right projections, left/right gates,
   out_gate) in a single matmul.
3️⃣ Apply sigmoid to the gate tensors.
4️⃣ If a mask is supplied, fuse the element‑wise product
   `proj * gate * mask` with a tiny Triton kernel (avoids extra memory passes).
   When no mask is present we simply multiply `proj * gate` (FP16).
5️⃣ Core O(N³) operation: batch GEMM with hidden dimension as the batch:
   `C = A @ Bᵀ` where `A` and `B` are (B·H, N, N) half‑precision tensors.
   This uses `torch.bmm`, which maps to a Tensor‑Core fused GEMM on H100.
6️⃣ LayerNorm over the hidden dimension (float32), multiply by the out‑gate,
   and project back to the original feature dimension with a half‑precision
   linear layer.
7️⃣ Return the result as float32.

Only the mask‑fusion step requires a Triton kernel – all heavy arithmetic is
performed with Tensor‑Core‑accelerated GEMMs in mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: proj * sigmoid(gate) * mask  (mask optional)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,    # fp16 projection (left or right)
    gate_ptr,    # fp16 raw gate (pre‑sigmoid)
    mask_ptr,    # fp16 mask, flattened, length = B * N * N
    out_ptr,     # fp16 destination
    hidden_dim: tl.constexpr,   # hidden dimension (for mask broadcasting)
    n_elements: tl.constexpr,   # total number of elements (B·N·N·H)
    BLOCK: tl.constexpr,        # compile‑time block size
    WITH_MASK: tl.constexpr,    # 1 → mask present, 0 → no mask
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load projection and raw gate (fp16 → fp32 for higher‑precision ops)
    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate_raw = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # Sigmoid (1 / (1 + exp(-x))) in fp32
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    if WITH_MASK:
        # Broadcast mask: each mask entry repeats `hidden_dim` times
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)
        out = proj * gate * mask_val
    else:
        out = proj * gate

    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, dim]
        torch.Tensor,          # mask tensor  [B, N, N]  (may be dummy if nomask)
        Dict[str, torch.Tensor],
        Dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and fuse all linear projections into one matmul
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                                   # [B, N, N, dim]

    # Stack all five weight matrices (left_proj, left_gate, right_proj,
    # right_gate, out_gate) into a single weight of shape
    # (5*hidden_dim, dim) to read the input only once.
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    # Shape check: all linear layers are bias‑free, weight shape = (out, in)
    combined_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)                                         # [5*H, dim]

    # Flatten the input to (B*N*N, dim) and apply the fused projection.
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                                 # [(B·N·N), dim]
    combined = F.linear(x_flat, combined_weight)               # [(B·N·N), 5*H] (fp16)

    # Reshape back to 4‑D tensor.
    combined = combined.view(B, N, N, 5 * hidden_dim)           # [B, N, N, 5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Slice the combined tensor into the individual components.
    # ------------------------------------------------------------------
    h0 = 0
    h1 = hidden_dim
    h2 = 2 * hidden_dim
    h3 = 3 * hidden_dim
    h4 = 4 * hidden_dim
    h5 = 5 * hidden_dim

    left_proj      = combined[..., h0:h1]   # [B, N, N, H]
    left_gate_raw  = combined[..., h1:h2]   # [B, N, N, H]
    right_proj     = combined[..., h2:h3]   # [B, N, N, H]
    right_gate_raw = combined[..., h3:h4]   # [B, N, N, H]
    out_gate_raw   = combined[..., h4:h5]   # [B, N, N, H]

    # Apply sigmoid to gate tensors (still in fp16).
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw).to(torch.float32)     # keep as fp32 later

    # ------------------------------------------------------------------
    # 4️⃣ Fuse (proj * gate * mask) with a Triton kernel (if mask present).
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast → fp16 → flattened
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                                 # [B·N·N]

        # Allocate output tensors for the gated‑and‑masked values
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Triton launch configuration
        BLOCK = 16384                       # number of elements per program
        n_elem = left_proj.numel()          # total # of elements (B·N·N·H)
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * sigmoid(left_gate_raw) * mask
        _proj_gate_mask_kernel[grid](
            left_proj,
            left_gate_raw,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
            WITH_MASK=1,
        )
        # right = right_proj * sigmoid(right_gate_raw) * mask
        _proj_gate_mask_kernel[grid](
            right_proj,
            right_gate_raw,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
            WITH_MASK=1,
        )
    else:
        # No mask – simple element‑wise product (fp16)
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³) operation: batch GEMM A @ Bᵀ (hidden as batch dim)
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) – note that we **avoid** an eager .contiguous()
    # call; the view is sufficient because the layout after permute is already
    # compatible for `torch.bmm`.
    left_batched  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication – Tensor‑Core GEMM (FP16×FP16→FP32)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))

    # Restore shape [B, N, N, H] and cast to FP32 for the following LayerNorm.
    out = out_batched.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, H] (float32)

    out = out * out_gate   # broadcast over hidden dim

    # Final projection back to the original feature dimension.
    # Perform it in FP16 for speed, then cast back to FP32.
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.to(torch.float16), to_out_weight_h).to(torch.float32)

    return out
