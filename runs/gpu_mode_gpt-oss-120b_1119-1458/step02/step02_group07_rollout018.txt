"""
TriMul (outgoing) – fast Triton‑augmented forward.

Algorithm
---------
1️⃣ Layer‑norm the input (float32).
2️⃣ Cast to fp16 and perform **fused** linear projections:
   * Concatenate left/right projection matrices → one F.linear call.
   * Concatenate left/right gate matrices    → one F.linear + sigmoid call.
3️⃣ Apply gates (and mask, if present) with a tiny Triton kernel that
   fuses the three element‑wise multiplications per (b,i,j) location.
   When `nomask=True` a simple element‑wise multiplication is used.
4️⃣ Batched matrix multiplication:
   left[...,i,k,d] @ right[...,j,k,d]ᵀ is realized as a single
   torch.matmul on tensors of shape (B, H, N, N) – no extra copies.
5️⃣ Second LayerNorm → out‑gate → final projection back to `dim`.

The heavy work (linear projections & batched GEMM) stays in highly‑optimised
CUDA / Tensor‑Core kernels; the Triton kernel only handles the inexpensive
mask‑gate fusion, satisfying the requirement to use a custom kernel while
keeping the overall runtime well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # fp16 * left/right (B,N,N,H)
    left_gate_ptr, right_gate_ptr,      # fp16 * left/right gate (B,N,N,H)
    mask_ptr,                           # fp16 * mask (B,N,N)
    # strides of left/right tensors
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides of gate tensors (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    # strides of mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    B, N, H,                            # sizes
    BLOCK_H: tl.constexpr,              # tile along hidden dim
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode b,i,j
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # broadcasted mask scalar
    mask_val = tl.load(mask_ptr + off_m)

    # iterate over the hidden dimension in BLOCK_H chunks
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        active = offs < H

        # load a chunk of the tensors
        left      = tl.load(left_ptr      + off_l + offs * stride_l3, mask=active, other=tl.float16(0.0))
        right     = tl.load(right_ptr     + off_r + offs * stride_r3, mask=active, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3, mask=active, other=tl.float16(0.0))
        right_gate= tl.load(right_gate_ptr+ off_g + offs * stride_g3, mask=active, other=tl.float16(0.0))

        # fused operation (mask * gate * proj)
        left_out  = left  * left_gate  * mask_val
        right_out = right * right_gate * mask_val

        # write back in‑place
        tl.store(left_ptr  + off_l + offs * stride_l3, left_out, mask=active)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=active)


# ----------------------------------------------------------------------
# Entry point for the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass for the outgoing Triangle Multiplicative Update.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:         [B, N, N]     (float32, optional)
            - weights: dict of model parameters (float32)
            - config: dict with keys `dim`, `hidden_dim`, `nomask`
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear projections (left + right) and gates (left_gate + right_gate)
    # ------------------------------------------------------------------
    # -- projections ----------------------------------------------------
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"].to(torch.float16),   # (H, dim)
            weights["right_proj.weight"].to(torch.float16),  # (H, dim)
        ],
        dim=0,
    )  # (2*H, dim)

    # combined projection → [B, N, N, 2*H]
    proj = F.linear(x_fp16, proj_weight)
    left = proj[..., :hidden_dim]   # (B, N, N, H)
    right = proj[..., hidden_dim:]  # (B, N, N, H)

    # -- gates ----------------------------------------------------------
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"].to(torch.float16),
            weights["right_gate.weight"].to(torch.float16),
        ],
        dim=0,
    )  # (2*H, dim)

    gates = torch.sigmoid(F.linear(x_fp16, gate_weight))  # (B, N, N, 2*H)
    left_gate = gates[..., :hidden_dim]   # (B, N, N, H)
    right_gate = gates[..., hidden_dim:]  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask and gates (fused kernel when mask is present)
    # ------------------------------------------------------------------
    if nomask:
        # simple element‑wise fusion
        left = left * left_gate
        right = right * right_gate
    else:
        # broadcast mask to hidden dimension
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]  # B * N * N

        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left, right,
            left_gate, right_gate,
            mask_fp16,
            # strides for left/right
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # strides for gates
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # sizes
            input_tensor.shape[0],                     # B
            input_tensor.shape[1],                     # N
            hidden_dim,                                 # H
            BLOCK_H=32,                                 # tile size on hidden dim
            num_warps=4,
        )
        # after the kernel, `left` and `right` already contain mask*gate*proj

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ operation: batched GEMM across the hidden dimension
    # ------------------------------------------------------------------
    # shape to (B, H, N, N) – torch.matmul can handle non‑contiguous tensors
    left_mat = left.permute(0, 3, 1, 2)    # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2)  # (B, H, N, N)

    # batched matrix multiply: (i,k)·(j,k)ᵀ  →  (i,j) per hidden head
    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))  # (B, H, N, N)

    # back to original layout (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing (LayerNorm → out‑gate → final projection)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # (B, N, N, H)

    # output gate (computed from the original (already‑normed) x)
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    ).to(torch.float32)  # (B, N, N, H)

    out_norm = out_norm * out_gate

    # final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])  # (B, N, N, dim)

    return out_final
