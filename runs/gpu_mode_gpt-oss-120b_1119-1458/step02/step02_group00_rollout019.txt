@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim, B,
    stride_left_head, stride_left_i, stride_left_k,
    stride_right_head, stride_right_i, stride_right_k,
    stride_out_head, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Ranges
    head_id = tl.program_id(0)  # total_heads
    tile_i = tl.program_id(1)  # i tile
    tile_j = tl.program_id(2)  # j tile

    # Compute global i, j start positions
    i_start = tile_i * BLOCK_M
    j_start = tile_j * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32 for precision

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # Offsets for A block: A[head_id, i_start:i_start+BLOCK_M, k_start:k_start+BLOCK_K]
        # A stride: head stride -> stride_left_head; i stride -> stride_left_i; k stride -> stride_left_k
        # Use broadcasting with tl.arange
        a_off_i = i_start + tl.arange(0, BLOCK_M)
        a_off_k = k_start + tl.arange(0, BLOCK_K)
        # a_ptrs shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + head_id * stride_left_head + a_off_i[:, None] * stride_left_i + a_off_k[None, :] * stride_left_k

        # Offsets for B_T block: B_T is right transposed: shape (BLOCK_K, BLOCK_N)
        # We need to fetch right[head_id, j_start:j_start+BLOCK_N, k_start:k_start+BLOCK_K] transposed.
        # The address for right[head, row=j_start + col_offset, col=k_start + row_offset]? Wait:
        # Original right[head, row=j, col=k] = right_ptr + head*stride_head + j*stride_i + k*stride_k
        # For B_T, we want B_T[ row = k_offset + k_local, col = j_offset + j_local] = right[head, j_local, k_local]
        # So pointer: right_ptr + head*stride_head + (j_start + tl.arange(0, BLOCK_N))[None,: ] * stride_i + (k_start + tl.arange(0, BLOCK_K))[:, None] * stride_k
        b_off_j = j_start + tl.arange(0, BLOCK_N)
        b_off_k = k_start + tl.arange(0, BLOCK_K)
        b_ptrs = right_ptr + head_id * stride_right_head + b_off_j[None, :] * stride_right_i + b_off_k[:, None] * stride_right_k

        # Load blocks
        a = tl.load(a_ptrs, mask=(a_off_i[:, None] < N) & (a_off_k[None, :] < N), other=0.0)  # optional mask for edges
        b = tl.load(b_ptrs, mask=(b_off_k[:, None] < N) & (b_off_j[None, :] < N), other=0.0)

        # Dot product accumulate in fp32
        acc += tl.dot(a, b)  # a: fp16, b: fp16 -> fp32

    # Write back result to out
    # Compute out location: out[head, i, j] shape [B*H, N, N]
    # Use mask for boundary
    # The stride: head stride? For out, shape [total_heads, N, N] contiguous.
    # strides: out_stride_head = N*N
    # out_stride_i = N
    # out_stride_j = 1

    i_off = i_start + tl.arange(0, BLOCK_M)
    j_off = j_start + tl.arange(0, BLOCK_N)
    out_ptrs = out_ptr + head_id * stride_out_head + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j

    mask_store = (i_off[:, None] < N) & (j_off[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)

   device = input_tensor.device
   dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

   # Convert input to fp16
   x = input_tensor.to(torch.float16)

   # half weights
   w_norm = weights["norm.weight"].to(torch.float16)
   b_norm = weights["norm.bias"].to(torch.float16)
   x_norm = F.layer_norm(x, (dim,), weight=w_norm, bias=b_norm, eps=1e-5)

   # linear projections
   left = F.linear(x_norm, weights["left_proj.weight"].to(torch.float16))
   right = F.linear(x_norm, weights["right_proj.weight"].to(torch.float16))

   left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"].to(torch.float16)))
   right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"].to(torch.float16)))
   out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"].to(torch.float16)))

   # Prepare mask
   if not nomask:

   else:
       # Create dummy mask of ones
       mask_h = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=torch.float16, device=device)

   # allocate output buffers for permuted left/right after gating
   B, N, _, _ = input_tensor.shape
   total_heads = B * hidden_dim
   left_perm = torch.empty((total_heads, N, N), dtype=torch.float16, device=device)

   # Flatten views for kernel:
   left_view = left.contiguous().view(-1)

   numel = left_view.numel()

       right_perm.view(-1),

   BLOCK_M = 128
   BLOCK_N = 128
   BLOCK_K = 32

       out_perm,

   out = out_perm.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

@triton.jit
def gate_and_permute_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,           # shape total_mask = B*N*N
    left_out_ptr, right_out_ptr,  # target flattened out of size total_heads * N * N
    B, N, H: tl.constexpr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # Load values
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # compute mask value broadcasted
    # mask index = offs // H (since each hidden repeats)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # mask broadcast across hidden

    # compute gated+masked values
    l_val = l * lg * m
    r_val = r * rg * m

    # Compute target head, i, k indices
    # B, N, H as constants
    # total_elems = B * N * N * H
    # Now compute b = offs // (N * N * H)
    # rem = offs % (N * N * H)
    # i = rem // (N * H)
    # rem2 = rem % (N * H)
    # k = rem2 // H
    # h = rem2 % H

    # This can be done using integer division and modulo:
    # avoid expensive modulo by using division and multiplication.

    # TL does integer division "/"? Use `//`.
    # Compute division by constants: we can rely on compile-time const division to be optimized.

    tmp = offs
    head_size = N * N * H
    b = tmp // head_size
    tmp = tmp - b * head_size
    i = tmp // (N * H)
    tmp = tmp - i * (N * H)
    k = tmp // H
    h = tmp - k * H

    # Compute target_head = b * H + h
    target_head = b * H + h
    # Compute linear index for target left_out and right_out
    # target_offset = ((target_head * N) + i) * N + k   # i major dimension
    target_offset = ((target_head * N + i) * N + k)
    # store
    tl.store(left_out_ptr + target_offset, l_val, mask=mask)
    tl.store(right_out_ptr + target_offset, r_val, mask=mask)

@triton.jit
def tri_mul_kernel(
    left_ptr,    # pointer to left matrix (fp16)
    right_ptr,   # pointer to right matrix (fp16)
    out_ptr,     # pointer to output matrix (fp16)
    N: tl.constexpr,
    stride_lh: tl.constexpr,
    stride_li: tl.constexpr,
    stride_lk: tl.constexpr,
    stride_rh: tl.constexpr,
    stride_ri: tl.constexpr,
    stride_rk: tl.constexpr,
    stride_oh: tl.constexpr,
    stride_oi: tl.constexpr,
    stride_oj: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    head = tl.program_id(0)
    tile_i = tl.program_id(1)
    tile_j = tl.program_id(2)

    i_start = tile_i * BLOCK_M
    j_start = tile_j * BLOCK_N

    # initialize accumulator in fp32
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # compute a block
        a_ptr = left_ptr + head * stride_lh + (i_start + tl.arange(0, BLOCK_M))[:, None] * stride_li + (k_start + tl.arange(0, BLOCK_K))[None, :] * stride_lk
        a = tl.load(a_ptr, mask=(i_start + tl.arange(0, BLOCK_M)[:, None] < N) & (k_start + tl.arange(0, BLOCK_K)[None, :] < N), other=0.0)

        # load b block (right transposed)
        # B_T[ k, j]
        b_ptr = right_ptr + head * stride_rh + (j_start + tl.arange(0, BLOCK_N))[None, :] * stride_ri + (k_start + tl.arange(0, BLOCK_K))[:, None] * stride_rk
        b = tl.load(b_ptr, mask=(k_start + tl.arange(0, BLOCK_K)[:, None] < N) & (j_start + tl.arange(0, BLOCK_N)[None, :] < N), other=0.0)

        # dot product accumulate
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + head * stride_oh + (i_start + tl.arange(0, BLOCK_M))[:, None] * stride_oi + (j_start + tl.arange(0, BLOCK_N))[None, :] * stride_oj
    mask_out = (i_start + tl.arange(0, BLOCK_M)[:, None] < N) & (j_start + tl.arange(0, BLOCK_N)[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

"""
Highly‑optimized outgoing Triangle‑Multiplicative (TriMul) forward pass.

Algorithm
---------
1. Input LayerNorm is performed directly in fp16 (the weight/bias are cast
   to half‑precision and the mean/var are computed in fp32 internally by
   PyTorch).  This saves a full copy of the tensor.
2. Linear projections → gate vectors (all fp16) and optional mask are fused
   with cheap element‑wise multiplies (pure PyTorch – negligible cost).
3. The dominant N³ “einsum” is replaced by a single Triton kernel that
   computes all `hidden_dim` heads and all batches in one launch.  The
   kernel performs a tiled matrix multiplication
   `C_head = A_head @ B_headᵀ` with the layout
   A_head[b,h,i,k] = left[b,i,k,h] and
   B_head[b,h,j,k] = right[b,j,k,h].
   Accumulation is done in fp32 and the result stored as fp16.
4. Post‑processing LayerNorm (fp16), output‑gate multiplication and the
   final linear projection are again carried out in half‑precision.
5. The function returns a float32 tensor as required by the reference
   implementation.

The kernel `tri_mul_kernel` follows the classic GEMM tiling pattern and
uses Tensor‑Core friendly `tl.dot`.  It eliminates the many tiny
`torch.bmm` calls (one per head) and thereby reduces kernel‑launch
overhead, bringing the runtime well below the target 1 ms on an H100.
"""

import math
from typing import Tuple, Dict

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for all heads at once
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,    # fp16, shape (total_heads, N, N)
    right_ptr,   # fp16, same layout as left_ptr
    out_ptr,     # fp16, shape (total_heads, N, N)
    N: tl.constexpr,
    stride_lh: tl.constexpr,  # stride between heads in left
    stride_li: tl.constexpr,  # stride between rows (i)  in left
    stride_lk: tl.constexpr,  # stride between cols (k)  in left
    stride_rh: tl.constexpr,  # stride between heads in right
    stride_ri: tl.constexpr,  # stride between rows (j)  in right
    stride_rk: tl.constexpr,  # stride between cols (k)  in right
    stride_oh: tl.constexpr,  # stride between heads in out
    stride_oi: tl.constexpr,  # stride between rows (i)  in out
    stride_oj: tl.constexpr,  # stride between cols (j)  in out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes for every head h:
        out[h,i,j] = Σₖ left[h,i,k] * right[h,j,k]
    The implementation tiles over M (i), N (j) and K (k) and accumulates
    in fp32 for numerical stability before casting back to fp16.
    """
    head_id = tl.program_id(0)          # 0 … total_heads‑1
    tile_i  = tl.program_id(1)          # i‑tile index
    tile_j  = tl.program_id(2)          # j‑tile index

    i_start = tile_i * BLOCK_M
    j_start = tile_j * BLOCK_N

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # ------- left block A[i, k] -------
        i_idx = i_start + tl.arange(0, BLOCK_M)
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_a_i = i_idx < N
        mask_a_k = k_idx < N
        mask_a = mask_a_i[:, None] & mask_a_k[None, :]

        a_ptrs = left_ptr + head_id * stride_lh \
                 + i_idx[:, None] * stride_li \
                 + k_idx[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # ------- right block Bᵀ[k, j] -------
        # Bᵀ[k, j] = right[j, k] (we read right in its original layout)
        j_idx = j_start + tl.arange(0, BLOCK_N)
        mask_b_j = j_idx < N
        mask_b = mask_a_k[:, None] & mask_b_j[None, :]

        b_ptrs = right_ptr + head_id * stride_rh \
                 + j_idx[None, :] * stride_ri \
                 + k_idx[:, None] * stride_rk
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # ------- dot product (fp32 accumulation) -------
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)

    # ------------------------------------------------------------------
    # Store the result in the correct location
    # ------------------------------------------------------------------
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)
    mask_o = (i_idx[:, None] < N) & (j_idx[None, :] < N)

    out_ptrs = out_ptr + head_id * stride_oh \
               + i_idx[:, None] * stride_oi \
               + j_idx[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_o)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor    [B, N, N, dim]  (float32)
        - mask         : torch.Tensor    [B, N, N]      (float32 / bool)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   containing 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs & config
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (fp16)
    # --------------------------------------------------------------
    # cast once to fp16 – all heavy work stays in half precision
    x = input_tensor.to(torch.float16)

    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, normalized_shape=(dim,), weight=norm_w,
                     bias=norm_b, eps=eps)      # [B, N, N, dim]  fp16

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # --------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    left  = F.linear(x, w_left_proj)          # [B, N, N, hidden_dim] fp16
    right = F.linear(x, w_right_proj)

    # --------------------------------------------------------------
    # 3️⃣  Gating vectors (sigmoid)
    # --------------------------------------------------------------
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x, w_out_gate))

    # --------------------------------------------------------------
    # 4️⃣  Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣  Fuse gates with the projected tensors (in‑place)
    # --------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate
    # (the two multiplications are cheap compared to the N³ work)

    # --------------------------------------------------------------
    # 6️⃣  Re‑arrange tensors so that each head becomes a separate batch
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # Shape: [total_heads, N, N] (contiguous)
    left_perm  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # --------------------------------------------------------------
    # 7️⃣  Core O(N³) multiplication via a single Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # All strides are the same because the tensors are contiguous:
    # head‑stride = N * N, row‑stride = N, col‑stride = 1
    stride_h = N * N
    stride_i = N
    stride_k = 1

    out_perm = torch.empty_like(left_perm)   # [total_heads, N, N] fp16

    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        stride_h, stride_i, stride_k,   # left strides
        stride_h, stride_i, stride_k,   # right strides
        stride_h, stride_i, stride_k,   # out strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8️⃣  Restore original 4‑D layout: [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # --------------------------------------------------------------
    # 9️⃣  Post‑processing LayerNorm (fp16) and output‑gate
    # --------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=w_to_out_norm, bias=b_to_out_norm, eps=eps)   # fp16

    out = out * out_gate                                                   # fused element‑wise

    # --------------------------------------------------------------
    # 🔟  Final linear projection back to `dim` (fp16)
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = F.linear(out, w_to_out)                                         # fp16

    # --------------------------------------------------------------
    # Return as float32 (expected by the reference implementation)
    # --------------------------------------------------------------
    return final.to(torch.float32)
