"""
TriMul (outgoing) forward‑pass, heavily fused with Triton.

Algorithm
---------
1️⃣  Layer‑Norm over the last channel (FP32 for numeric stability).
2️⃣  Cast to FP16 – the bulk of the compute (linear projections,
    gated element‑wise products and the N³ “einsum”) is performed
    in half‑precision.
3️⃣  Linear projections (left/right) and gate‑pre‑activations are
    computed with `torch.nn.functional.linear`.  The gate‑tensors
    are passed through a sigmoid.
4️⃣  Optional pair‑wise mask (broadcast over the hidden dimension) is
    applied in FP16 using a single torch multiplication (very cheap).
5️⃣  **Triton kernel** `apply_gates_kernel` fuses the two in‑place
    element‑wise products:
        left  ← left  * left_gate
        right ← right * right_gate
   This avoids two separate kernel launches and a temporary copy.
6️⃣  Core N³ operation:  `out[i,j,d] = Σₖ left[i,k,d]·right[j,k,d]`.
    By reshaping to (batch·head, N, N) we can use a batched
    half‑precision GEMM (`torch.bmm`), which on an H100 maps to
    Tensor‑Cores and is far faster than a naïve Triton triple‑loop.
7️⃣  Post‑projection LayerNorm (FP16 weights) followed by the
    output‑gate (`out_gate`) multiplied with another **Triton kernel**
    `elementwise_mul_kernel`.
8️⃣  Final linear projection back to the original `dim` (FP16 → FP16)
    and cast the result to FP32.

All heavy element‑wise work is performed by Triton kernels,
while the large matrix multiplication leverages the highly‑optimised
cuBLAS/Tensor‑Core path.  The function returns a float‑32 tensor
with shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,       # fp16,   shape (total_elems,)
    left_gate_ptr,  # fp16,   same shape as left_ptr
    right_ptr,      # fp16,   shape (total_elems,)
    right_gate_ptr, # fp16,   same shape as right_ptr
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place multiplication:
        left  = left  * left_gate
        right = right * right_gate
    The tensors are flattened beforehand.  One program processes
    BLOCK contiguous elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # left side
    l  = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    # right side
    r  = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def elementwise_mul_kernel(
    a_ptr,   # fp16 flattened
    b_ptr,   # fp16 flattened (same numel)
    out_ptr, # fp16 flattened
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    out = a * b  (both flattened)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask         : torch.Tensor  [B, N, N]      (bool or float)
        - weights      : dict of torch.Tensor – model parameters
        - config       : dict – contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack input / config
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5
    device     = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → keep stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast weights to FP16 (linear layers have no bias)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, w_right_proj)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gate pre‑activations (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] → fp16, broadcast via * 
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # (B,N,N,1)
        left  = left * mask_h
        right = right * mask_h
        # `out_gate` does **not** use the mask (as in the reference code)

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates using a tiny Triton kernel
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous before flattening – this guarantees
    # that the flattened view matches the memory layout expected by the kernel.
    left, right = left.contiguous(), right.contiguous()
    left_gate, right_gate = left_gate.contiguous(), right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536  # 64k elements per program (empirically good on H100)
    grid = ( (numel + BLOCK - 1) // BLOCK, )

    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Release gate tensors early to keep peak memory low
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation – batched GEMM (A @ Bᵀ)
    #    Shapes: left/right  -> (B, N, N, hidden_dim)
    #    We reshape to (B*hidden_dim, N, N) and use torch.bmm.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    total_heads = B * H  # B * hidden_dim

    # Permute to (B, hidden_dim, N, N) then fuse batch and head dimensions.
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Batched FP16 GEMM – leverages Tensor‑Core paths on H100.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (head, N, N)

    # Restore to original layout: (B, N, N, hidden_dim)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm (half‑precision weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 🔟 Multiply by output gate (another tiny Triton kernel)
    # ------------------------------------------------------------------
    out_gate = out_gate.contiguous()
    out_mul = torch.empty_like(out)

    numel_out = out.numel()
    grid_out = ( (numel_out + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim` (still fp16)
    # ------------------------------------------------------------------
    final = F.linear(out_mul, w_to_out)    # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 12️⃣ Cast to float32 as required by the surrounding model
    # ------------------------------------------------------------------
    return final.to(torch.float32)
