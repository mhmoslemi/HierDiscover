@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3, # for right similarly
    stride_lg0, stride_lg1, stride_lg2, stride_lg3, # for left_gate/right_gate same shape as left/right
    stride_m0, stride_m1, stride_m2,
    B, N, H,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets
    base_left = b*stride_l0 + i*stride_l1 + j*stride_l2
    base_right = b*stride_rg0 + i*stride_rg1 + j*stride_rg2
    base_left_gate = b*stride_lg0 + i*stride_lg1 + j*stride_lg2
    base_right_gate = b*stride_rg0 + i*stride_rg1 + j*stride_rg2 # same as right ptr layout

"""
Optimised outgoing Triangle Multiplicative Update (TriMul) for AlphaFold‑3.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32).
2️⃣ Cast to FP16 and perform *one* fused linear projection that simultaneously
   computes the left/right projections **and** their gates:
        [left_proj, left_gate, right_proj, right_gate] = x @ W_fused.T
   (W_fused = concat[left_proj, left_gate, right_proj, right_gate]).

3️⃣ Apply sigmoid to the gate slices (FP16).

4️⃣ If a mask is present, fuse the element‑wise triple product
   `proj * gate * mask` for the left and right tensors with a tiny Triton
   kernel (one program per (b,i,j) element).  When there is no mask the
   multiplication is done in‑place with a single torch kernel.

5️⃣ Reshape the masked‑gated tensors to (B·H, N, N) and perform a
   batched half‑precision GEMM:
        C_h = left_h @ right_hᵀ          (h = batch·hidden_dim)
   This is the core N³ computation realised as a strided‐batched GEMM.

6️⃣ Convert the result to float32, apply a second LayerNorm over the hidden
   dimension, multiply by the *out‑gate* (computed from the same x), and
   finally project back to the original channel dimension with a FP16 linear
   layer.

All heavy matrix ops are FP16 (Tensor‑Core accelerated) and the only
explicit Triton kernel fuses the mask‑gate‑projection step, reducing memory
traffic.  The implementation works for any (B, N, dim, hidden_dim) allowed
by the test matrix and returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 * left projection   (B,N,N,hd)
    right_ptr,              # fp16 * right projection  (B,N,N,hd)
    mask_ptr,               # fp16 * mask              (B,N,N)
    left_gate_ptr,          # fp16 * left gate         (B,N,N,hd)
    right_gate_ptr,         # fp16 * right gate        (B,N,N,hd)
    # strides for left, right (b,i,j,hd)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    # strides for gates (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,                # batch, seq_len, hidden_dim
    BLOCK_H: tl.constexpr, # tile size along hidden dimension
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for each tensor
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # broadcast mask scalar (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in BLOCK_H‑sized chunks
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left      = tl.load(left_ptr       + off_l + offs * stride_l3,
                            mask=mask_h, other=tl.float16(0.0))
        right     = tl.load(right_ptr      + off_r + offs * stride_r3,
                            mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr  + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate= tl.load(right_gate_ptr + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))

        # fused triple product
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val

        # store in‑place
        tl.store(left_ptr  + off_l + offs * stride_l3, left,
                 mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right,
                 mask=mask_h)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask        : [B, N, N]         (float32) – may be ignored when `nomask=True`
        - weights     : dict of tensors (float32)
        - config      : dict with keys
            * dim        – channel dimension of the input
            * hidden_dim – internal hidden dimension (usually 128)
            * nomask     – True → mask is all‑ones and can be skipped

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)    # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ One fused projection that yields left/right proj + gates
    # ------------------------------------------------------------------
    # concatenate the four weight matrices (out_features, in_features)
    proj_weight_fp16 = torch.cat(
        [
            weights["left_proj.weight"],      # (hd, dim)
            weights["left_gate.weight"],      # (hd, dim)
            weights["right_proj.weight"],     # (hd, dim)
            weights["right_gate.weight"],     # (hd, dim)
        ],
        dim=0,                               # (4*hd, dim)
    ).to(torch.float16)

    # fused linear → [B,N,N,4*hd]  fp16
    fused = F.linear(x_fp16, proj_weight_fp16)

    # split the four tensors
    hd = hidden_dim
    left_proj      = fused[..., 0:hd]          # (B,N,N,hd)
    left_gate_raw  = fused[..., hd:2*hd]       # (B,N,N,hd)
    right_proj     = fused[..., 2*hd:3*hd]     # (B,N,N,hd)
    right_gate_raw = fused[..., 3*hd:4*hd]     # (B,N,N,hd)

    # activate gates
    left_gate  = torch.sigmoid(left_gate_raw)   # fp16
    right_gate = torch.sigmoid(right_gate_raw)  # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate (separate linear, still fp16)
    # ------------------------------------------------------------------
    out_gate_weight_fp16 = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight_fp16))   # [B,N,N,hd] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask & gate – fused via Triton when mask is present
    # ------------------------------------------------------------------
    if not nomask:
        # mask must be fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        # make sure all tensors are contiguous for correct stride calculations
        left_proj  = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]  # B * N * N
        # Launch kernel which writes the masked‑gated results back in‑place
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left_proj,               # fp16 *
            right_proj,              # fp16 *
            mask_fp16,               # fp16 *
            left_gate,               # fp16 *
            right_gate,              # fp16 *
            # strides for left (b,i,j,hd)
            left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
            # strides for right (b,i,j,hd)
            right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
            # strides for mask (b,i,j)
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gates (both gates share layout with left/right)
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # sizes
            input_tensor.shape[0],       # B
            input_tensor.shape[1],       # N
            hidden_dim,                  # H
            BLOCK_H=32,                  # tile along hidden dim (covers 128‑dim heads)
            num_warps=4,
        )
        # after the kernel left_proj and right_proj already contain (proj * gate * mask)
        left  = left_proj
        right = right_proj
    else:
        # No mask → simple in‑place fusion
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ operation: batched GEMM per hidden dimension (FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    heads = B * hidden_dim

    # reshape to (heads, N, N) where each head = one hidden slice of one batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(heads, N, N)

    # GEMM: left @ rightᵀ  →  fp16 accumulation (Tensor‑Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]  (still fp16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, hidden_dim]  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (still in FP16 – PyTorch internally uses FP32 accum.)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final linear projection (all FP16)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                # element‑wise gate, fp16

    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)
    out_final = F.linear(out_norm, to_out_weight_fp16)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the API contract
    # ------------------------------------------------------------------
    return out_final.to(torch.float32)
