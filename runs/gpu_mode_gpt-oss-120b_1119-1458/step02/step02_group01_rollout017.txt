@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0) # i tile
    pid_n = tl.program_id(1) # j tile
    pid_z = tl.program_id(2) # batch*hidden channel

    # compute offsets for i, j within N
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # # Number of k-tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        # compute k start offset
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_z * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < K), other=0.0)

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N)
        # B layout [M, K, N] after permute & view
        b_ptrs = B_ptr + pid_z * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_j[None, :], other=0.0)

        # Perform dot and accumulate into fp32 accumulator
        acc = tl.dot(a, b, out_dtype=tl.float32, accumulate=acc)

    # Write result to C (FP16)
    c_ptrs = C_ptr + pid_z * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) with dims (i,k) for left and (k,j) for right
    # left: (b,i,k,h) -> permute(b,h,i,k)
    left_permuted = left.permute(0, 3, 1, 2).contiguous()
    A = left_permuted.view(B * H, N, N)  # shape (M,N,K) where M = B*H

    # right: (b,i,k,h) -> need (b,h,k,i)
    right_permuted = right.permute(0, 3, 2, 1).contiguous()
    B_ = right_permuted.view(B * H, N, N)  # shape (M,K,N)

    # allocate C
    C = torch.empty_like(A, dtype=torch.float16)

    # stride configuration
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Block size selection
    N_val = N
    # choose block sizes based on N
    if N_val <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    elif N_val <= 512:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    # reshape back
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C
```

"""
TriMul (outgoing) forward pass – highly optimized Triton kernel.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32 for numerical stability).
2️⃣  Compute the four linear maps (`left_proj`, `right_proj`,
    `left_gate`, `right_gate`) and the output gate in float32.
3️⃣  (Optional) mask the projected tensors.
4️⃣  Fuse the scalar gates with the projections (still float32).
5️⃣  Cast the gated tensors to fp16 and run a batched GEMM kernel that
    computes, for every hidden channel `h`

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    i.e. a matrix‑multiply `left @ rightᵀ`.  The kernel tiles the
    multiplication with a 64×64×64 (or 128×128×64) tile, uses
    `tl.dot(..., out_dtype=tl.float32, accumulate=acc)` so that the
    multiplication is performed on Tensor‑cores (fp16×fp16 → fp32) and
    the reduction stays in FP32.  Double‑buffering (`num_stages=4`) hides
    global‑memory latency.
6️⃣  Cast the result back to fp32, apply a second LayerNorm, multiply by
    the output gate and a final linear projection back to the original
    dimension.
7️⃣  Return the tensor in float32.

Only the O(N³) contraction is Triton‑accelerated; all other ops use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (A: [M, N, K], B: [M, K, N], C: [M, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                # *fp16  [M, N, K]
    B_ptr,                # *fp16  [M, K, N]
    C_ptr,                # *fp16  [M, N, N]
    M, N, K,              # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bk, stride_bj,   # B strides (M, k, j)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile index along i (first N dimension)
    pid_n = tl.program_id(1)   # tile index along j (second N dimension)
    pid_z = tl.program_id(2)   # batch * hidden index (M dimension)

    # ------------------------ tile offsets ---------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (high precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # total number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A tile: shape (BLOCK_M, BLOCK_K) -------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- load B tile (transposed) : shape (BLOCK_K, BLOCK_N) --
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # FP16×FP16 → FP32 accumulation on tensor‑cores
        acc = tl.dot(a, b, out_dtype=tl.float32, accumulate=acc)

    # -------------------------- store result -------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    using a custom Triton kernel.  `left` and `right` must be
    float16 tensors of shape [B, N, N, H].
    Returns a float16 tensor of the same shape.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension.
    # A: (M, i, k) ; B: (M, k, j)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Choose tile size based on sequence length.
    # For N > 512 a larger 128×128 tile yields higher arithmetic intensity.
    if N > 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 16
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Linear projections & gates (float32)
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_w)      # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_w)    # [B,N,N,hidden]

    # Optional mask (broadcast over hidden dim)
    if not nomask:
        mask_fp32 = mask_tensor.to(torch.float32).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp32
        right = right * mask_fp32

    # Gating (float32)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 3️⃣ Heavy O(N³) contraction via Triton (fp16 inputs/outputs)
    # -------------------------------------------------
    # Cast to fp16 only for the matrix multiply to save bandwidth.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = _batched_matmul_transpose_fp16(left_fp16, right_fp16)   # [B,N,N,hidden] fp16

    # Cast the contraction result back to float32 for the remaining ops.
    out = out_fp16.to(torch.float32)

    # -------------------------------------------------
    # 4️⃣ Second LayerNorm, output gating and final projection (float32)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] float32

    # -------------------------------------------------
    # Return result (float32)
    # -------------------------------------------------
    return out
