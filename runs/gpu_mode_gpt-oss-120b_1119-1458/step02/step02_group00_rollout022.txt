```python
@triton.jit
def proj_gate_transpose_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    numel = B * N * N * H
    mask = offs < numel

    # compute b,i,k,h using integer division
    tmp = offs // H
    h = offs - tmp * H
    tmp2 = tmp // N
    k = tmp % N
    b = tmp2 // N
    i = tmp2 % N

    # offsets
    # left and right original layout offset = ((b * N + i) * N + k) * H + h = offs itself
    # mask offset = (b * N + i) * N + k
    mask_off = (b * N + i) * N + k

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)

    l_out = left * left_gate * mask_val
    r_out = right * right_gate * mask_val

    # output offset: ((b * H + h) * N + i) * N + k
    out_off = ((b * H + h) * N + i) * N + k

    tl.store(left_out_ptr + out_off, l_out, mask=mask)
    tl.store(right_out_ptr + out_off, r_out, mask=mask)
```

def custom_kernel(data):
    # Unpack
    input_tensor, mask, weights, config = data
    # device
    device = input_tensor.device
    # config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps=1e-5
    # Layer norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_h = x.to(torch.float16)
    # cast weights to FP16
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)
    # Linear projections
    left_proj = F.linear(x_h, w_left_proj)      # [B,N,N,hidden_dim]
    right_proj = F.linear(x_h, w_right_proj)
    left_gate_raw = F.linear(x_h, w_left_gate)
    right_gate_raw = F.linear(x_h, w_right_gate)
    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))  # [B,N,N,hidden_dim]
    # mask
    if nomask:
        mask_fp16 = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
                               dtype=torch.float16,
                               device=device)
    else:
        mask_fp16 = mask.to(torch.float16)
    # Prepare output containers
    B, N, _, _ = input_tensor.shape
    left_out = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right_out = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    # Triton kernel
    total_elems = B * N * N * hidden_dim
    BLOCK = 65536
    grid = ((total_elems + BLOCK - 1) // BLOCK,)
    proj_gate_transpose_kernel[grid](
        left_proj, left_gate, right_proj, right_gate,
        mask_fp16,
        left_out, right_out,
        B, N, hidden_dim,
        BLOCK=BLOCK,
    )
    # free unnecessary tensors
    del left_proj, right_proj, left_gate_raw, right_gate_raw, left_gate, right_gate
    # Now perform batched GEMM
    total_heads = B * hidden_dim
    left_mat = left_out.view(total_heads, N, N)
    right_mat = right_out.view(total_heads, N, N)
    # bmm uses right_mat^T via transpose flag
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # post layer norm

    # multiply by out_gate

    # final projection back to dim
    final = F.linear(out, w_to_out)   # [B,N,N,dim] fp16
    return final.to(torch.float32)

"""
Optimized outgoing Triangle‑Multiplicative‑Update (TriMul) forward pass.

Key ideas
---------
* All heavy linear work is performed in FP16.
* A single Triton kernel (`proj_gate_transpose_kernel`) fuses
  - the element‑wise mask,
  - the left/right gating (including the sigmoid already computed in PyTorch),
  - the permutation from (B,N,N,H) → (B,H,N,N) (contiguous layout required by GEMM).

  This eliminates the two costly `permute(...).contiguous()` copies that the
  reference implementation performs.
* The dominant N³ contraction is carried out with a single batched GEMM
  (`torch.bmm`).  The left/right tensors are already in the optimal layout,
  so no extra transposition is required.
* The remaining element‑wise operations (LayerNorm, out‑gate multiplication,
  final projection) stay in PyTorch – they are negligible compared to the
  N³ work.

The whole forward pass therefore consists of:
    1. Input LayerNorm (FP32 → FP16)
    2. Four linear projections (left/right + gates)
    3. Sigmoid on the gates
    4. Triton kernel that applies mask, multiplies with gates and
       writes the transposed tensors `left_out`/`right_out` (B,H,N,N)
    5. Batched GEMM `out = left_out @ right_outᵀ`
    6. Output LayerNorm + out‑gate multiplication
    7. Final linear projection back to the original channel size
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gates and transpose
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_transpose_kernel(
    left_ptr,          # fp16  [B,N,N,H]  (flattened)
    left_gate_ptr,     # fp16  same shape as left_ptr
    right_ptr,         # fp16  [B,N,N,H]  (flattened)
    right_gate_ptr,    # fp16  same shape as right_ptr
    mask_ptr,          # fp16  [B,N,N]    (flattened)
    left_out_ptr,      # fp16  [B,H,N,N]  (flattened)
    right_out_ptr,     # fp16  same shape as left_out_ptr
    B, N, H,           # runtime ints
    BLOCK: tl.constexpr,
):
    """\
    For every element (b,i,k,h) compute

        left_out[b,h,i,k]  = left[b,i,k,h]  * left_gate[b,i,k,h]  * mask[b,i,k]
        right_out[b,h,i,k] = right[b,i,k,h] * right_gate[b,i,k,h] * mask[b,i,k]

    The result tensor layout (B,H,N,N) is contiguous, i.e. suitable for a
    batched GEMM without an extra permutation.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear index
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index ----
    # offs = ((b * N + i) * N + k) * H + h
    tmp = offs // H
    h   = offs - tmp * H                                  # h = offs % H
    tmp2 = tmp // N
    k   = tmp - tmp2 * N                                  # k = tmp % N
    b   = tmp2 // N
    i   = tmp2 - b * N                                    # i = tmp2 % N

    # ------ mask offset (B,N,N) -------
    mask_off = (b * N + i) * N + k

    # ------ load values -------------
    l   = tl.load(left_ptr       + offs, mask=mask, other=0.0)
    lg  = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    r   = tl.load(right_ptr      + offs, mask=mask, other=0.0)
    rg  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr       + mask_off, mask=mask, other=1.0)

    # ------ apply mask & gates -------
    l_out = l * lg * m
    r_out = r * rg * m

    # ------ write to (B,H,N,N) layout -------
    # out_off = ((b * H + h) * N + i) * N + k
    out_off = ((b * H + h) * N + i) * N + k
    tl.store(left_out_ptr  + out_off, l_out, mask=mask)
    tl.store(right_out_ptr + out_off, r_out, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no backward).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask         : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   contains 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)      # True → ignore mask
    eps         = 1e-5
    device      = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → FP16
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B,N,N,dim]  fp32
    x = x.to(torch.float16)            # fp16 for the heavy work

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (four weight matrices)
    # --------------------------------------------------------------
    w_left_proj  = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate  = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate   = weights["out_gate.weight"].to(torch.float16)

    left_proj   = F.linear(x, w_left_proj)        # [B,N,N,hidden_dim]
    right_proj  = F.linear(x, w_right_proj)
    left_gate   = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate  = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate    = torch.sigmoid(F.linear(x, w_out_gate))   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Mask tensor (FP16) – if no mask use all‑ones
    # --------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask, gates, and permutation with a Triton kernel
    # --------------------------------------------------------------
    left_out  = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right_out = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)

    total_elems = B * N * N * hidden_dim
    BLOCK = 65536                         # ~64K elements per program (tuned for H100)
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    proj_gate_transpose_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask_fp16,
        left_out, right_out,
        B, N, hidden_dim,
        BLOCK=BLOCK,
    )
    # Free intermediate tensors early
    del left_proj, right_proj, left_gate, right_gate, mask_fp16

    # --------------------------------------------------------------
    # 5️⃣ Core N³ contraction via a batched GEMM (L @ Rᵀ)
    # --------------------------------------------------------------
    total_heads = B * hidden_dim
    left_mat  = left_out.view(total_heads, N, N)            # [B*H, N, N]
    right_mat = right_out.view(total_heads, N, N)           # same layout

    # torch.bmm automatically treats the second argument as transposed when we call .transpose
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # Restore the original 4‑D layout
    out = (
        out_mat.view(B, hidden_dim, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)                      # [B, N, N, H]
        .contiguous()
    )                                             # fp16

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing LayerNorm (FP16)
    # --------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )          # fp16

    # --------------------------------------------------------------
    # 7️⃣ Multiply by the output gate
    # --------------------------------------------------------------
    out = out * out_gate               # element‑wise, cheap

    # --------------------------------------------------------------
    # 8️⃣ Final projection back to the original dimension
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    final = F.linear(out, w_to_out)          # [B,N,N,dim]  fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to float32 as required by the interface
    # --------------------------------------------------------------
    return final.to(torch.float32)
