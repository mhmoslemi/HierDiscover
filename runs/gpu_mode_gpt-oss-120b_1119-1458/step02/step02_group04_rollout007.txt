"""
Outgoing TriMul (AlphaFold‑3) – fast Triton implementation of the N³ contraction.

Algorithm
---------
1. Layer‑norm the input (float32 for numerical stability).
2. Cast to float16 and apply two linear projections (left/right) without bias.
3. Compute sigmoid gates (left_gate / right_gate) from the same input.
4. Apply optional mask, the linear‑gate product and store the gated tensors.
5. Perform the heavy N³ contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (B·H independent (N×N) matrix multiplications).
   A custom Triton kernel fuses the entire batch of GEMMs into a single
   launch, eliminating the per‑batch kernel‑launch overhead of cuBLAS.
6. Layer‑norm over the hidden dimension, an output gate and a final linear
   projection back to the original channel dimension.
7. Return the result as float32.

The kernel works on FP16 data (Tensor‑Core friendly) and accumulates in
FP32.  Only the contraction is on Triton; the surrounding linear layers,
layer‑norms and sigmoids use the highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_out_kernel(
    # pointers
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    out_ptr,                # fp16 *
    # problem size / meta
    B, N, H,                # int32 (batch, seq_len, hidden_dim)
    # strides of the tensors (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for every (batch, head) a matrix product:
        C = A @ Bᵀ   with   A[i,k] = left[b,i,k,h]   and   B[j,k] = right[b,j,k,h]
    The total number of independent GEMMs is total_batches = B * H.
    Grid layout:
        program_id(0) → batch*head index   (0 … B·H‑1)
        program_id(1) → block row (i)
        program_id(2) → block column (j)
    """
    pid_batch = tl.program_id(0)          # 0 … B·H‑1
    pid_i     = tl.program_id(1)          # block row
    pid_j     = tl.program_id(2)          # block column

    # ----- decode batch / head indices -----
    batch_idx = pid_batch // H               # 0 … B‑1
    head_idx  = pid_batch %  H               # 0 … H‑1

    # ----- compute absolute offsets for the current tile -----
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over the reduction dimension k -----
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)   A[i,k]
        a_ptrs = left_ptr + \
                 batch_idx * stride_left_b + \
                 head_idx  * stride_left_h + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                # fp16

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)   Bᵀ[k,j]
        b_ptrs = right_ptr + \
                 batch_idx * stride_right_b + \
                 head_idx   * stride_right_h + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                # fp16

        # accumulate:   acc[i,j] += Σ_k a[i,k] * b[k,j]
        acc += tl.dot(a, b)                  # fp16·fp16 → fp32 accumulation

    # ----- store the result (C) -----
    c_ptrs = out_ptr + \
             batch_idx * stride_out_b + \
             head_idx   * stride_out_h + \
             offs_i[:, None] * stride_out_i + \
             offs_j[None, :] * stride_out_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# Entry point required by the evaluation harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict of all model parameters (torch.Tensors)
        - config_dict  : contains ``dim`` and ``hidden_dim`` (int)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # 1️⃣ unpack
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]   # hidden_dim = hidden

    # -----------------------------------------------------------------
    # 2️⃣ helper – move a weight to fp16 on the target device
    # -----------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16, non_blocking=True)

    # -----------------------------------------------------------------
    # 3️⃣ load weights (half‑precision where possible)
    # -----------------------------------------------------------------
    norm_w            = weights["norm.weight"]
    norm_b            = weights["norm.bias"]
    left_proj_w       = to_half(weights["left_proj.weight"])
    right_proj_w      = to_half(weights["right_proj.weight"])
    left_gate_w       = to_half(weights["left_gate.weight"])
    right_gate_w      = to_half(weights["right_gate.weight"])
    out_gate_w        = to_half(weights["out_gate.weight"])
    to_out_norm_w     = to_half(weights["to_out_norm.weight"])
    to_out_norm_b     = to_half(weights["to_out_norm.bias"])
    to_out_w          = to_half(weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 4️⃣ Input LayerNorm (float32 → better numerical stability)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the heavy‑weight part of the model
    # -----------------------------------------------------------------
    x_half = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 6️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x_half, left_proj_w)    # [B, N, N, hidden] (fp16)
    right = F.linear(x_half, right_proj_w)   # [B, N, N, hidden] (fp16)

    # -----------------------------------------------------------------
    # 7️⃣ Optional mask (broadcast over hidden)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 8️⃣ Gating (sigmoid of a linear projection)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_half, left_gate_w))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))  # [B, N, N, hidden]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 9️⃣ N³ contraction – custom Triton batched GEMM
    # -----------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty_like(left)   # [B, N, N, hidden] (fp16)

    total_batches = B * hidden

    # strides (in elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()   # (B, N, N, H)
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_out_kernel[grid](
        left_ptr   = left.data_ptr(),
        right_ptr  = right.data_ptr(),
        out_ptr    = out.data_ptr(),
        B, N, hidden,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M = BLOCK_M,
        BLOCK_N = BLOCK_N,
        BLOCK_K = BLOCK_K,
        num_warps = 4,          # good default for fp16 on H100
    )

    # -----------------------------------------------------------------
    # 10️⃣ Output LayerNorm over the hidden dimension (float32 accumulator)
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)  # [B,N,N,hidden] fp32

    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # -----------------------------------------------------------------
    # 11️⃣ Output gate + final projection back to dim
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))  # [B,N,N,hidden] fp16
    gated = out_norm * out_gate                               # element‑wise gate

    final = F.linear(gated, to_out_w)                         # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 12️⃣ Cast to float32 as required by the API and return
    # -----------------------------------------------------------------
    return final.to(torch.float32)
