"""
Optimised outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑Norm the input (FP32).
2. Cast to FP16 and fuse the five linear projections (left, right,
   left‑gate, right‑gate, out‑gate) into a single GEMM (dim → 5·hidden_dim).
3. Split the fused result, apply sigmoid to the three gates and multiply
   the gated projections.  If a mask is present, apply it (broadcasted).
4. Reshape `left` and `right` to (B·H, N, N) and compute the expensive
   N³ reduction: for each head `d`,
        out[b·H+i,j] = Σₖ left[b·H,i,k] * right[b·H,j,k] .
   For modest head counts (≤32) a custom Triton kernel is used; otherwise
   we fall back to `torch.bmm` which maps to cuBLAS Tensor‑Core GEMM.
5. Cast the result to FP32, apply a second Layer‑Norm, the output‑gate,
   and the final linear projection back to `dim`.
6. Return a float‑32 tensor of shape [B, N, N, dim].

Both the fused linear projection and the N³ reduction are the
performance‑critical parts and are implemented with FP16 arithmetic,
while accumulation is done in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_am, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ  (A,B ∈ FP16, C ∈ FP16, accum in FP32)

    A: (batch, N, N)   – stride_am = batch‑stride, stride_ak = row‑stride, col‑stride = 1
    B: (batch, N, N)   – stride_bm = batch‑stride, stride_bk = row‑stride (k dimension), col‑stride = 1
    C: (batch, N, N)   – stride_cm = batch‑stride, stride_cn = row‑stride, col‑stride = 1
    """
    pid_batch = tl.program_id(0)   # batch * hidden_dim index
    pid_i = tl.program_id(1)       # tile row
    pid_j = tl.program_id(2)       # tile col

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (the inner sequence dimension)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_batch * stride_am
            + offs_i[:, None] * stride_ak
            + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ block: shape (BLOCK_K, BLOCK_N) – we load B as (k, j)
        b = tl.load(
            b_ptr
            + pid_batch * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # FP16 dot, accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Write back C (FP16)
    tl.store(
        c_ptr
        + pid_batch * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 and fuse the five linear projections
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] – FP16

    # Concatenate the five weight matrices (no bias)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # shape (5*hidden_dim, dim)

    # Single GEMM: (B*N*N, dim) @ (dim, 5*hidden_dim).T → (B,N,N,5*hidden_dim)
    fused = F.linear(x_h, w_fused)  # [B, N, N, 5*H] – FP16

    # ------------------------------------------------------------------
    # 3. Split fused tensor
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj = fused[..., 0:H]               # (B,N,N,H)
    right_proj = fused[..., H:2 * H]
    left_gate = fused[..., 2 * H:3 * H].sigmoid()
    right_gate = fused[..., 3 * H:4 * H].sigmoid()
    out_gate = fused[..., 4 * H :].sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Apply gates (still FP16) and optional mask
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        # mask shape: (B,N,N) → broadcast on hidden dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Prepare for batched N³ reduction
    #    reshape to (B*H, N, N) – contiguous layout
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)    # (BH, N, N) – FP16
    right_flat = right.view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 6. Compute batched Gram product: out_flat = left_flat @ right_flatᵀ
    # ------------------------------------------------------------------
    BH = B * hidden_dim
    use_triton = (BH <= 32) and (N <= 512)   # heuristic: Triton shines for small head counts

    if use_triton:
        out_flat = torch.empty_like(left_flat)  # FP16 output

        # Strides (in elements) for the three tensors
        stride_am, stride_ak, _ = left_flat.stride()
        stride_bm, stride_bk, stride_bn = right_flat.stride()
        stride_cm, stride_cn, stride_ck = out_flat.stride()

        # Tuned tiling parameters for H100 Tensor Cores
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (BH, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

        _tri_mul_gemm_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            stride_am,
            stride_ak,
            stride_bm,
            stride_bn,   # not used inside kernel (col‑stride = 1)
            stride_bk,
            stride_cm,
            stride_cn,
            stride_ck,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # Use cuBLAS batched GEMM (Tensor‑Cores) – fast for larger head counts
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H) and cast to FP32 for the remaining ops
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .float()
    )  # [B, N, N, H] – FP32

    # ------------------------------------------------------------------
    # 8. Output LayerNorm + output gating (still FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].float())

    return out
```

**Explanation of key optimisations**

* **Fused projection** – Instead of five separate `torch.nn.functional.linear` calls we fuse the four
  projections (`left`, `right`, `left_gate`, `right_gate`, `out_gate`) into a single GEMM.
  This reduces memory traffic and lets the GPU spend far fewer cycles on launching kernels.

* **FP16 arithmetic** – All heavy linear work (projections, gating, batched GEMM) runs in half precision,
  while the accumulators are kept in FP32 for numerical stability.

* **Custom Triton kernel** – For the cubic‑cost reduction we provide a specialised blocked GEMM that
  computes `C = A @ Bᵀ` with `A,B` in FP16 and accumulation in FP32.  A simple heuristic picks the
  kernel when the number of heads (`B·hidden_dim`) is small; otherwise we fall back to cuBLAS
  (`torch.bmm`) which is already highly optimised on H100.

* **In‑place casting & memory layout** – By permuting to `(B, H, N, N)` and using `.contiguous()`
  we obtain a stride‑1 inner dimension, which the Triton kernel can load efficiently.

* **Minimal data copies** – Apart from the necessary reshapes for the GEMM, the implementation never
  materialises large intermediate tensors in float32, keeping the memory footprint low.

The function returns a float‑32 tensor with the expected shape `[batch, seq_len, seq_len, dim]`.
