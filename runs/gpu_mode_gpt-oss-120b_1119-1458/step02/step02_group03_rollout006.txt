@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,   # shape: B,N,N,H, but we flatten batch*H, but we may pass M = B*H
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
   pid_b = tl.program_id(0)  # batch dimension over M = B*H
   pid_m = tl.program_id(1)  # block row
   pid_n = tl.program_id(2)  # block col

   # Compute starting indices
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # Mask if beyond N
   mask_m = offs_m < N
   mask_n = offs_n < N

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Compute batch offset
   batch_off = pid_b * stride_l_batch
   # Offsets for left and right pointers
   # We will compute offsets across K dimension in loop
   for k in range(0, N, BLOCK_K):
       cur_k = k + tl.arange(0, BLOCK_K)
       mask_k = cur_k < N

       # left tile: l[batch,b,i,k]
       # left_ptr offsets = batch_off + offs_m[:,None]*stride_l_i + cur_k[None,:]*stride_l_k
       # Since left shape (M,N,N) after flatten, stride_l_i = N (row stride = K?), stride_l_k = 1
       # confirm: left_stride0 = N*N, but we treat left after flatten: shape (M,N,N). So stride batch = N*N, stride row = N, stride col=1.
       # So:
       L = tl.load(
           left_ptr + batch_off + offs_m[:,None]*stride_l_i + cur_k[None,:]*stride_l_k,
           mask=mask_m[:,None] & mask_k[None,:],
           other=0.0
       )
       # right tile: shape (BLOCK_N, BLOCK_K)
       R = tl.load(
           right_ptr + batch_off + offs_n[:,None]*stride_r_i + cur_k[None,:]*stride_r_k,
           mask=mask_n[:,None] & mask_k[None,:],
           other=0.0
       )
       # Compute transposed of R
       R_T = tl.trans(R)  # shape (BLOCK_K, BLOCK_N)
       # Multiply accumulate
       acc += tl.dot(L, R_T)   # L shape (BLOCK_M,BLOCK_K), R_T shape (BLOCK_K,BLOCK_N)
   # Store
   out = acc.to(tl.float16)
   # Store only where mask_m & mask_n
   store_ptr = out_ptr + pid_b * stride_o_batch + offs_m[:,None]*stride_o_i + offs_n[None,:]*stride_o_j
   tl.store(store_ptr, out, mask=mask_m[:,None] & mask_n[None,:])

def tri_mul_triton(left: torch.Tensor, right: torch.Tensor, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    # left, right: shape [B,N,N,H]
    B, N, _, H = left.shape
    # permute
    left_t = left.permute(0,3,1,2).contiguous()
    right_t = right.permute(0,3,1,2).contiguous()
    M = B * H
    left_flat = left_t.view(M, N, N)
    right_flat = right_t.view(M, N, N)
    out = torch.empty_like(left_flat)

    # strides
    stride_l_batch, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_i, stride_r_k = right_flat.stride()
    stride_o_batch, stride_o_i, stride_o_j = out.stride()

    grid = (M, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out,
        M,
        N,
        N,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_i,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (FP32 → keep high precision).
2. Cast to FP16 and compute the four linear projections (left, right and three gates)
   using the provided weight tensors (all bias‑free).
3. Apply an optional pair‑wise mask (broadcast on the hidden dimension) to `left`
   and `right`.  Gating (sigmoid) is then applied element‑wise.
4. The expensive contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM using a custom Triton kernel.  The hidden
   dimension `h` is merged into the batch axis (M = B·H) so each head is a
   separate matrix multiplication.  The kernel executes a tiled FP16 matmul
   with a float32 accumulator (tensor‑core friendly) and writes the result
   in FP16.
5. Apply a second Layer‑norm over the hidden dimension, multiply by the output
   gate, and linearly project back to the original channel dimension.
6. Return the result as FP32.

The Triton kernel is the only non‑trivial part written in Triton, satisfying
the “must use Triton” rule while the surrounding bookkeeping is performed with
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM used in TriMul.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,            # fp16*, shape (M, N, K)  where K == N
    right_ptr,           # fp16*, shape (M, N, K)
    out_ptr,             # fp16*, shape (M, N, N)
    M, N, K,             # int64 scalars
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication:
    out[m, i, j] = sum_k left[m, i, k] * right[m, j, k]   (right is used transposed)
    """
    pid_batch = tl.program_id(0)   # batch index (M = B*H)
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block column index

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current matrix (batch slice)
    batch_off_l = pid_batch * stride_l_batch
    batch_off_r = pid_batch * stride_r_batch
    batch_off_o = pid_batch * stride_o_batch

    # Accumulator in float32 for higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles of size BLOCK_K
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile of `left`: shape (BLOCK_M, BLOCK_K)
        left_tile = tl.load(
            left_ptr
            + batch_off_l
            + offs_m[:, None] * stride_l_i
            + cur_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Load a tile of `right` (non‑transposed): shape (BLOCK_N, BLOCK_K)
        right_tile = tl.load(
            right_ptr
            + batch_off_r
            + offs_n[:, None] * stride_r_i
            + cur_k[None, :] * stride_r_k,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Transpose the right tile so we obtain (BLOCK_K, BLOCK_N)
        right_tile_T = tl.trans(right_tile)

        # Accumulate the dot product (fp16 × fp16 → fp32 accumulator)
        acc += tl.dot(left_tile, right_tile_T)

    # Write back the result in fp16
    out_tile = acc.to(tl.float16)
    out_ptrs = (
        out_ptr
        + batch_off_o
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j
    )
    tl.store(out_ptrs, out_tile, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel and restores the original tensor layout.
# ----------------------------------------------------------------------
def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                    BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Perform the batched contraction used in TriMul using the custom Triton kernel.

    Parameters
    ----------
    left, right : torch.Tensor
        Tensors of shape [B, N, N, H] and dtype torch.float16.
    BLOCK_M, BLOCK_N, BLOCK_K : int
        Tiling parameters for the Triton kernel.

    Returns
    -------
    torch.Tensor of shape [B, N, N, H] (dtype torch.float16).
    """
    B, N, _, H = left.shape
    device = left.device

    # Re‑order to [B, H, N, N] and flatten the first two dimensions → (M, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    M = B * H

    left_f = left_t.view(M, N, N)
    right_f = right_t.view(M, N, N)
    out_f = torch.empty_like(left_f)  # (M, N, N), fp16

    # Strides for the flattened tensors (these are Python ints)
    stride_l_batch, stride_l_i, stride_l_k = left_f.stride()
    stride_r_batch, stride_r_i, stride_r_k = right_f.stride()
    stride_o_batch, stride_o_i, stride_o_j = out_f.stride()

    # Launch a 3‑D grid: (batch, block_row, block_col)
    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_f,
        right_f,
        out_f,
        M,
        N,
        N,                     # K == N
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_i,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_f.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Public entry point – the forward pass of the outgoing TriMul module.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)   or None
        - weights_dict : dictionary containing all model weights
        - config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------
    # Unpack inputs and configuration
    # -----------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------------
    # Helper to cast a weight tensor to FP16 on the target device
    # -------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # -------------------------------------------------------
    # Load model parameters (bias‑free Linear layers and LayerNorms)
    # -------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # -----------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # -----------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C] – still float32

    # Cast to FP16 for the heavy path
    x_h = x_norm.to(torch.float16)

    # -----------------------------
    # 2️⃣ Linear projections (bias‑free)
    # -----------------------------
    left  = F.linear(x_h, left_proj_w)    # [B, N, N, hidden_dim], fp16
    right = F.linear(x_h, right_proj_w)   # [B, N, N, hidden_dim], fp16

    # -----------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # -----------------------------
    if mask is not None:
        # mask may be bool or float; ensure FP16 and broadcast over hidden_dim
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------
    # 4️⃣ Gating (sigmoid of bias‑free linear on the normalized input)
    # -----------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, hidden_dim], fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # fp16

    # Apply the gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------
    # 5️⃣ N³ contraction – batched GEMM via Triton
    # -----------------------------
    # Result: [B, N, N, hidden_dim] (fp16)
    out = _tri_mul_triton(left, right,
                          BLOCK_M=128,
                          BLOCK_N=128,
                          BLOCK_K=32)

    # -----------------------------
    # 6️⃣ Output LayerNorm (FP32 ↔ FP16 conversion around it)
    # -----------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # -----------------------------
    # 7️⃣ Gated projection back to the original channel dimension
    # -----------------------------
    gated = out_norm * out_gate          # shape [B, N, N, hidden_dim], fp16
    final = F.linear(gated, to_out_w)    # shape [B, N, N, C], fp16

    # -----------------------------
    # 8️⃣ Return FP32 tensor as required by the spec
    # -----------------------------
    return final.to(torch.float32)
