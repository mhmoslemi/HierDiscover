"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Cast the input to FP16 and apply a half‑precision LayerNorm.
2. Two bias‑free linear projections produce *left* and *right* tensors
   of shape [B, N, N, H] (H = hidden_dim, FP16).
3. Compute three sigmoid‑gates (left, right, out) from the same normalized input.
4. Fuse the per‑pair mask (if present) and the left/right gates with a
   custom Triton kernel – this eliminates two Python‑level elementwise passes.
5. Perform the dense N³ contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   by reshaping to (B·H, N, N) and calling a single torch.bmm (highly‑optimised
   Tensor‑Core GEMM).  This is the dominant cost and runs at close to peak TFLOPs.
6. Apply a second half‑precision LayerNorm on the hidden dimension,
   multiply by the output gate and linearly project back to the original channel dim.
7. Return the result as float32.

Only the mask‑gate fusion is performed in Triton; the heavy GEMM is left to
cuBLAS/Tensor‑Core which is still faster than a hand‑written Triton tiling.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and left/right gates into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def fuse_gate_mask_kernel(
    left_ptr,                # ptr to left tensor   [B,N,N,H]  FP16
    right_ptr,               # ptr to right tensor  [B,N,N,H]  FP16
    left_gate_ptr,           # ptr to left gate    [B,N,N,H]  FP16
    right_gate_ptr,          # ptr to right gate   [B,N,N,H]  FP16
    mask_ptr,                # ptr to mask tensor  [B,N,N,1] FP16  (broadcast on H)
    total_elements,          # total number of elements B*N*N*H (int64)
    hidden_dim,              # H (int64)
    BLOCK_SIZE: tl.constexpr,   # # elements processed per program
    USE_MASK: tl.constexpr       # 1 ⇒ mask present, 0 ⇒ no mask
):
    """In‑place: left = left * left_gate * mask, right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)  # 1‑D offsets
    mask = offs < total_elements

    # ------------------------------------------------------------------
    # Load mask (broadcast across hidden_dim) if required.
    # ------------------------------------------------------------------
    if USE_MASK:
        # mask index = floor(off / H) because mask has no H dimension.
        mask_off = offs // hidden_dim
        m = tl.load(mask_ptr + mask_off, mask=mask).to(tl.float16)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Load source tensors and their gates.
    # ------------------------------------------------------------------
    a = tl.load(left_ptr + offs, mask=mask)          # left
    b = tl.load(right_ptr + offs, mask=mask)         # right
    ag = tl.load(left_gate_ptr + offs, mask=mask)    # left gate
    bg = tl.load(right_gate_ptr + offs, mask=mask)   # right gate

    # ------------------------------------------------------------------
    # Apply gates and (optional) mask.
    # ------------------------------------------------------------------
    a = a * ag * m
    b = b * bg * m

    # ------------------------------------------------------------------
    # Write back in‑place.
    # ------------------------------------------------------------------
    tl.store(left_ptr + offs, a, mask=mask)
    tl.store(right_ptr + offs, b, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
def tri_mul_batched_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``
    using a single torch.bmm over the (B·H) batch dimension.
    All tensors are FP16 and have shape [B, N, N, H].

    Returns
    -------
    out : torch.Tensor   [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape

    # Move hidden dimension into the batch axis: (B, H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch×head → (B·H, N, N)
    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
        - input_tensor : torch.Tensor[ B, N, N, C ]      (float32)
        - mask_tensor  : torch.Tensor[ B, N, N ]      (bool/float) or None
        - weights_dict : dict of model parameters:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict : must contain "dim" (C) and "hidden_dim" (H).

    Returns
    -------
    torch.Tensor[ B, N, N, C ]   (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]                # input channel dimension (C)
    hidden_dim = config["hidden_dim"]  # H

    # ------------------------------------------------------------------
    # Convert model weights to FP16 on the target device (no gradients needed)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    norm_w        = to_half(weights["norm.weight"])
    norm_b        = to_half(weights["norm.bias"])
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (FP16)
    # ------------------------------------------------------------------
    x_h = input_tensor.to(torch.float16)
    x_norm = F.layer_norm(
        x_h,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # shape [B,N,N,dim], dtype=FP16

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_norm, left_proj_w)   # [B,N,N,hidden_dim] FP16
    right = F.linear(x_norm, right_proj_w)  # [B,N,N,hidden_dim] FP16

    # ------------------------------------------------------------------
    # 3️⃣  Gating tensors (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B,N,N,hidden_dim] FP16
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))  # [B,N,N,hidden_dim] FP16
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))    # [B,N,N,hidden_dim] FP16

    # ------------------------------------------------------------------
    # 4️⃣  Optional pair‑wise mask – broadcast on the hidden dimension
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1] FP16
    else:
        # Dummy 1‑element tensor used only when USE_MASK = 0
        mask_h = torch.empty((1,), dtype=torch.float16, device=device)

    # Ensure tensors are contiguous for the Triton kernel
    left       = left.contiguous()
    right      = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_h     = mask_h.contiguous()

    # ------------------------------------------------------------------
    # 5️⃣  Fuse mask and gates with a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    total_elems = left.numel()                     # B*N*N*H
    BLOCK_SIZE = 1024                              # tunable
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    fuse_gate_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_h,
        total_elements=total_elems,
        hidden_dim=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        USE_MASK=1 if mask is not None else 0,
    )
    # After the kernel `left` and `right` already contain the gated+masked values.
    # We can free the gate tensors now (optional, helps with memory pressure).
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Dense N³ contraction via a batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = tri_mul_batched_fp16(left, right)   # [B,N,N,hidden_dim] FP16

    # ------------------------------------------------------------------
    # 7️⃣  Second LayerNorm on the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # [B,N,N,hidden_dim] FP16

    # ------------------------------------------------------------------
    # 8️⃣  Output gating and final linear projection back to dim=C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate          # elementwise FP16
    final = F.linear(gated, to_out_w)    # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣  Cast back to float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
```

This implementation follows the reference PyTorch logic while:

* keeping the heavy‑path computations in FP16 to exploit Tensor‑Core throughput,
* fusing the mask‑application and left/right gating in a hand‑written Triton kernel,
* using a single batched `torch.bmm` for the O(N³) contraction (the performance‑critical step),
* and finally returning a float‑32 tensor as required.
