    left_proj = left_concat[..., :hidden]
    left_gate_raw = left_concat[..., hidden:]
    right_proj = right_concat[..., :hidden]
    right_gate_raw = right_concat[..., hidden:]

        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # compute O(N^3) contraction
    out = _batched_matmul_transpose_fp16(left, right)  # shape [B,N,N,hidden]

    # layernorm on hidden dim
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16))

    out_gate_raw = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)) # [B,N,N,hidden]
    out_gate = torch.sigmoid(out_gate_raw)
    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

"""
Optimized “outgoing” TriMul implementation (AlphaFold‑3).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32 → better numerics).
2️⃣ Cast to fp16 and fuse the two linear projections + their
   sigmoid‑gates (left / right) into two batched GEMMs:
        [B,N,N,dim] @ [hidden,dim]ᵀ  →  [B,N,N,2·hidden]
   The first half are the projections, the second half are the
   gate pre‑activations. This halves the number of GEMMs.
3️⃣ (Optional) mask the projected tensors.
4️⃣ Apply the sigmoid gates and obtain ``left`` and ``right`` tensors
   of shape ``[B,N,N,hidden]`` (fp16).
5️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Realised as a batched matrix‑multiply ``A @ Bᵀ`` with a custom
   Triton kernel.  The kernel works on tiles of 128×128×64
   (i‑j‑k) using Tensor‑Core MMA, accumulates in FP32 and writes the
   result as fp16.
6️⃣ Second LayerNorm (hidden dim), sigmoid out‑gate, and final linear
   projection back to ``dim`` (all fp16).
7️⃣ Cast the result to float32 (the harness expects float32).

Only the O(N³) contraction is written in Triton; all other ops use
high‑performance PyTorch primitives.  Mixed‑precision (fp16/FP32)
keeps memory traffic low while preserving accuracy.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # *fp16  [M, N, K]
    B_ptr,                     # *fp16  [M, N, K]
    C_ptr,                     # *fp16  [M, N, N]
    M, N, K,                  # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each batch‑hidden slice.
    Accumulation is in FP32; the final result is stored as FP16.
    """
    pid_m = tl.program_id(0)   # tile index over the first N‑axis (i)
    pid_n = tl.program_id(1)   # tile index over the second N‑axis (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M‑dimension)

    # ----- tile offsets on the N‑axes -----
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32) for the tile (BLOCK_M x BLOCK_N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A tile : [BLOCK_M, BLOCK_K] ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---- load B tile (transposed) : [BLOCK_K, BLOCK_N] ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # MMA (fp16 * fp16 → fp32) and accumulate
        acc += tl.dot(a, b)          # dot returns FP32

    # ----- write the tile back -----
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  fp16
    right: torch.Tensor,  # [B, N, N, H]  fp16
) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
    All tensors must be contiguous fp16 on the same CUDA device.
    Returns a fp16 tensor of identical shape.
    """
    B, N, _, H = left.shape

    # Merge batch and hidden dimensions → [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – 128×128 works well for N≥512; fall back to 64×64 otherwise
    if N >= 512:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 64
        BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,    # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,    # j‑tiles
        B * H,                           # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → stable)
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, dim]  float32
    x = x_norm.to(torch.float16)       # cast for the rest of the pipeline

    # -------------------------------------------------
    # 2️⃣ Fuse projection + gate for left and right sides
    # -------------------------------------------------
    # concatenate projection & gate weights (out_features stacked)
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)   # [2*H, dim]

    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # [2*H, dim]

    # single GEMM for each side
    left_cat = torch.nn.functional.linear(x, left_cat_w)    # [B, N, N, 2*H]  fp16
    right_cat = torch.nn.functional.linear(x, right_cat_w)  # [B, N, N, 2*H]  fp16

    # split projection / gate tensors
    left_proj = left_cat[..., :hidden_dim]                 # [B,N,N,H]
    left_gate_pre = left_cat[..., hidden_dim:]             # [B,N,N,H]
    right_proj = right_cat[..., :hidden_dim]
    right_gate_pre = right_cat[..., hidden_dim:]

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Apply sigmoid gates (all fp16)
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------
    # 5️⃣ O(N³) contraction via Triton (fp16 → fp16)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden dim) – fp16
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # -------------------------------------------------
    # 7️⃣ Out‑gate + final projection (fp16)
    # -------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16))
    out_gate = torch.sigmoid(out_gate_pre)               # [B,N,N,H] fp16
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 (required output dtype)
    # -------------------------------------------------
    return out.to(torch.float32)
