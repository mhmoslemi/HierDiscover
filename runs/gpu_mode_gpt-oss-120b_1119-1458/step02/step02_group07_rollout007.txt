"""
TriMul (outgoing) – fast Triton‑augmented forward pass

The operator works on a 4‑D tensor  [B, N, N, C] and performs:

    out[b,i,j,:] = Σₖ  ( (x·W_left)[:,i,k,:] ⊙ σ(x·W_left_gate)[:,i,k,:] ⊙ mask )
                  * ( (x·W_right)[:,j,k,:] ⊙ σ(x·W_right_gate)[:,j,k,:] ⊙ mask )

where “⊙” denotes element‑wise multiplication and the sum runs over the
sequence axis *k*.  The implementation

* uses FP16 for all heavy linear work,
* fuses “mask × gate” with a tiny Triton kernel (only when a mask is present),
* replaces the costly “permute‑contiguous‑view” of the reference code with a
  single `torch.einsum` that directly computes the N³ contraction without any
  extra copies,
* applies the second LayerNorm, the output‑gate, and the final projection in
  FP32 to preserve numerical stability.

The only Triton kernel is the mask‑gate fusion; everything else stays in PyTorch
so that we keep the code simple while still reaching sub‑millisecond latency on
an H100 for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                      # fp16 * left projection   [B,N,N,H]
    right_ptr,                     # fp16 * right projection  [B,N,N,H]
    mask_ptr,                      # fp16 * mask              [B,N,N]
    left_gate_ptr,                 # fp16 * left gate         [B,N,N,H]
    right_gate_ptr,                # fp16 * right gate        [B,N,N,H]
    # strides for left/right (b,i,j,h)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    # strides for gates (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,                       # batch, seq_len, hidden_dim
    BLOCK_H: tl.constexpr,        # tile size along hidden dimension
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # load scalar mask (broadcasted over hidden dim)
    mask_val = tl.load(mask_ptr + off_m)

    # loop over hidden dimension in chunks of BLOCK_H
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        # load vectors of length BLOCK_H
        left      = tl.load(left_ptr      + off_l + offs * stride_l3,
                            mask=mask_h, other=tl.float16(0.0))
        right     = tl.load(right_ptr     + off_r + offs * stride_r3,
                            mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate= tl.load(right_gate_ptr+ off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))

        # fused operation: proj * gate * mask
        left_out  = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        # store back in‑place
        tl.store(left_ptr  + off_l + offs * stride_l3, left_out,  mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)


# ----------------------------------------------------------------------
# Entry point used by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) – may be
                         ignored when `config["nomask"]` is True.
        - weights      : dict of model parameters (all float32)
        - config       : dict with keys
                         * "dim"        – input feature dimension C
                         * "hidden_dim" – hidden dimension used inside the module
                         * "nomask"     – bool, True → no mask is applied

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – cast weights once
    # ------------------------------------------------------------------
    left_w  = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_w = weights["right_proj.weight"].to(torch.float16)  # (H, C)

    left  = F.linear(x_fp16, left_w)   # [B, N, N, H]  fp16
    right = F.linear(x_fp16, right_w)  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # [B,N,N,H] fp16
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask and gates
    # ------------------------------------------------------------------
    if nomask:
        # Simple element‑wise fusion – no Triton launch
        left = left * left_gate
        right = right * right_gate
    else:
        # Triton kernel fuses mask, left_gate, right_gate in‑place
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]  # B * N * N
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left,
            right,
            mask_fp16,
            left_gate,
            right_gate,
            # strides for left/right
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gates (same layout as left/right)
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # sizes
            input_tensor.shape[0],               # B
            input_tensor.shape[1],               # N
            hidden_dim,                          # H
            BLOCK_H=32,                          # tile size along hidden dim
            num_warps=4,
        )
        # after the kernel `left` and `right` already contain mask·gate·proj

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ contraction via a single einsum (no extra copies)
    # ------------------------------------------------------------------
    # out shape: [B, N, N, H]  (still fp16)
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]  float32

    # Apply the output gate (converted to float32)
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]                     # shape (dim, H)
    out_final = F.linear(out_norm, to_out_w)                # [B, N, N, dim] float32

    return out_final
