def fused_proj_gate_kernel(
    a_ptr, left_proj_w_ptr, left_gate_w_ptr, right_proj_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    stride_am, stride_ak,
    stride_lp_wk, stride_lp_wh,
    stride_lg_wk, stride_lg_wh,
    stride_rp_wk, stride_rp_wh,
    stride_rg_wk, stride_rg_wh,
    stride_og_wk, stride_og_wh,
    stride_m0,
    stride_lo_m, stride_lo_h,
    stride_ro_m, stride_ro_h,
    stride_og_m, stride_og_h,
    M, K, H,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # Initialize accumulators
    left_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)
    left_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)
    right_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)
    right_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)
    out_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)

    # Load mask values (scalar per row)
    mask_val = tl.where(mask_m,
                        tl.load(mask_ptr + offs_m * stride_m0, mask=mask_m, other=tl.float16(0.0)),
                        tl.float16(0.0))

    for k in range(0, K, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        k_mask = k_off < K

        # Load A tile
        a = tl.load(a_ptr + (offs_m[:, None] * stride_am + k_off[None, :] * stride_ak),
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=tl.float16(0.0))

        # Load weight tiles
        lp = tl.load(left_proj_w_ptr + (k_off[:, None] * stride_lp_wk + offs_h[None, :] * stride_lp_wh),
                     mask=k_mask[:, None] & mask_h[None, :],
                     other=tl.float16(0.0))
        lg = tl.load(left_gate_w_ptr + (k_off[:, None] * stride_lg_wk + offs_h[None, :] * stride_lg_wh),
                     mask=k_mask[:, None] & mask_h[None, :],
                     other=tl.float16(0.0))
        rp = tl.load(right_proj_w_ptr + (k_off[:, None] * stride_rp_wk + offs_h[None, :] * stride_rp_wh),
                     mask=k_mask[:, None] & mask_h[None, :],
                     other=tl.float16(0.0))
        rg = tl.load(right_gate_w_ptr + (k_off[:, None] * stride_rg_wk + offs_h[None, :] * stride_rg_wh),
                     mask=k_mask[:, None] & mask_h[None, :],
                     other=tl.float16(0.0))
        og = tl.load(out_gate_w_ptr + (k_off[:, None] * stride_og_wk + offs_h[None, :] * stride_og_wh),
                     mask=k_mask[:, None] & mask_h[None, :],
                     other=tl.float16(0.0))

        # Accumulate dot products
        left_proj_acc += tl.dot(a, lp)
        left_gate_acc += tl.dot(a, lg)
        right_proj_acc += tl.dot(a, rp)
        right_gate_acc += tl.dot(a, rg)
        out_gate_acc += tl.dot(a, og)

    # Apply sigmoid on gates
    one = tl.float16(1.0)
    left_gate_sig = one / (one + tl.exp(-left_gate_acc))
    right_gate_sig = one / (one + tl.exp(-right_gate_acc))
    out_gate_sig = one / (one + tl.exp(-out_gate_acc))

    # Apply mask to left/right after gating
    mask_tile = mask_val[:, None]  # (BLOCK_M,1)
    left_out = left_proj_acc * left_gate_sig * mask_tile
    right_out = right_proj_acc * right_gate_sig * mask_tile

    # Store results
    store_mask = mask_m[:, None] & mask_h[None, :]
    tl.store(left_out_ptr + offs_m[:, None] * stride_lo_m + offs_h[None, :] * stride_lo_h,
             left_out, mask=store_mask)
    tl.store(right_out_ptr + offs_m[:, None] * stride_ro_m + offs_h[None, :] * stride_ro_h,
             right_out, mask=store_mask)
    tl.store(out_gate_out_ptr + offs_m[:, None] * stride_og_m + offs_h[None, :] * stride_og_h,
             out_gate_sig, mask=store_mask)

"""
Optimised outgoing Triangle Multiplicative (TriMul) update.

Algorithm
---------
1️⃣ Layer‑norm the input tensor (float32).
2️⃣ Cast to fp16 and launch a fused Triton kernel that:
   • computes the four linear projections (left/right + their gates) and the
     out‑gate in a single mat‑mul pass,
   • fuses the mask (or a dummy all‑ones mask) and the sigmoid gating.
   This replaces 5 separate torch.nn.functional.linear calls.
3️⃣ Reshape the projected tensors to (B·H, N, N) and perform a batched GEMM:
        C_h = left_h @ right_hᵀ
   where H = hidden_dim.  This is the N³ core of the TriMul.
4️⃣ Apply a second LayerNorm, multiply by the out‑gate and a final linear
   projection back to the original feature dimension.
All heavy arithmetic is in fp16 (Tensor‑Core accelerated); the final
output is float32.

The kernel is deliberately kept simple with three static tile sizes:
  BLOCK_M – rows (flattened B·N·N) per program,
  BLOCK_H – hidden‑dim columns per program,
  BLOCK_K – reduction over the input feature dimension (dim).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel: fused projection, gating, mask and out‑gate computation.
# ----------------------------------------------------------------------

def _tri_mul_fused_proj_gate_kernel(
    a_ptr,                # fp16 * input matrix A (M,K)  – flattened input
    lp_w_ptr,             # fp16 * left_proj weight  (K,H)
    lg_w_ptr,             # fp16 * left_gate weight  (K,H)
    rp_w_ptr,             # fp16 * right_proj weight (K,H)
    rg_w_ptr,             # fp16 * right_gate weight (K,H)
    og_w_ptr,             # fp16 * out_gate weight   (K,H)
    mask_ptr,             # fp16 * mask vector (M,)
    lo_ptr,               # fp16 * left   output (M,H)
    ro_ptr,               # fp16 * right  output (M,H)
    og_ptr,               # fp16 * out_gate output (M,H)

    # strides for A
    stride_am, stride_ak,
    # strides for all weight matrices (K,H)
    stride_lp_wk, stride_lp_wh,
    stride_lg_wk, stride_lg_wh,
    stride_rp_wk, stride_rp_wh,
    stride_rg_wk, stride_rg_wh,
    stride_og_wk, stride_og_wh,
    # stride for mask (M,)
    stride_m0,
    # strides for outputs (M,H)
    stride_lo_m, stride_lo_h,
    stride_ro_m, stride_ro_h,
    stride_og_m, stride_og_h,

    M, K, H,                      # problem sizes
    BLOCK_M: tl.constexpr,        # rows per program
    BLOCK_H: tl.constexpr,        # hidden‑dim tiles
    BLOCK_K: tl.constexpr,        # reduction tile
):
    pid_m = tl.program_id(0)               # row‑tile id
    pid_h = tl.program_id(1)               # hidden‑tile id

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    m_valid = offs_m < M
    h_valid = offs_h < H

    # ------------------------------------------------------------------
    # Accumulators for the five mat‑mul results
    # ------------------------------------------------------------------
    left_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)
    left_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)
    right_proj_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)
    right_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)
    out_gate_acc = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float16)

    # Load mask values for the rows (scalar per row)
    mask_val = tl.load(mask_ptr + offs_m * stride_m0,
                       mask=m_valid,
                       other=tl.float16(0.0))          # (BLOCK_M,)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        k_valid = k_off < K

        # ---- A tile -------------------------------------------------
        a = tl.load(a_ptr + (offs_m[:, None] * stride_am +
                             k_off[None, :] * stride_ak),
                     mask=m_valid[:, None] & k_valid[None, :],
                     other=tl.float16(0.0))          # (BLOCK_M, BLOCK_K)

        # ---- Weight tiles -------------------------------------------
        lp = tl.load(lp_w_ptr + (k_off[:, None] * stride_lp_wk +
                                 offs_h[None, :] * stride_lp_wh),
                     mask=k_valid[:, None] & h_valid[None, :],
                     other=tl.float16(0.0))
        lg = tl.load(lg_w_ptr + (k_off[:, None] * stride_lg_wk +
                                 offs_h[None, :] * stride_lg_wh),
                     mask=k_valid[:, None] & h_valid[None, :],
                     other=tl.float16(0.0))
        rp = tl.load(rp_w_ptr + (k_off[:, None] * stride_rp_wk +
                                 offs_h[None, :] * stride_rp_wh),
                     mask=k_valid[:, None] & h_valid[None, :],
                     other=tl.float16(0.0))
        rg = tl.load(rg_w_ptr + (k_off[:, None] * stride_rg_wk +
                                 offs_h[None, :] * stride_rg_wh),
                     mask=k_valid[:, None] & h_valid[None, :],
                     other=tl.float16(0.0))
        og = tl.load(og_w_ptr + (k_off[:, None] * stride_og_wk +
                                 offs_h[None, :] * stride_og_wh),
                     mask=k_valid[:, None] & h_valid[None, :],
                     other=tl.float16(0.0))

        # ---- Accumulate dot products --------------------------------
        left_proj_acc  += tl.dot(a, lp)      # (BLOCK_M, BLOCK_H)
        left_gate_acc  += tl.dot(a, lg)
        right_proj_acc += tl.dot(a, rp)
        right_gate_acc += tl.dot(a, rg)
        out_gate_acc   += tl.dot(a, og)

    # ------------------------------------------------------------------
    # Apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    one = tl.float16(1.0)
    left_gate_sig  = one / (one + tl.exp(-left_gate_acc))
    right_gate_sig = one / (one + tl.exp(-right_gate_acc))
    out_gate_sig   = one / (one + tl.exp(-out_gate_acc))

    # ------------------------------------------------------------------
    # Fuse mask (broadcast over hidden dim) and store results
    # ------------------------------------------------------------------
    mask_tile = mask_val[:, None]          # (BLOCK_M, 1)

    left_out  = left_proj_acc  * left_gate_sig  * mask_tile
    right_out = right_proj_acc * right_gate_sig * mask_tile

    store_mask = m_valid[:, None] & h_valid[None, :]

    tl.store(lo_ptr + offs_m[:, None] * stride_lo_m + offs_h[None, :] * stride_lo_h,
             left_out, mask=store_mask)
    tl.store(ro_ptr + offs_m[:, None] * stride_ro_m + offs_h[None, :] * stride_ro_h,
             right_out, mask=store_mask)
    tl.store(og_ptr + offs_m[:, None] * stride_og_m + offs_h[None, :] * stride_og_h,
             out_gate_sig, mask=store_mask)


# ----------------------------------------------------------------------
# Entrypoint expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation without gradients.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:        [B, N, N] (float32) – may be None when `nomask=True`
            - weights: dict of model parameters (float32)
            - config: dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and flatten to (M, K) where M = B·N·N, K = dim
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    B, N, _, _ = x_fp16.shape
    M = B * N * N                       # number of (b,i,k) triples
    K = dim

    a_flat = x_fp16.reshape(M, K).contiguous()        # (M, K)

    # ------------------------------------------------------------------
    # 3️⃣ Prepare (or fabricate) the mask – broadcasted over hidden_dim
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_flat = torch.ones(M, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(M).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Transpose & cast weight matrices to fp16 (K, H) layout
    # ------------------------------------------------------------------
    lp_w = weights["left_proj.weight"].t().to(torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].t().to(torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].t().to(torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].t().to(torch.float16).contiguous()
    og_w = weights["out_gate.weight"].t().to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Allocate output buffers (fp16)
    # ------------------------------------------------------------------
    left_out  = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right_out = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    out_gate  = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6️⃣ Launch the fused Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_H = 32
    BLOCK_K = 32

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(hidden_dim, BLOCK_H))

    _tri_mul_fused_proj_gate_kernel[grid](
        # pointers
        a_flat,
        lp_w,
        lg_w,
        rp_w,
        rg_w,
        og_w,
        mask_flat,
        left_out,
        right_out,
        out_gate,
        # strides
        a_flat.stride(0), a_flat.stride(1),
        lp_w.stride(0), lp_w.stride(1),
        lg_w.stride(0), lg_w.stride(1),
        rp_w.stride(0), rp_w.stride(1),
        rg_w.stride(0), rg_w.stride(1),
        og_w.stride(0), og_w.stride(1),
        mask_flat.stride(0),
        left_out.stride(0), left_out.stride(1),
        right_out.stride(0), right_out.stride(1),
        out_gate.stride(0), out_gate.stride(1),
        # problem sizes
        M, K, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape tensors back to [B, N, N, H]
    # ------------------------------------------------------------------
    left  = left_out .view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)   # sigmoid already applied

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation (batched GEMM)
    #     C_h = left_h @ right_hᵀ   (H independent GEMMs)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # fp16 GEMM – Tensor‑cores handle the accumulation in fp32 internally
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    # second LayerNorm over hidden_dim
    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # apply out‑gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B, N, N, dim] float32

    return out_final
