"""
Outgoing Triangle Multiplicative Update (TriMul) – fast Triton‑augmented implementation.

Algorithm
---------
1️⃣  Layer‑norm the input (float32).
2️⃣  Cast to fp16 and perform **one** huge GEMM that produces all
    projections and gates:
        • left_proj     (hidden_dim)
        • left_gate     (hidden_dim)
        • right_proj    (hidden_dim)
        • right_gate    (hidden_dim)
        • out_gate      (hidden_dim)
    The weight matrix is built by concatenating the five model weights,
    so the input tensor is loaded only once – the dominant memory‑bandwidth
    saving.
3️⃣  Apply sigmoid to the three gate tensors (still in fp16).
4️⃣  If a mask is present, a tiny Triton kernel fuses
        left = left * left_gate * mask
        right = right * right_gate * mask
    in‑place, avoiding two extra elementwise passes.
5️⃣  The core N³ operation   Σₖ left[..., i, k, d] * right[..., j, k, d]
    is realised as a batch of independent matrix multiplications
        C₍b,d₎ = left₍b,d₎ @ right₍b,d₎ᵀ
    using `torch.bmm`, which on H100 runs on Tensor‑Cores.
6️⃣  Second LayerNorm (float32), elementwise multiplication with the
    *out‑gate* and a final linear projection back to the original
    feature dimension (float32).

The function returns a tensor of shape `[B, N, N, dim]` (float32) and
contains a Triton kernel (step 4) as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * left (B,N,N,hd)
    right_ptr,         # fp16 * right (B,N,N,hd)
    mask_ptr,          # fp16 * mask (B,N,N)
    left_gate_ptr,     # fp16 * left_gate (B,N,N,hd)
    right_gate_ptr,    # fp16 * right_gate (B,N,N,hd)
    # strides for the 4‑D tensors
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    # strides for gate tensors (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,                           # batch, seq_len, hidden_dim
    BLOCK_H: tl.constexpr,             # tile size along hidden dimension
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode pid -> (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for each tensor
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    mask_val = tl.load(mask_ptr + off_m)      # scalar mask for this (b,i,j)

    # ------------------------------------------------------------------
    # loop over hidden dimension in blocks
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left  = tl.load(left_ptr  + off_l + offs * stride_l3,
                        mask=mask_h, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs * stride_r3,
                        mask=mask_h, other=tl.float16(0.0))
        left_gate  = tl.load(left_gate_ptr  + off_g + offs * stride_g3,
                             mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_g + offs * stride_g3,
                             mask=mask_h, other=tl.float16(0.0))

        # fused element‑wise operation
        left_out  = left  * left_gate  * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_ptr  + off_l + offs * stride_l3, left_out,  mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)


# ----------------------------------------------------------------------
# TriMul forward (outgoing) entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (float32). May be None.
            - weights:     dict of model parameters (float32)
            - config:      dict containing ``dim``, ``hidden_dim`` and ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣  Cast to fp16 and fuse *all* linear projections + gates
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16).contiguous()               # [B, N, N, dim]
    B, N, _, _ = x_fp16.shape
    M = B * N * N                                            # B·N²

    # flatten for the big GEMM
    x_flat = x_fp16.view(M, dim)                             # [M, dim]

    # concatenate every weight that depends on x  (left/right proj & gates + out_gate)
    w_left   = weights["left_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)

    # shape: (5*hidden_dim, dim)
    big_weight = torch.cat([w_left, w_lgate, w_right, w_rgate, w_outgate], dim=0)

    # single half‑precision GEMM
    proj = F.linear(x_flat, big_weight)                      # [M, 5*hd]  fp16

    # reshape back to 4‑D and split
    proj = proj.view(B, N, N, 5 * hidden_dim)
    left_raw, left_gate_raw, right_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )  # each: [B, N, N, hd]

    # ------------------------------------------------------------------
    # 3️⃣  Apply sigmoid to the three gate tensors (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)   # fp16
    right_gate = torch.sigmoid(right_gate_raw)  # fp16
    out_gate   = torch.sigmoid(out_gate_raw)    # fp16

    # ------------------------------------------------------------------
    # 4️⃣  Mask & gate fusion (if a mask is supplied)
    # ------------------------------------------------------------------
    if nomask:
        left  = left_raw * left_gate
        right = right_raw * right_gate
    else:
        # mask is required when nomask==False.  It may be None in edge cases;
        # treat a missing mask as all‑ones.
        if mask is None:
            mask = torch.ones(B, N, N, device=device, dtype=torch.float32)
        mask_fp16 = mask.to(torch.float16).contiguous()

        # ensure contiguous layout for the kernel
        left_raw   = left_raw.contiguous()
        right_raw  = right_raw.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = B * N * N
        _mask_gate_fuse_kernel[(total_programs,)](
            left_raw,
            right_raw,
            mask_fp16,
            left_gate,
            right_gate,
            # strides for left/right
            left_raw.stride(0), left_raw.stride(1), left_raw.stride(2), left_raw.stride(3),
            right_raw.stride(0), right_raw.stride(1), right_raw.stride(2), right_raw.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gates
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            B, N, hidden_dim,
            BLOCK_H=32,          # hidden‑dim tile (covers hd up to 128 efficiently)
            num_warps=4,
        )
        left  = left_raw
        right = right_raw

    # ------------------------------------------------------------------
    # 5️⃣  Core N³ multiplication: batched GEMM for each hidden head
    # ------------------------------------------------------------------
    # transform to (B*hd, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # BMM – fp16 Tensor‑Core accelerated
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hd]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣  Post‑processing: second LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # apply out‑gate (cast to float32 for the multiplication)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim] float32

    return out_final
