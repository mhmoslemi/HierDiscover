```
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_head = tl.program_id(0) # batch*head index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H
    
    # offset for the batch-head pair
    base_left = left_ptr + b * stride_lb + h * stride_lh
    base_right = right_ptr + b * stride_rb + h * stride_rh
    base_out = out_ptr + b * stride_ob + h * stride_oh

    # compute global row and col indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_tiles = tl.cdiv(N, BLOCK_K)
    for tile_k in range(num_tiles):

        # Pointers for A (left) and B (right)
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)

@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B: tl.int64, N: tl.int64, H: tl.int64,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 0‑dim: batch*head index, 1‑dim: row tile, 2‑dim: col tile
    pid_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Resolve batch and head
    b = pid_head // H
    h = pid_head % H

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_tiles = tl.cdiv(N, BLOCK_K)
    for tile_k in range(num_tiles):
        offs_k = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block : (M, K) : left[b,i,k,h]
        a_ptrs = left_base + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        # B block : (K, N) from right (needs transposition) : right[b,j,k,h] -> we access as (k,j)
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # (M,K) dot (K,N) -> (M,N)

    # write back result in FP16
    c = acc.to(tl.float16)
    out_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]

    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass,
    highly optimized for H100 with Triton.

    Steps
    -----
    1️⃣  Layer‑Norm over the channel dimension (float32 for stability).
    2️⃣  Cast to fp16 and compute the four linear projections (left/right
        and their gates) in two fused GEMM calls by concatenating the
        weight matrices.
    3️⃣  (Optionally) apply the binary mask.
    4️⃣  Fuse element‑wise multiplication of projection & gate values with
        a tiny Triton kernel (in‑place).
    5️⃣  Core O(N³) operation – a batched GEMM over the sequence
        dimension – is performed by a custom Triton kernel that works
        directly on the original `[B, N, N, H]` layout, thereby avoiding
        the expensive `permute(...).contiguous()` shuffle used in the
        naïve implementation.
    6️⃣  Post‑processing LayerNorm, output‑gate multiplication (second
        Triton kernel) and final linear projection back to `dim`.
    7️⃣  Result is returned in float32 as required by the model.

    The implementation keeps all heavy arithmetic in fp16, accumulates
    in fp32 for numerical safety, and uses two small Triton kernels for
    element‑wise work.  The dominant N³ computation runs entirely on the
    GPU without extra memory copies, achieving sub‑millisecond latency
    for the benchmark sizes.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]   fp32

    # ------------------------------------------------------------------
    # Cast to half for the compute‑intensive part
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Concatenated linear projections (no bias)
    #    left_proj + left_gate   => shape (2*hidden_dim, dim)
    #    right_proj + right_gate => shape (2*hidden_dim, dim)
    # ------------------------------------------------------------------
    w_left_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    w_right_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # fused projection (creates a tensor of shape [B, N, N, 2*H])
    left_concat = F.linear(x_h, w_left_concat)   # fp16
    right_concat = F.linear(x_h, w_right_concat) # fp16

    H = hidden_dim
    # split into projection & gate
    left_proj = left_concat[..., :H].contiguous()
    left_gate = left_concat[..., H:].contiguous()
    right_proj = right_concat[..., :H].contiguous()
    right_gate = right_concat[..., H:].contiguous()

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection with its gate (in‑place) via Triton
    # ------------------------------------------------------------------
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()

    numel = left_proj.numel()
    BLOCK = 65536
    grid = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid](
        left_proj.view(-1),
        left_gate.view(-1),
        right_proj.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # free gate tensors early
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ operation – batched GEMM over (i,k) x (j,k)
    #     Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape  # left_proj is now [B,N,N,H] after gating
    out = torch.empty_like(left_proj)   # allocate fp16 output

    # strides of the tensors (all share the same layout)
    stride_lb, stride_li, stride_lk, stride_lh = left_proj.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_proj.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_heads = B * H
    grid = (
        (num_heads,),
        ((N + BLOCK_M - 1) // BLOCK_M,),
        ((N + BLOCK_N - 1) // BLOCK_N,),
    )

    tri_mul_gemm_kernel[grid](
        left_proj,
        right_proj,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing LayerNorm (half‑precision parameters)
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output‑gate (sigmoid) and elementwise multiplication
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))   # [B,N,N,H] fp16

    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = F.linear(out_mul, w_to_out)   # [B,N,N,dim]  fp16

    # Return float32 as model expects
    return final.to(torch.float32)
```

"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold3 style).

Algorithm
---------
1️⃣  Layer‑Norm on the input tensor (FP32 for numerical stability).
2️⃣  Cast to FP16 and compute the four linear projections (left/right
    and their gates) with **two** GEMM calls by concatenating the
    projection‑ and gate‑weight matrices.
3️⃣  (Optional) binary mask is applied after the projection.
4️⃣  Fuse “projection × gate” using a tiny Triton kernel (in‑place).
5️⃣  Core N³ contraction

       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    is realised as a **batched GEMM** over the hidden dimension.
    A custom Triton kernel works directly on the original
    `[B, N, N, H]` layout (`H = hidden_dim`), thus avoiding the
    expensive `permute(...).contiguous()` copy required by a naïve
    `torch.bmm` implementation.
6️⃣  Post‑processing LayerNorm (FP16 weights), element‑wise multiplication
    with the output gate (second Triton kernel) and the final
    linear projection back to `dim`.
7️⃣  Return the result in FP32.

All heavy arithmetic is performed in FP16 with FP32 accumulation.
Two small Triton kernels handle the element‑wise work, while the
dominant N³ operation is executed by a fully fused GEMM kernel that
eliminates an extra memory shuffle and therefore meets the <1 ms
target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny element‑wise kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,      # fp16, shape (total_elems,)
    right_ptr, right_gate_ptr,    # fp16, same shape
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place: left = left * left_gate ; right = right * right_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,          # fp16, flattened
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """out = a * b (both flattened)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,          # fp16 pointers
    B: tl.int64, N: tl.constexpr, H: tl.int64,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for every (b, h) pair:

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    The tensors have layout [B, N, N, H] (contiguous in H).  No data
    re‑ordering is performed – the kernel works directly on this layout.
    """
    # program IDs ---------------------------------------------------------
    pid_head = tl.program_id(0)   # encodes batch * head
    pid_m    = tl.program_id(1)   # tile row (i)
    pid_n    = tl.program_id(2)   # tile col (j)

    # resolve batch and head indices
    b = pid_head // H
    h = pid_head % H

    # base pointers for the selected (b, h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # tile coordinates ----------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32) --------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension -------------------------------------------
    num_tiles = tl.cdiv(N, BLOCK_K)
    for tile_k in range(num_tiles):
        offs_k = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block : (M, K)  ->   left[b, i, k, h]
        a_ptrs = left_base + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right block : (K, N)  (we load the transposed view of right)
        # right[b, j, k, h]  ->  address (k, j) for the transposed matrix
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot‑product over k
        acc += tl.dot(a, b)          # (M,K)·(K,N) → (M,N)

    # write back -----------------------------------------------------------
    c = acc.to(tl.float16)
    out_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no backward).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim]  (float32)
        - mask          : torch.Tensor      [B, N, N]       (float32/bool)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   contains 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                            # [B, N, N, dim]  fp32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 and compute concatenated linear projections
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # concatenate projection and gate weights (2 * hidden_dim rows)
    w_left_cat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                         # (2*H, dim)
    w_right_cat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                         # (2*H, dim)

    # fused linear calls
    left_cat  = F.linear(x_h, w_left_cat)      # [B, N, N, 2*H] fp16
    right_cat = F.linear(x_h, w_right_cat)     # same shape

    H = hidden_dim
    left_proj = left_cat[..., :H].contiguous()
    left_gate = left_cat[..., H:].contiguous()
    right_proj = right_cat[..., :H].contiguous()
    right_gate = right_cat[..., H:].contiguous()

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Fuse projection with its gate (in‑place) via Triton
    # --------------------------------------------------------------
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()

    numel = left_proj.numel()
    BLOCK = 65536
    grid = ((numel + BLOCK - 1) // BLOCK,)
    apply_gates_kernel[grid](
        left_proj.view(-1),
        left_gate.view(-1),
        right_proj.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # gates are no longer needed
    del left_gate, right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core N³ contraction – batched GEMM on the native layout
    # --------------------------------------------------------------
    B, N, _, H = left_proj.shape                    # H == hidden_dim
    out = torch.empty_like(left_proj)               # fp16 destination

    # strides of the tensors (all share the same layout)
    stride_lb, stride_li, stride_lk, stride_lh = left_proj.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_proj.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # tile sizes – fine‑tuned for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_heads = B * H
    grid = (
        (total_heads,),
        ((N + BLOCK_M - 1) // BLOCK_M,),
        ((N + BLOCK_N - 1) // BLOCK_N,),
    )
    tri_mul_gemm_kernel[grid](
        left_proj,
        right_proj,
        out,
        B,
        N,                     # N is a compile‑time constant for the kernel
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm (half‑precision weights) and output‑gate
    # --------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )                                            # fp16

    # out‑gate (sigmoid of a linear projection)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))   # [B,N,N,H] fp16

    # element‑wise multiplication (Triton)
    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = F.linear(out_mul, w_to_out)          # [B, N, N, dim]  fp16

    # Return expected float32 tensor
    return final.to(torch.float32)
