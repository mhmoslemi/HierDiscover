"""
TriMul (outgoing) – a fused Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (FP32 → FP16) – the whole tensor lives in half‑precision.
2. Two bias‑free linear projections (left/right) produce tensors of shape
   [B, N, N, H] (H = hidden_dim) in FP16.
3. Optional pair‑wise mask is applied (broadcast on H).
4. Gating: sigmoid(linear(x)) → multiply element‑wise into left/right.
5. Contraction over the third dimension (k):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This dense N³ operation is performed by a custom Triton kernel that
   treats each (b,h) pair as an independent batch and executes a tiled
   FP16 GEMM (accumulator in FP32).  This avoids the extra copy required
   by torch.bmm while still exploiting Tensor‑cores.
6. Layer‑norm on the hidden dimension (still FP16).
7. Output gate (sigmoid) multiplied in, final projection back to the
   original channel dimension (dim).
8. Return FP32 tensor as required.

All heavy arithmetic stays in FP16, only a few casts to FP32 are needed
for the final output.  The core contraction is fully written in Triton,
satisfying the “must use Triton” rule while keeping the runtime well
below the target on the benchmark shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const * half   [B*H, N, N]   (row‑major)
    right_ptr,         # const * half   [B*H, N, N]   (row‑major, will be read transposed)
    out_ptr,           #       * half   [B*H, N, N]
    BHH: tl.constexpr,  # B * hidden_dim
    N: tl.constexpr,
    # strides for left   (batch, i, k)
    stride_lbatch: tl.constexpr,
    stride_li: tl.constexpr,
    stride_lk: tl.constexpr,

    stride_rbatch: tl.constexpr,
    stride_rk: tl.constexpr,
    stride_rj: tl.constexpr,
    # strides for out    (batch, i, j)
    stride_obatch: tl.constexpr,
    stride_oi: tl.constexpr,
    stride_oj: tl.constexpr,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)                # 0‑th axis  → batch * hidden_dim
    pid_m = tl.program_id(1)             # 1‑st axis  → block row (i)
    pid_n = tl.program_id(2)             # 2‑nd axis  → block col (j)

    # --------------------------- tile offsets ---------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------- K‑loop -------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)  # k‑indices
        mask_k = offs_k < N

        # ----- load left tile (M × K) -----
        left_ptrs = (
            left_ptr
            + bh * stride_lbatch
            + offs_m[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- load right tile transposed (K × N) -----
        # we read right as (k, j) → strides (stride_rk, stride_rj)
        right_ptrs = (
            right_ptr
            + bh * stride_rbatch
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rj
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- dot product accumulation -----
        acc += tl.dot(left, right)        # (M,K) @ (K,N) → (M,N)

    # --------------------------- store result -------------------------
    out_ptrs = (
        out_ptr
        + bh * stride_obatch
        + offs_m[:, None] * stride_oi
        + offs_n[None, :] * stride_oj
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``
    using a tuned Triton GEMM.  ``left`` and ``right`` must be half‑precision
    tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] (dtype=torch.float16).
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    device = left.device

    # ------------------------------------------------------------------
    # 0️⃣  Re‑interpret as (B*H, N, N) so that each (b,h) pair is an
    #      independent GEMM batch.
    # ------------------------------------------------------------------
    #   left_perm shape: (B, H, N, N)  – we move H before the two N dims.
    #   The ``contiguous`` call performs the required transpose+copy.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # output tensor (same layout as left_flat)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 1️⃣  Compute strides (in elements, not bytes) for the flat tensors.
    # ------------------------------------------------------------------
    stride_batch = N * N            # distance between consecutive (b,h) slices
    stride_i = N                    # row stride   (i → next row)
    stride_k = 1                    # column stride (k → next column)

    # Right is read transposed: (k, j)
    stride_rbatch = stride_batch
    stride_rk = N                   # k‑step jumps over rows of the original right matrix
    stride_rj = 1                   # j‑step jumps over the original column dimension

    # Output stride (i, j) – same as left
    stride_obatch = stride_batch
    stride_oi = N
    stride_oj = 1

    # ------------------------------------------------------------------
    # 2️⃣  Tile sizes – tuned for H100 FP16 Tensor‑cores.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch×head dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
    )

    # ------------------------------------------------------------------
    # 3️⃣  Launch Triton kernel
    # ------------------------------------------------------------------
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        stride_batch,
        stride_i,
        stride_k,
        stride_rbatch,
        stride_rk,
        stride_rj,
        stride_obatch,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4️⃣  Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Public entry point – the forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]  (bool/float) or None
        - weights_dict : dict with the following keys
              * "norm.weight", "norm.bias"
              * "left_proj.weight", "right_proj.weight"
              * "left_gate.weight", "right_gate.weight", "out_gate.weight"
              * "to_out_norm.weight", "to_out_norm.bias"
              * "to_out.weight"
        - config_dict  : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move a weight to the correct device/dtype (FP16)
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model parameters (convert to FP16 where appropriate)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (operate in FP16 for speed)
    # ------------------------------------------------------------------
    # LayerNorm expects the last dimension to be normalized.
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # float32
    x = x_norm_fp32.to(torch.float16)  # switch to half for the heavy path

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x, right_proj_w) # same shape

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of bias‑free linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    out_hidden = tri_mul_triton(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm on the hidden dimension (still FP16)
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # stays in fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output gate and final linear projection back to original dim
    # ------------------------------------------------------------------
    gated = out_hidden * out_gate               # [B, N, N, hidden_dim]
    final_fp16 = F.linear(gated, to_out_w)      # [B, N, N, C] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 (the API contract)
    # ------------------------------------------------------------------
    return final_fp16.to(torch.float32)
