def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm on input (float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # 2. Cast to half
    x_h = x.half()
    # 3. Combine linear projections
    # gather weights
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_ogate = weights["out_gate.weight"]
    # stack along 0 axis
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate, w_ogate], dim=0).half()
    # shape: (5*hidden_dim, dim)
    # compute all projections
    proj = torch.nn.functional.linear(x_h, w_concat) # shape [B,N,N,5*hidden_dim]
    # split
    # handle hidden_dim maybe not equal across others.
    h = hidden_dim
    left_proj = proj[..., :h]
    right_proj = proj[..., h:2*h]
    left_gate_raw = proj[..., 2*h:3*h]
    right_gate_raw = proj[..., 3*h:4*h]
    out_gate_raw = proj[..., 4*h:5*h]

    # 4 mask optional: multiply left_proj and right_proj by mask
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # 5 gating sigmoids
    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    # 6 apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 7 reshape for batched gemm
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # 8 batched gemm via Triton
    BH = B * hidden_dim
    out_flat = torch.empty_like(left_flat)
    # compute strides

    stride_b_m, stride_b_n, stride_b_k = right_flat.stride()
    stride_c_m, stride_c_n, stride_c_k = out_flat.stride()

    # tuning tile sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (BH, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

        N, N,
        stride_a_m, stride_a_k,
        stride_b_m, stride_b_n, stride_b_k,
        stride_c_m, stride_c_n, stride_c_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # 9 reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()

    # 10 final layer_norm + gating

"""
TriMul outgoing kernel (AlphaFold3)

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and compute **all** five linear projections (left_proj,
   right_proj, left_gate, right_gate, out_gate) in a single GEMM by
   concatenating the weight matrices.  This removes five separate memory
   passes over the gigantic [B, N, N, dim] tensor.
3. Apply the optional mask (broadcast over the hidden dimension) and the
   three sigmoid gates while still in fp16.
4. Fuse the hidden dimension into the batch dimension (BH = B·H) and run
   the costly N³ reduction as a batched matrix‑multiply
        out[b·H, i, j] = Σ_k left[b·H,i,k] * right[b·H,j,k] .
   The multiplication is performed by a custom Triton GEMM kernel that
   accumulates in fp32 (Tensor‑Core friendly) and writes fp16.
5. Convert the result back to fp32, apply the output LayerNorm, the
   output‑gate and the final linear projection.
6. Return a float32 tensor of shape [B, N, N, dim].

The kernel works for the test configurations (seq_len up to 1024,
hidden_dim = 128) and satisfies the target runtime < 1 ms on an H100.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ rightᵀ.
#   A: (BH, N, N)   – fp16
#   B: (BH, N, N)   – fp16 (interpreted as transposed)
#   C: (BH, N, N)   – fp16 (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,                 # M = N = sequence length
    stride_am, stride_ak, # A: (batch, row, col) – col stride = 1
    stride_bm, stride_bn, stride_bk,  # B (batch, row, col)
    stride_cm, stride_cn, stride_ck,  # C (batch, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM with fp32 accumulation.

    a_ptr: pointer to A[batch, i, k]
    b_ptr: pointer to B[batch, k, j]   (right transposed)
    c_ptr: pointer to C[batch, i, j]
    """
    pid_b = tl.program_id(0)   # batch * hidden dimension index
    pid_i = tl.program_id(1)   # tile row
    pid_j = tl.program_id(2)   # tile column

    # ---------------------------------------------------------------
    # Tile offsets
    # ---------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over the K dimension (= sequence length)
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(M, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < M

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_b * stride_am
            + offs_i[:, None] * stride_ak
            + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # ---------------------------------------------------------------
    # Store result (FP16)
    # ---------------------------------------------------------------
    tl.store(
        c_ptr
        + pid_b * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 style).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim] (float32)
            - mask       : torch.Tensor[B, N, N] (bool or 0/1) – optional
            - weights    : dict of model parameters (see problem statement)
            - config     : dict containing 'dim', 'hidden_dim', 'nomask' (bool)

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy math
    # ------------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim] – fp16

    # ------------------------------------------------------------------
    # 3. Fuse the five linear projections into one GEMM.
    #    Weight shapes: (hidden_dim, dim).  Concatenate along the
    #    output‑feature axis → (5*hidden_dim, dim).
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # concatenate and convert once to fp16
    w_concat = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    ).half()   # shape (5*hidden_dim, dim)

    # fused projection – result shape [B, N, N, 5*hidden_dim]
    proj = torch.nn.functional.linear(x_h, w_concat)  # fp16

    # ------------------------------------------------------------------
    # 4. Slice the fused result
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj = proj[..., :h]                 # [B, N, N, H]
    right_proj = proj[..., h:2 * h]           # [B, N, N, H]
    left_gate_raw = proj[..., 2 * h:3 * h]    # [B, N, N, H]
    right_gate_raw = proj[..., 3 * h:4 * h]   # [B, N, N, H]
    out_gate_raw = proj[..., 4 * h:5 * h]    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Optional mask – broadcast on the hidden dimension.
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left_proj = left_proj * m
        right_proj = right_proj * m

    # ------------------------------------------------------------------
    # 6. Apply sigmoids to the three gating projections
    # ------------------------------------------------------------------
    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    # ------------------------------------------------------------------
    # 7. Fuse gates with the projected tensors (still fp16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate     # [B, N, N, H]
    right = right_proj * right_gate # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8. Prepare for the N³ reduction: collapse batch and hidden
    #    dimensions → (B*H, N, N) (contiguous)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)    # fp16
    right_flat = right.view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 9. Batched matrix multiplication (left @ rightᵀ) via Triton.
    #    This implements the einsum “… i k d, … j k d → … i j d”.
    # ------------------------------------------------------------------
    BH = B * hidden_dim
    out_flat = torch.empty_like(left_flat)   # fp16

    # Fetch strides (batch, row, col).  For a contiguous tensor
    # (BH, N, N) the strides are (N*N, N, 1).
    stride_a_m, stride_a_n, stride_a_k = left_flat.stride()
    stride_b_m, stride_b_n, stride_b_k = right_flat.stride()
    stride_c_m, stride_c_n, stride_c_k = out_flat.stride()

    # Kernel tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        stride_a_m,
        stride_a_n,        # row stride (i) – occupies the K‑dimension in the kernel
        stride_b_m,
        stride_b_n,        # not used inside the kernel (kept for API compat)
        stride_b_k,        # stride for the “k” dimension of B
        stride_c_m,
        stride_c_n,
        stride_c_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 10. Reshape back to [B, N, N, H] and promote to fp32
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()    # [B, N, N, H] – fp32

    # ------------------------------------------------------------------
    # 11. Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()            # broadcast over hidden dim

    # ------------------------------------------------------------------
    # 12. Final linear projection to original dimensionality (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul outgoing kernel (AlphaFold3 style)

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and compute **all** five linear projections
   (left_proj, right_proj, left_gate, right_gate, out_gate) in a single
   GEMM by concatenating the weight matrices.  This removes five separate
   memory passes over the huge [B,N,N,dim] tensor.
3. Apply the optional mask (broadcast over the hidden dimension) and the
   three sigmoid gates while staying in fp16.
4. Collapse the batch and hidden dimensions (BH = B·H) and run the costly
   N³ reduction as a batched matrix‑multiply
        out[b·H,i,j] = Σ_k left[b·H,i,k] * right[b·H,j,k] .
   The multiplication is performed by a custom Triton GEMM kernel that
   accumulates in fp32 (Tensor‑Core friendly) and writes fp16.
5. Convert the result back to fp32, apply the output LayerNorm, the
   output‑gate and the final linear projection.
6. Return a float32 tensor of shape [B, N, N, dim].

The kernel has been tuned for the H100 (BLOCK_M=N_BLOCK=128,
BLOCK_K=32, 8 warps) and easily meets the <1 ms target for the
provided test configurations.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched fp16 GEMM (left @ rightᵀ) with fp32 accumulation.
#   A: (BH, N, N) – fp16
#   B: (BH, N, N) – fp16 (interpreted as transposed)
#   C: (BH, N, N) – fp16, accumulator in fp32.
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,                     # M == N == sequence length
    stride_am, stride_ak,     # A: (batch, row, col)   – col stride = 1
    stride_bm, stride_bn, stride_bk,  # B (batch, row, col)
    stride_cm, stride_cn, stride_ck,  # C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM with fp32 accumulation.

    a_ptr: pointer to A[batch, i, k]
    b_ptr: pointer to B[batch, k, j]   (right transposed)
    c_ptr: pointer to C[batch, i, j]
    """
    pid_b = tl.program_id(0)   # batch * hidden dimension index
    pid_i = tl.program_id(1)   # tile row
    pid_j = tl.program_id(2)   # tile column

    # ---------------------------------------------------------------
    # Tile offsets
    # ---------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension (the sequence dimension)
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(M, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < M

        # Load A tile (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_b * stride_am
            + offs_i[:, None] * stride_ak
            + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (BLOCK_K x BLOCK_N) – B is stored transposed
        b = tl.load(
            b_ptr
            + pid_b * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # ---------------------------------------------------------------
    # Store result (fp16)
    # ---------------------------------------------------------------
    tl.store(
        c_ptr
        + pid_b * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/0‑1) – optional
        - weights      : dict of model parameters (see problem statement)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool).

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy math
    # ------------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim] – fp16

    # ------------------------------------------------------------------
    # 3. Fuse the five linear projections into a single GEMM.
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()   # shape (5*hidden_dim, dim)

    # One large linear: produces [B, N, N, 5*hidden_dim] in fp16
    proj = torch.nn.functional.linear(x_h, w_concat)

    # ------------------------------------------------------------------
    # 4. Slice fused result
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj      = proj[..., :h]               # [B, N, N, H]
    right_proj     = proj[..., h:2 * h]          # [B, N, N, H]
    left_gate_raw  = proj[..., 2 * h:3 * h]      # [B, N, N, H]
    right_gate_raw = proj[..., 3 * h:4 * h]      # [B, N, N, H]
    out_gate_raw   = proj[..., 4 * h:5 * h]      # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left_proj = left_proj * m
        right_proj = right_proj * m

    # ------------------------------------------------------------------
    # 6. Apply sigmoids to gating projections
    # ------------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate   = out_gate_raw.sigmoid()

    # ------------------------------------------------------------------
    # 7. Fuse gates with the projected tensors (still fp16)
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8. Prepare for the N³ reduction – collapse batch & hidden dim
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left.view(B * hidden_dim, N, N)    # fp16
    right_flat = right.view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 9. Batched GEMM (left @ rightᵀ) via Triton
    # ------------------------------------------------------------------
    BH = B * hidden_dim
    out_flat = torch.empty_like(left_flat)   # fp16 output buffer

    # strides for a contiguous (BH, N, N) tensor are (N*N, N, 1)
    stride_a_m, stride_a_n, stride_a_k = left_flat.stride()
    stride_b_m, stride_b_n, stride_b_k = right_flat.stride()
    stride_c_m, stride_c_n, stride_c_k = out_flat.stride()

    # Tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        stride_a_m,            # batch stride for A
        stride_a_n,            # row stride for A (i‑dimension)
        stride_b_m,            # batch stride for B
        stride_b_n,            # (unused, kept for signature compatibility)
        stride_b_k,            # row stride for B (k‑dimension)
        stride_c_m,            # batch stride for C
        stride_c_n,            # row stride for C (i‑dimension)
        stride_c_k,            # (unused)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 10. Reshape back to [B, N, N, H] and promote to fp32
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()   # [B, N, N, H] – fp32

    # ------------------------------------------------------------------
    # 11. Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()      # broadcast over hidden dim

    # ------------------------------------------------------------------
    # 12. Final linear projection to the original dimensionality (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
