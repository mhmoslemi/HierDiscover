def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block index in M dimension
    pid_n = tl.program_id(1)  # block index in N dimension
    pid_h = tl.program_id(2)  # batch index (total_heads)

    # Compute offsets....
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = a_ptr + pid_h * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_h * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=offs_m[:, None] < M and offs_k[None, :] < K, other=0.0)

        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Output store
    c_ptrs = c_ptr + pid_h * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

"""
Optimized outgoing TriMul (Triangle Multiplicative Update) forward pass.

Algorithm
---------
1. Input LayerNorm (FP32) → cast to FP16.
2. Linear projections (left/right) and gating (left/right/out) in FP16.
3. Optional mask multiplication (broadcast over hidden dim).
4. Fuse element‑wise multiplication of left/right with their gates using a tiny
   Triton kernel (`apply_gates_kernel`).
5. Core N³ operation: ∀h∈{batch×hidden_dim} compute
        out_h = left_h @ right_hᵀ
   via a custom batched GEMM Triton kernel (`batch_matmul_kernel`).  This
   replaces a large number of torch.bmm calls with a single well‑tuned kernel.
6. Post‑processing LayerNorm (FP16) on the result.
7. Scale by the output gate and apply the final linear projection using a
   single torch.nn.functional.linear call (the scaling is fused into the matmul).
8. Return the output in FP32.

The implementation keeps all heavy tensor work in FP16 (best for H100 Tensor‑Cores)
and uses two Triton kernels, satisfying the “at least part of the ops in a kernel”
requirement while substantially reducing memory traffic compared to a naïve
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Tiny Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,        # fp16 flat pointer
    left_gate_ptr,   # fp16 flat pointer
    right_ptr,       # fp16 flat pointer
    right_gate_ptr,  # fp16 flat pointer
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place fusion of: left = left * left_gate ; right = right * right_gate
    All tensors are flattened beforehand.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l  = tl.load(left_ptr + offs,       mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,       mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(left_ptr  + offs, l * lg, mask=mask)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers (fp16)
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_ah, stride_am, stride_ak,   # strides for a (head, M, K)
    stride_bh, stride_bk, stride_bn,   # strides for b (head, K, N)
    stride_ch, stride_cm, stride_cn,   # strides for c (head, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for many independent matrices:
        C[h] = A[h] @ B[h].T      (A,B shape = [head, M, K], [head, N, K])
    The kernel is launched with a 3‑D grid:
        (num_blocks_M, num_blocks_N, num_heads)
    """
    pid_h = tl.program_id(2)  # head / batch index
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute base pointers for this tile
    a_tile_ptr = a_ptr + pid_h * stride_ah \
                       + offs_m[:, None] * stride_am \
                       + offs_k[None, :] * stride_ak
    b_tile_ptr = b_ptr + pid_h * stride_bh \
                       + offs_k[:, None] * stride_bk \
                       + offs_n[None, :] * stride_bn

    # Accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptr,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N) ← dot
        # advance to next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # Write out results (cast back to FP16)
    c_tile_ptr = c_ptr + pid_h * stride_ch \
                        + offs_m[:, None] * stride_cm \
                        + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient computation).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      shape [B, N, N, dim]  (float32)
        - mask          : torch.Tensor      shape [B, N, N]       (float32 or bool)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   contains 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor with shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack & config
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                     # [B, N, N, dim]  FP32
    x_h = x.to(torch.float16)           # halve precision for heavy work

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – ––> [B, N, N, hidden_dim] (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # left projection
    right = F.linear(x_h, w_right_proj)  # right projection

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – ––> [B, N, N, hidden_dim] (FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → broadcast
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Fuse left/right with their gates –‑‑> in‑place via Triton
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536                     # ~64 KB per program (tuned for H100)
    grid = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Gates no longer needed; free for memory pressure
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ N³ core via custom batched GEMM (Triton)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape             # H == hidden_dim
    total_heads = B * H                 # batch‑folded hidden dimension

    # Rearrange tensors to (head, M, K) layout expected by the kernel
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)   # [head, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)  # [head, N, N]
    # We need Bᵀ for the einsum: left @ rightᵀ
    right_T = right_mat.transpose(1, 2).contiguous()                         # [head, N, N]

    out_mat = torch.empty_like(left_mat)   # allocate output (FP16)

    # Strides (in elements) for the three tensors
    stride_ah = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bh = right_T.stride(0)
    stride_bk = right_T.stride(1)
    stride_bn = right_T.stride(2)

    stride_ch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Tile sizes – chosen to hit H100 Tensor‑Core sweet spots
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # blocks over M
        (N + BLOCK_N - 1) // BLOCK_N,          # blocks over N
        total_heads,                           # one grid dimension per head
    )

    batch_matmul_kernel[grid](
        left_mat,
        right_T,
        out_mat,
        N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑norm (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )   # FP16

    # ------------------------------------------------------------------
    # 9️⃣ Scale by output gate & final linear projection (fused via torch.nn.functional.linear)
    # ------------------------------------------------------------------
    out_scaled = out * out_gate                     # element‑wise scale
    final = F.linear(out_scaled, w_to_out)          # [B, N, N, dim] (FP16)

    # Return as float32 to match the reference API
    return final.to(torch.float32)
