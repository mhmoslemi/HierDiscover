5️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix‑multiply C = A @ Bᵀ for each (batch, hidden) pair.
   The contraction is performed by a Triton kernel that:
        – packs the (batch, hidden) dimensions into a single leading axis,
          i.e. A,B ∈ [M,N,N] with M = B·hidden,
        – uses a 128×128×64 tile and 8 warps (2 stages) – optimal for H100,
        – accumulates in FP32 (tensor‑core FP16 × FP16 → FP32) and writes FP16.
   The kernel is double‑buffered (prefetch next K‑tile while computing the
   current one) to hide global‑memory latency.
6️⃣ Layer‑norm on the hidden dimension, element‑wise *out_gate* (sigmoid of a
   separate linear), and final projection back to the original feature size.
7️⃣ Cast back to float32 (the format required by the harness).

Only the O(N³) part is written in Triton; all other ops use high‑performance
PyTorch primitives.  The implementation supports both masked and un‑masked
inputs, arbitrary batch sizes (1‑2), and the full range of sequence lengths
and dimensions required by the test suite.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ   (A,B ∈ [M,N,N]  fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]   (K == N)
    B_ptr,               # *fp16  [M, N, K]   (K == N)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)  – note B is accessed transposed
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs (tile coordinates)
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile
    pid_z = tl.program_id(2)          # (batch·hidden) tile

    # ---------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows   (i)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols   (j)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---------- accumulator (FP32) ----------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over K dimension ----------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # pre‑compute the K‑axis indices shared by all iterations
    offs_k = tl.arange(0, BLOCK_K)

    # Load the first K‑tile (k = 0)
    cur_k = 0
    a_ptrs = (
        A_ptr
        + pid_z * stride_am
        + offs_m[:, None] * stride_ai
        + (cur_k + offs_k)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_z * stride_bm
        + (cur_k + offs_k)[:, None] * stride_bk
        + offs_n[None, :] * stride_bj
    )
    a = tl.load(
        a_ptrs,
        mask=mask_m[:, None] & ((cur_k + offs_k) < K),
        other=0.0,
    )
    b = tl.load(
        b_ptrs,
        mask=((cur_k + offs_k)[:, None] < K) & mask_n[None, :],
        other=0.0,
    )

    for k_iter in range(num_k_tiles):
        # -------- pre‑fetch next K‑tile (if any) ----------
        next_k = cur_k + BLOCK_K
        if k_iter < num_k_tiles - 1:
            a_next_ptrs = (
                A_ptr
                + pid_z * stride_am
                + offs_m[:, None] * stride_ai
                + (next_k + offs_k)[None, :] * stride_ak
            )
            b_next_ptrs = (
                B_ptr
                + pid_z * stride_bm
                + (next_k + offs_k)[:, None] * stride_bk
                + offs_n[None, :] * stride_bj
            )
            a_next = tl.load(
                a_next_ptrs,
                mask=mask_m[:, None] & ((next_k + offs_k) < K),
                other=0.0,
            )
            b_next = tl.load(
                b_next_ptrs,
                mask=((next_k + offs_k)[:, None] < K) & mask_n[None, :],
                other=0.0,
            )
        # -------- compute dot for current tile ----------
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        # -------- advance to next tile ----------
        a = a_next if k_iter < num_k_tiles - 1 else a
        b = b_next if k_iter < num_k_tiles - 1 else b
        cur_k = next_k

    # ---------- write result ----------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a Triton kernel.  All tensors must be contiguous fp16.
    Returns a tensor of the same dtype & device.
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension → [M,N,N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile size selection – 128×128 works best for N ≥ 512 on H100.
    # For very small N we fall back to 64×64.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # Cast to half‑precision for the rest of the pipeline
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the four linear projections (left/right + their gates)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]

    # Concatenate all four weight matrices → shape (4·hidden, dim)
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    ).to(torch.float16)  # fp16

    # One big linear pass (reads `x` only once)
    fused_out = torch.nn.functional.linear(x, fused_weight)  # [B,N,N,4·hidden]

    # Split back into the four components
    left_proj, left_gate, right_proj, right_gate = torch.split(
        fused_out, hidden_dim, dim=-1
    )  # each: [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3️⃣ (Optional) mask the projected tensors (mask does NOT affect gates)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -----------------------------------------------------------------
    # 4️⃣ Sigmoid gates and fuse them with the projections
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)          # [B,N,N,hidden]
    right_gate = torch.sigmoid(right_gate)        # [B,N,N,hidden]

    left = left_proj * left_gate                  # [B,N,N,hidden]
    right = right_proj * right_gate               # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden]  fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm on the hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Out‑gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden]

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim]  fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
