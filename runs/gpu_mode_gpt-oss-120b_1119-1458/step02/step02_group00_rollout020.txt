```python
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)   # combined batch and hidden
    pid1 = tl.program_id(1)   # i tile
    pid2 = tl.program_id(2)   # j tile

    # Compute batch and hidden-index
    b = pid0 // H
    h = pid0 % H

    # compute tile start offsets
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # base pointer offsets for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over k dimension
    for k in range(0, N, BLOCK_K):
        # offsets within this tile
        offs_i = i0 + tl.arange(0, BLOCK_M)
        offs_j = j0 + tl.arange(0, BLOCK_N)
        offs_k = k + tl.arange(0, BLOCK_K)

        # mask for boundaries
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # pointers for A (left)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # pointers for B (right), note B's layout (k, j)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store result to out
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
Highly‑optimized outgoing Triangle‑Multiplicative update (TriMul) for AlphaFold‑3.

Algorithm
---------
1. Input LayerNorm (FP32) → cast to FP16.
2. Fuse the projection + gate linear layers:
       [left_proj , left_gate]  = X @ [W_left_proj ; W_left_gate]
       [right_proj, right_gate] = X @ [W_right_proj; W_right_gate]
   Split the concatenated results, apply the (optional) mask to the
   projections, then multiply by the sigmoid‑gates.
3. Core N³ operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that treats each (b,h) pair as a
   separate batch and performs a tiled FP16 GEMM with FP32 accumulation.
   No transposes or copies of the input tensors are required.
4. Output LayerNorm (FP16) → sigmoid(out_gate) → element‑wise multiply.
5. Final linear projection back to `dim`.
All heavy tensor work runs in FP16; LayerNorms use the standard FP32
reduction for numerical stability.  The only Triton kernels are the
core N³ GEMM and a tiny element‑wise multiply (steps 3 & 4); everything
else uses a few fused PyTorch operations (including a fused projection/
gate linear pass) to minimise memory traffic.

The implementation supports all test configurations listed in the
problem statement, works on H100 GPUs (Triton 3.3.1) and returns a
`torch.float32` tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B, N, H,           # int64
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single hidden channel `h`.  The kernel iterates over the
    reduction dimension `k` in tiles of size BLOCK_K.
    """
    pid0 = tl.program_id(0)   # combined batch & hidden index
    pid1 = tl.program_id(1)   # i‑tile index
    pid2 = tl.program_id(2)   # j‑tile index

    # ----- decode batch and hidden index ---------------------------------
    b = pid0 // H               # batch
    h = pid0 % H                # hidden channel

    # ----- tile offsets ---------------------------------------------------
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over k ---------------------------------------------
    for k in range(0, N, BLOCK_K):
        # Offsets inside the current tile
        offs_i = i0 + tl.arange(0, BLOCK_M)
        offs_j = j0 + tl.arange(0, BLOCK_N)
        offs_k = k  + tl.arange(0, BLOCK_K)

        # Boundary masks
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Load A = left[b,i,k,h]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B = right[b,j,k,h]   shape (BLOCK_K, BLOCK_N)
        #   note: we need the matrix transposed w.r.t. the original layout,
        #   i.e. rows are k, columns are j.
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate (FP16×FP16 → FP32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ----- store result ---------------------------------------------------
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


@triton.jit
def _elementwise_mul_kernel(
    a_ptr,    # fp16 input
    b_ptr,    # fp16 input
    out_ptr,  # fp16 output
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """out = a * b (both flattened)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

    """
    Forward pass of the outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------------
    # Unpack
    # ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ---------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) and cast to FP16
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_h = x.to(torch.float16)  # [B,N,N,dim]  fp16

    # ---------------------------------------------------------------
    # 2️⃣ Fuse projection + gate linear layers (2 GEMMs instead of 4)
    # ---------------------------------------------------------------
    # left + left_gate
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_left_comb = torch.cat([w_left_proj, w_left_gate], dim=0).to(torch.float16)   # (2*H, dim)

    left_comb = F.linear(x_h, w_left_comb)                         # [B,N,N,2*H] fp16
    left_proj = left_comb[..., :hidden_dim]                       # projection
    left_gate  = torch.sigmoid(left_comb[..., hidden_dim:])       # gate

    # right + right_gate
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_right_comb = torch.cat([w_right_proj, w_right_gate], dim=0).to(torch.float16)

    right_comb = F.linear(x_h, w_right_comb)                      # [B,N,N,2*H] fp16
    right_proj = right_comb[..., :hidden_dim]
    right_gate  = torch.sigmoid(right_comb[..., hidden_dim:])

    # ---------------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast over hidden dim) before gating
    # ---------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1]
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ---------------------------------------------------------------
    # 4️⃣ Fuse projection with its gate (element‑wise)
    # ---------------------------------------------------------------
    left  = left_proj * left_gate   # [B,N,N,H] fp16
    right = right_proj * right_gate

    # ---------------------------------------------------------------
    # 5️⃣ Core N³ multiplication (batched GEMM over the hidden dim)
    # ---------------------------------------------------------------

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        total_heads,                        # axis 0 : batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # axis 1 : i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # axis 2 : j‑tiles
    )

    # Strides (in element counts, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h   = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    # Allocate output tensor
    out = torch.empty_like(left)

    _tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP16) and output‑gate multiplication
    # ---------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # out_gate = sigmoid(x @ out_gate_weight)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))  # [B,N,N,H] fp16

    # Element‑wise multiplication – Triton kernel (in‑place style)
    out_mul = torch.empty_like(out_norm)
    numel = out_norm.numel()
    BLOCK = 65536
    grid_mul = ((numel + BLOCK - 1) // BLOCK,)
    _elementwise_mul_kernel[grid_mul](
        a_ptr=out_norm,
        b_ptr=out_gate,
        out_ptr=out_mul,
        numel=numel,
        BLOCK=BLOCK,
    )

    # ---------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ---------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = F.linear(out_mul, w_to_out)   # [B,N,N,dim] fp16

    # Return in the expected FP32 format
    return final.to(torch.float32)
```
