def tri_mul_triton(left, right, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Permute to shape (B, H, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    batch_head = B * H
    # Reshape to (batch_head, N, N)
    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)
    out_flat = torch.empty_like(left_flat)  # same dtype (fp16)
    # Get pointers
    left_ptr = left_flat.data_ptr()
    right_ptr = right_flat.data_ptr()
    out_ptr = out_flat.data_ptr()
    # Compute strides in number of elements (int64)
    # Strides for (batch_head, N, N) contiguous: (N*N, N, 1)
    stride_batch = N * N
    stride_m = N
    stride_k = 1
    # Kernel launch
    grid = (batch_head, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        N, N, N,
        stride_batch, stride_m, stride_k,
        stride_batch, stride_m, stride_k,
        stride_batch, stride_m, stride_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Bring back to original layout: out_perm shape (B, H, N, N)
    out_perm = out_flat.view(B, H, N, N)
    # Permute back to original shape (B, N, N, H)
    out = out_perm.permute(0,2,3,1).contiguous()
    return out

def tri_mul_triton(left: torch.Tensor, right: torch.Tensor, block_m=64, block_n=64, block_k=32):
    B, N, _, H = left.shape
    # Permute and contiguous: (B, H, N, N). Then flatten batch*head.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_head = B * H

    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)
    out_flat = torch.empty_like(left_flat)  # fp16

    # Setup strides
    stride_batch = N * N
    stride_row = N
    stride_col = 1

    grid = (batch_head, (N + block_m - 1) // block_m, (N + block_n - 1) // block_n)

    tri_mul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        N, N, N,
        stride_batch, stride_row, stride_col,
        stride_batch, stride_row, stride_col,
        stride_batch, stride_row, stride_col,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )

    # Restore shape: (B, H, N, N)
    out_perm = out_flat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    bh = tl.program_id(0)  # batch_head index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    block_m = pid_m * BLOCK_M
    block_n = pid_n * BLOCK_N

    # Compute base offsets for each tensor
    left_batch_offset = bh * stride_l_batch
    right_batch_offset = bh * stride_r_batch
    out_batch_offset = bh * stride_o_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, K, BLOCK_K):
        # Offsets for K
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile (M x K)
        offs_m = block_m + tl.arange(0, BLOCK_M)
        mask_m = offs_m < M

        left_ptrs = left_ptr + left_batch_offset + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # Load right tile (K x N) transposed (i.e., right[j,k] accessed)
        offs_n = block_n + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        # pointer: row=j (offs_n), col=k (offs_k), using original layout (row stride N, col stride 1)
        right_ptrs = right_ptr + right_batch_offset + offs_n[None, :] * stride_r_n + offs_k[:, None] * stride_r_k
        right_tile = tl.load(right_ptrs, mask=mask_n[None, :] & mask_k[:, None], other=0.0, dtype=tl.float16)

        # Accumulate matmul
        acc += tl.dot(left_tile, right_tile)  # fp16 inputs, fp32 accumulation

    # Write back to output
    offs_m = block_m + tl.arange(0, BLOCK_M)
    offs_n = block_n + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    out_ptrs = out_ptr + out_batch_offset + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    out_tile = acc.to(tl.float16)
    tl.store(out_ptrs, out_tile, mask=mask_m[:, None] & mask_n[None, :])

    out_gate_w = to_fp16(weights["out_gate.weight"])

    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])

    left = F.linear(x, left_proj_w)   # shape [B,N,N,hidden_dim]

    if mask is not None:

        right = right * mask_h

    out = tri_mul_triton(left, right, block_m=64, block_n=64, block_k=32)  # (B,N,N,hidden_dim) fp16

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel performing batched matrix multiplication for
# the N³ contraction in the outgoing TriMul operator.
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------------------
    # Program IDs
    # -------------------------------------------------------------
    bh = tl.program_id(0)       # batch * head dimension
    pid_m = tl.program_id(1)    # row block index
    pid_n = tl.program_id(2)    # column block index

    # -------------------------------------------------------------
    # Compute start offsets for this tile
    # -------------------------------------------------------------
    block_m = pid_m * BLOCK_M
    block_n = pid_n * BLOCK_N

    # Offsets for the batch/head slice
    left_batch_off = bh * stride_l_batch
    right_batch_off = bh * stride_r_batch
    out_batch_off = bh * stride_o_batch

    # -------------------------------------------------------------
    # Accumulator (FP32 for numeric stability)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Iterate over the K dimension (the sum axis of the contraction)
    # -------------------------------------------------------------
    for kk in range(0, K, BLOCK_K):
        # K‑axis offsets and bounds
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---------------------------------------------------------
        # Load a tile from the left operand:   left[bh, i, k]
        # ---------------------------------------------------------
        offs_i = block_m + tl.arange(0, BLOCK_M)
        mask_i = offs_i < M
        left_ptrs = left_ptr + left_batch_off \
                      + offs_i[:, None] * stride_l_m \
                      + offs_k[None, :] * stride_l_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # ---------------------------------------------------------
        # Load a tile from the right operand (transposed on‑the‑fly)
        #   we need right[j, k] but store it as a (K, N) tile:
        #   rows → k, cols → j
        # ---------------------------------------------------------
        offs_j = block_n + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N
        right_ptrs = right_ptr + right_batch_off \
                      + offs_j[None, :] * stride_r_n \
                      + offs_k[:, None] * stride_r_k
        b = tl.load(right_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0,
                    dtype=tl.float16)

        # ---------------------------------------------------------
        # GEMM accumulation. The dot product works on (M, K) × (K, N)
        # ---------------------------------------------------------
        acc += tl.dot(a, b)          # FP16 * FP16 → FP32 accumulation

    # -------------------------------------------------------------
    # Write the result tile back to memory (convert to FP16)
    # -------------------------------------------------------------
    offs_i = block_m + tl.arange(0, BLOCK_M)
    offs_j = block_n + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N
    out_ptrs = out_ptr + out_batch_off \
               + offs_i[:, None] * stride_o_m \
               + offs_j[None, :] * stride_o_n
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_triton(left: torch.Tensor,
                   right: torch.Tensor,
                   block_m: int = 128,
                   block_n: int = 128,
                   block_k: int = 32) -> torch.Tensor:
    """
    Batched contraction ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
    The implementation reshapes the tensors to a flat batch dimension
    (batch * head) and launches the Triton kernel defined above.
    All tensors are expected to be ``torch.float16`` and laid out
    contiguously in memory with shape ``[B, N, N, H]``.
    """
    B, N, _, H = left.shape
    # Bring the head dimension next to the batch dimension for a fast
    # contiguous layout: (B, H, N, N) → (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_head = B * H
    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)
    out_flat = torch.empty_like(left_flat)   # FP16 output

    # Strides for a contiguous (batch, M, K) tensor
    stride_batch = N * N
    stride_m = N
    stride_k = 1

    grid = (
        batch_head,
        (N + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
    )
    tri_mul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        N, N, N,
        stride_batch, stride_m, stride_k,           # left strides
        stride_batch, stride_m, stride_k,           # right strides (same layout)
        stride_batch, stride_m, stride_k,           # output strides
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    # Reshape back to the original layout
    out_perm = out_flat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or ``None``.
        - weights_dict : mapping with the following keys (all tensors on the same device):
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : must contain ``dim`` (input channel size) and ``hidden_dim``.
    Returns
    -------
    torch.Tensor
        The processed tensor of shape [B, N, N, C] in ``float32``.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to cast a weight to FP16 on the target device
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → FP16 after)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # still float32
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, hidden_dim] – FP16
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction – fully‑fused batched GEMM in Triton
    # ------------------------------------------------------------------
    # ``tri_mul_triton`` returns a tensor of shape [B, N, N, hidden_dim] in FP16.
    out = tri_mul_triton(left, right, block_m=128, block_n=128, block_k=32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------

"""
Outgoing Triangle‑Multiplicative‑Update (TriMul) – fast Triton implementation.

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm the input tensor (float32 → fp16 after the norm).
2️⃣  Two bias‑free linear projections create *left* and *right* tensors of
    shape [B, N, N, H] (H = hidden_dim) in fp16.
3️⃣  Optional pair‑wise mask is broadcast over the hidden dimension.
4️⃣  Gating: sigmoid(linear(x)) is multiplied into *left* and *right*.
5️⃣  Contraction over the third dimension (k):
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched GEMM with a custom Triton kernel that
    performs the N³ mat‑mul for all (b,h) slices in one launch.
6️⃣  Layer‑norm over the hidden dimension (float32⇔float16 conversion).
7️⃣  Multiply by the output gate and apply the final linear projection
    back to the original channel dimension C.
8️⃣  Return a float32 tensor.

The heavy N³ contraction dominates runtime; a hand‑written Triton
kernel (the only Triton code required) gets the same arithmetic
intensity as cuBLAS while keeping the whole operator in a single
PyTorch‑compatible function.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                     # all equal to seq_len = N
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for tensors of shape (B·H, N, N).

    left   : (batch_head, M, K)   – fp16
    right  : (batch_head, N, K)   – fp16  (rows = j, cols = k)
    out    : (batch_head, M, N)   – fp16 (accumulated in fp32)
    """
    bh = tl.program_id(0)          # batch_head (B·H)
    pid_m = tl.program_id(1)       # row‑tile index
    pid_n = tl.program_id(2)       # col‑tile index

    # Starting coordinates of the tile
    block_m = pid_m * BLOCK_M
    block_n = pid_n * BLOCK_N

    # Offsets for the current batch_head slice
    left_batch_off = bh * stride_l_batch
    right_batch_off = bh * stride_r_batch
    out_batch_off = bh * stride_o_batch

    # Accumulator in FP32 (higher precision, same speed on Tensor Cores)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k
    # ------------------------------------------------------------------
    for kk in range(0, K, BLOCK_K):
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- left tile (M × K) -------------------------------------------------
        offs_i = block_m + tl.arange(0, BLOCK_M)
        mask_i = offs_i < M
        left_ptrs = left_ptr + left_batch_off \
                    + offs_i[:, None] * stride_l_m \
                    + offs_k[None, :] * stride_l_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # ----- right tile (K × N) – loaded transposed on‑the‑fly --------------------
        # We need right[j,k]; we store it as (k,j) = (row, col) for the dot.
        offs_j = block_n + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N
        right_ptrs = right_ptr + right_batch_off \
                    + offs_j[None, :] * stride_r_n \
                    + offs_k[:, None] * stride_r_k
        b = tl.load(right_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0,
                    dtype=tl.float16)

        # ----- GEMM accumulation --------------------------------------------------
        acc += tl.dot(a, b)        # fp16 × fp16 → fp32

    # ------------------------------------------------------------------
    # Store the result tile (convert back to fp16)
    # ------------------------------------------------------------------
    offs_i = block_m + tl.arange(0, BLOCK_M)
    offs_j = block_n + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N
    out_ptrs = out_ptr + out_batch_off \
               + offs_i[:, None] * stride_o_m \
               + offs_j[None, :] * stride_o_n
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that reshapes the tensors, launches the kernel and reshapes back.
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor,
                   right: torch.Tensor,
                   block_m: int = 128,
                   block_n: int = 128,
                   block_k: int = 32) -> torch.Tensor:
    """
    Perform the core TriMul contraction using Triton.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (dtype torch.float16). ``H`` is the hidden dimension.
    block_m, block_n, block_k : int
        Tile sizes for the Triton kernel.

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (dtype torch.float16).
    """
    B, N, _, H = left.shape

    # Bring the head dimension next to the batch dimension → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_head = B * H

    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)
    out_flat = torch.empty_like(left_flat)  # fp16 output placeholder

    # Strides for a contiguous (batch, M, K) tensor in element units
    stride_batch = N * N      # distance to the next batch_head slice
    stride_row = N            # distance to the next row
    stride_col = 1            # distance to the next column

    # 3‑D launch grid: (batch_head, row‑tiles, col‑tiles)
    grid = (
        batch_head,
        (N + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
    )
    tri_mul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        N, N, N,                     # M, N, K
        stride_batch, stride_row, stride_col,   # left strides
        stride_batch, stride_row, stride_col,   # right strides (same layout)
        stride_batch, stride_row, stride_col,   # output strides
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # Reshape back to the original layout [B, N, N, H]
    out_perm = out_flat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    return out

# Public entry point required by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3/Chai/Protenix).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or ``None``
        - weights_dict : dictionary containing all linear / norm weights:
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : must contain ``dim`` (input channel size) and
          ``hidden_dim`` (size of the hidden projection).

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, C] in float32.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Utility to move a weight to FP16 on the correct device
    # --------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # Load all model parameters (bias‑free Linear layers → only weights)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (keep in float32 for numerical stability)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )                                    # shape [B, N, N, C] – float32
    x = x_norm.to(torch.float16)         # cast once for the heavy path

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B, N, N, hidden_dim] – fp16
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of a linear projection)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ N³ contraction – batched GEMM implemented in Triton
    # --------------------------------------------------------------
    # Output shape: [B, N, N, hidden_dim] (still fp16)
    out = tri_mul_triton(left, right,
                         block_m=128, block_n=128, block_k=32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm over hidden_dim
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate            # [B, N, N, hidden_dim] – fp16
    final = F.linear(gated, to_out_w)     # [B, N, N, C] – fp16

    # --------------------------------------------------------------
    # 8️⃣ Return tensor in float32 as required by the spec
    # --------------------------------------------------------------
    return final.to(torch.float32)
