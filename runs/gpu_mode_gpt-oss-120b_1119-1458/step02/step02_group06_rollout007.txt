def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on input (float32)
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half
    x_h = x.half()
    B, N, _, _ = x_h.shape
    # Flatten for linear
    x_flat = x_h.view(-1, dim)  # shape (B*N*N, dim)

    # Combine weights
    w_combined = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).half()  # shape (5*hidden_dim, dim)

    # Linear projection (half) -> combined output
    combined = torch.nn.functional.linear(x_flat, w_combined)  # (M,5*H)
    # reshape
    combined = combined.view(B, N, N, 5*hidden_dim)
    # split
    left = combined[..., 0:hidden_dim]
    right = combined[..., hidden_dim:2*hidden_dim]
    left_gate = combined[..., 2*hidden_dim:3*hidden_dim]
    right_gate = combined[..., 3*hidden_dim:4*hidden_dim]
    out_gate = combined[..., 4*hidden_dim:5*hidden_dim]

    # Apply sigmoid gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Apply optional mask (broadcast on hidden dim)
    if not nomask:
        m = mask.unsqueeze(-1).half()
        left = left * m
        right = right * m

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Compute batched Gram (N^3) product
    # Permute to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # view as (B*H, N, N)
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # choose kernel
    use_triton = (BH <= 32) and (N <= 512)
    if use_triton:
        out_flat = torch.empty_like(left_flat)
        stride_a_m, stride_a_n, stride_a_k = left_flat.stride()
        stride_b_m, stride_b_n, stride_b_k = right_flat.stride()
        stride_c_m, stride_c_n, stride_c_k = out_flat.stride()
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
        grid = (BH,
                (N + BLOCK_M - 1) // BLOCK_M,
                (N + BLOCK_N - 1) // BLOCK_N)
        triton_matmul_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            N,
            stride_a_m,
            stride_a_k,
            stride_b_m,
            stride_b_n,
            stride_b_k,
            stride_c_m,
            stride_c_n,
            stride_c_k,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Cast to float32 for rest
    out = out.float()

    # Output LayerNorm + gating
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.float()  # broadcast out_gate over B,N,N,H

    # final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].float())
    return out

"""
Fast outgoing TriMul kernel (AlphaFold3 “TriMul – outgoing”)

Algorithm
---------
1.  Layer‑norm the 4‑D input (FP32) → x.
2.  Cast x to fp16 and flatten to (B·N·N, dim).
3.  Fuse the four linear projections (left‑proj, right‑proj,
    left‑gate, right‑gate) and the output‑gate into a *single* matmul:
       concat_weight = [W_left, W_right, W_lgate, W_rgate, W_ogate] ∈ ℝ^{5·H×dim}
       combined = x_flat @ concat_weightᵀ                          (fp16)
    Split the result into the five tensors and apply sigmoid to the gates.
4.  (Optional) mask the left/right tensors.
5.  Apply element‑wise gate scalings → left, right  ∈ fp16.
6.  Compute the expensive N³ reduction
       out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]
    as a batched matrix‑multiply (B·H, N, N) @ (B·H, N, N)ᵀ.
    For very small batch·hidden (≤32) we launch a custom Triton kernel,
    otherwise we fall back to torch.bmm which uses cuBLAS Tensor‑Cores.
7.  Reshape to (B,N,N,H) and cast to fp32.
8.  Output LayerNorm (fp32) → out_norm.
9.  Multiply by the output‑gate (fp32) and apply the final linear
    projection W_out ∈ ℝ^{dim×H} (fp32) → result (B,N,N,dim).
10. Return the fp32 result.

All heavy matrix‑multiply work is performed in fp16 with fp32 accumulation,
maximising H100 Tensor‑Core utilisation while keeping the final result
in full precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for the N³ reduction.
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_am, stride_ak,          # A: (batch, M, K) – here K == N
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)       # batch * hidden
    pid_i = tl.program_id(1)       # tile row
    pid_j = tl.program_id(2)       # tile col

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduce over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr + pid_m * stride_am
                 + offs_i[:, None] * stride_ak
                 + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load Bᵀ block (BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr + pid_m * stride_bm
                 + offs_k[:, None] * stride_bk
                 + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # FP16 dot → fp32 accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Store result (fp16)
    # ------------------------------------------------------------------
    tl.store(
        c_ptr + pid_m * stride_cm
              + offs_i[:, None] * stride_cn
              + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float)
            - weights: dict of model parameters (see keys below)
            - config: dict containing "dim", "hidden_dim", "nomask"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and flatten for the fused projection
    # ------------------------------------------------------------------
    x_h = x.half()
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                     # (B·N·N, dim)

    # ------------------------------------------------------------------
    # 3. Fuse the four projections + output‑gate into a single matmul
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_ogate  = weights["out_gate.weight"]

    # concatenate along the out_features axis (dim 0)
    w_combined = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_ogate],
        dim=0,
    ).half()                                      # (5·H, dim)

    # one big linear (fp16) → (B·N·N, 5·H)
    combined = F.linear(x_flat, w_combined)       # fp16
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # split the concatenated tensor
    left      = combined[..., 0:hidden_dim]                  # (B,N,N,H)
    right     = combined[..., hidden_dim:2 * hidden_dim]
    left_gate = combined[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate= combined[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate  = combined[..., 4 * hidden_dim:5 * hidden_dim]

    # ------------------------------------------------------------------
    # 4. Gate activations (sigmoid) – stay in fp16
    # ------------------------------------------------------------------
    left_gate  = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate   = out_gate.sigmoid()

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)   # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 6. Apply element‑wise gates (still fp16)
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Prepare tensors for the N³ batched GEMM
    #    (B, N, N, H) → (B·H, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)                # fp16
    right_flat = right.view(BH, N, N)              # fp16

    # ------------------------------------------------------------------
    # 8. Batched Gram product: out_flat[b*h,i,j] = Σ_k left[i,k]·right[j,k]
    # ------------------------------------------------------------------
    use_triton = (BH <= 32) and (N <= 512)   # small‑batch heuristic
    if use_triton:
        out_flat = torch.empty_like(left_flat)
        # strides are in elements (not bytes)
        stride_a_m, stride_a_n, stride_a_k = left_flat.stride()
        stride_b_m, stride_b_n, stride_b_k = right_flat.stride()
        stride_c_m, stride_c_n, stride_c_k = out_flat.stride()

        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )
        triton_matmul_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,               # M (square)
            N,               # N (square)
            stride_a_m,
            stride_a_k,
            stride_b_m,
            stride_b_n,
            stride_b_k,
            stride_c_m,
            stride_c_n,
            stride_c_k,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # cuBLAS batched GEMM (Tensor‑Cores) – highly optimised
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H) and cast to fp32 for the rest
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()                               # (B,N,N,H) fp32

    # ------------------------------------------------------------------
    # 10. Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()                    # broadcast over H

    # ------------------------------------------------------------------
    # 11. Final linear projection back to dimension `dim` (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].float())

    return out
