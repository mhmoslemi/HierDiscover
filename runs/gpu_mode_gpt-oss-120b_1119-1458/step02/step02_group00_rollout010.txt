@triton.jit
def gate_and_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,   # broadcast shape (total_elems/hidden_dim)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + offs, mask=mask, other=1.0) if mask_ptr != 0 else 1.0 # but we can pass a pointer always. 
    left = left * left_gate * mask_val
    right = right * right_gate * mask_val
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
Optimized outgoing Triangle‑Multiplicative update (TriMul) forward pass.

The implementation follows the reference PyTorch module but moves the
dominant N³ “einsum” (`... i k d, ... j k d -> ... i j d`) to a custom
Triton kernel that performs a batched GEMM (A @ Bᵀ) on FP16 data.
All linear projections, gating, the final projection and the intermediate
LayerNorms are kept in FP16 to exploit H100 Tensor‑Cores while the final
output is returned in float32.

Key steps
---------
1️⃣ Layer‑Norm over the last channel (dim) – FP32 for numerical stability.
2️⃣ Cast to fp16, compute left/right projections and the three gates.
3️⃣ (Optional) apply the binary mask.
4️⃣ Fuse left/right with their gates (in‑place element‑wise mul).
5️⃣ Core N³ operation → batched GEMM via `batched_gemm_kernel`.
6️⃣ Post‑GEMM LayerNorm (hidden_dim) – FP16.
7️⃣ Multiply by the output gate (element‑wise).
8️⃣ Final linear projection back to `dim` (FP16 → FP16) and cast to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                # fp16*: left  [batch, M, K]
    b_ptr,                # fp16*: right [batch, N, K] – we will read it as (K,N)
    c_ptr,                # fp16*: out   [batch, M, N]
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, row, col) strides
    stride_bh, stride_bk, stride_bn,   # B (as transposed view): (batch, K, N) strides
    stride_ch, stride_cm, stride_cn,   # C: (batch, row, col) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision matrix multiplication:
        C[b, i, j] = Σₖ A[b, i, k] * B[b, j, k]   (i.e. A @ Bᵀ)
    The kernel is launched with a 3‑D grid:
        (grid_m, grid_n, batch)  →  program_id(0/1/2)
    """
    batch_id = tl.program_id(2)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col

    # ------ tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------ iterate over K dimension ------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile (transposed view): shape (BLOCK_K, BLOCK_N)
        #   Bᵀ[k, n] == B[n, k] in the original right matrix.
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # ------ write back ---------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim]   (float32)
        - mask          : torch.Tensor      [B, N, N]        (float32 / bool)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict   contains 'dim', 'hidden_dim', optional
                         'nomask' (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Load weights (cast to fp16)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, w_right_proj)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates (in‑place)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    # tensors are now ready for the N³ contraction

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation – batched GEMM (A @ Bᵀ) in Triton
    # ------------------------------------------------------------------
    # reshape to (batch * hidden_dim, N, N) so each head is an independent GEMM
    total_heads = B * hidden_dim
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16 output buffer

    # strides for the contiguous tensors
    stride_ah, stride_am, stride_ak = left_mat.stride()          # (batch, row, col)
    stride_bh, stride_bn, stride_bk = right_mat.stride()          # note: bn = stride for N, bk = stride for K
    stride_ch, stride_cm, stride_cn = out_mat.stride()

    # Tunable block sizes – empirically good for 80 GB H100 cards
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over M (i dimension)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over N (j dimension)
        total_heads,                    # one program per head
    )

    batched_gemm_kernel[grid](
        a_ptr=left_mat.data_ptr(),
        b_ptr=right_mat.data_ptr(),
        c_ptr=out_mat.data_ptr(),
        M=N,
        N=N,
        K=N,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,   # original col‑stride (==1)
        stride_bn=stride_bn,   # original row‑stride (==N)
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 🔟 Multiply by the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 🟢 Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)   # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # Return in the requested float32 format
    # ------------------------------------------------------------------
    return out.to(torch.float32)
