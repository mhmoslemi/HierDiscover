Implementation details:

We will produce a new kernel:

```python
@triton.jit
def _gate_mask_kernel_dual(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    hidden_dim: tl.constexpr,
    n_elts: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elts

    # Load the projections and gates
    left_proj = tl.load(left_proj_ptr + offs, mask=mask).to(tl.float32)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask).to(tl.float32)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask).to(tl.float32)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask).to(tl.float32)

    # Broadcast mask across hidden_dim
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    left_val = left_proj * left_gate * mask_val
    right_val = right_proj * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_val.to(tl.float16), mask=mask)
    tl.store(right_out_ptr + offs, right_val.to(tl.float16), mask=mask)
```

"""
Highly‑optimised outgoing Triangle‑Multiplicative Update (TriMul) forward
pass.  

Algorithm
---------
1️⃣  Layer‑Norm over the feature dimension (float32 for numerical safety).  
2️⃣  Cast to FP16 and apply the four linear projections (left / right) and the
    three gating projections.  Sigmoid is applied to the gating outputs.  
3️⃣  Fuse the element‑wise product *projection × gate × (optional) mask* for
    **both** left and right tensors in a single Triton kernel.  This halves
    the memory traffic and avoids two kernel launches.  
4️⃣  Core O(N³) operation: for each hidden channel perform a matrix
    multiplication `C = A @ Bᵀ`.  By reshaping the tensors we obtain a batch
    of `B·H` independent `N×N` GEMMs, which are executed by `torch.bmm`
    (FP16 × FP16 → FP32 accumulation) and run at full Tensor‑Core speed on
    an H100.  The right matrices are pre‑transposed, so no extra transpose
    is needed inside the GEMM.  
5️⃣  Reshape back to `[B, N, N, H]`, apply a second Layer‑Norm, multiply by
    the output gate and a final linear projection back to the original
    feature dimension.  The final result is returned as `float32`.  

At least one operation (the gating + mask fusion) is performed with a
Triton kernel, satisfying the problem constraints while keeping the
overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_dual_kernel(
    left_proj_ptr,      # fp16 left projection   (B·N·N·H)
    right_proj_ptr,     # fp16 right projection  (B·N·N·H)
    left_gate_ptr,      # fp16 left gate         (B·N·N·H)
    right_gate_ptr,     # fp16 right gate        (B·N·N·H)
    mask_ptr,           # fp16 mask (flattened) (B·N·N)
    left_out_ptr,       # fp16 left output
    right_out_ptr,      # fp16 right output
    hidden_dim: tl.constexpr,
    n_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """Apply (proj * gate * mask) for left and right tensors in one launch."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load projection and gate values (convert to fp32 for a more accurate mul)
    left_proj = tl.load(left_proj_ptr + offs, mask=mask).to(tl.float32)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask).to(tl.float32)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask).to(tl.float32)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask).to(tl.float32)

    # The mask is broadcast over the hidden dimension.
    mask_idx = offs // hidden_dim                     # integer division
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    left_val = left_proj * left_gate * mask_val
    right_val = right_proj * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_val.to(tl.float16), mask=mask)
    tl.store(right_out_ptr + offs, right_val.to(tl.float16), mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : Tensor[B, N, N, dim]   (float32)
        - mask         : Tensor[B, N, N] (float32/float16/bool) or None
        - weights      : dict with all layer‑norm and linear weights
        - config       : dict containing at least "dim" and "hidden_dim",
                         optionally "nomask" (bool)

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # when True we skip the mask step

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------ #
    # 2️⃣ Cast to FP16 and compute linear projections & gating tensors
    # ------------------------------------------------------------------ #
    x_h = x.to(torch.float16)   # [B,N,N,dim] fp16

    # Linear weights – keep them fp16 to stay in the fast pathway
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # Projections
    left_proj  = F.linear(x_h, left_proj_w)    # [B,N,N,hidden_dim] fp16
    right_proj = F.linear(x_h, right_proj_w)   # [B,N,N,hidden_dim] fp16

    # Gates (sigmoid, keep fp16)
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w)).to(torch.float32)

    # ------------------------------------------------------------------ #
    # 3️⃣ Fuse (proj * gate * mask) for left and right tensors
    # ------------------------------------------------------------------ #
    if not nomask and mask is not None:
        # Convert mask to fp16 and flatten (shape B·N·N)
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                     # [B·N·N]

        # Allocate output buffers
        left  = torch.empty_like(left_proj)   # fp16
        right = torch.empty_like(right_proj)  # fp16

        # Kernel launch configuration
        BLOCK = 65536
        n_elem = left_proj.numel()                       # B·N·N·H
        grid = (triton.cdiv(n_elem, BLOCK),)

        # One launch handles both left and right tensors
        _gate_mask_dual_kernel[grid](
            left_proj,            # proj left
            right_proj,           # proj right
            left_gate,            # gate left
            right_gate,           # gate right
            mask_flat,            # mask (broadcast)
            left,                 # out left
            right,                # out right
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask → simple element‑wise product (still fp16)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------ #
    # 4️⃣ Core O(N³) operation: batch GEMM A @ Bᵀ
    # ------------------------------------------------------------------ #
    B, N, _, _ = left.shape

    # Reshape to (B*H, N, N).  Left stays as (i,k), right is pre‑transposed to (k,j)
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication in FP16 (Tensor‑Core GEMM, FP32 accu)
    out_batched = torch.bmm(left_batched, right_batched)  # [B*H, N, N] fp16

    # Restore original layout: [B, N, N, hidden_dim] (float32 for the next layers)
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)                     # B, N, N, H
        .contiguous()
        .to(torch.float32)
    )   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------ #
    # 5️⃣ Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear proj
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B,N,N,hidden_dim] (float32)

    out = out * out_gate   # broadcast over hidden_dim

    # Final projection back to original feature dim (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
