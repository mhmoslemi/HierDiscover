import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def apply_gates_kernel(left_ptr, left_gate_ptr, right_ptr, right_gate_ptr, numel: tl.int64, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)

@triton.jit
def mul_inplace_kernel(a_ptr, b_ptr, numel: tl.int64, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    The kernel consists of:
    1) Input LayerNorm (FP32)
    2) FP16 casts of inputs and all linear weights.
    3) Linear projections (left/right) and gate projections (left_gate, right_gate, out_gate).
    4) Optional pairwise mask applied to left/right.
    5) Triton kernel ``apply_gates_kernel`` fusing left/right tensor
       element‑wise multiplication with their respective gates (in‑place).
    6) Core N³ interaction computed as a batched matrix multiplication:
       ``out[b, d] = left[b, d] @ right[b, d].T``.
    7) Post‑multiplicative LayerNorm (FP16).
    8) Triton kernel ``mul_inplace_kernel`` multiplies the normalized tensor
       by the output gate (in‑place).
    9) Final Linear projection back to ``dim`` (FP16) and cast to FP32.

    All heavy arithmetic runs in FP16 to exploit H100 Tensor‑Cores.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim], fp32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for dense work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Load weights in fp16
    # --------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right_proj)  # same shape

    # --------------------------------------------------------------
    # 5️⃣ Gate pre‑activations (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # --------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] then broadcast
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates (in‑place) via Triton
    # --------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    numel = left.numel()
    BLOCK_SIZE = 65536  # multiples of 128 for good alignment
    grid = ((numel + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # gates are no longer needed
    del left_gate, right_gate

    # --------------------------------------------------------------
    # 8️⃣ Core N³ operation: batched matrix multiplication.
    #    Each hidden channel is treated as an independent head.
    # --------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    batch_heads = B * H

    # shape to (batch_heads, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(batch_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch_heads, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T   -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # --------------------------------------------------------------
    # 9️⃣ Post‑norm (fp16 weights)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # --------------------------------------------------------------
    # 🔟 Multiply by output gate (in‑place) with Triton
    # --------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # out_gate no longer needed
    del out_gate

    # --------------------------------------------------------------
    # 11️⃣ Final linear projection to original dim
    # --------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim], fp16

    # Return as float32 (as in reference impl)
    return final.to(torch.float32)

"""
Optimized outgoing TriMul forward pass (AlphaFold3 style).

The implementation follows the reference PyTorch module but moves the
computationally dominant N³ “einsum” to a highly‑optimized batched matrix
multiplication (torch.bmm) that executes in FP16 on H100 Tensor‑Cores.
Two tiny Triton kernels are used:

* ``apply_gates_kernel``  – fuses the element‑wise multiplication of the
  left/right tensors with their respective gates (in‑place).
* ``mul_inplace_kernel``   – multiplies the post‑norm tensor by the output
  gate (in‑place).

All heavy arithmetic runs in FP16; LayerNorms are kept in FP32 (or FP16
where safe) for numerical stability.  The final output is returned in
float32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr,        # fp16*, shape (total_elements,)
    left_gate_ptr,   # fp16*, same shape as left_ptr
    right_ptr,       # fp16*, shape (total_elements,)
    right_gate_ptr,  # fp16*, same shape as right_ptr
    numel: tl.int64,          # total number of elements per tensor
    BLOCK: tl.constexpr,      # block size (tiles per program)
):
    """
    In‑place multiplication:
        left  = left  * left_gate
        right = right * right_gate
    The tensors are flattened beforehand.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # left side
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    # right side
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def mul_inplace_kernel(
    a_ptr,         # fp16*, output tensor (will be overwritten)
    b_ptr,         # fp16*, multiplication operand (same shape as a_ptr)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication:
        a = a * b
    Both tensors are flattened.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim]    (float32)
        - mask         : torch.Tensor      [B, N, N]         (bool/float)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   must contain 'dim', 'hidden_dim' and may contain
                           'nomask' (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for dense work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Convert weights to half‑precision
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, w_right_proj)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Gate pre‑activations (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Fuse left/right with their gates using a small Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536  # tuned block size (~64 k elements)
    grid = ((numel + BLOCK - 1) // BLOCK,)

    apply_gates_kernel[grid](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Gates no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Core N³ operation: batched matrix multiplication (A @ Bᵀ)
    #    Each hidden channel is treated as an independent “head”.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape            # H == hidden_dim
    total_heads = B * H

    # reshape to (total_heads, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # (total_heads, N, N) @ (total_heads, N, N)ᵀ → (total_heads, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original layout: [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Post‑processing LayerNorm (half‑precision weights)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 🔟 Multiply by the output gate (in‑place) via Triton
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)

    mul_inplace_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    del out_gate  # free memory

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(out, w_to_out)   # [B, N, N, dim], fp16

    # Return in expected float32 precision
    return final.to(torch.float32)
