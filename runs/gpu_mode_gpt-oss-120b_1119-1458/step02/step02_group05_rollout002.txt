"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

Algorithm
---------
1️⃣  Layer‑normalize the input (float32) over the feature dimension `dim`.
2️⃣  Cast to FP16 for all matrix‑multiplication work.
3️⃣  Fuse the two linear projections **and** their gates per side by packing
    the weights (`proj` + `gate`) into a single matrix.  This halves the number
    of passes over the massive `[B, N, N, dim]` tensor.
4️⃣  Apply the sigmoid gate and (optional) mask in one Triton kernel that
    computes `proj * gate * mask` (mask is broadcast across the hidden dim).
5️⃣  Core O(N³) update: for each hidden channel perform a matrix–multiply
    `C = A @ Bᵀ`.  We reshape the tensors to `(B*hidden, N, N)` and use the
    highly‑optimized `torch.bmm` (Tensor‑Core GEMM, FP16 × FP16 → FP32).
6️⃣  Layer‑normalize the hidden result, multiply by the `out_gate`,
    and finally project back to `dim` with a linear layer.
   All heavy arithmetic stays in FP16/F32, while the final tensor is returned
   as float32 to match the reference implementation.

A Triton kernel is used for the fused element‑wise `proj * gate * mask`
operation (step 4); the remaining arithmetic leverages cuBLAS/Tensor‑Core
kernels via PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16*  projection (left or right)
    gate_ptr,      # fp16*  gate (sigmoid output)
    mask_ptr,      # fp16*  mask flattened, length = B * N * N
    out_ptr,       # fp16*  destination
    hidden_dim,    # i64    hidden feature size
    n_elements,    # i64    total number of proj/gate elements = B*N*N*hidden
    BLOCK: tl.constexpr,  # compile‑time block size
):
    """Element‑wise multiplication proj * gate * mask (mask broadcasted)."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # load projection & gate (convert to fp32 for precise mul)
    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # map the linear index to mask index (mask repeats hidden_dim times)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – all heavy ops will use half‑precision
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate linear layers (per side)
    # ------------------------------------------------------------------
    #   left: proj + gate → 2*hidden output
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden, dim)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden, dim)

    # Compute combined linear maps (no bias)
    left_comb = F.linear(x_h, left_comb_weight)   # [B,N,N,2*hidden]
    right_comb = F.linear(x_h, right_comb_weight)  # [B,N,N,2*hidden]

    # Split back into proj / gate
    left_proj = left_comb[..., :hidden_dim]            # fp16
    left_gate_raw = left_comb[..., hidden_dim:]        # fp16
    right_proj = right_comb[..., :hidden_dim]          # fp16
    right_gate_raw = right_comb[..., hidden_dim:]      # fp16

    # Apply sigmoid to gate values
    left_gate = torch.sigmoid(left_gate_raw)   # fp16
    right_gate = torch.sigmoid(right_gate_raw) # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask and gating with a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # Prepare mask: [B, N, N, 1] -> flatten to [B*N*N] (fp16)
        mask_f = mask_tensor.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)  # length B*N*N, dtype fp16

        # Allocate fused tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Triton launch configuration
        BLOCK = 65536  # can be tuned; 64 KB of work per program
        n_elem = left_proj.numel()                     # B*N*N*hidden
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask – simple element‑wise product (still half‑precision)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³) update: batched GEMM (A @ Bᵀ) over each hidden channel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden, N, N) – required shape for torch.bmm
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Tensor‑Core GEMM (FP16 x FP16 → FP32 accumulation)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))

    # Back to [B, N, N, hidden] and promote to FP32 for the following LayerNorm
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    # Hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,hidden] (float32)

    # Compute out_gate (sigmoid, keep FP32 for numeric stability)
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16))).to(torch.float32)
    out = out * out_gate  # broadcast over hidden dim

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim] (float32)

    return out
