"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input (float32, stable).
2️⃣ Cast to FP16 and perform **one** large linear projection that concatenates the
   five weight matrices (left_proj, right_proj, left_gate, right_gate, out_gate).
   This reduces the number of passes over the huge input tensor [B,N,N,dim].
3️⃣ Slice the concatenated tensor, apply sigmoid to the gate portions.
4️⃣ Fuse element‑wise multiplication of projection × gate × (optional) mask with a
   tiny Triton kernel, saving two extra memory passes when a mask is present.
5️⃣ Core O(N³) computation: batch GEMM `C = A @ Bᵀ` where `A,B ∈ ℝ^{B·H×N×N}`
   (H = hidden_dim).  The GEMM runs in mixed‑precision (FP16×FP16→FP32) using
   :func:`torch.bmm`, which on an H100 exploits Tensor‑Core acceleration.
6️⃣ Layer‑norm over the hidden dimension, multiply by the output gate, and project
   back to the original feature dimension.
All heavy arithmetic stays in FP16 (except the two LayerNorms and the final
output‑gate multiplication), and the final result is returned as float32.

The implementation satisfies the “use Triton” requirement via the mask‑gate
kernel while keeping the overall runtime well below the 1000 µs target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) for the masked case.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,                     # fp16*  projection (left or right)
    gate_ptr,                     # fp16*  corresponding gate
    mask_ptr,                     # fp16*  mask flattened (B·N·N)
    out_ptr,                      # fp16*  destination
    hidden_dim,                   # i64   hidden dimension (for broadcasting mask)
    n_elem,                       # i64   total number of elements (B·N·N·H)
    BLOCK: tl.constexpr,          # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < n_elem

    # Load projection and gate as FP16, promote to FP32 for the multiplication.
    proj = tl.load(proj_ptr + offs, mask=valid).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=valid).to(tl.float32)

    # Each mask element is repeated `hidden_dim` times in the flattened view.
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid).to(tl.float32)

    out = proj * gate * mask_val      # FP32 intermediate
    tl.store(out_ptr + offs, out.to(tl.float16), mask=valid)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float)
        - weights      : dict containing all model parameters (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optionally
                         ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32) matching the reference PyTorch
        implementation.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (dim) – keep in FP32.
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy linear/gate work.
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Combine all five linear weight matrices into one big matrix.
    #    This reduces the number of passes over the huge tensor x_h.
    # ------------------------------------------------------------------
    # Expected shapes: each weight is (hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*hidden_dim, dim)

    # Single linear pass: result shape [B, N, N, 5*hidden_dim] (FP16)
    combined = F.linear(x_h, combined_weight)

    # ------------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor into the five components.
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = combined[..., 0 : H]
    right_proj  = combined[..., H : 2 * H]
    left_gate_p = combined[..., 2 * H : 3 * H]
    right_gate_p = combined[..., 3 * H : 4 * H]
    out_gate_p  = combined[..., 4 * H : 5 * H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid to gates.  Keep left/right gates in FP16 for speed;
    #    output gate is cast to FP32 because it later multiplies a FP32 tensor.
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)          # FP16
    right_gate = torch.sigmoid(right_gate_p)         # FP16
    out_gate   = torch.sigmoid(out_gate_p).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse proj × gate × mask (if present) using Triton.
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] -> FP16 -> flatten
        mask_f    = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)  # (B·N·N,)

        # Allocate result tensors
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Triton launch configuration
        BLOCK = 65536                     # tuneable compile‑time block size
        n_elem = left_proj.numel()        # B·N·N·H
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask – simple element‑wise product (FP16)
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core O(N³) operation: batched GEMM  (A @ Bᵀ) with Tensor‑Core speed.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Reshape to (B*H, N, N) – treat the hidden dimension as a batch axis.
    left_batched  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm does FP16×FP16→FP32 accumulation automatically.
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))  # (B*H, N, N), FP16

    # Re‑shape back to [B, N, N, H] and promote to FP32 for the following LayerNorm.
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing: hidden‑dim LayerNorm, output gate, final projection.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    out = out * out_gate                     # broadcast multiplication (FP32)

    # Final linear projection back to `dim` (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
