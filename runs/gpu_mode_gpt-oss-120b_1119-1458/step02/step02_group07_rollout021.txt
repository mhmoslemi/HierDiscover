@triton.jit
def _gate_mask_fuse_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    mask_ptr,
    stride_lp0, stride_lp1, stride_lp2, stride_lp3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rp0, stride_rp1, stride_rp2, stride_rp3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_out0, stride_out1, stride_out2, stride_out3,
    stride_m0, stride_m1, stride_m2,
    B, N, H,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N
    off_lp = b*stride_lp0 + i*stride_lp1 + j*stride_lp2
    off_lg = b*stride_lg0 + i*stride_lg1 + j*stride_lg2
    off_rp = b*stride_rp0 + i*stride_rp1 + j*stride_rp2
    off_rg = b*stride_rg0 + i*stride_rg1 + j*stride_rg2
    off_m = b*stride_m0 + i*stride_m1 + j*stride_m2

    mask_val = tl.load(mask_ptr + off_m)
    
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask = offs < H
        left_proj = tl.load(left_proj_ptr + off_lp + offs*stride_lp3, mask=mask)
        left_gate = tl.load(left_gate_ptr + off_lg + offs*stride_lg3, mask=mask)
        right_proj = tl.load(right_proj_ptr + off_rp + offs*stride_rp3, mask=mask)
        right_gate = tl.load(right_gate_ptr + off_rg + offs*stride_rg3, mask=mask)

        # sigmoid
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

        left_out = left_proj * left_gate_sig * mask_val
        right_out = right_proj * right_gate_sig * mask_val

        tl.store(out_left_ptr + off_lp + offs*stride_lp3, left_out, mask=mask) # in-place
        tl.store(out_right_ptr + off_rp + offs*stride_rp3, right_out, mask=mask)
```

```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _gate_mask_fuse_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    # strides
    stride_lp0, stride_lp1, stride_lp2, stride_lp3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rp0, stride_rp1, stride_rp2, stride_rp3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_m0, stride_m1, stride_m2,
    B, N, H,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    # decode b,i,j
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets
    off_lp = b * stride_lp0 + i * stride_lp1 + j * stride_lp2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rp = b * stride_rp0 + i * stride_rp1 + j * stride_rp2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    mask_val = tl.load(mask_ptr + off_m)

    # iterate hidden dim
    for start in range(0, H, BLOCK_H):
        cur_off = start + tl.arange(0, BLOCK_H)
        active = cur_off < H

        left = tl.load(left_proj_ptr + off_lp + cur_off * stride_lp3,
                       mask=active, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_lg + cur_off * stride_lg3,
                            mask=active, other=tl.float16(0.0))
        right = tl.load(right_proj_ptr + off_rp + cur_off * stride_rp3,
                        mask=active, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_rg + cur_off * stride_rg3,
                             mask=active, other=tl.float16(0.0))

        # sigmoid via 1/(1+exp(-x))
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_proj_ptr + off_lp + cur_off * stride_lp3, left_out, mask=active)
        tl.store(right_proj_ptr + off_rp + cur_off * stride_rp3, right_out, mask=active)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.
    The implementation fuses the five linear transformations into a single
    batched GEMM, applies the gating and optional mask via a Triton kernel,
    performs the N³ pairwise multiplication as a batched GEMM, and finishes
    with a second LayerNorm, an output gate, and a final linear projection.
    All heavy‑compute kernels run in fp16; the final result is returned in fp32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # 1️⃣ LayerNorm over channel dimension (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]  float32

    # 2️⃣ Cast to fp16 for the main linear work
    x_fp16 = x.to(torch.float16)

    # 3️⃣ Fuse the five linear maps into a single GEMM
    #   combined weight shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    #   B * N * N is the “M” dimension for the GEMM
    combined = F.linear(x_fp16, combined_weight)  # [B,N,N,5*hidden_dim] fp16

    # 4️⃣ Slice the combined tensor
    H = hidden_dim
    left_proj = combined[..., :H]               # [B,N,N,H]
    right_proj = combined[..., H : 2 * H]       # [B,N,N,H]
    left_gate = combined[..., 2 * H : 3 * H]    # [B,N,N,H]
    right_gate = combined[..., 3 * H : 4 * H]   # [B,N,N,H]
    out_gate = combined[..., 4 * H :]           # [B,N,N,H] (pre‑sigmoid)

    # 5️⃣ Apply sigmoid to gate tensors and optional mask via Triton
    #    We handle both the masked and un‑masked path with the same kernel.
    if nomask:
        # create a mask of ones (fp16) to reuse the same kernel – negligible cost
        mask_fp16 = torch.ones_like(mask, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    # Ensure all tensors are contiguous (required for correct stride handling)
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp16 = mask_fp16.contiguous()

    total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]
    # launch kernel: computes left = left_proj * sigmoid(left_gate) * mask
    #              and right = right_proj * sigmoid(right_gate) * mask
    _gate_mask_fuse_kernel[
        (total_programs,)
    ](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_fp16,
        # strides
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        input_tensor.shape[0],
        input_tensor.shape[1],
        hidden_dim,
        BLOCK_H=32,
        num_warps=4,
    )

    # 6️⃣ Compute sigmoid for the output gate (outside the kernel, cheap)
    out_gate = torch.sigmoid(out_gate)  # fp16

    # 7️⃣ Core N³ multiplication via batched GEMM (Tensor‑Core accelerated)
    B, N, _, _ = left_proj.shape
    # reshape to (B*H, N, N) -> treat each head as an independent batch
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to [B, N, N, H] (fp16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 8️⃣ Second LayerNorm (float32) → output gate → final linear
    out_f32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate.to(torch.float32)  # apply output gate

    # 9️⃣ Final projection back to original feature dim
    out_final = F.linear(out_norm, weights["to_out.weight"])  # shape [B,N,N,dim] float32
    return out_final
```

"""
Optimised **outgoing** Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input (float32).
2️⃣ Cast to fp16 and compute *all five* linear maps (left‑proj, right‑proj,
   left‑gate, right‑gate, out‑gate) with a single GEMM – this eliminates five
   redundant reads of the large [B,N,N,dim] tensor.
3️⃣ A custom Triton kernel fuses:
      – sigmoid on the gate tensors,
      – optional mask multiplication,
      – writes the gated, masked *left* and *right* tensors back in‑place,
      – sigmoid on the output‑gate (no mask needed).
   The kernel works for both masked and un‑masked cases (mask = 1 when it is
   absent) and processes the hidden dimension in 32‑element tiles.
4️⃣ Core N³ operation Σₖ left[...,i,k,d]·right[...,j,k,d] is carried out as a
   batched half‑precision GEMM (Tensor‑Core accelerated):
        C[d] = left[d] @ right[d]ᵀ
   for every hidden channel *d*.
5️⃣ Convert the result to fp32, apply a second LayerNorm, multiply by the
   (sigmoid‑activated) out‑gate and finish with the final linear projection
   back to the original channel dimension.
All heavy arithmetic is performed in fp16; the final tensor is returned as
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_fuse_kernel(
    # tensors
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_ptr, mask_ptr,
    # strides for left projection
    stride_lp0, stride_lp1, stride_lp2, stride_lp3,
    # strides for left gate
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    # strides for right projection
    stride_rp0, stride_rp1, stride_rp2, stride_rp3,
    # strides for right gate
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    # strides for out gate (pre‑sigmoid)
    stride_og0, stride_og1, stride_og2, stride_og3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    # sizes
    B, N, H,
    BLOCK_H: tl.constexpr,        # tile size along the hidden dimension
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets (no hidden‑dim offset yet)
    off_lp = b * stride_lp0 + i * stride_lp1 + j * stride_lp2
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2
    off_rp = b * stride_rp0 + i * stride_rp1 + j * stride_rp2
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2
    off_og = b * stride_og0 + i * stride_og1 + j * stride_og2
    off_m  = b * stride_m0  + i * stride_m1  + j * stride_m2

    mask_val = tl.load(mask_ptr + off_m)       # fp16 scalar (1.0 when not masked)

    # loop over the hidden dimension in chunks of BLOCK_H
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        active = offs < H                        # mask for tail

        # ----- load -----
        left      = tl.load(left_proj_ptr + off_lp + offs * stride_lp3,
                            mask=active, other=tl.float16(0.0))
        left_g    = tl.load(left_gate_ptr + off_lg + offs * stride_lg3,
                            mask=active, other=tl.float16(0.0))
        right     = tl.load(right_proj_ptr + off_rp + offs * stride_rp3,
                            mask=active, other=tl.float16(0.0))
        right_g   = tl.load(right_gate_ptr + off_rg + offs * stride_rg3,
                            mask=active, other=tl.float16(0.0))
        out_g_pre = tl.load(out_gate_ptr  + off_og + offs * stride_og3,
                            mask=active, other=tl.float16(0.0))

        # ----- sigmoid (in fp16) -----
        left_g  = 1.0 / (1.0 + tl.exp(-left_g))
        right_g = 1.0 / (1.0 + tl.exp(-right_g))
        out_g   = 1.0 / (1.0 + tl.exp(-out_g_pre))

        # ----- apply mask and gates -----
        left_out  = left  * left_g  * mask_val
        right_out = right * right_g * mask_val

        # ----- store results back in‑place -----
        tl.store(left_proj_ptr  + off_lp + offs * stride_lp3,
                 left_out,  mask=active)
        tl.store(right_proj_ptr + off_rp + offs * stride_rp3,
                 right_out, mask=active)
        tl.store(out_gate_ptr   + off_og + offs * stride_og3,
                 out_g,    mask=active)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask:         [B, N, N]       (float32, optional)
            - weights: dict of model parameters (float32)
            - config: dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and compute all five linear maps in ONE GEMM
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*H, dim)

    # shape -> [B, N, N, 5*hidden_dim]  (fp16)
    combined = F.linear(x_fp16, combined_weight)

    H = hidden_dim
    left_proj  = combined[..., :H]                 # [B, N, N, H]
    right_proj = combined[..., H : 2 * H]          # [B, N, N, H]
    left_gate  = combined[..., 2 * H : 3 * H]      # [B, N, N, H]
    right_gate = combined[..., 3 * H : 4 * H]      # [B, N, N, H]
    out_gate   = combined[..., 4 * H :]            # [B, N, N, H] (pre‑sigmoid)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse gate, mask and out‑gate sigmoid via a Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        # a tensor of ones – avoids branching in the kernel
        mask_fp16 = torch.ones_like(mask, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    # ensure contiguous layout for stride‑correctness
    left_proj   = left_proj.contiguous()
    right_proj  = right_proj.contiguous()
    left_gate   = left_gate.contiguous()
    right_gate  = right_gate.contiguous()
    out_gate    = out_gate.contiguous()
    mask_fp16   = mask_fp16.contiguous()

    total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]

    _gate_mask_fuse_kernel[
        (total_programs,)
    ](
        left_proj, left_gate,
        right_proj, right_gate,
        out_gate, mask_fp16,
        # strides for left projection
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        # strides for left gate
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        # strides for right projection
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        # strides for right gate
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        # strides for out gate (pre‑sigmoid)
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        # strides for mask
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        # sizes
        input_tensor.shape[0],      # B
        input_tensor.shape[1],      # N
        hidden_dim,                 # H
        BLOCK_H=32,
        num_warps=4,
    )
    # after the kernel:
    #   left_proj  = gated, masked left  (fp16)
    #   right_proj = gated, masked right (fp16)
    #   out_gate   = sigmoid(out_gate) (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ update – batched GEMM (half‑precision, Tensor‑Core)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    # reshape to (B*H, N, N) where each “batch” corresponds to one hidden channel
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_mat[i] = left_mat[i] @ right_mat[i].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] (fp16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Post‑processing: LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # out_gate is still fp16 → cast to fp32 for the multiplication
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B, N, N, dim] float32

    return out_final
