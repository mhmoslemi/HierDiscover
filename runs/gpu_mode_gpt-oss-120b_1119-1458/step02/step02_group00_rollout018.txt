custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm:
    x = F.layer_norm(input_tensor, (dim,), weights["norm.weight"], weights["norm.bias"], eps=1e-5)

    # cast to fp16
    x_h = x.to(torch.float16)

    # combine weights for left/right projections:
    w_left_right = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"]], dim=0).to(torch.float16)  # shape (2*hidden_dim, dim)
    # linear projection
    proj = F.linear(x_h, w_left_right)   # shape (B,N,N,2H)
    left = proj[..., :hidden_dim]
    right = proj[..., hidden_dim:]

    # combine gate weights
    w_gate = torch.cat([weights["left_gate.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    gates = torch.sigmoid(F.linear(x_h, w_gate))
    left_gate = gates[..., :hidden_dim]
    right_gate = gates[..., hidden_dim:]

    # optionally mask
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # fuse left/right with gates
    left = left * left_gate
    right = right * right_gate

    # core multiplication using tri_mul_kernel
    out = tri_mul(left, right)  # shape (B,N,N,H)

    # out_norm
    out = F.layer_norm(out, (hidden_dim,), weights["to_out_norm.weight"].to(torch.float16), weights["to_out_norm.bias"].to(torch.float16), eps=1e-5)

    # out_gate
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    # multiply out by out_gate
    out = out * out_gate

    # final out projection
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast to float32
    return out.to(torch.float32)

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) for AlphaFold3.

The implementation follows the reference forward pass but moves the
dominant N³ contraction to a fused Triton kernel that directly computes
the batched matrix multiplication

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

without any intermediate permutations or copies.  Linear projections,
gates and the final projection are kept in fp16 (Tensor‑Core friendly)
while LayerNorms use fp16 weights for speed.  Two tiny element‑wise
kernels fuse mask‑gate multiplication and the final gate application.

The overall data flow (shapes shown) is:

    x (fp32) ──► LayerNorm (fp32) ──► cast fp16
    ├─► left+right projections (fp16) –‑► gate projections (fp16) –‑► apply gates (Triton)
    │
    └─► right‑hand side (fp16) –‑► gate projections (fp16) –‑► apply gates (Triton)

    left, right ──► fused Triton `tri_mul_kernel` (batched GEMM) ──► out (fp16)

    out ──► LayerNorm (fp16) ──► out_gate (fp16) ──► element‑wise mul (Triton)
    ──► final linear (fp16) ──► cast fp32 → result [B, N, N, dim]
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny element‑wise Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, left_gate_ptr,      # fp16, shape (total_elems,)
    right_ptr, right_gate_ptr,    # fp16, shape (total_elems,)
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place multiplication of left/right with their gates."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    l  = tl.load(left_ptr   + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(left_ptr + offs, l * lg, mask=mask)

    r  = tl.load(right_ptr   + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    tl.store(right_ptr + offs, r * rg, mask=mask)


@triton.jit
def mul_out_gate_kernel(
    out_ptr, gate_ptr, out_mul_ptr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """out_mul = out * gate (both fp16, flattened)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    o = tl.load(out_ptr   + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr  + offs, mask=mask, other=0.0)
    tl.store(out_mul_ptr + offs, o * g, mask=mask)


# ----------------------------------------------------------------------
# Core N³ Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,           # fp16 tensors
    B, N, H,                               # sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    All tensors are stored as 4‑D contiguous slabs (B,N,N,H).
    """
    pid_batch = tl.program_id(0)      # index over (batch * head)
    pid_i = tl.program_id(1)          # tile over output dimension i
    pid_j = tl.program_id(2)          # tile over output dimension j

    # recover batch and head indices
    b = pid_batch // H
    h = pid_batch %  H

    # offsets inside the i / j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base offsets that include batch & head
    base_left  = b * stride_left_b  + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out   = b * stride_out_b   + h * stride_out_h

    # accumulator (FP32 for increased precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask each axis
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # left[i,k] tile
        a_ptrs = left_ptr + base_left \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[j,k] tile  (note the transpose: k is the first dim)
        b_ptrs = right_ptr + base_right \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate dot product (FP32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # write result (down‑cast to fp16)
    c = acc.to(tl.float16)
    out_ptrs = out_ptr + base_out \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask         : torch.Tensor  [B, N, N]    (float32/bool)
        - weights      : dict of torch.Tensor   model parameters
        - config       : dict   contains 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # fp32
    x = x.to(torch.float16)  # fp16 for the heavy work

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections (single linear)
    # ------------------------------------------------------------------
    w_proj_cat = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                     # (2·hidden_dim, dim)
    proj = F.linear(x, w_proj_cat)          # (B, N, N, 2·hidden_dim)
    left, right = proj[..., :hidden_dim], proj[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right gates (single linear + sigmoid)
    # ------------------------------------------------------------------
    w_gate_cat = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2·hidden_dim, dim)
    gates = torch.sigmoid(F.linear(x, w_gate_cat))
    left_gate, right_gate = gates[..., :hidden_dim], gates[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Fuse left/right with corresponding gates (tiny Triton kernel)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()
    BLOCK = 65536
    grid_gate = ((numel + BLOCK - 1) // BLOCK,)
    apply_gates_kernel[grid_gate](
        left.view(-1),
        left_gate.view(-1),
        right.view(-1),
        right_gate.view(-1),
        numel,
        BLOCK=BLOCK,
    )
    # Gates are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ contraction using a fused Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty_like(left)  # placeholder for the result

    # Strides (int64) – required by the kernel
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * H,                                    # batch * head dimension
        (N + BLOCK_M - 1) // BLOCK_M,             # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,             # j‑tiles
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_i, sR_k, sR_h,   # note: right's j‑stride == sR_i
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid) and fused element‑wise multiplication
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float16))
    )  # (B, N, N, hidden_dim) fp16

    out_gate = out_gate.contiguous()
    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    mul_out_gate_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to `dim` (fp16 → fp16)
    # ------------------------------------------------------------------
    final = F.linear(
        out_mul,
        weights["to_out.weight"].to(torch.float16),
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10️⃣ Cast to the expected float32 output
    # ------------------------------------------------------------------
    return final.to(torch.float32)
