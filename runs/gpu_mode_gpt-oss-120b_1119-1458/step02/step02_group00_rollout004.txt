"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Key ideas
---------
* **Mixed‑precision** – all heavy linear work is performed in FP16 while
  the surrounding LayerNorms stay in FP32 for numeric stability.
* **Fused linear projections** – left/right projections and their gates are
  computed in a single GEMM each by concatenating the two weight matrices.
  This cuts the number of large matrix‑multiplication launches from 6 to 4.
* **Triton kernels** – two tiny kernels fuse element‑wise work:
    1. `apply_mask_gate_kernel` multiplies the left/right tensors with the
       mask (when present) **and** their respective sigmoid‑gates in a
       single pass.
    2. `elementwise_mul_kernel` multiplies the post‑norm tensor with the
       output gate.
* **Batched GEMM for the N³ step** – the core `einsum('...ikd,...jkd->...ijd')`
  is exactly a batched matrix multiplication `A @ Bᵀ` where each head
  (batch × hidden_dim) is an independent N×N GEMM.  A single `torch.bmm`
  call processes the whole `(B*hidden_dim)` batch efficiently on H100.

The implementation respects the required signature:

    def custom_kernel(data: Tuple[torch.Tensor,
                                  torch.Tensor,
                                  Dict[str, torch.Tensor],
                                  Dict]) -> torch.Tensor:
        ...

and works for all test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    left_ptr,          # fp16, shape (total_elems,)
    left_gate_ptr,     # fp16, same shape as left_ptr
    right_ptr,         # fp16, same shape as left_ptr
    right_gate_ptr,    # fp16, same shape as left_ptr
    mask_ptr,          # fp16, shape (B * N * N,)
    hidden_dim: tl.constexpr,
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask,
       right = right * right_gate * mask.
       `mask` is broadcasted over the hidden dimension, i.e.
       mask_idx = offset // hidden_dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < numel

    # load mask (broadcasted over hidden_dim)
    mask_offs = offs // hidden_dim
    mask = tl.load(mask_ptr + mask_offs, mask=m, other=1.0)

    l  = tl.load(left_ptr + offs, mask=m, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    r  = tl.load(right_ptr + offs, mask=m, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    tl.store(left_ptr + offs,  l * lg * mask, mask=m)
    tl.store(right_ptr + offs, r * rg * mask, mask=m)


@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # fp16 input (out_norm)
    b_ptr,          # fp16 input (out_gate)
    out_ptr,        # fp16 output
    numel: tl.int64,
    BLOCK: tl.constexpr,
):
    """out = a * b   (both flattened)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < numel

    a = tl.load(a_ptr + offs, mask=m, other=0.0)
    b = tl.load(b_ptr + offs, mask=m, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=m)


# ----------------------------------------------------------------------
# Custom kernel (forward pass)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask          : torch.Tensor      [B, N, N]    (bool/float)
        - weights       : dict of torch.Tensor   model parameters
        - config        : dict with keys ``dim`` and ``hidden_dim``.
          Optional key ``nomask`` (bool) specifies if a mask is present.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half for all subsequent heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projections + gates (two GEMMs instead of four)
    # ------------------------------------------------------------------
    # left projection + left gate
    w_left_comb = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden_dim, dim)
    left_comb = F.linear(x_h, w_left_comb)  # [B, N, N, 2*hidden_dim]   fp16
    left = left_comb[..., :hidden_dim]                     # proj
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # gate

    # right projection + right gate
    w_right_comb = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden_dim, dim)
    right_comb = F.linear(x_h, w_right_comb)  # [B, N, N, 2*hidden_dim]
    right = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and fuse the gates via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    numel = left.numel()               # B * N * N * hidden_dim
    BLOCK = 65536                      # ~64k elements per kernel launch
    grid = ((numel + BLOCK - 1) // BLOCK,)

    if nomask:
        # simple gate‑only kernel (already defined earlier)
        @triton.jit
        def apply_gates_kernel(
            left_ptr, left_gate_ptr,
            right_ptr, right_gate_ptr,
            numel: tl.int64,
            BLOCK: tl.constexpr,
        ):
            pid = tl.program_id(0)
            offs = pid * BLOCK + tl.arange(0, BLOCK)
            mask = offs < numel

            l  = tl.load(left_ptr + offs,  mask=mask, other=0.0)
            lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
            r  = tl.load(right_ptr + offs, mask=mask, other=0.0)
            rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

            tl.store(left_ptr + offs,  l * lg, mask=mask)
            tl.store(right_ptr + offs, r * rg, mask=mask)

        apply_gates_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            right.view(-1),
            right_gate.view(-1),
            numel,
            BLOCK=BLOCK,
        )
    else:
        # broadcast mask over hidden_dim inside the kernel
        mask_h = mask.to(torch.float16).contiguous()  # [B, N, N]
        mask_flat = mask_h.view(-1)                  # [B*N*N]
        apply_mask_gate_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            right.view(-1),
            right_gate.view(-1),
            mask_flat,
            hidden_dim,
            numel,
            BLOCK=BLOCK,
        )
    # Gates are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ reduction – batched GEMM (equivalent to einsum)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    total_heads = B * H

    # (B, N, N, H) -> (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # A @ Bᵀ for each head   (FP16 GEMM on H100 Tensor‑Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [head, N, N]

    # Restore shape → [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))  # [B, N, N, hidden_dim]  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing LayerNorm (FP16 weights)
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by the output gate (Triton)
    # ------------------------------------------------------------------
    out_mul = torch.empty_like(out)
    numel_out = out.numel()
    grid_out = ((numel_out + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_out](
        out.view(-1),
        out_gate.contiguous().view(-1),
        out_mul.view(-1),
        numel_out,
        BLOCK=BLOCK,
    )
    # we can free out_gate now
    del out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final = F.linear(out_mul, w_to_out)  # [B, N, N, dim]   fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (the interface expects fp32)
    # ------------------------------------------------------------------
    return final.to(torch.float32)
