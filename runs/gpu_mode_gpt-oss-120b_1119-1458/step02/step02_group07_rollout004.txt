```
@triton.jit
def _apply_mask_kernel(left_ptr, right_ptr, mask_ptr,
                       stride_l0, stride_l1, stride_l2, stride_l3,
                       stride_r0, stride_r1, stride_r2, stride_r3,
                       stride_m0, stride_m1, stride_m2,
                       B, N, H,
                       BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    b = pid // (N*N)
    ij = pid % (N*N)
    i = ij // N
    j = ij % N
    off_l = b*stride_l0 + i*stride_l1 + j*stride_l2
    off_r = b*stride_r0 + i*stride_r1 + j*stride_r2
    off_m = b*stride_m0 + i*stride_m1 + j*stride_m2
    
    mask_val = tl.load(mask_ptr + off_m)
    # loop over hidden
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H
        left = tl.load(left_ptr + off_l + offs*stride_l3, mask=mask_h, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs*stride_r3, mask=mask_h, other=tl.float16(0.0))
        left = left * mask_val
        right = right * mask_val
        tl.store(left_ptr + off_l + offs*stride_l3, left, mask=mask_h)
        tl.store(right_ptr + off_r + offs*stride_r3, right, mask=mask_h)
```

"""
Highly‑optimized Triton‑augmented forward pass for the *outgoing* Triangle
Multiplicative Update (TriMul) from AlphaFold‑3.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) → fp16.
2️⃣  Fuse the linear projection and its gate for the *left* and *right*
    tensors by concatenating the two weight matrices and performing a single
    GEMM per side.  Apply the sigmoid gate and obtain the gated tensors.
3️⃣  Compute the *out* gate separately (single GEMM + sigmoid).
4️⃣  If a mask is present, launch a tiny Triton kernel that multiplies the
    two gated tensors *in‑place* by the scalar mask (broadcast over the hidden
    dimension).  This avoids two extra memory passes.
5️⃣  The core N³ operation Σₖ left[i,k,d]·right[j,k,d] is realised as a batch
    of independent half‑precision GEMMs:
        C[d] = left[d] @ right[d]ᵀ
    performed by a single torch.bmm call on a (B·H, N, N) tensor.
6️⃣  Second Layer‑norm (half‑precision), multiply by the *out* gate and a
    final linear projection back to the original feature dimension.
    All heavy work stays in fp16; the final result is cast back to float32.

Only the mask‑application step lives in Triton; the rest leverages cuBLAS/Tensor
Cores for maximal throughput on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _apply_mask_kernel(
    left_ptr,     # fp16 * left tensor   [B,N,N,hidden]
    right_ptr,    # fp16 * right tensor  [B,N,N,hidden]
    mask_ptr,     # fp16 * mask tensor   [B,N,N]
    # strides for left tensor
    stride_l0, stride_l1, stride_l2, stride_l3,
    # strides for right tensor
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask tensor
    stride_m0, stride_m1, stride_m2,
    B, N, H,                 # dimensions
    BLOCK_H: tl.constexpr,   # tile size for hidden dimension
):
    """In‑place: left, right ← left * mask, right * mask."""
    pid = tl.program_id(0)                     # one thread per (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets (element strides, not byte strides)
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # scalar mask value for this (b,i,j)
    mask_val = tl.load(mask_ptr + off_m)

    # loop over the hidden dimension
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        active = offs < H

        left = tl.load(left_ptr + off_l + offs * stride_l3,
                       mask=active, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs * stride_r3,
                        mask=active, other=tl.float16(0.0))

        left = left * mask_val
        right = right * mask_val

        tl.store(left_ptr + off_l + offs * stride_l3, left, mask=active)
        tl.store(right_ptr + off_r + offs * stride_r3, right, mask=active)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:          torch.Tensor [B, N, N]    (float32, may be all‑ones)
            - weights:       dict of model parameters (float32)
            - config:        dict with keys `dim`, `hidden_dim`, `nomask`

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    x_fp16 = x.to(torch.float16)

    # 2️⃣  Fused projection + gate for left & right (single GEMM per side)
    # ------------------------------------------------------------------
    # concatenate projection & gate weights (shape: (2*H, dim))
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # combined GEMMs
    left_comb = F.linear(x_fp16, left_comb_w)   # [B,N,N,2*H]  fp16
    right_comb = F.linear(x_fp16, right_comb_w)

    # split projection & gate pre‑activations
    left_proj = left_comb[..., :hidden_dim]
    left_gate_pre = left_comb[..., hidden_dim:]
    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]

    # sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # gated tensors
    left = left_proj * left_gate          # [B,N,N,H] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3️⃣  Out‑gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )  # [B,N,N,H] fp16

    # ------------------------------------------------------------------

        left = left.contiguous()
        right = right.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        _apply_mask_kernel[
            (total_programs,)
        ](
            left,
            right,
            mask_fp16,
            # strides for left
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            # strides for right
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # sizes
            input_tensor.shape[0],
            input_tensor.shape[1],
            hidden_dim,
            BLOCK_H=32,          # 32 works for hidden ≤ 128
            num_warps=4,
        )
        # left/right now contain the masked values

    # ------------------------------------------------------------------
    # 5️⃣  Batched GEMM: Σₖ left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    batch_heads = B * hidden_dim

    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch_heads, N, N)

    # half‑precision GEMM (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (batch_heads, N, N), fp16

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm (half‑precision) → out‑gate → final projection
    # ------------------------------------------------------------------
    to_out_norm_w_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_fp16 = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(

        weight=to_out_norm_w_fp16,
        bias=to_out_norm_b_fp16,
        eps=eps,
    )  # fp16

    # apply out‑gate (already fp16)
    out = out * out_gate

    # final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))  # fp16
