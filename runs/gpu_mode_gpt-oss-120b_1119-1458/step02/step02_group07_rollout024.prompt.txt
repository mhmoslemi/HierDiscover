```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
Optimized Triton‑augmented forward pass of the *outgoing* Triangle
Multiplicative Update (TriMul) from AlphaFold‑3.

Algorithm
---------
1️⃣ Layer‑norm the input tensor (float32).
2️⃣ Cast to fp16 and apply three linear projections (left, right and gates).
3️⃣ If a mask is present, a tiny Triton kernel fuses  
   `mask * left_gate` and `mask * right_gate` into the projected tensors,
   removing two extra memory passes.
4️⃣ The core N³ operation  Σₖ left[..., i, k, d] * right[..., j, k, d]   
   is realised as a batch of independent half‑precision GEMMs:
        C[d] = left[d] @ right[d].T
   which is executed by a single `torch.bmm` call (Tensor‑Core accelerated).
5️⃣ Convert back to float32, apply a second LayerNorm, the output gate
   and a final linear projection to the original feature dimension.

The heavy work (mask‑gate fusion) is done in Triton, while the large
matrix multiplication leverages cuBLAS/Tensor‑Cores for maximal throughput.
All tensors are kept contiguous to avoid extra copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                      # fp16 * left projection (B,N,N,head)
    right_ptr,                     # fp16 * right projection (B,N,N,head)
    mask_ptr,                      # fp16 * mask (B,N,N)
    left_gate_ptr,                 # fp16 * left gate (B,N,N,head)
    right_gate_ptr,                # fp16 * right gate (B,N,N,head)
    # strides for left/right (head,i,j,head_dim)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (b,i,j)
    stride_m0, stride_m1, stride_m2,
    # strides for gates (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,                       # batch, seq_len, hidden_dim (head_dim)
    BLOCK_H: tl.constexpr,        # tile size along the hidden dimension
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # broadcasted mask scalar (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in BLOCK_H sized chunks
    # ------------------------------------------------------------------
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        # load loads (vectorised across hidden dim)
        left = tl.load(left_ptr + off_l + offs * stride_l3,
                       mask=mask_h, other=tl.float16(0.0))
        right = tl.load(right_ptr + off_r + offs * stride_r3,
                        mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_g + offs * stride_g3,
                             mask=mask_h, other=tl.float16(0.0))

        # fused operation:  proj * gate * mask
        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        # store back (in‑place)
        tl.store(left_ptr + off_l + offs * stride_l3, left_out, mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out, mask=mask_h)


# ----------------------------------------------------------------------
# Entry‑point expected by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]      (float32)
            - mask:        [B, N, N]            (float32, optional)
            - weights: dict of model parameters (float32)
            - config: dict with keys `dim`, `hidden_dim`, `nomask`
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – cast weights to fp16 once
    # ------------------------------------------------------------------
    left_w = weights["left_proj.weight"].to(torch.float16)
    right_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_fp16, left_w)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_fp16, right_w)  # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # fp16
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask & gates
    # ------------------------------------------------------------------
    if nomask:
        # simple element‑wise fusion – no Triton launch
        left = left * left_gate
        right = right * right_gate
    else:
        # mask shape [B, N, N] → broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).contiguous()

        # ensure tensors are contiguous for correct strides
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]  # B * N * N
        # Triton launch
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left,
            right,
            mask_fp16,
            left_gate,
            right_gate,
            # strides
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # sizes
            input_tensor.shape[0],       # B
            input_tensor.shape[1],       # N
            hidden_dim,                  # H
            BLOCK_H=32,                  # hidden tile size (covers 128‑dim heads)
            num_warps=4,
        )
        # after the kernel, `left` and `right` already contain mask*gate*proj

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: for each head compute C = A @ Bᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    heads = B * hidden_dim

    # reshape to (heads, N, N) where each head corresponds to one hidden dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(heads, N, N)

    # fp16 GEMM – Tensor Cores take care of fp32 accumulation internally
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing (LayerNorm → out‑gate → final projection)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # out_gate was fp16 → cast to fp32 for the multiplication
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]  # already float32
    out_final = F.linear(out_norm, to_out_w)  # [B, N, N, dim] float32

    return out_final
```
