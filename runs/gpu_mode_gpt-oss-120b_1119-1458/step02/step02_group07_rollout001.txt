"""
TriMul outgoing kernel – high‑performance forward pass.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to fp16 and perform **one fused linear** that produces
   * left_proj, right_proj, left_gate, right_gate, out_gate* in a single
   mat‑mul (reduces kernel launches & memory traffic).
3. Apply sigmoid to the three gate slices.
4. If a mask is present, a tiny Triton kernel fuses
   `proj * gate * mask` into the left/right tensors in‑place.
   Otherwise we fuse `proj * gate` with a simple element‑wise product.
5. The N³ “pairwise” operation is a batch of independent GEMMs:
        C[d] = left[..., :, :, d] @ right[..., :, :, d].T
   Implemented with `torch.bmm` on (B·hidden_dim) matrices, which
   uses cuBLAS/Tensor‑Cores (fp16 → fp32 accumulation).
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to `dim` (float32).

All heavy element‑wise work (mask + gate fusion) is done in Triton,
while the large matrix multiplications are delegated to cuBLAS for
maximum throughput on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * left_proj (B,N,N,H)
    right_ptr,         # fp16 * right_proj (B,N,N,H)
    mask_ptr,          # fp16 * mask (B,N,N)
    left_gate_ptr,     # fp16 * left_gate (B,N,N,H)
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H)
    # Strides (in elements) for the above tensors
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_m0, stride_m1, stride_m2,
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,                       # sizes
    BLOCK_H: tl.constexpr,         # core tile along hidden dim
):
    """Fuse mask and gate into the left/right projections in‑place."""
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode (b,i,j) from flat pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets (no hidden dim)
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # mask is a scalar for a given (b,i,j)
    mask_val = tl.load(mask_ptr + off_m)

    # loop over hidden dimension
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        left      = tl.load(left_ptr      + off_l + offs * stride_l3,
                           mask=mask_h, other=tl.float16(0.0))
        right     = tl.load(right_ptr     + off_r + offs * stride_r3,
                           mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                           mask=mask_h, other=tl.float16(0.0))
        right_gate = tl.load(right_gate_ptr + off_g + offs * stride_g3,
                            mask=mask_h, other=tl.float16(0.0))

        # fused operation
        left_out  = left  * left_gate  * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_ptr  + off_l + offs * stride_l3, left_out,
                 mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out,
                 mask=mask_h)


# ----------------------------------------------------------------------
# Entry‑point: expected by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]        (float32)
            - mask:         [B, N, N] (float32)   (may be ignored if config['nomask'])
            - weights: dict of model parameters (float32)
            - config:  dict with ``dim``, ``hidden_dim``, ``nomask``

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16, fused linear for all projections & gates
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                       # [B,N,N,dim] fp16

    # Concatenate all five weight matrices (out_features stacked)
    left_w     = weights["left_proj.weight"]    .to(torch.float16)   # (H, dim)
    right_w    = weights["right_proj.weight"]   .to(torch.float16)   # (H, dim)
    left_g_w   = weights["left_gate.weight"]    .to(torch.float16)   # (H, dim)
    right_g_w  = weights["right_gate.weight"]   .to(torch.float16)   # (H, dim)
    out_g_w    = weights["out_gate.weight"]     .to(torch.float16)   # (H, dim)

    # (5*H, dim)
    fused_w = torch.cat([left_w, right_w, left_g_w, right_g_w, out_g_w], dim=0)

    # One fused mat‑mul: result shape [B,N,N,5*H] (fp16)
    proj_all = F.linear(x_fp16, fused_w)   # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Slice and apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    H = hidden_dim
    left         = proj_all[...,                     :H]               # [B,N,N,H]
    right        = proj_all[...,                 H:2*H]               # [B,N,N,H]
    left_gate_raw  = proj_all[...,           2*H:3*H]               # [B,N,N,H]
    right_gate_raw = proj_all[...,           3*H:4*H]               # [B,N,N,H]
    out_gate_raw   = proj_all[...,           4*H:5*H]               # [B,N,N,H]

    left_gate  = torch.sigmoid(left_gate_raw)    # fp16
    right_gate = torch.sigmoid(right_gate_raw)   # fp16
    out_gate   = torch.sigmoid(out_gate_raw)     # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask & gating
    # ------------------------------------------------------------------
    if nomask:
        # mask‑free path – element‑wise multiplication only
        left  = left  * left_gate
        right = right * right_gate
    else:
        # broadcast mask to hidden dim and fuse with gates via Triton
        mask_fp16 = mask.to(torch.float16).contiguous()
        # ensure tensors are contiguous for correct strides
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]  # B*N*N
        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left, right,
            mask_fp16,
            left_gate, right_gate,
            # strides for left
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            # strides for right
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            # strides for mask
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            # strides for gates (same layout as left/right)
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # sizes
            input_tensor.shape[0],           # B
            input_tensor.shape[1],           # N
            hidden_dim,                      # H
            BLOCK_H=32,                      # tile along hidden dim (covers hidden_dim up to 128)
            num_warps=4,
        )
        # after the kernel, `left` & `right` already contain mask*gate*proj

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ operation: batch of GEMMs (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # fp16

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N).T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, retains fp32 accumulation

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)   # back to fp32 for the remaining FP32 ops

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # fp32

    # out_gate is still fp16; cast to fp32 for the element‑wise multiplication
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim) float32
    out_final = F.linear(out_norm, to_out_w)    # [B, N, N, dim] float32

    return out_final
