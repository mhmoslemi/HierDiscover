"""
Optimized outgoing Triangular Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input (FP32) and cast to FP16 for all heavy ops.
2. Fuse the two linear projections (proj + gate) for the left and right
   branches by concatenating their weight matrices:
       [proj; gate]  →  F.linear → split → sigmoid(gate) (in‑place).
   This reduces the number of GEMM passes from 5 → 3.
3. A single Triton kernel (`proj_gate_mask_kernel`) multiplies the
   projected tensors with their sigmoid‑gates and the optional mask, producing
   `left_out` and `right_out` (both shape [B,N,N,hidden_dim]).
4. Perform the cubic‑cost “einsum” via a batched FP16 GEMM:
       out[b,h,i,j] = Σ_k left_out[b,i,k,h] · right_out[b,j,k,h]
   Implemented with `torch.bmm` on a view (B·hidden_dim, N, N),
   which on H100 executes on Tensor‑Cores.
5. Apply a second LayerNorm (FP16), multiply by `out_gate` (sigmoid‑gated)
   and finish with a linear projection back to `dim`.
6. Return the result in FP32.

A Triton kernel is used for the element‑wise fusion in step 3,
satisfying the requirement that “part of the operations” are performed in a kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    left_proj_ptr, left_gate_ptr,        # fp16, [total_elems]
    right_proj_ptr, right_gate_ptr,      # fp16, [total_elems]
    mask_ptr,                            # fp16, [B*N*N]  (broadcasted) – may be dummy
    left_out_ptr, right_out_ptr,         # fp16, [total_elems]
    hidden_dim: tl.constexpr,            # scalar hidden_dim (for mask broadcast)
    mask_present: tl.constexpr,          # 0/1 compile‑time flag
    numel: tl.int64,                     # total number of elements per tensor
    BLOCK: tl.constexpr,                 # thread block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel

    # Load the four tensors
    lproj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rproj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if mask_present:
        # Each group of `hidden_dim` elements shares the same mask entry
        mask_off = offs // hidden_dim
        m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)
        lout = lproj * lgate * m
        rout = rproj * rgate * m
    else:
        lout = lproj * lgate
        rout = rproj * rgate

    tl.store(left_out_ptr + offs, lout, mask=mask)
    tl.store(right_out_ptr + offs, rout, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float)
        - weights      : dict of torch.Tensor, model parameters
        - config       : dict, must contain 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor with shape [B, N, N, dim] (float32)
    """
    # -------------------------- unpack -----------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True → ignore mask
    eps = 1e-5
    device = input_tensor.device

    # -------------------- 1️⃣ Input LayerNorm (FP32) -------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] fp32

    # -------------------- cast to FP16 ---------------------------
    x_h = x.to(torch.float16)

    # -------------------- 2️⃣ Prepare fused weights ---------------
    # left side: concat proj and gate weights → (2*hidden_dim, dim)
    w_left_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    # right side
    w_right_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    # out‑gate weight
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    # final layer‑norm and projection weights
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # -------------------- 3️⃣ Linear + gate (fused) ---------------
    # left branch
    left_comb = F.linear(x_h, w_left_concat)                     # [B,N,N,2*H]
    left_proj = left_comb[..., :hidden_dim]                      # view of proj part
    left_gate = left_comb[..., hidden_dim:].sigmoid_()           # in‑place sigmoid on gate part

    # right branch
    right_comb = F.linear(x_h, w_right_concat)                   # [B,N,N,2*H]
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid_()         # in‑place

    # -------------------- 4️⃣ out‑gate ----------------------------
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))         # [B,N,N,hidden_dim]

    # -------------------- 5️⃣ optional mask -----------------------
    if not nomask:
        mask_fp16 = mask.to(dtype=torch.float16)                 # [B,N,N]
    else:
        # dummy tensor – will never be accessed because mask_present=False
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)

    # -------------------- 6️⃣ fused gate+mask kernel --------------
    # Allocate output tensors for the gated & masked left/right streams
    left_out = torch.empty_like(left_proj)   # [B,N,N,hidden_dim] fp16
    right_out = torch.empty_like(right_proj)

    # flatten for kernel launch
    total_el = left_proj.numel()
    BLOCK = 65536  # 64 K elements per program – works well on H100
    grid = (triton.cdiv(total_el, BLOCK),)

    proj_gate_mask_kernel[grid](
        left_proj.view(-1),
        left_gate.view(-1),
        right_proj.view(-1),
        right_gate.view(-1),
        mask_fp16.view(-1),
        left_out.view(-1),
        right_out.view(-1),
        hidden_dim,               # constexpr for mask broadcast
        mask_present=int(not nomask),  # 0/1 constexpr
        numel=total_el,
        BLOCK=BLOCK,
    )
    # free intermediate tensors early
    del left_proj, left_gate, right_proj, right_gate, left_comb, right_comb

    # -------------------- 7️⃣ N³ batched GEMM --------------------
    B, N, _, H = left_out.shape               # H == hidden_dim
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left_out.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_out.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # A @ Bᵀ   (Tensor‑Core FP16 GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))       # [B*H, N, N]

    # restore original layout: [B,N,N,hidden_dim]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # -------------------- 8️⃣ post‑norm ---------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # fp16

    # -------------------- 9️⃣ gate after norm ----------------------
    out = out * out_gate  # element‑wise FP16 multiplication (torch)

    # -------------------- 🔟 final projection --------------------
    final = F.linear(out, w_to_out)   # [B,N,N,dim] fp16

    # -------------------- return FP32 ---------------------------
    return final.to(torch.float32)
