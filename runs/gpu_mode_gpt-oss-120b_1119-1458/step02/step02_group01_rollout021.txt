```
"""
optimized outgoing TriMul kernel using fused projections and a custom Triton matmul.

Algorithm:
1. LayerNorm over input dim in fp32.
2. Cast to fp16.
3. Compute left/right projections and per‑position gates with torch.nn.functional.linear
   (bias=False).  Mask (if any) is applied after projection.
4. Fuse projections with gates: left = (left_proj * left_gate) * mask, right = …
5. Heavy contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched matrix‑multiply A @ Bᵀ where A,B have shape
   [B·H, N, N] (fp16).  The Triton kernel uses 128×128×64 tiles,
   FP16 tensor‑core multiplication, and FP32 accumulation.
6. Second LayerNorm (fp16), element‑wise out_gate (sigmoid), and final linear
   projection back to `dim`.  Result is returned in fp32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batch‑wise matrix multiply C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch*hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile index along output‑row dimension (i)
    pid_n = tl.program_id(1)   # tile index along output‑col dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # -------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 to keep numerical accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # load A tile  (M‑dim × K‑dim)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # load B tile transposed (K‑dim × N‑dim)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )   # fp16

        # core MMA (fp16 * fp16 -> fp32)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # -------- write result ----------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Helper: launch the Triton kernel for the `(i,k) * (j,k)` contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched MatMul:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        tensors of shape [B, N, N, H] with dtype torch.float16 and contiguous memory.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, H] (fp16) containing the contraction result.
    """
    B, N, _, H = left.shape
    # pack batch and hidden into a single leading dimension: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [M,N,K]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # strides (in element counts)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``, optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # cast to fp16 for the rest of the pipeline
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Cast weights to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -------------------------------------------------
    # 4️⃣ Optional mask
    # -------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> broadcast over hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # apply gates
    left = left_proj * left_gate   # [B,N,N,hidden]
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton
    # -------------------------------------------------
    # Ensure tensors are contiguous before passing to Triton
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating and projection (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (as required)
    # -------------------------------------------------
    return out.to(torch.float32)

"""
Optimized outgoing TriMul kernel (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D tensor over the feature axis (float32 for stability).
2️⃣  Cast to fp16 – all remaining heavy work runs in half precision.
3️⃣  Linear projections (left/right) and per‑position sigmoid gates
    (left_gate, right_gate, out_gate) are computed with bias‑free `torch.nn.functional.linear`.
    An optional mask (broadcast over the hidden dimension) is applied after the
    projections.
4️⃣  Fuse projection & gate by element‑wise multiplication:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
5️⃣  Contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    This is a batched matrix‑multiply `A @ Bᵀ` where
        A,B ∈ ℝ^{M×N×K}, M = B·H, N = seq_len, K = seq_len.
    It is executed by a custom Triton kernel that:
        • uses 128×128×64 tiles,
        • employs tensor‑core `tl.dot` (fp16 → fp32 accumulation),
        • pipelines loads with 4 stages,
        • runs with 8 warps per block.
6️⃣  Second layer‑norm (fp16) over the hidden dimension.
7️⃣  Apply the `out_gate` (sigmoid) and a final linear projection back to `dim`.
8️⃣  Cast the result to float32 – the required output type.

Only the O(N³) contraction is implemented in Triton; all other
operations use high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A, B : [M, N, K] (M = batch * hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile along output‑row dimension (i)
    pid_n = tl.program_id(1)   # tile along output‑col dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # ---------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- load A tile (M‑dim × K‑dim) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----- load B tile transposed (K‑dim × N‑dim) -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )   # fp16

        # core MMA (fp16 * fp16 → fp32)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ---------- write result ----------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Helper that prepares tensors, launches the kernel and restores layout
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        shape [B, N, N, H], dtype torch.float16, contiguous.

    Returns
    -------
    torch.Tensor
        shape [B, N, N, H], dtype torch.float16
    """
    B, N, _, H = left.shape
    # pack (batch, hidden) into one leading dimension: [M, N, K]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # strides in element counts
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H, N, N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Entry point consumed by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, on CUDA)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool, on CUDA)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # Cast everything to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 2️⃣ Cast model weights to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # fuse projection with gate
    left = left_proj * left_gate   # shape [B,N,N,hidden]
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (fp16 inputs → fp16 output)
    # -------------------------------------------------
    # ensure contiguity for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection back to the original dimension
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 (as required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
