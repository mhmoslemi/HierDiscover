@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    stride_lp_b, stride_lp_i, stride_lp_j,
    stride_lg_b, stride_lg_i, stride_lg_j,
    stride_rp_b, stride_rp_i, stride_rp_j,
    stride_rg_b, stride_rg_i, stride_rg_j,
    stride_mask_b, stride_mask_i, stride_mask_j,
    hidden_dim, B, N,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    total = B * hidden_dim * N * N
    mask = offs < total

    # compute coordinates
    # Using integer division
    tmp = offs // (hidden_dim * N * N)
    b = tmp
    rem = offs - b * (hidden_dim * N * N)
    h = rem // (N * N)
    rem2 = rem - h * (N * N)
    i = rem2 // N
    j = rem2 - i * N

    # pointers
    # left_proj offset
    lp_offset = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j + h
    lg_offset = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h
    rp_offset = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j + h
    rg_offset = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h

    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load values as fp16 then cast to fp32 for multiplier
    lp = tl.load(left_proj_ptr + lp_offset, mask=mask).to(tl.float32)
    lg = tl.load(left_gate_ptr + lg_offset, mask=mask).to(tl.float32)
    rp = tl.load(right_proj_ptr + rp_offset, mask=mask).to(tl.float32)
    rg = tl.load(right_gate_ptr + rg_offset, mask=mask).to(tl.float32)
    m = tl.load(mask_ptr + mask_offset, mask=mask).to(tl.float32)

    left_val = lp * lg * m
    right_val = rp * rg * m

    # compute out offsets (contiguous layout)
    out_stride_b = hidden_dim * N * N
    out_stride_h = N * N
    out_stride_i = N
    out_stride_j = 1

    left_out_offset = b * out_stride_b + h * out_stride_h + i * out_stride_i + j
    right_out_offset = left_out_offset  # same layout for both but separate pointer

    tl.store(left_out_ptr + left_out_offset, left_val.to(tl.float16), mask=mask)
    tl.store(right_out_ptr + right_out_offset, right_val.to(tl.float16), mask=mask)

Implementation specifics:

```
@triton.jit
def _proj_gate_mask_fused_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    stride_lp_b, stride_lp_i, stride_lp_j,
    stride_lg_b, stride_lg_i, stride_lg_j,
    stride_rp_b, stride_rp_i, stride_rp_j,
    stride_rg_b, stride_rg_i, stride_rg_j,
    stride_mask_b, stride_mask_i, stride_mask_j,
    B, N, hidden_dim,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    total = B * hidden_dim * N * N
    mask = offs < total

    # compute indices
    # compute b = offs // (hidden_dim * N * N)
    prod = hidden_dim * N * N
    b = offs // prod
    remainder = offs - b * prod
    h = remainder // (N * N)
    remainder2 = remainder - h * (N * N)
    i = remainder2 // N
    j = remainder2 - i * N

    # compute offsets for inputs
    # left_proj offset: b*stride_lp_b + i*stride_lp_i + j*stride_lp_j + h*1
    lp_offset = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j + h
    lg_offset = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h
    rp_offset = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j + h
    rg_offset = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h

    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load
    lp = tl.load(left_proj_ptr + lp_offset, mask=mask).to(tl.float32)
    lg = tl.load(left_gate_ptr + lg_offset, mask=mask).to(tl.float32)
    rp = tl.load(right_proj_ptr + rp_offset, mask=mask).to(tl.float32)
    rg = tl.load(right_gate_ptr + rg_offset, mask=mask).to(tl.float32)
    m = tl.load(mask_ptr + mask_offset, mask=mask).to(tl.float32)

    left_out_val = lp * lg * m
    right_out_val = rp * rg * m

    # compute output offsets (contiguous)
    out_stride_b = hidden_dim * N * N
    out_stride_h = N * N
    out_stride_i = N
    out_stride_j = 1

    left_out_offset = b * out_stride_b + h * out_stride_h + i * out_stride_i + j
    right_out_offset = left_out_offset  # since same layout

    tl.store(left_out_ptr + left_out_offset, left_out_val.to(tl.float16), mask=mask)
    tl.store(right_out_ptr + right_out_offset, right_out_val.to(tl.float16), mask=mask)
```

"""
TriMul (outgoing) – fast forward pass.

The implementation follows the reference PyTorch module but
* fuses the elementwise product of the two linear projections,
  their sigmoid gates and the optional mask in a single Triton kernel,
* stores the result in a layout (B, H, N, N) that is already
  contiguous for the subsequent batched GEMM,
* performs the O(N³) contraction as a batch of matrix‑multiplications
  (A @ Bᵀ) with torch.bmm which on an H100 uses Tensor‑Core FP16→FP32.
All heavy arithmetic is done in FP16; LayerNorms and the final
projection are carried out in FP32 to match the reference output.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: proj × gate × mask → contiguous (B, hidden, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    # inputs
    left_proj_ptr, left_gate_ptr,        # [B, N, N, H]  (fp16)
    right_proj_ptr, right_gate_ptr,      # [B, N, N, H]  (fp16)
    mask_ptr,                            # [B, N, N]    (fp16)
    # outputs (contiguous)
    left_out_ptr, right_out_ptr,         # [B, H, N, N] (fp16)
    # strides of the 4‑D tensors (elements, not bytes)
    stride_lp_b, stride_lp_i, stride_lp_j,
    stride_lg_b, stride_lg_i, stride_lg_j,
    stride_rp_b, stride_rp_i, stride_rp_j,
    stride_rg_b, stride_rg_i, stride_rg_j,
    stride_mask_b, stride_mask_i, stride_mask_j,
    # problem size
    B, N, H,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    total = B * H * N * N
    mask = offs < total

    # -------------------- decode linear index ------------------------
    # idx = b·(H·N·N) + h·(N·N) + i·N + j
    prod_h_n_n = H * N * N
    b = offs // prod_h_n_n
    offs_rem = offs - b * prod_h_n_n

    h = offs_rem // (N * N)
    offs_rem = offs_rem - h * (N * N)

    i = offs_rem // N
    j = offs_rem - i * N

    # -------------------- compute input offsets ----------------------
    # left/right projection/gate tensors: (B, N, N, H)
    lp_off = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j + h
    lg_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h
    rp_off = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j + h
    rg_off = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h

    # mask tensor (B, N, N)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # -------------------- load values -------------------------------
    lp = tl.load(left_proj_ptr + lp_off, mask=mask).to(tl.float32)
    lg = tl.load(left_gate_ptr + lg_off, mask=mask).to(tl.float32)
    rp = tl.load(right_proj_ptr + rp_off, mask=mask).to(tl.float32)
    rg = tl.load(right_gate_ptr + rg_off, mask=mask).to(tl.float32)
    m  = tl.load(mask_ptr + mask_off, mask=mask).to(tl.float32)

    # -------------------- fuse -------------------------------
    left_val  = lp * lg * m
    right_val = rp * rg * m

    # -------------------- compute output offsets --------------------
    # contiguous layout (B, H, N, N) → strides:
    #   stride_out_b = H * N * N
    #   stride_out_h = N * N
    #   stride_out_i = N
    #   stride_out_j = 1
    out_off = b * (H * N * N) + h * (N * N) + i * N + j

    tl.store(left_out_ptr + out_off,  left_val .to(tl.float16), mask=mask)
    tl.store(right_out_ptr + out_off, right_val.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Triton‑driven TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (AlphaFold3) module.
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   (float32)
            mask         : [B, N, N]        (bool/float)
            weights      : dict of model parameters
            config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # Cast to FP16 for all linear / gate ops
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # Gating values (sigmoid) – keep FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16))).to(torch.float32)

    # ------------------------------------------------------------------
    # Prepare mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if nomask:
        # all‑ones mask – cheapest possible
        mask_fp16 = torch.ones((mask.shape[0], mask.shape[1], mask.shape[2]),
                               dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # --------------------------------------------------------------
    # Fuse (proj × gate × mask) into contiguous (B, H, N, N) tensors
    # --------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    # allocate contiguous storage for the fused tensors
    left_out  = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right_out = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)

    # launch Triton kernel (single launch computes both left and right)
    BLOCK = 32768                      # number of elements per program
    total_elements = B * hidden_dim * N * N
    grid = (triton.cdiv(total_elements, BLOCK),)

    _proj_gate_mask_kernel[grid](
        # input pointers
        left_proj,   left_gate,
        right_proj,  right_gate,
        mask_fp16,
        # output pointers
        left_out,    right_out,
        # strides – note: strides are in *elements* (not bytes)
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        # problem size
        B, N, hidden_dim,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 2️⃣ Core O(N³) operation – batched GEMM (A @ Bᵀ)
    # --------------------------------------------------------------
    # left_out/right_out : [B, H, N, N] → reshape to (B·H, N, N)
    left_flat  = left_out.view(B * hidden_dim, N, N)
    right_flat = right_out.view(B * hidden_dim, N, N)

    # FP16 × FP16 → FP32 accumulation (Tensor Cores)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # [B·H, N, N] (fp16)

    # reshape back to [B, N, N, H] and promote to FP32 for the next layers
    out = (
        out_flat
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # B,N,N,H
        .contiguous()
        .to(torch.float32)
    )

    # --------------------------------------------------------------
    # 3️⃣ Post‑processing: hidden‑dim LayerNorm, out‑gate, final proj
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                          # [B,N,N,H] (float32)

    out = out * out_gate                       # broadcast over H

    # final linear back to original dimension
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
