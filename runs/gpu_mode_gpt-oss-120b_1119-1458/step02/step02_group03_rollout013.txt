In PyTorch, default strides: stride[0] = M*K, stride[1] = K, stride[2] = 1 for contiguous layout. For B*N*N*H that after permute and .contiguous, these hold.

Thus we need stride_ab = a.stride(0) (for batch) = M*K; stride_al = a.stride(1) = K; stride_ak = a.stride(2) = 1.

Similarly B: shape (batch, K, N) (where K size is K dimension and N is column), stride for batch = K*N; stride_Bk = B.stride(1) = N; stride_Bn = B.stride(2) = 1.

C: shape (batch, M, N): stride_cb = M*N; stride_cm = N (if contiguous), stride_cn = 1.

    _, K2, N = B.shape
    assert K == K2
    # allocate output
    C = torch.empty((BATCH, M, N), dtype=torch.float16, device=A.device)

    # compute strides as ints
    stride_a_batch = A.stride(0)
    stride_al = A.stride(1)
    stride_ak = A.stride(2)

    stride_b_batch = B.stride(0)
    stride_bk = B.stride(1)
    stride_bn = B.stride(2)

    stride_c_batch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    BLOCK_M = 64   # tune
    BLOCK_N = 64
    BLOCK_K = 64

        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

    pid_m = tl.program_id(0)

    b_ptr_batch = b_ptr + pid_b * stride_Bb

"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
(AlphaFold3).  The heavy O(N³·H) contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is executed by a custom batched GEMM kernel written in Triton.
All linear projections are merged (projection + gating) to reduce the
number of passes over the input tensor, and the whole forward pass is
performed in mixed precision (FP16 for the compute‑intensive parts,
FP32 for the layer‑norms).  The final result is returned as FP32.

The kernel operates on tensors of shape
    A : (batch, M, K)   – left matrix (M=N, K=N)
    B : (batch, K, N)   – right matrix transposed (K=N, N=N)
and produces
    C : (batch, M, N)   – the contracted tensor.
Both A and B are expected to be contiguous in memory, which is ensured
by a cheap ``.contiguous()`` after the ``permute`` that moves the
hidden dimension to the batch axis.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel (A @ B) – pure Triton
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_Ab, stride_Al, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes C[b, m, n] = Σ_k A[b, m, k] * B[b, k, n] for a batch of
    independent GEMMs.  All tensors are in FP16; accumulation is done in
    FP32 to preserve numerical stability.
    """
    pid_b = tl.program_id(2)          # batch id
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_Ab
    b_batch = b_ptr + pid_b * stride_Bb
    c_batch = c_ptr + pid_b * stride_Cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Pointers to the current tiles of A and B
        a_ptrs = a_batch + (offs_m[:, None] * stride_Al
                            + cur_k[None, :] * stride_Ak)
        b_ptrs = b_batch + (cur_k[:, None] * stride_Bk
                            + offs_n[None, :] * stride_Bn)

        # Load with masks (zero‑pad out‑of‑bounds)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Matrix–matrix multiply (FP16 inputs → FP32 accumulation)
        acc += tl.dot(a, b)

    # Write the result back to C (down‑cast to FP16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_Cm
                        + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_triton(A: torch.Tensor, B: torch.Tensor,
                        BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM kernel.
    A – (batch, M, K) FP16 tensor, contiguous.
    B – (batch, K, N) FP16 tensor, contiguous.
    Returns C – (batch, M, N) FP16 tensor.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements (not bytes)
    stride_Ab = A.stride(0)
    stride_Al = A.stride(1)
    stride_Ak = A.stride(2)

    stride_Bb = B.stride(0)
    stride_Bk = B.stride(1)
    stride_Bn = B.stride(2)

    stride_Cb = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)

    grid = (triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            batch)

    batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        stride_Ab, stride_Al, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # good default for FP16 matmul
    )
    return C

# Public entry‑point – the forward pass of the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict: see code comments for required keys
        - config_dict : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, C = x.shape
    dim = cfg["dim"]
    H = cfg["hidden_dim"]                      # hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input layer‑norm (float32 for stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, C]  (float32)

    # Convert to FP16 for the heavy path
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Merge projection weights (left + right) and compute both
    # ------------------------------------------------------------------
    proj_w = torch.cat([w["left_proj.weight"], w["right_proj.weight"]], dim=0)
    proj_w = proj_w.to(device=device, dtype=torch.float16)          # (2*H, C)

    proj = F.linear(x_h, proj_w)                                    # [B, N, N, 2*H]
    left  = proj[..., :H]                                          # [B, N, N, H]
    right = proj[..., H:]                                          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Merge gating weights (left_gate + right_gate) and compute
    # ------------------------------------------------------------------
    gate_w = torch.cat([w["left_gate.weight"], w["right_gate.weight"]], dim=0)
    gate_w = gate_w.to(device=device, dtype=torch.float16)          # (2*H, C)

    gates = torch.sigmoid(F.linear(x_h, gate_w))                    # [B, N, N, 2*H]
    left_gate  = gates[..., :H]                                    # [B, N, N, H]
    right_gate = gates[..., H:]                                    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = w["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional pair‑wise mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6️⃣ Apply per‑head gating
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Batched contraction via Triton GEMM
    #    left : (B, N, N, H)   → (B*H, N, N)
    #    right: (B, N, N, H)   → (B*H, N, N)  and will be transposed on‑the‑fly
    # ------------------------------------------------------------------
    # Move hidden dimension to batch axis and make the matrices contiguous
    left_t  = left.permute(0, 3, 1, 2).contiguous()               # (B, H, N, N)  (i,k)
    right_t = right.permute(0, 3, 2, 1).contiguous()              # (B, H, N, N)  (k,j)

    # Collapse batch×head → single batch dimension
    batch = B * H
    left_flat  = left_t.view(batch, N, N)          # (batch, M=N, K=N)
    right_flat = right_t.view(batch, N, N)         # (batch, K=N, N)

    # Perform the batched GEMM with Triton
    out_flat = batched_gemm_triton(left_flat, right_flat)  # (batch, N, N) FP16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8️⃣ Output layer‑norm (float32 ↔ FP16)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"].to(torch.float32),
        bias=w["to_out_norm.bias"].to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 9️⃣ Final gated projection back to original channel dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B,N,N,H)
    to_out_w = w["to_out.weight"].to(device=device, dtype=torch.float16)  # (C, H)
    final = F.linear(gated, to_out_w)              # (B,N,N,C)  FP16

    # ------------------------------------------------------------------
    # 10️⃣ Return FP32 tensor as required by the API
    # ------------------------------------------------------------------
    return final.to(torch.float32)
```
