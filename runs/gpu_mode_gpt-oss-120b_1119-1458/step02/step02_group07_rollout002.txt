"""
Optimised outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the [B,N,N,dim] input (float32).
2️⃣  Cast to fp16 and perform **fused** linear projections:
       - left_proj  + left_gate  (concatenated → one matmul)
       - right_proj + right_gate (concatenated → one matmul)
   Split the results; apply sigmoid to the gate halves.
3️⃣  If a mask is present, a tiny Triton kernel fuses
       `mask * left_gate` and `mask * right_gate` into the projected tensors,
   otherwise a simple element‑wise multiplication is performed.
4️⃣  The core N³ operation Σₖ left[i,k,d]·right[j,k,d] is carried out as a
   batched half‑precision GEMM:
       C[d] = left[d] @ right[d]ᵀ
   using a single `torch.bmm` over the stacked head dimension.
5️⃣  Convert back to float32, apply a second LayerNorm, the output gate,
   and a final fp32 linear projection to the original feature dimension.

Only the mask‑gate fusion (step 3) is written in Triton; all large matrix
multiplications use cuBLAS/Tensor‑Cores, which on an H100 gives the best
throughput.  Mixed‑precision reduces memory traffic while keeping the final
output in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * left projection   [B,N,N,H]
    right_ptr,         # fp16 * right projection  [B,N,N,H]
    mask_ptr,          # fp16 * mask             [B,N,N]
    left_gate_ptr,     # fp16 * left gate         [B,N,N,H]
    right_gate_ptr,    # fp16 * right gate        [B,N,N,H]
    # strides for left/right (B,N,N,H)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for mask (B,N,N)
    stride_m0, stride_m1, stride_m2,
    # strides for gates (same layout as left/right)
    stride_g0, stride_g1, stride_g2, stride_g3,
    B, N, H,                     # compile‑time constants
    BLOCK_H: tl.constexpr,      # tile size along hidden dimension
):
    """One program per (b,i,j) tuple.  Applies:
           left  = left  * left_gate  * mask
           right = right * right_gate * mask
       in‑place on the fp16 tensors."""
    pid = tl.program_id(0)
    total = B * N * N
    # decode (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for the 4‑D tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2
    off_g = b * stride_g0 + i * stride_g1 + j * stride_g2
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # load the scalar mask value (fp16)
    mask_val = tl.load(mask_ptr + off_m)

    # Iterate over the hidden dimension
    for start in range(0, H, BLOCK_H):
        offs = start + tl.arange(0, BLOCK_H)
        mask_h = offs < H

        # load vectors of length BLOCK_H
        left      = tl.load(left_ptr      + off_l + offs * stride_l3,
                           mask=mask_h, other=tl.float16(0.0))
        right     = tl.load(right_ptr     + off_r + offs * stride_r3,
                           mask=mask_h, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + off_g + offs * stride_g3,
                           mask=mask_h, other=tl.float16(0.0))
        right_gate= tl.load(right_gate_ptr+ off_g + offs * stride_g3,
                           mask=mask_h, other=tl.float16(0.0))

        # fused mask‑gate‑project
        left_out  = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        # store back (in‑place)
        tl.store(left_ptr  + off_l + offs * stride_l3, left_out,
                 mask=mask_h)
        tl.store(right_ptr + off_r + offs * stride_r3, right_out,
                 mask=mask_h)


# ----------------------------------------------------------------------
# Entry‑point expected by the harness
# ----------------------------------------------------------------------
@torch.no_grad()
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor, shape [B, N, N] (float32) – may be ignored
            - weights:     dict of model parameters (float32 tensors)
            - config:      dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack everything
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last (feature) dimension – stay in fp32
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and perform *fused* linear projections
    #    left_proj+left_gate   →   2·hidden_dim columns
    #    right_proj+right_gate →   2·hidden_dim columns
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # concatenate weights for the fused matmuls
    left_cat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)   # (2·H, dim)
    right_cat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)   # (2·H, dim)

    # fused linear: [B,N,N,2·H]
    left_cat = F.linear(x_fp16, left_cat_w)   # fp16
    right_cat = F.linear(x_fp16, right_cat_w) # fp16

    # split into projection & gate tensors
    left = left_cat[..., :hidden_dim]          # [B,N,N,H]   proj
    left_gate_raw = left_cat[..., hidden_dim:] # [B,N,N,H]   gate pre‑sigmoid
    right = right_cat[..., :hidden_dim]
    right_gate_raw = right_cat[..., hidden_dim:]

    # sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask and gate
    # ------------------------------------------------------------------
    if not nomask:
        # mask is expected to be a float tensor (0/1). Convert to fp16.
        mask_fp16 = mask.to(device).to(torch.float16).contiguous()

        # Ensure tensors are contiguous for stride calculation
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_programs = input_tensor.shape[0] * input_tensor.shape[1] * input_tensor.shape[1]

        _mask_gate_fuse_kernel[
            (total_programs,)
        ](
            left,
            right,
            mask_fp16,
            left_gate,
            right_gate,
            # strides
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            # dimensions
            input_tensor.shape[0],                # B
            input_tensor.shape[1],                # N
            hidden_dim,                           # H
            BLOCK_H=32,                           # tile over H
            num_warps=4,
        )
        # after the kernel `left` and `right` already contain mask·gate·proj
    else:
        # No mask – simple element‑wise fuse
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ computation as batched GEMM
    #    For each hidden head we compute C = A @ Bᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                # B,N,N,H
    heads = B * hidden_dim

    # reshape to (heads, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(heads, N, N)

    # half‑precision batched matmul (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim] (still fp16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Post‑processing: LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # fp32

    # out gate (computed earlier in fp16, now cast)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # fp16
    out_gate_f32 = out_gate.to(torch.float32)

    out_norm = out_norm * out_gate_f32

    # final linear back to `dim` (float32)
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
