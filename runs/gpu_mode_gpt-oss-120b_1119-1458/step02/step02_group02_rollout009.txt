    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # layernorm
    x_norm = F.layer_norm(inp, (dim,), weights['norm.weight'], weights['norm.bias'], eps=eps).to(torch.float16)

    # fused linear for proj and gates
    w_concat = torch.cat([...], dim=0).to(torch.float16)

    combined = F.linear(x_norm, w_concat)  # shape B,N,N,5*hidden

    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(combined, hidden, dim=-1)

    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        mask = mask.to(torch.float16).unsqueeze(-1).expand(-1,-1,-1,hidden)
        left = fused_mul3(left_proj_raw, mask, left_gate)
        right = fused_mul3(right_proj_raw, mask, right_gate)

    # core contraction using triton batched gemm
    out = batched_gemm_outgoing(left, right, B, N, hidden)  # returns [B,N,N,hidden] half

    # out_norm (layernorm) across hidden
    out_norm = F.layer_norm(out, (hidden,), weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps=eps)

    # apply gate
    out_norm = out_norm * out_gate

    # final linear back to dim
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

@triton.jit
def batched_gemm_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    B, N, hidden,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bi, stride_Bk,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   batch_id = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute b, h
   b = batch_id // hidden
   h = batch_id % hidden

   # base offsets
   offset_A = b * stride_Ab + h
   offset_B = b * stride_Bb + h
   offset_C = b * stride_Cb + h

   # block offsets
   offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)

   # masks
   mask_i = offs_i < N
   mask_j = offs_j < N

   # accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # tile loop
   for k in range(0, tl.cdiv(N, BLOCK_K)):
       cur_offs_k = k * BLOCK_K + offs_k
       mask_k = cur_offs_k < N

       # Load A block
       a_ptrs = A_ptr + offset_A + offs_i[:,None] * stride_Ai + (cur_offs_k[None,:]) * stride_Ak
       a = tl.load(a_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0).to(tl.float16)
       # Load B block (transposed)
       b_ptrs = B_ptr + offset_B + (cur_offs_k[:,None]) * stride_Bk + offs_j[None,:] * stride_Bj
       b = tl.load(b_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0).to(tl.float16)

       acc = acc + tl.dot(a, b).to(tl.float32)

   # store
   c_ptrs = C_ptr + offset_C + offs_i[:,None] * stride_Ci + offs_j[None,:] * stride_Cj
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:,None] & mask_j[None,:])

"""
TriMul (outgoing) – optimized Triton forward pass
================================================

Algorithm
---------
1. **Input LayerNorm** –  FP32 → FP16.
2. **Fused projection + gate logits** – concatenate the five weight matrices
   (left_proj, left_gate, right_proj, right_gate, out_gate) and perform a
   single half‑precision linear layer.  This yields the raw projections and the
   three gate‑logits in one memory pass.
3. **Mask + gate fusion** – when a mask is present we fuse the three
   element‑wise products (mask, projection, sigmoid‑gate) with a tiny Triton
   kernel (`fused_mul3`).  Without a mask a simple binary multiplication is
   used.
4. **Core contraction (outgoing update)** – compute
   ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
   This is a batched GEMM over the hidden dimension; a custom Triton kernel
   `batched_gemm_outgoing_kernel` processes each hidden channel as a separate
   batch, eliminating the costly `permute → contiguous → torch.bmm` steps.
5. **Output LayerNorm + gating + final linear** – layer‑norm over the hidden
   channel, element‑wise multiply by the output‑gate, then a half‑precision linear
   projection back to the original dimension.  The result is cast back to the
   input dtype (usually float32).

All heavy arithmetic stays in FP16 (or FP32 accumulation inside the kernels)
to exploit Tensor‑cores on the H100, while memory traffic is minimized by
fusing element‑wise ops and removing unnecessary copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Element‑wise triple product (mask * proj * gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    size: tl.constexpr,            # total elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused a * b * c using Triton (inputs must be contiguous and same shape)."""
    assert a.shape == b.shape == c.shape
    a, b, c = (x.contiguous() for x in (a, b, c))
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣  Batched GEMM for the outgoing update
#    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers to left, right, output
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,   # dimensions
    stride_Ab, stride_Ai, stride_Ak,   # strides for A (left)
    stride_Bb, stride_Bi, stride_Bk,   # strides for B (right)
    stride_Cb, stride_Ci, stride_Cj,   # strides for C (output)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)           # flattened (batch * hidden) index
    pid_m    = tl.program_id(1)           # tile over i (rows)
    pid_n    = tl.program_id(2)           # tile over j (cols)

    # De‑compose batch_id → batch and hidden‑channel
    hidden = H
    b = batch_id // hidden
    h = batch_id % hidden

    # Base address for this hidden channel
    a_base = A_ptr + b * stride_Ab + h
    b_base = B_ptr + b * stride_Bb + h
    c_base = C_ptr + b * stride_Cb + h

    # Offsets inside the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (the summed index)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # --- Load A block (shape BLOCK_M × BLOCK_K) ---
        a_ptrs = a_base + offs_i[:, None] * stride_Ai + (cur_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float16)

        # --- Load B block (as B^T: shape BLOCK_K × BLOCK_N) ---
        # Original B layout: (b, j, k, h) → we want B_T[k, j]
        b_ptrs = b_base + (cur_k[:, None] * stride_Bk) + (offs_j[None, :] * stride_Bi)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float16)

        acc = acc + tl.dot(a, b).to(tl.float32)

    # Store result C (shape BLOCK_M × BLOCK_N) back to memory (FP16)
    c_ptrs = c_base + offs_i[:, None] * stride_Ci + (offs_j[None, :] * stride_Cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def batched_outgoing_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing TriMul contraction with a custom Triton kernel.
    left/right: [B, N, N, H] in FP16, contiguous.
    Returns: same shape and dtype.
    """
    B, N, _, H = left.shape

    # Output tensor (same layout as inputs)
    out = torch.empty_like(left)

    # Strides of the (B, N, N, H) layout (contiguous)
    stride_b = left.stride(0)   # N * N * H
    stride_i = left.stride(1)   # N * H
    stride_k = left.stride(2)   # H
    # stride_h = left.stride(3) == 1  (implicit, added via `+ h` below)

    # Block sizes – tuned for H100 FP16 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                        # one program per hidden channel
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
    )

    batched_gemm_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_b, stride_i, stride_k,   # A strides
        stride_b, stride_i, stride_k,   # B strides (same layout)
        stride_b, stride_i, stride_k,   # C strides: (i‑stride, j‑stride=H)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# 3️⃣  Full forward pass (entry point)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul.
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim] (float32, CUDA)
            - mask_tensor  : [B, N, N]  (bool/float, CUDA) – may be all‑ones when nomask=True
            - weights_dict : model weights (see skeleton)
            - config_dict  : {"dim": ..., "hidden_dim": ..., "eps": ..., "nomask": ...}
    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    H = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fused projection + gate logits
    # ------------------------------------------------------------------
    # Concatenate five weight matrices (each of shape H×dim) → (5H × dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    combined = F.linear(x_norm, w_concat)                      # [B,N,N,5H]
    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(
        combined, H, dim=-1
    )

    # Sigmoid gates (still half‑precision)
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # ------------------------------------------------------------------
    # 3) Apply mask + gates (fused element‑wise product)
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj_raw * left_gate            # [B,N,N,H]
        right = right_proj_raw * right_gate
    else:
        # Expand mask to hidden dimension; ensure contiguity for Triton
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        mask_h = mask_h.expand(-1, -1, -1, H).contiguous()
        left = fused_mul3(left_proj_raw, mask_h, left_gate)
        right = fused_mul3(right_proj_raw, mask_h, right_gate)

    # ------------------------------------------------------------------
    # 4) Core contraction via custom Triton GEMM
    # ------------------------------------------------------------------
    # Both tensors are already FP16 and contiguous
    out_hidden = batched_outgoing_matmul(left, right)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                      # element‑wise gate

    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),
    )

    # Cast back to the original dtype (normally float32)
    return out_final.to(inp.dtype)
