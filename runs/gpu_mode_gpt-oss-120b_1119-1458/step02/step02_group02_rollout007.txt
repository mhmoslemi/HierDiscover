```
@triton.jit
def proj_gate_kernel(
    combined_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_elem: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK_ELEM: tl.constexpr,
    BLOCK_HIDDEN: tl.constexpr,
):
    pid_e = tl.program_id(0)
    pid_h = tl.program_id(1)
    offs_e = pid_e * BLOCK_ELEM + tl.arange(0, BLOCK_ELEM)
    offs_h = pid_h * BLOCK_HIDDEN + tl.arange(0, BLOCK_HIDDEN)
    mask_e = offs_e < total_elem
    mask_h = offs_h < hidden
    mask = mask_e[:, None] & mask_h[None, :]  # shape (BLOCK_ELEM, BLOCK_HIDDEN)

    # Load mask values for current elements (scalar per element)
    mask_scalar = tl.load(mask_ptr + offs_e[:, None], mask=mask_e[:, None], other=1.0)  # shape (BLOCK_ELEM,1)

    # compute strides for combined
    stride_comb = 5 * hidden
    # calculate offsets for each segment
    # base offset for combined: offs_e * stride_comb + offset_segment + offs_h
    # segments: left_proj_raw:0*hidden, left_gate:1*hidden, right_proj_raw:2*hidden, right_gate:3*hidden, out_gate:4*hidden

    # load raw values
    left_raw = tl.load(combined_ptr + offs_e[:, None] * stride_comb + offs_h[None, :], mask=mask, other=0.0)
    left_gate = tl.load(combined_ptr + offs_e[:, None] * stride_comb + hidden + offs_h[None, :], mask=mask, other=0.0)
    right_raw = tl.load(combined_ptr + offs_e[:, None] * stride_comb + 2*hidden + offs_h[None, :], mask=mask, other=0.0)
    right_gate = tl.load(combined_ptr + offs_e[:, None] * stride_comb + 3*hidden + offs_h[None, :], mask=mask, other=0.0)
    out_gate_logit = tl.load(combined_ptr + offs_e[:, None] * stride_comb + 4*hidden + offs_h[None, :], mask=mask, other=0.0)

    # sigmoid function
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    left_gate_sig = sigmoid(left_gate)
    right_gate_sig = sigmoid(right_gate)
    out_gate_sig = sigmoid(out_gate_logit)

    # apply mask and gates
    left_scaled = left_raw * left_gate_sig * mask_scalar
    right_scaled = right_raw * right_gate_sig * mask_scalar

    # store to outputs
    # left_ptr, right_ptr, out_gate_ptr shape: (total_elem, hidden)
    offset_out = offs_e[:, None] * hidden + offs_h[None, :]

    tl.store(left_ptr + offset_out, left_scaled, mask=mask)
    tl.store(right_ptr + offset_out, right_scaled, mask=mask)
    tl.store(out_gate_ptr + offset_out, out_gate_sig, mask=mask)
```

"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
=======================================================

The kernel fuses the three projection/gate linear layers (left_proj,
right_proj, left_gate, right_gate, out_gate) into a single
matrix‑multiply followed by a 2‑D Triton kernel that:
    • computes the sigmoid logits,
    • applies the optional binary mask,
    • multiplies the projections by their gates,
    • produces the three tensors required for the core contraction:
        left  : [B, N, N, H]
        right : [B, N, N, H]
        out_gate : [B, N, N, H]

The heavy N³ contraction `out[i,j,h] = Σₖ left[i,k,h] * right[j,k,h]`
remains a batched GEMM (`torch.bmm`), which on an H100 uses Tensor‑cores
and is already optimal.  All other operations (layer‑norms, final
linear) are kept in half‑precision to minimise memory traffic.

The implementation therefore reduces the number of memory passes,
eliminates two separate element‑wise kernels, and keeps the total
kernel launch count low while still satisfying the requirement to
use Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused projection + gating + (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    combined_ptr,            # [total_elem, 5*H]   (fp16, contiguous)
    mask_ptr,                # [total_elem]        (fp16)
    left_ptr,                # [total_elem, H]     (fp16)
    right_ptr,               # [total_elem, H]     (fp16)
    out_gate_ptr,            # [total_elem, H]     (fp16)
    total_elem: tl.int64,    # total number of (b,i,k) positions = B*N*N
    hidden: tl.int64,        # hidden dimension H
    BLOCK_ELEM: tl.constexpr,
    BLOCK_HIDDEN: tl.constexpr,
):
    """
    For each (elem, h) compute:
        left  = (combined[...,0*H+h]) * sigmoid(combined[...,1*H+h]) * mask
        right = (combined[...,2*H+h]) * sigmoid(combined[...,3*H+h]) * mask
        out_gate = sigmoid(combined[...,4*H+h])
    """
    pid_e = tl.program_id(0)                     # element‑tile index
    pid_h = tl.program_id(1)                     # hidden‑tile index

    # 1‑D offsets inside the flattened tensors
    offs_e = pid_e * BLOCK_ELEM + tl.arange(0, BLOCK_ELEM)
    offs_h = pid_h * BLOCK_HIDDEN + tl.arange(0, BLOCK_HIDDEN)

    # bounds checking
    mask_e = offs_e < total_elem
    mask_h = offs_h < hidden
    mask_2d = mask_e[:, None] & mask_h[None, :]  # (BLOCK_ELEM, BLOCK_HIDDEN)

    # ------------------------------------------------------------------
    # Load the required slices from the concatenated linear result
    # ------------------------------------------------------------------
    stride = 5 * hidden                           # distance between rows

    # raw projections
    left_raw   = tl.load(
        combined_ptr + offs_e[:, None] * stride + offs_h[None, :],
        mask=mask_2d, other=0.0)

    left_gate  = tl.load(
        combined_ptr + offs_e[:, None] * stride + hidden + offs_h[None, :],
        mask=mask_2d, other=0.0)

    right_raw  = tl.load(
        combined_ptr + offs_e[:, None] * stride + 2 * hidden + offs_h[None, :],
        mask=mask_2d, other=0.0)

    right_gate = tl.load(
        combined_ptr + offs_e[:, None] * stride + 3 * hidden + offs_h[None, :],
        mask=mask_2d, other=0.0)

    out_gate_raw = tl.load(
        combined_ptr + offs_e[:, None] * stride + 4 * hidden + offs_h[None, :],
        mask=mask_2d, other=0.0)

    # ------------------------------------------------------------------
    # Sigmoid (using the exact definition – works with fp16)
    # ------------------------------------------------------------------
    @tl.inline
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    left_gate_sig  = sigmoid(left_gate)
    right_gate_sig = sigmoid(right_gate)
    out_gate_sig   = sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # Load mask (defaults to 1.0 if mask is all‑ones)
    # ------------------------------------------------------------------
    # mask is a 1‑D tensor; broadcast to hidden dimension.
    mask_val = tl.load(mask_ptr + offs_e, mask=mask_e, other=1.0)  # (BLOCK_ELEM,)
    mask_val = mask_val[:, None]                                 # (BLOCK_ELEM,1)

    # ------------------------------------------------------------------
    # Apply mask and gates
    # ------------------------------------------------------------------
    left_out  = left_raw  * left_gate_sig  * mask_val
    right_out = right_raw * right_gate_sig * mask_val

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    # layout for the three outputs: row‑major [total_elem, hidden]
    out_offset = offs_e[:, None] * hidden + offs_h[None, :]

    tl.store(left_ptr + out_offset,   left_out,   mask=mask_2d)
    tl.store(right_ptr + out_offset,  right_out,  mask=mask_2d)
    tl.store(out_gate_ptr + out_offset, out_gate_sig, mask=mask_2d)

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C] (float32, CUDA)
            - mask_tensor : [B, N, N] (bool/float, CUDA) – may be None
            - weights_dict: model parameters (see skeleton)
            - config_dict : contains "dim", "hidden_dim", optional "eps" and "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    device = inp.device
    dtype = inp.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (still in fp32 → fp16 for the rest)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # half‑precision for downstream work

    # ------------------------------------------------------------------
    # 2) Fused projection + gate linear map
    # ------------------------------------------------------------------
    # Concatenate the 5 weight matrices (each of shape [hidden, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)           # (5*hidden, dim)

    total_elem = B * N * N

    # Collapse the first three dims to a single batch dimension for the
    # large GEMM that produces all five linear results at once.
    x_flat = x_norm.view(total_elem, dim)                 # (B·N·N, dim)
    combined = F.linear(x_flat, w_concat)                 # (B·N·N, 5*hidden)

    # Ensure the tensor is contiguous for Triton loads.
    combined = combined.contiguous()

    # ------------------------------------------------------------------
    # 3) Mask handling (broadcast to all hidden positions)
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_flat = torch.full(
            (total_elem,), 1.0, dtype=torch.float16, device=device
        )
    else:
        mask_flat = mask.to(torch.float16).view(total_elem)

    # ------------------------------------------------------------------
    # 4) Fused projection / gating / (optional) mask kernel
    # ------------------------------------------------------------------
    left  = torch.empty((total_elem, hidden), dtype=torch.float16, device=device)
    right = torch.empty((total_elem, hidden), dtype=torch.float16, device=device)
    out_gate = torch.empty((total_elem, hidden), dtype=torch.float16, device=device)

    # Block sizes – tuned for H100 FP16 workloads.
    BLOCK_ELEM = 128          # number of (b,i,k) elements per program
    BLOCK_HIDDEN = 32         # hidden sub‑tiles per program (must divide hidden)

    grid = (
        triton.cdiv(total_elem, BLOCK_ELEM),   # over elements
        triton.cdiv(hidden, BLOCK_HIDDEN),     # over hidden
    )

    proj_gate_kernel[grid](
        combined,
        mask_flat,
        left,
        right,
        out_gate,
        total_elem,
        hidden,
        BLOCK_ELEM=BLOCK_ELEM,
        BLOCK_HIDDEN=BLOCK_HIDDEN,
    )

    # Reshape back to the 4‑D layout expected by the contraction.
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)
    out_gate = out_gate.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5) Core contraction: out[i,j,h] = Σₖ left[i,k,h] * right[j,k,h]
    #    (batched GEMM across the hidden dimension)
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the second axis so that we can use a
    # single bmm call over a batch of size B*hidden.
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N)

    left_flat = left_mat.view(B * hidden, N, N)        # (B*H, N, N)
    right_flat = right_mat.view(B * hidden, N, N)      # (B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat)       # (B*H, N, N)

    # Restore the original layout: (B, N, N, hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # (B,N,N,hidden) half‑precision

    # Apply the learned output gate.
    out_norm = out_norm * out_gate

    # Final linear projection back to the original dimensionality.
    # Again we flatten to use a single GEMM.
    out_flat = out_norm.view(total_elem, hidden)
    out_final = F.linear(out_flat, weights["to_out.weight"].to(torch.float16))
    out_final = out_final.view(B, N, N, dim)

    # Cast back to the original (usually float32) dtype.
    return out_final.to(dtype)
