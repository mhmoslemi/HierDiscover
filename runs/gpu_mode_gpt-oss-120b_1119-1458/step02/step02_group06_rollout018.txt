"""
Optimised outgoing TriMul (AlphaFold‑3 style) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input in FP32.
2. Cast to FP16 – the heavy linear projections are executed in half‑precision.
3. Fuse *proj* and *gate* for the left and right branches:
      [left_proj, left_gate] = Linear(x, cat(W_left_proj, W_left_gate))
      [right_proj, right_gate] = Linear(x, cat(W_right_proj, W_right_gate))
   – this halves the number of passes over the huge tensor (B·N²·dim) and
     reduces memory traffic.
4. Apply the optional mask together with the sigmoid gates (element‑wise
   multiplication – cheap compared to the GEMMs).
5. Reshape the tensors to (B·H, N, N) and perform the N³ reduction.
   For modest (B·H) a custom Triton GEMM kernel is used, otherwise we fall
   back to torch.bmm which maps to cuBLAS Tensor‑Core kernels.
6. Layer‑norm the result, apply the output‑gate and the final linear
   projection back to the original dimension (all in FP32).
7. Return a float‑32 tensor of shape [B, N, N, dim].

The kernel for the batched GEMM (step 5) uses fp16×fp16→fp32 Tensor‑Core
dot‑products with block‑size tuned for H100 (BLOCK_M=N_BLOCK=128, BLOCK_K=32).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ rightᵀ  (M = B·H, K = N, N = N)
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_am, stride_ak,          # a: (M, N, K) – we interpret K=N
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # batch * hidden
    pid_i = tl.program_id(1)   # tile row   (i)
    pid_j = tl.program_id(2)   # tile col   (j)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduce over the K dimension (the sequence axis)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + pid_m * stride_am
                 + offs_i[:, None] * stride_ak
                 + offs_k[None, :] * 1,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load Bᵀ block: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr + pid_m * stride_bm
                 + offs_k[:, None] * stride_bk
                 + offs_j[None, :] * 1,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # FP16 dot, accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write result (FP16)
    # ------------------------------------------------------------------
    tl.store(
        c_ptr + pid_m * stride_cm
              + offs_i[:, None] * stride_cn
              + offs_j[None, :] * 1,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor (B, N, N, dim)    – float32
            - mask         : torch.Tensor (B, N, N)       – 0/1 int or bool
            - weights      : dict of pre‑initialised model weights
            - config       : dict with keys
                * dim        – input feature dimension
                * hidden_dim – hidden dimension after projection
                * nomask     – bool, whether the mask should be ignored

    Returns:
        torch.Tensor of shape (B, N, N, dim) in float32.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 (keep contiguous)
    # ------------------------------------------------------------------
    x_h = x.contiguous().half()          # [B, N, N, dim]  FP16
    B, N, _, _ = x_h.shape                # B, N, dim

    # ------------------------------------------------------------------
    # 3. Flatten for the massive linear projections
    # ------------------------------------------------------------------
    x_flat = x_h.view(-1, dim)            # (B·N·N, dim)

    # ------------------------------------------------------------------
    # 4. Fuse left‑proj & left‑gate, and right‑proj & right‑gate
    # ------------------------------------------------------------------
    # left side
    w_left_full = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()                              # (2·H, dim)
    left_full = torch.nn.functional.linear(x_flat, w_left_full)  # (BNN, 2·H)
    left_proj = left_full[:, :hidden_dim]
    left_gate_raw = left_full[:, hidden_dim:]

    # right side
    w_right_full = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()                              # (2·H, dim)
    right_full = torch.nn.functional.linear(x_flat, w_right_full)  # (BNN, 2·H)
    right_proj = right_full[:, :hidden_dim]
    right_gate_raw = right_full[:, hidden_dim:]

    # out‑gate (separate)
    w_out_gate = weights["out_gate.weight"].half()
    out_gate = torch.nn.functional.linear(x_flat, w_out_gate).sigmoid()  # (BNN, H)

    # ------------------------------------------------------------------
    # 5. Reshape back to 4‑D tensors
    # ------------------------------------------------------------------
    left_proj = left_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate_raw.view(B, N, N, hidden_dim).sigmoid()
    right_proj = right_proj.view(B, N, N, hidden_dim)
    right_gate = right_gate_raw.view(B, N, N, hidden_dim).sigmoid()
    out_gate = out_gate.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 6. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(torch.float16)   # (B,N,N,1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # ------------------------------------------------------------------
    # 7. Apply gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 8. Batched GEMM: out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Transform to (B·H, N, N) and use either the Triton kernel or
    #    torch.bmm (cuBLAS Tensor‑Cores) depending on size.
    # ------------------------------------------------------------------
    # move hidden dim to the second axis and make it contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left.view(B * hidden_dim, N, N)    # FP16
    right_flat = right.view(B * hidden_dim, N, N)

    BH = B * hidden_dim
    # Heuristic: Triton is faster for a modest number of batches
    use_triton = (BH <= 64) and (N <= 512)

    if use_triton:
        out_flat = torch.empty_like(left_flat)    # FP16 result

        stride_a_m, stride_a_n, stride_a_k = left_flat.stride()
        stride_b_m, stride_b_n, stride_b_k = right_flat.stride()
        stride_c_m, stride_c_n, stride_c_k = out_flat.stride()

        # Block sizes tuned for H100 – 128×128 tiles, 32‑wide reduction
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )
        triton_matmul_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            N,
            stride_a_m,
            stride_a_k,
            stride_b_m,
            stride_b_n,
            stride_b_k,
            stride_c_m,
            stride_c_n,
            stride_c_k,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # Very well‑optimised Tensor‑Core implementation from cuBLAS
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H) and cast to FP32 for the rest
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()                       # now FP32

    # ------------------------------------------------------------------
    # 10. Output layer‑norm + out‑gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()            # broadcasted element‑wise

    # ------------------------------------------------------------------
    # 11. Final linear projection back to original dim (FP32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].float()
    )

    return out
