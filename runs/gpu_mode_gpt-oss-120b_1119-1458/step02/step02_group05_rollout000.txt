import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                      hidden_dim, n_elements,
                      BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK
    offs = offset + tl.arange(0, BLOCK)
    mask = offs < n_elements

    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # mask repeats hidden_dim times
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Implements the heavy element‑wise gate×mask fusion with a tiny Triton
    kernel and computes the O(N³) interaction using a single batched GEMM
    (bmm) over the hidden dimension. The whole pipeline runs in mixed
    precision (FP16 for the bulk of the arithmetic, FP32 for the final
    LayerNorms) and returns a float32 tensor of shape [B, N, N, dim].
    """
    # Unpack
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = inp.device
    eps = 1e-5

    # ---------- LayerNorm on input (float32) ----------
    x = F.layer_norm(inp,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)  # [B,N,N,dim] float32

    # ---------- Cast to FP16 ----------
    x_h = x.to(torch.float16)

    # ---------- Linear projections (FP16) ----------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_w) # [B,N,N,hidden]

    # ---------- Gating (sigmoid, keep in FP16) ----------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w)).to(torch.float32)

    # ---------- Apply mask & fuse with gates ----------
    if not nomask:
        # mask: [B,N,N] -> fp16 flattened, broadcast over hidden dim
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                # [B*N*N]

        # allocate output tensors
        left_fused = torch.empty_like(left)
        right_fused = torch.empty_like(right)

        BLOCK = 65536
        n_elem = left.numel()
        grid = (triton.cdiv(n_elem, BLOCK),)

        _gate_mask_kernel[grid](
            left,            # proj_ptr
            left_gate,       # gate_ptr
            mask_flat,       # mask_ptr
            left_fused,      # out_ptr
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        _gate_mask_kernel[grid](
            right,
            right_gate,
            mask_flat,
            right_fused,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        left = left_fused
        right = right_fused
    else:
        left = left * left_gate
        right = right * right_gate

    # ---------- Core O(N³) via batched GEMM ----------
    B, N, _, _ = left.shape
    # reshape to (B*hidden, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul: left @ right.T   (FP16 × FP16 → FP32 accumulation)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # reshape back to [B,N,N,hidden] as FP32
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ---------- Post‑processing ----------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate

    # final projection back to dim (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing Triangle‑Multiplicative Update (TriMul) – forward pass.

Algorithm
---------
1. Layer‑Norm over the feature dimension (float32) for numerical stability.
2. Cast to FP16 and apply three linear projections:
   * left  = x @ W_left          (hidden_dim)
   * right = x @ W_right         (hidden_dim)
   * Gating projections → sigmoid:
       left_gate, right_gate, out_gate.
3. Fuse element‑wise product (projection × gate) with an optional mask
   using a tiny Triton kernel.  The mask is broadcast over the hidden
   dimension, so each mask entry is multiplied with `hidden_dim` values.
4. Core O(N³) interaction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batched matrix multiplication:
       left_b : (B·H, N, N)
       right_b: (B·H, N, N)
   Compute   out_b = left_b @ right_bᵀ   (FP16×FP16 → FP32 accumulation).
5. Reshape back to [B,N,N,H] (float32), apply a second LayerNorm,
   multiply by the out‑gate and finally project back to the original
   feature dimension with a linear layer.
All heavy arithmetic runs in mixed precision; the final tensor is
float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: projection × gate × (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16: projection (left or right)
    gate_ptr,          # fp16: gate (same shape as proj)
    mask_ptr,          # fp16: broadcast mask (shape B·N·N)
    out_ptr,           # fp16: destination (can be the same as proj_ptr)
    hidden_dim,        # i64: size of hidden dimension (for mask broadcasting)
    n_elements,        # i64: total number of elements in proj/gate (B·N·N·H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load projection & gate (promote to fp32 for a more accurate product)
    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    # Each mask element repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)

# ----------------------------------------------------------------------

        - config       : dict with keys:
            * dim        : feature dimension
            * hidden_dim : hidden dimension (c_z)
            * nomask     : bool, True ⇒ mask should be ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """

    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all linear / gating ops
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear projection weights (cast once to FP16)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Gating logits & sigmoid (stay in FP16 except out_gate)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w)).to(torch.float32)  # keep FP32 for scaling

    # ------------------------------------------------------------------
    # 5️⃣ Fuse (projection × gate × mask) with Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → FP16 → flattened
        mask_fp = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)                     # shape = B·N·N (broadcast)

        # In‑place rewrite of left & right to avoid extra allocations
        # (proj_ptr and out_ptr are the same tensor)
        BLOCK = 65536
        n_elem = left.numel()                           # B·N·N·hidden_dim
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left * left_gate * mask
        _gate_mask_kernel[grid](
            left,                # proj_ptr
            left_gate,           # gate_ptr
            mask_flat,           # mask_ptr
            left,                # out_ptr (in‑place)
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right * right_gate * mask
        _gate_mask_kernel[grid](
            right,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask → just element‑wise multiply with gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³) interaction – batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B·hidden_dim, N, N) where hidden_dim becomes part of the batch
    left_b  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiply – utilizes Tensor Cores (FP16×FP16→FP32)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))   # [B·H, N, N]

    # Reshape back to [B, N, N, hidden_dim] and promote to FP32
    out = (
        out_b
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)      # [B, N, N, hidden_dim]
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim] (float32)

    out = out * out_gate  # broadcast over hidden_dim

    # Final projection back to `dim` (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
```
