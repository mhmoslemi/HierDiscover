def custom_kernel(data):
    inp, mask, weights, config = data

    B,N,_,dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps",1e-5)
    nomask = config.get("nomask", False)  # indicates whether mask should be used

    # Input layernorm
    x_norm = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)

    # Apply mask to x_norm if needed BEFORE projection (If nomask=False)
    if nomask: # no mask, just pass x_norm as masked_x
        masked_x = x_norm
    else:
        # mask shape [B,N,N], cast to half and unsqueeze
        mask_fp = mask.to(torch.float16).unsqueeze(-1) # [B,N,N,1]
        masked_x = x_norm * mask_fp  # broadcast to dim

    # fused linear for left/right projections
    # concat left_proj.weight and right_proj.weight
    proj_weight = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"]], dim=0).to(torch.float16) # (2*hidden, dim)
    left_right = F.linear(masked_x, proj_weight) # shape [B,N,N,2*hidden]
    left_proj_raw, right_proj_raw = torch.split(left_right, hidden, dim=-1)

    # Gates (linear on original x_norm without mask)
    gate_weight = torch.cat([weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]], dim=0).to(torch.float16) # (3*hidden, dim)
    gate_logits = F.linear(x_norm, gate_weight) # shape [B,N,N,3*hidden]
    left_gate_logit, right_gate_logit, out_gate_logit = torch.split(gate_logits, hidden, dim=-1)
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # Apply gating (multiplicative)
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # Now compute TriMul contraction using Triton kernel
    # shape left/right: [B,N,N,hidden] (half)
    B_val = B
    N_val = N
    hidden_val = hidden

    left = left.contiguous()
    right = right.contiguous()
    # Allocate output tensor
    out_hidden = torch.empty_like(left)  # shape [B,N,N,hidden] half

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid
    grid = lambda meta: (
        B_val * hidden_val,
        triton.cdiv(N_val, meta["BLOCK_M"]),
        triton.cdiv(N_val, meta["BLOCK_N"]),

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, Hidden,
    stride_b, stride_i, stride_k, stride_d,
    out_stride_b, out_stride_i, out_stride_j, out_stride_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    hidden = Hidden
    # compute batch and hidden index
    b = pid_bh // hidden
    h = pid_bh % hidden

    # base pointers for left/right/out for this (b,h)
    left_base = left_ptr + b * stride_b + h * stride_d
    right_base = right_ptr + b * stride_b + h * stride_d
    out_base = out_ptr + b * out_stride_b + h * out_stride_d

    # compute offsets for M and N dimensions (i and j)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_m[:, None] * stride_i) + (offs_k[None, :] * stride_k)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right block transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs_T = right_base + (offs_k[:, None] * stride_k) + (offs_n[None, :] * stride_i)
        right_T = tl.load(right_ptrs_T, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product (left (M,K) dot right_T (K,N)) -> (M,N)
        acc += tl.dot(left, right_T)

    # store result to out
    out_ptrs = out_base + (offs_m[:, None] * out_stride_i) + (offs_n[None, :] * out_stride_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass
==================================================

Algorithm
---------
1. **Input Layer‑Norm** – Normalise the last dimension of the input.
2. **Mask pre‑projection** – If a mask is supplied, multiply the
   normalised tensor by the mask (broadcasted).  This allows us to
   fuse the mask into the two projection linear layers, saving a
   whole element‑wise pass.
3. **Fused projection & gating** – 
   * Concatenate the two projection matrices (left/right) and apply a
     single homogeneous linear layer to the masked input.
   * Concatenate the three gating matrices (left_gate, right_gate,
     out_gate) and apply a second linear layer to the *unmasked*
     normalised input.
   * Split the results, apply `sigmoid` to the gates and multiply
     the gates into the projected tensors.
4. **Core contraction** – Compute the outgoing update
   `out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]`.
   This is a batched matrix multiplication across the hidden
   dimension.  A custom Triton kernel (`tri_mul_kernel`) performs the
   GEMM for every `(batch, hidden)` pair without the expensive
   `permute(...).contiguous()` copy required by a naïve `torch.bmm`.
5. **Output Layer‑Norm, gating and final projection** – Apply a
   second layer‑norm over the hidden dimension, multiply by the
   `out_gate`, and linearly project back to the original channel
   dimension.

All heavy linear algebra is executed in `float16`; the final result
is cast back to the input dtype (usually `float32`).  The implementation
uses a single custom Triton kernel (step 4), satisfying the requirement
to provide a non‑trivial kernel while improving runtime over the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul update
#   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
#   left/right shape: [B, N, N, H]   (H = hidden_dim)
#   out shape      : [B, N, N, H]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime dimensions
    stride_b, stride_i, stride_k, stride_d,  # left/right strides (same)
    out_stride_b, out_stride_i, out_stride_j, out_stride_d,  # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication over the hidden channel.

    Program ID layout:
        0 -> (batch * hidden) dimension
        1 -> tile over output rows (i)
        2 -> tile over output columns (j)
    """
    # ----- decode program IDs ------------------------------------------------
    pid_bh = tl.program_id(0)   # batch * hidden
    pid_m  = tl.program_id(1)   # i‑tile
    pid_n  = tl.program_id(2)   # j‑tile

    # batch index and hidden‑channel index
    b = pid_bh // H
    h = pid_bh %  H

    # base pointers for the selected (b, h)
    left_base = left_ptr  + b * stride_b + h * stride_d
    right_base = right_ptr + b * stride_b + h * stride_d
    out_base  = out_ptr   + b * out_stride_b + h * out_stride_d

    # ----- tile offsets ------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over k -------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_i) + (offs_k[None, :] * stride_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)   # fp16

        # right block transposed to (BLOCK_K, BLOCK_N)
        # right[j,k] -> we need right_T[k,j]
        right_ptrs_T = right_base + (offs_k[:, None] * stride_k) + (offs_j[None, :] * stride_i)
        right_T = tl.load(right_ptrs_T,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)   # fp16

        # matrix multiply for this k‑tile
        acc += tl.dot(left, right_T)   # acc stays in FP32
    # -------------------------------------------------------------------------

    # ----- write back ---------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * out_stride_i) + (offs_j[None, :] * out_stride_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
    # -------------------------------------------------------------------------

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor:  shape [B, N, N] (bool/float, on CUDA) or ``None``.
            - weights_dict: model weights (see skeleton in the prompt)
            - config_dict: contains at least "dim", "hidden_dim",
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (stay in FP32 for numerical stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)   # everything after this is fp16

    # ------------------------------------------------------------------
    # 2) Optional mask – fuse it into the two projection linear layers
    # ------------------------------------------------------------------
    if nomask or mask is None:
        masked_x = x_norm
    else:
        # mask: [B,N,N] -> [B,N,N,1] and broadcast over the feature dim
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        masked_x = x_norm * mask_fp

    # ------------------------------------------------------------------
    # 3) Fuse projections (left/right) and gates (left_gate/right_gate/out_gate)
    # ------------------------------------------------------------------
    # ------- projections -------------------------------------------------
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                     # (2*hidden, dim)
    proj = F.linear(masked_x, proj_weight)  # (B,N,N,2*hidden)
    left_proj_raw, right_proj_raw = torch.split(proj, hidden, dim=-1)

    # ------- gates -------------------------------------------------------
    gate_weight = torch.cat(
        [weights["left_gate.weight"],
         weights["right_gate.weight"],
         weights["out_gate.weight"]],
        dim=0,
    ).to(torch.float16)                     # (3*hidden, dim)
    gate = F.linear(x_norm, gate_weight)    # (B,N,N,3*hidden)
    left_gate_logit, right_gate_logit, out_gate_logit = torch.split(gate, hidden, dim=-1)

    left_gate  = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate   = torch.sigmoid(out_gate_logit)

    # ------- apply gates (mask already fused) ---------------------------
    left  = (left_proj_raw * left_gate).contiguous()
    right = (right_proj_raw * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 4) Core contraction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    Implemented with a custom Triton kernel (no permute / copy)
    # ------------------------------------------------------------------
    out_hidden = torch.empty_like(left)   # (B,N,N,hidden), fp16

    # Triton block size – tuned for H100 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in elements (PyTorch returns element‑wise strides)
    stride_b = left.stride(0)   # B‑stride
    stride_i = left.stride(1)   # i‑stride (row index)
    stride_k = left.stride(2)   # k‑stride (depth index)
    stride_d = left.stride(3)   # hidden‑stride (fastest)

    out_stride_b = out_hidden.stride(0)
    out_stride_i = out_hidden.stride(1)
    out_stride_j = out_hidden.stride(2)
    out_stride_d = out_hidden.stride(3)

    # Grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B, N, hidden,
        stride_b, stride_i, stride_k, stride_d,
        out_stride_b, out_stride_i, out_stride_j, out_stride_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5) Output layer‑norm + gating + final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # Apply the output gate (element‑wise)
    out_norm = out_norm * out_gate

    # Final linear back to original `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # Return in the original dtype (usually float32)
    return out_final.to(inp.dtype)
