```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     hidden_dim, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    # Load projection and gate
    proj = tl.load(proj_ptr + offs, mask=mask)  # fp16
    gate = tl.load(gate_ptr + offs, mask=mask)  # fp16

    # Determine mask index (broadcast mask across hidden_dim)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)  # fp16

    # Compute product in fp32 for precision
    prod = proj.to(tl.float32) * gate.to(tl.float32) * mask_val.to(tl.float32)

    tl.store(out_ptr + offs, prod.to(tl.float16), mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    This implementation fuses:
        • Linear projections for left/right together with their
          corresponding gates (two matmuls → two fused matmuls).
        • Element‑wise multiplication with optional mask using a
          custom Triton kernel (projection × sigmoid(gate) × mask).
        • The heavy O(N³) interaction as a batched GEMM (FP16 × FP16 → FP32).
        • Remaining post‑processing (LayerNorm, out‑gate, final
          projection) are performed with torch kernels in mixed precision.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # 1. LayerNorm over feature dim (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # 2. Cast to FP16 for all matmul heavy ops
    x_h = x.to(torch.float16)

    # 3. Fuse projection + gate weights for left and right
    left_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # Linear (B*N*N, dim) @ (2*hidden, dim).T -> (B*N*N, 2*hidden)
    left_concat = F.linear(x_h, left_w)   # shape [B, N, N, 2*hidden]
    right_concat = F.linear(x_h, right_w)

    # Split projection & gate logits
    left_proj, left_gate_logits = torch.split(left_concat, hidden_dim, dim=-1)
    right_proj, right_gate_logits = torch.split(right_concat, hidden_dim, dim=-1)

    # Sigmoid gating (keep in FP16)
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    # 4. Apply mask and element‑wise product (projected value × gate)
    if not nomask:
        # mask: [B, N, N] → flatten and cast to fp16
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                     # shape B·N·N

        # Allocate output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        n_elem = left_proj.numel()          # B·N·N·hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 5. Core O(N³) operation: batch GEMM (left @ right^T) per hidden dim
    B, N, _, _ = left.shape
    # reshape for batched matmul: (B*hidden, N, N)
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Compute C = A @ Bᵀ  (FP16×FP16 → FP32 accumulation)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))  # [B*hidden, N, N] fp16

    # Reshape back to [B, N, N, hidden]
    out = out_batched.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # 6. Post‑processing: hidden‑dim LayerNorm, out‑gate, output projection
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # out_gate computation (already done earlier)
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16))).to(torch.float32)

    # Apply out gate (broadcasted over hidden dim)
    out = out * out_gate

    # Final projection (hidden -> dim)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    hidden_dim, n_elements, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    proj = tl.load(proj_ptr + offs, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass optimized with fused linear layers
    and a custom Triton kernel for mask‑gated element‑wise multiplication.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # LayerNorm over feature dimension (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Cast to FP16 for matmul‑heavy work
    x_h = x.to(torch.float16)

    # ----------- 1) Linear projection + gate (fused) -------------
    # Concatenate proj and gate weights (out_features = 2*hidden_dim)
    left_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # Compute concatenated outputs: shape (B, N, N, 2*hidden_dim)
    left_concat = F.linear(x_h, left_w)
    right_concat = F.linear(x_h, right_w)

    # Split into projection and gate logits
    left_proj, left_gate_logits = torch.split(left_concat, hidden_dim, dim=-1)
    right_proj, right_gate_logits = torch.split(right_concat, hidden_dim, dim=-1)

    # Sigmoid gating (FP16)
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    # ----------- 2) Optional mask & element‑wise multiplication -------------
    if not nomask:
        # Convert mask to fp16 and flatten
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)  # shape B*N*N

        # Allocate result tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        n_elem = left_proj.numel()
        BLOCK = 1024
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------- 3) Core O(N³) interaction (batched GEMM) -------------
    B, N, _, _ = left.shape
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Compute C = A @ Bᵀ  (FP16*FP16 → FP32 accumulation)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ----------- 4) Post‑processing ---------
    # Hidden‑dim LayerNorm (float32)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Compute out_gate (sigmoid of a linear projection)
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    ).to(torch.float32)

    # Apply out_gate
    out = out * out_gate

    # Final projection back to original dimensionality
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Key ideas
---------
1️⃣  Layer‑norm on the input (float32) → keep numerical stability.
2️⃣  Cast to fp16 for all heavy matrix‐multiplications.
3️⃣  Fuse each side’s projection and gate linear layers by concatenating the
    two weight matrices (proj | gate) and performing a single matmul.
    The gate logits are turned into sigmoids afterwards.
4️⃣  Fuse the element‑wise product (proj × gate × mask) into a tiny Triton
    kernel – this replaces three separate PyTorch kernels with one.
5️⃣  The O(N³) core is a batched GEMM (A @ Bᵀ) on fp16 tensors; cuBLAS/Tensor‑Core
    delivers the highest throughput.
6️⃣  Post‑processing (hidden‑dim LayerNorm, out‑gate, final projection) stays in
    PyTorch but uses mixed precision (fp16 where possible) and avoids extra
    memory passes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 projection (left or right)
    gate_ptr,          # fp16 gate (sigmoid already applied)
    mask_ptr,          # fp16 mask (flattened, length = B·N·N)
    out_ptr,           # fp16 destination
    hidden_dim,        # i64 – hidden dimension (for mask broadcasting)
    n_elements,        # i64 – total number of operands (=B·N·N·hidden_dim)
    BLOCK: tl.constexpr,   # compile‑time thread‑block size (≤1024)
):
    """proj * gate * mask   (all in fp16; computation done in fp32)."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)                     # [0, BLOCK)
    valid = offs < n_elements

    # Load projection & gate (fp16 → fp32)
    proj = tl.load(proj_ptr + offs, mask=valid).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=valid).to(tl.float32)

    # Broadcast mask: every mask entry repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid).to(tl.float32)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out.to(tl.float16), mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data = (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]      (float32)
        mask         : [B, N, N]           (float32/bool, optional)
        weights      : dict of model parameters
        config       : dict with at least
                       {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for all matmuls
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fused projection + gate for left and right sides
    #    (concatenate weight matrices → single matmul per side)
    # ------------------------------------------------------------------
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)            # shape (2*hidden, dim)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)            # shape (2*hidden, dim)

    # Linear -> [B, N, N, 2*hidden]
    left_concat = F.linear(x_h, left_w)
    right_concat = F.linear(x_h, right_w)

    # Split into projection part and gate‑logit part
    left_proj, left_gate_logits = torch.split(left_concat, hidden_dim, dim=-1)
    right_proj, right_gate_logits = torch.split(right_concat, hidden_dim, dim=-1)

    # Sigmoid on gate logits (still fp16)
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    # ------------------------------------------------------------------
    # 4. Optional mask – fuse proj * gate * mask via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16 → flatten
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                     # length = B·N·N

        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        n_elem = left_proj.numel()                      # B·N·N·hidden_dim
        BLOCK = 1024                                     # ≤ max threads per block
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) – batched GEMM (A @ Bᵀ) per hidden dimension
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden, N, N)
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # FP16 × FP16 → FP32 accumulation
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))

    # Back to [B, N, N, hidden] (float32)
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # out_gate = sigmoid(Linear(x_h))
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    ).to(torch.float32)

    out = out * out_gate                              # broadcast over hidden

    # Final linear back to original feature dimension (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
