@triton.jit
def outgoing_tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program ids
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile col
    pid_batch = tl.program_id(2) # combined batch index (b*hidden+h)
    # Derive b and h from pid_batch
    h = pid_batch % hidden
    b = pid_batch // hidden

    # Compute base pointer offsets for left and right (start of slice for this b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Define offset indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K axis loop
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left block (i,k) shape MxK
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(left_ptrs, mask = mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block (j,k) shape NxK (we later transpose)
        right_ptrs = right_base + offs_j[:, None] * stride_right_i + cur_k[None, :] * stride_right_k
        b_buf = tl.load(right_ptrs, mask = mask_j[:, None] & mask_k[None, :], other=0.0)

        # b_buf shape (BLOCK_N, BLOCK_K), we need (BLOCK_K, BLOCK_N) for dot
        # Use tl.trans to transpose
        acc += tl.dot(a, tl.trans(b_buf))

    # Write out
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask = mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Input Layer‑Norm (FP32 → FP16) – keep the heavy‑weight reduction on the
   tensor‑core‑friendly datatype.
2️⃣ Fuse the five Linear layers (left_proj, left_gate, right_proj, right_gate,
   out_gate) by concatenating their weight matrices.  A single `F.linear`
   produces all five tensors at once, cutting the number of passes over the
   massive `[B, N, N, dim]` tensor from five to one.
3️⃣ Apply the (optional) mask together with the three gate tensors using a
   tiny Triton kernel (`fused_mul3`).  This fuses three element‑wise
   products into one memory pass.
4️⃣ Core outgoing update  
   `out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]`  (i.e. a batch of
   matrix‑multiplications `left @ rightᵀ` for each (b, d) pair).  
   A custom Triton kernel (`outgoing_gemm_kernel`) treats the hidden
   channel as an extra batch dimension and performs a tiled GEMM directly
   on the FP16 tensors, accumulating in FP32 (Tensor‑cores).  This removes
   the expensive `torch.bmm` + permute + copy pattern.
5️⃣ Output Layer‑Norm + gate + final Linear – all stay in FP16 until the
   very last cast back to the original dtype.

Only the element‑wise mask/gate product (step 3) and the cubic contraction
(step 4) are written in Triton; the remaining reductions are handled by the
highly‑optimised PyTorch kernels.

The implementation works for all shapes listed in the statement and for
both “mask‑present” and “mask‑absent” configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel 1: fused element‑wise product of three tensors
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    size: tl.constexpr,            # total number of elements
    BLOCK: tl.constexpr,           # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, a * b * c, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fuse a * b * c with a single Triton kernel."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 8192  # empirically good for H100
    grid = (triton.cdiv(total, BLOCK),)
    fused_mul3_kernel[grid](a, b, c, out, total, BLOCK=BLOCK)
    return out


# ----------------------------------------------------------------------
# Triton kernel 2: outgoing Tensor‑core GEMM (batched left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def outgoing_gemm_kernel(
    # tensors
    left_ptr, right_ptr, out_ptr,
    # geometry
    B, N, hidden,
    # strides (elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_m = tl.program_id(0)            # tile row (i)
    pid_n = tl.program_id(1)            # tile col (j)
    pid_batch = tl.program_id(2)        # combined (b, h)

    # ----- un‑pack batch & hidden indices ---------------------------------
    h = pid_batch % hidden
    b = pid_batch // hidden

    # ----- base pointers for this (b, h) ----------------------------------
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # ----- tile offsets ----------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over k ----------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left[i, k]   -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[j, k]  -> shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_base + offs_j[:, None] * stride_right_i + cur_k[None, :] * stride_right_k
        b = tl.load(right_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)

        # dot: (M,K) @ (K,N)  – we transpose `b` on‑the‑fly
        acc += tl.dot(a, tl.trans(b))

    # ----- write back ------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry‑point: fully‑fused TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32, on CUDA)
            - mask_tensor : [B, N, N]      (bool/float, on CUDA)
            - weights_dict: model parameters
            - config_dict : contains at least
                * "dim" (int) – input channel dimension
                * "hidden_dim" (int) – hidden dimension used inside TriMul
              optional:
                * "nomask" (bool) – if True, ignore the mask
                * "eps" (float) – epsilon for LayerNorm

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → float16 for the rest)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # keep FP16 for subsequent ops

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + the three gating linear layers
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (all have shape (hidden, dim))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # (5*hidden, dim)

    # one linear pass gives us all five tensors at once
    combined = F.linear(x_norm, w_concat)   # shape [B, N, N, 5*hidden]

    left_proj_raw, left_gate_logit, right_proj_raw, right_gate_logit, out_gate_logit = torch.split(
        combined, hidden, dim=-1
    )

    # sigmoid gates (still in FP16)
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # ------------------------------------------------------------------
    # 3️⃣ Mask + gate fusion (optional)
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # expand mask to hidden dimension and fuse three element‑wise products
        mask_exp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden).contiguous()
        left = fused_mul3(left_proj_raw, mask_exp, left_gate)
        right = fused_mul3(right_proj_raw, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – outgoing update
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # allocate FP16 output tensor

    # strides for the original layout [B, N, N, hidden]
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # launch the Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows (i)
        triton.cdiv(N, BLOCK_N),      # cols (j)
        B * hidden,                   # combined batch × hidden
    )

    outgoing_gemm_kernel[grid](
        left, right, out,
        B, N, hidden,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_i, s_right_k, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Output Layer‑Norm + gate + final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    out_norm = out_norm * out_gate                     # element‑wise gating

    # final linear back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # convert back to original dtype (float32) for the model
    return out_final.to(inp.dtype)
