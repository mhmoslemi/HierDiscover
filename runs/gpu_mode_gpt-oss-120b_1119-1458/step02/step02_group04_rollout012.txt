    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)

def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.is_contiguous() and b.is_contiguous()
    out = torch.empty_like(a)
    total = a.numel()

    elementwise_mul_kernel[grid](
        a, b, out,
        total_elts=total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out

def custom_kernel(data):
    """
    Fast forward implementation of the outgoing TriMul operator.
    The heavy linear projections are fused into two large matmul
    operations, the N³ contraction is performed with a single batched
    torch.matmul, and the element‑wise gate+mask multiplication is
    carried out by a tiny Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Helper to cast weight to half on device
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # Load weights and cast to half (except layernorm where fp32 may be used)
    norm_w = to_half(weights["norm.weight"])
    norm_b = to_half(weights["norm.bias"])
    left_proj_w = to_half(weights["left_proj.weight"])
    right_proj_w = to_half(weights["right_proj.weight"])
    left_gate_w = to_half(weights["left_gate.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w = to_half(weights["to_out.weight"])

    # ---------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ---------------------------------------------------------
    # Keep LN in fp32 because bias/weight are stored in fp32 in most configs.
    # Convert to half afterwards.
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w.to(torch.float32),
        bias=norm_b.to(torch.float32),
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ---------------------------------------------------------
    # 2. Cast to fp16 for all subsequent matmuls
    # ---------------------------------------------------------
    x = x_norm_fp32.to(torch.float16)

    # ---------------------------------------------------------
    # 3. One‑shot projection + out‑gate linear
    # ---------------------------------------------------------
    # combined projection weight shape (3*hidden_dim, dim)
    proj_weight = torch.cat([left_proj_w, right_proj_w, out_gate_w], dim=0)  # (3*H, dim)
    proj_all = F.linear(x, proj_weight)  # [B, N, N, 3*H]
    left_proj, right_proj, out_gate_pre = torch.split(proj_all, hidden_dim, dim=-1)

    # ---------------------------------------------------------
    # 4. One‑shot gate linear (left + right gates)
    # ---------------------------------------------------------
    gate_weight = torch.cat([left_gate_w, right_gate_w], dim=0)   # (2*H, dim)
    gate_all = F.linear(x, gate_weight)                         # [B, N, N, 2*H]
    left_gate, right_gate = torch.split(gate_all, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate_pre)

    # ---------------------------------------------------------
    # 5. Fuse projection, gate and optional mask via Triton
    # ---------------------------------------------------------
    # element‑wise product left = left_proj * left_gate (mask applied later)
    # element‑wise product right = right_proj * right_gate
    left = elementwise_mul(left_proj, left_gate)   # half
    right = elementwise_mul(right_proj, right_gate)  # half

    if mask is not None:
        # mask: [B, N, N] -> broadcast on hidden dim
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ---------------------------------------------------------
    # 6. N³ contraction: batched matmul per head
    # ---------------------------------------------------------
    # reshape to (B, H, N, N) and use torch.matmul
    left_view = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    # Contract over the k dimension: left @ rightᵀ
    out_view = torch.matmul(left_view, right_view.transpose(-2, -1))  # (B, H, N, N)
    # bring back to (B, N, N, H)
    out = out_view.permute(0, 2, 3, 1).contiguous()  # half

    # ---------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension
    # ---------------------------------------------------------
    # Convert to fp32 for higher‑precision LN (still cheap)
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ---------------------------------------------------------
    # 8. Apply output gate and final linear projection
    # ---------------------------------------------------------
    gated = out_norm * out_gate  # half * half
    final = F.linear(gated, to_out_w)  # [B,N,N,dim] half

    # ---------------------------------------------------------
    # 9. Return float32 as required
    # ---------------------------------------------------------
    return final.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → x_norm.
2. Cast to fp16 for the compute‑heavy part.
3. Fuse the three linear projections (left, right, out‑gate) into a single
   matmul; fuse the two gate projections (left‑gate, right‑gate) into a
   second matmul.
4. Apply sigmoid to the gate tensors.
5. Element‑wise fuse “proj * gate” for left and right using a tiny Triton
   kernel (reduces Python overhead).
6. Optional mask (broadcast on the hidden dimension) is applied with a
   regular PyTorch multiply (the mask is very cheap compared to the matmul).
7. Contract over the sequence dimension k via a batched GEMM:
   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]  →  left @ rightᵀ.
   Implemented with a single torch.matmul on tensors shaped (B, H, N, N).
8. Layer‑norm over the hidden dimension H, followed by the output gate.
9. Final linear projection back to the original channel dimension C.
10. Return float32 output.

Only the element‑wise “proj * gate” multiplication is performed by a
custom Triton kernel; all large linear / matmul operations stay in H100‑
optimized cuBLAS/Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton helper: element‑wise multiply (a * b) for half tensors
# -------------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total_elts: tl.constexpr, BLOCK_SIZE: tl.constexpr,
):
    """out = a * b  (all half‑precision)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the above Triton kernel; a and b must be half‑contiguous."""
    assert a.is_contiguous() and b.is_contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 32768                     # 2¹⁵ elements per program (tunable)
    grid = (triton.cdiv(total, BLOCK),)   # one dimension grid
    elementwise_mul_kernel[grid](
        a, b, out,
        total_elts=total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out

        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (torch.Tensor, float32)
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns

    torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # number of hidden channels (c_z)

    # -----------------------------------------------------------------
    # Helper: move a weight to fp16 on the target device
    # -----------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # Load and cast model weights
    # -----------------------------------------------------------------
    norm_w = to_half(weights["norm.weight"])
    norm_b = to_half(weights["norm.bias"])
    left_proj_w   = to_half(weights["left_proj.weight"])   # (H, dim)
    right_proj_w  = to_half(weights["right_proj.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numeric stability)
    # -----------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w.to(torch.float32),
        bias=norm_b.to(torch.float32),
        eps=1e-5,
    )                                   # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16, the heavy material now lives in half‑precision
    # -----------------------------------------------------------------
    x = x_norm_fp32.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ One‑shot projection + out‑gate linear (3 * hidden_dim columns)
    # -----------------------------------------------------------------
    proj_w = torch.cat([left_proj_w, right_proj_w, out_gate_w], dim=0)   # (3H, dim)
    proj_all = F.linear(x, proj_w)                                        # [B, N, N, 3H]
    left_proj, right_proj, out_gate_pre = torch.split(proj_all,
                                                      hidden_dim,
                                                      dim=-1)

    # -----------------------------------------------------------------
    # 4️⃣ One‑shot gate linear (2 * hidden_dim columns)
    # -----------------------------------------------------------------
    gate_w = torch.cat([left_gate_w, right_gate_w], dim=0)               # (2H, dim)
    gate_all = F.linear(x, gate_w)                                        # [B, N, N, 2H]
    left_gate, right_gate = torch.split(gate_all, hidden_dim, dim=-1)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate_pre)                             # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5️⃣ Fuse projection & gate (element‑wise) with a tiny Triton kernel
    # -----------------------------------------------------------------
    # Both inputs are half‑contiguous → can be processed directly.
    left  = elementwise_mul(left_proj, left_gate)        # [B, N, N, H]
    right = elementwise_mul(right_proj, right_gate)      # [B, N, N, H]

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 7️⃣ N³ contraction → batched GEMM (left @ rightᵀ) per head
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) to let torch.matmul do a batched matmul.
    left_view  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # out_view[b, h, i, j] = Σₖ left_view[b, h, i, k] * right_view[b, h, j, k]
    out_view = torch.matmul(left_view, right_view.transpose(-2, -1))   # (B, H, N, N)

    # Back to the original layout [B, N, N, H]
    out = out_view.permute(0, 2, 3, 1).contiguous()    # half

    # -----------------------------------------------------------------
    # 8️⃣ Output LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    # Perform the norm in fp32 (a tiny extra copy, but guarantees numerical
    # parity with the reference implementation).
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)          # back to half

    # -----------------------------------------------------------------
    # 9️⃣ Apply the output gate and the final linear projection
    # -----------------------------------------------------------------
    gated = out_norm * out_gate                         # half * half
    final = F.linear(gated, to_out_w)                  # [B, N, N, dim] (half)

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 as required by the specification
    # -----------------------------------------------------------------
    return final.to(torch.float32)
