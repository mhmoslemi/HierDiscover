"""
Optimised outgoing TriMul (AlphaFold‑3) forward pass.

The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed with a custom Triton kernel that implements a
batched GEMM C = A @ Bᵀ for each hidden slice (A/B have shape
[batch*hidden, N, N]).  The kernel uses a 64×64×64 tiling, fuses the
FP16→FP32 accumulation internally (no explicit .to conversion) and
employs three pipelining stages (num_stages=3) to hide global‑memory
latency.  All other operations (LayerNorm, Linear, Sigmoid) are kept
in Torch with fp16 arithmetic; the first LayerNorm stays in fp32 for
numerical stability.  The output is cast back to float32 as required
by the harness.

Both mask‑aware and mask‑free variants are supported.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (batched matmul)
#   A : [M, N, K]  (M = B*hidden, N = seq_len, K = seq_len)
#   B : [M, N, K]  (will be accessed as [M, K, N] to achieve the transpose)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, K, N]   (right already transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bk, stride_bj,   # strides for B (note the swapped K‑N order)
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) tile indices
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)                # tile row index (i)
    pid_n = tl.program_id(1)                # tile col index (j)
    pid_z = tl.program_id(2)                # batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2) accumulator (FP32 for accuracy)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) iterate over the reduction dimension K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --- load a tile of A (shape BLOCK_M x BLOCK_K) ---
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # --- load a tile of B (shape BLOCK_K x BLOCK_N) ---
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # K‑dimension stride
            + offs_n[None, :] * stride_bj   # N‑dimension stride (post‑transpose)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # --- FP16×FP16 → FP32 accumulation using tensor‑cores ---
        acc += tl.dot(a, b)   # no explicit type conversion needed

    # ------------------------------------------------------------------
    # 4) write out the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that reshapes tensors, launches the Triton kernel and restores the
# original layout [B, N, N, hidden].
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor,   # [B, N, N, H]  (fp16)
    right: torch.Tensor,  # [B, N, N, H]  (fp16)
) -> torch.Tensor:
    B, N, _, H = left.shape

    # Pack batch & hidden together: [M, N, K] with M = B*H
    #   left  → [M, N, K]   (i‑k)
    #   right → [M, K, N]   (k‑j)   – we transpose the last two dims
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # k‑j

    # Output buffer
    out = torch.empty_like(left_, dtype=torch.float16)

    # Strides expressed in element counts (not bytes)
    stride_am, stride_ai, stride_ak = left_.stride()
    stride_bm, stride_bk, stride_bj = right_.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – empirically good for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        left_, right_, out,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,          # double‑buffered pipeline
    )

    # Restore to [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no grads).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → better numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse the gates (in‑place to keep memory traffic low)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via the Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gating + projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
