def proj_gate_fused_kernel(
    x_ptr, left_ptr, right_ptr, out_gate_ptr,
    left_proj_w_ptr, right_proj_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    C, H, N,
    stride_x_bc, stride_x_in, stride_x_jn, stride_x_c,
    stride_l_bc, stride_l_in, stride_l_jn, stride_l_h,
    stride_r_bc, stride_r_in, stride_r_jn, stride_r_h,
    stride_o_bc, stride_o_in, stride_o_jn, stride_o_h,
    stride_w_c, stride_w_h,
    BLOCK_C: tl.constexpr,
):
   # Compute global index
   pid = tl.program_id(0)
   total = B * N * N
   if pid >= total:
       return
   b = pid // (N * N)
   nid = pid % (N * N)
   i = nid // N
   j = nid % N
   # compute offsets
   x_offset_base = b * stride_x_bc + i * stride_x_in + j * stride_x_jn
   # load x vector for this element
   acc_left = tl.zeros([H], dtype=tl.float16)
   acc_right = tl.zeros([H], dtype=tl.float16)
   acc_left_gate = tl.zeros([H], dtype=tl.float16)
   acc_right_gate = tl.zeros([H], dtype=tl.float16)
   acc_out_gate = tl.zeros([H], dtype=tl.float16)
   for c_start in range(0, C, BLOCK_C):
        cur_c = tl.arange(0, BLOCK_C)
        c_offsets = c_start + cur_c
        mask = c_offsets < C
        x_ptrs = x_ptr + x_offset_base + c_offsets * stride_x_c
        x = tl.load(x_ptrs, mask=mask, other=0.0)  # shape (BLOCK_C)
        # reshape to (1, BLOCK_C)
        x_vec = x[None, :]
        # Load weight blocks
        w_left = tl.load(left_proj_w_ptr + c_offsets[:, None] * stride_w_c + tl.arange(0, H)[None,:] * stride_w_h, mask=mask[:, None], other=0.0)
        w_right = tl.load(right_proj_w_ptr + c_offsets[:, None] * stride_w_c + tl.arange(0, H)[None,:] * stride_w_h, mask=mask[:, None], other=0.0)

        w_rgate = tl.load(right_gate_w_ptr + c_offsets[:, None] * stride_w_c + tl.arange(0, H)[None,:] * stride_w_h, mask=mask[:, None], other=0.0)
        w_outgate = tl.load(out_gate_w_ptr + c_offsets[:, None] * stride_w_c + tl.arange(0, H)[None,:] * stride_w_h, mask=mask[:, None], other=0.0)
        # Compute contributions via dot

        acc_left += tl.dot(x_vec, w_left)[0]
        acc_right += tl.dot(x_vec, w_right)[0]

   # Apply mask if needed outside kernel (optional).

```
@triton.jit
def proj_gate_kernel(
    x_ptr, left_ptr, right_ptr,
    left_proj_w_ptr, right_proj_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr,
    mask_ptr,
    B, N, C, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_w_c, stride_w_h,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # input pointer for this element
    x_base = x_ptr + b * stride_x_b + i * stride_x_i + j * stride_x_j
    # load input vector
    # compute dot in loop
    left_proj_acc = tl.zeros([H], dtype=tl.float16)
    right_proj_acc = tl.zeros([H], dtype=tl.float16)
    left_gate_acc = tl.zeros([H], dtype=tl.float16)
    right_gate_acc = tl.zeros([H], dtype=tl.float16)

    for c_start in range(0, C, BLOCK_C):
        c_off = tl.arange(0, BLOCK_C) + c_start
        mask_c = c_off < C
        # load x
        x = tl.load(x_base + c_off * stride_x_c, mask=mask_c, other=0.0)  # shape (BLOCK_C)
        # reshape to (1, BLOCK_C)
        x_vec = x[None, :]  # shape (1,BLOCK_C)

        # Load left_proj weights
        w_left = tl.load(left_proj_w_ptr + c_off[:, None] * stride_w_c + tl.arange(0, H)[None,:] * stride_w_h,
                         mask=mask_c[:, None], other=0.0)  # shape (BLOCK_C, H)
        left_proj_acc += tl.dot(x_vec, w_left)[0]

        # Load right_proj weights
        w_right = tl.load(right_proj_w_ptr + c_off[:, None] * stride_w_c + tl.arange(0, H)[None,:] * stride_w_h,
                          mask=mask_c[:, None], other=0.0)
        right_proj_acc += tl.dot(x_vec, w_right)[0]

        # Load left_gate weights
        w_lgate = tl.load(left_gate_w_ptr + c_off[:, None] * stride_w_c + tl.arange(0, H)[None,:] * stride_w_h,
                          mask=mask_c[:, None], other=0.0)
        left_gate_acc += tl.dot(x_vec, w_lgate)[0]

        # Load right_gate weights
        w_rgate = tl.load(right_gate_w_ptr + c_off[:, None] * stride_w_c + tl.arange(0, H)[None,:] * stride_w_h,
                          mask=mask_c[:, None], other=0.0)
        right_gate_acc += tl.dot(x_vec, w_rgate)[0]

    # Apply mask if needed.
    if mask_ptr != 0:
        mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)
        left_proj_acc = left_proj_acc * mask_val
        right_proj_acc = right_proj_acc * mask_val

    # Apply sigmoid gating
    left_gate_val = tl.sigmoid(left_gate_acc)
    right_gate_val = tl.sigmoid(right_gate_acc)
    left_final = left_proj_acc * left_gate_val
    right_final = right_proj_acc * right_gate_val

    # Store
    left_ptrs = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + tl.arange(0, H) * stride_left_h
    tl.store(left_ptrs, left_final)
    right_ptrs = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + tl.arange(0, H) * stride_right_h
    tl.store(right_ptrs, right_final)
```

"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

This file implements the forward pass of the *outgoing* TriMul module
used in AlphaFold‑3.  The heavy N³ contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

is realised as a custom batched GEMM written in Triton.  The surrounding
linear projections, gating, LayerNorms and the final projection are
performed with torch’s highly optimised cuBLAS kernels (mixed FP16/FP32)
so that the overall latency is dominated by the Triton kernel.

The algorithm
-------------
1️⃣  Input LayerNorm (float32 → float16)  
2️⃣  Two bias‑free linear projections (left/right) + two gate
    projections (left/right) computed with torch.nn.functional.linear.  
3️⃣  Optional pair‑wise mask (broadcast on the hidden dimension).  
4️⃣  Apply sigmoid gates and mask to the projected tensors.  
5️⃣  **Batched GEMM** – a Triton kernel computes
    `out = left @ rightᵀ` for every head independently
    (`batch = B·hidden_dim`).  The kernel follows the classic
    tiled‑matmul pattern and uses `tl.dot` (Tensor‑core friendly).  
6️⃣  Output LayerNorm (float32) and sigmoid output gate.  
7️⃣  Final linear projection back to the original channel dimension.  
8️⃣  Return a float32 tensor.

The implementation keeps the API identical to the reference PyTorch
module:

    custom_kernel((input, mask, weights, config)) → Tensor[B,N,N,dim]

and satisfies the “must use Triton” rule by providing the batched GEMM
kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # Pointers
    A_ptr,               # left  [batch, M, K] (FP16)
    B_ptr,               # right [batch, K, N] – will be accessed as transposed
    C_ptr,               # output[batch, M, N] (FP16)
    # Matrix sizes – compile‑time constants
    M: tl.constexpr,     # = N (seq_len)
    N: tl.constexpr,     # = N (seq_len)
    K: tl.constexpr,     # = N (seq_len)
    # Strides
    stride_am, stride_ak,            # A: row‑stride, col‑stride
    stride_br, stride_bc,            # B (transposed): row‑stride (= original col‑stride), col‑stride (= original row‑stride)
    stride_cm, stride_cn,            # C: row‑stride, col‑stride
    stride_batch_a, stride_batch_b, stride_batch_c,
    # Tile sizes (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices using a tiled Triton GEMM.
    A, B and C are FP16 tensors.  The batch dimension is encoded in
    program_id(0); program_id(1) and program_id(2) index the output tiles.
    """
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # ------------ tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                    # [BLOCK_K]

    # ------------ base pointers for this batch ---------------------------------
    a_batch = A_ptr + batch_id * stride_batch_a
    b_batch = B_ptr + batch_id * stride_batch_b
    c_batch = C_ptr + batch_id * stride_batch_c

    # ------------ pointer arithmetic for the current tile -----------------------
    a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_br) + (offs_n[None, :] * stride_bc)

    # ------------ accumulator ---------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ------------ loop over K dimension ----------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute masks for edge handling
        cur_k = k * BLOCK_K
        a_mask = (offs_m[:, None] < M) & (cur_k + offs_k[None, :] < K)
        b_mask = (cur_k + offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)          # (BLOCK_K, BLOCK_N)

        # dot‑product using Tensor‑cores
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_br

    # ------------ write back ----------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc)

    K = N
    N_dim = N

    # (batch, M, K) and (batch, K, N) – we will read B as transposed
    left_flat = left_t.view(batch, M, K)
    right_flat = right_t.view(batch, K, N_dim)   # original layout; will be accessed transposed

    # ------------------------------------------------------------
    # 2. Allocate output tensor
    # ------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------
    # 3. Gather strides (int64 → Python int is fine for Triton)
    # ------------------------------------------------------------
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)

    # Right matrix is accessed as Bᵀ:
    #   original strides: (batch, row, col) = (s_batch, s_row=N, s_col=1)
    #   after transposition we need:
    #       row‑stride = original col‑stride (=1)
    #       col‑stride = original row‑stride (=N)
    stride_br = right_flat.stride(2)   # = 1  (row stride of transposed view)
    stride_bc = right_flat.stride(1)   # = N  (col stride of transposed view)

    stride_cm = out_flat.stride(1)      # row stride (M)
    stride_cn = out_flat.stride(2)      # col stride (N)

    stride_batch_a = left_flat.stride(0)
    stride_batch_b = right_flat.stride(0)
    stride_batch_c = out_flat.stride(0)

    # ------------------------------------------------------------
    # 4. Launch Triton kernel
    # ------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_dim + BLOCK_N - 1) // BLOCK_N,
    )
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N_dim,
        K,
        stride_am,
        stride_ak,
        stride_br,
        stride_bc,
        stride_cm,
        stride_cn,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------
    # 5. Reshape back to the original layout
    # ------------------------------------------------------------
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)   # (B, N, N, H)
        .contiguous()
    )
    return out


# ----------------------------------------------------------------------
# Entry point – the forward pass of the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype=float32
        - mask_tensor  : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights_dict : dict with all linear / norm parameters
        - config_dict  : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (dtype=float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]  # H

    # ------------------------------------------------------------------
    # Helper to move weights to the correct device / dtype
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model parameters (all stored in fp32 in the checkpoint)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    left_proj_w  = to_fp16(weights["left_proj.weight"])      # (C, H)
    right_proj_w = to_fp16(weights["right_proj.weight"])     # (C, H)

    left_gate_w  = to_fp16(weights["left_gate.weight"])      # (C, H)
    right_gate_w = to_fp16(weights["right_gate.weight"])     # (C, H)
    out_gate_w   = to_fp16(weights["out_gate.weight"])       # (C, H)

    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )
    x_h = x_norm.to(torch.float16)                     # [B, N, N, C]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)                 # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)                # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – applied after masking
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_triton(left, right)                # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32 ↔ float16 conversion around it)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7️⃣ Gated projection back to the original channel dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, N, N, H)
    final = F.linear(gated, to_out_w)               # (B, N, N, C)  fp16

    # ------------------------------------------------------------------
    # 8️⃣ Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
