@triton.jit
def triton_matmul_batched_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    M: tl.constexpr, Ndim: tl.constexpr, Kdim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_batch = tl.program_id(0)  # combined batch*hidden dimension
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute actual batch and hidden indices
    # flatten BH: total_batch = B*H
    # batch index f = pid_batch
    # base offsets: compute B index and H index
    b = pid_batch // H
    h = pid_batch % H

    # compute base offsets for left, right, out using strides
    left_base = left_ptr + b*stride_left_batch + h*stride_left_h
    right_base = right_ptr + b*stride_right_batch + h*stride_right_h
    out_base = out_ptr + b*stride_out_batch + h*stride_out_h

    # compute i and j offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < Ndim

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over k dimension
    for k in range(0, tl.cdiv(Kdim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < Kdim

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,

"""
TriMul (outgoing) forward pass – Triton implementation.

Algorithm
---------
1️⃣  Layer‑norm the input (half‑precision) → `x`.
2️⃣  Linear projections (`left_proj`, `right_proj`) and three gate‑projections,
    all in fp16.
3️⃣  Optional mask: a tiny Triton kernel fuses   left = left * left_gate * mask
    and right = right * right_gate * mask   (mask broadcasted over the hidden
    dimension).
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is performed by a custom Triton batched GEMM that works on the original
    memory layout (no permute‑copy).  The kernel treats each pair (batch,
    hidden‑channel) as an independent GEMM, using Tensor‑Core `tl.dot`.
5️⃣  Second LayerNorm (hidden dimension) → out.
6️⃣  Out‑gate (sigmoid) multiplication.
7️⃣  Final linear projection back to the original dimension.
8️⃣  Cast output back to float32 (required by the harness).

Only the forward pass is provided – no gradients.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse elementwise multiplication with mask (broadcasted on hidden)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (flattened B·N·N)
    hidden_dim,     # i32   : hidden dimension H
    size,           # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < size

    # mask index = linear index // hidden_dim  (since hidden is innermost)
    mask_offs = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=active, other=0.0, dtype=tl.float16)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0, dtype=tl.float16)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0, dtype=tl.float16)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
#   left  : [B, N, N, H]  (float16)
#   right : [B, N, N, H]  (float16)
#   out   : [B, N, N, H]  (float16)
#   The kernel computes for each (b, h):
#       out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    # shape / config
    B, N, H,
    # strides for left
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    # strides for out
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program IDs (batch‑hidden, tile‑row, tile‑col)
    # ------------------------------------------------------------------
    batch_h_id = tl.program_id(0)        # 0 … B*H-1
    pid_m      = tl.program_id(1)        # tile row   0 … ceil(N/BLOCK_M)-1
    pid_n      = tl.program_id(2)        # tile col   0 … ceil(N/BLOCK_N)-1

    # ------------------------------------------------------------------
    # decode batch and hidden indices
    # ------------------------------------------------------------------
    b = batch_h_id // H
    h = batch_h_id % H

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # ------------------------------------------------------------------
    # block offsets (row = i, col = j)
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (float16 is enough – the values are summed over K in fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over the reduction dimension K (= N)
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # load A tile: left[b, i, k, h]  → shape (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0, dtype=tl.float16)

        # ------------------------------------------------------------------
        # load B tile transposed on‑the‑fly:
        #   we need right[b, j, k, h]  (j = column, k = reduction)
        #   for the dot we want B_T[k, j] = right[b, j, k, h]
        # ------------------------------------------------------------------
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=(mask_k[:, None] & mask_j[None, :]), other=0.0, dtype=tl.float16)

        # ------------------------------------------------------------------
        # accumulate block product
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # write the result
    # ------------------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that calls the Triton batched GEMM
# ----------------------------------------------------------------------
def _batched_matmul_triton(left: torch.Tensor,
                           right: torch.Tensor,
                           block_m: int = 128,
                           block_n: int = 128,
                           block_k: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication for tensors of shape [B,N,N,H] (half‑precision).
    Returns a tensor of the same shape with the contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides of the original tensors (contiguous in memory)
    s_batch_l, s_i_l, s_k_l, s_h_l = left.stride()
    s_batch_r, s_i_r, s_k_r, s_h_r = right.stride()
    s_batch_o, s_i_o, s_j_o, s_h_o = out.stride()

    total_batch = B * H
    grid = (total_batch,
            (N + block_m - 1) // block_m,
            (N + block_n - 1) // block_n)

    batched_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=int(B), N=int(N), H=int(H),
        stride_left_batch=int(s_batch_l),
        stride_left_i=int(s_i_l),
        stride_left_k=int(s_k_l),
        stride_left_h=int(s_h_l),
        stride_right_batch=int(s_batch_r),
        stride_right_i=int(s_i_r),
        stride_right_k=int(s_k_r),
        stride_right_h=int(s_h_r),
        stride_out_batch=int(s_batch_o),
        stride_out_i=int(s_i_o),
        stride_out_j=int(s_j_o),
        stride_out_h=int(s_h_o),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,          # empirical choice for H100 Tensor‑Cores
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool) or None
        - weights_dict : dict of model parameters (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Cast everything to half‑precision early (except mask, which may be bool)
    # --------------------------------------------------------------
    # Model weights → fp16 (one‑time conversion)
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (half‑precision)
    # --------------------------------------------------------------
    # Cast input to fp16 before LayerNorm – small loss of numeric stability is acceptable
    inp_fp16 = inp.to(torch.float16)
    x = torch.nn.functional.layer_norm(
        inp_fp16,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim] (fp16)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) – still fp16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 4️⃣ Optional mask – fuse elementwise multiply with the mask
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask → fp16, contiguous (size B·N·N)
        mask_fp16 = mask.to(torch.float16).contiguous()
        total_elems = left.numel()                         # B·N·N·hidden_dim
        BLOCK = 1024                                        # ≤ 1024 threads per block
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple elementwise multiplication
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via custom Triton batched GEMM
    # --------------------------------------------------------------
    out = _batched_matmul_triton(left, right,
                                 block_m=128, block_n=128, block_k=32)   # [B,N,N,hidden] fp16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden dimension) – fp16
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate (sigmoid) multiplication
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the evaluation harness) and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
