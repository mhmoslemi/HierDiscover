"""
Outgoing TriMul kernel (AlphaFold‑3) – forward only.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣ Four linear projections (left / right) + three gating linear layers (FP16).
3️⃣ If a mask is present, fuse mask multiplication and gating for *both* left and right
   in a single Triton kernel (in‑place, reduces memory traffic).
4️⃣ Heavy O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM: C = A @ Bᵀ where A,B ∈ ℝ^{(B·H)×N×N}.
5️⃣ Second LayerNorm (FP16) → element‑wise out‑gate (sigmoid) → final linear
   projection back to the original `dim`.
6️⃣ Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is performed in Triton; the rest uses
high‑performance PyTorch kernels (FP16 Tensor‑core GEMM, LayerNorm, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * mask * left_gate
#   right ← right * mask * right_gate
# Tensors are contiguous FP16 with shapes:
#   left/right/left_gate/right_gate : [B, N, N, H]
#   mask                           : [B, N, N]   (broadcasted over H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i32   (B·N·N·H)
    H,                 # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat indices
    mask = offs < total_elements                                 # OOB guard

    # Compute the mask index (broadcast across hidden dimension)
    mask_offs = offs // H

    # Load data (masked load returns 0 for OOB)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask (0/1). For OOB positions we give a neutral value 1.0.
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Write back in‑place
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for A,B ∈ ℝ^{B×N×N×H}
    using a batched GEMM where the batch dimension is (B·H).

    Returns:
        C of shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # Collapse batch & hidden dims → (B·H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (FP16 Tensor‑core GEMM)
    C = torch.bmm(A, B_.transpose(1, 2))

    # Restore original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, dim] (float32)
        torch.Tensor,          # mask tensor   [B, N, N]  (bool/float) – may be dummy if nomask=True
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast linear weights to FP16 (same device as inputs)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Gating projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # used later

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if nomask:
        # No mask – just multiply by the gates
        left = left * left_gate
        right = right * right_gate
    else:
        # Convert mask to FP16 (0.0 / 1.0)
        mask_fp16 = mask_tensor.to(dtype_fp16)

        # Ensure contiguous layout for the Triton kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()
        BLOCK = 1024  # stable block size for mask‑gate fusion
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gate & mask tensors are no longer needed
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched FP16 GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)   # [B, N, N, H] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply final out‑gate and project back to original `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                # element‑wise gating, FP16
    out = F.linear(out, to_out_w)       # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by downstream code)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
