def gating_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.constexpr,
    size: tl.constexpr,
    BLOCK: tl.constexpr,
    DO_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # load values
    left_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(right_ptr + offs, right_out, mask=mask)

@triton.jit
def left_right_gate_mask_kernel(
    left_ptr,
    left_gate_ptr,
    right_ptr,
    right_gate_ptr,
    mask_ptr,
    hidden_dim,
    size,
    BLOCK,
    DO_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = offs < size

    left_val = tl.load(left_ptr + offs, mask=mask_valid, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask_valid, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask_valid, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask_valid, other=0.0)

    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    if DO_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)
    else:
        m = 1.0

    left_out = left_val * left_gate * m
    right_out = right_val * right_gate * m

    tl.store(left_ptr + offs, left_out, mask=mask_valid)
    tl.store(right_ptr + offs, right_out, mask=mask_valid)
```

"""
Custom Triton kernel implementation for the outgoing TriMul operation used in AlphaFold3.

Algorithm:
1. Layer‑norm the input (float32) → FP16.
2. Fuse the five linear projections (left_proj, right_proj, left_gate, right_gate, out_gate)
   into a single matrix‑multiply with concatenated weights (GPU GEMM, fp16).
3. Apply sigmoid on the three gate tensors.
4. Fuse element‑wise multiplication of left/right projections with the corresponding
   gates and optional mask using a custom Triton kernel (mask is broadcast across the
   hidden dimension). This eliminates separate kernels for masking and gating.
5. Perform the heavy O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   via a batched GEMM (torch.bmm) on tensors reshaped to [B*hidden_dim, N, N].
6. Second Layer‑norm (fp16), gating with out_gate, and final linear projection back
   to the original channel dimension.
7. Cast the result back to float32.

All heavy compute uses fp16 for Tensor‑Core acceleration; the Triton kernel
covers the gating/mask fusion, satisfying the “use Triton” requirement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused gate, mask and projection scaling for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def left_right_gate_mask_kernel(
    left_ptr,          # *fp16 left_proj (flattened)
    left_gate_ptr,     # *fp16 left_gate_raw (flattened)
    right_ptr,         # *fp16 right_proj (flattened)
    right_gate_ptr,    # *fp16 right_gate_raw (flattened)
    mask_ptr,          # *fp16 mask (flattened, length B*N*N) – ignored if DO_MASK=False
    hidden_dim,        # i32 hidden dimension size
    size,              # i64 total number of elements (B*N*N*hidden_dim)
    BLOCK: tl.constexpr,
    DO_MASK: tl.constexpr,  # compile‑time boolean
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load data
    left_val = tl.load(left_ptr + offs, mask=valid, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=valid, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Sigmoid via 1/(1+exp(-x))
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # Mask
    if DO_MASK:
        mask_index = offs // hidden_dim
        # Prevent OOB when offs >= size (mask_index could be out‑of‑range)
        mask_index = tl.where(valid, mask_index, 0)
        m = tl.load(mask_ptr + mask_index, mask=valid, other=1.0)
    else:
        m = 1.0

    # Fuse multiplication
    left_out = left_val * left_gate * m
    right_out = right_val * right_gate * m

    # Store in‑place
    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)

# ----------------------------------------------------------------------
# Helper – batched GEMM (FP16) for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] using a single
    batched matrix‑multiply (torch.bmm).  All inputs must be fp16 and
    contiguous.  The result will have dtype fp16.
    """
    B, N, _, H = left.shape  # left shape: (B,N,N,H)
    # reshape to [B*H, N, N] and perform A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))  # shape [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Entrypoint – the forward pass for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; may contain ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)  # [B, N, N, dim] fp16
    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 3️⃣ Concatenate projection & gate weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # Shape: (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    ).contiguous()

    # -----------------------------------------------------------------
    # 4️⃣ Single fused linear projection (FP16)
    # -----------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # -----------------------------------------------------------------
    # 5️⃣ Slice the combined tensor
    # -----------------------------------------------------------------
    H = hidden_dim
    left_proj      = combined[..., 0:H]
    right_proj     = combined[..., H:2*H]
    left_gate_raw  = combined[..., 2*H:3*H]
    right_gate_raw = combined[..., 3*H:4*H]
    out_gate_raw   = combined[..., 4*H:5*H]

    # -----------------------------------------------------------------
    # 6️⃣ Compute out_gate (sigmoid) – stays in FP16
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 7️⃣ Fuse left/right projection with gates (and optional mask) via Triton
    # -----------------------------------------------------------------
    # Flatten tensors for the kernel
    left_flat  = left_proj.reshape(-1)
    right_flat = right_proj.reshape(-1)
    left_gate_flat = left_gate_raw.reshape(-1)
    right_gate_flat = right_gate_raw.reshape(-1)

    # Prepare mask – always allocate a tensor; when nomask=True we set DO_MASK=False
    if nomask:
        do_mask = False
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy, never read
    else:
        do_mask = True
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.reshape(-1)  # shape [B*N*N]

    total_elems = left_flat.numel()
    BLOCK = 1024  # threads per block (tuned for H100)
    grid = (triton.cdiv(total_elems, BLOCK),)

    left_right_gate_mask_kernel[grid](
        left_flat,
        left_gate_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        DO_MASK=do_mask,
        num_warps=4,
    )

    # Reshape back to original 4‑D shape
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction – batched GEMM in fp16
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 🔟 Gating with out_gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)     # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🎯 Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul (Triangle‑Multiplicative‑Update) forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → FP16.
2️⃣  Fuse the five linear layers (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single matrix‑multiply with concatenated weights.
3️⃣  Compute the out‑gate (sigmoid) on the fly.
4️⃣  Fuse element‑wise multiplication of left/right projections with the
    corresponding sigmoid‑gates and an optional mask using a custom Triton
    kernel (one kernel handles both sides).  This eliminates separate mask and
    gate kernels.
5️⃣  Perform the O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    as a batched GEMM (torch.bmm) on tensors reshaped to [B*H, N, N].
6️⃣  Second Layer‑norm (FP16), element‑wise out‑gate, and final linear
    projection back to the original channel dimension.
7️⃣  Cast the result to float32 (the harness expects float32).

All heavy compute uses FP16 to exploit Tensor‑Cores on H100; the Triton kernel
covers the gating/mask fusion, satisfying the “use Triton” requirement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused left/right projection scaling with sigmoid‑gate and
# optional mask.  Operates on flattened tensors of shape
#   [B * N * N * hidden_dim].
# ----------------------------------------------------------------------
@triton.jit
def left_right_gate_mask_kernel(
    left_ptr,            # *fp16 – left projection (flattened, will be overwritten)
    left_gate_ptr,       # *fp16 – raw left‑gate values (flattened)
    right_ptr,           # *fp16 – right projection (flattened, will be overwritten)
    right_gate_ptr,      # *fp16 – raw right‑gate values (flattened)
    mask_ptr,            # *fp16 – mask (flattened)  – ignored if DO_MASK=False
    hidden_dim,          # i32   – size of the hidden dimension H
    size,                # i64   – total number of elements (B*N*N*H)
    BLOCK: tl.constexpr, # compile‑time tile size
    DO_MASK: tl.constexpr # compile‑time boolean, true iff a mask is present
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the four tensors
    l_val = tl.load(left_ptr + offs, mask=valid, other=0.0)
    l_gate_raw = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=valid, other=0.0)
    r_gate_raw = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Sigmoid (1/(1+exp(-x))) – works in fp16
    l_gate = 1.0 / (1.0 + tl.exp(-l_gate_raw))
    r_gate = 1.0 / (1.0 + tl.exp(-r_gate_raw))

    # Optional mask (broadcast over the hidden dimension)
    if DO_MASK:
        mask_idx = offs // hidden_dim
        mask_idx = tl.where(valid, mask_idx, 0)   # avoid OOB when offs >= size
        m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
    else:
        m = 1.0

    # Fuse multiplication
    l_out = l_val * l_gate * m
    r_out = r_val * r_gate * m

    # Write results back in‑place
    tl.store(left_ptr + offs,  l_out, mask=valid)
    tl.store(right_ptr + offs, r_out, mask=valid)


# ----------------------------------------------------------------------
# Helper – batched GEMM for the O(N³) contraction (FP16)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a single batched matrix‑multiply (torch.bmm).  Input tensors must be
    FP16, shape [B, N, N, H]; output is also FP16.
    """
    B, N, _, H = left.shape
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B*H, N, N]
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [B*H, N, N]
    c = torch.bmm(a, b.transpose(1, 2))                           # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]
    return out


# ----------------------------------------------------------------------
# Entrypoint – Forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial normalization
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)           # [B, N, N, dim], fp16
    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 3️⃣ Concatenate all five weight matrices (fp16) – one GEMM instead of five
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    ).contiguous()

    # -----------------------------------------------------------------
    # 4️⃣ Single fused linear projection (fp16)
    # -----------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*H)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    # -----------------------------------------------------------------
    # 5️⃣ Slice combined tensor into its constituent parts
    # -----------------------------------------------------------------
    H = hidden_dim
    left_proj      = combined[..., 0:H]
    right_proj     = combined[..., H:2*H]
    left_gate_raw  = combined[..., 2*H:3*H]
    right_gate_raw = combined[..., 3*H:4*H]
    out_gate_raw   = combined[..., 4*H:5*H]

    # -----------------------------------------------------------------
    # 6️⃣ Out‑gate (sigmoid)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)   # fp16, shape [B,N,N,H]

    # -----------------------------------------------------------------
    # 7️⃣ Fuse left/right projections with their gates and optional mask
    # -----------------------------------------------------------------
    # Flatten for Triton kernel
    left_flat  = left_proj.reshape(-1)
    right_flat = right_proj.reshape(-1)
    left_gate_flat  = left_gate_raw.reshape(-1)
    right_gate_flat = right_gate_raw.reshape(-1)

    if nomask:
        do_mask = False
        # Dummy mask tensor – never accessed because DO_MASK=False
        mask_flat = torch.empty(1, dtype=torch.float16, device=device)
    else:
        do_mask = True
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.reshape(-1)   # shape [B*N*N]

    total_elements = left_flat.numel()
    BLOCK = 1024  # threads per block (tuned for H100)
    grid = (triton.cdiv(total_elements, BLOCK),)

    left_right_gate_mask_kernel[grid](
        left_flat,
        left_gate_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        hidden_dim,
        total_elements,
        BLOCK=BLOCK,
        DO_MASK=do_mask,
        num_warps=4,
    )

    # Restore 4‑D shapes after in‑place scaling
    left  = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 8️⃣ O(N³) contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H], fp16

    # -----------------------------------------------------------------
    # 9️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # -----------------------------------------------------------------
    # 🔟 Gating with out_gate and final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                       # element‑wise gating (fp16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim], fp16

    # -----------------------------------------------------------------
    # 🎯 Cast back to float32 (expected by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
