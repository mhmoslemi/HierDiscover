@triton.jit
def trimul_compute_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # get program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # compute global offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    h_offset = pid_h * BLOCK_H

    # ranges
    i = i_offset + tl.arange(0, BLOCK_M)
    j = j_offset + tl.arange(0, BLOCK_N)
    h = h_offset + tl.arange(0, BLOCK_H)

    # masks for bounds
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load left tile
        left_off = (i[:, None, None] * stride_left_i
                    + k[None, :, None] * stride_left_k
                    + h[None, None, :] * stride_left_h)
        left = tl.load(left_ptr + left_off,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0)

        # Load right tile
        right_off = (j[:, None, None] * stride_right_i
                     + k[None, :, None] * stride_right_k
                     + h[None, None, :] * stride_right_h)
        right = tl.load(right_ptr + right_off,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                        other=0.0)

        # Unroll over K dimension
        for kk in range(0, BLOCK_K):
            # If k index out-of-bounds skip
            kk_valid = (k_start + kk) < N

            # left slice for this kk
            left_kk = left[:, kk, :]         # shape (BLOCK_M, BLOCK_H)
            right_kk = right[:, kk, :]       # shape (BLOCK_N, BLOCK_H)
            # multiplication and accumulate
            # broadcast left_kk over j dimension, right_kk over i dimension
            acc += tl.where(kk_valid,
                            left_kk[:, None, :] * right_kk[None, :, :],
                            0.0)

    # store result
    out_off = (i[:, None, None] * stride_out_i
               + j[None, :, None] * stride_out_j
               + h[None, None, :] * stride_out_h)
    # cast accumulator to fp16
    out_val = acc.to(tl.float16)
    tl.store(out_ptr + out_off,
             out_val,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
```

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    device = input_tensor.device

    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # get program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # compute starting offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    h_base = pid_h * BLOCK_H

    # indices
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    h = h_base + tl.arange(0, BLOCK_H)

    # masks
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over K dimension
    for k_base in range(0, N, BLOCK_K):
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # load left tile (M x K x H)
        # offsets shape broadcast to (BLOCK_M, BLOCK_K, BLOCK_H)
        left_offsets = (i[:, None, None] * stride_left_i
                       + k[None, :, None] * stride_left_k
                       + h[None, None, :] * stride_left_h)
        left = tl.load(left_ptr + left_offsets,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0)

        # load right tile (K x N x H)
        # offsets shape (BLOCK_K, BLOCK_N, BLOCK_H)
        right_offsets = (k[:, None, None] * stride_right_k
                         + j[None, :, None] * stride_right_i
                         + h[None, None, :] * stride_right_h)
        right = tl.load(right_ptr + right_offsets,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                        other=0.0)

        # now compute
        # ROW-wise loop over kk:
        for kk in range(0, BLOCK_K):
            # left contribution: shape (BLOCK_M, BLOCK_H)
            left_kk = left[:, kk, :]
            # right contribution: shape (BLOCK_N, BLOCK_H)
            right_kk = right[kk, :, :]
            # accumulate using broadcasting
            # left_kk[:, None, :] shape (M,1,H)
            # right_kk[None, :, :] shape (1,N,H)
            acc += left_kk[:, None, :] * right_kk[None, :, :]

    # store the result
    out_offsets = (i[:, None, None] * stride_out_i
                   + j[None, :, None] * stride_out_j
                   + h[None, None, :] * stride_out_h)
    tl.store(out_ptr + out_offsets,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
Highly‑optimized Triton implementation of the “outgoing” Triangle‑Multiplicative
Update (TriMul) forward pass (AlphaFold‑3).  

Algorithm
---------
1️⃣ Layer‑Norm the input (FP32 → FP16 for speed).
2️⃣ Four linear projections (left, right, three gates) computed with Torch’s
   fused FP16 linear.
3️⃣ Fuse the mask (if present) and gating using a tiny in‑place Triton mul
   kernel – eliminates temporary tensors.
4️⃣ TriMul contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is realised as a custom Triton GEMM‑style kernel that processes a tile of
   (i,j) together with a small hidden‑dim block.  The kernel loads a
   (BLOCK_M×BLOCK_K×BLOCK_H) tile from *left* and a (BLOCK_K×BLOCK_N×BLOCK_H)
   tile from *right*, performs the K‑reduction with vectorised broadcasted
   multiplication, and writes the result back as FP16.  This fuses all
   hidden‑dim channels and avoids launching 𝑩·𝐻 separate cuBLAS GEMMs.
5️⃣ Second Layer‑Norm (over the hidden dimension), out‑gate multiplication,
   final linear projection back to the original `dim`.
6️⃣ Cast the result back to float32 (required by the harness).

Only the element‑wise gating and the cubic contraction are performed with
Triton; the remaining operations stay in PyTorch (highly‑optimised FP16
kernels).  The kernel is written to respect the HW limits of an H100 (e.g.
small per‑thread tile sizes to keep the register usage low).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Pointwise in‑place multiplication (used for gating & optional mask)
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    out_ptr, mul_ptr, N,  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _pointwise_mul_inplace(x: torch.Tensor, y: torch.Tensor) -> None:
    """In‑place element‑wise multiplication: x ← x * y (both FP16, contiguous)."""
    total = x.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        x, y, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# TriMul contraction kernel:
#   out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]    (per‑hidden‑dim)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,                # fp16*  [N, N, H]
    right_ptr,               # fp16*  [N, N, H]
    out_ptr,                 # fp16*  [N, N, H]
    N, H,                    # matrix dimensions (int)
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,   # tile size in i‑direction
    BLOCK_N: tl.constexpr,   # tile size in j‑direction
    BLOCK_K: tl.constexpr,   # tile size in k‑reduction
    BLOCK_H: tl.constexpr,   # hidden‑dim block size
):
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile column (j)
    pid_h = tl.program_id(2)          # hidden‑block index

    # ------------------------------------------------------------------
    # Global offsets for this tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    h_base = pid_h * BLOCK_H

    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    h = h_base + tl.arange(0, BLOCK_H)

    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop: iterate over the contracted dimension
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----------- load left tile  (M, K, H) -----------------------
        left_off = (i[:, None, None] * stride_left_i
                    + k[None, :, None] * stride_left_k
                    + h[None, None, :] * stride_left_h)
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # shape (BLOCK_M, BLOCK_K, BLOCK_H)

        # ----------- load right tile (K, N, H) -----------------------
        right_off = (k[:, None, None] * stride_right_k
                     + j[None, :, None] * stride_right_i
                     + h[None, None, :] * stride_right_h)
        right = tl.load(
            right_ptr + right_off,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # shape (BLOCK_K, BLOCK_N, BLOCK_H)

        # ----------- reduction over K (blocked) --------------------
        # Unrolled loop over the inner K dimension (constexpr)
        for kk in range(0, BLOCK_K):
            # left slice: (M, H)
            left_kk = left[:, kk, :]          # [BLOCK_M, BLOCK_H]
            # right slice: (N, H)
            right_kk = right[kk, :, :]        # [BLOCK_N, BLOCK_H]
            # broadcast multiplication → (M, N, H)
            acc += left_kk[:, None, :] * right_kk[None, :, :]

    # ------------------------------------------------------------------
    # Write the computed tile back to global memory (FP16)
    # ------------------------------------------------------------------
    out_off = (i[:, None, None] * stride_out_i
               + j[None, :, None] * stride_out_j
               + h[None, None, :] * stride_out_h)
    tl.store(
        out_ptr + out_off,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)  # [B, N, N, dim]   fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to fp16 (on‑device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating values (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Fuse gating (and mask, if any) with an in‑place Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _pointwise_mul_inplace(left, left_gate)   # left ← left * left_gate
    _pointwise_mul_inplace(right, right_gate) # right ← right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy cubic contraction (TriMul) using a custom Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty_like(left, dtype=dtype_fp16)  # allocate [B,N,N,H]

    # Tile sizes – chosen to keep register pressure modest on H100
    BLOCK_M = 8   # rows per tile
    BLOCK_N = 8   # cols per tile
    BLOCK_K = 16  # reduction block
    BLOCK_H = 4   # hidden‑dim sub‑block

    # Strides for the *left* tensor (after slicing the batch dimension)
    # left_ptr will receive a tensor of shape [N, N, H] (no batch dim)
    stride_left_i = left.stride(1)  # N * H
    stride_left_k = left.stride(2)  # H
    stride_left_h = left.stride(3)  # 1

    stride_right_i = right.stride(1)  # N * H
    stride_right_k = right.stride(2)  # H
    stride_right_h = right.stride(3)  # 1

    stride_out_i = out.stride(1)   # N * H
    stride_out_j = out.stride(2)   # H
    stride_out_h = out.stride(3)   # 1

    # Grid: (i‑tiles, j‑tiles, hidden‑blocks)
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_h = (H + BLOCK_H - 1) // BLOCK_H
    grid = (grid_i, grid_j, grid_h)

    # Launch one kernel per batch element (B ≤ 2)
    for b in range(B):
        left_ptr = left[b]
        right_ptr = right[b]
        out_ptr = out[b]

        _trimul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            H,
            stride_left_i,
            stride_left_k,
            stride_left_h,
            stride_right_i,
            stride_right_k,
            stride_right_h,
            stride_out_i,
            stride_out_j,
            stride_out_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
            num_warps=4,
        )
    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden_dim) → out‑gate → final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the evaluation harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
