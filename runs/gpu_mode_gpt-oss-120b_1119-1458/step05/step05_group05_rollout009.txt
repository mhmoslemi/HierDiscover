import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel...
@triton.jit
def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr, hidden_dim, total_numel, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_numel

    # load
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)   # fp16
    g_raw = tl.load(gate_ptr + offs, mask=valid, other=0.0) # fp16

    # compute sigmoid
    # g_raw may be fp16, promote to fp32 for stability? Use tl.float16 in computation,
    # but tl.exp for fp16 may be slower; promote to fp32
    g_fp32 = g_raw.to(tl.float32)
    sigmoid = 1.0 / (1.0 + tl.exp(-g_fp32))

    # compute mask index
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = x * sigmoid.to(tl.float16) * mask_val

    tl.store(x_ptr + offs, out, mask=valid)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: for each hidden dimension compute
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Implemented as a reshape + torch.bmm (leverages cuBLAS).
    """
    B, N, _, H = left.shape
    # reshape
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

4️⃣ Fuse ‑ sigmoid, optional mask and element‑wise multiplication for the left
   and right tensors using a tiny Triton kernel
   (`mask_gate_fuse_kernel`).  This kernel computes
        out = proj * sigmoid(gate) * mask
   in‑place, avoiding extra temporaries.
5️⃣ Batched GEMM (cuBLAS) computes the O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
6️⃣ Apply a second LayerNorm (fp16), multiply by the out‑gate (sigmoid),
   and finally project back to the original channel dimension.
7️⃣ Cast the result back to float32 (the harness expects float32).

All heavy linear work is done in fp16 using Tensor‑Cores; only the two
LayerNorms stay in PyTorch.  At least one custom Triton kernel is used,
fulfilling the “must‑use‑Triton” requirement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (proj * sigmoid(gate) * mask) – in‑place on proj.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # *fp16  → projection tensor (flattened)
    gate_ptr,          # *fp16  → pre‑sigmoid gate tensor (flattened)
    mask_ptr,          # *fp16  → mask tensor (flattened, length = B*N*N)
    hidden_dim,        # i32   → hidden dimension H
    total_numel,       # i64   → total number of elements in proj (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_numel

    # Load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)   # fp16
    gate_raw = tl.load(gate_ptr + offs, mask=mask, other=0.0)  # fp16

    # Sigmoid(gate)  = 1 / (1 + exp(‑gate))
    # Compute in fp32 for numerical stability, convert back to fp16
    gate_fp32 = gate_raw.to(tl.float32)
    sigmoid = 1.0 / (1.0 + tl.exp(-gate_fp32))

    # Determine which mask entry belongs to this element.
    # The mask does *not* have a hidden dimension – each block of
    # ``hidden_dim`` consecutive proj entries shares the same mask entry.
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # fp16

    # Fuse everything
    out = proj * sigmoid.to(tl.float16) * mask_val

    # Write back in‑place (overwrites the original projection)
    tl.store(proj_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper – batched matrix multiply for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single batched GEMM (cuBLAS).

    All tensors are fp16 and contiguous.
    """
    B, N, _, H = left.shape
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                     # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass (no autograd).
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 – all following heavy work will be in half‑precision
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Concatenated linear projection (left/right + all three gates)
    # -----------------------------------------------------------------
    # Gather the five weight matrices (bias‑less Linear layers) and cast once
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # (5 * hidden_dim, dim) – row‑major as required by torch.nn.functional.linear
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0
    ).to(torch.float16)

    # Linear projection → shape [B, N, N, 5*hidden_dim] (fp16)
    proj = torch.nn.functional.linear(x, combined_weight)

    # -----------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor into the five components
    # -----------------------------------------------------------------
    h = hidden_dim
    left_proj   = proj[..., 0:h]               # [B, N, N, h]
    left_gate   = proj[..., h:2 * h]           # pre‑sigmoid
    right_proj  = proj[..., 2 * h:3 * h]
    right_gate  = proj[..., 3 * h:4 * h]
    out_gate_pr = proj[..., 4 * h:5 * h]       # pre‑sigmoid

    # -----------------------------------------------------------------
    # 5️⃣ Fuse sigmoid, mask and element‑wise multiplication (Triton)
    # -----------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=torch.float16,
            device=device,
        )
    else:
        mask_fp16 = mask_tensor.to(torch.float16)

    mask_flat = mask_fp16.contiguous().view(-1)          # [B*N*N]
    total_el = left_proj.numel()                         # B*N*N*hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_el, BLOCK),)

    # left side
    mask_gate_fuse_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        hidden_dim,
        total_el,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right side
    mask_gate_fuse_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        hidden_dim,
        total_el,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Compute out‑gate (sigmoid) – no mask needed
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_pr)   # [B, N, N, h] (fp16)

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left_proj, right_proj)   # [B, N, N, h] (fp16)

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )   # [B, N, N, h] (fp16)

    # -----------------------------------------------------------------
    # 9️⃣ Gating with out‑gate and final projection back to ``dim``
    # -----------------------------------------------------------------
    out = out * out_gate                                      # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)     # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)           # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (the benchmark harness expects fp32)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
