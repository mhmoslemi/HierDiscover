import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton mask-gate fusion kernel
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,
    gate_ptr,
    mask_ptr,
    size,
    hidden_dim,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    x = tl.load(x_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # broadcast mask per hidden_dim
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=tl.float16(1.0))

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: for each head h compute left[b,i,:,h] @ right[b,j,:,h]^T.
    left/right shape: [B, N, N, H] (FP16)
    Returns tensor of same shape (FP16).
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]

    """
    Forward pass for the outgoing TriMul operator (AlphaFold3).

    The implementation fuses:
      • The five linear projections (left_proj, right_proj, left_gate, right_gate,
        out_gate) into a single GEMM.
      • The mask‑gate element‑wise product for the left/right tensors via a
        custom Triton kernel.
    Heavy contraction is performed with a batched GEMM (torch.bmm) which
    maps to cuBLAS Tensor‑Core kernels on H100.

    Arguments
    ---------
    data : tuple
        (x, mask, weights, config)
        - x : Tensor [B, N, N, C] (float32)
        - mask : Tensor [B, N, N] (float32 or bool)
        - weights : dict of model weights (float32)
        - config : dict must contain `dim` (C) and `hidden_dim` (c)

    Returns
    -------
    Tensor [B, N, N, C] (float32)
    """
    # Unpack inputs
    x, mask, weights, config = data
    dim = config["dim"]          # input channels: c_z
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = x.device

    # 1️⃣ First LayerNorm (FP32 → FP16)
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.half()   # cast only once

    # 2️⃣ Fuse all five linear layers into one GEMM.
    # Concatenate weight matrices along the output dimension.
    weight_concat = torch.cat([
        weights["left_proj.weight"],     # (H, dim)
        weights["right_proj.weight"],    # (H, dim)
        weights["left_gate.weight"],     # (H, dim)
        weights["right_gate.weight"],    # (H, dim)
        weights["out_gate.weight"],      # (H, dim)
    ], dim=0).half()  # shape (5*H, dim)

    # Project once.
    all_proj = torch.nn.functional.linear(x, weight_concat)  # [B, N, N, 5*H] FP16

    # Split the concatenated result.
    H = hidden_dim
    left_proj = all_proj[..., 0            : H]
    right_proj = all_proj[..., H           : 2*H]
    left_gate_pre = all_proj[..., 2*H      : 3*H]
    right_gate_pre = all_proj[..., 3*H     : 4*H]
    out_gate_pre = all_proj[..., 4*H       : 5*H]

    # Apply sigmoid to gating tensors.
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # 3️⃣ Mask‑gate fuse (or plain elementwise multiply when mask is omitted).
    if nomask:
        # No mask → just element‑wise gating.
        left = left_proj.mul_(left_gate)   # in‑place to keep memory low
        right = right_proj.mul_(right_gate)
    else:
        # Prepare flattened mask (broadcast over hidden_dim inside the kernel).
        mask_fp16 = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        total_elements = left_proj.numel()          # B*N*N*H
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Re‑use the tensors after fusion.
        left = left_proj
        right = right_proj

    # Release now‑unused tensors to free memory.
    del left_proj, right_proj, left_gate, right_gate, left_gate_pre, right_gate_pre

    # 4️⃣ Heavy O(N³) contraction via batched GEMM.
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H] FP16

    # 5️⃣ Second LayerNorm (over hidden_dim) – keep FP16.
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # 6️⃣ Out‑gate multiplication and final projection.
    out = out * out_gate                           # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())  # -> [B,N,N,dim] FP16

    # 7️⃣ Cast back to FP32 for the caller.
    return out.float()

"""
TriMul (outgoing) forward pass – fused five‑linear projections, mask‑gate
fusion (Triton) and a batched GEMM for the O(N³) contraction.
All heavy arithmetic runs in FP16; the final tensor is cast back to FP32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (tensor * gate * mask) fusion.
# The mask is broadcast over the hidden dimension `hidden_dim`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,           # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,     # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by the block
    mask = offs < size                         # boundary mask

    # Load values
    x = tl.load(x_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Broadcast mask: every `hidden_dim` consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=tl.float16(1.0))

    # Fused multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM (left @ rightᵀ) for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a strided
    batched GEMM (cuBLAS Tensor‑Core).  left/right are FP16 tensors of shape
    [B, N, N, H] and the result has the same shape.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and compute A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                     # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",

            total_elements,

            BLOCK=BLOCK,
            num_warps=4,
        )

        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left, right = left_proj, right_proj

    # Release temporaries that are no longer needed
    del left_proj, right_proj, left_gate, right_gate, left_gate_pre, right_gate_pre

    # ------------------------------------------------------------------
    return out.float()
