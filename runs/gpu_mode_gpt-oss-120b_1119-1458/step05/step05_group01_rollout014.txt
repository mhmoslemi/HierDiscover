```
@triton.jit
def fuse_kernel(
    combined_ptr, mask_ptr, left_ptr, right_ptr, out_gate_ptr,
    B, N, H,
    stride_combined_b, stride_combined_i, stride_combined_j, stride_combined_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_gate_b, stride_out_gate_i, stride_out_gate_j, stride_out_gate_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of h tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Compute combined base pointer offsets
        # Combined tensor index for a given projection/gate:
        # base = pid_b * stride_combined_b + offs_i[:, None, None] * stride_combined_i + offs_j[None, :, None] * stride_combined_j + offs_h[None, None, :] * stride_combined_h

        # For left_proj raw (offset 0)
        proj_left_ptrs = combined_ptr + pid_b * stride_combined_b + \
            offs_i[:, None, None] * stride_combined_i + \
            offs_j[None, :, None] * stride_combined_j + \
            offs_h[None, None, :] * stride_combined_h

        left_proj_raw = tl.load(
            proj_left_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        # left_gate raw at offset H
        proj_left_gate_ptrs = proj_left_ptrs + H * stride_combined_h
        left_gate_raw = tl.load(
            proj_left_gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        # right_proj raw at offset 2*H
        proj_right_ptrs = proj_left_ptrs + 2 * H * stride_combined_h
        right_proj_raw = tl.load(
            proj_right_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        # right_gate raw at offset 3*H
        proj_right_gate_ptrs = proj_left_ptrs + 3 * H * stride_combined_h
        right_gate_raw = tl.load(
            proj_right_gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        # out_gate raw at offset 4*H
        proj_out_gate_ptrs = proj_left_ptrs + 4 * H * stride_combined_h
        out_gate_raw = tl.load(
            proj_out_gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # compute sigmoid gates
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # mask factor
        if HAS_MASK:
            mask_tile = tl.load(
                mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j,
                mask=mask_i[:, None] & mask_j[None, :],
                other=0.0,
            )
            mask_tile = mask_tile[:, :, None]  # broadcast H
        else:
            mask_tile = 1.0

        # left = left_proj_raw * left_gate * mask
        left_tile = left_proj_raw * left_gate * mask_tile
        # right = right_proj_raw * right_gate * mask
        right_tile = right_proj_raw * right_gate * mask_tile

        # store left, right, out_gate
        left_ptrs = left_ptr + pid_b * stride_left_b + offs_i[:, None, None] * stride_left_i + offs_j[None, :, None] * stride_left_j + offs_h[None, None, :] * stride_left_h
        right_ptrs = right_ptr + pid_b * stride_right_b + offs_i[:, None, None] * stride_right_i + offs_j[None, :, None] * stride_right_j + offs_h[None, None, :] * stride_right_h
        out_gate_ptrs = out_gate_ptr + pid_b * stride_out_gate_b + offs_i[:, None, None] * stride_out_gate_i + offs_j[None, :, None] * stride_out_gate_j + offs_h[None, None, :] * stride_out_gate_h

        tl.store(left_ptrs, left_tile, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
        tl.store(right_ptrs, right_tile, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
        tl.store(out_gate_ptrs, out_gate, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
```

@triton.jit
def fused_proj_gate_kernel(
    combined_ptr, mask_ptr, left_ptr, right_ptr, out_gate_ptr,
    B, N, H,
    stride_combined_b, stride_combined_i, stride_combined_j, stride_combined_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_gate_b, stride_out_gate_i, stride_out_gate_j, stride_out_gate_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    i_valid = offs_i < N
    j_valid = offs_j < N
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        h_valid = offs_h < H
        # indices for combined raw slices
        # layout: left_proj (0:H), left_gate (H:2H), right_proj (2H:3H), right_gate (3H:4H), out_gate (4H:5H)
        # compute base offset for left_proj (0)
        base = pid_b * stride_combined_b + \
               offs_i[:, None, None] * stride_combined_i + \
               offs_j[None, :, None] * stride_combined_j + \
               offs_h[None, None, :] * stride_combined_h
        left_proj = tl.load(
            combined_ptr + base,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        left_gate = tl.load(
            combined_ptr + base + H * stride_combined_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right_proj = tl.load(
            combined_ptr + base + 2 * H * stride_combined_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right_gate = tl.load(
            combined_ptr + base + 3 * H * stride_combined_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            combined_ptr + base + 4 * H * stride_combined_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        # gating activation sigmoid
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate_val = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # mask factor
        if HAS_MASK:
            mask_tile = tl.load(
                mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j,
                mask=i_valid[:, None] & j_valid[None, :],
                other=0.0,
            )
            mask_factor = mask_tile[:, :, None]
        else:
            mask_factor = tl.full((BLOCK_I, BLOCK_J, 1), 1.0, dtype=tl.float16)

        # compute left,right
        left_tile = left_proj * left_gate * mask_factor
        right_tile = right_proj * right_gate * mask_factor

        # store
        tl.store(
            left_ptr + pid_b * stride_left_b + offs_i[:, None, None] * stride_left_i + offs_j[None, :, None] * stride_left_j + offs_h[None, None, :] * stride_left_h,
            left_tile,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr + pid_b * stride_right_b + offs_i[:, None, None] * stride_right_i + offs_j[None, :, None] * stride_right_j + offs_h[None, None, :] * stride_right_h,
            right_tile,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            out_gate_ptr + pid_b * stride_out_gate_b + offs_i[:, None, None] * stride_out_gate_i + offs_j[None, :, None] * stride_out_gate_j + offs_h[None, None, :] * stride_out_gate_h,
            out_gate_val,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    device = input_tensor.device

    # 1. LayerNorm x (float32)
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # shape [B,N,N,dim], dtype float32
    x = x.to(torch.float16)  # to fp16 for rest

    # 2. Combine projection weights
    # All projection weights have shape [hidden_dim, dim] (fp32)
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0)  # shape [5*hidden_dim, dim]
    combined_weight = combined_weight.to(device=device, dtype=torch.float16)

    # 3. Compute combined projection
    combined = torch.nn.functional.linear(x, combined_weight) # shape [B,N,N,5*hidden_dim], dtype fp16

    # 4. Gate and mask fusion using Triton
    if not nomask:
        mask_fp16 = mask_tensor.to(device=device, dtype=torch.float16)  # [B,N,N], broadcast
    else:
        # still need a placeholder for mask (unused)
        mask_fp16 = mask_tensor  # dummy, but not used inside kernel

    left, right, out_gate = _fused_proj_gate(combined, mask_fp16, nomask, hidden_dim)  # function returns those

    # combined tensor can be freed (let GC collect) via del combined maybe
    del combined

    # 5. O(N^3) contraction batched GEMM
    out = _batched_matmul_transpose_fp16(left, right)  # [B,N,N,hidden_dim] fp16

    # 6. Second layernorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # 7. Apply out_gate gating
    out = out * out_gate

    # 8. Final linear projection to dim
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim] fp16

    # 9. Return float32 output
    return out.to(torch.float32)

```
def _batched_matmul_transpose_fp16(left, right):
    B,N,_,H = left.shape
    # combine batch and hidden
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out
```

"""
TriMul (outgoing) forward pass – highly‑optimized Triton + PyTorch implementation.

Algorithm
---------
1️⃣  Layer‑norm the input (FP32) → cast to FP16.
2️⃣  Fuse the five linear layers (left_proj, left_gate, right_proj, right_gate, out_gate)
    into a **single** FP16 matrix multiply:   X @ W⁽ᵀ⁾   where
    W = [left_proj; left_gate; right_proj; right_gate; out_gate] (5·H × dim).
3️⃣  Fuse mask‑application, sigmoid‑gates and projection‐splitting with a Triton
    kernel that reads the combined projection tensor and writes
    * left  = left_proj * sigmoid(left_gate) * mask
    * right = right_proj * sigmoid(right_gate) * mask
    * out_gate = sigmoid(out_gate_raw)          (mask is optional).
4️⃣  Heavy O(N³) contraction is a batched GEMM:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  ⇔  left @ rightᵀ
    performed with torch.bmm on a (B·H)‑batch of NxN matrices (FP16).
5️⃣  Second LayerNorm on the hidden dimension (FP16).
6️⃣  Apply the out‑gate (element‑wise) and a final linear projection back to `dim`.
7️⃣  Cast the result to FP32 (the harness expects float32).

Only the mask‑gate‑fusion step is written in Triton; all large matrix
multiplications are delegated to cuBLAS tensor‑core kernels, giving
sub‑millisecond runtimes for the benchmark shapes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: split the combined projection tensor and apply
# mask + sigmoid gates.
# ----------------------------------------------------------------------
@triton.jit
def _fused_proj_gate_kernel(
    # inputs
    combined_ptr,                 # *fp16   [B, N, N, 5*H]
    mask_ptr,                     # *fp16   [B, N, N]   (unused if HAS_MASK == 0)
    # outputs
    left_ptr,                     # *fp16   [B, N, N, H]
    right_ptr,                    # *fp16   [B, N, N, H]
    out_gate_ptr,                 # *fp16   [B, N, N, H]
    # meta‑parameters
    B, N, H,
    stride_comb_b, stride_comb_i, stride_comb_j, stride_comb_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_og_b, stride_og_i, stride_og_j, stride_og_h,
    # compile‑time constants
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Split the combined projection tensor [5·H] into left/right
    projections and their gates, apply a mask (if present) and
    write the three resulting tensors.
    """
    pid_b = tl.program_id(0)   # batch axis
    pid_i = tl.program_id(1)   # i‑tile axis
    pid_j = tl.program_id(2)   # j‑tile axis

    # tile coordinates
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of hidden‑dim tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)

    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # base pointer for the left‑proj slice (offset 0)
        base = (
            pid_b * stride_comb_b
            + offs_i[:, None, None] * stride_comb_i
            + offs_j[None, :, None] * stride_comb_j
            + offs_h[None, None, :] * stride_comb_h
        )
        # ----- load the five raw tensors -----
        left_proj = tl.load(
            combined_ptr + base,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        left_gate = tl.load(
            combined_ptr + base + H * stride_comb_h,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        right_proj = tl.load(
            combined_ptr + base + 2 * H * stride_comb_h,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        right_gate = tl.load(
            combined_ptr + base + 3 * H * stride_comb_h,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            combined_ptr + base + 4 * H * stride_comb_h,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----- apply sigmoid gates -----
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # ----- optional mask loading -----
        if HAS_MASK:
            mask_tile = tl.load(
                mask_ptr
                + pid_b * stride_mask_b
                + offs_i[:, None] * stride_mask_i
                + offs_j[None, :] * stride_mask_j,
                mask=mask_i[:, None] & mask_j[None, :],
                other=0.0,
            )                      # (BLOCK_I, BLOCK_J)
            mask_factor = mask_tile[:, :, None]   # broadcast over H
            left_tile = left_proj * left_gate * mask_factor
            right_tile = right_proj * right_gate * mask_factor
        else:
            left_tile = left_proj * left_gate
            right_tile = right_proj * right_gate

        # ----- store results -----
        store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]

        tl.store(
            left_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_j[None, :, None] * stride_left_j
            + offs_h[None, None, :] * stride_left_h,
            left_tile,
            mask=store_mask,
        )
        tl.store(
            right_ptr
            + pid_b * stride_right_b
            + offs_i[:, None, None] * stride_right_i
            + offs_j[None, :, None] * stride_right_j
            + offs_h[None, None, :] * stride_right_h,
            right_tile,
            mask=store_mask,
        )
        tl.store(
            out_gate_ptr
            + pid_b * stride_og_b
            + offs_i[:, None, None] * stride_og_i
            + offs_j[None, :, None] * stride_og_j
            + offs_h[None, None, :] * stride_og_h,
            out_gate,
            mask=store_mask,
        )


# ----------------------------------------------------------------------
# Helper: fuse the combined projection + mask + gates.
# ----------------------------------------------------------------------
def _fuse_proj_gate(
    combined: torch.Tensor,
    mask: torch.Tensor,
    nomask: bool,
    hidden_dim: int,
) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Input:
        combined : [B, N, N, 5·H]  (fp16)
        mask     : [B, N, N]      (fp16, may be ignored)
        nomask   : if True → mask is not applied
        hidden_dim (H)

    Returns:
        left   : [B, N, N, H] (fp16)
        right  : [B, N, N, H] (fp16)
        out_gate: [B, N, N, H] (fp16)
    """
    B, N, _, five_h = combined.shape
    H = hidden_dim
    assert five_h == 5 * H, "combined tensor does not match hidden_dim"

    left = torch.empty((B, N, N, H), dtype=combined.dtype, device=combined.device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # strides in *elements*
    s_comb = combined.stride()
    s_mask = mask.stride()
    s_left = left.stride()
    s_right = right.stride()
    s_og = out_gate.stride()

    # tile sizes – 32×32 tiles for (i,j) give good occupancy,
    # hidden‑dim tiles of 32 keep register pressure modest.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _fused_proj_gate_kernel[grid](
        combined,
        mask,
        left,
        right,
        out_gate,
        B,
        N,
        H,
        # strides for the combined tensor
        s_comb[0],
        s_comb[1],
        s_comb[2],
        s_comb[3],
        # strides for the mask tensor
        s_mask[0],
        s_mask[1],
        s_mask[2],
        # strides for outputs
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_og[0],
        s_og[1],
        s_og[2],
        s_og[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        HAS_MASK=0 if nomask else 1,
        num_warps=8,
        num_stages=4,
    )
    return left, right, out_gate


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a strided‑batched GEMM (FP16 inputs/outputs).
    """
    B, N, _, H = left.shape
    # Reshape to (B·H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ   (each GEMM is N×N × N×Nᵀ)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to FP16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x.to(torch.float16)  # continue in FP16

    # -------------------------------------------------
    # 2️⃣ Fuse the five linear layers into one matmul
    # -------------------------------------------------
    # All projection / gate matrices have shape (hidden_dim, dim)
    proj_weights = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )  # (5·hidden_dim, dim)
    proj_weights = proj_weights.to(device=device, dtype=torch.float16)

    # Combined projection → [B, N, N, 5·hidden_dim] (FP16)
    combined = torch.nn.functional.linear(x, proj_weights)  # no bias, fp16

    # -------------------------------------------------
    # 3️⃣ Mask + gate fusion (Triton)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(device=device, dtype=torch.float16)
    else:
        # mask will be ignored by the kernel when HAS_MASK == 0
        mask_fp16 = mask_tensor.to(device=device, dtype=torch.float16)

    left, right, out_gate = _fuse_proj_gate(combined, mask_fp16, nomask, hidden_dim)

    # Release the intermediate combined tensor early
    del combined
    torch.cuda.empty_cache()

    # -------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (FP16)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] FP16

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # FP16

    # -------------------------------------------------
    # 6️⃣ Apply out‑gate (element‑wise) and final projection
    # -------------------------------------------------
    out = out * out_gate  # gating after LN

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim] FP16

    # -------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
```
