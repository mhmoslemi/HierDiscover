@triton.jit
def gated_projection_kernel(
    out_ptr,        # fp16 output [B, N, N, H]
    comb_ptr,       # fp16 input [B, N, N, 2*H]
    mask_ptr,       # fp16 mask [B, N, N] maybe
    total_elems,    # total num elements in output: B*N*N*H
    hidden_dim: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elems
    # compute idx and h
    idx = offs // hidden_dim
    h = offs % hidden_dim

    # base offset in comb_ptr = idx * (2*hidden_dim)
    comb_base = idx * (2 * hidden_dim)
    proj_offset = comb_base + h
    gate_offset = comb_base + hidden_dim + h
    proj = tl.load(comb_ptr + proj_offset, mask=mask, other=0)
    gate_logit = tl.load(comb_ptr + gate_offset, mask=mask, other=0)
    gate = tl.sigmoid(gate_logit)
    # mask value
    if has_mask:
        mask_val = tl.load(mask_ptr + idx, mask=mask, other=0)
    else:
        mask_val = tl.full([1], 1.0, dtype=tl.float16)[0]  # maybe tl.full

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Define the fused gating kernel
@triton.jit
def fused_gate_mask_kernel(
    out_left_ptr, out_right_ptr,
    left_comb_ptr, right_comb_ptr,
    mask_ptr,
    total_elements,      # total number of elements = B * N * N * H
    hidden_dim: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Compute base idx for (b,i,k) and hidden index h
    # idx corresponds to which element in left/right output (i.e., output position)
    idx = offs // hidden_dim
    h = offs % hidden_dim

    # Compute offsets in combined tensors
    # Each (b,i,k) entry has 2*hidden_dim values in combined tensor: first hidden_dim for proj, next for gate
    # offset for proj: idx * (2*hidden_dim) + h
    # offset for gate: idx * (2*hidden_dim) + hidden_dim + h
    base = idx * (2 * hidden_dim)
    proj_off = base + h
    gate_off = base + hidden_dim + h

    # Load values
    left_proj = tl.load(left_comb_ptr + proj_off, mask=mask, other=tl.float16(0))
    left_gate_logit = tl.load(left_comb_ptr + gate_off, mask=mask, other=tl.float16(0))
    right_proj = tl.load(right_comb_ptr + proj_off, mask=mask, other=tl.float16(0))
    right_gate_logit = tl.load(right_comb_ptr + gate_off, mask=mask, other=tl.float16(0))

    left_gate = tl.sigmoid(left_gate_logit)
    right_gate = tl.sigmoid(right_gate_logit)
    left_val = left_proj * left_gate
    right_val = right_proj * right_gate

    if has_mask:
        # mask reference: one mask value per (b,i,k) -> same idx as idx
        mask_val = tl.load(mask_ptr + idx, mask=mask, other=tl.float16(0))
        left_val = left_val * mask_val
        right_val = right_val * mask_val

    tl.store(out_left_ptr + offs, left_val, mask=mask)
    tl.store(out_right_ptr + offs, right_val, mask=mask)

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def fused_gate_mask_kernel(
    out_left_ptr, out_right_ptr,           # *fp16 - output
    left_comb_ptr, right_comb_ptr,         # *fp16 - input with proj and gate logits
    mask_ptr,                              # *fp16 - mask (if has_mask)
    total_elements,                        # int - total number of output elements (B*N*N*hidden_dim)
    hidden_dim: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Map linear index of output element to (batch*seq*seq) idx and hidden index
    base_idx = offs // hidden_dim      # index within the (B,N,N) dimension
    hid = offs % hidden_dim            # hidden channel

    # Offsets within concatenated projection+gate buffer
    # For each (b,i,k) we have 2*hidden_dim values: [proj(0..H-1), gate(0..H-1)]
    comb_base = base_idx * (2 * hidden_dim)
    proj_off = comb_base + hid                # offset for projection value
    gate_off = comb_base + hidden_dim + hid    # offset for gate logit

    # Load left and right projection and gate logits
    left_proj = tl.load(left_comb_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    left_gate_logit = tl.load(left_comb_ptr + gate_off, mask=mask, other=tl.float16(0.0))
    right_proj = tl.load(right_comb_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    right_gate_logit = tl.load(right_comb_ptr + gate_off, mask=mask, other=tl.float16(0.0))

    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_logit))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_logit))

    left_val = left_proj * left_gate
    right_val = right_proj * right_gate

    if has_mask:
        # mask has size (B,N,N) => one value per base_idx
        m = tl.load(mask_ptr + base_idx, mask=mask, other=tl.float16(0.0))
        left_val = left_val * m
        right_val = right_val * m

    tl.store(out_left_ptr + offs, left_val, mask=mask)
    tl.store(out_right_ptr + offs, right_val, mask=mask)

def _fused_gate_and_mask(left_comb, right_comb, mask, hidden_dim, has_mask):
    """
    Applies sigmoid gates, optional mask and writes gated tensors.
    Returns (left, right) tensors, both shape [B, N, N, hidden_dim] in fp16.
    """
    assert left_comb.is_contiguous()
    assert right_comb.is_contiguous()
    B, N, _, _ = left_comb.shape
    # allocate outputs
    left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left_comb.device)
    right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=right_comb.device)

    total = B * N * N * hidden_dim
    BLOCK = 1024  # each program processes 1024 elements
    grid = ((total + BLOCK - 1) // BLOCK,)

    mask_ptr = mask if has_mask else left_comb  # dummy pointer when no mask
    fused_gate_mask_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_comb.data_ptr(),
        right_comb.data_ptr(),
        mask_ptr.data_ptr(),
        total,
        hidden_dim=hidden_dim,
        has_mask=1 if has_mask else 0,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Optimized forward pass for the outgoing TriMul module (no gradients).

    The implementation fuses:
      * Linear projection + gate projection into a single matmul.
      * Sigmoid gating + optional masking into a custom Triton kernel.
    The heavy cubic contraction is performed via a batched GEMM (torch.bmm)
    using FP16/Tensor‑core arithmetic.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    has_mask = not nomask

    # ---------- 1️⃣ First LayerNorm (FP32 for numeric stability) ----------
    # Equivalent to nn.LayerNorm(dim) applied to last dimension.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=norm_weight,
                                       bias=norm_bias)
    # Convert to half‑precision for subsequent ops.
    x = x.to(torch.float16)  # [B, N, N, dim]

    # ---------- 2️⃣ Fuse projection + gate for left and right ----------
    # Concatenate projection and gate weights → shape (2*hidden_dim, dim)
    left_concat_w = torch.cat([weights["left_proj.weight"],
                               weights["left_gate.weight"]], dim=0
                              ).to(device=device, dtype=torch.float16)
    right_concat_w = torch.cat([weights["right_proj.weight"],
                                weights["right_gate.weight"]], dim=0
                               ).to(device=device, dtype=torch.float16)

    left_comb = torch.nn.functional.linear(x, left_concat_w)   # [B,N,N,2*H] fp16
    right_comb = torch.nn.functional.linear(x, right_concat_w) # [B,N,N,2*H] fp16

    # ---------- 3️⃣ Compute out‑gate (sigmoid after a linear) ----------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_logit = torch.nn.functional.linear(x, out_gate_w)  # [B,N,N,hidden_dim] fp16
    out_gate = torch.sigmoid(out_gate_logit)  # fp16

    # ---------- 4️⃣ Fuse sigmoid‑gate, optional mask, and projection ----------
    if has_mask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        # dummy tensor (won't be accessed)
        mask_fp16 = left_comb  # any tensor with right size

    left, right = _fused_gate_and_mask(
        left_comb,
        right_comb,
        mask_fp16,
        hidden_dim,
        has_mask,
    )
    # At this point left & right are [B,N,N,hidden_dim] in fp16.

    # ---------- 5️⃣ Heavy O(N³) contraction (batched GEMM) ----------
    B, N, _, H = left.shape  # note H == hidden_dim
    # Reshape to [B*H, N, N] for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # [B*H, N, N]
    # Compute out_flat[b*h, i, j] = Σ_k left_flat[b*h,i,k] * right_flat[b*h,j,k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore original layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 6️⃣ Second LayerNorm, out‑gate scaling, final projection ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # element‑wise gating (fp16)

    # final linear projection back to original dimension
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] fp16

    # ---------- 7️⃣ Cast back to FP32 ----------
    return out.to(torch.float32)

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1️⃣ First LayerNorm is performed in FP32 for numerical stability, then the
   tensor is cast to FP16 for all subsequent heavy work.
2️⃣ For each side (left/right) we fuse the projection linear layer and the
   gating linear layer into a *single* matmul (weight concatenation).  The
   resulting tensor has shape [..., 2·hidden_dim] = [proj, gate_logit].
3️⃣ A custom Triton kernel fuses the sigmoid gate, optional mask multiplication
   and the final projection‑gate element‑wise product for **both** sides
   simultaneously, eliminating extra temporaries and extra kernel launches.
4️⃣ The cubic contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is exactly a batch of GEMMs (left @ rightᵀ).  We reshape the tensors to
   [B·hidden_dim, N, N] and invoke `torch.bmm`, letting cuBLAS/Tensor‑cores
   handle the heavy work.
5️⃣ A second LayerNorm (FP16) is applied, then the out‑gate (computed from the
   original normalized input) scales the hidden representation, followed by the
   final linear projection back to `dim`.
6️⃣ The output is returned in FP32 as required by the harness.

Only the element‑wise gating step is implemented in Triton, satisfying the
requirement to use a custom kernel while keeping the overall runtime low.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _fused_gate_mask_kernel(
    out_left_ptr, out_right_ptr,      # *fp16 – output tensors  [B,N,N,H]
    left_comb_ptr, right_comb_ptr,    # *fp16 – concatenated proj+gate logits [B,N,N,2H]
    mask_ptr,                         # *fp16 – mask tensor [B,N,N] (ignored if has_mask=0)
    total_elems,                      # int – B·N·N·H
    hidden_dim: tl.constexpr,         # compile‑time H
    has_mask: tl.constexpr,           # compile‑time flag
    BLOCK: tl.constexpr,              # block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # 1) Map linear index to (b,i,k) coordinate (named base_idx) and hidden channel (hid)
    base_idx = offs // hidden_dim      # index within the (B,N,N) dimension
    hid = offs % hidden_dim            # hidden dimension index

    # 2) Offsets inside the concatenated buffers (proj at [0:H), gate at [H:2H))
    combo_base = base_idx * (2 * hidden_dim)
    proj_off = combo_base + hid
    gate_off = combo_base + hidden_dim + hid

    # 3) Load the two sides' projection and gate logits
    left_proj = tl.load(left_comb_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    left_gate_logit = tl.load(left_comb_ptr + gate_off, mask=mask, other=tl.float16(0.0))
    right_proj = tl.load(right_comb_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    right_gate_logit = tl.load(right_comb_ptr + gate_off, mask=mask, other=tl.float16(0.0))

    # 4) Sigmoid gate (1/(1+exp(-x)))
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_logit))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_logit))

    # 5) Apply gate (and optional mask)
    left_val = left_proj * left_gate
    right_val = right_proj * right_gate

    if has_mask:
        # mask has one value per (b,i,k) → same base_idx as above
        m = tl.load(mask_ptr + base_idx, mask=mask, other=tl.float16(0.0))
        left_val = left_val * m
        right_val = right_val * m

    # 6) Store results
    tl.store(out_left_ptr + offs, left_val, mask=mask)
    tl.store(out_right_ptr + offs, right_val, mask=mask)


def _apply_gate_and_mask(
    left_comb: torch.Tensor,
    right_comb: torch.Tensor,
    mask: torch.Tensor,
    hidden_dim: int,
    has_mask: bool,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Executes the Triton kernel that fuses:
       sigmoid(gate) * projection * (mask ?)
    for *both* left and right sides.
    Returns two tensors of shape [B, N, N, hidden_dim] (FP16).
    """
    B, N, _, _ = left_comb.shape
    total = B * N * N * hidden_dim
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    # Allocate output buffers
    left_out = torch.empty((B, N, N, hidden_dim),
                           dtype=torch.float16, device=left_comb.device)
    right_out = torch.empty((B, N, N, hidden_dim),
                            dtype=torch.float16, device=right_comb.device)

    # When there is no mask we pass a dummy pointer (left_comb) – it will never be loaded
    dummy_ptr = left_comb if has_mask else left_comb

    _fused_gate_mask_kernel[grid](
        left_out.data_ptr(),
        right_out.data_ptr(),
        left_comb.data_ptr(),
        right_comb.data_ptr(),
        mask.data_ptr() if has_mask else dummy_ptr.data_ptr(),
        total,
        hidden_dim=hidden_dim,
        has_mask=1 if has_mask else 0,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor   [B, N, N]     (float32 or bool)
        - weights_dict : dict of named model parameters (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    has_mask = not nomask
    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(inp,
                                       (dim,),
                                       weight=norm_w,
                                       bias=norm_b)      # FP32
    x = x.to(torch.float16)                               # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (concatenate weights)
    # ------------------------------------------------------------------
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)               # (2H, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)               # (2H, dim)

    left_comb = torch.nn.functional.linear(x, left_concat_w)   # [B,N,N,2H] FP16
    right_comb = torch.nn.functional.linear(x, right_concat_w) # [B,N,N,2H] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Compute out‑gate (sigmoid of a linear layer)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_logits = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N,H] FP16
    out_gate = torch.sigmoid(out_gate_logits)                     # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Fuse sigmoid, optional mask, and projection (Triton)
    # ------------------------------------------------------------------
    if has_mask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        # dummy tensor – kernel will ignore it when has_mask==0
        mask_fp16 = left_comb

    left, right = _apply_gate_and_mask(
        left_comb,
        right_comb,
        mask_fp16,
        hidden_dim,
        has_mask,
    )   # both [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    # reshape to [B*H, N, N] – each slice performs a matmul
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)    # [B·H, N, N]
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # [B·H, N, N]

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h,i,k] * right_flat[b*h,j,k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out‑gate scaling + final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                 # FP16

    out = out * out_gate               # element‑wise gating (FP16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
