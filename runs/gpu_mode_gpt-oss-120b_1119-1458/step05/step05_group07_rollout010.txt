"""
Highly‑optimised “outgoing” TriMul implementation (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 (for numerical stability) and cast to FP16.
2️⃣  Fuse the two linear projections (proj + gate) for the left and right branches
    by concatenating the weight matrices.  A single torch.nn.functional.linear per
    branch yields two tensors `[B, N, N, 2*H]` that are then split into the projection
    and gate tensors.  The gate is passed through a sigmoid.
3️⃣  Fuse element‑wise multiplication with the optional mask in a tiny Triton kernel:
       `proj = proj * gate * mask`   (or `proj = proj * gate` if there is no mask).
4️⃣  Compute the heavy O(N³) contraction
       `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.
    This is equivalent to a batched GEMM `C = A @ Bᵀ` with batch size `B*H`.  The
    required memory layout is obtained by a single `permute(...).contiguous()` which
    re‑orders the data only once per branch.
5️⃣  Apply a second Layer‑Norm on the hidden dimension, multiply by the output‑gate
    (computed from the first‑normed tensor) and finish with a final linear projection
    back to the original feature dimension.
6️⃣  Cast the result back to FP32 for the test harness.

Only the forward pass is required – no autograd.  The Triton kernel is used for
the (proj × gate × mask) fusion; all large matrix multiplies are delegated to
cuBLAS via torch.bmm, which is optimal on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated (proj)
    gate_ptr,              # *fp16 : gate tensor (same shape as proj)
    mask_ptr,              # *fp16 : mask tensor, broadcasted over hidden_dim
    total_elts: tl.int32,  # total number of elements in the tensors (B*N*N*H)
    hidden_dim: tl.int32,  # hidden dimension H
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_elts

    # Index into the broadcasted mask (mask repeats every `hidden_dim` entries)
    mask_idx = offs // hidden_dim

    x_val   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x_val * gate_val * mask_val, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM (left @ rightᵀ) for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.Tensor [B, N, N, H] (FP16)
    Returns:       torch.Tensor [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape

    # Re‑order to [B*H, N, N] (contiguous matrices)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ -> (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))

    # Restore the original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) – ignored if config["nomask"] is True
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣  Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                         # [B, N, N, dim], FP32
    x = x_norm.to(torch.float16)   # cast once for the heavy part

    # ------------------------------------------------------------------
    # 2️⃣  Fuse proj + gate linear layers
    # ------------------------------------------------------------------
    # concatenate weights to form a single matmul per branch
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*H, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*H, dim)

    # (B,N,N,2*H) – linear without bias
    left_comb = torch.nn.functional.linear(x, left_combined_w)   # [B,N,N,2*H]
    right_comb = torch.nn.functional.linear(x, right_combined_w)

    # split into projection and gate halves
    left_proj = left_comb[..., :hidden_dim]               # [B,N,N,H]
    left_gate_pre = left_comb[..., hidden_dim:]           # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]

    # sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask – fuse element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcasted over the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()              # B * N * N * H
        BLOCK = 1024                                 # ≤1024 threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left_proj = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right_proj = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # no mask – cheap element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣  Compute out = Σₖ left[...,k,:] * right[...,k,:]  → batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul(left, right)   # [B, N, N, H]  (FP16)

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # [B, N, N, H]  FP16

    # ------------------------------------------------------------------
    # 6️⃣  Output gate (computed from the first‑normed tensor)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] FP16

    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  FP16

    # ------------------------------------------------------------------
    # 8️⃣  Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
