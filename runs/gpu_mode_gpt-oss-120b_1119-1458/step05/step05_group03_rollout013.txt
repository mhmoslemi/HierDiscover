def _batched_gemm_fp16(left, right):
    B,N,_,H = left.shape
    # reshape to [B*H, N, N] for left and right
    # left_permuted = left.permute(0,3,1,2).contiguous()  # shape (B,H,N,N)

    # view
    a = left_permuted.reshape(B*H, N, N)
    b = right_permuted.reshape(B*H, N, N)
    # compute batched matmul: a @ b.transpose(1,2) (or b.transpose(-2,...)?)

    # reshape back
    out = c.reshape(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

     the input tensor is read once instead of five times.
2️⃣   In the mask‑present case we fuse the three element‑wise steps
     (mask, gate sigmoid, multiplication) into one Triton kernel
     (`mask_gate_multi_kernel`).  The kernel works on a flat view of the
     tensors and broadcasts the mask over the hidden dimension.
3️⃣   The heavy O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (`_batched_gemm_fp16`).  For each head
   we compute `A @ Bᵀ` with cuBLAS – the fastest path on an H100.
4️⃣   The second Layer‑Norm, out‑gate, and final linear projection are
   carried out with native PyTorch kernels (FP16).  The final result is
   cast back to float32 for the harness.

Only the forward pass is required; no gradients are computed.
At least one custom Triton kernel is used (`mask_gate_multi_kernel`).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate sigmoid and element‑wise multiplication.
# Works on the flattened view of each tensor: [B*N*N*H] where H = hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_multi_kernel(
    left_ptr,          # *fp16 : left_proj buffer (output in‑place)
    left_gate_ptr,    # *fp16 : left_gate pre‑sigmoid buffer
    right_ptr,        # *fp16 : right_proj buffer (output in‑place)
    right_gate_ptr,   # *fp16 : right_gate pre‑sigmoid buffer
    out_gate_ptr,     # *fp16 : out_gate pre‑sigmoid (overwritten by sigmoid)
    mask_ptr,         # *fp16 : mask buffer of shape (B*N*N,) – broadcasted
    size,              # i32   : total number of elements (B*N*N*H)
    hidden_dim,        # i32   : size of hidden dimension H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load data
    left      = tl.load(left_ptr      + offs,      mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs,      mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs,     mask=mask, other=0.0)
    out_gate  = tl.load(out_gate_ptr  + offs,      mask=mask, other=0.0)

    # Broadcast mask over hidden_dim (every H consecutive positions share the same mask entry)
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)   # 1.0 = no‑mask

    # Sigmoid = 1/(1+exp(-x))
    left_gate_sig  = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate_sig   = 1.0 / (1.0 + tl.exp(-out_gate))

    # Apply mask, gate and store results in‑place
    left_out  = left  * left_gate_sig  * m
    right_out = right * right_gate_sig * m

    tl.store(left_ptr      + offs, left_out,  mask=mask)
    tl.store(right_ptr     + offs, right_out, mask=mask)
    tl.store(out_gate_ptr  + offs, out_gate_sig, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a
    batched GEMM (A @ Bᵀ) with cuBLAS.

    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H] (FP16).

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # Move H to the batch dimension: (B, H, N, N) → (B*H, N, N)
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    # GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.matmul(a, b.transpose(-2, -1))
    # Restore original layout
    out = c.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool, or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # input channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection dimension (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32
    x = x_norm.half()                 # cast to FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections into one GEMM
    # --------------------------------------------------------------
    # Gather weights (FP16)
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()

    # Concatenate:  [left_proj; left_gate; right_proj; right_gate; out_gate]
    all_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Linear projection (B,N,N,5*hidden_dim) in FP16
    proj_all = torch.nn.functional.linear(x, all_w)  # FP16

    # Split into the five components
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = proj_all.split(
        [hidden_dim] * 5, dim=-1
    )  # each: [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask / gating
    # --------------------------------------------------------------
    if nomask:
        # No mask: simple element‑wise ops (still in FP16)
        left_gate = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)
        out_gate = torch.sigmoid(out_gate_pre)

        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Prepare flattened views for the Triton kernel
        B, N, _, _ = left_proj.shape
        total_elements = left_proj.numel()  # B * N * N * hidden_dim

        # Ensure contiguous memory layout
        left_proj_flat    = left_proj.contiguous().view(-1)
        left_gate_pre_flat = left_gate_pre.contiguous().view(-1)
        right_proj_flat   = right_proj.contiguous().view(-1)
        right_gate_pre_flat = right_gate_pre.contiguous().view(-1)
        out_gate_pre_flat = out_gate_pre.contiguous().view(-1)

        # Mask (flattened) – broadcasted over hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # shape (B*N*N,)

        # Triton launch configuration
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # In‑place fusion: mask, sigmoid gates, and multiplication
        mask_gate_multi_kernel[grid](
            left_proj_flat,
            left_gate_pre_flat,
            right_proj_flat,
            right_gate_pre_flat,
            out_gate_pre_flat,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )

        # After kernel, the buffers contain the final tensors
        left = left_proj      # shape [B, N, N, hidden_dim] (already mutated)
        right = right_proj    # same shape
        out_gate = out_gate_pre  # now holds sigmoid(out_gate)

        # Free temporary views
        del (
            left_proj_flat,
            left_gate_pre_flat,
            right_proj_flat,
            right_gate_pre_flat,
            out_gate_pre_flat,
        )

    # --------------------------------------------------------------
    # 4️⃣ Heavy contraction (batched GEMM) – O(N³)
    # --------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim], FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # --------------------------------------------------------------
    # 6️⃣ Out‑gate gating + final linear projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate                       # element‑wise gating (FP16)
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].half()
    )  # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 7️⃣ Cast back to float32 for the harness
    # --------------------------------------------------------------
    return out.float()
```
