"""
TriMul – outgoing version (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) → keep FP32 for numerical exactness.
2️⃣ Four linear projections (left/right) + three gating projections
   (left_gate, right_gate, out_gate).  All are bias‑free as in the reference.
3️⃣ Optional mask handling:
   – When a mask is present we fuse `mask * left_gate` and `mask * right_gate`
     with the element‑wise multiplication `left = left * left_gate`,
     `right = right * right_gate` by a small Triton kernel.
   – When there is no mask we simply multiply by the gates in PyTorch.
4️⃣ Heavy O(N³) contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = A @ Bᵀ where A,B ∈
   ℝ^{(B·H)×N×N}.  We use `torch.bmm`, which on H100 is tensor‑core
   accelerated.
5️⃣ Second LayerNorm over the hidden dimension → element‑wise product
   with `out_gate`.
6️⃣ Final linear projection back to the original `dim`.
7️⃣ Cast back to float32 (the harness expects FP32).

A tiny Triton kernel is used for step 3 (mask + gate fusion),
fulfilling the “must contain a kernel” requirement while keeping the
implementation correct and reasonably fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp32  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp32  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp32  [B,N,N,H]
    right_gate_ptr,    # *fp32  [B,N,N,H]
    mask_ptr,          # *fp32  [B,N,N]    (broadcasted)
    total_elements,    # int    B·N·N·H
    H,                 # int    hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse mask and gate multiplications:

        left  ← left  * left_gate  * mask
        right ← right * right_gate * mask

    All tensors are contiguous float32.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)        # flat offsets
    mask = offs < total_elements                               # OOB guard

    # Corresponding mask index (broadcast over the hidden dimension)
    mask_offs = offs // H

    # Load operands (masked loads return 0.0 / 1.0 for OOB entries)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask (float32 0.0 or 1.0).  For OOB we use 1.0 (neutral element).
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store the result back in‑place
    tl.store(left_ptr  + offs, left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp32(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Implements the operation as a batched GEMM on the last two dimensions.
    Arguments are expected to be FP32 tensors of shape [B, N, N, H].
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    # Collapse batch and hidden dimensions: (B·H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    # Restore original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float, optional)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32, keep full precision)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # LayerNorm normalizes over the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (still float32)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # Projections → [B, N, N, hidden_dim]
    left  = F.linear(x, left_proj_w)    # bias=False in reference
    right = F.linear(x, right_proj_w)

    # Gating projections → [B, N, N, hidden_dim] after sigmoid
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # used later

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple elementwise gating
        left  = left * left_gate
        right = right * right_gate
    else:
        # Prepare mask as float32 (0.0 / 1.0) – broadcasting over the hidden dim
        mask_fp32 = mask_tensor.to(dtype=torch.float32)      # [B, N, N]
        # Ensure contiguous layout for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp32 = mask_fp32.contiguous()

        total = left.numel()                     # B·N·N·H
        BLOCK = 1024                             # ~1K threads per block (good occupancy)
        grid = ( (total + BLOCK - 1) // BLOCK, ) # 1‑D grid

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp32,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gate tensors are no longer needed
        del left_gate, right_gate, mask_fp32

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (float32)
    # ------------------------------------------------------------------
    out = _batched_matmul_fp32(left, right)   # [B, N, N, hidden_dim] (float32)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (float32) over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, hidden_dim] (float32)

    # ------------------------------------------------------------------
    # 6️⃣ Multiply by out_gate and project back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise (float32)
    out = F.linear(out, to_out_w)            # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # 7️⃣ Return float32 tensor (already FP32)
    # ------------------------------------------------------------------
    return out
