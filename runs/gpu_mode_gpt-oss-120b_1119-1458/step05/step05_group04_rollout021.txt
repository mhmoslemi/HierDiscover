    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Using torch.nn.functional.layer_norm (requires specifying normalized shape)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

        left_gate_w,
        right_proj_w,

        out_gate_w,

    B, N, _, _ = x.shape # x shape [B,N,N,dim]; we ignore 'dim' store?

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input in FP32 (numerical stability) and cast to FP16.
2️⃣  Fuse the five linear projections (left/right and their three gates) into a
    *single* matrix multiply by concatenating the weight matrices.
    This reads the input token‑wise only once and produces the five tensors
    (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`) in one
    pass.
3️⃣  Apply the sigmoid to the three gate tensors.
4️⃣  Fuse element‑wise gating **and** optional masking into a tiny Triton kernel.
    The kernel multiplies `proj *= gate * mask` in‑place, thus eliminating
    temporaries and extra launches.
5️⃣  Perform the cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    as a batched GEMM:
        left   → (B, H, N, N)   –‑contiguous
        right  → (B, H, N, N)   –‑contiguous
        out = torch.matmul(left, right.transpose(-1, -2))
6️⃣  Second Layer‑Norm (FP16) → multiply by the `out_gate`.
7️⃣  Final linear projection back to the original `dim`.
8️⃣  Cast the result to FP32 (the harness expects float32).

Only the element‑wise gating is written in Triton; all dense matmuls use
cuBLAS/Tensor‑Cores, which on an H100 gives > 60 TFLOP/s and easily satisfies
the sub‑millisecond budget for the provided test cases.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel – fused gate * mask (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mul_kernel(
    out_ptr,         # *fp16  – tensor to be multiplied in‑place
    gate_ptr,        # *fp16  – gate values (same shape as out)
    mask_ptr,        # *fp16  – 1‑D mask (B·N·N), broadcast over hidden dim
    TOTAL_ELEMS,     # total number of elements = B·N·N·H
    HIDDEN_DIM,      # hidden dimension (size of the last axis)
    MASK_LEN,        # length of the mask = B·N·N
    HAS_MASK: tl.constexpr,   # compile‑time flag (True → apply mask)
    BLOCK_SIZE: tl.constexpr, # block size (threads per program)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load data
    out_val  = tl.load(out_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Index of the corresponding mask entry (ignore hidden dim)
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_idx < MASK_LEN, other=1.0)
        out_val = out_val * gate_val * mask_val
    else:
        out_val = out_val * gate_val

    tl.store(out_ptr + offs, out_val, mask=mask)


def _gate_mul_inplace(
    tensor: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
    nomask: bool,
) -> None:
    """
    In‑place element‑wise multiplication:
        tensor ← tensor * gate * mask   (if mask is present)

    All tensors must be contiguous, of dtype torch.float16 and share the same
    shape ``[B, N, N, hidden_dim]``.
    """
    assert tensor.is_contiguous() and gate.is_contiguous()
    total = tensor.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    if nomask:
        # dummy mask (will never be read because HAS_MASK=False)
        dummy = torch.empty(0, device=tensor.device, dtype=tensor.dtype)
        mask_ptr = dummy
        mask_len = 0
    else:
        assert mask is not None
        mask_ptr = mask.view(-1)
        mask_len = mask_ptr.numel()

    _gate_mul_kernel[grid](
        tensor,
        gate,
        mask_ptr,
        total,
        hidden_dim,
        mask_len,
        HAS_MASK=not nomask,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # LayerNorm over the last dimension ‘dim’
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)                      # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into one matmul
    # ------------------------------------------------------------------
    # Convert weight tensors to fp16 once
    def w16(key):
        return weights[key].to(device=device, dtype=dtype_fp16)

    # Order matters – must match the splitting later
    combined_weight = torch.cat(
        [
            w16("left_proj.weight"),
            w16("left_gate.weight"),
            w16("right_proj.weight"),
            w16("right_gate.weight"),
            w16("out_gate.weight"),
        ],
        dim=0,  # (5*hidden_dim, dim)
    )                     # → fp16

    B, N, _, _ = x.shape
    tokens = B * N * N
    x_flat = x.view(tokens, dim)                      # [B·N·N, dim]

    # One big linear → shape (tokens, 5*hidden_dim)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # fp16

    # Split back to the five tensors
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        combined, hidden_dim, dim=1
    )  # each is [tokens, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Activations & reshape
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # Reshape everything to [B, N, N, hidden_dim]
    left  = left_proj_raw.view(B, N, N, hidden_dim).contiguous()
    right = right_proj_raw.view(B, N, N, hidden_dim).contiguous()
    left_gate  = left_gate.view(B, N, N, hidden_dim).contiguous()
    right_gate = right_gate.view(B, N, N, hidden_dim).contiguous()
    out_gate   = out_gate.view(B, N, N, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask (if present) and gate multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(dtype_fp16).contiguous()
    else:
        mask_fp16 = None
    _gate_mul_inplace(left, left_gate, mask_fp16, hidden_dim, nomask)
    _gate_mul_inplace(right, right_gate, mask_fp16, hidden_dim, nomask)

    # ------------------------------------------------------------------
    # 5️⃣ Cubic contraction – batched GEMM (Tensor‑Core)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Move hidden dimension to the batch axis and make tensors contiguous.
    # Shape after permute: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Batch‑matmul over the ‘k’ dimension.
    # Result shape: (B, H, N, N)
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))
    # Back to [B, N, N, H]
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out‑gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to ‘dim’
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)  # (dim, hidden_dim)
    out_flat = out.view(tokens, hidden_dim)            # [B·N·N, hidden_dim]
    out = torch.nn.functional.linear(out_flat, to_out_w)  # → [B·N·N, dim]
    out = out.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
