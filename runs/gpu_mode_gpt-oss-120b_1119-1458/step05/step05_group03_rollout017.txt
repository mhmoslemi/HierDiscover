"""
TriMul (outgoing) – fast forward pass.

The implementation follows the reference PyTorch module but:
1. Performs the first LayerNorm directly in FP16 (no intermediate FP32 pass).
2. Fuses each projection+gate linear layer by concatenating the two weight
   matrices (proj and gate) → a single matmul per side, then splits the result.
3. If a mask is present, a tiny Triton kernel (`mask_gate_fuse_kernel`) fuses
   element‑wise multiplication of (proj * gate * mask) for the left and right
   streams, avoiding extra memory passes.
4. The heavy contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is executed with a single batched GEMM using `torch.bmm`.  The tensors are
   reshaped to (B*H, N, N) and multiplied with the transpose of the right
   operand.  This leverages cuBLAS Tensor‑Core kernels on H100.
5. A second LayerNorm, out‑gate sigmoid, and the final linear projection are
   performed in FP16; the result is cast back to FP32 for the harness.

All steps are fully GPU‑resident; the only Triton kernel is the mask‑gate
fusion, satisfying the requirement to implement part of the operation in
Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) – mask is broadcast over the
# hidden dimension `hidden_dim`.  The kernel works on a flat view of the
# 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,    # *fp16 : projected tensor (shape: total_elements)
    gate_ptr,    # *fp16 : gate tensor, same shape as proj
    mask_ptr,    # *fp16 : mask tensor of shape (B*N*N,)
    total_elems, # i32   : total number of elements (B*N*N*hidden_dim)
    hidden_dim,  # i32   : hidden dimension size (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load projected value and gate
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask across the hidden dimension
    mask_off = offs // hidden_dim
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(proj_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# Input tensors are FP16 of shape [B, N, N, H].
# Output shape is the same.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]  – each head becomes an independent batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM (FP16 → FP32 accumulation)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]  # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm in FP16
    # ------------------------------------------------------------------
    # Convert weights to half precision once
    norm_w = weights["norm.weight"].to(device).half()
    norm_b = weights["norm.bias"].to(device).half()

    # Convert the input to FP16 *before* layer‑norm to keep everything in half
    x_fp16 = input_tensor.to(device).half()
    x = torch.nn.functional.layer_norm(
        x_fp16,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )   # shape [B, N, N, dim], dtype=fp16

    # ------------------------------------------------------------------
    # 2️⃣ Concatenate projection & gating weights (half‑precision)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).half()
    left_gate_w  = weights["left_gate.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w   = weights["out_gate.weight"].to(device).half()
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    to_out_w      = weights["to_out.weight"].to(device).half()

    # Concatenate proj+gate weight matrices (2*hidden_dim, dim)
    left_concat_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (proj + gate) – single matmul per side
    # ------------------------------------------------------------------
    # Output shape: [B, N, N, 2*hidden_dim]
    left_combined  = torch.nn.functional.linear(x, left_concat_w)   # (B,N,N,2H)
    right_combined = torch.nn.functional.linear(x, right_concat_w) # (B,N,N,2H)

    # Split back into projection and pre‑gate tensors
    left_proj, left_gate_pre = left_combined.split(hidden_dim, dim=-1)
    right_proj, right_gate_pre = right_combined.split(hidden_dim, dim=-1)

    # Apply sigmoid to gating streams
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + element‑wise gating fusion (Triton)
    # ------------------------------------------------------------------
    if nomask:
        # In‑place multiply proj * gate (no mask)
        left = left_proj.mul_(left_gate)
        right = right_proj.mul_(right_gate)
    else:
        # Flatten mask for broadcast across hidden_dim
        mask_fp16 = mask_tensor.to(device).float().half().contiguous().view(-1)  # (B*N*N,)
        total_elems = left_proj.numel()               # B*N*N*H
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=8,
        )
        # right = right_proj * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=8,
        )
        left = left_proj
        right = right_proj
        # Release gate tensors early
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – batched GEMM
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) on the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate (sigmoid) and final projection back to `dim`
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_w)  # shape [B,N,N,hidden_dim]
    )
    out = out * out_gate                               # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)   # final projection, FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
