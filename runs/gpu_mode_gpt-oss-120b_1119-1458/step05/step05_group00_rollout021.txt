"""
Optimized outgoing TriMul kernel (AlphaFold‑3).
------------------------------------------------
Algorithm
---------
1️⃣ Layer‑norm the input (FP32) → cast to FP16.
2️⃣ Fuse the two linear projections (proj + gate) for the *left* and *right*
   branches by concatenating the weight matrices and performing a single GEMM.
3️⃣ Split the concatenated results, apply sigmoid to the gate halves.
4️⃣ (Optional) Fuse mask, left‑gate and right‑gate with a tiny Triton kernel.
5️⃣ Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using a custom Triton kernel that works on the
   original memory layout (no transposes or copies).
6️⃣ Second LayerNorm (FP16) over the hidden dimension.
7️⃣ Apply the out‑gate and a final linear projection back to `dim`.
8️⃣ Cast the result to FP32 (the harness expects FP32).

All tensors are kept in FP16 for the compute‑heavy part and only the two
LayerNorms use the provided FP16 / FP32 weights.  The implementation respects
the `nomask` flag and works for the full test‑matrix of shapes and distributions.
"""

import math
from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate in one pass.
#   left  ← left  * left_gate  * mask
#   right ← right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]   (broadcasted)
    total_elements,    # i32   total number of elements in left/right
    H,                 # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # broadcast mask over the hidden dimension
    mask_offs = offs // H

    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=mask)
    tl.store(right_ptr + offs, right_out,  mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the O(N³) contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# Memory layout of left/right/out: [B, N, N, H] (contiguous, H is innermost)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_matmul_kernel(
    left_ptr,          # *fp16  [B,N,N,H]
    right_ptr,         # *fp16  [B,N,N,H]
    out_ptr,           # *fp16  [B,N,N,H]
    seq_len,           # i32   N
    hidden_dim,        # i32   H
    stride_i,          # i32   stride for the i‑axis (N*H)
    stride_k,          # i32   stride for the k‑axis (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)          # block row   (i)
    pid_j = tl.program_id(1)          # block col   (j)
    pid_bh = tl.program_id(2)         # flattened batch*head index (b·H + h)

    # ------------------------------------------------------------------
    # Resolve (batch, head) from flattened index
    # ------------------------------------------------------------------
    b = pid_bh // hidden_dim          # batch index
    h = pid_bh % hidden_dim           # hidden‑dim index

    # Base pointer for this (b,h) pair.
    #   offset = b * (N*N*H) + h   (all strides are counted in elements)
    stride_batch = seq_len * seq_len * hidden_dim
    base = b * stride_batch + h

    a_ptr = left_ptr  + base
    b_ptr = right_ptr + base
    c_ptr = out_ptr   + base

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the summed dimension k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seq_len

        # left[A] : (i, k)   -> shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_ptr + offs_i[:, None] * stride_i
                              + offs_k[None, :] * stride_k,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[B] : (j, k) transposed, i.e. we read B[k, j]
        b = tl.load(b_ptr + offs_k[:, None] * stride_k
                               + offs_j[None, :] * stride_i,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate the outer product for this k‑tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result (convert back to FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)

    tl.store(c_ptr + offs_i[:, None] * stride_i
                       + offs_j[None, :] * stride_i,
              c,
              mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper for the fused matmul
# ----------------------------------------------------------------------
def _batched_trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing TriMul contraction via the custom Triton kernel.
    Input tensors must be FP16, contiguous, shape [B,N,N,H].
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    # Strides in elements (not bytes) – left/right have identical layout
    stride_i = left.stride(1)            # N * H
    stride_k = left.stride(2)            # H

    out = torch.empty_like(left)

    # Tiling parameters – tuned for H100 tensor‑core usage
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )                     # flattened (batch, head)

    _trimul_outgoing_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        seq_len=N,
        hidden_dim=H,
        stride_i=stride_i,
        stride_k=stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )   # [B, N, N, dim] – FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # --------------------------------------------------------------
    x_fp16 = x.to(dtype_fp16)

    # --------------------------------------------------------------
    # 3️⃣ Prepare weight tensors (FP16, on‑device)
    # --------------------------------------------------------------
    # Projection + gate weights (concatenated to halve linear ops)
    left_proj_w  = weights["left_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]

    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(
        device=device, dtype=dtype_fp16
    )   # (2*hidden_dim, dim)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(
        device=device, dtype=dtype_fp16
    )

    out_gate_w = weights["out_gate.weight"].to(
        device=device, dtype=dtype_fp16
    )
    to_out_norm_w = weights["to_out_norm.weight"].to(
        device=device, dtype=dtype_fp16
    )
    to_out_norm_b = weights["to_out_norm.bias"].to(
        device=device, dtype=dtype_fp16
    )
    to_out_w = weights["to_out.weight"].to(
        device=device, dtype=dtype_fp16
    )

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (proj + gate fused)
    # --------------------------------------------------------------
    # left_branch: [B,N,N,2*hidden_dim] = x @ left_combined_w.T
    left_cat = F.linear(x_fp16, left_combined_w)        # FP16
    right_cat = F.linear(x_fp16, right_combined_w)      # FP16

    # Split into projection and gate halves
    left_proj, left_gate_pre = torch.split(left_cat, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_cat, hidden_dim, dim=-1)

    left_gate = torch.sigmoid(left_gate_pre)    # FP16
    right_gate = torch.sigmoid(right_gate_pre)  # FP16

    # --------------------------------------------------------------
    # 5️⃣ Optional mask + gate fusion (Triton kernel)
    # --------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure contiguous layout for the kernel
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()  # [B,N,N]

        total = left.numel()
        BLOCK = 1024
        grid_mask = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid_mask](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # free temporary tensors early
        del left_gate, right_gate, mask_fp16

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via custom Triton GEMM
    # --------------------------------------------------------------
    out = _batched_trimul_matmul(left, right)   # [B,N,N,hidden_dim] FP16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden dimension)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # FP16

    # --------------------------------------------------------------
    # 8️⃣ Out‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # [B,N,N,hidden_dim] FP16
    out = out * out_gate                                     # element‑wise FP16
    out = F.linear(out, to_out_w)                            # [B,N,N,dim] FP16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to float32 (as expected by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
