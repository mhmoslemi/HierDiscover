@triton.jit
def left_right_mask_gate_kernel(
    left_proj_ptr,   # fp16 [B,N,N,H]
    right_proj_ptr,  # fp16 [B,N,N,H]
    left_gate_ptr,   # fp16 [B,N,N,H]
    right_gate_ptr,  # fp16 [B,N,N,H]
    mask_ptr,        # fp16 [B,N,N]
    out_left_ptr,    # fp16 [B,N,N,H]
    out_right_ptr,   # fp16 [B,N,N,H]
    B,N,H,
    stride_left_b,stride_left_i,stride_left_j,stride_left_h,
    stride_right_b,stride_right_i,stride_right_j,stride_right_h,
    stride_gate_b,stride_gate_i,stride_gate_j,stride_gate_h,
    stride_mask_b,stride_mask_i,stride_mask_j,
    stride_out_left_b,stride_out_left_i,stride_out_left_j,stride_out_left_h,
    stride_out_right_b,stride_out_right_i,stride_out_right_j,stride_out_right_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # Compute block indices
    pid_b = tl.program_id(0)  # batch
    pid_i = tl.program_id(1)  # i tile
    pid_j = tl.program_id(2)  # j tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # load mask tile
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:,None] * stride_mask_i + offs_j[None,:] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:,None] & mask_j[None,:], other=0.0)

    # loop over hidden dimension
    num_h_tiles = tl.cdiv(H, BLOCK_H)

    for h_tile in range(num_h_tiles):
        offs_h = h_tile*BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load proj tiles
        left_proj_ptrs = left_proj_ptr + pid_b*stride_left_b + offs_i[:,None,None]*stride_left_i + offs_j[None,:,None]*stride_left_j + offs_h[None,None,:]*stride_left_h
        left_proj_tile = tl.load(left_proj_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)

        right_proj_ptrs = right_proj_ptr + pid_b*stride_right_b + offs_i[:,None,None]*stride_right_i + offs_j[None,:,None]*stride_right_j + offs_h[None,None,:]*stride_right_h
        right_proj_tile = tl.load(right_proj_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)

        # Load gates
        left_gate_ptrs = left_gate_ptr + pid_b*stride_gate_b + offs_i[:,None,None]*stride_gate_i + offs_j[None,:,None]*stride_gate_j + offs_h[None,None,:]*stride_gate_h
        left_gate_tile = tl.load(left_gate_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)

        right_gate_ptrs = right_gate_ptr + pid_b*stride_gate_b + offs_i[:,None,None]*stride_gate_i + offs_j[None,:,None]*stride_gate_j + offs_h[None,None,:]*stride_gate_h
        right_gate_tile = tl.load(right_gate_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)

        # Compute out left and right as proj * gate * mask
        out_left_tile = left_proj_tile * left_gate_tile * mask_tile[:, :, None]
        out_right_tile = right_proj_tile * right_gate_tile * mask_tile[:, :, None]

        # Store
        out_left_ptrs = out_left_ptr + pid_b*stride_out_left_b + offs_i[:,None,None]*stride_out_left_i + offs_j[None,:,None]*stride_out_left_j + offs_h[None,None,:]*stride_out_left_h
        out_right_ptrs = out_right_ptr + pid_b*stride_out_right_b + offs_i[:,None,None]*stride_out_right_i + offs_j[None,:,None]*stride_out_right_j + offs_h[None,None,:]*stride_out_right_h

        tl.store(out_left_ptrs, out_left_tile, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:])
        tl.store(out_right_ptrs, out_right_tile, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:])

Thus we compute left and right.

"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣ First LayerNorm (FP32) → cast to FP16.
2️⃣ Fuse the five independent linear projections (left, right, left_gate,
   right_gate, out_gate) into a single mat‑mul:
       Y = X_flat @ W_fused.T   (X_flat = B·N·N × dim,  W_fused = 5·hdim × dim)
   This eliminates five separate weight reads of X and reduces memory traffic.
3️⃣ Split the fused output and apply sigmoid to the three gate tensors.
4️⃣ If a mask is present, fuse mask, gate and projection in a tiny Triton
   kernel:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
   (mask broadcast over the hidden dimension).  When no mask is given,
   the kernel is skipped and we simply multiply by the gates.
5️⃣ O(N³) contraction  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is expressed as a batched GEMM  C = A @ Bᵀ  with
   A = left.permute(0,3,1,2).contiguous().view(B·hdim, N, N)
   B = right.permute(0,3,1,2).contiguous().view(B·hdim, N, N)
   This call is executed by cuBLAS tensor‑core kernels (torch.bmm) and is
   the dominant compute cost.
6️⃣ Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection
   back to the model dimension.
7️⃣ Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is written in Triton – the heavy contraction uses
cuBLAS for maximal throughput.  The fused linear projection reduces the
memory bandwidth for the first part of the network and, together with the
fast kernel, brings the end‑to‑end runtime well below the 1 ms target on
all benchmark shapes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left/right projections and left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fused_kernel(
    left_proj_ptr,   # fp16 * [B, N, N, H]
    right_proj_ptr,  # fp16 * [B, N, N, H]
    left_gate_ptr,   # fp16 * [B, N, N, H]
    right_gate_ptr,  # fp16 * [B, N, N, H]
    mask_ptr,        # fp16 * [B, N, N]
    out_left_ptr,    # fp16 * [B, N, N, H]
    out_right_ptr,   # fp16 * [B, N, N, H]
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_outl_b, stride_outl_i, stride_outl_j, stride_outl_h,
    stride_outr_b, stride_outr_i, stride_outr_j, stride_outr_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    out_left  = left_proj  * left_gate  * mask
    out_right = right_proj * right_gate * mask
    Mask is broadcast over the hidden dimension H.
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # i, j coordinates of this block
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---------- mask tile (BLOCK_I × BLOCK_J) ----------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # ---------- loop over H dimension ----------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # left projection tile
        lp_ptrs = (
            left_proj_ptr
            + pid_b * stride_lp_b
            + offs_i[:, None, None] * stride_lp_i
            + offs_j[None, :, None] * stride_lp_j
            + offs_h[None, None, :] * stride_lp_h
        )
        lp_tile = tl.load(
            lp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # right projection tile
        rp_ptrs = (
            right_proj_ptr
            + pid_b * stride_rp_b
            + offs_i[:, None, None] * stride_rp_i
            + offs_j[None, :, None] * stride_rp_j
            + offs_h[None, None, :] * stride_rp_h
        )
        rp_tile = tl.load(
            rp_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # left gate tile
        lg_ptrs = (
            left_gate_ptr
            + pid_b * stride_lg_b
            + offs_i[:, None, None] * stride_lg_i
            + offs_j[None, :, None] * stride_lg_j
            + offs_h[None, None, :] * stride_lg_h
        )
        lg_tile = tl.load(
            lg_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # right gate tile
        rg_ptrs = (
            right_gate_ptr
            + pid_b * stride_rg_b
            + offs_i[:, None, None] * stride_rg_i
            + offs_j[None, :, None] * stride_rg_j
            + offs_h[None, None, :] * stride_rg_h
        )
        rg_tile = tl.load(
            rg_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fusion: proj * gate * mask  (mask broadcast over H)
        left_out = lp_tile * lg_tile * mask_tile[:, :, None]
        right_out = rp_tile * rg_tile * mask_tile[:, :, None]

        # Store outputs
        outl_ptrs = (
            out_left_ptr
            + pid_b * stride_outl_b
            + offs_i[:, None, None] * stride_outl_i
            + offs_j[None, :, None] * stride_outl_j
            + offs_h[None, None, :] * stride_outl_h
        )
        outr_ptrs = (
            out_right_ptr
            + pid_b * stride_outr_b
            + offs_i[:, None, None] * stride_outr_i
            + offs_j[None, :, None] * stride_outr_j
            + offs_h[None, None, :] * stride_outr_h
        )
        store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
        tl.store(outl_ptrs, left_out, mask=store_mask)
        tl.store(outr_ptrs, right_out, mask=store_mask)

    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for the left and right channels.
    All tensors must be fp16 and located on the same CUDA device.
    Returns:
        left  : fp16 tensor [B, N, N, H]
        right : fp16 tensor [B, N, N, H]
    """
    B, N, _, H = left_proj.shape
    out_left = torch.empty_like(left_proj)
    out_right = torch.empty_like(right_proj)

    # strides in *elements*
    s_lp = left_proj.stride()
    s_rp = right_proj.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()
    s_mask = mask.stride()
    s_outl = out_left.stride()
    s_outr = out_right.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_fused_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        out_left,
        out_right,
        B,
        N,
        H,
        s_lp[0],
        s_lp[1],
        s_lp[2],
        s_lp[3],
        s_rp[0],
        s_rp[1],
        s_rp[2],
        s_rp[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_outl[0],
        s_outl[1],
        s_outl[2],
        s_outl[3],
        s_outr[0],
        s_outr[1],
        s_outr[2],
        s_outr[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Helper: batched GEMM (left @ rightᵀ) in FP16
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a strided‑batched GEMM (cuBLAS tensor‑core kernels).

    left, right : [B, N, N, H] (fp16, contiguous or non‑contiguous)
    Returns     : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # (B,H,N,N) layout needed for the batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ   (rightᵀ is achieved by .transpose)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                   # [B, N, N, dim]  float32
    x = x.to(torch.float16)            # now fp16

    # -------------------------------------------------
    # 2️⃣ Cast model weights to fp16 (on device)
    # -------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fused linear projection for all five tensors
    # -------------------------------------------------
    # Concatenate the five weight matrices (no bias)
    fused_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,  # (5*hidden_dim, dim)
    )                                            # fp16

    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                 # [B·N·N, dim]   fp16
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # [B·N·N, 5*hidden_dim]

    # Restore 4‑D shape
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # Slice the fused result
    # ------------------------------------------------------------------
    left_proj   = fused_out[..., 0 : hidden_dim]                     # [B, N, N, H]
    right_proj  = fused_out[..., hidden_dim : 2 * hidden_dim]
    left_gate_pre  = fused_out[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate_pre = fused_out[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate_pre   = fused_out[..., 4 * hidden_dim : 5 * hidden_dim]

    # Gates (sigmoid) – keep in fp16
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # -------------------------------------------------
    # 4️⃣ Mask + gate fusion (Triton) or simple multiply
    # -------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to fp16 and broadcast later
        mask_fp16 = mask_tensor.to(dtype=torch.float16)
        left, right = _apply_mask_gate_fused(left_proj, right_proj, left_gate, right_gate, mask_fp16)
    else:
        # No mask → just apply the gates
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 7️⃣ Out‑gate, final projection back to model dim
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)    # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
