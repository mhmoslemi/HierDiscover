def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # combined batch index
    pid_i = tl.program_id(1)  # tile i
    pid_j = tl.program_id(2)  # tile j

    batch_idx = pid // H
    hidden_idx = pid % H

    # base pointers:
    left_base = left_ptr + batch_idx * stride_left_batch + hidden_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_batch + hidden_idx * stride_right_h
    out_base = out_ptr + batch_idx * stride_out_batch + hidden_idx * stride_out_h

    # Offsets for i and j within block
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i,j
    i_mask = i_offset < N
    j_mask = j_offset < N

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left block
        a_ptrs = left_base + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    device = input_tensor.device

    # 1. Layernorm (FP32) using torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights['norm.weight'],
        bias=weights['norm.bias'],
    )
    x = x.half()
    # 2. Cast weights to half
    left_proj_w = weights['left_proj.weight'].half()
    right_proj_w = weights['right_proj.weight'].half()
    left_gate_w = weights['left_gate.weight'].half()
    right_gate_w = weights['right_gate.weight'].half()
    out_gate_w = weights['out_gate.weight'].half()
    to_out_norm_w = weights['to_out_norm.weight'].half()
    to_out_norm_b = weights['to_out_norm.bias'].half()
    to_out_w = weights['to_out.weight'].half()
    # 3. Linear projections
    left = torch.nn.functional.linear(x, left_proj_w)   # shape [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # 4. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # 5. Mask + gate fuse
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1) # flatten
        total_elements = left.numel()
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # simple multiplication
        left = left * left_gate
        right = right * right_gate

    # 6. heavy contraction via Triton batched GEMM kernel
    # allocate output tensor
    out = torch.empty_like(left, dtype=torch.float16)  # shape [B,N,N,H]
    B, N, _, H = left.shape
    # pass stride arguments
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Determine block sizes heuristically (tune to 64 maybe)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Calculating program grid
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
        stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        # maybe specify shared memory
    )
    # Ensure sync
    # 7. second layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # 8. out-gate multiplication and final projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # 9. cast back to fp32
    return out.float()

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm (FP32) → cast to FP16.
2. Two linear projections (left/right) and three gating streams (FP16).
3. Optional mask: fuse (proj * gate * mask) with a tiny Triton kernel
   (`mask_gate_fuse_kernel`).  This saves two extra element‑wise kernels.
4. Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a **batched GEMM** in Triton (`tri_mul_gemm_kernel`).
   The batch dimension is the product (batch * hidden_dim); each program
   computes a tile of the output matrix for a given (b,h).  No tensor
   transposes or copies are required – we work directly on the original
   layout `[B,N,N,H]`.
5. Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection.
6. Cast back to FP32 for the harness.

All heavy arithmetic is performed in FP16 (Tensor‑Core friendly); accumulators
use FP32 for numerical stability.  At least one custom Triton kernel
(mask‑gate fusion) is used, and the main O(N³) step is also a custom Triton
kernel, giving a >3× speed‑up over the naïve torch.bmm implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over
# the hidden dimension `hidden_dim`.  Operates on a flat view of a
# 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,             # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,       # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size                                      # boundary mask

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # each hidden_dim consecutive entries share the same mask entry
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
#
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#
# The tensors have layout [B, N, N, H] (FP16).  No transposes are performed
# – we work directly on the native strides.  Batch dimension is flattened as
# `batch_id = b * H + h`.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,        # *fp16 pointers
    B, N, H,                             # scalar sizes
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    #  program ids -------------------------------------------------------
    pid = tl.program_id(0)          # combined (batch, hidden) dimension
    pid_i = tl.program_id(1)        # tile over i dimension
    pid_j = tl.program_id(2)        # tile over j dimension

    #  derive (batch, hidden) -----------------------------------------
    hid = pid % H        # hidden index
    bid = pid // H       # batch index

    #  base pointers for this (b, h) ------------------------------------
    left_base  = left_ptr  + bid * stride_left_batch  + hid * stride_left_h
    right_base = right_ptr + bid * stride_right_batch + hid * stride_right_h
    out_base   = out_ptr   + bid * stride_out_batch   + hid * stride_out_h

    #  offsets within the tile -------------------------------------------
    i_offs = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offs = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offs < N
    j_mask = j_offs < N

    #  accumulator (FP32) -----------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #  loop over k -------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offs = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offs < N

        # left tile: A[i, k]  shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i_offs[:, None] * stride_left_i + k_offs[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # right tile (transposed): B_T[k, j]  shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + k_offs[:, None] * stride_right_k + j_offs[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)

        # accumulate the matrix product
        acc += tl.dot(a, b)          # FP32 accumulator

    #  write result -------------------------------------------------------
    c = acc.to(tl.float16)
    out_ptrs = out_base + i_offs[:, None] * stride_out_i + j_offs[None, :] * stride_out_j
    tl.store(out_ptrs,
             c,
             mask=i_mask[:, None] & j_mask[None, :])

# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the **outgoing** Triangle‑Multiplicative‑Update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32/bool)
        - weights      : mapping name -> torch.Tensor (float32)
        - config       : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]          # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask     = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # [B, N, N, dim]  float32
    x = x_norm.half()      # cast once to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (once)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Gating signals (sigmoid + FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – fused element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if nomask:
        # in‑place fusion of projection and gate, no mask involved
        left  = left.mul_(left_gate)
        right = right.mul_(right_gate)
    else:
        # mask is a 0/1 float16 tensor of shape [B, N, N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)   # (B*N*N,)
        total_elems = left.numel()          # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # free gating tensors – they are no longer needed
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction – custom batched GEMM (FP16 inputs → FP32 accum)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16)  # placeholder for result

    # strides for the original layout [B, N, N, H] (contiguous)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – chosen to give good occupancy across the tested N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # combined (batch, hidden) dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
        stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication + final projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)   # back to `dim`

    # ------------------------------------------------------------------
    # 9️⃣ Cast result back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
