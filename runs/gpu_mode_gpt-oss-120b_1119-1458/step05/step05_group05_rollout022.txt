@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_k_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    hid_batch = tl.program_id(0)
    p_i = tl.program_id(1)
    p_j = tl.program_id(2)

    # Compute base indices
    b = hid_batch // H
    h = hid_batch % H

    # offsets within matrices
    offs_i = p_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = p_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks to avoid accessing beyond N
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Compute base pointer for each hidden slice
    left_base = left_ptr + b * stride_b_l + h * stride_h_l
    right_base = right_ptr + b * stride_b_r + h * stride_h_r
    out_base = out_ptr + b * stride_b_o + h * stride_h_o

    # Pointers for tiles (using broadcasting)
    # A tile shape (BLOCK_M, BLOCK_K)
    a_ptrs = left_base + offs_i[:, None] * stride_i_l + offs_k[None, :] * stride_k_l
    # B tile shape (BLOCK_K, BLOCK_N) (transposed layout)
    b_ptrs = right_base + offs_k[:, None] * stride_k_r + offs_j[None, :] * stride_i_r

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute actual offsets for this iteration
        cur_k = start_k * BLOCK_K

### Implementing 3D hidden batch and mask functions in Triton.

First define the kernel signature:

```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    mask_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    stride_bm, stride_im, stride_km,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    """Row-group kernel for TriMul contraction:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] * mask_factor (optional)
    """
    # Program IDs
    hid_batch = tl.program_id(0)   # flattened batch*hidden id
    pid_i = tl.program_id(1)      # tile row
    pid_j = tl.program_id(2)      # tile col

    # Compute batch and hidden channel from hid_batch
    b = hid_batch // H
    h = hid_batch % H

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointer base for hidden slice
    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_ir
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        if HAS_MASK:
            # Load mask tiles for A and B side (shape matches a and b). Mask dtype fp16.
            mask_a_ptrs = mask_ptr + b * stride_bm + offs_i[:, None] * stride_im + offs_k[None, :] * stride_km
            mask_a = tl.load(mask_a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=1.0)
            a = a * mask_a.to(tl.float16)

            mask_b_ptrs = mask_ptr + b * stride_bm + offs_j[:, None] * stride_im + offs_k[None, :] * stride_km
            mask_b = tl.load(mask_b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=1.0)
            b = b * mask_b.to(tl.float16)

        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_ij)
```

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    mask_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    stride_bm, stride_im, stride_km,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr):
    """
    Triton kernel that computes the core TriMul contraction.
    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] *
                           (mask_factor if provided)
    All tensors are fp16. Accumulator is fp32.
    """
    # program ids
    hid_batch = tl.program_id(0)   # flattened batch*hidden id
    pid_i = tl.program_id(1)      # block row index
    pid_j = tl.program_id(2)      # block column index

    # decode batch and hidden channel
    b = hid_batch // H
    h = hid_batch % H

    # offsets within N
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (left) : shape (BLOCK_M, BLOCK_K)
        a_ptr = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (right transposed) : shape (BLOCK_K, BLOCK_N)
        b_ptr = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_ir
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Optional mask multiplication
        if HAS_MASK:
            # mask slice for left side (i,k)
            mask_a_ptr = mask_ptr + b * stride_bm + offs_i[:, None] * stride_im + offs_k[None, :] * stride_km
            mask_a = tl.load(mask_a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=1.0)
            a = a * mask_a

            # mask slice for right side (j,k)
            mask_b_ptr = mask_ptr + b * stride_bm + offs_j[:, None] * stride_im + offs_k[None, :] * stride_km
            mask_b = tl.load(mask_b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=1.0)
            b = b * mask_b

        # Dot product accumulate
        acc += tl.dot(a, b)

    # Store result to out (fp16)
    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward of the outgoing TriMul module, using a custom Triton kernel
    for the O(N³) contraction.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------- 1. First LayerNorm (float32) -------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # float32

    # Cast to fp16 for heavy computation
    x = x.to(torch.float16)

    # ------- 2. Project and compute gates (fp16) -------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_w)          # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)        # [B,N,N,H]

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates (mask is fused later inside kernel)
    left = left * left_gate
    right = right * right_gate

    # ------- 3. Heavy contraction via Triton -------
    B, N, _, H = left.shape  # H == hidden_dim

    # Allocate output tensor (fp16)
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # Prepare mask (if any)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        # Dummy mask (not used). Provide a tensor of size 1, device only.
        mask_fp16 = torch.empty((1,), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()
    stride_bm, stride_im, stride_km = mask_fp16.stride() if not nomask else (0, 0, 0)

    # Tiling parameters (tuned for H100, fp16 tensor cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    total_hidden = B * H
    grid = (total_hidden, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        mask_fp16,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        stride_bm, stride_im, stride_km,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=int(not nomask),
        num_warps=8,
        num_stages=2,
    )

    # ------- 4. Second LayerNorm -------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------- 5. Gating and final projection -------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    # Cast back to float32 as expected by the harness
    return out.to(torch.float32)

"""
custom_kernel – outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ First LayerNorm (float32) on the 4‑D input.
2️⃣ Cast to fp16 and compute the two linear projections (left / right) and
   the three gating signals (left_gate, right_gate, out_gate).
3️⃣ Fuse gating via simple element‑wise multiplication; the optional mask
   is handled *inside* the heavy contraction kernel (no extra kernel launch).
4️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] * mask(b,i,k) * mask(b,j,k)
   This is performed by a custom Triton kernel that treats the hidden
   dimension as an extra batch axis (total hidden batches = B·H).  The
   kernel implements a batched GEMM using `tl.dot`, works on fp16
   tensors, accumulates in fp32, and writes the result back in fp16.
5️⃣ Second LayerNorm (fp16) on the contracted tensor.
6️⃣ Multiply by the out‑gate and apply the final linear projection back to
   the original channel dimension.
7️⃣ Cast the result to float32 (the harness expects float32).

The Triton kernel is the only custom kernel – all other ops use PyTorch
functionalities.  Tile sizes (BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) are
chosen for H100 tensor‑core throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # *fp16 left projection (B,N,N,H)
    right_ptr,              # *fp16 right projection (B,N,N,H)
    out_ptr,                # *fp16 output tensor (B,N,N,H)
    mask_ptr,               # *fp16 mask tensor (B,N,N) – only read when HAS_MASK
    B, N, H,                # runtime constants
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_ir, stride_kr, stride_hr,   # right strides
    stride_bo, stride_io, stride_jo, stride_ho,   # out strides
    stride_bm, stride_im, stride_km,             # mask strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr):
    """
    Computes:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] *
                       (mask(b,i,k) * mask(b,j,k)   if HAS_MASK else 1)
    All tensors are fp16; accumulation is in fp32.
    """
    # -------------------------------------------------
    # 0. Program IDs – hidden‑batch, tile‑row, tile‑col
    # -------------------------------------------------
    hid_batch = tl.program_id(0)        # 0 … B*H-1  (flattened batch·hidden)
    pid_i = tl.program_id(1)            # tile row
    pid_j = tl.program_id(2)            # tile col

    # -------------------------------------------------
    # 1. Decode batch index and hidden channel index
    # -------------------------------------------------
    b = hid_batch // H
    h = hid_batch % H

    # -------------------------------------------------
    # 2. Offsets inside the N dimension
    # -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # -------------------------------------------------
    # 3. Base pointers for the current (b,h) slice
    # -------------------------------------------------
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # -------------------------------------------------
    # 4. Accumulator (FP32)
    # -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # 5. Loop over the K dimension (tiles of size BLOCK_K)
    # -------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- A tile (left) : shape (BLOCK_M, BLOCK_K) -----------------
        a_ptr = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # fp16

        # ---- B tile (right, transposed) : shape (BLOCK_K, BLOCK_N) ----
        b_ptr = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_ir
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)          # fp16

        # ---- Optional mask multiplication ---------------------------------
        if HAS_MASK:
            # mask for left side (i,k)
            m_a_ptr = mask_ptr + b * stride_bm + offs_i[:, None] * stride_im + offs_k[None, :] * stride_km
            m_a = tl.load(m_a_ptr,
                          mask=mask_i[:, None] & mask_k[None, :],
                          other=1.0)      # fp16, 1.0 == no masking
            a = a * m_a

            # mask for right side (j,k)
            m_b_ptr = mask_ptr + b * stride_bm + offs_j[:, None] * stride_im + offs_k[None, :] * stride_km
            m_b = tl.load(m_b_ptr,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=1.0)      # fp16
            b = b * m_b

        # ---- Accumulate dot product ---------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) FP32

    # -------------------------------------------------
    # 6. Write out result (cast back to fp16)
    # -------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij)

    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Load model weights (cast once to fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections and gating signals (fp16)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)      # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)    # [B,N,N,H]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # fuse projection with its gate
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy contraction – Triton kernel (batched GEMM over hidden dim)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # allocate output buffer (fp16)
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # Prepare mask (if any). If nomask we still pass a dummy tensor – the kernel
    # will keep HAS_MASK=0 and therefore never read it.
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.empty((1,), dtype=torch.float16, device=device)  # unused

    # Tensor strides (in number of elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()
    stride_bm, stride_im, stride_km = mask_fp16.stride() if not nomask else (0, 0, 0)

    # Tile sizes – tuned for H100 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (hidden‑batch, i‑tiles, j‑tiles)
    total_hidden = B * H
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total_hidden, grid_i, grid_j)

    # Launch the kernel
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        mask_fp16,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        stride_bm, stride_im, stride_km,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=int(not nomask),
        num_warps=8,
        num_stages=2,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Final gating + projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to [B,N,N,dim]

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
