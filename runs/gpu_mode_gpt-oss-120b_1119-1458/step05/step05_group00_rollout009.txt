"""
Optimized Triton‑accelerated outgoing TriMul (AlphaFold‑3).

Algorithm
---------
1. **First LayerNorm** (FP32 → keeps numerical stability).
2. **Cast to FP16** – the heavy path runs in half‑precision.
3. **Fused linear projection** – concatenate the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) into a single
   weight of shape `(5·hidden_dim, dim)` and perform one `torch.nn.functional.linear`
   call.  This saves five kernel launches and avoids repeated reads of the
   input tensor.
4. **Slice & activation** – split the concatenated result into tensors
   `left`, `right`, `left_gate`, `right_gate`, `out_gate`.  Apply `sigmoid`
   to the three gating tensors.
5. **Mask & gate fusion** – when a mask is present a tiny Triton kernel
   fuses the element‑wise mask, left‑gate and right‑gate multiplications:
   `left = left * left_gate * mask`, `right = right * right_gate * mask`.
   If no mask is supplied the element‑wise products are performed with
   standard PyTorch broadcasting.
6. **O(N³) contraction** – compute
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is implemented as a batched matrix multiply:
   `C = (B·H, N, N) @ (B·H, N, N)ᵀ` using `torch.bmm`.  The tensors are
   reshaped with a cheap `permute`+`contiguous` before the GEMM.
7. **Second LayerNorm** (FP16) over the hidden dimension.
8. **Out‑gate & final projection** – multiply by the sigmoid `out_gate`
   and linearly project back to `dim`.
9. **Cast back to FP32** – the harness expects a float32 output.

Only the mask‑gate fusion is performed in a Triton kernel, satisfying the
requirement that “part of the operations” live in Triton while the rest
leverages highly‑optimized cuBLAS GEMMs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate with the projected
# tensors.  All tensors are expected to be contiguous FP16 with shape
# [B, N, N, H] (left/right/gate) and mask of shape [B, N, N] broadcast
# over H.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]
    total_elements,    # i32  total number of fp16 elements (B·N·N·H)
    H,                 # i32  hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat indices
    mask = offs < total_elements                                 # OOB guard

    # Each hidden dimension is stored contiguously, therefore the mask index
    # is the flat offset divided by H (integer division).
    mask_offs = offs // H

    # Load data (OOB positions return a neutral value 0.0 for tensors,
    # 1.0 for the mask – multiplying by 1 leaves the values unchanged).
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse mask and gates.
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store the results back in‑place.
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for A,B ∈ ℝ^{(B·H)×N×N} in FP16.
    Input tensors have shape [B, N, N, H] (contiguous, fp16).
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    # Bring hidden dimension to the batch axis.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()
    # Merge B and H into a single batch dimension.
    left_flat = left_t.view(B * H, N, N)     # (B·H, N, N)
    right_flat = right_t.view(B * H, N, N)   # (B·H, N, N)

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout.
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float, may be None)
        - weights_dict : mapping from parameter name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)          # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections (left/right and the three gates)
    # ------------------------------------------------------------------
    # Concatenate weight matrices along the output‑feature dimension.
    w_left      = weights["left_proj.weight"]   # (hidden_dim, dim)
    w_right     = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate= weights["right_gate.weight"]
    w_out_gate  = weights["out_gate.weight"]
    # Resulting big weight: (5*hidden_dim, dim)
    big_weight = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate], dim=0
    ).to(device=device, dtype=torch.float16)

    # Single linear call → shape [B, N, N, 5*hidden_dim]
    proj = F.linear(x, big_weight)   # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Slice tensors & apply sigmoid to gating tensors
    # ------------------------------------------------------------------
    # Helper to split the last dimension.
    split = torch.split(proj, hidden_dim, dim=-1)
    left      = split[0]                     # (B,N,N,H)
    right     = split[1]
    left_gate = torch.sigmoid(split[2])
    right_gate= torch.sigmoid(split[3])
    out_gate  = torch.sigmoid(split[4])     # used later after second LN

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask + gate fusion (in‑place Triton kernel)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Ensure tensors are contiguous.
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_tensor.to(dtype=torch.float16).contiguous()

        total = left.numel()                # B·N·N·H
        BLOCK = 1024                         # ~1K threads per block
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # After fusion (mask already applied) the gated tensors are ready.
        # No extra multiplication required.
    else:
        # No mask → simply apply gates.
        left  = left * left_gate
        right = right * right_gate
        # Gates are no longer needed.
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)    # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias" ].to(device=device, dtype=torch.float16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Apply final out‑gate and project back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)            # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (expected by the caller)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
