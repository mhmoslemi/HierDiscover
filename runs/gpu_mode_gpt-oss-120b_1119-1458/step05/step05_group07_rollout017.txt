"""
TriMul (outgoing) – fast Triton‑fused implementation
====================================================

The forward pass follows the reference PyTorch code but
* performs the first LayerNorm in FP32 (numerical stability) and
  immediately casts to FP16,
* fuses the left/right linear projections + their gates in two
  batched matmuls,
* fuses the element‑wise `tensor * gate * mask` using a tiny Triton kernel,
* carries out the heavy O(N³) contraction
  `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
  with a custom Triton batched‑GEMM kernel that works directly on
  the original NHWH layout (no costly permute/contiguous copies),
* applies the second LayerNorm, out‑gate, and final projection
  (still in FP16), converting back to FP32 at the end.

Only the forward pass is needed; no autograd support is required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: fused (tensor * gate * mask) in‑place.
#   x   : [B, N, N, H] (FP16) – written with the fused result
#   gate: [B, N, N, H] (FP16) – element‑wise gate (already sigmoid‑ed)
#   mask: [B, N, N]    (FP16) – broadcasted over the hidden dimension H
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,          # *fp16 : tensor to be updated (size = total_elts)
    gate_ptr,       # *fp16 : pre‑computed gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,   # B * N * N * H
    hidden_dim: tl.int32,   # H
    BLOCK: tl.constexpr,    # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    valid = offs < total_elts

    # index into the broadcasted mask (mask repeats every `hidden_dim` entries)
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the O(N³) contraction.
#   Computes for each (b, h) pair:
#       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#   All tensors have layout (B, N, N, H) – contiguous in the last dim.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_gemm_kernel(
    left_ptr,               # *fp16 : left tensor
    right_ptr,              # *fp16 : right tensor
    out_ptr,                # *fp16 : output tensor
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,            # hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile rows  (i‑axis)
    BLOCK_N: tl.constexpr,  # tile cols  (j‑axis)
    BLOCK_K: tl.constexpr,  # reduction tile (k‑axis)
):
    # 0‑dim: combined batch (b*H + h), 1‑dim: tile‑row, 2‑dim: tile‑col
    pid0 = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode combined batch index
    b = pid0 // H
    h = pid0 % H

    # start offsets for the current tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # generate range indices
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows/cols
    i_mask = i < N
    j_mask = j < N

    # accumulator in FP32 (better precision and utilizes Tensor‑Cores)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for kk in range(0, num_k_tiles):
        k_start = kk * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # ----- load a tile from left (shape M×K) -----
        a_ptrs = (
            left_ptr
            + b * stride_l_b
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
            + h * stride_l_h
        )
        a = tl.load(a_ptrs,
                    mask=(i_mask[:, None] & k_mask[None, :]),
                    other=0.0)

        # ----- load a tile from right (shape N×K) -----
        b_ptrs = (
            right_ptr
            + b * stride_r_b
            + j[:, None] * stride_r_i
            + k[None, :] * stride_r_k
            + h * stride_r_h
        )
        b = tl.load(b_ptrs,
                    mask=(j_mask[:, None] & k_mask[None, :]),
                    other=0.0)

        # ----- matrix multiply + accumulate (Tensor‑Core friendly) -----
        # a, b are FP16; tl.dot accumulates in FP32 automatically.
        acc += tl.dot(a, b)

    # ----- write the result (cast back to FP16) -----
    c_ptrs = (
        out_ptr
        + b * stride_o_b
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(c_ptrs, acc.to(tl.float16),
              mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (AlphaFold‑3) without autograd.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool)
        - weights_dict : dict of named tensors (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         ``nomask`` (bool) may be supplied.

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    x = x_norm.to(torch.float16)               # work in FP16 from now on

    # ------------------------------------------------------------------
    # 2️⃣ convert weight tensors to FP16 once
    # ------------------------------------------------------------------
    # linear projection + gate weights (combined to reduce the number of GEMMs)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # cast all to half‑precision
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # (2H, dim)
    out_gate_w   = out_gate_w.to(torch.float16)
    to_out_norm_w = to_out_norm_w.to(torch.float16)
    to_out_norm_b = to_out_norm_b.to(torch.float16)
    to_out_w     = to_out_w.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ linear + gate for the left side (produces proj & pre‑gate in one matmul)
    # ------------------------------------------------------------------
    left_comb = torch.nn.functional.linear(x, left_comb_w)           # [B, N, N, 2H]
    left_proj = left_comb[..., :hidden_dim]                         # [B, N, N, H]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ linear + gate for the right side
    # ------------------------------------------------------------------
    right_comb = torch.nn.functional.linear(x, right_comb_w)        # [B, N, N, 2H]
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 5️⃣ optional mask + gate fusion
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcasted over the hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()               # B * N * N * H
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # no mask → element‑wise multiplication (cheap)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ heavy O(N³) contraction using a custom Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape  # H == hidden_dim
    out = torch.empty_like(left_proj)  # allocate output fp16 tensor

    # strides for the three tensors (all have identical layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_proj.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_proj.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tiling configuration (tuned for H100 Tensor‑Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * hidden_dim,                     # combined batch (b * H + h)
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles along j
    )
    tri_mul_batch_gemm_kernel[grid](
        left_proj.data_ptr(),
        right_proj.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # 8 warps ≈ 256 threads per block fits H100 well
    )

    # ------------------------------------------------------------------
    # 7️⃣ second LayerNorm (FP16) → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # out‑gate is computed from the *normalized* x (the same x after step 1)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    out = out * out_gate                      # element‑wise gate
    out = torch.nn.functional.linear(out, to_out_w)  # Project back to dim

    # ------------------------------------------------------------------
    # 8️⃣ cast back to FP32 (required by the surrounding harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
