def layernorm_gate_kernel(
    out_ptr, # B,N,N,H
    x_ptr,
    gate_ptr,
    norm_weight_ptr,
    norm_bias_ptr,
    hidden_dim: tl.constexpr,
    eps: tl.constexpr,
):

    pid = tl.program_id(0)
    # compute 3D indices:
    # total_positions = B * N * N
    # linear position index pos = pid
    # compute batch, i, j from pos via division?
    # We can compute base pointer: start = pos * hidden_dim
    base_offset = pid * hidden_dim
    # accumulation
    sum_ = tl.zeros([], dtype=tl.float32)
    sumsq = tl.zeros([], dtype=tl.float32)
    # loop over hidden_dim in chunks of BLOCK_H
    for offset in range(0, hidden_dim, BLOCK_H):
        cur_h = offset + tl.arange(0, BLOCK_H)
        # guard for overflow
        mask = cur_h < hidden_dim
        # load x and gate
        x = tl.load(x_ptr + base_offset + cur_h, mask=mask, other=0.0).to(tl.float32)
        sum_ += tl.sum(x, axis=0) # but backward across block?
        sumsq += tl.sum(x*x, axis=0)

    var = sumsq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    # second loop: compute normalized and store
    for offset in range(0, hidden_dim, BLOCK_H):
        cur_h = offset + tl.arange(0, BLOCK_H)
        mask = cur_h < hidden_dim
        x = tl.load(x_ptr + base_offset + cur_h, mask=mask, other=0.0).to(tl.float32)

def custom_kernel(data):
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Move tensors to same device
    device = inp.device

    # First layernorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    x = torch.nn.functional.layer_norm(inp,
                (dim,),
                weight=norm_weight,
                bias=norm_bias)
    # cast to fp16
    x = x.to(torch.float16)

    # concatenate weights for projects and gates
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # create concat weight
    concat_weight = torch.cat([
        left_proj_w, right_proj_w,
        left_gate_w, right_gate_w,
        out_gate_w
    ], dim=0).to(torch.float16)

    # Linear projection (combined)
    proj = torch.nn.functional.linear(x, concat_weight)  # shape (B,N,N,5*hidden)
    # free x maybe
    del x

    # reshape view
    batch_size, seq_len, _, total_out = proj.shape
    # total_out = 5*hidden_dim
    proj = proj.view(batch_size, seq_len, seq_len, 5, hidden_dim)
    # slices
    left_proj = proj[..., 0, :]
    right_proj = proj[..., 1, :]
    left_gate_raw = proj[..., 2, :]
    right_gate_raw = proj[..., 3, :]
    out_gate_raw = proj[..., 4, :]

    # allocate output tensors for left, right
    left = torch.empty_like(left_proj)   # fp16
    right = torch.empty_like(right_proj)

    # allocate out_gate
    out_gate = torch.empty_like(out_gate_raw)  # fp16

    # compute out_gate via torch.sigmoid
    out_gate = torch.sigmoid(out_gate_raw)

    # compute mask in fp16 if needed
    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        # create ones mask of same size
        mask_fp16 = torch.ones((batch_size, seq_len, seq_len), dtype=torch.float16, device=device)

    # flatten left_proj etc for launching kernels
    total_positions = batch_size * seq_len * seq_len
    total_elems = total_positions * hidden_dim

    BLOCK = 1024  # threads per block
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    # left compute kernel
    mask_gate_fuse_sigmoid_kernel[grid](
        left_proj,
        left_gate_raw,
        mask_fp16,
        left,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right compute kernel
    mask_gate_fuse_sigmoid_kernel[grid](
        right_proj,
        right_gate_raw,
        mask_fp16,
        right,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # Delete intermediate proj slices to free memory
    del proj
    del left_proj
    del right_proj
    del left_gate_raw
    del right_gate_raw
    del out_gate_raw
    # free mask if needed?

    # Contract left and right via batched matmul
    out = _batched_matmul_torch(left, right)  # shape [B,N,N,hidden] fp16

    # free left/right
    del left
    del right

    # second layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # multiply by out_gate
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast back to float32
    return out.to(torch.float32)

```
@triton.jit
def mask_gate_fuse_sigmoid_kernel(
    proj_ptr,    # *fp16
    gate_ptr,   # *fp16
    mask_ptr,   # *fp16
    out_ptr,    # *fp16
    hidden_dim, # i32
    size,       # i64 (# total elements)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_active = offs < size
    # compute mask offset
    mask_offs = offs // hidden_dim

    # load values
    x = tl.load(proj_ptr + offs, mask=mask_active, other=0.0)
    g_raw = tl.load(gate_ptr + offs, mask=mask_active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask_active, other=1.0)   # ones for out-of-bound

    # sigmoid of g_raw
    g_raw_f32 = g_raw.to(tl.float32)
    g_sig_f32 = 1.0 / (1.0 + tl.exp(-g_raw_f32))
    g_sig = g_sig_f32.to(tl.float16)

    out = x * g_sig * m
    tl.store(out_ptr + offs, out, mask=mask_active)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# fused kernel for left/right multiply by mask and sigmoid(gate)
@triton.jit
def mask_gate_fuse_sigmoid_kernel(
    proj_ptr,  # *fp16 left projection or right projection
    gate_raw_ptr,  # *fp16 raw gate
    mask_ptr,  # *fp16 mask
    out_ptr,  # *fp16 output
    hidden_dim,  # i32
    total_elems,  # i64
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elems
    # compute mask offset (mask one per hidden group)
    mask_offs = offs // hidden_dim

    x = tl.load(proj_ptr + offs, mask=active, other=0.0)
    g_raw = tl.load(gate_raw_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # sigmoid in fp32 for stability
    g_raw_f32 = g_raw.to(tl.float32)
    g = 1.0 / (1.0 + tl.exp(-g_raw_f32))
    g_f16 = g.to(tl.float16)

    out = x * g_f16 * m
    tl.store(out_ptr + offs, out, mask=active)

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left/right shape [B, N, N, hidden]
    B, N, _, H = left.shape
    # move hidden into batch dim
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # bmm with right transposed (k dim)
    c = torch.bmm(a, b.transpose(1, 2))  # [B*H,N,N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    """
    Fast outgoing TriMul forward (no autograd). Implements:
        1. LayerNorm (float32)
        2. One fused Linear for left/right projections and gates.
        3. Mask + sigmoid gating fused via Triton.
        4. Batched GEMM for the O(N³) contraction.
        5. Second LayerNorm + out‑gate, then final Linear.
    All heavy ops run in fp16; final output is float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # 1️⃣ First LayerNorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32
    # Cast to fp16 for the rest
    x = x.to(torch.float16)

    # 2️⃣ Concatenated projection (left, right, left_gate, right_gate, out_gate)
    # All weight matrices have shape (hidden_dim, dim)
    concat_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape (5*hidden_dim, dim)

    proj = torch.nn.functional.linear(x, concat_weight)  # [B,N,N,5*hidden_dim] fp16
    del x  # free earlier tensor

    B, N, _, _ = proj.shape
    proj = proj.view(B, N, N, 5, hidden_dim)  # view to separate channels

    left_proj = proj[..., 0, :]          # [B,N,N,hidden]
    right_proj = proj[..., 1, :]         # [B,N,N,hidden]
    left_gate_raw = proj[..., 2, :]      # [B,N,N,hidden]
    right_gate_raw = proj[..., 3, :]     # [B,N,N,hidden]
    out_gate_raw = proj[..., 4, :]       # [B,N,N,hidden]

    # out_gate = sigmoid(out_gate_raw)   # fp16
    out_gate = torch.sigmoid(out_gate_raw)

    # 3️⃣ Apply mask and sigmoid‑gate to left/right
    # Prepare mask in fp16, broadcasting over hidden dim.
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # Allocate output tensors for left/right
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_positions = B * N * N
    total_elems = total_positions * hidden_dim
    BLOCK = 1024  # threads per block (max 1024)
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    # Fuse mask, sigmoid(gate) and elementwise multiplication
    mask_gate_fuse_sigmoid_kernel[grid](
        left_proj,
        left_gate_raw,
        mask_fp16,
        left,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )

        right_gate_raw,
        mask_fp16,
        right,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # Release intermediate tensors early to free memory
    del proj
    del left_proj
    del right_proj
    del left_gate_raw
    del right_gate_raw
    del out_gate_raw
    del mask_fp16

    # 4️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core)
    out = _batched_matmul_torch(left, right)   # [B,N,N,hidden] fp16

    # Free left/right early
    del left
    del right

    # 5️⃣ Second LayerNorm (fp16) → out‑gate → final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,

        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    # final projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32 as required by the harness
    return out.to(torch.float32)

"""
Fast outgoing TriMul forward (no autograd) – Triton‑accelerated core.

The implementation follows the reference PyTorch module:
1️⃣ Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣ Fuse the five linear projections (left/right and three gates) into a single
   matmul by concatenating their weight matrices. This reduces the number of
   full‑tensor matmuls from five to one, dramatically cutting memory traffic.
3️⃣ Fuse mask‑multiplication, sigmoid‑gate, and element‑wise multiplication for
   the left and right terms with a tiny Triton kernel (still fp16‑only).
   The out‑gate is obtained with a regular `torch.sigmoid`.
4️⃣ Perform the O(N³) contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM using cuBLAS/Tensor‑Core (`torch.bmm`).
5️⃣ Second LayerNorm (fp16) → multiply by the pre‑computed out‑gate → final
   linear projection back to the original dimension.
All heavy work runs in fp16; the final result is cast back to float32
as required by the harness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: mask • sigmoid(gate) • projection (left or right term)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_sigmoid_kernel(
    proj_ptr,        # *fp16 : projected tensor (left_proj or right_proj)
    gate_raw_ptr,    # *fp16 : raw gate tensor (pre‑sigmoid)
    mask_ptr,        # *fp16 : mask tensor of shape (B, N, N)
    out_ptr,         # *fp16 : output tensor (left or right)
    hidden_dim,      # i32   : hidden dimension size H
    total_elements,  # i64   : total number of elements in the projection tensor
    BLOCK: tl.constexpr,  # compile‑time block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element indices
    active = offs < total_elements                     # boundary guard

    # Corresponding mask index (each hidden element shares the same mask entry)
    mask_offs = offs // hidden_dim

    # Load tensors
    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)          # (H) projection
    gate_raw = tl.load(gate_raw_ptr + offs, mask=active, other=0.0)  # raw gate
    mask_val = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # Sigmoid in fp32 for numerical stability, then cast back to fp16
    gate_f32 = gate_raw.to(tl.float32)
    gate_sig_f32 = 1.0 / (1.0 + tl.exp(-gate_f32))
    gate_sig = gate_sig_f32.to(tl.float16)

    out = proj * gate_sig * mask_val
    tl.store(out_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
# left/right: [B, N, N, H] (fp16)
# returns    : [B, N, N, H] (fp16)
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right is transposed on the last two dims to realize Σ_k left[i,k]·right[j,k]
    c = torch.bmm(a, b.transpose(1, 2))               # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point for the custom kernel.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable on the raw input.
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                   # [B, N, N, dim]  float32
    x = x.to(torch.float16)            # cast for the bulk of the work

    # --------------------------------------------------------------
    # 2️⃣ One fused linear (left/right + three gates) in fp16.
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each hidden_dim × dim)
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # shape (5*H, dim)

    # Linear projection – result shape [B, N, N, 5*H]
    proj = torch.nn.functional.linear(x, concat_weight)   # fp16
    del x  # free early

    B, N, _, _ = proj.shape
    proj = proj.view(B, N, N, 5, hidden_dim)               # split the 5 streams

    # Slice the five streams
    left_proj = proj[..., 0, :]          # [B, N, N, H]
    right_proj = proj[..., 1, :]         # [B, N, N, H]
    left_gate_raw = proj[..., 2, :]      # [B, N, N, H]
    right_gate_raw = proj[..., 3, :]     # [B, N, N, H]
    out_gate_raw = proj[..., 4, :]       # [B, N, N, H]

    # out_gate = sigmoid(out_gate_raw)   # fp16
    out_gate = torch.sigmoid(out_gate_raw)

    # --------------------------------------------------------------
    # 3️⃣ Mask + sigmoid‑gate fusion for left/right terms.
    # --------------------------------------------------------------
    if not nomask:
        # mask is given – cast to fp16 and make it contiguous.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        # No mask → a tensor full of ones (broadcasted over hidden dim).
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # Allocate destination buffers for the gated tensors.
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Flatten the per‑position × hidden layout for kernel launch.
    total_positions = B * N * N
    total_elements = total_positions * hidden_dim
    BLOCK = 1024                         # ≤ 1024 threads per block
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    # Fuse mask, sigmoid(gate) and multiplication.
    mask_gate_fuse_sigmoid_kernel[grid](
        left_proj,
        left_gate_raw,
        mask_fp16,
        left,
        hidden_dim,
        total_elements,
        BLOCK=BLOCK,
        num_warps=4,
    )
    mask_gate_fuse_sigmoid_kernel[grid](
        right_proj,
        right_gate_raw,
        mask_fp16,
        right,
        hidden_dim,
        total_elements,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # Ensure kernels have finished before releasing temporary tensors.
    torch.cuda.synchronize()

    # Release intermediate tensors that are no longer needed.
    del proj
    del left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw
    del mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ O(N³) contraction via a batched GEMM (Tensor‑Core).
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H] fp16

    # Release tensors we no longer need.
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16) → out‑gate → final linear projection.
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                                   # gating
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
    )                                                      # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # Cast back to float32 (required by the harness) and return.
    # --------------------------------------------------------------
    return out.to(torch.float32)
