def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # Unpack weights
    # Cast weights to fp16:
    def to_fp16(t): return t.to(device=device, dtype=torch.float16)
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias =weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # 1. LN (FP16)
    if torch.version.hip or torch.backends.cudnn.version(): Not relevant.

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, dtype=torch.float16)  # directly FP16

    # 2. Concatenated linear projection
    combined_weight = torch.cat(
        [left_proj_weight, right_proj_weight,
         left_gate_weight, right_gate_weight,
         out_gate_weight],
        dim=0)  # shape (5*hidden_dim, dim)
    combined_weight_fp16 = combined_weight.to(device=device, dtype=torch.float16)

    # compute projection
    combined = torch.nn.functional.linear(x, combined_weight_fp16)  # shape [B,N,N,5*hidden_dim] fp16

    # split
    h = hidden_dim
    left = combined[..., :h]
    right = combined[..., h:2*h]
    left_gate = combined[..., 2*h:3*h]
    right_gate = combined[..., 3*h:4*h]
    out_gate = combined[..., 4*h:5*h]

    # apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:
        # apply mask and gate with Triton fused
        mask_fp16 = mask_tensor.to(dtype=torch.float16)  # shape [B,N,N]
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr = left,
            right_ptr = right,
            left_gate_ptr = left_gate,
            right_gate_ptr = right_gate,
            mask_ptr = mask_fp16,
            total_elements = total,
            H = hidden_dim,
            BLOCK_SIZE = BLOCK,
            num_warps = 4,
            num_stages = 1,
        )
    # free left_gate right_gate mask
    # heavy contraction: out = left @ right^T per hidden.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)
    # alternative: use batched bmm with permute + view after we have left and right in shape (B,N,N,H).
    # For minimal copy, use einsum. Or optionally use batched matmul if needed.
    # Second LayerNorm

    out = torch.nn.functional.linear(out, to_out_weight.to(device=device, dtype=torch.float16))
    # cast back to fp32
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast forward implementation

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP16 (the reference implementation uses FP32,
    but casting the result to FP16 is safe for the test tolerances and
    reduces bandwidth).

2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single GEMM by concatenating their weight matrices.
    This replaces five separate `linear` calls with one large matrix
    multiplication, cutting the number of kernel launches and the reads of
    the large input tensor.

3️⃣  Split the fused tensor into the four intermediate tensors and the
    out‑gate; apply `sigmoid` to the three gate tensors.

4️⃣  Optional mask: if a mask is present we fuse the element‑wise mask,
    left‑gate and right‑gate multiplication in a tiny Triton kernel (zero
    extra memory traffic).

5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is exactly `einsum('b i k h, b j k h -> b i j h')`.
    Using `torch.einsum` avoids the costly permute+contiguous copy used
    by the naïve batched `bmm` version.

6️⃣  Second LayerNorm over the hidden dimension (still FP16).

7️⃣  Multiply by the out‑gate and apply the final linear projection back
    to the original `dim`.

8️⃣  Cast the result back to FP32 (the harness expects float32).

All heavy tensor‑core‑friendly work is performed in FP16, while only the
layer‑norm statistics (mean/var) are computed in FP32 (via the functional
layer‑norm call).  The only Triton kernel required is the mask‑gate fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * left_gate  * mask
#   right ← right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i64   total number of elements (B·N·N·H)
    H,                 # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat indices
    active = offs < total_elements

    # Map flat index to mask index (broadcast on hidden dim)
    mask_offs = offs // H

    # Load tensors (masked loads guarantee zero for OOB)
    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)

    # Load mask (0/1). For OOB we use a neutral value 1.0.
    mask_val = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # Fuse the three factors
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Write results back in‑place
    tl.store(left_ptr  + offs, left_out ,  mask=active)
    tl.store(right_ptr + offs, right_out,  mask=active)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 0️⃣  Cast needed model parameters to FP16 (one‑time cost)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=dtype_fp16)
    norm_b = weights["norm.bias"].to(device=device, dtype=dtype_fp16)

    # Concatenate the five projection weight matrices:
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)      # shape (5*hidden_dim, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (float32 -> cast to FP16)
    # ------------------------------------------------------------------
    # Use the FP32 LayerNorm implementation then cast to half.
    # This maintains numerical stability for the mean/var computation.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                          # [B, N, N, dim] float32
    x = x.to(dtype_fp16)                       # now half

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections (single GEMM)
    # ------------------------------------------------------------------
    # Result shape: [B, N, N, 5*hidden_dim]
    fused = F.linear(x, concat_w)              # FP16

    # ------------------------------------------------------------------
    # 3️⃣ Slice the fused tensor into the four intermediates and the out‑gate
    # ------------------------------------------------------------------
    h = hidden_dim
    left      = fused[..., 0:h]                # [B, N, N, h]
    right     = fused[..., h:2 * h]            # [B, N, N, h]
    left_gate = fused[..., 2 * h:3 * h]        # [B, N, N, h]
    right_gate= fused[..., 3 * h:4 * h]        # [B, N, N, h]
    out_gate  = fused[..., 4 * h:5 * h]        # [B, N, N, h]

    # Apply the sigmoid activations to the three gate tensors
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)      # used later

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → just multiply by the gates
        left  = left * left_gate
        right = right * right_gate
    else:
        mask_fp16 = mask_tensor.to(dtype_fp16)          # [B, N, N]
        # Ensure all tensors are contiguous (required for the kernel)
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gate tensors are no longer needed after fusion
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction using einsum (avoids copy‑permute)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                          # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)             # FP16 → [B, N, N, dim]

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the external harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
