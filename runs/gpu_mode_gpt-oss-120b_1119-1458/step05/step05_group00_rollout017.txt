@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program id
    pid_i = tl.program_id(0)  # block row index i
    pid_j = tl.program_id(1)  # block col index j
    pid_head = tl.program_id(2)

    # Compute batch and head
    b = pid_head // H
    h = pid_head % H

    # Offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for i and j
    mask_i = i < N
    mask_j = j < N

    # Base pointers (head)
    left_base = left_ptr + b * stride_left_b + h  # h offset as innermost
    right_base = right_ptr + b * stride_right_b + h
    out_base = out_ptr + b * stride_out_b + h

    # ptr offsets for i and j
    i_offs = i * stride_left_i  # shape (BLOCK_M)
    # j_offs = j * stride_right_i (since right's i dimension is stride_i)
    j_offs = j * stride_right_i

    # Prepare accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile (M x K)
        # pointer shape: left_base + i_offs[:,None] + k_offset[None,:] * stride_left_k
        a_ptrs = left_base + i_offs[:, None] + k_offset[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)  # (BLOCK_M, BLOCK_K) fp16
        # Load right tile (K x N)
        # right tile pointer: right_base + k_offset[:,None] * stride_right_k + j_offs[None,:]
        b_ptrs = right_base + k_offset[:, None] * stride_right_k + j_offs[None, :]
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)  # (BLOCK_K, BLOCK_N) fp16

        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # Write output
    # ptr out: out_base + i_offs[:, None] + j_offs[None, :]
    out_ptrs = out_base + i_offs[:, None] + j_offs[None, :]
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – High‑performance Triton implementation
---------------------------------------------------------

Algorithm
~~~~~~~~~
1. **LayerNorm** (float32) over the channel dimension `dim`.
2. **Linear projections** (left/right) and **gates** (left/right/out) → FP16.
3. **Mask & gate fusion** (optional) – a tiny Triton kernel that
   multiplies the projected tensors by the (optional) mask and by the
   sigmoid gates, all in‑place.
4. **Heavy O(N³) contraction**:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication `C = A @ Bᵀ` with
   `A = left` and `B = right`.  We implement it with a custom Triton
   kernel that loops over the reduction dimension `k`.  Each kernel
   instance works on a tile `(BLOCK_M, BLOCK_N)` of the output for a
   single head `(b, h)`.  Accumulation is performed in float32 for
   numerical stability.
5. **Second LayerNorm** (FP16), multiplication by the `out_gate`,
   and final linear projection back to the original `dim`.
6. Cast the result back to float32 (the harness expects FP32).

Only part of the op (mask + gate fusion and the N³ contraction) is
implemented in Triton; the remaining lightweight operations use
torch.nn.functional for maximal throughput.

The kernel is tailored for the “outgoing” version used in AlphaFold‑3:
∎
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1) In‑place mask & gate fusion kernel
#    left  ← left  * mask * left_gate
#    right ← right * mask * right_gate
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in‑place)
    right_ptr,         # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcast over H)
    total_elements,    # i64: B·N·N·H
    H,                 # i64: hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # The mask is broadcast along the hidden dimension.
    mask_offs = offs // H                     # index in [B,N,N]

    # Load tensors (masked loads give 0 for OOB)
    left       = tl.load(left_ptr       + offs, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + offs, mask=valid, other=0.0)
    left_gate  = tl.load(left_gate_ptr  + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    mask_val   = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


# ----------------------------------------------------------------------
# 2) Batched N³ contraction kernel
#    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,      # *fp16  [B,N,N,H] (input)
    right_ptr,     # *fp16  [B,N,N,H] (input)
    out_ptr,       # *fp16  [B,N,N,H] (output)
    B, N, H,       # i64 scalars
    stride_left_b, stride_left_i, stride_left_k,  # i64
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b,   stride_out_i,   stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------- program ids -------------------------------------------------
    pid_i   = tl.program_id(0)                     # output row block index
    pid_j   = tl.program_id(1)                     # output col block index
    pid_h   = tl.program_id(2)                     # flattened (batch, head) index

    # ------------- decode head index -----------------------------------------
    b = pid_h // H                                   # batch index
    h = pid_h % H                                    # hidden‑dim index

    # ------------- offsets for the tiles ---------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)     # [BLOCK_M]
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)     # [BLOCK_N]

    mask_i = i < N
    mask_j = j < N

    # ------------- base pointers for this (b,h) -------------------------------
    # Strides of the hidden dim are 1, so we simply add `h`.
    left_base  = left_ptr  + b * stride_left_b  + h
    right_base = right_ptr + b * stride_right_b + h
    out_base   = out_ptr   + b * stride_out_b   + h

    # Offsets for i and j dimensions (h is already baked into the base)
    i_off = i * stride_left_i      # (BLOCK_M)
    j_off = j * stride_out_j       # (BLOCK_N)

    # ------------- accumulator -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------- reduction over k -------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)      # (BLOCK_K)
        mask_k = k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i_off[:, None] + k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + k[:, None] * stride_right_k + j_off[None, :]
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ------------- write output ------------------------------------------------
    out_ptrs = out_base + i_off[:, None] + j_off[None, :]
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 3) Helper: wrapper around the Triton N³ kernel
# ----------------------------------------------------------------------
def _contract_tri_mul(left: torch.Tensor, right: torch.Tensor, config: Dict[str, Any]) -> torch.Tensor:
    """
    Perform the heavy contraction using the custom Triton kernel.
    Parameters
    ----------
    left, right : torch.Tensor [B, N, N, H] (fp16, contiguous)
    config : dict containing possible tuning parameters (none required now)

    Returns
    -------
    out : torch.Tensor [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    device = left.device

    # Allocate output
    out = torch.empty_like(left)

    # Strides (int64) – use Python int for Triton
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tile sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M ),
        ( (N + BLOCK_N - 1) // BLOCK_N ),
        B * H,
    )
    _tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=s_lb,
        stride_left_i=s_li,
        stride_left_k=s_lk,
        stride_right_b=s_rb,
        stride_right_i=s_ri,
        stride_right_k=s_rk,
        stride_out_b=s_ob,
        stride_out_i=s_oi,
        stride_out_j=s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# 4) Entry point for the custom kernel
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights_dict  : mapping name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``; optional
                          ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm in float32 (as in the reference implementation)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)          # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Load model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Gating projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # used later

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Prepare mask as FP16 (0/1) with broadcasting over hidden dimension
        mask_fp16 = mask_tensor.to(dtype_fp16)                  # [B,N,N]
        # Make sure all tensors are contiguous
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # We no longer need the gate tensors
        del left_gate, right_gate, mask_fp16
    else:
        # No mask → just apply the gates
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction (custom Triton kernel)
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous before handing them to the kernel
    left  = left.contiguous()
    right = right.contiguous()
    out = _contract_tri_mul(left, right, config)   # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 9️⃣ Apply final out_gate and project back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise (FP16)
    out = F.linear(out, to_out_w)            # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
