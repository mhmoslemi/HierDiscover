def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    # load weights
    norm_w = weights["norm.weight"]; norm_b = weights["norm.bias"]; ...
    # convert to appropriate dtype
    # Step 1: Layer norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)
    x_norm = x_norm.to(torch.float16) # maybe keep as fp16
    # reshape x_norm for fused linear
    B,N,N,_ = x_norm.shape
    M = B * N * N
    x_flat = x_norm.view(M, dim)
    # Fuse linear weights
    weight_list = [weights["left_proj.weight"], weights["right_proj.weight"],
                   weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]]
    weight_concat = torch.cat(weight_list, dim=0).to(x.device, dtype=torch.float16)
    # fused projection: shape M x (5*hidden)
    proj = torch.nn.functional.linear(x_flat, weight_concat) # bias=None
    # reshape back
    proj = proj.view(B, N, N, hidden_dim*5)
    left_proj = proj[..., 0:hidden_dim]
    right_proj = proj[..., hidden_dim:2*hidden_dim]
    left_gate = proj[..., 2*hidden_dim:3*hidden_dim]
    right_gate = proj[..., 3*hidden_dim:4*hidden_dim]
    out_gate = proj[..., 4*hidden_dim:5*hidden_dim]
    # mask
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16
    # gating sigmoids
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)
    # gating multiplication kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _gate_fused_mul_inplace(left_proj, left_gate, right_proj, right_gate)
    # contraction
    # left_proj and right_proj shape [B,N,N,H] (H=hidden_dim)
    B, N, _, H = left_proj.shape
    left_flat = left_proj.permute(0,3,1,2).contiguous().view(B*H, N, N)  # shape [B*H, N, N]
    right_flat = right_proj.permute(0,3,1,2).contiguous().view(B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2)) # bs N N
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    # second LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    # out gate multiplication (elementwise)
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16)
    )
    # Cast back to float32
    return out.to(torch.float32)

@triton.jit
def _gate_fused_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    N, BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # mul
    left_out = left * left_gate
    right_out = right * right_gate
    # store back
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)

def _gate_fused_mul_inplace(left: torch.Tensor, left_gate: torch.Tensor,

    assert left_gate.is_contiguous()
    assert right.is_contiguous()

        BLOCK_SIZE=BLOCK,

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for fused elementwise multiplication of left/right with gates
@triton.jit
def _gate_fused_mul_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    N, BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    left_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    left_val = left_val * left_gate_val
    right_val = right_val * right_gate_val

    tl.store(left_ptr + offs, left_val, mask=mask)
    tl.store(right_ptr + offs, right_val, mask=mask)

def _gate_fused_mul_inplace(left: torch.Tensor, left_gate: torch.Tensor,
                            right: torch.Tensor, right_gate: torch.Tensor):
    assert left.is_contiguous() and left_gate.is_contiguous()
    assert right.is_contiguous() and right_gate.is_contiguous()
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _gate_fused_mul_kernel[grid](
        left, left_gate,
        right, right_gate,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul (AlphaFold‑3) forward pass with heavy operations fused.
    1️⃣   LayerNorm in FP32 → cast to FP16.
    2️⃣   Fuse the five linear projections (left/right and gates, plus out_gate) into one GEMM.
    3️⃣   Optional mask multiplication (torch) then fused gate multiplication via Triton.
    4️⃣   Batched GEMM (torch.bmm) computes the cubic contraction.
    5️⃣   Second LayerNorm + out‑gate scaling.
    6️⃣   Final linear projection back to ``dim``.
    Returns FP32 tensor of shape [B, N, N, dim].
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm across last dimension (dim)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Cast to FP16 for the heavy work
    x = x.to(torch.float16)  # shape [B, N, N, dim]

    B, N, _, _ = x.shape
    M = B * N * N

    # --------------------------------------------------------------
    # 2️⃣ Fuse linear projections (5·hidden_dim) into a single matmul
    # --------------------------------------------------------------
    # Gather weight tensors and cast to FP16
    w_left_proj = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    w_left_gate = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Concatenate along output dimension (axis 0)
    w_concat = torch.cat([w_left_proj, w_right_proj,
                          w_left_gate, w_right_gate,
                          w_out_gate], dim=0)  # shape [(5*hidden_dim), dim]

    # Perform a single GEMM: (B·N·N, dim) @ (dim, 5·hidden_dim)ᵀ
    x_flat = x.reshape(M, dim)  # [M, dim]
    proj = torch.nn.functional.linear(x_flat, w_concat)  # bias=None → shape [M, 5*hidden_dim]

    # Reshape back to 4‑D and split into the five tensors
    proj = proj.view(B, N, N, hidden_dim * 5)
    left = proj[..., 0:hidden_dim]                     # [B,N,N,hidden]
    right = proj[..., hidden_dim:2*hidden_dim]
    left_gate = proj[..., 2*hidden_dim:3*hidden_dim]
    right_gate = proj[..., 3*hidden_dim:4*hidden_dim]
    out_gate = proj[..., 4*hidden_dim:5*hidden_dim]

    # Free temporary concatenated weight to save memory
    del w_concat, x_flat, proj

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates (still FP16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 5️⃣ Fuse the two element‑wise multiplications via Triton
    #    left ← left * left_gate
    #    right ← right * right_gate
    # --------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _gate_fused_mul_inplace(left, left_gate, right, right_gate)

    # --------------------------------------------------------------
    # 6️⃣ Cubic contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM over the hidden dimension.
    # --------------------------------------------------------------
    # Reshape to [B·hidden, N, N] before the batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout: [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) → scale by out_gate → final projection
    # --------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    b_out_norm = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=w_out_norm,
                                         bias=b_out_norm)

    # Apply out‑gate (element‑wise)
    out = out * out_gate

    # Final linear back to dim (FP16)
    w_to_out = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, w_to_out)  # bias=False

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  LayerNorm (FP32) on the [B,N,N,dim] tensor, then cast to FP16.
2️⃣  Fuse the five linear projections (left/right and three gates) into a single
    GEMM by concatenating the weight matrices.  This avoids loading the huge
    input tensor five times.
3️⃣  (Optional) apply the mask on `left`/`right`.
4️⃣  Compute sigmoid gates (FP16) and fuse the two element‑wise multiplications
    `left *= left_gate` and `right *= right_gate` with a custom Triton kernel.
5️⃣  Perform the cubic contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM (`torch.bmm`) after reshaping to [B·hidden,N,N].
6️⃣  Second LayerNorm (FP16) → element‑wise scaling by `out_gate`.
7️⃣  Final linear projection back to the original `dim`.
8️⃣  Cast the result back to FP32.

Only the element‑wise gate multiplication is performed with Triton; all dense
matrix multiplications use cuBLAS/Tensor‑cores which is optimal on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused in‑place element‑wise multiplication
#    left ← left * left_gate
#    right ← right * right_gate
# ----------------------------------------------------------------------
@triton.jit
def _gate_fused_mul_kernel(
    left_ptr,          # *fp16   left tensor (input & output)
    left_gate_ptr,     # *fp16   left gate
    right_ptr,         # *fp16   right tensor (input & output)
    right_gate_ptr,    # *fp16   right gate
    N,                 # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(left_ptr + offs, mask=mask, other=0.0)
    g_a = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    b = tl.load(right_ptr + offs, mask=mask, other=0.0)
    g_b = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    a = a * g_a
    b = b * g_b

    tl.store(left_ptr + offs, a, mask=mask)
    tl.store(right_ptr + offs, b, mask=mask)


def _gate_fused_mul_inplace(left: torch.Tensor,
                            left_gate: torch.Tensor,
                            right: torch.Tensor,
                            right_gate: torch.Tensor) -> None:
    """
    In‑place element‑wise multiplication:
        left  ← left  * left_gate
        right ← right * right_gate
    All tensors must be contiguous FP16 of the same shape.
    """
    assert left.is_contiguous() and left_gate.is_contiguous()
    assert right.is_contiguous() and right_gate.is_contiguous()
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _gate_fused_mul_kernel[grid](
        left, left_gate, right, right_gate, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 for stability) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)      # [B, N, N, dim] in FP16

    B, N, _, _ = x.shape
    M = B * N * N                # flatten size

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the five linear projections (left/right + gates) into one GEMM
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device, dtype=torch.float16)
    w_left_gate = weights["left_gate.weight"].to(device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device, dtype=torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device, dtype=torch.float16)

    # concatenate along output dimension: (5*hidden_dim, dim)
    w_concat = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0
    )

    # Perform a single GEMM: (M, dim) @ (dim, 5*hidden_dim).T → (M, 5*hidden_dim)
    x_flat = x.reshape(M, dim)
    proj = torch.nn.functional.linear(x_flat, w_concat)  # bias=None

    # Reshape back and split into the five tensors
    proj = proj.view(B, N, N, hidden_dim * 5)
    left = proj[..., 0:hidden_dim]
    right = proj[..., hidden_dim:2 * hidden_dim]
    left_gate = proj[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = proj[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate = proj[..., 4 * hidden_dim:5 * hidden_dim]

    # free large temporary tensors early
    del w_concat, x_flat, proj

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣  Compute sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5️⃣  Fuse gate multiplication for left and right via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _gate_fused_mul_inplace(left, left_gate, right, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣  Heavy cubic contraction – batched GEMM (cuBLAS/Tensor‑cores)
    # ------------------------------------------------------------------
    # reshape to [B·hidden, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 8️⃣  Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
