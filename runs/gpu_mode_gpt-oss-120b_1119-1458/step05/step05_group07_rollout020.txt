def batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left/right shape: [B, N, N, H] (fp16)
    B, N, _, H = left.shape
    # reshape
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    # allocate output
    out = torch.empty_like(left_reshaped)
    # get strides in elements
    stride_ab = left_reshaped.stride(0)
    stride_am = left_reshaped.stride(1)
    stride_ak = left_reshaped.stride(2)
    stride_bb = right_reshaped.stride(0)
    stride_bn = right_reshaped.stride(1)
    stride_bk = right_reshaped.stride(2)
    stride_cb = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)
    batch_total = B * H
    # grid dims
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32 # or 64
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), batch_total)
    batched_gemm_kernel[grid](
        left_reshaped, right_reshaped, out,
        batch_total, N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bn, stride_bk,
        stride_cb, stride_cm, stride_cn,

def custom_kernel(data):
    """
    Fast outgoing TriMul forward pass.

    Combines layernorm, linear projections, gating, mask multiplication,
    and a batched matrix multiplication fused in Triton.
    """
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # 1. First LayerNorm (float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # cast for heavy ops

    # 2. Prepare combined projection weight
    w_combined = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)   # shape (5*hidden_dim, dim)

    # 3. Linear projection for all five tensors at once
    proj = torch.nn.functional.linear(x, w_combined)   # shape [bs, seq_len, seq_len, 5*hidden_dim], fp16

    # 4. Split
    left, right, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        proj, hidden_dim, dim=-1
    )
    # 5. Activation
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # 6. Elementwise gating (with optional mask) – fused kernel
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)
        # left = left * left_gate * mask
        total_elts = left.numel()
        BLOCK = 1024
        grid = ( (total_elts + BLOCK - 1) // BLOCK, )
        fused_gate_mask_kernel[grid](
            left, left_gate, mask_fp16,
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        fused_gate_mask_kernel[grid](
            right, right_gate, mask_fp16,
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: simple elementwise multiply
        left = left * left_gate
        right = right * right_gate

    # 7. Batched GEMM using Triton (out = left @ right.T for each hidden channel)
    out = batched_matmul_triton(left, right)  # shape [bs, seq_len, seq_len, hidden_dim]

    # 8. Second LayerNorm (fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # 9. Out gate multiplication
    out = out * out_gate

    # 10. Final linear projection (fp16)
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # 11. Cast to float32 and return
    return out.to(torch.float32)

@triton.jit
def fused_gate_mask_kernel(
    x_ptr,            # *fp16 : tensor to update (size = total_elts)
    gate_ptr,         # *fp16 : gate tensor (same shape as x)
    mask_ptr,         # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate
    hidden_dim: tl.int32,  # size of the hidden dimension
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    valid = offs < total_elts

    # Decode the mask index (mask repeats every `hidden_dim` entries)
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM  C = A @ Bᵀ
# A : [batch, M, K]   (fp16)
# B : [batch, N, K]   (fp16)  – we treat it as Bᵀ inside the kernel
# C : [batch, M, N]   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # *fp16 pointers
    M, N, K,                        # matrix dimensions (int32)
    stride_ab, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bb, stride_bn, stride_bk,   # B strides: batch, N (row), K (col)
    stride_cb, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM using Tensor‑Cores.
    Each (batch_id) processes its own matrix pair.
    """
    pid_b = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)          # block row index
    pid_n = tl.program_id(1)          # block column index

    # Linear offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Batch offsets
    a_batch = pid_b * stride_ab
    b_batch = pid_b * stride_bb
    c_batch = pid_b * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + a_batch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        # Bᵀ tile: [BLOCK_K, BLOCK_N]  (loaded from B as B[batch, N, K])
        b_ptrs = b_ptr + b_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # fp16 dot → fp32 accumulation

    # Cast back to fp16 and store
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + c_batch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: batched GEMM implemented with the Triton kernel above.
# ----------------------------------------------------------------------
def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a Triton kernel.
    left/right: shape [B, N, N, H] (fp16)
    Returns: tensor of same shape (fp16)
    """
    B, N, _, H = left.shape
    batch_total = B * H                     # flattened batch dimension

    # Transpose to [batch, M, K] where M=N, K=N
    a = left.permute(0, 3, 1, 2).contiguous().view(batch_total, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(batch_total, N, N)

    # Allocate output
    c = torch.empty_like(a)

    # Strides in elements
    stride_ab = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    stride_bb = b.stride(0)
    stride_bn = b.stride(1)   # stride over N dimension (row)
    stride_bk = b.stride(2)   # stride over K dimension (col)

    stride_cb = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    # Tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (tl.cdiv(N, BLOCK_M),),   # blocks along M (i)
        (tl.cdiv(N, BLOCK_N),),   # blocks along N (j)
        (batch_total,)            # batch dimension
    )

    batched_gemm_kernel[grid](
        a, b, c,
        N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bn, stride_bk,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,           # more warps → better utilization on H100
        num_stages=3,
    )

    # Reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) – ignored if config["nomask"]==True
        - weights_dict  : dict name → torch.Tensor (float32)
        - config_dict   : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → cast to FP16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # work in FP16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into one matmul
    # ------------------------------------------------------------------
    # Concatenate weight matrices: each has shape (hidden_dim, dim)
    w_combined = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape (5*hidden_dim, dim)

    # Single linear projection → shape [B, N, N, 5*hidden_dim]
    proj = torch.nn.functional.linear(x, w_combined)  # no bias

    # ------------------------------------------------------------------
    # 3️⃣ Split into the five tensors
    # ------------------------------------------------------------------
    left, right, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        proj, hidden_dim, dim=-1
    )  # each → [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Activations for the three gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse element‑wise gating (+ mask) with a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 and ensure contiguous layout
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()                     # B * N * N * hidden_dim
        BLOCK = 1024                                   # ≤ 1024 threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left, left_gate, mask_fp16,
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right, right_gate, mask_fp16,
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – cheap element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (Triton)
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # shape [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # → [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
