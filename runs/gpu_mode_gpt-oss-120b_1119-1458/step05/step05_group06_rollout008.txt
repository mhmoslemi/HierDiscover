def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    # layernorm = first norm (float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to fp16
    x = x.half()
    
    # combine left_proj and left_gate
    left_comb_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_comb_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    # fuse linear pass for left+gate
    left_comb = F.linear(x, left_comb_w)  # shape [B,N,N,2*hidden_dim]
    left_proj = left_comb[..., :hidden_dim]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])
    # fuse linear for right+gate
    right_comb = F.linear(x, right_comb_w)
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])
    
    # apply mask if needed, before gating multiplication
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # compute out_gate gate on x
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].half()))  # shape [B,N,N,hidden_dim]
    
    # contraction
    seq_len = input_tensor.shape[1]
    if seq_len <= 256: # or maybe always call triton?
        contracted = _batched_matmul_triton(left, right)
    else:
        contracted = _batched_matmul_torch(left, right)

    # second layernorm
    contracted = torch.nn.functional.layer_norm(contracted, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # gating and final projection
    contracted = contracted * out_gate
    out = F.linear(contracted, weights["to_out.weight"] .half())
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fused forward pass.

Algorithm
---------
1️⃣ Apply LayerNorm over the feature dimension (`dim`) in fp32.
2️⃣ Cast to fp16 for the heavy compute.
3️⃣ Fuse the two projection Linear layers (`left_proj`/`left_gate`,
   `right_proj`/`right_gate`) into a single matmul per side:
   • Stack the projection‑weight and gate‑weight (both have shape
     [hidden_dim, dim]) → weight of shape [2·hidden_dim, dim].
   • One `F.linear` gives both the raw projection and the gate logits.
   • Apply sigmoid to the gate logits, mask (if present) the raw
     projection, then multiply by the gate.
4️⃣ Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   – For `seq_len ≤ 256` we launch a custom Triton kernel that performs a
     batched GEMM (A @ Bᵀ) with fp16 inputs and fp32 accumulation.
   – For larger sequences we fall back to `torch.bmm` (cuBLAS) which is
     faster on H100 for big matrices.
5️⃣ Second LayerNorm over the hidden dimension (`hidden_dim`) in fp32.
6️⃣ Compute the out‑gate (`out_gate`) from the original normalized input,
   apply it element‑wise, and finally project back to the original
   `dim` with a bias‑free Linear layer.
7️⃣ Cast the result back to fp32.

All heavy kernels run in fp16 (mixed‑precision) while the two LayerNorms
are performed in fp32 for numerical stability.  The implementation
covers every test configuration while keeping the amount of kernel
launches minimal (only the O(N³) contraction uses Triton).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (fp16 → fp16, accumulator fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction:
       C[m, i, j] = Σₖ A[m, i, k] * B[m, j, k]   (B is accessed transposed).
    """
    pid_m = tl.program_id(0)   # tile along output row i
    pid_n = tl.program_id(1)   # tile along output col j
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # Offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                    # [BLOCK_K]

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_tile * BLOCK_K
        cur_offs_k = cur_k + offs_k

        # Load A[i, k]
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + cur_offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # Load Bᵀ[j, k]   (B is stored as (j, k), we read it as (k, j))
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + cur_offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot product
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # Store the result
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left & right (both [B, N, N, H]) using a custom Triton GEMM.
    Returns a tensor of shape [B, N, N, H] in fp16.
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B * H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # tile size – 64×64 works well on H100 for fp16 tensor‑core ops
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left & right using cuBLAS via torch.bmm.
    Returns [B, N, N, H] in fp16.
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool)
                         – may be all‑ones if ``nomask`` is True.
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 3️⃣ fp16 cast for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fuse projection + gate linear layers
    # ------------------------------------------------------------------
    # left side: [proj, gate] stacked
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)
    # right side: [proj, gate] stacked
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # fused linear (proj + gate logits)
    left_comb = F.linear(x, left_comb_weight)   # [B, N, N, 2·hidden_dim]
    right_comb = F.linear(x, right_comb_weight) # [B, N, N, 2·hidden_dim]

    # split projection & gate logits
    left_proj = left_comb[..., :hidden_dim]
    left_gate_logits = left_comb[..., hidden_dim:]
    right_proj = right_comb[..., :hidden_dim]
    right_gate_logits = right_comb[..., hidden_dim:]

    # gate activations (sigmoid)
    left_gate = torch.sigmoid(left_gate_logits)   # fp16
    right_gate = torch.sigmoid(right_gate_logits)

    # ------------------------------------------------------------------
    # 5️⃣ optional mask (broadcasted to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B, N, N, hidden_dim]  fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ out‑gate (computed only once from the original normalized x)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 8️⃣ heavy contraction (batched GEMM)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)  # fp16
    else:
        contracted = _batched_matmul_torch(left, right)  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (float32) → out_gate → final Linear
    # ------------------------------------------------------------------
    # Cast to fp32 for stable LayerNorm computation
    contracted_fp32 = contracted.to(torch.float32)
    contracted_norm_fp32 = torch.nn.functional.layer_norm(
        contracted_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Back to fp16 for the following element‑wise ops
    contracted_norm = contracted_norm_fp32.to(torch.float16)

    # Apply out‑gate
    contracted_norm = contracted_norm * out_gate

    # Final projection back to `dim`
    out = F.linear(contracted_norm, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the test harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
