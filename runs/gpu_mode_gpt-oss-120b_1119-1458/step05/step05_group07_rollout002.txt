"""
Outgoing TriMul forward pass (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32 → FP16).
2️⃣  Linear projections (left/right) and three gates (left‑gate, right‑gate,
    out‑gate) in half‑precision.
3️⃣  If a mask is present, fuse `tensor * gate * mask` with a tiny Triton
    kernel (the mask is broadcast over the hidden dimension).
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed with a single ``torch.einsum`` which avoids any
    intermediate transposes/copies.
5️⃣  Second Layer‑norm (FP16) → multiply by ``out_gate`` → final linear
    projection back to the original feature dimension.
6️⃣  Cast the result back to FP32 (required by the testing harness).

Only the forward pass is needed; gradients are not computed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                # *fp16  : output tensor (in‑place)
    gate_ptr,             # *fp16  : gate tensor (same shape as x)
    mask_ptr,             # *fp16  : mask tensor (shape = B*N*N, broadcasted over H)
    total_elts: tl.int32, # total number of elements in x/gate
    hidden_dim: tl.int32, # size of the hidden dimension H
    BLOCK: tl.constexpr,  # tile size (≤ 1024)
):
    """
    In‑place:  x ← x * gate * mask
    ``mask`` is broadcast across the hidden dimension.  Each ``hidden_dim``
    consecutive elements in ``x`` share the same mask entry.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    mask = offs < total_elts

    # which mask element corresponds to this linear offset?
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (True ⇒ ignore the mask).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # layer_norm works on the last dim only – we pass (dim,) as normalized_shape
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                # [B, N, N, dim]  (float32)
    x = x_norm.to(torch.float16)   # cast for the heavy part

    # ------------------------------------------------------------------
    # 2️⃣  Cast model weights once (FP16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Linear projections (half‑precision)
    # ------------------------------------------------------------------
    # `torch.nn.functional.linear` expects weight shape (out_features, in_features)
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣  Gating signals (half‑precision)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣  Optional mask – fused element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; broadcast across hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()               # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication (in‑place)
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Heavy O(N³) contraction with a single einsum
    # ------------------------------------------------------------------
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # The einsum works directly on the original layout and avoids any
    # explicit transpose / copy required for a batched GEMM.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣  Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣  Final gating + projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
