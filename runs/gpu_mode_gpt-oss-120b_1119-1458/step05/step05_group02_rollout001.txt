"""
Outgoing TriMul (AlphaFold‑3) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) – the only operation kept in FP32 for numerical
    stability.

2️⃣  Cast to FP16 and perform **fused linear+gate** projections:
        - left_proj  + left_gate   → one matmul, split & sigmoid(left_gate)
        - right_proj + right_gate  → one matmul, split & sigmoid(right_gate)
        - out_gate  (sigmoid) computed separately.

3️⃣  If a mask is present, fuse the three element‑wise multiplications
    ``left = left * left_gate * mask`` and ``right = right * right_gate * mask``
    into a **single Triton kernel** that updates both tensors in‑place
    (dramatically cuts launch overhead and memory traffic).

4️⃣  Heavy O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Implemented as a batched GEMM on tensors reshaped to
    [B·H, N, N] (H = hidden_dim) – leverages cuBLAS Tensor‑Core kernels.

5️⃣  Second LayerNorm over the hidden dimension (FP16).

6️⃣  Gating with ``out_gate`` and final linear projection back to the original
    ``dim`` (FP16 → cast to FP32 for the harness).

All heavy element‑wise work (mask + gating) is performed by Triton;
the O(N³) contraction stays in cuBLAS because it is already near the
hardware peak on an H100.

The implementation works for any ``B∈{1,2}``, ``N∈{128,256,512,1024}``,
``dim∈{128,384}``, ``hidden_dim=128`` and supports both masked and unmasked
inputs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate applications.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_two_kernel(
    left_ptr,          # *fp16  : left projection tensor (B,N,N,H)
    left_gate_ptr,     # *fp16  : left gate tensor    (same shape)
    right_ptr,         # *fp16  : right projection tensor (B,N,N,H)
    right_gate_ptr,    # *fp16  : right gate tensor   (same shape)
    mask_ptr,          # *fp16  : mask tensor (B,N,N) – broadcast on H
    hidden_dim,        # i32    : H
    size,              # i32    : total number of elements in left/right
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear indices
    active = offs < size

    # mask index = element index divided by hidden_dim (since H is the innermost stride)
    mask_offs = offs // hidden_dim

    # load tensors
    l   = tl.load(left_ptr      + offs, mask=active, other=0.0)
    lg  = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    r   = tl.load(right_ptr       + offs, mask=active, other=0.0)
    rg  = tl.load(right_gate_ptr  + offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr        + mask_offs, mask=active, other=1.0)

    # fused multiplication
    l_out = l * lg * m
    r_out = r * rg * m

    # store back in‑place
    tl.store(left_ptr  + offs, l_out, mask=active)
    tl.store(right_ptr + offs, r_out, mask=active)


# ----------------------------------------------------------------------
# Helper: batched GEMM that implements the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a batched GEMM (cuBLAS Tensor‑Core).
    All tensors must be fp16.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] (each hidden channel becomes a separate batch)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # bmm with right transposed on the second dimension → matches einsum
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input tensor [B, N, N, dim] (float32)
        torch.Tensor,                     # mask tensor  [B, N, N]   (float32/bool) or None
        Dict[str, torch.Tensor],          # model weights (float32)
        Dict[str, Any]                    # config dict (contains dim, hidden_dim, optional nomask)
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Returns:
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)               # True → no mask, otherwise mask is present

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – keep high precision here
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left and right streams.
    #    Packing weights reduces the number of matmuls from 4 → 2.
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]      # (hidden_dim, dim)
    left_gate_w  = weights["left_gate.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]       # (hidden_dim, dim)

    # concatenate projection & gate weights (bias=False → simple cat)
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16) # (2*H, dim)

    # single linear → [B,N,N,2*H]
    left_comb   = torch.nn.functional.linear(x, left_comb_w)        # fp16
    right_comb  = torch.nn.functional.linear(x, right_comb_w)       # fp16

    # split back to projection & gate streams
    left  = left_comb[..., :hidden_dim]                # [B,N,N,H] (proj)
    left_gate_pre = left_comb[..., hidden_dim:]        # raw gate pre‑sigmoid
    left_gate = torch.sigmoid(left_gate_pre)           # [B,N,N,H] (gate)

    right  = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate (independent linear)
    # ------------------------------------------------------------------
    out_gate_w = out_gate_w.to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N,H] fp16
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 5️⃣ If a mask is present, fuse mask * gate * projection via Triton.
    #    The kernel updates ``left`` and ``right`` in‑place.
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()                      # B·N·N·H
        # Choose a block size that is a multiple of 32 (warp size)
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        mask_gate_fuse_two_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication.
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)          # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Gating with out_gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
