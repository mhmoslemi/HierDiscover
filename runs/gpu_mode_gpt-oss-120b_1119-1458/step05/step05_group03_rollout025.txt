def proj_gate_mask_kernel(
    x_ptr,
    w_proj_ptr,
    w_gate_ptr,
    mask_ptr,
    out_ptr,
    M, K, H,
    stride_xm, stride_xk,
    stride_wh, stride_wk, # but weight shape (H, K): strid0 = K (row), strid1 = 1 (col)
    stride_outm, stride_outh,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # grid: (num_pid_m * num_pid_n)
    pid = tl.program_id(0) # grid is 1D
    # Compute pid_m and pid_n
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(H, BLOCK_H)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n
    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_H + tl.arange(0, BLOCK_H)
    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < H
    # Initialize accumulators
    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Compute mask value per row (input)
    if mask_ptr is not None:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0) # default 1.0
        mask_row = mask_val[:, None] # shape (BLOCK_M,1)
    else:
        mask_row = tl.full((BLOCK_M,1), 1.0, dtype=tl.float32) # not needed? Actually mask unconditional.

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load A tile
        a = tl.load(
            x_ptr + (offs_m[:,None] * stride_xm + (k_offset + tl.arange(0, BLOCK_K))[None,:] * stride_xk),
            mask=mask_m[:, None] & ((k_offset + tl.arange(0,BLOCK_K)) < K),
            other=0.0
        ) # shape (BLOCK_M,BLOCK_K)
        # load B_proj tile
        w_proj = tl.load(
            w_proj_ptr + ((k_offset + tl.arange(0,BLOCK_K))[:,None] + offs_n[None,:] * stride_wh),
            mask=((k_offset + tl.arange(0,BLOCK_K)) < K)[:,None] & mask_n[None,:],
            other=0.0
        )
        w_gate = tl.load(
            w_gate_ptr + ((k_offset + tl.arange(0,BLOCK_K))[:,None] + offs_n[None,:] * stride_wh),
            mask=((k_offset + tl.arange(0,BLOCK_K)) < K)[:,None] & mask_n[None,:],

    hidden_dim = config['hidden_dim']

    device = input_tensor.device

        bias=weights['norm.bias']

    K = dim
    H = hidden_dim

    left_gate_w = left_gate_w.contiguous()

@triton.jit
def left_proj_gate_out_gate_kernel(
    x_ptr, left_proj_ptr, left_gate_ptr,
    out_gate_weight_ptr,
    mask_ptr,
    left_out_ptr, out_gate_out_ptr,
    M, K, H,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
    USE_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute grid dims
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_h = tl.cdiv(H, BLOCK_H)
    pid_m = pid // num_pid_h
    pid_h = pid % num_pid_h

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # accumulators
    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    K_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(K_tiles):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        valid_k = k_range < K

        # Load X tile
        # off = offs_m[:,None] * K + k_range[None,:]  (since stride_xm=K, stride_xk=1)
        a = tl.load(x_ptr + (offs_m[:, None] * K + k_range[None, :]), mask=mask_m[:, None] & valid_k[None, :], other=0.0)   # shape (BLOCK_M, BLOCK_K)
        # Load weight tiles
        w_left_proj = tl.load(left_proj_ptr + (k_range[:, None] + offs_h[None,:] * K), mask=valid_k[:, None] & mask_h[None,:], other=0.0)
        w_left_gate = tl.load(left_gate_ptr + (k_range[:, None] + offs_h[None,:] * K), mask=valid_k[:, None] & mask_h[None,:], other=0.0)
        w_out_gate = tl.load(out_gate_weight_ptr + (k_range[:, None] + offs_h[None,:] * K), mask=valid_k[:, None] & mask_h[None,:], other=0.0)
        # Accumulate
        acc_proj += tl.dot(a, w_left_proj)
        acc_gate += tl.dot(a, w_left_gate)
        acc_out_gate += tl.dot(a, w_out_gate)
    # Compute left output
    left_gate_sig = 1.0 / (1.0 + tl.exp(-acc_gate))
    left_val = acc_proj * left_gate_sig

    if USE_MASK:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)   # fp16 mask
        # cast mask to fp32
        mask_fp = mask_val.to(tl.float32)
        left_val = left_val * mask_fp[:, None]

    # Store left output
    left_ptrs = left_out_ptr + (offs_m[:, None] * H + offs_h[None, :])
    tl.store(left_ptrs, left_val.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])

    # Compute out_gate output
    out_gate_sig = 1.0 / (1.0 + tl.exp(-acc_out_gate))
    out_gate_val = out_gate_sig
    out_gate_ptrs = out_gate_out_ptr + (offs_m[:, None] * H + offs_h[None, :])
    tl.store(out_gate_ptrs, out_gate_val.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])
```

def tri_mul_fwd_kernel(
    x_ptr,  # (M,K) input
    left_proj_ptr,
    left_gate_ptr,
    right_proj_ptr,
    right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    left_out_ptr,
    right_out_ptr,
    out_gate_out_ptr,
    M, K, H,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
    USE_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_h = tl.cdiv(H, BLOCK_H)
    pid_m = pid // num_pid_h
    pid_h = pid % num_pid_h

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_m = offs_m < M
    mask_h = offs_h < H

    # accumulators
    a_proj_left = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    a_gate_left = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    a_proj_right = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    a_gate_right = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    a_out_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for k in range(tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        valid_k = k_range < K

        # input tile
        a = tl.load(x_ptr + (offs_m[:, None] * K + k_range[None, :]), mask=mask_m[:, None] & valid_k[None, :], other=0.0)

        # weight tiles
        w_left_proj = tl.load(left_proj_ptr + (k_range[:, None] + offs_h[None,:] * K), mask=valid_k[:, None] & mask_h[None,:], other=0.0)
        w_left_gate = tl.load(left_gate_ptr + (k_range[:, None] + offs_h[None,:] * K), mask=valid_k[:, None] & mask_h[None,:], other=0.0)
        w_right_proj = tl.load(right_proj_ptr + (k_range[:, None] + offs_h[None,:] * K), mask=valid_k[:, None] & mask_h[None,:], other=0.0)
        w_right_gate = tl.load(right_gate_ptr + (k_range[:, None] + offs_h[None,:] * K), mask=valid_k[:, None] & mask_h[None,:], other=0.0)
        w_out_gate = tl.load(out_gate_ptr + (k_range[:, None] + offs_h[None,:] * K), mask=valid_k[:, None] & mask_h[None,:], other=0.0)

        # accum dot
        a_proj_left += tl.dot(a, w_left_proj)
        a_gate_left += tl.dot(a, w_left_gate)
        a_proj_right += tl.dot(a, w_right_proj)
        a_gate_right += tl.dot(a, w_right_gate)
        a_out_gate += tl.dot(a, w_out_gate)

    # compute gates
    left_gate_sig = 1.0 / (1.0 + tl.exp(-a_gate_left))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-a_gate_right))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-a_out_gate))

    left = a_proj_left * left_gate_sig
    right = a_proj_right * right_gate_sig

    if USE_MASK:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)
        mask_fp = mask_val.to(tl.float32)
        left = left * mask_fp[:, None]
        right = right * mask_fp[:, None]

    # store left
    left_ptrs = left_out_ptr + (offs_m[:, None] * H + offs_h[None,:])
    tl.store(left_ptrs, left.to(tl.float16), mask=mask_m[:, None] & mask_h[None,:])

    # store right

    tl.store(right_ptrs, right.to(tl.float16), mask=mask_m[:, None] & mask_h[None,:])

    # store out_gate
    out_gate_ptrs = out_gate_out_ptr + (offs_m[:, None] * H + offs_h[None,:])
    tl.store(out_gate_ptrs, out_gate_sig.to(tl.float16), mask=mask_m[:, None] & mask_h[None,:])

"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑Norm the input tensor (FP32) → cast to FP16.
2. A single Triton kernel fuses **all three linear projections** (left, right,
   out‑gate) together with the two gating MLPs and optional mask broadcasting.
   The kernel computes for each position (b,i,k):
        left    = (x·W_left)    * sigmoid(x·W_left_gate)    * mask
        right   = (x·W_right)   * sigmoid(x·W_right_gate)   * mask
        out_gate= sigmoid(x·W_out_gate)
   This removes five separate matmul‑gate‑mask passes and avoids extra memory
   copies.
3. Contract the two hidden tensors with an efficient batched GEMM via
   `torch.einsum` (equivalent to per‑head `A @ Bᵀ`).  No further copies or
   permutes are required.
4. Second Layer‑Norm on the contracted tensor, out‑gate multiplication and
   final linear projection back to the original channel dimension.
5. Cast the result back to FP32 (the harness expects FP32).

Only the forward pass is required; no gradients are computed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse left‑projection, left‑gate, right‑projection,
# right‑gate, out‑gate, and (optional) mask multiplication.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_fwd_kernel(
    # pointers
    x_ptr,                     # *fp16 : input flattened [M, K]
    left_proj_ptr,             # *fp16 : weight [H, K] (left projection)
    left_gate_ptr,             # *fp16 : weight [H, K] (left gate)
    right_proj_ptr,            # *fp16 : weight [H, K] (right projection)
    right_gate_ptr,            # *fp16 : weight [H, K] (right gate)
    out_gate_ptr,              # *fp16 : weight [H, K] (out gate)
    mask_ptr,                  # *fp16 : mask [M] (broadcasted), may be dummy
    left_out_ptr,              # *fp16 : output left   [M, H]
    right_out_ptr,             # *fp16 : output right  [M, H]
    out_gate_out_ptr,          # *fp16 : output out‑gate [M, H]
    # sizes
    M,                         # i32 : number of rows = B·N·N
    K,                         # i32 : input channel dim
    H,                         # i32 : hidden dimension
    # compile‑time constants
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_H: tl.constexpr,     # hidden columns per program
    BLOCK_K: tl.constexpr,     # K‑tile size
    USE_MASK: tl.constexpr,    # 0 → ignore mask, 1 → apply mask
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 2‑D grid: (row‑tile, hidden‑tile)
    # --------------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_h = tl.cdiv(H, BLOCK_H)
    pid_m = pid // num_pid_h          # row tile index
    pid_h = pid %  num_pid_h          # hidden tile index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Accumulators (FP32 for accuracy)
    # ------------------------------------------------------------------
    acc_left_proj  = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_left_gate  = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_right_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_out_gate   = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Tile‑wise multiplication over K
    # ------------------------------------------------------------------
    for k_tile in range(tl.cdiv(K, BLOCK_K)):
        k_off = k_tile * BLOCK_K
        k_range = k_off + tl.arange(0, BLOCK_K)
        valid_k = k_range < K

        # Load a block of X  [BLOCK_M, BLOCK_K]
        x = tl.load(
            x_ptr + (offs_m[:, None] * K + k_range[None, :]),
            mask=mask_m[:, None] & valid_k[None, :],
            other=0.0,
        )  # fp16 -> will be up‑cast in dot

        # Load weight tiles  [BLOCK_K, BLOCK_H]  (weights are stored as [H,K])
        w_lp = tl.load(
            left_proj_ptr + (k_range[:, None] + offs_h[None, :] * K),
            mask=valid_k[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_lg = tl.load(
            left_gate_ptr + (k_range[:, None] + offs_h[None, :] * K),
            mask=valid_k[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_rp = tl.load(
            right_proj_ptr + (k_range[:, None] + offs_h[None, :] * K),
            mask=valid_k[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_rg = tl.load(
            right_gate_ptr + (k_range[:, None] + offs_h[None, :] * K),
            mask=valid_k[:, None] & mask_h[None, :],
            other=0.0,
        )
        w_og = tl.load(
            out_gate_ptr + (k_range[:, None] + offs_h[None, :] * K),
            mask=valid_k[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Accumulate the GEMM products
        acc_left_proj  += tl.dot(x, w_lp)   # (Mblk, Hblk)
        acc_left_gate  += tl.dot(x, w_lg)
        acc_right_proj += tl.dot(x, w_rp)
        acc_right_gate += tl.dot(x, w_rg)
        acc_out_gate   += tl.dot(x, w_og)

    # ------------------------------------------------------------------
    # Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate_sig  = 1.0 / (1.0 + tl.exp(-acc_left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-acc_right_gate))
    out_gate_sig   = 1.0 / (1.0 + tl.exp(-acc_out_gate))

    left  = acc_left_proj  * left_gate_sig
    right = acc_right_proj * right_gate_sig

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if USE_MASK:
        m = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)   # fp16
        m_f32 = m.to(tl.float32)                               # broadcast
        left  = left * m_f32[:, None]
        right = right * m_f32[:, None]

    # ------------------------------------------------------------------
    # Store results (back to FP16)
    # ------------------------------------------------------------------
    out_left_ptr  = left_out_ptr  + (offs_m[:, None] * H + offs_h[None, :])
    out_right_ptr = right_out_ptr + (offs_m[:, None] * H + offs_h[None, :])
    out_gate_ptr  = out_gate_out_ptr + (offs_m[:, None] * H + offs_h[None, :])

    tl.store(out_left_ptr,  left .to(tl.float16),  mask=mask_m[:, None] & mask_h[None, :])
    tl.store(out_right_ptr, right.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr,  out_gate_sig.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry‑point called by the harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional key ``nomask`` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    M = B * N * N                      # total number of (b,i,k) positions
    K = dim
    H = hidden_dim

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (FP32) → FP16
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_norm.to(torch.float16)                     # [B,N,N,dim] FP16
    x_flat = x.reshape(M, K).contiguous()           # [M, K]

    # --------------------------------------------------------------
    # 3️⃣ Cast all required model weights to FP16 (contiguous)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16).contiguous()
    left_gate_w   = weights["left_gate.weight"].to(torch.float16).contiguous()
    right_proj_w  = weights["right_proj.weight"].to(torch.float16).contiguous()
    right_gate_w  = weights["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w    = weights["out_gate.weight"].to(torch.float16).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).contiguous()
    to_out_w      = weights["to_out.weight"].to(torch.float16).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (flatten & FP16)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.to(torch.float16).reshape(M).contiguous()
    else:
        # dummy tensor – will never be read because USE_MASK=0
        mask_flat = torch.empty((0,), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 5️⃣ Allocate output buffers for left, right and out‑gate
    # --------------------------------------------------------------
    left_out      = torch.empty((M, H), dtype=torch.float16, device=device)
    right_out     = torch.empty((M, H), dtype=torch.float16, device=device)
    out_gate_out  = torch.empty((M, H), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 6️⃣ Triton kernel launch – fused projections, gates and mask
    # --------------------------------------------------------------
    # Tunable block sizes – chosen to keep register pressure modest.
    BLOCK_M = 64
    BLOCK_H = 64
    BLOCK_K = 32
    num_pid_m = (M + BLOCK_M - 1) // BLOCK_M
    num_pid_h = (H + BLOCK_H - 1) // BLOCK_H
    grid = (num_pid_m * num_pid_h,)

    tri_mul_fwd_kernel[grid](
        x_flat,
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
        mask_flat,
        left_out,
        right_out,
        out_gate_out,
        M,
        K,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        USE_MASK=int(not nomask),
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to 4‑D tensors
    # --------------------------------------------------------------
    left      = left_out.view(B, N, N, H)          # [B, N, N, H]
    right     = right_out.view(B, N, N, H)         # [B, N, N, H]
    out_gate  = out_gate_out.view(B, N, N, H)      # [B, N, N, H]

    # --------------------------------------------------------------
    # 8️⃣ Heavy contraction  (einsum ≡ batched GEMM)
    # --------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm on the contracted tensor (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # still FP16

    # --------------------------------------------------------------
    # 🔟 Gating with out_gate (produced by the kernel) and final projection
    # --------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating, FP16
    out = torch.nn.functional.linear(out, to_out_w)   # back to `dim`, FP16

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to FP32 for the harness and return
    # --------------------------------------------------------------
    return out.float()
