"""
Outgoing TriMul kernel (AlphaFold‑3) – fused, mixed‑precision implementation.

Steps
-----
1️⃣  Layer‑norm the 4‑D input (FP32) and cast to FP16.
2️⃣  Stack the two projection matrices (left/right) and the three gating
    matrices (left_gate, right_gate, out_gate) so that each pair is computed
    with a single `torch.nn.functional.linear` call.  This reduces kernel
    launch overhead and improves memory locality.
3️⃣  Optional mask + gating fusion is performed by a tiny Triton kernel.
    The kernel multiplies `left`, `right` with their respective gates and
    applies the binary mask (0/1) in‑place, cutting two extra memory passes.
4️⃣  Core O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    which is a batched matrix multiplication C = A @ Bᵀ with
    A,B ∈ ℝ^{(B·H)×N×N}.  We flatten the leading dimensions,
    run `torch.bmm` (Tensor‑core‑accelerated FP16 GEMM) and reshape back.
5️⃣  Second LayerNorm (FP16) over the hidden channel, element‑wise multiplication
    with the out‑gate and a final projection back to `dim`.
6️⃣  Cast the result back to FP32 (the harness expects float32).

Only the mask‑gate fusion is performed with Triton; the heavy GEMM is
delegated to cuBLAS via `torch.bmm`, which gives the best performance on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * left_gate  * mask
#   right ← right * right_gate * mask
# All tensors are contiguous FP16 with shape [B,N,N,H] (left/right/gates)
# and mask has shape [B,N,N] (broadcast along H).
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i32   total number of elements in left/right (B·N·N·H)
    H,                 # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    # Flat indices handled by this program instance
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # each hidden element shares the same mask entry -> divide by H
    mask_offs = offs // H

    # Load tensors (masked loads guarantee 0 for OOB)
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)  # 1.0 keeps value unchanged if OOB

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store results back (in‑place)
    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform C = A @ Bᵀ where A and B are [B, N, N, H] (FP16).
    Returns tensor of shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # (B, H, N, N) – non‑contiguous view, then flatten batch & heads
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B·H, N, N)

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))

    # Restore original layout (B, N, N, H)
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32 (same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32
    x = x.to(dtype_fp16)  # FP16 for the heavy part

    # ------------------------------------------------------------------
    # 2️⃣ Cast all model weights to FP16 on the same device
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Stacked linear projections:
    #     - left/right (2 × hidden_dim)
    #     - left_gate, right_gate, out_gate (3 × hidden_dim)
    # ------------------------------------------------------------------
    # (B,N,N,2*H)
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)   # (2*H, dim)
    proj = F.linear(x, proj_weight)
    left  = proj[..., :hidden_dim]           # (B,N,N,H)
    right = proj[..., hidden_dim:]           # (B,N,N,H)

    # (B,N,N,3*H) → split into gate tensors
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*H, dim)
    gates = F.linear(x, gate_weight)
    left_gate  = torch.sigmoid(gates[..., :hidden_dim])
    right_gate = torch.sigmoid(gates[..., hidden_dim:2*hidden_dim])
    out_gate   = torch.sigmoid(gates[..., 2*hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if nomask:
        # No mask: only gate multiplication
        left  = left * left_gate
        right = right * right_gate
    else:
        # Ensure contiguous layout for the kernel
        left        = left.contiguous()
        right       = right.contiguous()
        left_gate   = left_gate.contiguous()
        right_gate  = right_gate.contiguous()
        mask_fp16   = mask_tensor.to(dtype_fp16).contiguous()  # [B,N,N]

        total = left.numel()
        BLOCK = 1024  # ~1K threads per block – good occupancy
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release intermediate tensors early
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched FP16 GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dim (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                        # element‑wise gating (FP16)
    out = F.linear(out, to_out_w)               # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast to float32 (expected by the test harness) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
