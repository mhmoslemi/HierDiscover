def triton_batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH_HEAD,
    N,
    HIDDEN_DIM, // Actually we combine hidden dimension into batch each head already
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # grid: (num_i_blocks, num_j_blocks, batch_head)
    batch_head_idx = tl.program_id(2)
    i_block = tl.program_id(0)
    j_block = tl.program_id(1)
    # Compute offsets
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # Ensure bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # compute base pointer offsets for this batch_head
    a_base = a_ptr + batch_head_idx * stride_a_batch
    b_base = b_ptr + batch_head_idx * stride_b_batch
    c_base = c_ptr + batch_head_idx * stride_c_batch

    # load A block (M x K)
    a_ptrs = a_base + i_offset[:, None] * stride_a_i + k_offset[None, :] * stride_a_k
    a = tl.load(a_ptrs, mask=i_mask[:, None] & (k_offset[None, :] < N), other=0.0).to(tl.float32)

    # load B_T block (K x N)
    b_ptrs = b_base + k_offset[:, None] * stride_b_k + j_offset[None, :] * stride_b_i
    b = tl.load(b_ptrs, mask=(k_offset[:, None] < N) & j_mask[None, :], other=0.0).to(tl.float32)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        # same as above with proper offsets
        ...
        acc += tl.dot(a_block, b_block) ...
    # after accumulate
    # cast to fp16 (maybe store fp16 for later steps)
    c_ptrs = c_base + i_offset[:, None] * stride_c_i + j_offset[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

```
@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH_HEAD, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_head = tl.program_id(2)  # BH id
    i_tile = tl.program_id(0)  # tile in M dimension (i)
    j_tile = tl.program_id(1)  # tile in N dimension (j)

    # Offsets
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute masks for boundaries
    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointers

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    device = input_tensor.device

    # 1) first layernorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x_norm.half()

    # Cast weight to half
    left_proj_weight = weights["left_proj.weight"].half()
    left_gate_weight = weights["left_gate.weight"].half()
    right_proj_weight = weights["right_proj.weight"].half()
    right_gate_weight = weights["right_gate.weight"].half()
    out_gate_weight = weights["out_gate.weight"].half()
    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias = weights["to_out_norm.bias"].half()
    to_out_weight = weights["to_out.weight"].half()

    # 2) compute concatenated left and right linear outputs
    left_cat_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # [2*hidden_dim, dim]
    right_cat_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)

    left_cat = torch.nn.functional.linear(x, left_cat_weight)   # [B,N,N,2H] fp16
    right_cat = torch.nn.functional.linear(x, right_cat_weight)

    # split
    left_proj = left_cat[..., :hidden_dim]
    left_gate_pre = left_cat[..., hidden_dim:]
    right_proj = right_cat[..., :hidden_dim]
    right_gate_pre = right_cat[..., hidden_dim:]

    # gate activations
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # mask handling
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
    else:
        mask_fp16 = None

    # apply gate and mask in place
    if mask_fp16 is None:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16

    # 3) out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # 4) heavy contraction using custom Triton kernel
    out = _batched_matmul_triton(left, right)   # [B,N,N,hidden_dim] fp16

    # 5) second layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # 6) apply out_gate
    out = out * out_gate

    # 7) final linear projection to original dim
    out = torch.nn.functional.linear(out, to_out_weight)

    # 8) cast to float32
    return out.float()

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Kernel for batched matrix multiplication (C = A @ B^T)
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bi, stride_bk,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    batch_head = tl.program_id(2)          # which batch+head
    tile_i = tl.program_id(0)              # block row index
    tile_j = tl.program_id(1)              # block col index

    # Offsets for block
    i_offset = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Base pointers for this batch head
    a_base = a_ptr + batch_head * stride_ah
    b_base = b_ptr + batch_head * stride_bh
    c_base = c_ptr + batch_head * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load block of A (size M x K)
        a_ptrs = a_base \
                 + i_offset[:, None] * stride_ai \
                 + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load block of B transposed (size K x N)
        # B is stored as [batch_head, row=j, col=k], we need B_T[k, j]
        b_ptrs = b_base \
                 + k_offset[:, None] * stride_bk \
                 + j_offset[None, :] * stride_bi
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)  # (M,K) x (K,N) -> (M,N)

    # Store result in FP16
    c_ptrs = c_base \
             + i_offset[:, None] * stride_ci \
             + j_offset[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

Now define wrapper:

```
def batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: tensors of shape [B, N, N, H] dtype half, contiguous.
    Returns out of same shape.
    """
    B, N, _, H = left.shape
    # permute to [B, H, N, N] for easier view
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    # view as [BH, N, N]
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)  # same dtype FP16

    # strides (per element, in units of the tensor dtype)
    stride_ah = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    # right strides
    stride_bh = right_flat.stride(0)
    stride_bi = right_flat.stride(1)
    stride_bk = right_flat.stride(2)

    # output strides
    stride_ch = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bi, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣ Fuse the two Linear + gate streams per side:
   left = Linear(x, [proj;gate])   →  split to proj and gate,
   right = Linear(x, [proj;gate])  →  split to proj and gate.
   Apply sigmoid to the gate half and multiply *proj*·*gate*.
   If a mask is provided (shape [B,N,N]), broadcast it over the hidden
   dimension and fuse it with the multiplication (∏proj·gate·mask).
3️⃣ Heavy O(N³) contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a custom Triton batched GEMM: for each head (h) and batch
   we compute C = A @ Bᵀ where A,B ∈ ℝ^{N×N}_fp16 using Tensor‑Cores.
   The kernel loops over K in 32‑element tiles, accumulates in FP32 and
   stores the result in FP16.
4️⃣ Second LayerNorm (FP16) over the hidden dimension.
5️⃣ Multiply by the out‑gate (computed in step 2) and apply the final
   Linear projection back to the original channel dimension.
6️⃣ Cast the result to FP32 (the harness expects FP32).

At least one Triton kernel is used (the batched GEMM).  All other
operations stay in Torch but are heavily fused (concatenated Linear
weights, combined gating, in‑place masking) to reduce memory traffic.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
#  Triton kernel: batched GEMM computing C = A @ Bᵀ for many (batch*head)
#  matrices of size N×N (A,B ∈ ℝ^{N×N}_fp16).
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                    # *fp16 : left  tensor, shape (BH, N, N)
    b_ptr,                    # *fp16 : right tensor, shape (BH, N, N)
    c_ptr,                    # *fp16 : output tensor, shape (BH, N, N)
    N,                        # i32   : sequence length
    stride_ah, stride_ai, stride_ak,   # i32 : strides of A (batch‑head, i, k)
    stride_bh, stride_bi, stride_bk,   # i32 : strides of B (batch‑head, i, k)
    stride_ch, stride_ci, stride_cj,   # i32 : strides of C (batch‑head, i, j)
    BLOCK_M: tl.constexpr,    # compile‑time tile size along i
    BLOCK_N: tl.constexpr,    # compile‑time tile size along j
    BLOCK_K: tl.constexpr,    # compile‑time tile size along k
):
    # --------------------------------------------------------------
    #  Program IDs – 3‑D grid: (tile_i, tile_j, batch_head)
    # --------------------------------------------------------------
    bh_id  = tl.program_id(2)        # which (batch * head) piece
    tile_i = tl.program_id(0)        # block row (i)
    tile_j = tl.program_id(1)        # block column (j)

    # --------------------------------------------------------------
    #  Offsets for the current tile
    # --------------------------------------------------------------
    i_off = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i_off < N
    mask_j = j_off < N

    # Base pointers for this batch‑head
    a_base = a_ptr + bh_id * stride_ah
    b_base = b_ptr + bh_id * stride_bh
    c_base = c_ptr + bh_id * stride_ch

    # Accumulator in FP32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over the reduction dimension k (tiles of size BLOCK_K)
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # ---- Load A tile (M×K) ------------------------------------------------
        a_ptrs = a_base \
                 + i_off[:, None] * stride_ai \
                 + k_off[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- Load Bᵀ tile (K×N) ----------------------------------------------
        # B is stored as [..., j, k]; we need its transpose (k, j)
        b_ptrs = b_base \
                 + k_off[:, None] * stride_bk \
                 + j_off[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ---- Accumulate dot product -------------------------------------------
        acc += tl.dot(a, b)      # (M×K)·(K×N) → (M×N)

    # --------------------------------------------------------------
    #  Store C tile back (cast to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_base \
             + i_off[:, None] * stride_ci \
             + j_off[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_triton(left: torch.Tensor,
                           right: torch.Tensor,
                           block_m: int = 32,
                           block_n: int = 32,
                           block_k: int = 32) -> torch.Tensor:
    """
    Invoke `batched_matmul_kernel` on tensors of shape [B, N, N, H] (FP16).
    Returns a tensor of the same shape (FP16).
    """
    B, N, _, H = left.shape

    # (B, H, N, N) → view as (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)     # FP16 output

    # Compute strides (in elements, not bytes)
    stride_ah = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bi = right_flat.stride(1)
    stride_bk = right_flat.stride(2)

    stride_ch = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Grid configuration
    grid = (triton.cdiv(N, block_m),
            triton.cdiv(N, block_n),
            BH)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bi, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,               # good default for fp16 Tensor‑Cores
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
#  Triton helper: optional fused mask‑gate multiply (keeps the kernel count low)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # *fp16 : projection tensor [size]
    gate_ptr,          # *fp16 : gate tensor   [size] (already after sigmoid)
    mask_ptr,          # *fp16 : mask tensor   [B*N*N] – broadcasted
    size,              # i32   : total number of elements = B·N·N·H
    hidden_dim,        # i32   : hidden dimension (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < size

    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)

    # each `hidden_dim` consecutive entries share one mask value
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    out = proj * gate * m
    tl.store(proj_ptr + offs, out, mask=active)   # store in‑place


# ----------------------------------------------------------------------
#  Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      shape [B, N, N]   (float32/bool) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional `nomask` (True → ignore mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]          # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask     = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                           # [B, N, N, dim]  fp32
    x = x_norm.half()                           # cast once to fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (one‑time)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ Fused Linear → [proj ; gate] for left and right sides
    # ------------------------------------------------------------------
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, dim)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    left_cat  = torch.nn.functional.linear(x, left_cat_w)      # [B,N,N,2H] fp16
    right_cat = torch.nn.functional.linear(x, right_cat_w)     # [B,N,N,2H] fp16

    # Split projection and gate halves
    left_proj = left_cat[..., :hidden_dim]
    left_gate = left_cat[..., hidden_dim:]                     # pre‑sigmoid

    right_proj = right_cat[..., :hidden_dim]
    right_gate = right_cat[..., hidden_dim:]                  # pre‑sigmoid

    # Apply sigmoid on the gate halves
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – broadcasted over the hidden dimension.
    # ------------------------------------------------------------------
    if not nomask:
        # mask is shape [B, N, N]; broadcast to [B, N, N, 1]
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
    else:
        mask_fp16 = None

    # Element‑wise fusion: proj * gate * mask (in‑place to keep memory low)
    if mask_fp16 is None:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16

    # Ensure contiguous layout for the Triton GEMM
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Out‑gate (computed from the same normalized input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction – custom batched GEMM (fp16 inputs → fp32 accum)
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)   # back to original dim (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
