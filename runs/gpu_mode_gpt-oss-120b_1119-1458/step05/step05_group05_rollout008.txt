@triton.jit
def mask_proj_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr,
    out_ptr,
    hidden_dim,
    size,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # mask index for each element
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # compute sigmoid
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * m * gate_sig

    tl.store(out_ptr + offs, out, mask=mask)
```

"""
custom_kernel – highly‑optimised TriMul (outgoing) forward pass.

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32) → cast to fp16.
2.  Fuse the five linear layers (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a *single* matmul by concatenating the
    weights.  This reduces the number of cuBLAS calls from 5 to 1.
3.  Split the concatenated output into the five tensors.
4.  Fuse mask multiplication, sigmoid‑gate, and element‑wise product for
    *left* and *right* tensors with a tiny Triton kernel
    ``mask_proj_gate_kernel``.  The kernel works for both masked and
    un‑masked inputs (mask of ones yields the un‑masked path).
5.  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM (A @ Bᵀ) using a single ``torch.bmm`` on the
    reshaped tensors – this is already heavily tuned on H100.
6.  Second Layer‑norm (hidden_dim) → element‑wise gating with ``out_gate``.
7.  Final linear projection back to the original channel dimension.
8.  Cast the result to float32 (the harness expects float32).

Only the mask‑gate fusion is performed in Triton, satisfying the
“use at least one Triton kernel” requirement while the rest of the
pipeline stays in high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask * sigmoid(gate) * proj
# ----------------------------------------------------------------------
@triton.jit
def mask_proj_gate_kernel(
    proj_ptr,          # *fp16  → projection tensor (flattened)
    gate_ptr,          # *fp16  → gate tensor (same flattening as proj)
    mask_ptr,          # *fp16  → mask tensor flattened, shape [B*N*N]
    out_ptr,           # *fp16  → write‑back pointer (can be same as proj_ptr)
    hidden_dim,        # i32   → size of the hidden dimension (H)
    size,              # i32   → total number of elements in proj (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Map each element to its mask entry (mask has no hidden dimension)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Sigmoid in‑kernel (stable enough for fp16)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # Fused multiply: proj * mask * sigmoid(gate)
    out = proj * m * gate_sig

    # Write back (in‑place if out_ptr == proj_ptr)
    tl.store(out_ptr + offs, out, mask=valid)

# ----------------------------------------------------------------------
# Helper – batched GEMM (A @ Bᵀ) with fp16 tensors using cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left, right : fp16 tensors of shape [B, N, N, H] (contiguous)

    Returns
    -------
    out : fp16 tensor of shape [B, N, N, H]
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and use torch.bmm
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # BMM with Bᵀ (i.e. a @ bᵀ)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32, on target device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → stable statistics)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x = x.contiguous().to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse ALL projection & gating linear layers
    # -----------------------------------------------------------------
    # Weight order: left_proj, right_proj, left_gate, right_gate, out_gate
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)               # shape [5*H, dim]

    # Single large linear projection
    proj = torch.nn.functional.linear(x, fused_weight)   # [B, N, N, 5*H]

    # -----------------------------------------------------------------
    # 4️⃣ Split concatenated tensor
    # -----------------------------------------------------------------
    proj_chunks = torch.split(proj, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate, right_gate, out_gate = map(
        lambda t: t.contiguous(), proj_chunks

    # -----------------------------------------------------------------
    # 5️⃣ Mask & gate fusion via Triton (runs even if mask is all ones)
    # -----------------------------------------------------------------
    mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    mask_flat = mask_fp16.view(-1)                     # [B*N*N]

    size = left_proj.numel()                           # B*N*N*H
    BLOCK = 1024                                        # ≤ 1024 threads per block
    grid = (triton.cdiv(size, BLOCK),)

    # left = left_proj * mask * sigmoid(left_gate)
    mask_proj_gate_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        left_proj,                     # in‑place write‑back
        hidden_dim,
        size,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right = right_proj * mask * sigmoid(right_gate)

        right_proj,
        right_gate,
        mask_flat,
        right_proj,
        hidden_dim,
        size,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Compute sigmoid(out_gate) (no mask needed)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction (batched GEMM)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left_proj, right_proj)   # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection
    # -----------------------------------------------------------------
    out = out * out_gate                               # element‑wise gating

    to_out_weight = weights["to_out.weight"].to(device, torch.float16)  # shape [dim, H]
    out = torch.nn.functional.linear(out, to_out_weight)               # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
