    Triton kernel:   X = X * GATE * MASK   (in‑place, FP16).
5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    realised as a batched GEMM C = A @ Bᵀ with torch.bmm (cuBLAS) on the
    H100 – this is already optimal for the hardware.
6️⃣  Second Layer‑norm (FP16) → apply the out‑gate → final linear projection
    back to the original feature dimension.
7️⃣  Cast the result to FP32 for the external harness.

Only the forward pass is implemented; gradients are not required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,              # *fp16 : gate tensor, same shape as x
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate
    hidden_dim: tl.int32,  # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_elts

    # Compute the corresponding mask index (mask repeats each hidden_dim)
    mask_idx = offs // hidden_dim

    x_val   = tl.load(x_ptr + offs,   mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x_val * gate_val * mask_val, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM via torch.bmm (cuBLAS).  Core O(N³) step.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (FP16).

    Arguments
    ---------
    left, right : torch.Tensor
        [B, N, N, H] (FP16)

    Returns
    -------
    out : torch.Tensor
        [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel is a separate batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ  → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) – ignored if config["nomask"] is True
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x_norm.to(torch.float16)  # switch to fp16 for the heavy part

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single matmul
    # ------------------------------------------------------------------
    # Cast all projection/gate weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Stack them: order = [left_proj, right_proj, left_gate, right_gate, out_gate]
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Single linear call → result shape [B, N, N, 5*hidden_dim] (fp16)
    combined = torch.nn.functional.linear(x, combined_weight)  # bias=False

    # ------------------------------------------------------------------
    # 3️⃣ Split & apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    H = hidden_dim
    left          = combined[..., 0:H]            # [B, N, N, H]
    right         = combined[..., H:2*H]          # [B, N, N, H]
    left_gate_raw = combined[..., 2*H:3*H]        # [B, N, N, H]
    right_gate_raw = combined[..., 3*H:4*H]       # [B, N, N, H]
    out_gate_raw  = combined[..., 4*H:5*H]        # [B, N, N, H]

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Cast mask to fp16 and ensure it is contiguous & flattened
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()               # B * N * N * H
        BLOCK = 1024                             # ≤ 1024 threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: simple element‑wise multiplication (tiny cost)
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) → out‑gate gating → final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Apply out‑gate (broadcasted across hidden dimension)
    out = out * out_gate

    # Final projection back to original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # -> [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (required by testing harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
