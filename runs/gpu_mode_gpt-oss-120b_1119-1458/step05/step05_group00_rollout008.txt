def _mask_gate_fusion_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements,
    H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Compute broadcast over hidden dim
    mask_offset = offs // H  # each hidden slice repeats each mask for H values

    # Load values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=1.0)

    right_out = right * right_gate * mask_val

    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul forward (AlphaFold‑3) – optimized Triton implementation
    ---------------------------------------------------------------
    1. Layer‑norm the 4‑D pairwise tensor (FP32) → cast to FP16.
    2. Fuse the four linear projections (left/right + their gates) into
       a single FP16 GEMM (`left|right|gate` weight = 4·hidden x dim),
       then split the result into left_proj, left_gate, right_proj,
       right_gate.  Apply a sigmoid to the gate slices.
    3. Compute a third linear projection for the output‑gate.
    4. If a mask is provided, launch a tiny Triton kernel that
       fuses *mask* × *gate* × *proj* for both left and right streams,
       overwriting the projection tensors in‑place.
       If ``nomask`` is True the gating is performed with a simple
       element‑wise multiplication.
    5. Heavy contraction:

    7. Cast the result back to FP32 (the harness expects FP32).

    # unpack
    # ------------------------------------------------------------------

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # LayerNorm expects normalized_shape=(dim,)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,

    # Cast to FP16 for all subsequent heavy work
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projections + gates into a single GEMM
    # ------------------------------------------------------------------
    # Convert weight tensors to FP16 on the same device as inputs
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Concatenate the four projection / gate weight matrices
    # each has shape (hidden_dim, dim)
    # The concatenated weight has shape (4*hidden_dim, dim)
    fused_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
        ],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)

    # One linear pass => shape [B, N, N, 4*hidden_dim] (FP16)
    fused = F.linear(x, fused_weight)  # FP16

    # Split the fused tensor
    # Note: slicing on the last dimension yields views (no copy)
    h = hidden_dim
    left_proj = fused[..., :h]                            # [B,N,N,h]
    left_gate = torch.sigmoid(fused[..., h : 2 * h])      # [B,N,N,h]
    right_proj = fused[..., 2 * h : 3 * h]                # [B,N,N,h]
    right_gate = torch.sigmoid(fused[..., 3 * h :])      # [B,N,N,h]

    # ------------------------------------------------------------------
    # 3️⃣ Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------

        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure tensors are contiguous for the Triton kernel
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()  # shape [B,N,N]; values 0/1

        total_elements = left_proj.numel()  # B * N * N * hidden_dim
        # launch ~1‑K threads per block; good occupancy
        BLOCK = 1024

            right_ptr=right_proj,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total_elements,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # ``left_proj`` and ``right_proj`` now hold the masked & gated tensors
        left = left_proj
        right = right_proj
        # free intermediate gate tensors early

    # 5️⃣ Heavy O(N³) pairing via batched GEMM (FP16)
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)   # shape [B,N,N,hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden dimension) – FP16
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)

    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)  # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply out_gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate    # element‑wise gating, FP16

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = F.linear(out, to_out_w)  # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)

# Triton kernel: fuse mask and gate on left/right projections.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]   (broadcasted along H)
    total_elements,    # int    total number of fp16 elements (B·N·N·H)
    H,                 # int    hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    # flat index range for this program
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # for each element in the (B,N,N,H) tensor we need the corresponding mask
    # that lives on the (B,N,N) tensor.  The mask index is the flat index
    # integer-divided by H because H is the innermost (fastest‑changing) dim.
    mask_offs = offs // H

    # Load data (value 0.0 for OOB loads – never used)
    left_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Load mask, using 1.0 for OOB (neutral element for multiplication)
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse mask, gate and projection
    left_out = left_val * left_gate_val * mask_val
    right_out = right_val * right_gate_val * mask_val

    # Store the fused results back in‑place
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
) -> torch.Tensor:
    """
    Perform batched matrix multiplication C = A @ Bᵀ where A and B are
    tensors on shape [B, N, N, H] (FP16).  Returns a tensor of the same
    shape and dtype.
    """
    B, N, _, H = left.shape
    # Rearrange to (B·H, N, N) for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched GEMM with transposed right matrix
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

Outgoing TriMul (AlphaFold‑3) – forward only.

Algorithm (FP16‑centric)
------------------------
1. First LayerNorm (float32 for stability) → cast to FP16.
2. Fuse the four per‑pair projections (left/right) and their gates
   into a single GEMM:
       [left_proj, left_gate, right_proj, right_gate] = Linear(x, W_fused)
   where W_fused has shape (4·hidden, dim).  Apply sigmoid to the
   gate slices.
3. Compute the separate output‑gate (sigmoid(Linear(x, W_out_gate))).
4. If a mask is supplied, launch a tiny Triton kernel that fuses
   *mask* × *gate* × *proj* for both left and right tensors,
   overwriting the projections in‑place.
   Without a mask we simply multiply by the gates.
5. Heavy contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication C = A @ Bᵀ with
   A,B ∈ ℝ^{(B·H)×N×N}.  It is carried out by `torch.bmm` on FP16
   (Tensor‑core accelerated).
6. Second LayerNorm over the hidden dimension (FP16), multiply by the
   output‑gate, and project back to the original `dim` with a final
   linear layer.
7. Cast the result back to FP32 (the harness expects FP32).
"""

    mask_offs = offs // H

    left: torch.Tensor,
    right: torch.Tensor,
) -> torch.Tensor:
    """
    Batched matrix multiply C = A @ Bᵀ for tensors of shape [B,N,N,H] (FP16).
    Returns a tensor of identical shape and dtype.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) – contiguous layout for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    right_mat  = right_perm.view(B * H, N, N)     # (B·H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
