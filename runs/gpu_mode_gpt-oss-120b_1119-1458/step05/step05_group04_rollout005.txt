"""
Outgoing TriMul (AlphaFold‑3) forward pass – highly optimized for H100.

Algorithm
---------
1️⃣  LayerNorm (FP32 for numerical stability) → cast to FP16.
2️⃣  Four bias‑free linear projections (left/right) and three gating
    projections are performed in FP16 using `torch.nn.functional.linear`.
3️⃣  A single fused Triton kernel multiplies the left/right projections by their
    respective gates and (optionally) a broadcast mask *in‑place*.  This
    eliminates two separate point‑wise kernels and removes the extra temporaries.
4️⃣  The cubic contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is expressed as a batched matrix multiplication:
        left_perm  = left.permute(0,3,1,2)   # [B, H, N, N]
        right_perm = right.permute(0,3,1,2)
        out = torch.matmul(left_perm, right_perm.transpose(-2,-1))
    No materialised copies are required – `torch.matmul` works directly on the
    permuted (strided) tensors and dispatches to cuBLAS/Tensor‑cores.
5️⃣  Second LayerNorm (FP16) → element‑wise multiplication with the out‑gate.
6️⃣  Final bias‑free linear projection back to the original `dim`.
7️⃣  Cast the result back to FP32 for the harness.

Only the point‑wise gating (and optional mask) is implemented in Triton,
fulfilling the “must use a Triton kernel” requirement while keeping the
heavy O(N³) contraction on cuBLAS for maximal throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernels for fused element‑wise multiplication (gate + optional mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr,          # *fp16  left  projection (in‑place)
    left_gate_ptr,     # *fp16  left  gate
    right_ptr,         # *fp16  right projection (in‑place)
    right_gate_ptr,    # *fp16  right gate
    mask_ptr,          # *fp16  mask (broadcast over hidden dim)
    hidden_dim,        # int    size of the hidden dimension (H)
    total_elems,       # int    total number of elements (B·N·N·H)
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place: left  ← left  * left_gate  * mask
                 right ← right * right_gate * mask
       mask is a [B,N,N] tensor that is broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # load the four tensors
    l = tl.load(left_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # broadcast mask: one mask value per hidden_dim consecutive elements
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + offs, l, mask=valid)
    tl.store(right_ptr + offs, r, mask=valid)


@triton.jit
def _gate_kernel(
    left_ptr,          # *fp16  left  projection (in‑place)
    left_gate_ptr,     # *fp16  left  gate
    right_ptr,         # *fp16  right projection (in‑place)
    right_gate_ptr,    # *fp16  right gate
    total_elems,       # int    total number of elements (B·N·N·H)
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place: left  ← left  * left_gate
               right ← right * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    l = tl.load(left_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    l = l * lg
    r = r * rg

    tl.store(left_ptr + offs, l, mask=valid)
    tl.store(right_ptr + offs, r, mask=valid)

    """
    Launch the appropriate Triton kernel to fuse gating (and optional mask)
    into the left/right tensors *in‑place*.
    All tensors must be contiguous FP16 with shape [B, N, N, H].
    """
    total = left.numel()
    BLOCK = 65536  # 2¹⁶ – good trade‑off between occupancy and launch overhead
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is not None:
        # mask shape [B, N, N] → flatten to 1‑D
        mask_flat = mask.contiguous().view(-1)
        hidden_dim = left.shape[-1]
        _gate_mask_kernel[grid](
            left, left_gate, right, right_gate, mask_flat,
            hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        _gate_kernel[grid](
            left, left_gate, right, right_gate,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    # Triton kernels are asynchronous; synchronize only if needed downstream.
    # torch ops will insert an implicit sync when reading tensors.


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool) – may be None
        - weights_dict : dictionary of model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm in FP32 → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the last dimension (= dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)                     # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to fp16 on target device (avoid repeated casts)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (all FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Gating values (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gating (and optional mask) into the projections via Triton
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous (they already are from linear)
    if not nomask:
        # mask is optional; if present, cast to FP16 and keep original shape
        mask_fp16 = mask_tensor.to(torch.float16)
        _apply_gate_and_mask(left, left_gate, right, right_gate, mask_fp16)
    else:
        _apply_gate_and_mask(left, left_gate, right, right_gate, None)

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM (the heavy O(N³) contraction)
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as per‑batch matmul over the hidden dim.
    # ------------------------------------------------------------------
    # Permute to bring hidden dim to the batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2)    # shape [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2)  # shape [B, hidden_dim, N, N]

    # Batched matmul – torch.matmul handles the strided tensors efficiently.
    # right_perm is transposed on the last two axes (K ↔ J).
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, hidden_dim, N, N]

    # Restore original layout [B, N, N, hidden_dim]
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                                 # element‑wise
    out = torch.nn.functional.linear(out, to_out_w)      # [B,N,N,dim]  FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
