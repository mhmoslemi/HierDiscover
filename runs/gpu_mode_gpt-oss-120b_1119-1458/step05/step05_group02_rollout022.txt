"""
Highly‑optimized Triton implementation of the outgoing TriMul (AlphaFold‑3) forward
pass.

Algorithm
---------
1️⃣  First LayerNorm (float32) on the raw input.
2️⃣  Cast to fp16 and fuse **all** linear projections (left/right projections,
    left/right gates, and the out‑gate) into a **single** matrix multiplication.
    This reduces the number of passes over the large [B,N,N,dim] tensor.
3️⃣  Apply the sigmoid gates.  If a mask is present a tiny Triton kernel fuses
    the three element‑wise multiplications `proj * gate * mask`.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is carried out as a batched GEMM (torch.bmm) on tensors reshaped to
    [B·H, N, N] – each hidden channel becomes an independent batch.
5️⃣  Second LayerNorm, out‑gate multiplication, and the per‑channel scaling
    (to_out_norm) are fused in a Triton kernel.  The kernel reads the
    contraction result and the out‑gate, computes mean/variance per (b,i,j)
    position, normalises, applies the per‑channel weight & bias, and finally
    multiplies by the out‑gate.  The result overwrites the contraction tensor,
    so no extra memory is allocated.
6️⃣  Final linear projection back to the original dimension (fp16 → fp32).

Only the forward pass is provided – no autograd information is required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -----------------------------------------------------------------------
# Triton kernel: fuse element‑wise multiplication with broadcast mask
# -----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be multiplied (left_proj or right_proj)
    gate_ptr,       # *fp16 : sigmoid gate (same shape as x)
    mask_ptr,       # *fp16 : mask (shape B·N·N, broadcast over hidden dim)
    hidden_dim,     # i32   : size of hidden dimension H
    total_elems,    # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elems

    # each hidden entry repeats the same mask -> integer division to map
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=active)


# -----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + out‑gate + to_out_norm scaling (in‑place)
# -----------------------------------------------------------------------
@triton.jit
def ln_gate_fuse_kernel(
    hidden_ptr,          # *fp16 : contraction result, shape [B·N·N·H]
    gate_ptr,            # *fp16 : out_gate, same shape
    norm_weight_ptr,     # *fp16 : to_out_norm.weight   (H,)
    norm_bias_ptr,       # *fp16 : to_out_norm.bias     (H,)
    H: tl.constexpr,     # hidden dimension (compile‑time)
    eps: tl.constexpr,   # epsilon for LayerNorm
):
    pos = tl.program_id(0)               # flattened position index: 0 .. B·N·N‑1
    # offsets for the hidden dimension (contiguous innermost stride = 1)
    h_off = tl.arange(0, H)               # shape (H,)
    base = pos * H                         # start of this position in memory

    # ---------------------------------------------------------------
    # 1) Load the hidden vector and gate vector (fp16)
    # ---------------------------------------------------------------
    hidden_vec = tl.load(hidden_ptr + base + h_off)    # (H,)
    gate_vec   = tl.load(gate_ptr   + base + h_off)    # (H,)

    # promote to fp32 for a numerically stable norm
    hidden_f32 = hidden_vec.to(tl.float32)
    gate_f32   = gate_vec.to(tl.float32)

    # ---------------------------------------------------------------
    # 2) Compute mean and variance (per‑position)
    # ---------------------------------------------------------------
    ones = tl.full((H,), 1.0, dtype=tl.float32)       # constant 1s
    sum_h   = tl.dot(hidden_f32, ones)                # Σ h_i
    sum_sq  = tl.dot(hidden_f32 * hidden_f32, ones)   # Σ h_i²

    inv_H = 1.0 / H
    mean = sum_h * inv_H
    var  = sum_sq * inv_H - mean * mean
    inv_std = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # 3) Normalise, apply to_out_norm weight&bias, then out_gate
    # ---------------------------------------------------------------
    norm_weight = tl.load(norm_weight_ptr + h_off).to(tl.float32)
    norm_bias   = tl.load(norm_bias_ptr   + h_off).to(tl.float32)

    normalized   = (hidden_f32 - mean) * inv_std            # (H,)
    scaled       = normalized * norm_weight + norm_bias    # (H,)
    gated_hidden = scaled * gate_f32                       # (H,)

    # cast back to fp16 and store in‑place
    out_fp16 = gated_hidden.to(tl.float16)
    tl.store(hidden_ptr + base + h_off, out_fp16)


# -----------------------------------------------------------------------
# Helper: batched GEMM that implements the O(N³) contraction
# -----------------------------------------------------------------------
def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, H] (fp16)
    returns    : [B, N, N, H] (fp16)   =   Σₖ left[...,k,:] * right[...,k,:]
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]; each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # bmm with right transposed (i‑k × j‑kᵀ) → matches the einsum in the paper
    c = torch.bmm(a, b.transpose(1, 2))
    # restore layout [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# -----------------------------------------------------------------------
# Main entry‑point
# -----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input tensor [B, N, N, dim] (float32)
        torch.Tensor,                # mask tensor   [B, N, N]   (float32 or bool)
        Dict[str, torch.Tensor],     # model weights  (float32)
        Dict[str, Any]               # config dict   contains "dim" and "hidden_dim"
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Returns a float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable on raw input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the work
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse all linear projections (proj, gate, out_gate) into one matmul
    #    Weight shapes: (hidden_dim, dim)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # concatenate: [left_proj; left_gate; right_proj; right_gate; out_gate]
    concat_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(torch.float16).contiguous()                 # shape (5*H, dim)

    # single linear pass → shape [B,N,N,5*H]
    lin = torch.nn.functional.linear(x, concat_w)    # fp16

    H = hidden_dim
    # slice the concatenated result
    left_proj      = lin[..., :H]
    left_gate_raw  = lin[..., H:2*H]
    right_proj     = lin[..., 2*H:3*H]
    right_gate_raw = lin[..., 3*H:4*H]
    out_gate_raw   = lin[..., 4*H:5*H]

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)          # [B,N,N,H]

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – fused element‑wise multiplication using Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()               # B·N·N·H
        BLOCK = 1024                                   # ≤ 1024 threads per block
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # after the kernel the tensors `left_proj` and `right_proj` already hold the
        # gated‑and‑masked values → rename for clarity
        left  = left_proj
        right = right_proj
    else:
        # No mask – ordinary elementwise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (batched GEMM)
    # -----------------------------------------------------------------
    out_hidden = _batched_matmul(left, right)   # [B,N,N,H]  fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + to_out_norm (fused Triton kernel)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)

    B, N, _, _ = out_hidden.shape
    total_positions = B * N * N
    grid = (total_positions,)               # one program per (b,i,j)

    ln_gate_fuse_kernel[grid](
        out_hidden,                 # in‑place write
        out_gate,                   # out‑gate (fp16)
        to_out_norm_weight,
        to_out_norm_bias,
        H=hidden_dim,
        eps=1e-5,
        num_warps=4,
    )
    # at this point `out_hidden` contains the gated‑and‑normalised hidden vector

    # -----------------------------------------------------------------
    # 7️⃣ Final linear projection back to original dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, H)
    out = torch.nn.functional.linear(out_hidden, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
