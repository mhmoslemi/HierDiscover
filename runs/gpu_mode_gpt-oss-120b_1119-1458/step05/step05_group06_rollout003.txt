def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # First LayerNorm (float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)

    # Concatenate linear weights into a single weight for the fused projection
    # All weights have shape [out_features, in_features] (dim hidden), but in given weights they are stored as shape (hidden_dim, dim)
    # left_proj.weight etc. Are (hidden_dim, dim). Great.

    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"]
        ],
        dim=0
    ).to(torch.float16)
    # Perform fused linear
    proj = torch.nn.functional.linear(x, weight_concat) # shape [B,N,N,5*hidden_dim]

    # split
    # Use torch.split on last dim
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(proj, hidden_dim, dim=-1)

    # gating with sigmoid
    # left_gate = torch.sigmoid(left_gate)
    # right_gate = torch.sigmoid(right_gate)
    # out_gate = torch.sigmoid(out_gate) (maybe need later after contraction)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    # apply gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # heavy contraction
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right) # returns fp16
    else:
        contracted = _batched_matmul_torch(left, right)

    # second LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    contracted = torch.nn.functional.layer_norm(contracted, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # gating after normalization
    out_gate = torch.sigmoid(out_gate) # out_gate shape same as contracted (float16)
    contracted = contracted * out_gate

    # final projection to dim
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(contracted, to_out_weight)
    # shape [B,N,N,dim] fp16

    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fused linear projections + Triton batched GEMM.

The implementation fuses the five weight matrices (two projections, three gates)
into a single fp16 linear layer, applies sigmoid gating and an optional mask,
then performs the O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
as a batched GEMM using a custom Triton kernel for N ≤ 256 (otherwise torch.bmm).
A second LayerNorm, out‑gate and final projection produce the float32 output.
All heavy arithmetic runs in fp16 with fp32 accumulation for speed,
while the two LayerNorms run in fp16 for a good trade‑off between stability
and performance.
"""

from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 × fp16 → fp32)
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for TriMul contraction (C = A @ Bᵀ)."""
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------------------------------------------------------
    # Offsets for the output tile (i, j)
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator (mixed‑precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over k‑tiles
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---------- load A[i, k] ----------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---------- load Bᵀ[j, k] ----------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k first
            + offs_j[None, :] * stride_bj   # j second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract `left` and `right` (both [B, N, N, H]) using a Triton GEMM.
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()    # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()   # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()    # (N*N, N, 1)

    # tile size – 64×64 works well for H100 fp16 mma
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract `left` and `right` using cuBLAS via torch.bmm.
    Operates in fp16 with fp32 accumulation (default for torch.bmm on GPU).
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – may be all‑ones
        - weights_dict  : mapping name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``.
                          Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ fuse all five Linear layers into a single fp16 GEMM
    #    order: left_proj, left_gate, right_proj, right_gate, out_gate
    # ------------------------------------------------------------------
    # Concatenate weight matrices (shape: [out_features, in_features])
    W_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*hidden_dim, dim)

    # One fused projection (fp16)
    proj = F.linear(x, W_concat)  # [B, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ split and apply sigmoid gating
    # ------------------------------------------------------------------
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=-1
    )  # each: [B, N, N, hidden_dim] fp16

    # Sigmoid in fp16
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)  # will be used after 2nd LN

    # Apply element‑wise gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ optional mask (applied after gating – equivalent mathematically)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ heavy contraction (batched GEMM)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16
    else:
        contracted = _batched_matmul_torch(left, right)    # fp16

    # ------------------------------------------------------------------
    # 7️⃣ second LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    contracted = contracted * out_gate  # element‑wise gate (fp16)

    out = F.linear(
        contracted,
        weights["to_out.weight"].to(torch.float16),
    )  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ cast back to float32 for the test harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
