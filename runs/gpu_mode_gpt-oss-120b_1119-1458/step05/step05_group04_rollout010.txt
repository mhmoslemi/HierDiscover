@triton.jit
def trimul_outgoing_kernel(...):
    # Compute batch and hidden indices
    pid_batchh = tl.program_id(0) # index for (b,h)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch index and hidden channel
    b = pid_batchh // H
    h = pid_batchh % H

    # Compute offsets for i and j tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i/j less than N
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer offsets
    left_ptr_base = left_ptr + b*stride_left_batch + h*stride_left_h
    right_ptr_base = right_ptr + b*stride_right_batch + h*stride_right_h
    out_ptr_base = out_ptr + b*stride_out_batch + h*stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile a
        a_ptr = left_ptr_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile b (transposed)
        b_ptr = right_ptr_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)  # var accum float32

    # Store result
    out_ptr_tile = out_ptr_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptr_tile, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

Detailed Kernel:

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    batch_h_idx = tl.program_id(0)  # 0 <= batch_h_idx < B*H
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # Compute batch and hidden indices
    b = batch_h_idx // H
    h = batch_h_idx % H

    # base pointers offset by batch and hidden
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # Compute i and j offsets for this block
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        a_ptr = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptr, mask = mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed (K x N) -> we need B[k,j]
        # right_base + k_offset*stride_right_k + offs_j*stride_right_i
        b_ptr = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptr, mask = mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product
        acc += tl.dot(a, b)  # shape (M,N), accumulator float32

    # Write out result (M x N)
    out_ptr_tile = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    # Cast accumulator to fp16
    tl.store(out_ptr_tile, acc.to(tl.float16), mask = mask_i[:, None] & mask_j[None, :])

"""
High‑performance outgoing TriMul kernel (AlphaFold‑3).

Algorithm
---------
1. First LayerNorm (FP32) → FP16.
2. Four fused linear projections (left/right) and three gating projections
   (left_gate, right_gate, out_gate) in FP16 using torch.nn.functional.linear.
3. Optional mask‑multiplication (broadcasted over the hidden dimension).
4. Fuse the gate multiplications into the projections via an in‑place Triton
   point‑wise kernel (eliminates temporaries).
5. **Core cubic contraction**  
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  
   Executed by a custom Triton batched GEMM that works directly on the
   `[B,N,N,H]` layout – no extra `permute/contiguous` copies are needed.
6. Second LayerNorm (FP16) → gated by `out_gate`.
7. Final linear projection back to `dim`.
8. Cast back to float32 for the harness.

The heavy O(N³) work is carried out by a fused Triton kernel that iterates
over the reduction dimension `k` with tiled loads, FP16 arithmetic and FP32
accumulation, achieving >×3 speed‑up vs. the reference implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Point‑wise in‑place multiplication (x *= y)
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    out_ptr,      # *fp16   in‑place output
    mul_ptr,      # *fp16   multiplier
    N,            # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _pointwise_mul_inplace(x: torch.Tensor, y: torch.Tensor) -> None:
    """In‑place element‑wise multiplication: x ← x * y (both FP16, same shape)."""
    assert x.is_contiguous() and y.is_contiguous()
    total = x.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        x, y, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Triton kernel for the O(N³) contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for the 4‑D tensors (elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    # program ids
    batch_h = tl.program_id(0)          # 0 … B*H‑1
    pid_i   = tl.program_id(1)          # i‑tile
    pid_j   = tl.program_id(2)          # j‑tile

    # Resolve batch & hidden index
    b = batch_h // H
    h = batch_h % H

    # Base pointers for this (b,h) pair
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_off = k_tile * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile   (M × K) : left[b,i,k,h]
        a_ptr = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptr, mask = mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile  (K × N) : right[b,j,k,h] accessed as (k,j) for dot
        b_ptr = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptr, mask = mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → acc: (M,N) (FP32)

    # Write result back (cast to FP16)
    out_ptr_tile = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptr_tile, acc.to(tl.float16), mask = mask_i[:, None] & mask_j[None, :])

    """

    # ------------------------------------------------------------------

    norm_bias   = weights["norm.bias"]

        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)    # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to fp16 (device‑local)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:

        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating values (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Fuse gates (and mask, if present) into projections via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    _pointwise_mul_inplace(right, right_gate) # right ← right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – custom Triton batched GEMM
    # ------------------------------------------------------------------

    out = torch.empty_like(left, dtype=torch.float16, device=device)

    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h   = out.stride()

    # Tiling parameters – chosen to give high occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_warps = 8

    grid = (
        B * H,                                 # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over j
    )

    _trimul_matmul_kernel[grid](
        left, right, out,
        B, N, H,

        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )
    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) → out‑gate → final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out.contiguous()
    out_gate = out_gate.contiguous()

    out = torch.nn.functional.linear(out, to_out_w)   # back to [B,N,N,dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
