"""
Outgoing TriMul (AlphaFold‑3) – high‑performance forward pass.

Algorithm
---------
1. Layer‑norm the input (float32) for numerical stability.
2. Cast to fp16 and apply four bias‑free linear projections:
   * left_proj  : x → [B,N,N,hidden]
   * right_proj : x → [B,N,N,hidden]
   * left_gate  : sigmoid(x → hidden)
   * right_gate : sigmoid(x → hidden)
   * out_gate   : sigmoid(x → hidden)   (used after the contraction)
   The two linear layers of each side are fused by stacking the projection &
   gate weights and performing a single dense mat‑mul (torch.nn.functional.linear).
3. Apply the optional mask (broadcasted on the hidden dim) and fuse the
   corresponding element‑wise multiplication with the gates:
   `left  = left  * left_gate  * mask`  
   `right = right * right_gate * mask`
4. Heavy O(N³) contraction:
   `out[i,j,:] = Σₖ left[i,k,:] * right[j,k,:]`
   Implemented as a batched GEMM `C = A @ Bᵀ` where
   `A = left.permute(0,3,1,2).reshape(B*hidden,N,N)`,
   `B = right.permute(0,3,1,2).reshape(B*hidden,N,N)`.
   • For `seq_len ≤ 256` we launch a custom Triton kernel that
     tiles the multiplication with a 128×128×64 layout (fp16 × fp16 → fp32
     accumulation).  This reduces the number of program launches and the
     total memory traffic by a factor ~2 vs. a 64×64 tile.
   • For larger sequences we fall back to cuBLAS via `torch.bmm`,
     which is the fastest path on H100 for big GEMMs.
5. Apply a second Layer‑Norm over the hidden dimension (fp16) followed by
   the `out_gate` element‑wise multiplication.
6. Final linear projection back to the original `dim`.
7. Cast the result back to float32 (required by the test harness).

All heavy arithmetic is performed in float16 with float32 accumulation for
numeric stability; only the two LayerNorms use float32 statistics.
"""  # noqa: E501

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (fp16 → fp16, accumulate fp32)
#   A, B : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   Tiling: 128×128 output tile, 64‑wide reduction tile.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction (fp16 inputs, fp32 acc)."""
    pid_m = tl.program_id(0)   # tile over output rows (i)
    pid_n = tl.program_id(1)   # tile over output columns (j)
    pid_z = tl.program_id(2)   # batch*hidden index

    # ------------------------------------------------------------------
    # Compute tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A[i, k]   (shape: BLOCK_M × BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am                     # batch*hidden offset
            + offs_i[:, None] * stride_ai           # i axis
            + offs_k[None, :] * stride_ak           # k axis
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # --------------------------------------------------------------
        # Load Bᵀ[j, k]  (shape: BLOCK_K × BLOCK_N)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm                     # batch*hidden offset
            + offs_k[:, None] * stride_bk           # k axis first (Bᵀ)
            + offs_j[None, :] * stride_bj           # j axis second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the tile back to C
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using a custom Triton GEMM.
    Returns: tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # Reshape to [M, N, N] with M = B*H.
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # Tile size tuned for N ≤ 256 on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using cuBLAS (torch.bmm).  Operates in fp16
    with fp32 accumulation (default on GPU).
    Returns: tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # A @ Bᵀ  (batched GEMM)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm, out‑gate, and final projection
    # --------------------------------------------------------------
    contracted = F.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    contracted = contracted * out_gate
    out = F.linear(contracted, to_out_w)   # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
