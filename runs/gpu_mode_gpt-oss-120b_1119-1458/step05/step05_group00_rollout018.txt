"""
Optimized outgoing TriMul kernel (AlphaFold‑3).

Algorithm
---------
1.  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2.  Four linear projections (left/right) + three sigmoid‑gated
    linear layers (all FP16).  The left/right projections are
    multiplied **in‑place** by the masks and gates with a tiny
    Triton kernel (one pass → very low memory traffic).
3.  Heavy contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix‑multiply where the batch
   dimension is ``B·hidden_dim``.  We avoid the costly
   ``permute → contiguous → view`` dance of the reference
   implementation by letting ``torch.einsum`` perform the
   contraction directly on the original layout.
4.  Second LayerNorm (FP16) → element‑wise out‑gate → final
    linear projection back to ``dim``.
5.  Cast the result back to FP32 (the harness expects FP32).

The only Triton code is the mask+gate‑fusion kernel, satisfying
the “implement part of the op in a kernel” rule while keeping
the critical O(N³) contraction fully on the highly‑optimised
cuBLAS/cuTensor path.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate in a single pass.
#   left  ← left  * mask * left_gate
#   right ← right * mask * right_gate
# All tensors are contiguous FP16 with shape [B, N, N, H] (left/right/gates)
# and mask has shape [B, N, N] (broadcast over H).
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i64   total number of elements in left/right
    H,                 # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Index of the mask element (same for left/right & gates because mask is
    # broadcast over the hidden dimension)
    mask_offs = offs // H

    # Load values (out‑of‑bounds loads return 0 / 1 as guarded)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    mask_val  = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _einsum_contraction_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Both ``left`` and ``right`` are expected to be FP16 tensors of shape
    [B, N, N, H].  The contraction is performed with a single ``einsum``
    call, which under the hood uses cuTensor/cuBLAS to avoid any extra
    data copies.
    """
    # torch.einsum works on FP16 and will dispatch to the fastest
    # tensor‑core kernel on the H100.
    out = torch.einsum('bikd,bjkd->bijd', left, right)
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float32) or None
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                          # [B, N, N, dim]  FP32
    x = x.to(dtype_fp16)       # cast once for the heavy part

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (no extra copy of FP32 version)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) (FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Mask + gate fusion (Triton)  (in‑place)
    # ------------------------------------------------------------------
    if nomask or mask_tensor is None:
        # No mask → just gate the projections (single fused multiply)
        left  = left * left_gate
        right = right * right_gate
    else:
        # Ensure contiguous layout for the kernel
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()   # [B,N,N]

        total = left.numel()
        BLOCK = 1024                         # max threads per block on H100
        grid  = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gate tensors are no longer needed
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – a single einsum
    # ------------------------------------------------------------------
    out = _einsum_contraction_fp16(left, right)   # [B, N, N, H] (FP16)

    # Release projection tensors early (they’re large)
    del left, right

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                             # [B, N, N, H] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Apply final out‑gate and project back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate                          # element‑wise (FP16)
    out = F.linear(out, to_out_w)                 # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Return FP32 tensor (the harness expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
