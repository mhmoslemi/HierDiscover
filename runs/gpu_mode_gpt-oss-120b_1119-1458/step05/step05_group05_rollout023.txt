def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_batch, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # i tiles
    pid_j = tl.program_id(1)  # j tiles
    pid_bh = tl.program_id(2) # batch*hidden
    # calculate actual indices
    b = pid_bh // H
    h = pid_bh % H

    # compute tile start positions
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for bounds
    mask_i = i < N
    mask_j = j < N

    # compute base offset for this batch and hidden
    batch_off = b * stride_batch + h

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_k_tiles):
        k = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left loads: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_off + i[:, None] * stride_i + k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right loads: we need transposed shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + batch_off + k[:, None] * stride_k + j[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # store results
    c_ptrs = out_ptr + batch_off + i[:, None] * stride_i + j[None, :] * stride_k

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, H,
                  stride_batch, stride_i, stride_k,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)   # block row
    pid_j = tl.program_id(1)   # block col
    pid_bh = tl.program_id(2)  # batch*hidden index

    # derive batch and hidden index
    bh = pid_bh
    b = bh // H
    h = bh % H

    # start indices for tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base offset for this batch and hidden dim
    base = b * stride_batch + h   # stride_h = 1

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile (i,k)
        a_ptr = left_ptr + base + i[:, None] * stride_i + k[None, :] * stride_k
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed (k,j)

        acc += tl.dot(a, b)

    # store result
    c_ptr = out_ptr + base + i[:, None] * stride_i + j[None, :] * stride_k

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → cast to fp16.
2. Two linear projections (left/right) and three gating linear layers
   (left_gate, right_gate, out_gate) are computed in fp16.
3. If a mask is supplied we fuse the three element‑wise factors
   (tensor * gate * mask) with a tiny Triton kernel (`mask_gate_fuse_kernel`).
4. The O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton kernel (`trimul_kernel`) that
   implements a batched GEMM directly on the original (B,N,N,H) layout,
   thus avoiding the costly permute‑contiguous copies required by a
   naïve torch.bmm solution.
5. A second Layer‑norm (hidden_dim) is applied, followed by the out‑gate.
6. Final linear projection back to the original channel dimension.
7. Cast result to float32 (the harness expects float32).

All heavy work is done in fp16 with float32 accumulation for numerical
stability.  At least two Triton kernels are used (mask‑gate fuse and the
core O(N³) contraction).  The implementation is self‑contained and
works for the full range of test configurations.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1: fused multiplication of {tensor * gate * mask}
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16   → tensor to be updated (flattened)
    gate_ptr,       # *fp16   → gate tensor (same flattening as x)
    mask_ptr,       # *fp16   → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32    → size of the hidden dimension (H)
    size,           # i32    → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # mask does not have a hidden dimension → compute its index
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fused multiplication
    out = x * g * m

    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: core TriMul contraction (batched GEMM on [B,N,N,H])
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,       # *fp16   left tensor   [B,N,N,H]
    right_ptr,      # *fp16   right tensor  [B,N,N,H]
    out_ptr,        # *fp16   output tensor [B,N,N,H]
    B, N, H,        # i32    dimensions
    stride_batch,   # i64    stride of batch dimension (B stride)
    stride_i,       # i64    stride of first N dimension (i / j)
    stride_k,       # i64    stride of second N dimension (k)
    BLOCK_M: tl.constexpr,    # tile rows   (i)
    BLOCK_N: tl.constexpr,    # tile cols   (j)
    BLOCK_K: tl.constexpr,    # reduction   (k)
):
    # program ids encode tile row, tile col, and (batch, hidden) pair
    pid_i = tl.program_id(0)          # tile over i
    pid_j = tl.program_id(1)          # tile over j
    pid_bh = tl.program_id(2)         # combined batch‑hidden index

    # recover batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # starting indices of the current tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base offset for this (b, h) – hidden dimension is contiguous (stride 1)
    base = b * stride_batch + h

    # accumulator in fp32 (tensor cores accumulate in fp32 for fp16 inputs)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # --------------------------------------------------------------
        # Load left tile  [i, k]  → shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        a_ptr = left_ptr + base + i[:, None] * stride_i + k[None, :] * stride_k
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # --------------------------------------------------------------
        # Load right tile transposed [k, j] → shape (BLOCK_K, BLOCK_N)
        # --------------------------------------------------------------
        b_ptr = right_ptr + base + k[:, None] * stride_k + j[None, :] * stride_i
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # fused dot product (uses tensor cores on H100)
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Write back the result tile
    # --------------------------------------------------------------
    c_ptr = out_ptr + base + i[:, None] * stride_i + j[None, :] * stride_k
    tl.store(c_ptr, acc.to(left_ptr.dtype),
              mask=mask_i[:, None] & mask_j[None, :])

# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights once to fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w) # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (sigmoid of a linear projection)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → fp16, flattened for the kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)   # [B*N*N]
        size = left.numel()                      # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → plain element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – custom Triton kernel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)                # [B,N,N,hidden_dim]  fp16

    # strides for the original layout [B, N, N, H]
    stride_batch = left.stride(0)   # N * N * H
    stride_i      = left.stride(1)   # N * H   (i‑dimension and also j‑dimension)
    stride_k      = left.stride(2)   # H       (the second N‑dimension)

    # Tile sizes – product ≤ 1024 (max threads per Triton program)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H,                     # batch × hidden
    )
    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden_dim)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Gating with out_gate and final projection
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
