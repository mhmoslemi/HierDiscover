import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for fusing mask and gate multiplication
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : hidden dimension size
    size,           # i32   : total number of elements in x
    BLOCK: tl.constexpr,   # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    # identify corresponding mask entry (mask repeats for each hidden channel)
    mask_offs = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out_{bijh} = Σ_k left_{b i k h} * right_{b j k h}
    using a single batched torch.matmul on permuted tensors.
    All tensors are expected in fp16.
    """
    # left: [B, N, N, H] ; right: same
    # permute to [B, H, N, N] where the inner dimensions are (i,k) and (k,j)
    # then torch.matmul will compute the sum over k for each (i,j)
    out = torch.matmul(
        left.permute(0, 3, 1, 2),       # (B, H, I, K)
        right.permute(0, 3, 2, 1)       # (B, H, K, J) => K,J order
    )
    # out shape (B, H, I, J) → bring back to (B, I, J, H)
    return out.permute(0, 2, 3, 1).contiguous()

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    This implementation fuses the mask‑gate multiplication with a small
    Triton kernel, concatenates the four linear projections/gates into a
    single matmul to cut down on kernel launches, and performs the O(N³)
    contraction with a single batched ``torch.matmul`` on permuted tensors.
    All heavy arithmetic is carried out in fp16; the final result is cast
    back to fp32.
    """
    # -----------------------------------------------------------------
    # 0️⃣ unpack arguments
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm on the raw (float32) input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the remaining work
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse left/right projections and gating weights into one large matmul
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]

    # concatenate in the order: left_proj, left_gate, right_proj, right_gate
    # shape (4*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
        ],
        dim=0,
    ).to(torch.float16)

    # combined linear: output shape [B, N, N, 4*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_weight)

    # -----------------------------------------------------------------
    # 4️⃣ Split combined tensor back into the four needed pieces
    # -----------------------------------------------------------------
    # combined: (..., 4*hidden_dim)
    split = torch.split(combined, hidden_dim, dim=-1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw = split

    # -----------------------------------------------------------------
    # 5️⃣ Apply sigmoid to the gating tensors
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # -----------------------------------------------------------------
    # 6️⃣ Apply mask and element‑wise gating (fused via Triton)
    # -----------------------------------------------------------------
    if not nomask:
        # mask must be half‑precision and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        total_elems = left_proj_raw.numel()          # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)
        # left = left_proj_raw * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj_raw,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj_raw * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj_raw,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj_raw
        right = right_proj_raw
    else:
        # No mask – just elementwise multiplication with gate
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction (batched matmul)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm on the hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Out‑gate gating
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)      # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(out_gate_pre)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 🔟 Final linear projection back to original dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] (fp16)

    # -----------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to float32 as required by the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)

Algorithm
---------

2️⃣  Fuse the four linear projections (left/right and their gates) into a

3️⃣  If a mask is present, fuse the element‑wise multiplication
    ``proj * gate * mask`` for both left and right streams with a tiny
    Triton kernel (mask_gate_fuse_kernel).
4️⃣  O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a single batched matrix multiply using
    ``torch.matmul`` on permuted tensors – no intermediate copies.
5️⃣  Second layer‑norm (fp16) over the hidden dimension.
6️⃣  Apply the output gate (sigmoid) and the final linear projection back
    to the original channel dimension.
7️⃣  Cast the result back to float32 (required by the harness).

All heavy arithmetic is carried out in fp16 for maximal Tensor‑Core
throughput on an H100.  Only the mask‑gate fusion is written in Triton,
but the overall pipeline has far fewer kernel launches and memory copies
than a naïve implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse element‑wise mask and gate multiplication.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (flattened B·N·N)
    hidden_dim,     # i32   : hidden dimension size (H)
    size,           # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,      # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    active = offs < size                               # boundary guard

    # each hidden channel repeats the same mask entry → integer division
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)   # mask = 1 outside range

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction using a single batched matrix multiply.

    left, right: [B, N, N, H] (fp16)
    returns     : [B, N, N, H] (fp16)
    """
    # permute to (B, H, N, N) – the last two dims are the matrix dimensions
    # right is additionally transposed on its last two dims (K,J)
    out = torch.matmul(
        left.permute(0, 3, 1, 2),   # (B, H, I, K)
        right.permute(0, 3, 2, 1)   # (B, H, K, J)
    )                               # → (B, H, I, J)
    # bring the hidden dimension back to the trailing axis
    return out.permute(0, 2, 3, 1).contiguous()   # (B, I, J, H)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor with shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣  Layer‑norm on the raw float32 input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣  Cast to fp16 for the bulk of the computation
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣  Fuse the four linear layers (proj + gate) into one matmul
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]

    # concatenate in order: left_proj | left_gate | right_proj | right_gate
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w],
        dim=0
    ).to(torch.float16)                     # shape (4*hidden, dim)

    # one big linear transformation → shape [B, N, N, 4*hidden]
    combined = torch.nn.functional.linear(x, combined_weight)

    # -----------------------------------------------------------------
    # 4️⃣  Split back into the four components
    # -----------------------------------------------------------------
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw = torch.split(
        combined, hidden_dim, dim=-1
    )   # each : [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 5️⃣  Compute sigmoid gates
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # [B, N, N, hidden]
    right_gate = torch.sigmoid(right_gate_raw)

    # -----------------------------------------------------------------
    # 6️⃣  Apply mask (if any) and fuse multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dimension; make it half‑precision
        mask_fp16 = mask.to(torch.float16).contiguous()
        total_elems = left_proj_raw.numel()               # B·N·N·hidden
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj_raw * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj_raw,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj_raw * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj_raw,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj_raw
        right = right_proj_raw
    else:
        # no mask – simple elementwise gating
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # -----------------------------------------------------------------
    # 7️⃣  O(N³) contraction (batched matrix multiply)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden] (fp16)

    # -----------------------------------------------------------------
    # 8️⃣  Second LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣  Output gate
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N,hidden]
    out_gate = torch.sigmoid(out_gate_pre)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 🔟  Final projection back to the original channel size
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)           # [B,N,N,dim] (fp16)

    # -----------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to float32 for the outside world
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
