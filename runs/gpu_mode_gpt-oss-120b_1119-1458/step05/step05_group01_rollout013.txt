def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # Convert weights to half (if not already)
    norm_weight = weights['norm.weight'].to(device=device, dtype=torch.float16)
    norm_bias = weights['norm.bias'].to(device=device, dtype=torch.float16)

    left_proj_weight = weights['left_proj.weight'].to(device=device, dtype=torch.float16)
    right_proj_weight = weights['right_proj.weight'].to(device=device, dtype=torch.float16)
    left_gate_weight = weights['left_gate.weight'].to(device=device, dtype=torch.float16)
    right_gate_weight = weights['right_gate.weight'].to(device=device, dtype=torch.float16)
    out_gate_weight = weights['out_gate.weight'].to(device=device, dtype=torch.float16)
    to_out_norm_weight = weights['to_out_norm.weight'].to(device=device, dtype=torch.float16)
    to_out_norm_bias = weights['to_out_norm.bias'].to(device=device, dtype=torch.float16)
    to_out_weight = weights['to_out.weight'].to(device=device, dtype=torch.float16)

    # Cast input to fp16
    x = x.to(torch.float16)

    # First layernorm (fp16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections (fp16)
    # note: linear expects weight shape (out_features, in_features) i.e., hidden_dim, dim
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply mask and gates
    if not nomask:
        # mask broadcasted and same for both tensors
        mask_fp16 = mask.to(torch.float16)
        left = _apply_mask_gate(left, left_gate, mask_fp16)
        right = _apply_mask_gate(right, right_gate, mask_fp16)
    else:
        left = left * left_gate
        right = right * right_gate

    # Contraction: sum over k dimension
    # Use einsum to avoid copies
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # Second LayerNorm (fp16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Out gate and final linear projection back to dim
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim] half

    # Cast to float32 for output
    return out.to(torch.float32)

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Cast the FP32 input to FP16 and apply the first LayerNorm in FP16.
2️⃣  Fuse the two linear projections and their corresponding sigmoid gates
    (left_proj+left_gate, right_proj+right_gate) by concatenating the weight
    matrices and performing a single GEMM per side.
3️⃣  If a mask is present, fuse mask, projection and gate in a tiny Triton
    kernel:  out = proj * gate * mask  (mask broadcast over the hidden dim).
    Otherwise just multiply by the gates in place.
4️⃣  Perform the O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single `torch.einsum` which maps internally to a batched
    GEMM (Tensor‑core friendly) and therefore avoids the expensive permute &
    copy needed for a manual `bmm`.
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection
    back to the model dimension.
6️⃣  Cast the result back to FP32 for the harness.

All heavy linear operations stay in FP16 (Tensor‑core rate) and only the
final output is up‑cast.  The only custom Triton kernel is the mask‑gate
fusion; the rest leverages PyTorch's highly‑optimized cuBLAS/cuTensor kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden dim H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                # *fp16  [B, N, N, H]
    gate_ptr,                # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,                # *fp16  [B, N, N]
    out_ptr,                 # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute:  out = proj * gate * mask    (mask broadcast over hidden dim H).
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # i‑, j‑coordinates for this block
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – will be broadcast over H later
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles (BLOCK_H)
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile  (shape: BLOCK_I × BLOCK_J × BLOCK_H)
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask (mask broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store the fused result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H]  (fp16)
    mask       : [B, N, N]    (fp16)

    Returns out = proj * gate * mask  (mask broadcast over H)  as fp16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides in *elements* (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes – 32×32 works well for the N dimension,
    # and a hidden‑dim tile of 8 keeps per‑block shared memory small.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # -------------------------------------------------
    # Cast everything we need to fp16 for speed
    # -------------------------------------------------
    # First LayerNorm
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=torch.float16)

    # Linear weights (bias‑free)
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Output LayerNorm and final projection
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (stay in fp16)
    # -------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  fp16

    # -------------------------------------------------
    # 2️⃣ Fused projection + gate for the left side
    # -------------------------------------------------
    # Concatenate weights: [proj; gate]  -> shape (2*hidden_dim, dim)
    left_weight_cat = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)
    left_cat = torch.nn.functional.linear(x, left_weight_cat)       # [B, N, N, 2*H]

    left_proj, left_gate_pre = torch.split(left_cat, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)                       # [B, N, N, H]

    # -------------------------------------------------
    # 3️⃣ Fused projection + gate for the right side
    # -------------------------------------------------
    right_weight_cat = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)
    right_cat = torch.nn.functional.linear(x, right_weight_cat)        # [B, N, N, 2*H]

    right_proj, right_gate_pre = torch.split(right_cat, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)                         # [B, N, N, H]

    # -------------------------------------------------
    # 4️⃣ Out‑gate (still using the original x)
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B, N, N, H]

    # -------------------------------------------------
    # 5️⃣ Optional mask‑gate fusion (Triton)
    # -------------------------------------------------
    if not nomask:
        # mask is float32 or bool → convert to fp16 once
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # No mask → just element‑wise multiply by the sigmoid gates
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via a single einsum (tensor‑core friendly)
    # -------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # -------------------------------------------------
    # 8️⃣ Apply out‑gate and final projection back to model dim
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
