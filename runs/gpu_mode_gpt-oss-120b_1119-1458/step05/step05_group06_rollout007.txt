"""
Outgoing TriMul (AlphaFold‑3) – fused forward pass.

Algorithm
---------
1️⃣  Layer‑Norm the input (float32) → shape [B, N, N, C].
2️⃣  Cast to fp16 and flatten to [B·N·N, C].
3️⃣  Fuse *all* linear projections (left/right projections, 3 gates) into a
    single large matrix‑multiply:
        Y = x_flat @ W_concat.T
    where W_concat = [left_proj; left_gate; right_proj; right_gate; out_gate]
    (shape (5·H, C)).  This reads the input only once and produces the
    five intermediate tensors in one pass.
4️⃣  Apply sigmoid‑gates, optional mask, and obtain the gated tensors
    `left` and `right` (shape [B, N, N, H]).
5️⃣  Perform the O(N³) contraction
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    as a batched GEMM C = A @ Bᵀ.  For N ≤ 256 a handcrafted Triton kernel
    (fp16 → fp32 accumulation) is used; for larger N we fall back to cuBLAS
    via ``torch.bmm``.
6️⃣  Second Layer‑Norm over the hidden dimension, multiply by the pre‑computed
    ``out_gate``, and apply the final output projection (hidden → C).
7️⃣  Cast the result back to float32.

All heavy arithmetic (steps 2‑6) runs in fp16; the two LayerNorms are
performed in fp32 for stability.  The kernel reduces memory traffic by
fusing five linear layers into a single matrix‑multiply and by avoiding
intermediate materialisation of the gated tensors.

The function signature follows the problem statement:
    custom_kernel((input, mask, weights, config)) → torch.Tensor
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ Bᵀ) for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply C = A @ Bᵀ."""
    pid_m = tl.program_id(0)   # tile along i (row)
    pid_n = tl.program_id(1)   # tile along j (col)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A[i, k]
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Bᵀ[j, k]  (i.e. B[k, j])
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        acc += tl.dot(a, b)   # mixed‑precision accumulate

    # store C[i, j]
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction using a custom Triton kernel.
    Input tensors are expected to be *fp16* and have shape [B, N, N, H].
    Returns a tensor of the same shape (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides in elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Fallback contraction using cuBLAS (torch.bmm).  Operates on fp16 tensors.
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))         # A @ Bᵀ
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient computation).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)  # now fp16

    # ------------------------------------------------------------------
    # 4️⃣ Fuse all linear projections (projections + gates) into a single GEMM
    # ------------------------------------------------------------------
    # Cast all involved weight tensors to fp16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output dimension: shape (5*hidden_dim, dim)
    w_concat = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0
    )  # fp16

    # Flatten input to (B*N*N, dim) for a single large matmul
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # [B*N*N, dim]   fp16, contiguous

    # Perform the fused linear projection (no bias)
    y = F.linear(x_flat, w_concat)        # shape (B*N*N, 5*hidden_dim), fp16

    # Reshape back to spatial layout
    y = y.view(B, N, N, 5 * hidden_dim)   # [B, N, N, 5*H]

    # Split into the five components
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        y, hidden_dim, dim=-1
    )  # each [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates and optional mask
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)     # [B,N,N,H] fp16
    right_gate = torch.sigmoid(right_gate_raw)   # [B,N,N,H] fp16
    out_gate = torch.sigmoid(out_gate_raw)       # [B,N,N,H] fp16

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj_raw * left_gate * mask_fp16
        right = right_proj_raw * right_gate * mask_fp16
    else:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (TriMul)
    # ------------------------------------------------------------------
    seq_len = N
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # [B,N,N,H] fp16
    else:
        contracted = _batched_matmul_torch(left, right)   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden_dim) → out_gate → final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    contracted = F.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    contracted = contracted * out_gate  # element‑wise gated, fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out = F.linear(contracted, to_out_w)  # fp16 → shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
