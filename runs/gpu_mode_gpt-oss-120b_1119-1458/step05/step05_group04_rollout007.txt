```python
"""
Optimized outgoing TriMul kernel.

Algorithm:
1. LayerNorm on the 4‑D input (FP32) and cast to FP16.
2. Fuse the four linear projections (left/right) and the three gating projections
   into a single matmul by concatenating the weight matrices. This reduces the
   number of kernel launches and halves the reads of the large input tensor.
3. Split the fused tensor into projection and gate tensors, apply sigmoid to the
   gates and fuse the element‑wise gating (and optional mask) with a small Triton
   kernel.
4. Perform the cubic contraction as a batched GEMM using `torch.bmm` on a view
   reshaped to `[B·H, N, N]`. This lets cuBLAS/Tensor‑cores handle the heavy work.
5. Apply a second LayerNorm, out‑gate (sigmoid) and the final linear projection
   back to the original channel dimension.
6. Cast the result back to FP32.

The Triton kernel is used for the in‑place element‑wise product of the
projections with their corresponding gates (fused pointwise multiplication).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication
#   x ← x * y
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    x_ptr,                # *fp16    in‑place operand
    y_ptr,                # *fp16    multiplier
    N,                    # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(x_ptr + offs, mask=mask, other=0.0)
    b = tl.load(y_ptr + offs, mask=mask, other=0.0)

    tl.store(x_ptr + offs, a * b, mask=mask)


def _inplace_mul(x: torch.Tensor, y: torch.Tensor) -> None:
    """
    In‑place element‑wise multiplication using Triton.
    Both tensors must be contiguous, same shape, and fp16.
    """
    assert x.is_contiguous() and y.is_contiguous()
    total = x.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK, )
    _pointwise_mul_kernel[grid](
        x, y, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.nn.functional.layer_norm expects weight/bias with same dtype as input
    # we keep them FP32 to preserve accuracy, then cast the result.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear projections (left/right + gates) into one GEMM
    # ------------------------------------------------------------------
    # Load and cast weight matrices to fp16 once.
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)   # [H, dim]
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Concatenate along the output‑feature axis (rows) → shape (5*H, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )

    # Linear projection: will produce a tensor of shape [B, N, N, 5*H]
    fused = torch.nn.functional.linear(x, combined_weight)  # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor, apply sigmoids, and fuse gates with a Triton kernel
    # ------------------------------------------------------------------
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        fused, hidden_dim, dim=-1
    )
    # Sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate_pre)   # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_pre) # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)     # [B,N,N,H]

    # In‑place multiplication of projections with their gates using Triton
    # (left = left_proj * left_gate, right = right_proj * right_gate)
    left = left_proj.contiguous()
    right = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _inplace_mul(left, left_gate)   # left ← left * left_gate
    _inplace_mul(right, right_gate) # right ← right * right_gate

    # Optional mask (broadcast over hidden dimension)
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Cubic contraction via batched GEMM (Tensor‑Core accelerated)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # fused optional mask not needed here

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Cast back to FP32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

"""
Optimized outgoing TriMul kernel (AlphaFold‑3 style).

Algorithm Overview
------------------
1️⃣  First LayerNorm (float‑32) → cast to float‑16.
2️⃣  Fuse the 5 linear projections (left/right + three gates) into a single
    matrix‑multiply by concatenating their weight matrices.  This reduces the
    number of kernel launches and halves the reads of the large `[B,N,N,dim]`
    tensor.
3️⃣  Split the fused result, apply sigmoid to the gate slices and fuse the
    element‑wise gating (proj * gate) with an ultra‑light Triton kernel.
    (Masking, if present, is applied afterwards with a standard torch `*`.)
4️⃣  Perform the cubic contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    as a batched GEMM (`torch.bmm`) on a view shaped
    `[B·H, N, N]`, letting cuBLAS/Tensor‑Cores do the heavy lifting.
5️⃣  Second LayerNorm (float‑16) → out‑gate (sigmoid) → final linear projection
    back to the original channel dimension.
6️⃣  Cast the result back to float‑32 (required by the harness).

A Triton kernel is used for the in‑place element‑wise product of the
projections with their respective gates – the only mandatory Triton component.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication (x ← x * y)
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    x_ptr,          # *fp16  in‑place tensor
    y_ptr,          # *fp16  multiplier
    N,              # total number of elements (scalar)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(x_ptr + offs, mask=mask, other=0.0)
    b = tl.load(y_ptr + offs, mask=mask, other=0.0)
    tl.store(x_ptr + offs, a * b, mask=mask)


def _inplace_mul(x: torch.Tensor, y: torch.Tensor) -> None:
    """
    In‑place point‑wise multiplication using Triton.
    Both tensors must be contiguous, identical shape and dtype (fp16).
    """
    assert x.is_contiguous() and y.is_contiguous()
    total = x.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        x, y, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float‑32) → FP16
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # LayerNorm in float32 for numerical stability
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(torch.float16)                     # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear layers (left/right + 3 gates) into one GEMM
    # ------------------------------------------------------------------
    # Cast weight matrices once
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Concatenate along the output‑feature axis (rows)
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                            # shape (5*H, dim)

    # One fused linear: → [B, N, N, 5*H]
    fused = torch.nn.functional.linear(x, fused_weight)   # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split, sigmoid gates, and fuse projection*gate with Triton
    # ------------------------------------------------------------------
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        fused, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate_pre)   # fp16
    right_gate = torch.sigmoid(right_gate_pre) # fp16
    out_gate = torch.sigmoid(out_gate_pre)     # fp16

    # In‑place gate multiplication (left = left_proj * left_gate, etc.)
    left = left_proj.contiguous()
    right = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _inplace_mul(left, left_gate)   # left ← left * left_gate
    _inplace_mul(right, right_gate) # right ← right * right_gate

    # Optional mask (broadcast over hidden dim)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Cubic contraction via batched GEMM (Tensor‑Core accelerated)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to [B·H, N, N] for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore layout → [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                            # fp16 gating

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # -> [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Cast back to float‑32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
