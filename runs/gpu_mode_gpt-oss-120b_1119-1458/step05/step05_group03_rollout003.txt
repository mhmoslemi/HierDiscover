"""
TriMul (outgoing) – fast forward pass.
------------------------------------------------
* 1️⃣  Layer‑norm the input (FP32 → half).
* 2️⃣  Fuse **all** linear projections (left/right and three gates) into a **single**
      matmul by concatenating the five weight matrices.  This reduces the number
      of times the large input tensor `[B,N,N,dim]` is read from 5 → 1.
* 3️⃣  Split the concatenated result into
      left_proj, right_proj, left_gate, right_gate, out_gate.
      Apply sigmoid on the three gating streams.
* 4️⃣  Optional mask‑gate fusion – a tiny Triton kernel multiplies
      `tensor * gate * mask` in‑place, broadcasting the mask over the hidden
      dimension.
* 5️⃣  Heavy contraction  
      `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`  
      is performed as a batched GEMM (`B*H` independent matrix multiplies)
      using cuBLAS (`torch.bmm`).  The inputs are half‑precision, accumulation is
      in FP32 (Tensor‑Core friendly).
* 6️⃣  Second LayerNorm (FP16), out‑gate multiplication and final projection
      back to the original channel dimension.
* 7️⃣  Cast back to float32 for the harness.

Only the forward pass is required – no gradients are computed.
At least one Triton kernel (mask‑gate fusion) is used.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place `x = x * gate * mask`  (mask broadcast over hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, flatten view
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,)
    size,             # i32   : total number of elements in x (B*N*N*H)
    hidden_dim,       # i32   : hidden dimension (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load the data and the gate
    x   = tl.load(x_ptr   + offs, mask=mask, other=0.0).to(tl.float16)
    g   = tl.load(gate_ptr+ offs, mask=mask, other=0.0).to(tl.float16)

    # Compute the index of the mask entry (mask is broadcast over hidden_dim)
    mask_offs = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0).to(tl.float16)

    # In‑place fused multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the heavy O(N³) contraction
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]` using
    cuBLAS batched GEMM.  Both inputs must be `torch.float16` tensors
    of shape `[B, N, N, H]`.  Returns a tensor of the same shape (FP16).
    """
    B, N, _, H = left.shape
    # Bring H into the batch dimension: (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM A @ Bᵀ  (fp16 inputs → fp32 accumulation)
    c = torch.bmm(a, b.transpose(1, 2))
    # Reshape back to (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Entry point called by the evaluation harness
# ----------------------------------------------------------------------

    """

        - config_dict  : must contain `dim` and `hidden_dim`;

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if nomask:
        # Simple in‑place element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate
    else:
        # Prepare mask (broadcasted over hidden_dim)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        total_elements = left.numel()                # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Gating tensors are no longer needed
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM (cuBLAS Tensor‑Core)
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]  (FP16)
    out = _batched_gemm_fp16(left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate  # element‑wise gating (FP16)
    to_out_w = weights["to_out.weight"].half()   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)  # → [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast result back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```
