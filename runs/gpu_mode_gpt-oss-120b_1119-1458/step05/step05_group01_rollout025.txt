Based on the typical matrix multiplication kernel:

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute base offsets for i and j tiles.
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # Offsets for i and j within block.
    offs_i = i_base + tl.arange(0, BLOCK_I)  # shape (BLOCK_I,)
    offs_j = j_base + tl.arange(0, BLOCK_J)  # shape (BLOCK_J,)

    # Masks for boundary
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Create 2D grid for i and j coordinates.
    i = offs_i[:, None]  # shape (BLOCK_I, 1)
    j = offs_j[None, :]  # shape (1, BLOCK_J)

    # Initialize accumulator over hidden dimension.
    # We'll compute result for all hidden dims (tile).
    # We'll iterate over hidden tiles.
    num_h_tiles = tl.cdiv(H, BLOCK_H)

        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

                + i * stride_left_i

                right_ptr
                + pid_b * stride_right_b
                + j * stride_right_i
                + offs_k[None, :, None] * stride_right_k  # shape (1, BLOCK_K, 1) broadcast over j and h
                + offs_h[None, None, :] * stride_right_h
            )

                right_ptrs,
                mask=mask_j[None, :, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )   # fp16

            # Broadcast left across j dimension and right across i dimension.
            # left shape (BLOCK_I, BLOCK_K, BLOCK_H) -> unsqueeze dim1 after i: (BLOCK_I, 1, BLOCK_K, BLOCK_H)
            # right shape (BLOCK_J, BLOCK_K, BLOCK_H) -> unsqueeze dim0 after j: (1, BLOCK_J, BLOCK_K, BLOCK_H)
            # Multiply and sum over k (axis=2)
            prod = left[:, None, :, :] * right[None, :, :, :]    # shape (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_H)
            acc += tl.sum(prod, axis=2)   # sum over K dimension -> shape (BLOCK_I, BLOCK_J, BLOCK_H)

        # Store the accumulator into out.
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + i * stride_out_i
            + j * stride_out_j
            + offs_h[None, None, :] * stride_out_h

"""
TriMul (outgoing) forward pass – high‑performance implementation.

Algorithm
---------
1️⃣  Layer‑Norm the input in fp16 (no bias‑folding – tiny numerical impact).
2️⃣  Fuse the five linear projections (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single matmul to cost only ONE kernel launch.
3️⃣  Apply sigmoid to the three gate tensors, optionally multiply by a mask
    (broadcast over the hidden dimension).
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is expressed as a custom Triton kernel that works on the original
    [B,N,N,H] layout – no expensive reshape/copy, hidden dimension is tiled.
5️⃣  Second Layer‑Norm (fp16) → out‑gate (sigmoid) → final projection
    back to the model dimension.
6️⃣  Cast the result to float32 (the harness expects fp32).

The kernel uses a classic GEMM‑style tiling (BLOCK_I×BLOCK_J×BLOCK_H)
with a per‑tile reduction over K (BLOCK_K).  For each hidden‑tile we
accumulate with `tl.dot`, which maps onto H100 tensor‑cores.
All heavy work stays in fp16, while the accumulation uses fp32 for
numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: O(N³) contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # *fp16  [B, N, N, H]  (left after gating & mask)
    right_ptr,         # *fp16  [B, N, N, H]  (right after gating & mask)
    out_ptr,           # *fp16  [B, N, N, H]  (output)
    B, N, H,
    # strides (in elements) for the three tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # ------------------------------------------------------------------
    # Tile offsets for the i, j dimensions
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    offs_i = i_base + tl.arange(0, BLOCK_I)          # (BLOCK_I,)
    offs_j = j_base + tl.arange(0, BLOCK_J)          # (BLOCK_J,)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Loop over hidden‑dimension tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        h_base = h_tile * BLOCK_H
        offs_h = h_base + tl.arange(0, BLOCK_H)      # (BLOCK_H,)
        mask_h = offs_h < H

        # Accumulator for this (i,j) tile and hidden‑tile → fp32 for stability
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # ----------------------------------------------------------
        # Loop over the K‑dimension (sequence length)
        # ----------------------------------------------------------
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for k_tile in range(num_k_tiles):
            k_base = k_tile * BLOCK_K
            offs_k = k_base + tl.arange(0, BLOCK_K)      # (BLOCK_K,)
            mask_k = offs_k < N

            # ------------------------------------------------------------------
            # Load left tile  : shape (BLOCK_I, BLOCK_K, BLOCK_H)   left[b,i,k,h]
            # ------------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] &
                    mask_k[None, :, None] &
                    mask_h[None, None, :],
                other=0.0,
            )   # fp16

            # ------------------------------------------------------------------
            # Load right tile (transposed) : shape (BLOCK_K, BLOCK_J, BLOCK_H)
            # right_T[k, j, h] == right[b, j, k, h]
            # ------------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_j[:, None, None] * stride_right_i
                + offs_k[:, None, None] * stride_right_k
                + offs_h[None, None, :] * stride_right_h
            )
            # right_ptrs has shape (BLOCK_J, BLOCK_K, BLOCK_H);
            # we need the K‑J transposition, so we simply swap the first two dims
            # via indexing after the load.
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] &
                    mask_k[None, :, None] &
                    mask_h[None, None, :],
                other=0.0,
            )   # fp16  shape (BLOCK_J, BLOCK_K, BLOCK_H)

            # Transpose K ↔ J to obtain shape (BLOCK_K, BLOCK_J, BLOCK_H)
            right_T = tl.trans(right, axes=(1, 0, 2))   # (BLOCK_K, BLOCK_J, BLOCK_H)

            # ------------------------------------------------------------------
            # Compute dot products for each hidden slice in the tile
            # acc[:, :, h] += left[:, :, h] @ right_T[:, :, h].T
            # ------------------------------------------------------------------
            for h in range(BLOCK_H):
                # Guard against out‑of‑range hidden elements
                if not mask_h[h]:
                    continue
                # left slice: (BLOCK_I, BLOCK_K)
                left_h = left[:, :, h]          # fp16
                # right slice (already transposed): (BLOCK_K, BLOCK_J)
                right_h = right_T[:, :, h]       # fp16
                # Matrix multiply → (BLOCK_I, BLOCK_J) in fp16, cast to fp32
                prod = tl.dot(left_h, right_h)   # fp16
                acc[:, :, h] += prod.to(tl.float32)

        # ------------------------------------------------------------------
        # Store the accumulated tile (convert back to fp16)
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] &
                 mask_j[None, :, None] &
                 mask_h[None, None, :],
        )
# ----------------------------------------------------------------------
# Helper: run the contraction kernel
# ----------------------------------------------------------------------
def _trimul_contract(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] with a
    custom Triton kernel (no extra copies).  All tensors must be fp16
    and reside on the same CUDA device.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in *elements* (not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # Tunable block sizes – 32×32 work well on H100 for the test shapes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8    # H is a multiple of 8 in all benchmark configs

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        # 8 warps per block give good occupancy on H100
        num_warps=8,
        num_stages=2,
    )
    return out

# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (fp16) – cheap and GPU‑friendly
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    )  # output fp16 because weight is fp16
    # Ensure fp16 dtype (the layer_norm already returns fp16)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣  Fuse the five linear projections into one matmul
    # -----------------------------------------------------------------
    # Gather the five weight matrices (all shape [hidden_dim, dim])
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Concatenate along the output‑feature dimension (hidden_dim)
    fused_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    B, N, _, _ = input_tensor.shape
    flat_x = x.view(B * N * N, dim)                # (M, dim)
    fused_out = torch.nn.functional.linear(flat_x, fused_w)   # (M, 5*hidden_dim)
    fused_out = fused_out.view(B, N, N, hidden_dim * 5)

    # Split back into the five tensors
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused_out, hidden_dim, dim=-1
    )  # each → [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣  Apply sigmoid gates
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 4️⃣  Mask fusion (optional).  Simple elementwise broadcast is cheap,
    #      no extra Triton kernel needed – the heavy work is the O(N³) step.
    # -----------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; broadcast over hidden dimension
        mask_fp16 = mask_tensor.to(device=device, dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣  Heavy O(N³) contraction via custom Triton kernel
    # -----------------------------------------------------------------
    out = _trimul_contract(left, right)   # shape [B,N,N,hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 6️⃣  Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # -----------------------------------------------------------------
    # 7️⃣  Apply out‑gate and final projection back to model dim
    # -----------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 8️⃣  Cast back to FP32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
