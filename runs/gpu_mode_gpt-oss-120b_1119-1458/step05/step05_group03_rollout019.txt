@triton.jit
def layernorm_fwd_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    n_slices, dim,
    eps: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)            # slice index
    # offset base for slice
    base = pid * dim

    sum = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    # loop over dim
    for offset in range(0, dim, BLOCK):
        cur = tl.arange(0, BLOCK)
        offs = base + offset + cur
        mask = offs < base + dim
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)  # FP32
        sum += tl.sum(x * mask.float())    # mask conversion to float?
        sum_sq += tl.sum(x * x * mask.float())
    # Compute mean and var
    mean = sum / dim
    var = sum_sq / dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass: write normalized
    for offset in range(0, dim, BLOCK):
        cur = tl.arange(0, BLOCK)
        offs = base + offset + cur
        mask = offs < base + dim
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        # weight and bias are per channel (dim). load them similarly
        weight = tl.load(weight_ptr + offset + cur, mask=mask, other=1.0)
        bias = tl.load(bias_ptr + offset + cur, mask=mask, other=0.0)
        out = (norm * weight + bias).to(tl.float16)
        tl.store(out_ptr + offs, out, mask=mask)

@triton.jit
def layernorm_fwd_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    dim: tl.constexpr,
    N_slices,
    eps: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)  # slice ID
    base = pid * dim
    # reduction
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)
    for start in range(0, dim, BLOCK):
        offs = base + start + tl.arange(0, BLOCK)
        mask = offs < base + dim
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)  # float32
        sum_val += tl.sum(x)
        sum_sq_val += tl.sum(x * x)
    mean = sum_val / dim
    var = sum_sq_val / dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    # second loop
    for start in range(0, dim, BLOCK):
        offs = base + start + tl.arange(0, BLOCK)
        mask = offs < base + dim
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        weight = tl.load(weight_ptr + start + tl.arange(0, BLOCK), mask=mask, other=1.0)
        bias = tl.load(bias_ptr + start + tl.arange(0, BLOCK), mask=mask, other=0.0)
        # cast to fp16 after scaling
        out_val = (norm * weight + bias).to(tl.float16)
        tl.store(out_ptr + offs, out_val, mask=mask)
```

"""
TriMul (outgoing) – highly‑optimized forward pass.

Main ideas
-----------
1. **LayerNorms in Triton** – a custom kernel normalises each
   (B,N,N, C) slice across the channel dimension in FP32 and writes
   back in FP16.  This replaces torch.nn.functional.layer_norm which
   incurs many small kernels.

2. **Fuse the five linear projections** – the three gating streams and
   the two value streams are concatenated into one weight matrix
   (shape (5·hidden_dim, dim)) and evaluated with a single
   `torch.nn.functional.linear`.  This removes four separate GEMMs
   and reduces memory traffic.

3. **Mask‑gate fusion** – if a mask is present we multiply
   `proj * gate * mask` in‑place with a tiny Triton kernel,
   avoiding extra element‑wise passes.

4. **Heavy contraction** – the O(N³) contraction
   ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`` is performed
   as a batched GEMM (`C = A @ Bᵀ`) using cuBLAS (torch.bmm).  This
   exploits Tensor‑Cores on H100 and is the fastest path for N ≤ 1024.

5. **Second LayerNorm + out‑gate + final projection** – a second Triton
   LayerNorm normalises the contraction result, then we apply the out‑gate
   (sigmoid) and finish with a single linear projection back to the
   original `dim`.

All tensors stay on the device, the final output is cast to FP32 as
required by the harness.  The implementation works for both masked and
unmasked inputs and supports `dim ∈ {128,384}` and `hidden_dim = 128`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernels

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,               # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,            # *fp16 : gate tensor, same shape as x
    mask_ptr,            # *fp16 : mask tensor of shape (S,) – S = B*N*N
    size,                # i32   : total number of elements (B*N*N*hidden_dim)
    hidden_dim,          # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # each hidden‑dim consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


@triton.jit
def layernorm_fp32_to_fp16_kernel(
    x_ptr,               # *fp32 : input tensor (flattened) [S*D]
    out_ptr,             # *fp16 : output tensor (flattened) [S*D]
    weight_ptr,          # *fp16 : weight (channel) [D]
    bias_ptr,            # *fp16 : bias   (channel) [D]
    *, DIM: tl.constexpr,    # channel dimension (D)
    EPS: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                # one program per slice
    base = pid * DIM                       # start of slice in 1‑D view

    # ----- reduction phase ------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for i in range(tl.cdiv(DIM, BLOCK)):
        off = i * BLOCK
        offs = base + off + tl.arange(0, BLOCK)
        mask = offs < base + DIM
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)         # fp32
        sum_val += tl.sum(x)
        sum_sq += tl.sum(x * x)

    mean = sum_val / DIM
    var = sum_sq / DIM - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ----- normalisation phase --------------------------------------------
    for i in range(tl.cdiv(DIM, BLOCK)):
        off = i * BLOCK
        offs = base + off + tl.arange(0, BLOCK)
        mask = offs < base + DIM

        x = tl.load(x_ptr + offs, mask=mask, other=0.0)         # fp32
        norm = (x - mean) * inv_std

        w = tl.load(weight_ptr + off + tl.arange(0, BLOCK), mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + off + tl.arange(0, BLOCK), mask=mask, other=0.0).to(tl.float32)

        out = (norm * w + b).to(tl.float16)
        tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def layernorm_fp16_to_fp16_kernel(
    x_ptr,               # *fp16 : input tensor (flattened) [S*D]
    out_ptr,             # *fp16 : output tensor (flattened) [S*D]
    weight_ptr,          # *fp16 : weight (channel) [D]
    bias_ptr,            # *fp16 : bias   (channel) [D]
    *, DIM: tl.constexpr,    # channel dimension (D)
    EPS: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    base = pid * DIM

    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for i in range(tl.cdiv(DIM, BLOCK)):
        off = i * BLOCK
        offs = base + off + tl.arange(0, BLOCK)
        mask = offs < base + DIM
        x = tl.load(x_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)
        sum_sq += tl.sum(x * x)

    mean = sum_val / DIM
    var = sum_sq / DIM - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    for i in range(tl.cdiv(DIM, BLOCK)):
        off = i * BLOCK
        offs = base + off + tl.arange(0, BLOCK)
        mask = offs < base + DIM

        x = tl.load(x_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        norm = (x - mean) * inv_std

        w = tl.load(weight_ptr + off + tl.arange(0, BLOCK), mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + off + tl.arange(0, BLOCK), mask=mask, other=0.0).to(tl.float32)

        out = (norm * w + b).to(tl.float16)
        tl.store(out_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper functions (pure‑torch + Triton)
# ----------------------------------------------------------------------


def layernorm_fp32_to_fp16(x_fp32: torch.Tensor,
                           weight_fp16: torch.Tensor,
                           bias_fp16: torch.Tensor,
                           eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm across the last dimension of a float32 tensor.
    Returns FP16 normalised tensor.
    """
    B, N, _, D = x_fp32.shape
    total_slices = B * N * N
    x_flat = x_fp32.contiguous().view(total_slices, D)
    out_flat = torch.empty_like(x_flat, dtype=torch.float16)

    BLOCK = 64
    grid = (total_slices,)
    layernorm_fp32_to_fp16_kernel[grid](
        x_flat,
        out_flat,
        weight_fp16,
        bias_fp16,
        DIM=D,
        EPS=eps,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_flat.view(B, N, N, D)


def layernorm_fp16_to_fp16(x_fp16: torch.Tensor,
                           weight_fp16: torch.Tensor,
                           bias_fp16: torch.Tensor,
                           eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm across the last dimension of a float16 tensor.
    Returns FP16 normalised tensor.
    """
    B, N, _, D = x_fp16.shape
    total_slices = B * N * N
    x_flat = x_fp16.contiguous().view(total_slices, D)
    out_flat = torch.empty_like(x_flat, dtype=torch.float16)

    BLOCK = 64
    grid = (total_slices,)
    layernorm_fp16_to_fp16_kernel[grid](
        x_flat,
        out_flat,
        weight_fp16,
        bias_fp16,
        DIM=D,
        EPS=eps,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_flat.view(B, N, N, D)


def _batched_gemm_fp16(left: torch.Tensor,
                       right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    performed as a batched GEMM with cuBLAS Tensor‑Cores.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B·H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))            # FP16 * FP16 → FP32 accumulation
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entrypoint – the custom kernel expected by the harness
# ----------------------------------------------------------------------


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradient computation).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor  : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor   : torch.Tensor [B, N, N]  (float32 or bool)
        - weights_dict  : mapping name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (True → mask ignored).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]  # projection width (c)
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)
    x = layernorm_fp32_to_fp16(input_tensor, norm_weight, norm_bias, eps=eps)  # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (proj + gates + out_gate)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)    # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # concatenate into one big weight matrix: (5·hidden_dim, dim)
    big_weight = torch.cat(
        [left_proj_w,
         right_proj_w,
         left_gate_w,
         right_gate_w,
         out_gate_w],
        dim=0,
    )  # FP16

    B, N, _, _ = x.shape
    total_slices = B * N * N
    x_flat = x.contiguous().view(total_slices, dim)       # [S, dim]
    # Linear: [S, dim] @ (5·hidden_dim, dim).T → [S, 5·hidden_dim]
    y = torch.nn.functional.linear(x_flat, big_weight)   # FP16

    y = y.view(B, N, N, 5 * hidden_dim)                  # [B,N,N,5·hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Split & apply sigmoid gates
    # ------------------------------------------------------------------
    left          = y[..., :hidden_dim]                                 # (proj)
    right         = y[..., hidden_dim:2 * hidden_dim]                     # (proj)
    left_gate_raw = y[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate_raw = y[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate_raw  = y[..., 4 * hidden_dim:5 * hidden_dim]

    left_gate  = torch.sigmoid(left_gate_raw)    # FP16
    right_gate = torch.sigmoid(right_gate_raw)   # FP16
    out_gate   = torch.sigmoid(out_gate_raw)     # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Mask‑gate fusion (if a mask is present)
    # ------------------------------------------------------------------
    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:
        # mask shape → (B·N·N,) broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elements = left.numel()                  # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Release gates early
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction via batched GEMM
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B,N,N,hidden_dim] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------
    out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = layernorm_fp16_to_fp16(out, out_norm_weight, out_norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # gating (FP16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    out_flat = out_norm.contiguous().view(total_slices, hidden_dim)   # [S, hidden_dim]
    final_fp16 = torch.nn.functional.linear(out_flat, to_out_w)      # [S, dim] FP16

    out_tensor = final_fp16.view(B, N, N, dim).float()   # cast to FP32 for harness
    return out_tensor
