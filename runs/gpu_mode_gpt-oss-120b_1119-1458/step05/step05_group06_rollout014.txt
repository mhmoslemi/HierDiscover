def _trimul_matmul_kernel(

"""
Outgoing TriMul – fast forward pass.

The heavy O(N³) contraction 
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] 
is expressed as a batched GEMM C = A @ Bᵀ where each hidden channel h is an
independent batch (M = B * hidden_dim).  A and B are stored as
[M, N, N] (batch, row, col) in FP16; the kernel accumulates in FP32
using Triton‐based tiling (128×128×64).  For larger sequences a cuBLAS
fallback (torch.bmm) is used.  All linear layers and gating are fused as
much as possible (proj+gate weights are concatenated) to reduce kernel
launch overhead.  LayerNorms remain in FP32 for numerical stability, the
final linear projection is in FP16, and the output is cast back to FP32
as required by the test harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (FP16 → FP16, accumulator FP32)
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes (M = batch*hidden)
    stride_am, stride_ai, stride_ak,   # A strides (batch, row(i), col(k))
    stride_bm, stride_bi, stride_bk,   # B strides (batch, row(j), col(k))
    stride_cm, stride_ci, stride_cj,   # C strides (batch, row(i), col(j))
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix‑multiply kernel for the TriMul contraction.
    Each program instance works on a (BLOCK_M × BLOCK_N) tile of the output
    for a single (batch × hidden) slice (pid_h).
    """
    pid_m = tl.program_id(0)                # tile along i (output rows)
    pid_n = tl.program_id(1)                # tile along j (output cols)
    pid_h = tl.program_id(2)                # batch * hidden index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # FP32 accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile: [BLOCK_M, BLOCK_K]   (row i, col k)
        a_ptrs = (
            A_ptr
            + pid_h * stride_am                     # batch*hidden offset
            + offs_i[:, None] * stride_ai           # i offset
            + offs_k[None, :] * stride_ak           # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Bᵀ tile: [BLOCK_K, BLOCK_N]   (row k, col j)
        b_ptrs = (
            B_ptr
            + pid_h * stride_bm                     # batch*hidden offset
            + offs_k[:, None] * stride_bk           # k offset (row)
            + offs_j[None, :] * stride_bi           # j offset (col)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result in FP16
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_h * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using a Triton GEMM.
    Returns tensor of shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bi, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # Tile size – 128×128 works well on H100 for FP16 TensorCores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bi,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using cuBLAS (torch.bmm). Operates in FP16
    with FP32 accumulation (torch.bmm default on recent GPUs). Returns
    [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float32)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fuse linear projections & gates (fp16 weights)
    # ------------------------------------------------------------------
    # left: proj + gate
    left_proj_gate_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # shape [(2*hidden_dim), dim]

    # right: proj + gate
    right_proj_gate_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                    # shape [(2*hidden_dim), dim]

    # out_gate (single linear)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # to_out projection
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # to_out LayerNorm parameters (keep in float32 – torch will up‑cast internally)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    # 5️⃣ fused linear pass for left & right (fp16)
    #    Output layout after linear: [..., 2*hidden_dim]
    # ------------------------------------------------------------------
    # Combined linear for left side
    left_combined = F.linear(x, left_proj_gate_w)         # [B, N, N, 2*hidden_dim]
    # Split into proj and gate
    left_proj, left_gate = torch.split(left_combined, hidden_dim, dim=-1)

    # Combined linear for right side
    right_combined = F.linear(x, right_proj_gate_w)       # [B, N, N, 2*hidden_dim]
    right_proj, right_gate = torch.split(right_combined, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 6️⃣ mask handling (optional) – broadcast over hidden_dim
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; bring to fp16 and expand
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates in-place to reduce memory pressure
    left = left_proj * left_gate         # [B, N, N, hidden_dim] (fp16)
    right = right_proj * right_gate      # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ heavy contraction (O(N³))
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    # Use Triton for truly small sequences; otherwise cuBLAS (torch.bmm)
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16
    else:
        contracted = _batched_matmul_torch(left, right)    # fp16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    # LayerNorm on contracted tensor (fp16).  Weight/bias are kept in float32.
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    contracted = contracted * out_gate
    out = F.linear(contracted, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
