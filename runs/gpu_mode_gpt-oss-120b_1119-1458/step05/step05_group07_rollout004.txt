"""
Fast outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float16) using the provided LN weights.
2. Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single GEMM:
        Y = x @ W_combined.T   where
        W_combined = concat([W_left, W_lgate, W_right, W_rgate, W_outgate]).
   Slice Y to obtain the four projected tensors and their gate logits.
3. Apply sigmoid to the three gate logits.
4. Fuse element‑wise multiplication of the projected tensors with their
   respective gates (and optional mask) using a tiny Triton kernel:
        proj = proj * gate * mask   (mask broadcast over hidden_dim)
5. Perform the O(N³) contraction as a batched half‑precision GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented via torch.bmm on reshaped tensors.
6. Second Layer‑norm over the hidden dimension (FP16) with its own
   LN parameters.
7. Multiply by the out‑gate, then a final linear projection back to
   the original feature dimension.
8. Cast the result back to float32.

The kernel fuses the mask & gate multiplication (two launches) and
combines the five initial linear layers into one, reducing both compute
and memory traffic. All heavy matrix‑multiplications are done in FP16
to exploit Tensor‑Cores on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# mask shape: (B, N, N) → broadcast across hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated (proj)
    gate_ptr,              # *fp16 : gate tensor (sigmoid)
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate
    hidden_dim: tl.int32,  # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,   # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear index
    mask = offs < total_elts

    # each hidden_dim entries share the same mask value
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM helper (uses cuBLAS / Tensor‑Core)
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using torch.bmm (batched half‑precision GEMM).

    left, right : [B, N, N, H] (FP16)
    Returns      : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]; treat each hidden slice as a batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))            # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    out = torch.nn.functional.linear(out_hidden, to_out_w)  # fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the test harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
