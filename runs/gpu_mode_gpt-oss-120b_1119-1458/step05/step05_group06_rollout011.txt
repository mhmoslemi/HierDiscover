```python
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # sort out dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # 1. First LayerNorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # fp32

    # 2. Cast to fp16 for projection
    x_fp16 = x.to(torch.float16)

    # 3. fuse linear transforms
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # concatenate
    fused_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0).to(torch.float16) # shape [5*hidden_dim, dim]

    # fused linear
    fused = torch.nn.functional.linear(x_fp16, fused_w)  # shape [B,N,N,5*hidden_dim]

    # split
    splits = torch.split(fused, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = splits

    # gating
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # apply gates to proj
    left = left_proj * left_gate
    right = right_proj * right_gate

    # mask
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # contraction
    seq_len = input_tensor.shape[1]
    # optionally choose kernel; set threshold 256 perhaps to use triton for all
    if seq_len <= 512:
        contracted = _batched_matmul_triton(left, right)  # fp16
    else:
        contracted = _batched_matmul_torch(left, right)  # fp16

    # second layernorm in fp32
    # cast contracted to fp32
    contracted_fp32 = contracted.to(torch.float32)
    # to_out_norm weight and bias are fp32 as loaded
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    contracted_fp32 = torch.nn.functional.layer_norm(contracted_fp32, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out_gate scaling
    out_gate_fp32 = out_gate.to(torch.float32)
    contracted_fp32 = contracted_fp32 * out_gate_fp32

    # final projection
    to_out_w = weights["to_out.weight"].to(torch.float16)
    # cast to fp16 for linear
    out_fp16 = torch.nn.functional.linear(contracted_fp32.to(torch.float16), to_out_w)

    # cast back to float32
    out = out_fp16.to(torch.float32)
    return out
```

"""Outgoing TriMul (AlphaFold‑3) – highly‑fused forward pass.

Algorithm
---------
1️⃣  First LayerNorm in float32 for numerical stability.
2️⃣  Cast to fp16 and fuse **all** linear projections (left_proj,
    right_proj, left_gate, right_gate, out_gate) into a single
    matrix‑multiply using a concatenated weight matrix.  This reduces
    the number of passes over the large `[B, N, N, dim]` tensor from
    five to one.
3️⃣  Apply sigmoid gates, optional mask and obtain the gated tensors
    `left` and `right` (shape `[B, N, N, hidden_dim]`).
4️⃣  Heavy O(N³) contraction   
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  
    is performed as a **batched GEMM** `C = A @ Bᵀ` where each hidden
    channel `h` (and each batch) is an independent GEMM.  For
    sequence lengths ≤ 512 a custom Triton kernel computes the GEMM
    with fp16·fp16 → fp32 accumulation (Tensor‑Core friendly).  For
    longer sequences we fall back to cuBLAS via `torch.bmm`, which is
    optimal on H100 for large matrices.
5️⃣  Second LayerNorm (float32) on the contracted tensor, followed by
    the out‑gate scaling.
6️⃣  Final linear projection back to the original `dim` (fp16) and
    cast the result to float32.

All heavy arithmetic runs in fp16, while LayerNorms run in fp32 for
stability.  The implementation respects the required signature
`custom_kernel(data)` and works for the full set of test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton batched GEMM kernel: C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile along output row   (i)
    pid_n = tl.program_id(1)   # tile along output col   (j)
    pid_z = tl.program_id(2)   # batch * hidden index   (M)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---------------------------------------------------------
        # Load A[i, k]  (shape BLOCK_M x BLOCK_K)
        # ---------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=(mask_i[:, None]) & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ---------------------------------------------------------
        # Load Bᵀ[j, k]  -> B loaded as (k, j) for the transpose
        # ---------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (mask_j[None, :]),
            other=0.0,
        )   # fp16

        # mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Batched contraction with a pure Triton GEMM:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    left/right are FP16 tensors of shape [B, N, N, H].
    Returns a FP16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B * H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()    # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()   # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()    # (N*N, N, 1)

    # Tile size – 128×128 works well for N up to 512, 64×64 for larger
    # matrices on H100.  Empirically 128 gives the best occupancy for
    # N ≤ 512, while 64 is safer for N = 1024.
    if N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 64
        BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """Fallback contraction using cuBLAS (torch.bmm)."""
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – may be all‑ones
        - weights_dict : mapping of weight name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack & config
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the remaining pipeline
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fuse all five per‑position linear projections into one
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]      # [hidden, dim]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # concatenate along the out‑feature dimension: shape (5*hidden_dim, dim)
    fused_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)  # FP16 for Tensor‑Core friendliness

    # single linear pass (A @ Wᵀ) => shape [B, N, N, 5*hidden_dim]
    fused = F.linear(x_fp16, fused_weight)

    # ------------------------------------------------------------------
    # 5️⃣ split & apply sigmoid gates
    # ------------------------------------------------------------------
    splits = torch.split(fused, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = splits

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 6️⃣ gated projections (fp16) – optional masking
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    # For medium sizes Triton is faster; for very large N cuBLAS wins.
    if seq_len <= 512:
        contracted = _batched_matmul_triton(left, right)      # fp16
    else:
        contracted = _batched_matmul_torch(left, right)       # fp16

    # ------------------------------------------------------------------
    # 8️⃣ second LayerNorm (float32) → out‑gate → final projection
    # ------------------------------------------------------------------
    # Cast to float32 for the LayerNorm to retain the original model’s
    # numerical stability.
    contracted_f32 = contracted.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"]   # float32
    to_out_norm_b = weights["to_out_norm.bias"]

    contracted_f32 = torch.nn.functional.layer_norm(
        contracted_f32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # out_gate is still fp16 – promote to fp32 for the scaling.
    out_gate_f32 = out_gate.to(torch.float32)
    contracted_f32 = contracted_f32 * out_gate_f32

    # final projection back to `dim` (still in fp16 for speed)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # [dim, hidden_dim]
    out_fp16 = F.linear(contracted_f32.to(torch.float16), to_out_w)  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ cast result back to float32 as required by downstream code
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    return out
