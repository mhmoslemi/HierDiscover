"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) → cast to FP16.
2. Linear projections (left/right) and three gating streams are computed in FP16.
3. If a mask is present we fuse the three element‑wise ops
       (proj * gate * mask)
   with a tiny Triton kernel (`mask_gate_fuse_kernel`).  The mask is
   broadcast over the hidden dimension, so only a single kernel launch
   per tensor is needed.
4. The heavy contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM: for each head `h` we compute
   `C = A @ Bᵀ` with `torch.bmm`, which uses cuBLAS Tensor‑Core kernels.
5. Second Layer‑norm (FP16) → out‑gate (sigmoid) → final linear projection
   back to the original channel dimension.
6. Cast the result back to FP32 (the harness expects FP32).

Only the forward pass is required; gradients are not computed.
A Triton kernel is used for the mask‑gate fusion.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over the
# hidden dimension.  Operates on a flat view of a contiguous tensor
# of shape [B, N, N, hidden_dim] (size = B·N·N·hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, size = total_elements
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor of shape (B·N·N,) – broadcasted
    total_elements,  # i32 : total number of elements in x
    hidden_dim,      # i32 : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask_bounds = offs < total_elements        # guard for the tail of the tensor

    # Load the tensors
    x   = tl.load(x_ptr + offs, mask=mask_bounds, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask_bounds, other=0.0)

    # Every `hidden_dim` consecutive elements share the same mask entry.
    # Compute the mask index by integer division.
    mask_idx = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_idx, mask=mask_bounds, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store the result back in‑place
    tl.store(x_ptr + offs, out, mask=mask_bounds)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using cuBLAS.
# left/right are FP16 tensors of shape [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via
    a batched matrix multiplication (A @ Bᵀ) on FP16 data.
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] and perform A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # bmm does (B*H, N, N) @ (B*H, N, N).T → (B*H, N, N)
    # cuBLAS will accumulate in FP32 internally for FP16 inputs.
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------

        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed after fusion
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction (batched GEMM) – cuBLAS implementation
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast result back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```
