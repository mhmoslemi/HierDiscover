```python
@triton.jit
def trig_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)  # tile row i dimension
    pid_j = tl.program_id(1)  # tile col j dimension
    pid_bh = tl.program_id(2) # batch/hidden index

    bh = pid_bh
    b = bh // hidden
    h = bh % hidden

    # compute offsets for this bh
    off_batch_left = b * stride_left_batch + h * stride_left_h
    off_batch_right = b * stride_right_batch + h * stride_right_h
    off_batch_out = b * stride_out_batch + h * stride_out_h

    # starting i and j
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # create offsets within block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bound rows and cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_s in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offset for this k tile
        k_start = k_s * BLOCK_K
        # offsets for k inside tile
        k_offs = k_start + offs_k

        # mask for valid k
        mask_k = k_offs < N

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + off_batch_left + offs_i[:, None] * stride_left_i + (k_offs[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B_T tile: shape (BLOCK_K, BLOCK_N) - need to load B[j,k] = right[j,k] using k as row.
        b_ptrs = right_ptr + off_batch_right + (k_offs[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product over K dimension
        # accumulate: for each i in M, j in N, we want sum_k a[i,k] * b[k,j]
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # write back result to out_fp32? out will store FP16, we need final output in fp16 for downstream
    # convert acc to fp16 possibly after later normalization
    out_tile = acc.to(tl.float16)
    c_ptrs = out_ptr + off_batch_out + offs_i[:, None] * stride_out_i + (offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs, out_tile, mask=mask_i[:, None] & mask_j[None, :])
```

```python
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # LayerNorm (FP32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)  # now fp16

    # Convert weight tensors to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_w)   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w) # shape [B,N,N,hidden]

    # Optional mask multiplication (in-place)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left.mul_(mask_fp16)   # in-place
        right.mul_(mask_fp16)

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))  # [B,N,N,hidden]
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gating via Triton pointwise multiplication
    # Ensure contiguous contiguous before kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _pointwise_mul_inplace(left, left_gate)
    _pointwise_mul_inplace(right, right_gate)

    # Batched contraction using Triton
    B, N, _, H = left.shape
    # allocate out
    out = torch.empty_like(left)  # shape [B,N,N,H] fp16

    # Strides
    stride_l_batch = left.stride(0)
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    stride_l_h = left.stride(3)

    stride_r_batch = right.stride(0)
    stride_r_i = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)   # j dimension stride
    stride_out_h = out.stride(3)

    # define tile sizes
    BLOCK_M = 256
    BLOCK_N = 256
    BLOCK_K = 32

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B * H)
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_batch,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,

    )

    out = torch.nn.functional.linear(out, to_out_w)     # shape [B,N,N,dim] fp16
```

```python
@triton.jit
def _pointwise_mul_kernel(out_ptr, mul_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)
```

```python
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)   # tile row index
    pid_j = tl.program_id(1)   # tile col index
    pid_bh = tl.program_id(2)  # batch * hidden

    # compute batch and hidden index
    bh = pid_bh
    hidden_dim = H
    b = bh // hidden_dim
    h = bh % hidden_dim

    # base offsets for this batch+head
    off_batch_left = b * stride_l_batch + h * stride_l_h
    off_batch_right = b * stride_r_batch + h * stride_r_h
    off_batch_out = b * stride_out_batch + h * stride_out_h

    # compute start indices for block
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets within block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds check masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_s in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_s * BLOCK_K
        k_offs = k_start + offs_k
        mask_k = k_offs < N

        # Load A tile (left)
        a_ptrs = left_ptr + off_batch_left + offs_i[:, None] * stride_l_i + k_offs[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B_T tile (right) - using original right layout (right[j,k])
        b_ptrs = right_ptr + off_batch_right + k_offs[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(a, b)   # shape (BLOCK_M, BLOCK_N)

    # Write back result
    out_tile = acc.to(tl.float16)
    c_ptrs = out_ptr + off_batch_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, out_tile, mask=mask_i[:, None] & mask_j[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑fused implementation
-----------------------------------------------------------------
Algorithm
1️⃣  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣  Linear projections (left/right) and three gating projections (all FP16).
3️⃣  In‑place mask multiplication (if required) and a tiny Triton kernel that
    fuses the element‑wise left‑gate/right‑gate multiplication.
4️⃣  Heavy cubic contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    is performed by a custom Triton batched GEMM kernel that works on the
    original `[B,N,N,D]` layout, thus avoiding the costly `permute+contiguous`
    copies of the reference implementation.
5️⃣  Second LayerNorm over the hidden dimension, out‑gate (sigmoid) and the
    final linear projection back to `dim`.
6️⃣  Cast the result back to FP32 (as expected by the harness).

All heavy compute (steps 3‑4) is executed on the GPU with fused kernels,
while the lighter linear / norm ops use PyTorch’s highly‑optimized kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: point‑wise in‑place multiplication (x ← x * y)
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    out_ptr,        # *fp16  – input & output
    mul_ptr,        # *fp16  – multiplier
    N,              # total number of elements (int64)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _pointwise_mul_inplace(x: torch.Tensor, y: torch.Tensor) -> None:
    """In‑place element‑wise multiplication `x ← x * y` (FP16 tensors)."""
    assert x.is_contiguous() and y.is_contiguous()
    total = x.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        x,
        y,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

# Triton kernel: batched triangular multiplicative update
#   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
#   Input & output layout: [B, N, N, D] (row‑major, D is fastest)
# ----------------------------------------------------------------------
@triton.jit

    left_ptr,          # *fp16  left tensor  [B,N,N,D]
    right_ptr,         # *fp16  right tensor [B,N,N,D]

    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_i = tl.program_id(0)      # tile row (i)
    pid_j = tl.program_id(1)      # tile col (j)
    pid_bh = tl.program_id(2)     # combined batch * hidden index

    # Recover batch and hidden index
    bh = pid_bh
    b = bh // D                   # batch
    d = bh % D                    # hidden channel

    # Base offsets for this (b,d) slice
    off_l = b * stride_l_batch + d * stride_l_d
    off_r = b * stride_r_batch + d * stride_r_d
    off_o = b * stride_o_batch + d * stride_o_d

    # Start indices for this tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Offsets inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Bounds masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        k_offs = k0 + offs_k
        mask_k = k_offs < N

        # Load A_tile = left[b,i,k,d]   shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr + off_l + offs_i[:, None] * stride_l_i + k_offs[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_tile = right[b,j,k,d]   (need the transposed view for dot)
        #   We load right[j,k] and treat rows as k, cols as j → B_T shape (BLOCK_K, BLOCK_N)
        b_ptr = right_ptr + off_r + k_offs[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate the matrix product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write back the tile (convert back to FP16)
    out_tile = acc.to(tl.float16)
    c_ptr = out_ptr + off_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptr, out_tile, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float32)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for stability) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)                     # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ FP16 model weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (in‑place)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left.mul_(mask_fp16)
        right.mul_(mask_fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Gating values (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Fuse gating (left/right) using Triton point‑wise kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _pointwise_mul_inplace(left, left_gate)    # left ← left * left_gate
    _pointwise_mul_inplace(right, right_gate)  # right ← right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – custom Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, D = left.shape                     # D == hidden_dim
    # allocate output tensor
    out = torch.empty_like(left)                # [B,N,N,D]  fp16

    # Strides (in elements)
    sL_b, sL_i, sL_k, sL_d = left.stride()
    sR_b, sR_i, sR_k, sR_d = right.stride()
    sO_b, sO_i, sO_j, sO_d = out.stride()

    # Tile sizes (empirically good for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * D,                           # batch * hidden
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        D,
        sL_b,
        sL_i,
        sL_k,
        sL_d,
        sR_b,
        sR_i,
        sR_k,
        sR_d,
        sO_b,
        sO_i,
        sO_j,
        sO_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                     # a few warps give good occupancy on H100
    )

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)               # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
