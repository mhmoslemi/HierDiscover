    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bi, stride_bk,
    stride_cb, stride_ci, stride_cj,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden indices
    b_idx = batch_id // H
    h_idx = batch_id % H

    # Base pointers
    a_base = a_ptr + b_idx * stride_ab + h_idx
    b_base = b_ptr + b_idx * stride_bb + h_idx
    c_base = c_ptr + b_idx * stride_cb + h_idx

    # Bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Compute batch index
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets in the 2‑D matrix tiling
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_base = a_ptr + batch_id * stride_ab
    b_base = b_ptr + batch_id * stride_bb
    c_base = c_ptr + batch_id * stride_cb

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        start_k = k * BLOCK_K
        # Load A block
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + (start_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & ((start_k + tl.arange(0, BLOCK_K)) < N)),
                    other=0.0).to(tl.float32)

        # Load B block (transposed)
        b_ptrs = b_base + ((start_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=(((start_k + tl.arange(0, BLOCK_K)) < N)[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write out the result
    c_ptrs = c_base + (offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul “outgoing” operator (AlphaFold‑3) – highly‑optimized forward pass.

Algorithm Overview
------------------
1. **First LayerNorm** (float32) – stabilises the input.
2. **Cast to FP16** – everything from this point is FP16 (Tensor‑core friendly).
3. **Fused projection + gating** – a single linear layer produces
   ``left_proj, left_gate, right_proj, right_gate`` for all hidden
   channels (4·hidden_dim outputs).  The gating vectors are passed
   through a sigmoid and multiplied into the projections.  If a mask is
   supplied it is broadcast‑multiplied afterwards.
4. **“Outgoing” contraction** – for each hidden slice we need
   ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]``.
   This is a batched GEMM ``C = A @ Bᵀ`` where
   ``A = left`` and ``B = right`` have shape
   ``(B·hidden_dim, N, N)``.  The heavy O(N³) work is performed by a
   custom Triton kernel that loads tiles of ``A`` and ``Bᵀ`` and
   accumulates in FP32 (`tl.dot`).  The kernel works on the
   *contiguous* layout obtained by ``permute(0,3,1,2).contiguous()``.
5. **Second LayerNorm** (FP16) over the hidden dimension.
6. **Out‑gate & final projection** – element‑wise sigmoid gate
   followed by a linear layer back to ``dim``.
7. **Cast back to float32** for the harness.

All heavy numeric work (batch GEMM) is done in Triton; the rest uses
high‑performance PyTorch kernels.  The implementation avoids unnecessary
memory copies and reduces the number of separate linear calls by fusing
the left/right projections with their gates.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
#   C[b,i,j] = Σₖ A[b,i,k] * B[b,j,k]    (B = batch*hidden_dim)
# The input matrices A and B are stored in FP16, row‑major.
# B is accessed in transposed form (Bᵀ) by swapping the strides for the
# K (contracting) and N (output‑column) dimensions.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # *fp16 pointers
    N,                                   # int, size of each square matrix
    stride_ab, stride_ai, stride_ak,    # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,    # B (used as transposed): batch, row‑K, col‑N strides
    stride_cb, stride_ci, stride_cj,    # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the current batch (batch = original batch * hidden_dim)
    a_base = a_ptr + batch_id * stride_ab
    b_base = b_ptr + batch_id * stride_bb
    c_base = c_ptr + batch_id * stride_cb

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the contraction dimension K (= N)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # ----- Load A block (row‑major) ---------------------------------
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + (k_start + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & ((k_start + tl.arange(0, BLOCK_K)) < N)),
                    other=0.0).to(tl.float32)

        # ----- Load Bᵀ block (by swapping strides) ----------------------
        # Bᵀ row = original column (k), Bᵀ col = original row (j)
        b_ptrs = b_base + ((k_start + tl.arange(0, BLOCK_K))[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=(((k_start + tl.arange(0, BLOCK_K)) < N)[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # ----- Accumulate ------------------------------------------------
        acc += tl.dot(a, b)          # FP16 inputs, FP32 accumulator

    # ----- Write the result ------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    left, right: tensors of shape [B, H, N, N] (contiguous, FP16)
    Returns tensor of the same shape containing left @ rightᵀ per (B,H).
    """
    B, H, N, _ = left.shape
    total_batch = B * H

    # Flatten the leading batch dimensions so the kernel sees [total_batch, N, N]
    left_mat = left.view(total_batch, N, N)
    right_mat = right.view(total_batch, N, N)
    out_mat = torch.empty_like(left_mat)

    # Strides expressed in element counts (not bytes)
    stride_ab = left_mat.stride(0)   # N*N
    stride_ai = left_mat.stride(1)   # N
    stride_ak = left_mat.stride(2)   # 1

    stride_bb = right_mat.stride(0)  # N*N
    # For Bᵀ we need row‑stride = original col‑stride (1) and col‑stride = original row‑stride (N)
    stride_bk = right_mat.stride(2)  # 1
    stride_bn = right_mat.stride(1)  # N

    stride_cb = out_mat.stride(0)    # N*N
    stride_ci = out_mat.stride(1)    # N
    stride_cj = out_mat.stride(2)    # 1

    # Choose tile sizes – 64 is a good trade‑off for H100 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        N=N,
        stride_ab=stride_ab,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_bb=stride_bb,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cb=stride_cb,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )
    # Restore the original (B, H, N, N) shape
    return out_mat.view(B, H, N, N)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul block (no gradients).

    Parameters
    ----------
    data
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – may be omitted
        - weights_dict : mapping of weight names → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )                     # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(fp16)        # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Prepare FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse left/right projections with their gates (single Linear)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices → shape (4*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )                     # (4*H, dim)

    # One fused linear produces all four tensors at once
    combined = F.linear(x, combined_weight)                     # [B, N, N, 4*H]

    h = hidden_dim
    left_proj = combined[..., :h]                               # [B, N, N, H]
    left_gate = torch.sigmoid(combined[..., h:2 * h])          # sigmoid
    right_proj = combined[..., 2 * h:3 * h]
    right_gate = torch.sigmoid(combined[..., 3 * h:4 * h])

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(fp16).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Compute out‑gate (separate Linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Rearrange for the batched GEMM: (B, H, N, N) contiguous
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]

    # ------------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via the custom Triton kernel
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left_perm, right_perm)        # [B, H, N, N] (fp16)

    # Return to the original layout expected by the rest of the model
    out = out.permute(0, 2, 3, 1).contiguous()                # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 🔟 Apply final out‑gate and project back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate                              # element‑wise (fp16)
    out = F.linear(out, to_out_w)                     # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to float32 (the harness expects fp32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
