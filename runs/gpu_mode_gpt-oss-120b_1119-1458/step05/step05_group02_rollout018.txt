def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # 1. layernorm (float32)
    norm_w = weights["norm.weight"]; norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_w, norm_b)
    x_fp16 = x.to(torch.float16)

    # 2. combined linear for left & right proj+gate
    # combine weights
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    # each shape (hidden_dim, dim)
    combined_linear_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0).to(torch.float16)  # shape (4*hidden_dim, dim)

    # Linear
    combined = torch.nn.functional.linear(x_fp16, combined_linear_w)  # shape (B,N,N,4*hidden_dim) fp16

    # slice
    left_proj = combined[..., 0:hidden_dim]  # shape (B,N,N,hidden_dim)
    left_gate = combined[..., hidden_dim:2*hidden_dim]
    right_proj = combined[..., 2*hidden_dim:3*hidden_dim]
    right_gate = combined[..., 3*hidden_dim:4*hidden_dim]

    # gate activations
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # 3. mask and gating
    if not nomask:
       mask_fp16 = mask_tensor.to(torch.float16).contiguous()
       total = left_proj.numel()
       BLOCK = 1024
       grid = ( (total + BLOCK - 1) // BLOCK, )
       # left = left_proj * left_gate * mask
       mask_gate_fuse_kernel[grid](
           left_proj,
           left_gate,
           mask_fp16,
           hidden_dim,
           total,
           BLOCK=BLOCK,
           num_warps=4,
       )
       mask_gate_fuse_kernel[grid](
           right_proj,
           right_gate,
           mask_fp16,
           hidden_dim,
           total,
           BLOCK=BLOCK,
           num_warps=4,
       )
    else:
       left_proj = left_proj * left_gate
       right_proj = right_proj * right_gate

    # At this point left and right are final left and right (named left_final and right_final)
    left = left_proj
    right = right_proj

    # 4. batched matmul
    out = _tri_mul_outgoing(left, right)  # implemented as custom Triton kernel

    # 5. second layernorm
    to_out_norm_w = weights["to_out_norm.weight"]; to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # 6. out_gate
    out_gate_w = weights["out_gate.weight"]
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w.to(torch.float16))
    out_gate = torch.sigmoid(out_gate)

    out_norm = out_norm * out_gate

    # 7. final linear
    to_out_w = weights["to_out.weight"]
    final_out = torch.nn.functional.linear(out_norm, to_out_w.to(torch.float16))

    # 8. cast to float32
    return final_out.to(torch.float32)

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, hidden_dim,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)   # combined B * hidden_dim
    pid_m = tl.program_id(1)    # row tile
    pid_n = tl.program_id(2)    # column tile

    # Compute batch and hidden indices
    bh = pid_bh
    # Division
    b = bh // hidden_dim
    h = bh - b * hidden_dim   # avoid modulo maybe faster (works)
    # base offset for this batch/h
    # offset = batch_offset + h
    base_a = a_ptr + b * stride_a_batch + h
    base_b = b_ptr + b * stride_b_batch + h
    base_c = c_ptr + b * stride_c_batch + h

    # Offsets for i (M) and j (N)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bound rows and columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        # k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers for A and B tiles
        # A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_a + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        # B: shape (BLOCK_K, BLOCK_N) (since B needs to be transposed)
        # We need B element: right[j,k] => row j dim i, col k dimension is column.
        # But we want B_T where rows = k, cols = j.
        b_ptrs = base_b + offs_n[None, :] * stride_b_i + offs_k[:, None] * stride_b_k

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)

    # Write back C
    c_ptrs = base_c + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input (float32) → fp16.
2️⃣ Fuse the four 1×1 linear projections (left_proj, left_gate,
   right_proj, right_gate) into **two** larger linear layers:
      *W_left* = [left_proj; left_gate]   (2·H × D)
      *W_right* = [right_proj; right_gate] (2·H × D)
   This halves the number of passes over the huge [B,N,N,D] tensor.
3️⃣ Split the result, apply sigmoid to the gate halves and (optionally)
   fuse mask·gate·proj elementwise with a tiny Triton kernel.
4️⃣ Heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is carried out by a custom Triton batched GEMM that works directly on
   the original strided layout (no permute‑and‑contiguous copy).  The
   kernel loads tiles of size 128×128×32 (M,N,K) and uses Tensor‑Core
   `tl.dot`, accumulating in fp32.
5️⃣ Second Layer‑Norm (fp16) → out_gate (sigmoid) → elementwise gating.
6️⃣ Final projection back to the original dimension (fp16) and cast to
   float32 for the harness.

Only the forward pass is implemented – no autograd bookkeeping.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
#  Mask‑gate fused multiplication (broadcast mask over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (flattened B·N·N)
    hidden_dim,     # i32   : hidden dimension H
    size,           # i32   : total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices of this block
    active = offs < size                               # bounds guard

    # mask index = element index // H  (mask is broadcast over H)
    mask_offs = offs // hidden_dim

    x = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
#  Batched GEMM for the O(N³) contraction.
#  Operates on tensors with layout (B, N, N, H) where H is the innermost
#  dimension.  No extra copies – we work directly on the original strides.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                       # *fp16
    B, N, H,                                    # i32
    stride_a_batch, stride_a_i, stride_a_k,     # i32
    stride_b_batch, stride_b_i, stride_b_k,     # i32
    stride_c_batch, stride_c_i, stride_c_j,     # i32
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)          # 0 … B*H‑1   (combined batch·hidden)
    pid_m = tl.program_id(1)              # row tile   (i‑index)
    pid_n = tl.program_id(2)              # col tile   (j‑index)

    # Decode batch and hidden indices from the combined ID
    bh = pid_batch
    b = bh // H                           # batch index 0 … B‑1
    h = bh - b * H                         # hidden index 0 … H‑1

    # Base pointer for this (b,h) matrix
    a_base = a_ptr + b * stride_a_batch + h
    b_base = b_ptr + b * stride_b_batch + h
    c_base = c_ptr + b * stride_c_batch + h

    # Row / column offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds masks for rows and columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the summed dimension)
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        mask_a = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (right matrix is used transposed: we need B_T[k, j])
        b_ptrs = b_base + offs_n[None, :] * stride_b_i + offs_k[:, None] * stride_b_k
        mask_b = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Matrix‑multiply‑accumulate (Tensor‑Core if possible)
        acc += tl.dot(a, b)

    # Write out the result tile (FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _tri_mul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    using the Triton kernel above.  All tensors must be fp16 and on the same
    CUDA device.
    """
    B, N, _, H = left.shape
    # allocate output
    out = torch.empty_like(left)

    # strides (in elements) for the three tensors
    a_batch, a_i, a_k, a_h = left.stride()
    b_batch, b_i, b_k, b_h = right.stride()
    c_batch, c_i, c_j, c_h = out.stride()

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        a_batch, a_i, a_k,
        b_batch, b_i, b_k,
        c_batch, c_i, c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # H100 prefers many warps per block
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (AlphaFold‑3) – no autograd.

    Arguments
    ---------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : [B, N, N, dim]  (float32, CUDA)
        mask_tensor  : [B, N, N]      (float32/bool)  – can be None
        weights_dict : model weights (float32)
        config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                       ``nomask`` (True ⇒ ignore mask)

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – performed on the raw input.
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                       # [B, N, N, dim]  float32
    x = x.to(torch.float16)  # work in fp16 from now on

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the four 1×1 projections into two large linear layers.
    # -----------------------------------------------------------------
    # Concatenate weights for (proj + gate) for left and right sides.
    left_proj_w = weights["left_proj.weight"]      # (hidden, dim)
    left_gate_w = weights["left_gate.weight"]      # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]

    # Shape: (4*hidden, dim)
    fused_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w],
        dim=0,
    ).to(torch.float16)

    # One combined linear pass (cost: B·N·N × dim × (4·hidden))
    fused = torch.nn.functional.linear(x, fused_w)   # [B,N,N,4*hidden] fp16

    # Split the concatenated result
    h = hidden_dim
    left_proj = fused[..., 0:h]               # [B,N,N,h]
    left_gate = fused[..., h:2 * h]           # [B,N,N,h]
    right_proj = fused[..., 2 * h:3 * h]      # [B,N,N,h]
    right_gate = fused[..., 3 * h:4 * h]      # [B,N,N,h]

    # Gate activations (sigmoid)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # -----------------------------------------------------------------
    # 3️⃣ Mask & gate fusion (optional)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total = left_proj.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)
        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # No mask → simple element‑wise multiplication.
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via custom Triton GEMM.
    # -----------------------------------------------------------------
    out = _tri_mul_outgoing(left, right)      # [B,N,N,hidden] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (hidden dim) – fp16.
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Output gate (sigmoid) and final projection.
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden]

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 for the harness.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
