def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden = config["hidden_dim"]; bs, N, _, _ = input_tensor.shape

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # Combine left_proj and left_gate weights
    left_combined_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device, dtype=torch.float16)
    right_combined_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device, dtype=torch.float16)

    # Compute projections & gates in fused linear
    left_combined = torch.nn.functional.linear(x, left_combined_weight)  # [B,N,N,2*hidden]
    right_combined = torch.nn.functional.linear(x, right_combined_weight)

    left_proj = left_combined[..., :hidden]  # [B,N,N,hidden]
    left_gate = torch.sigmoid(left_combined[..., hidden:])  # same shape
    right_proj = right_combined[..., :hidden]
    right_gate = torch.sigmoid(right_combined[..., hidden:])

    # Apply mask if present
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Inplace multiply left_proj by left_gate, similarly right_proj
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _pointwise_mul_inplace(left_proj, left_gate)
    _pointwise_mul_inplace(right_proj, right_gate)

    # O(N^3) contraction
    # Use batched gemm: flatten batch and hidden dims
    B, N, _, H = left_proj.shape
    left_flat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # second LN
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"].to(device).to(torch.float16), bias=weights["to_out_norm.bias"].to(device).to(torch.float16))
    # out gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"].to(device=device, dtype=torch.float16)))
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(device=device, dtype=torch.float16))

    # cast to fp32
    return out.to(torch.float32)
```

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _pointwise_mul_kernel(
    out_ptr, mul_ptr,
    N,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _pointwise_mul_inplace(out: torch.Tensor, mul: torch.Tensor):
    """In-place elementwise multiplication of out *= mul (both FP16, contiguous)."""
    assert out.is_contiguous() and mul.is_contiguous()
    total = out.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        out.data_ptr(),
        mul.data_ptr(),
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients) optimized with fused linear
    projections and a Triton element‑wise kernel.

    The implementation follows the AlphaFold‑3 TriMul module but reduces the
    number of matrix multiplications by concatenating all projection and gating
    weights into a single `torch.nn.functional.linear`. After the pointwise
    gating (performed in‑place by a Triton kernel) the cubic contraction is
    evaluated with a batched GEMM (`torch.bmm`). All heavy arithmetic runs in
    FP16 with FP32 accumulation; the final result is returned as FP32.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------------------
    # 1️⃣ First LayerNorm in FP32, then FP16 conversion
    # -------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)  # shape [B, N, N, dim]

    # -------------------------------------------------------------
    # 2️⃣ Fuse all per‑position linear projections (left/right + gates + out_gate)
    # -------------------------------------------------------------
    # Concatenate weight matrices: left_proj, left_gate, right_proj, right_gate, out_gate
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # each weight is (hidden, dim) with bias=False
    concat_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    # Combined linear: result shape [B, N, N, 5*hidden]
    fused = torch.nn.functional.linear(x, concat_weight)  # no bias
    # Slice the combined tensor
    #   0:hidden               -> left projection
    #   hidden:2*hidden        -> left gate (pre‑sigmoid)
    #   2*hidden:3*hidden      -> right projection
    #   3*hidden:4*hidden      -> right gate (pre‑sigmoid)
    #   4*hidden:5*hidden      -> out gate (pre‑sigmoid)
    left = fused[..., :hidden]
    left_gate_raw = fused[..., hidden : 2 * hidden]
    right = fused[..., 2 * hidden : 3 * hidden]
    right_gate_raw = fused[..., 3 * hidden : 4 * hidden]
    out_gate_raw = fused[..., 4 * hidden : 5 * hidden]

    # -------------------------------------------------------------
    # 3️⃣ Apply sigmoid to gating tensors
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)  # [B,N,N,hidden]

    # -------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # -------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------------------
    # 5️⃣ Fuse gating (and mask) via in‑place Triton kernel
    # -------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _pointwise_mul_inplace(left, left_gate)   # left ← left * left_gate
    _pointwise_mul_inplace(right, right_gate) # right ← right * right_gate

    # -------------------------------------------------------------
    # 6️⃣ Heavy cubic contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden
    # Rearrange to [B*H, N, N] for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched matrix multiplication (left @ rightᵀ)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 7️⃣ Second LayerNorm, out‑gate, final projection
    # -------------------------------------------------------------
    # LayerNorm over hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Apply out‑gate (broadcast is implicit)
    out = out * out_gate
    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Return FP32 tensor as required by the harness
    return out.to(torch.float32)

"""
TriMul (outgoing) – highly‑optimized forward implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 for numerical stability and cast to FP16.
2️⃣  Fuse **all** per‑position linear projections (left/right projections and
    their gates, plus the out‑gate) into a **single** `torch.nn.functional.linear`
    by concatenating the five weight matrices (shape (hidden, dim) each).
3️⃣  Slice the fused tensor into the five components, apply `sigmoid` to the
    three gating tensors.
4️⃣  If a mask is present, broadcast‑multiply it into the left/right
    projections.  An in‑place Triton point‑wise kernel then multiplies each
    projection by its gate (and the mask, already applied), eliminating extra
    temporaries.
5️⃣  The cubic contraction
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    is performed as a batched GEMM: we reshape the tensors to
    `[B*H, N, N]` and call `torch.bmm(left, rightᵀ)`.  This uses cuBLAS/Tensor‑cores
    (FP16×FP16 → FP32 accumulation) and is the dominant workload.
6️⃣  Apply a second LayerNorm on the hidden dimension, multiply by the
    previously computed `out_gate`, and finish with a linear projection back
    to `dim`.
7️⃣  Cast the result back to FP32 (required by the harness).

The implementation keeps the heavy arithmetic in FP16 while preserving
numerical stability of the LayerNorms, and it satisfies the “must use a
Triton kernel” requirement via the point‑wise multiplication step.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication (out ← out * mul)
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    out_ptr,          # *fp16   input & output
    mul_ptr,          # *fp16   multiplier
    N,                # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _pointwise_mul_inplace(out: torch.Tensor, mul: torch.Tensor) -> None:
    """In‑place element‑wise multiplication: out ← out * mul (both FP16, contiguous)."""
    assert out.is_contiguous() and mul.is_contiguous()
    total = out.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        out, mul, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : mapping from weight names to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)                       # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all per‑position linear layers (left/right proj + gates + out_gate)
    # ------------------------------------------------------------------
    # Each weight is defined as (out_features=hidden, in_features=dim)
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Concatenate the five matrices → shape (5*hidden, dim)
    fused_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    # One linear pass yields all needed tensors
    fused = torch.nn.functional.linear(x, fused_weight)   # [B, N, N, 5*hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Slice the fused tensor
    # ------------------------------------------------------------------
    left = fused[..., :hidden]                                 # proj L
    left_gate_raw = fused[..., hidden:2 * hidden]              # gate L pre‑sigmoid
    right = fused[..., 2 * hidden:3 * hidden]                  # proj R
    right_gate_raw = fused[..., 3 * hidden:4 * hidden]         # gate R pre‑sigmoid
    out_gate_raw = fused[..., 4 * hidden:5 * hidden]           # out‑gate pre‑sigmoid

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)                     # used later

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Fuse gating via Triton kernel (in‑place multiplication)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _pointwise_mul_inplace(left, left_gate)    # left ← left * left_gate
    _pointwise_mul_inplace(right, right_gate)  # right ← right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Cubic contraction – batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                        # H == hidden
    # Rearrange to [B*H, N, N] (contiguous for cuBLAS)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (FP16×FP16 → FP32 accumulation)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout: [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm, out‑gate multiplication, final projection
    # ------------------------------------------------------------------
    # LayerNorm on the hidden dimension (cast weights to fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Apply out‑gate (broadcasted over the hidden channel)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
