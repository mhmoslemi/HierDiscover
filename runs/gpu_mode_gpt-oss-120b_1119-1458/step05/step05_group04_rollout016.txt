```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Unpack weights
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    device = input_tensor.device

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # cast weights to fp16
    left_proj_w = left_proj_w.to(device=device, dtype=torch.float16)
    right_proj_w = right_proj_w.to(device=device, dtype=torch.float16)
    left_gate_w = left_gate_w.to(device=device, dtype=torch.float16)
    right_gate_w = right_gate_w.to(device=device, dtype=torch.float16)
    out_gate_w = out_gate_w.to(device=device, dtype=torch.float16)
    to_out_norm_w = to_out_norm_w.to(device=device, dtype=torch.float16)
    to_out_norm_b = to_out_norm_b.to(device=device, dtype=torch.float16)
    to_out_w = to_out_w.to(device=device, dtype=torch.float16)

    # 2. Stacked linear for left_proj, right_proj, left_gate, right_gate
    combined_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0)  # [4*hidden, dim]
    proj = torch.nn.functional.linear(x, combined_weight)  # shape [B,N,N,4*hidden]

    # slice
    hidden = hidden_dim
    left = proj[..., 0:hidden]
    right = proj[..., hidden:2*hidden]
    left_gate = proj[..., 2*hidden:3*hidden]
    right_gate = proj[..., 3*hidden:4*hidden]

    # 3. compute gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # 4. optional mask: apply before gating in fused kernel
    if not nomask:
        # mask need to be broadcasted to hidden dim; shape [B,N,N,1] in fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        # fused multiply mask and gate
        _masked_mul_gate_inplace(left, left_gate, mask_fp16)
        _masked_mul_gate_inplace(right, right_gate, mask_fp16)
    else:
        # just gate multiply
        _mul_gate_inplace(left, left_gate)
        _mul_gate_inplace(right, right_gate)

    # 5. Compute out via einsum (batched GEMM)
    # Could try torch.einsum
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # 6. second layer norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # 7. out_gate (from x)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    out = out * out_gate

    # 8. final linear to dim
    out = torch.nn.functional.linear(out, to_out_w)

    # 9. cast to float32
    return out.to(torch.float32)
```

Optimized outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input (FP32) → cast to FP16.
2️⃣ Four linear projections (left/right + their gates) are executed with a
   single nn.functional.linear call for each projection.  The ``left_gate``
   and ``right_gate`` are turned into sigmoids (FP16).
3️⃣ An in‑place Triton kernel fuses the element‑wise mask (if present) and
   gate multiplication:
        out ← out * gate * mask               # mask is broadcast over the hidden dim
   This eliminates extra temporary allocations and reduces kernel launch
   overhead.
4️⃣ The cubic contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed with ``torch.einsum`` which internally uses a batched GEMM
   without the explicit permute+contiguous copy required by the naïve
   implementation.
5️⃣ Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection.
6️⃣ Cast the result back to FP32.

Only the element‑wise gating/mask step is implemented in Triton (as required);
all heavy matrix multiplications are delegated to cuBLAS/Tensor‑cores.
"""

import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: in‑place gated (and optionally masked) element‑wise multiplication

    out_ptr,          # *fp16  input & output
    gate_ptr,         # *fp16  gate tensor
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEM

    a = tl.load(out_ptr + offs, mask=mask)          # original value
    g = tl.load(gate_ptr + offs, mask=mask)         # gate (already sigmoid‑ed)
    tl.store(out_ptr + offs, a * g, mask=mask)


@triton.jit
def _masked_gate_mul_kernel(
    out_ptr,          # *fp16  input & output
    gate_ptr,         # *fp16  gate tensor
    mask_ptr,         # *fp16  mask tensor (flattened, shape = B*N*N)
    N_ELEM: tl.constexpr,   # total number of elements in out/gate tensors
    HIDDEN: tl.constexpr,   # hidden dimension (size of the last dim)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEM

    a = tl.load(out_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)
    # mask is stored without the hidden dimension, so we divide the flat index
    # by the hidden size to obtain the corresponding mask element.
    mask_idx = offs // HIDDEN
    m = tl.load(mask_ptr + mask_idx, mask=mask)
    tl.store(out_ptr + offs, a * g * m, mask=mask)


def _gate_mul_inplace(tensor: torch.Tensor, gate: torch.Tensor) -> None:
    """In‑place multiply ``tensor`` by ``gate`` (both fp16, contiguous)."""
    assert tensor.is_contiguous() and gate.is_contiguous()
    total = tensor.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _gate_mul_kernel[grid](
        tensor, gate, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


def _masked_gate_mul_inplace(tensor: torch.Tensor,
                             gate: torch.Tensor,
                             mask_flat: torch.Tensor,
                             hidden_dim: int) -> None:
    """In‑place multiply ``tensor`` by ``gate`` and broadcasted ``mask``."""
    assert tensor.is_contiguous() and gate.is_contiguous() and mask_flat.is_contiguous()
    total = tensor.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _masked_gate_mul_kernel[grid](
        tensor, gate, mask_flat, total, hidden_dim,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32, on CUDA)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32 or bool)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : must contain 'dim' and 'hidden_dim';
                         optional ``nomask`` (bool, defaults to True).

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for statistics) → FP16 for speed
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )
    x = x.to(torch.float16)        # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to fp16 (on‑device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (each is fp16 and returns a contiguous tensor)
    # ------------------------------------------------------------------
    left      = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, hidden]
    right     = torch.nn.functional.linear(x, right_proj_w)   # [B, N, N, hidden]
    left_gate = torch.nn.functional.linear(x, left_gate_w)   # [B, N, N, hidden]
    right_gate= torch.nn.functional.linear(x, right_gate_w)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 4️⃣ Gating sigmoids
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – fused with gating via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # flatten mask to 1‑D (B·N·N) in fp16 for the kernel; no explicit expansion needed
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        _masked_gate_mul_inplace(left, left_gate, mask_fp16, hidden_dim)
        _masked_gate_mul_inplace(right, right_gate, mask_fp16, hidden_dim)
    else:
        _gate_mul_inplace(left, left_gate)
        _gate_mul_inplace(right, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Compute out‑gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 7️⃣ Cubic contraction via einsum (batched GEMM under the hood)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Using torch.einsum avoids the explicit permute+contiguous copies used in the
    # reference implementation.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # shape [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
