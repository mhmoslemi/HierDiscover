```
@triton.jit
def fused_gate_mask_kernel_combined(
    left_proj_ptr, left_gate_pre_ptr,
    right_proj_ptr, right_gate_pre_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elts, hidden_dim,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elts

    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid)
    lgate_pre = tl.load(left_gate_pre_ptr + offs, mask=valid)
    rp = tl.load(right_proj_ptr + offs, mask=valid)
    rgate_pre = tl.load(right_gate_pre_ptr + offs, mask=valid)
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # compute sigmoid
    lgate = 1.0 / (1.0 + tl.exp(-lgate_pre))
    rgate = 1.0 / (1.0 + tl.exp(-rgate_pre))

    left_out = lp * lgate * m
    right_out = rp * rgate * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)
```

```
    # unpacks
    # compute x_norm = layer_norm
    x_fp16 = x_norm.float16()
    # Combine left_proj and left_gate weight
    left_combined_weight = torch.cat([weights['left_proj.weight'], weights['left_gate.weight']], dim=0).to(dtype=torch.float16)

    right_combined = torch.nn.functional.linear(x_fp16, right_combined_weight)

    left_gate_pre = left_combined[..., hidden_dim:]
    right_proj = right_combined[..., :hidden_dim]
    right_gate_pre = right_combined[..., hidden_dim:]

    # gating and mask fuse
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)
        total_elts = left_proj.numel()
        block = 1024

            left_gate_pre.data_ptr(),
            right_proj.data_ptr(),
            right_gate_pre.data_ptr(),

            total_elts,
            hidden_dim,

            num_warps=4,
```

We implement the **outgoing** Triage‑Multiplicative‑Update (TriMul) forward pass in a
single entry‑point called `custom_kernel`.  
The implementation follows the reference PyTorch code but heavily reduces
overhead:

| Step | Original code | Optimised version |
|------|---------------|-------------------|
| Layer‑norm (first) | FP32 → FP16 cast | Direct FP16 layer‑norm (no extra copy) |
| `left_proj` + `left_gate` | two separate linear layers | one fused linear (`[2·h, d]` weight) → split |
| `right_proj` + `right_gate` | two separate linear layers | one fused linear (`[2·h, d]` weight) → split |
| Mask + gate application | two Triton kernels (mask only) | **one** Triton kernel that does *sigmoid*, *mask* and *multiplication* for *both* left and right tensors |
| Heavy contraction `out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]` | `permute → view → torch.bmm` (copy) | unchanged (cuBLAS Tensor‑Core batched GEMM is already optimal) |
| Layer‑norm (second) | FP16 | unchanged |
| `out_gate` | linear + sigmoid | unchanged |
| Final projection | linear | unchanged |
| dtype handling | many FP32↔FP16 casts | only final cast back to FP32 |

Overall we cut the number of expensive mat‑muls from six to **four** and replace
four Triton launches by a single launch, while keeping all heavy arithmetic in
FP16 (Tensor‑Core friendly). The only custom Triton kernel is the fused gate‑ &
mask‑application kernel.

```python
"""
Fast outgoing TriMul forward pass (AlphaFold‑3) – Triton‑accelerated.

Algorithm
---------
1️⃣ FP16 layer‑norm on the input tensor.
2️⃣ Fuse left projection + left gate, and right projection + right gate, by
   concatenating the two weight matrices and performing a single mat‑mul per
   side (shape …‑> …×[2·hidden_dim]).
3️⃣ Split the result, apply the sigmoid gate and (optional) mask in a **single**
   Triton kernel that works on both left‑ and right‑tensors at once.
4️⃣ Compute the heavy O(N³) contraction `out = left @ rightᵀ` as a batched GEMM
   (`torch.bmm`).  This uses cuBLAS/Tensor‑Core and is already optimal.
5️⃣ Second FP16 layer‑norm.
6️⃣ Compute `out_gate = sigmoid(linear(x, out_gate_weight))`.
7️⃣ Multiply by `out_gate` and apply the final linear projection.
8️⃣ Cast back to FP32 for the harness.

The only Triton kernel is the fused gate‑mask kernel (step 3).  All other
operations use highly‑optimised PyTorch cuBLAS kernels in half‑precision.
"""

def fused_gate_mask_combined_kernel(
    # pointers (in‑place update)
    left_proj_ptr,          # *fp16  left projection
    left_gate_pre_ptr,      # *fp16  left gate (pre‑sigmoid)
    right_proj_ptr,         # *fp16  right projection
    right_gate_pre_ptr,     # *fp16  right gate (pre‑sigmoid)
    mask_ptr,               # *fp16  mask (broadcasted over hidden_dim)
    # run‑time parameters
    total_elts: tl.int32,   # B·N·N·H
    hidden_dim: tl.int32,   # H
    BLOCK: tl.constexpr,    # threads per program (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element id
    mask = offs < total_elts

    # mask is broadcast along the hidden dimension
    mask_idx = offs // hidden_dim

    # load data
    lp   = tl.load(left_proj_ptr   + offs, mask=mask, other=0.0)
    lgp  = tl.load(left_gate_pre_ptr + offs, mask=mask, other=0.0)
    rp   = tl.load(right_proj_ptr   + offs, mask=mask, other=0.0)
    rgp  = tl.load(right_gate_pre_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # sigmoid (fast fp16 approximation via exp)
    lgate = 1.0 / (1.0 + tl.exp(-lgp))
    rgate = 1.0 / (1.0 + tl.exp(-rgp))

    # apply gate and mask in‑place
    tl.store(left_proj_ptr   + offs, lp * lgate * m, mask=mask)
    tl.store(right_proj_ptr  + offs, rp * rgate * m, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    (Both left and right are [B, N, N, H] half‑precision tensors)
    """
    B, N, _, H = left.shape
    # (B, H, N, N) -> (B·H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # batched GEMM  (B·H, N, N) @ (B·H, N, N)ᵀ
    c = torch.bmm(a, b.transpose(1, 2))
    # restore layout (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack & cast
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm – keep everything in FP16
    # ------------------------------------------------------------------
    norm_w  = weights["norm.weight"].to(torch.float16)
    norm_b  = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        inp.to(torch.float16),            # cast input to fp16 first
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                                   # → [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (left & right)
    # ------------------------------------------------------------------
    # concatenate projection & gate weights (shape: [2·H, dim])
    left_comb_w  = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                 # (2·H, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                 # (2·H, dim)

    # one linear per side
    left_comb = torch.nn.functional.linear(x, left_comb_w)   # (B, N, N, 2·H)
    right_comb = torch.nn.functional.linear(x, right_comb_w)

    # split into projection & gate‑pre‑sigmoid
    left_proj      = left_comb[..., :hidden_dim]        # (B,N,N,H)
    left_gate_pre  = left_comb[..., hidden_dim:]        # (B,N,N,H)
    right_proj     = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 3️⃣ Apply sigmoid‑gate (and optional mask) with a single Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcasted over hidden_dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        total = left_proj.numel()          # B·N·N·H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        fused_gate_mask_combined_kernel[grid](
            left_proj.data_ptr(),
            left_gate_pre.data_ptr(),
            right_proj.data_ptr(),
            right_gate_pre.data_ptr(),
            mask_fp16.data_ptr(),
            total,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # after the kernel `left_proj` and `right_proj` already hold
        # the gated & masked values
        left  = left_proj
        right = right_proj
    else:
        # no mask → just sigmoid & element‑wise multiply
        left_gate  = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                           # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate (linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)   # (B,N,N,H) fp16
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 7️⃣ Gate, final projection & cast back to FP32
    # ------------------------------------------------------------------
    out = out * out_gate                         # (B,N,N,H) fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)     # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)          # (B,N,N,dim) fp16

    return out.to(torch.float32)
```
