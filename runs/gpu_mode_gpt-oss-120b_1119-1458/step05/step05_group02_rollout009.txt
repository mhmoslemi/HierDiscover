@triton.jit
def mask_gate_fuse_two_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    hidden_dim: tl.constexpr, size: tl.constexpr, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    mask_offs = offs // hidden_dim

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    new_left = left * left_gate * mask_val
    new_right = right * right_gate * mask_val

    tl.store(left_ptr + offs, new_left, mask=mask)
    tl.store(right_ptr + offs, new_right, mask=mask)
```

def apply_gate_and_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.constexpr,
    total_elems: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    mask_idx = offs // hidden_dim

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr+offs, left_out, mask=mask)
    tl.store(right_ptr+offs, right_out, mask=mask)
```

def mask_gate_fuse_kernel(
    left_ptr, right_ptr,

    mask_ptr,

    active = offs < total_elems

    # Compute flat mask index
    mask_idx = offs // hidden_dim

    x = tl.load(left_ptr + offs, mask=active, other=0.0)
    y = tl.load(right_ptr + offs, mask=active, other=0.0)
    g1 = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    g2 = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    tl.store(left_ptr + offs, x * g1 * m, mask=active)
    tl.store(right_ptr + offs, y * g2 * m, mask=active)

TriMul outgoing kernel (AlphaFold‑3) – forward only.

Algorithm
---------
1️⃣ First LayerNorm (float32) on the input tensor.
2️⃣ Cast to fp16 – this is the precision used for the bulk of the computation.
3️⃣ Fuse the five linear layers (left‑proj, right‑proj, three gates) into a single
   matrix‑multiply (`torch.nn.functional.linear`).  This removes four kernel launches
   and halves the amount of memory traffic.
4️⃣ Apply the three sigmoids to obtain the gate tensors.
5️⃣ If a mask is provided, a custom Triton kernel fuses the three element‑wise
   multiplications  `left = left * left_gate * mask` and
   `right = right * right_gate * mask`.  The kernel processes *both* tensors in a
   single launch, drastically cutting launch overhead.
   If there is no mask we fall back to a simple PyTorch element‑wise product.
6️⃣ Heavy O(N³) contraction is performed as a batched GEMM:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   By reshaping to `[B·H, N, N]` we can reuse cuBLAS Tensor‑Core kernels
   (`torch.bmm`) which are highly optimized on H100.
7️⃣ Second LayerNorm (half‑precision) over the hidden dimension.
8️⃣ Multiply by the output gate (sigmoid) and project back to the original
   feature dimension with a final linear layer.
9️⃣ Cast the result back to float32 (required by the harness) and return.

Only the forward pass is implemented – no autograd is needed.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel that fuses: left = left * left_gate * mask
#                     right = right * right_gate * mask
# The mask is broadcast over the hidden dimension (the innermost stride).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr,          # *fp16  : tensor to be updated (left)
    right_ptr,         # *fp16  : tensor to be updated (right)
    left_gate_ptr,     # *fp16  : left‑gate tensor (same shape as left)
    right_gate_ptr,    # *fp16  : right‑gate tensor (same shape as right)
    mask_ptr,          # *fp16  : mask tensor (shape B·N·N)
    hidden_dim,        # i32    : size of the hidden dimension H
    total_elems,       # i32    : total number of elements in left/right (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices for this block
    active = offs < total_elems                       # boundary guard

    # Because the hidden dimension is the innermost stride we can obtain the
    # mask index by integer division.
    mask_idx = offs // hidden_dim

    l = tl.load(left_ptr      + offs, mask=active, other=0.0)
    r = tl.load(right_ptr     + offs, mask=active, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rg = tl.load(right_gate_ptr+ offs, mask=active, other=0.0)
    m = tl.load(mask_ptr      + mask_idx, mask=active, other=1.0)

    tl.store(left_ptr  + offs, l * lg * m, mask=active)
    tl.store(right_ptr + offs, r * rg * m, mask=active)


def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction as a batched GEMM.
    left/right: [B, N, N, H] (fp16)
    Returns:    [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] – each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # bmm with the second operand transposed – matches the einsum in the reference
    c = torch.bmm(a, b.transpose(1, 2))               # [B·H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool/None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # if True we skip mask handling
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable on the raw input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                            # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy computation.
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single matmul.
    #    Shapes:
    #      - left_proj, right_proj, left_gate, right_gate, out_gate
    #        are all (hidden_dim, dim) without bias.
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # concatenate along the out‑feature dimension
    combined_w = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    )                                            # (5*hidden_dim, dim)

    B, N, _, _ = x.shape
    # flatten the first three dims: (B·N·N, dim)
    x_flat = x.reshape(-1, dim)                  # [B*N*N, dim]

    # single large linear (no bias)
    combined_out = torch.nn.functional.linear(x_flat, combined_w)  # [B*N*N, 5*hidden_dim]

    # -----------------------------------------------------------------
    # 4️⃣ Split and reshape projections + apply sigmoids to the gates.
    # -----------------------------------------------------------------
    splits = torch.split(combined_out, hidden_dim, dim=1)
    # left & right projections
    left = splits[0].view(B, N, N, hidden_dim)
    right = splits[1].view(B, N, N, hidden_dim)
    # gates – apply sigmoid in fp16
    left_gate  = torch.sigmoid(splits[2].view(B, N, N, hidden_dim))
    right_gate = torch.sigmoid(splits[3].view(B, N, N, hidden_dim))
    out_gate   = torch.sigmoid(splits[4].view(B, N, N, hidden_dim))

    # free the large temporary
    del combined_out, x_flat, combined_w

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fused element‑wise multiplication with gates.
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask is broadcast over the hidden dimension; convert to fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elems = left.numel()                         # B·N·N·H
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # single kernel processes both left and right tensors
        mask_gate_fuse_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # after the kernel left/right already contain the correct gated values
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # gates are no longer needed
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core).
    # -----------------------------------------------------------------
    out = _batched_matmul(left, right)   # [B, N, N, hidden_dim]  fp16

    # free left/right early to keep peak memory low
    del left, right

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (fp16).
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Output‑gate scaling and final projection back to `dim`.
    # -----------------------------------------------------------------
    out = out * out_gate                       # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected by the harness) and return.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
