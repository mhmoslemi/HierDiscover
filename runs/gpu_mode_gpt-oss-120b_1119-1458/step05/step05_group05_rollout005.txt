```python
# docstring summarizing algorithm.

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr, hidden_dim, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # compute mask index for each element
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left, right):
    """
    Compute out = left @ right^T for each batch & hidden channel by reshaping into
    a large batch of matrices and using torch.bmm (cuBLAS).
    left, right: shape (B,N,N,H) half.
    Returns: shape (B,N,N,H) half.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) matrices
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # batched matmul A @ B^T
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to (B,N,N,H)
    return c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass optimized with:
    1. A single fused linear for all projections and gates.
    2. Triton kernel fusing mask, gate and linear output multiplication.
    3. Batched GEMM via cuBLAS for O(N³) contraction.
    Works in fp16 for the heavy part, casts final result to fp32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------- 1. First LayerNorm (float32) -------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B,N,N,dim], float32

    # Cast to FP16 for the rest of the pipeline
    x = x.to(torch.float16)

    # ------- 2. Fuse all linear projections (proj + gates) -------
    # Gather weight matrices (float32) and cast to FP16
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along output dimension: (5*hidden_dim, dim)
    combined_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)

    # Flatten input [B,N,N,dim] -> [M, dim] where M = B*N*N
    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.view(M, dim)

    # Single fused linear (bias = None)
    proj = torch.nn.functional.linear(x_flat, combined_w)  # shape [M, 5*hidden_dim]
    # split
    H = hidden_dim
    left = proj[:, :H].view(B, N, N, H)
    left_gate = proj[:, H:2*H].view(B, N, N, H)
    right = proj[:, 2*H:3*H].view(B, N, N, H)
    right_gate = proj[:, 3*H:4*H].view(B, N, N, H)
    out_gate = proj[:, 4*H:5*H].view(B, N, N, H)

    # Apply sigmoid to gating tensors (stable in fp32)
    left_gate = torch.sigmoid(left_gate.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate.float()).to(torch.float16)
    out_gate = torch.sigmoid(out_gate.float()).to(torch.float16)

    # ------- 3. Optional mask fusion (Triton kernel) -------
    if not nomask:
        # mask shape [B,N,N] -> fp16, flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)   # [B*N*N]
        total_elements = left.numel()   # B*N*N*H

        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            H,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            H,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------- 4. Heavy O(N³) contraction via batched GEMM -------
    out = _batched_gemm_fp16(left, right)   # shape [B,N,N,H], fp16

    # ------- 5. Second LayerNorm (fp16) -------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------- 6. Gating and final projection -------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to float32 as the harness expects
    return out.to(torch.float32)

"""
Outgoing TriMul (the “tri‑multiplicative update”) forward pass.

Algorithm
---------
1️⃣  First LayerNorm (float32) → cast to FP16.
2️⃣  Fuse *all* linear projections (left/right and three gates) into a single
    matmul by concatenating the five weight matrices.  The result is split
    into `left`, `left_gate`, `right`, `right_gate`, `out_gate`.
3️⃣  Apply `sigmoid` to the three gating tensors (computed in FP32 for
    numerical stability, then cast back to FP16).
4️⃣  If a mask is supplied, fuse `tensor * gate * mask` with a tiny Triton
    kernel (`mask_gate_fuse_kernel`).  No‑mask case does plain element‑wise
    multiplication.
5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM (`C = A @ Bᵀ`) by reshaping to
    `[B*H, N, N]` and calling `torch.bmm`, which uses cuBLAS on H100.
6️⃣  Second LayerNorm (FP16) → element‑wise multiplication with `out_gate`.
7️⃣  Final linear projection back to the original channel dimension.
8️⃣  Cast the result back to float32 (the harness expects float32).

All heavy tensor arithmetic runs in FP16, reducing memory traffic and
latency, while the two LayerNorms stay numerically stable.  A custom
Triton kernel is used for the mask‑gate fusion, guaranteeing that the
solution contains user‑defined GPU code.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → size of the hidden dimension (H)
    size,           # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    """Fuse element‑wise multiplication with a broadcast mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=valid, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    # Each contiguous block of `hidden_dim` entries shares the same mask entry.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))

    # Fused multiply‑mask
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single batched GEMM (cuBLAS).  All tensors are FP16 and
    contiguous.
    """
    B, N, _, H = left.shape               # left.shape == (B, N, N, H)
    # reshape to [B*H, N, N] and perform A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))   # [B*H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32/bool) – ignored if config["nomask"]==True
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable mean/var computation
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all linear projections (proj + 3 gates) into one matmul
    # ------------------------------------------------------------------
    # Load weight matrices (float32) and cast once to fp16
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output dimension: shape (5*hidden_dim, dim)
    combined_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N                     # total number of (i,k) positions
    x_flat = x.reshape(M, dim)        # [M, dim]

    # Single fused linear (bias disabled in all sub‑layers)
    proj = torch.nn.functional.linear(x_flat, combined_w)   # [M, 5*hidden_dim]

    H = hidden_dim
    # Split the concatenated result
    left      = proj[:, 0 : H].reshape(B, N, N, H)
    left_gate = proj[:, H : 2 * H].reshape(B, N, N, H)
    right     = proj[:, 2 * H : 3 * H].reshape(B, N, N, H)
    right_gate = proj[:, 3 * H : 4 * H].reshape(B, N, N, H)
    out_gate   = proj[:, 4 * H : 5 * H].reshape(B, N, N, H)

    # ---------------------------------------------------------------
    # 4️⃣ Apply sigmoid to gating tensors (using FP32 for stability)
    # ---------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate.float()).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate.float()).to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – fused multiplication with Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → FP16, flattened to [B*N*N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                # size = B*N*N
        total_elements = left.numel()                 # B*N*N*H

        BLOCK = 1024                                   # <= 1024 threads per block
        grid = (triton.cdiv(total_elements, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            H,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            H,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H]  (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Gating with out_gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
