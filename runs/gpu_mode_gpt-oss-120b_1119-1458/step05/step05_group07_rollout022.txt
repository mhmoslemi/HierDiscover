"""
TriMul outgoing implementation (AlphaFold‑3) – highly‑optimized forward pass.

Key ideas
---------
1️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    into a single `torch.nn.functional.linear` call by concatenating the weight
    matrices.  This eliminates five separate matrix‑multiplications and reduces
    memory traffic on the large `[B, N, N, dim]` tensor.

2️⃣  Perform the heavy O(N³) contraction with a single batched GEMM via
    `torch.matmul`.  By permuting the hidden dimension to the batch axis
    (`(B, H, N, N)`) we avoid the costly `contiguous()` copies required by the
    reference implementation.

3️⃣  Fuse the per‑element mask‑*‑gate multiplication into a tiny Triton kernel.
    The mask has shape `[B, N, N]` and is broadcast over the hidden dimension;
    the kernel computes `x *= gate * mask` in‑place.

4️⃣  Keep all compute‑intensive stages in FP16 (half precision) – the H100
    tensor‑core kernels for `matmul` and the Triton kernel run at peak
    throughput.  Only the two LayerNorms are performed in FP32 → FP16 to keep
    numerical stability.

The function `custom_kernel` follows the signature required by the harness:
    (input_tensor, mask_tensor, weights_dict, config_dict) -> output_tensor
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,              # *fp16 : gate tensor, same shape as x
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate
    hidden_dim: tl.int32,  # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    valid = offs < total_elts

    # mask repeats every `hidden_dim` entries (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (no autograd).

    Parameters
    ----------
    data :
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool)
        - weights_dict : dict with the model weights (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → cast to FP16 afterwards)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all subsequent heavy work
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five projections into a single matmul
    # ------------------------------------------------------------------
    # Concatenate the weight matrices (shape: (out_features, in_features))
    # Order: left_proj, right_proj, left_gate, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, dim)

    # Linear projection: [B, N, N, dim] @ (5*H, dim).T → [B, N, N, 5*H]
    proj = torch.nn.functional.linear(x, combined_weight)  # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Split into the individual tensors
    # ------------------------------------------------------------------
    left, right, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        proj, hidden_dim, dim=-1
    )  # each: [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Gating sigmoids (still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()                     # B * N * N * H
        BLOCK = 1024                                   # ≤ 1024 threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # Fuse left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: cheap element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via a single batched GEMM (no extra copy)
    # ------------------------------------------------------------------
    # Rearrange tensors so the hidden dimension becomes the “batch” axis:
    #   left_perm : (B, H, N, N)   – -> rows are i, cols are k
    #   right_perm: (B, H, N, N)   – -> rows are j, cols are k
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)   # (B, H, N, N)

    # GEMM: out[b, h, i, j] = Σ_k left_perm[b, h, i, k] * right_perm[b, h, j, k]
    #   we compute left_perm @ right_permᵀ  (K dimension is the inner one)
    out = torch.matmul(
        left_perm,                     # (B, H, N, K)
        right_perm.transpose(-2, -1)  # (B, H, K, N)
    )  # -> (B, H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gating + projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)  FP16

    # ------------------------------------------------------------------
    # 🔟 Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
