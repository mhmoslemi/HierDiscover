import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def fused_gate_mask_kernel(
    x_ptr, gate_ptr, mask_ptr,
    total_elts, hidden_dim,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # Index into mask per hidden dimension tile
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=mask)

def _batched_matmul_fused(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] using
    torch.bmm on the reshaped tensors. All tensors are expected to be
    half‑precision and contiguous."""
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) tensors
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right needs to be transposed on the last two axes before reshaping
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    # batched matmul
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)
    # restore shape
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
Fast outgoing TriMul forward (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input in FP32 (stable) and cast to FP16.
2️⃣  Fuse the two linear projections (`left_proj`, `right_proj`) into a single
    GEMM by concatenating their weight matrices → obtain `left` and `right`.
3️⃣  Fuse the two gate projections (`left_gate`, `right_gate`) similarly.
    Apply a sigmoid to obtain the gating tensors.
4️⃣  Compute the optional mask‑gate fusion with a tiny Triton kernel:
        tensor = tensor * gate * mask   (broadcast mask over `hidden_dim`).
    If `nomask==True` a simple element‑wise multiply is used.
5️⃣  The O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM:
        * reshape `left` → (B·H, N, N)
        * reshape and transpose `right` → (B·H, N, N)
        * torch.bmm on the two 3‑D tensors.
6️⃣  Second Layer‑Norm (FP16), multiply by `out_gate`, and a final linear
    projection back to the original feature dimension.
7️⃣  Cast the result to FP32 for the harness.

All heavy linear work is done in half‑precision, the mask‑gate fusion is
implemented in Triton, and the number of large GEMMs is reduced from five to
three by weight concatenation.  This yields a substantial speed‑up on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask shape is (B, N, N); it is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,              # *fp16 : gate tensor, same shape as x
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate
    hidden_dim: tl.int32,  # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    valid = offs < total_elts

    # Compute index into the broadcasted mask (mask repeats every `hidden_dim`)
    mask_idx = offs // hidden_dim

    x_val   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g_val   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m_val   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(x_ptr + offs, x_val * g_val * m_val, mask=valid)


def _batched_matmul_fused(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]` using
    a single batched GEMM.  Expected datatype: torch.float16, contiguous.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right needs to be transposed on the last two axes before reshaping
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    # batched matrix multiplication (C = A @ B) → shape (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)
    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # → [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to FP32 – required by the testing harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

This implementation:

* **Reduces the number of large GEMMs** by concatenating the projection matrices (`left_proj`/`right_proj`) and the gate matrices (`left_gate`/`right_gate`).  
* **Fuses masking and gating** with a compact Triton kernel (`fused_gate_mask_kernel`).  
* **Performs the cubic contraction** with a single batched `torch.bmm`, reshaping tensors to avoid extra transposes.  
* **Keeps the heavy arithmetic in FP16** while preserving numerical stability on the initial LayerNorm (FP32).  

The kernel should run on Triton 3.3.1 and H100 GPUs, delivering a substantial speed‑up over the baseline implementation.
