    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm 1: use fp32 to compute mean/var, then cast to fp16.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    ).half()

    # cast weights to FP16
    left_proj_gate_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_proj_gate_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    out_gate_weight = weights["out_gate.weight"].half()
    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias = weights["to_out_norm.bias"].half()
    to_out_weight = weights["to_out.weight"].half()

    # left + left_gate linear: shape (2*hidden_dim, dim)
    left_concat = torch.nn.functional.linear(x, left_proj_gate_weight)  # [B,N,N,2*hidden_dim]
    left_proj = left_concat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])  # [B,N,N,hidden_dim]

    # right + right_gate
    right_concat = torch.nn.functional.linear(x, right_proj_gate_weight)
    right_proj = right_concat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])

    # out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Multiply gates (and mask)
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # flatten mask to 1D float16
        mask_flat = mask_tensor.to(torch.float16).contiguous().view(-1)  # shape B*N*N
        total_elements = left_proj.numel()
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

            left_proj,

            hidden_dim,

        # fuse right
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )

        right = right_proj

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm on the original channel dimension (FP32) → cast to FP16.
2️⃣  Fuse the two projection linear layers (proj & gate) per side by
    concatenating their weights → a single matmul produces both tensors.
3️⃣  Apply sigmoid to the gate half, then fuse element‑wise multiplication
    (proj * gate) and optional mask using one Triton kernel per side
    (in‑place → no extra memory).
4️⃣  Compute the heavy contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM (left @ rightᵀ) via `torch.bmm`.  Each head is
    processed in the same cuBLAS tensor‑core kernel.
5️⃣  Second Layer‑norm (FP16) over the hidden dimension.
6️⃣  Multiply by the out‑gate (sigmoid of a third linear layer) and
    apply the final projection back to the original dimension.
7️⃣  Cast the result back to FP32 for the harness.

Only the mask‑gate fusion is implemented in Triton, but the critical
linear‑projection passes are halved by weight concatenation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,         # *fp16 : projection output (size,)
    gate_ptr,         # *fp16 : gate output (same shape)
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,)
    size,             # i32   : total number of elements in proj/gate
    hidden_dim,       # i32   : width of hidden dimension (channel count)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block id
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets for this block
    mask = offs < size                         # guard against OOB

    # Load the projection and gate values
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Every hidden_dim consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)   # 1.0 = no‑mask

    # Fuse: proj * gate * mask
    out = proj * gate * m

    # Store back in‑place
    tl.store(proj_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right tensors:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Both inputs are FP16 of shape [B, N, N, H].
    The operation is performed as a batched GEMM:
        (B*H, N, N) @ (B*H, N, N)ᵀ
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and transpose the second argument on‑the‑fly
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))               # (B*H, N, N)
    # restore original layout [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask_tensor  : [B, N, N]       (float32 / bool) – may be ignored
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (True → mask is not used)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]       # projection width (c)
    nomask = config.get("nomask", True)    # True → ignore mask

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32
    x = x_norm.half()  # cast once to fp16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to fp16 (once)
    # ------------------------------------------------------------------
    # Concatenate proj and gate weights for each side – reduces matmul launches
    left_proj_gate_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # (2*hidden_dim, dim)
    right_proj_gate_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # (2*hidden_dim, dim)

    out_gate_w = weights["out_gate.weight"].half()               # (hidden_dim, dim)
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w = weights["to_out.weight"].half()                   # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection+gate linear layers (one matmul per side)
    # ------------------------------------------------------------------
    # ---- left side ----------------------------------------------------
    left_cat = torch.nn.functional.linear(x, left_proj_gate_w)   # [B,N,N,2*hidden_dim]
    left_proj = left_cat[..., :hidden_dim]                      # proj part
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])       # gate part (after sigmoid)

    # ---- right side ---------------------------------------------------
    right_cat = torch.nn.functional.linear(x, right_proj_gate_w)
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ out‑gate (independent linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Fuse element‑wise multiplication (proj * gate) and optional mask
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate          # in‑place could be used, but this is fine
        right = right_proj * right_gate
    else:
        # Flatten mask to 1‑D fp16 for the Triton kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        total_elements = left_proj.numel()  # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction (batched GEMM) – using cuBLAS via torch.bmm
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim]   FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the evaluation harness
    # ------------------------------------------------------------------
    return out.float()
