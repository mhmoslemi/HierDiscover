"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input in float32 (as in the reference model).
2. Compute left/right projections, left/right/out gates with fp32 linear layers.
3. Cast the projections and gates to fp16.
4. If a mask is present, fuse the three element‑wise multiplications
   (tensor * gate * mask) with a small Triton kernel.
   Otherwise a simple PyTorch element‑wise product is used.
5. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is carried out as a batched half‑precision GEMM by reshaping the tensors
   to [B·H, N, N] and calling ``torch.bmm`` (Tensor‑Core kernels on H100).
6. Second LayerNorm (still in fp16) over the hidden dimension.
7. Multiply by the out‑gate (sigmoid), then linear‑project back to the original
   dimension with a half‑precision weight matrix.
8. Cast the final result back to float32.

Only the forward pass is implemented; no autograd bookkeeping is required.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) for fp16 tensors.
# x, gate have shape (B, N, N, H) → flattened length = B*N*N*H.
# mask has shape (B, N, N) and is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (flattened B·N·N)
    hidden_dim,     # i32   : size of the hidden dimension H
    size,           # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,  # compile‑time block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element indices
    active = offs < size                               # guard out‑of‑bounds

    # The mask index for a given element is the element index divided by H
    mask_offs = offs // hidden_dim

    x_val   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g_val   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m_val   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)  # missing entries = 1

    out_val = x_val * g_val * m_val
    tl.store(x_ptr + offs, out_val, mask=active)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction using a batched half‑precision GEMM.

    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H] and dtype torch.float16.

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, H] (dtype torch.float16).
    """
    B, N, _, H = left.shape
    # Move hidden dimension to the batch axis: [B, H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out[b*h, i, j] = Σ_k a[b*h, i, k] * b[b*h, j, k]   →  a @ b^T
    c = torch.bmm(a, b.transpose(1, 2))

    # Reshape back to original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) or None
        - weights_dict : mapping of weight names to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``, optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – mirrors the reference implementation.
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections and gate projections (float32 for maximal accuracy).
    # -----------------------------------------------------------------
    left      = torch.nn.functional.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right     = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3️⃣ Cast tensors needed for the heavy contraction to fp16.
    # -----------------------------------------------------------------
    left      = left.to(torch.float16)
    right     = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – fuse element‑wise multiplication with a Triton kernel.
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16 and make it contiguous.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()                     # B·N·N·hidden
        BLOCK = 1024                                    # max 1024 threads per program
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple elementwise multiplication in fp16.
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via a batched half‑precision GEMM.
    # -----------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)   # [B, N, N, hidden] (fp16)

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16).
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Gating with out_gate (cast to fp16) and final projection.
    # -----------------------------------------------------------------
    out_gate = out_gate.to(torch.float16)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 and return.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
