"""
TriMul (outgoing) forward pass – highly‑fused Triton implementation
===================================================================

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣  Five linear projections are required:
     * left_proj   + left_gate   (concatenated → single GEMM)
     * right_proj  + right_gate  (concatenated → single GEMM)
     * out_gate    (stand‑alone GEMM)
   All GEMMs run on half‑precision using ``torch.nn.functional.linear``.
3️⃣  Fuse the three element‑wise ops (gate‑sigmoid, optional mask, and
    multiplication) into a tiny Triton kernel.  The kernel works on the
    flattened tensor ``[B·N·N·H]`` and broadcasts the mask over the hidden
    dimension.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
    is carried out as a *batched* GEMM (C = A @ Bᵀ) with
    ``torch.bmm`` – each hidden channel becomes a batch element.
5️⃣  Second Layer‑norm (fp16), gated by ``out_gate`` and a final linear
    projection back to the original feature dimension.
6️⃣  Cast the result back to float32 (the harness expects fp32).

Only the forward pass is provided – no autograd bookkeeping is needed.
The implementation contains a Triton kernel for the fused gating+mask step,
which cuts two memory passes (gate + mask) into one.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * sigmoid(gate) * mask) in‑place.
#   x_ptr   – pointer to the projected tensor (size = total_elts)
#   g_ptr   – pointer to the *raw* gate tensor (same shape as x)
#   m_ptr   – pointer to the broadcast mask (size = total_elts / hidden_dim)
#   total_elts = B * N * N * hidden_dim
#   hidden_dim = size of the last axis of x/g
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_sigmoid_kernel(
    x_ptr,                     # *fp16 : tensor to be gated (output in‑place)
    g_ptr,                     # *fp16 : raw gate tensor (same shape as x)
    m_ptr,                     # *fp16 : mask tensor (broadcasted)
    total_elts: tl.int32,      # total #elements in x/g
    hidden_dim: tl.int32,      # hidden dimension (= last dim of x)
    BLOCK: tl.constexpr,       # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elts

    # mask is broadcast across the hidden dimension:
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(g_ptr + offs, mask=mask, other=0.0)

    # sigmoid(g) = 1 / (1 + exp(-g))
    g_sig = 1.0 / (1.0 + tl.exp(-g))

    # mask (0 / 1) – when nomask we simply pass 1.0
    m = tl.load(m_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x * g_sig * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM using torch.bmm (the fastest path on H100 for large N)
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    Arguments are FP16 tensors of shape [B, N, N, H].
    Returns a FP16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) → each hidden channel becomes a batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)ᵀ
    c = torch.bmm(a, b.transpose(1, 2))
    # restore the original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    # Left side: concat(proj, gate) → single GEMM, then split
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)          # (2*H, dim)
    left_concat   = torch.nn.functional.linear(x, left_concat_w)         # [B,N,N,2*H]
    left_proj, left_gate_raw = left_concat.chunk(2, dim=-1)              # each [B,N,N,H]

    # Right side: concat(proj, gate) → single GEMM, then split
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)      # (2*H, dim)
    right_concat    = torch.nn.functional.linear(x, right_concat_w)     # [B,N,N,2*H]
    right_proj, right_gate_raw = right_concat.chunk(2, dim=-1)          # each [B,N,N,H]

    # Out‑gate (single linear)
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)            # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_raw)                               # [B,N,N,H]

    # --------------------------------------------------------------
    # 4️⃣ Fuse gate‑sigmoid + optional mask (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dimension; store as FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                     # B * N * N * H
        BLOCK = 1024                                        # tile size (≤1024 threads)
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * sigmoid(left_gate_raw) * mask
        fused_gate_mask_sigmoid_kernel[grid](
            left_proj.data_ptr(),
            left_gate_raw.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * sigmoid(right_gate_raw) * mask
        fused_gate_mask_sigmoid_kernel[grid](
            right_proj.data_ptr(),
            right_gate_raw.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # discard the raw gate tensors (they are no longer needed)
    else:
        # No mask – just apply the sigmoids
        left_proj = left_proj * torch.sigmoid(left_gate_raw)
        right_proj = right_proj * torch.sigmoid(right_gate_raw)

    # Now ``left_proj`` and ``right_proj`` hold the gated tensors.
    left  = left_proj   # shape [B, N, N, H]
    right = right_proj  # shape [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H]  (FP16)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection back to original dim
    # --------------------------------------------------------------
    out = out * out_gate                         # element‑wise scaling
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
