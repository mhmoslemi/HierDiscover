"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 (for numeric stability) and cast to FP16.
2️⃣  Fuse the four linear projections (left‑/right‑proj) into a single
    `nn.functional.linear` with concatenated weights → 2·hidden_dim outputs.
    Likewise fuse the three gating projections (left‑gate, right‑gate,
    out‑gate) into a second linear → 3·hidden_dim outputs.
3️⃣  Apply the sigmoid to the gating tensors.
4️⃣  Fuse the element‑wise mask (if present) and gating multiplications
    into a tiny Triton kernel – one launch per tensor (left/right).  This
    eliminates two temporary copies.
5️⃣  Heavy cubic contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM.  By reshaping
    `[B,N,N,H] → [B·H, N, N]` we let cuBLAS/Tensor‑cores handle the work.
6️⃣  Second Layer‑norm (FP16) → multiply by the out‑gate (elementwise).
7️⃣  Final linear projection back to `dim`.
8️⃣  Cast result back to float32 (required by the harness).

All heavy arithmetic is done in FP16 with FP32 accumulation (cuBLAS default).
Only the gating‑mask fusion is implemented in Triton, satisfying the “must
use a Triton kernel” requirement while drastically reducing the number of
separate `nn.Linear` calls (5 → 2) and memory traffic.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication (mask optional)
#   out ← out * gate * mask   (mask broadcasted over hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mul_kernel(
    out_ptr,          # *fp16  tensor to be multiplied (in‑place)
    gate_ptr,         # *fp16  gate tensor (same shape as out)
    mask_ptr,         # *fp16  mask tensor (flattened, shape = total_rows = B·N·N)
    hidden_dim: tl.constexpr,  # size of the hidden dimension H
    total_elements: tl.constexpr,  # total number of elements in out (B·N·N·H)
    USE_MASK: tl.constexpr,   # 0 → ignore mask, 1 → apply it
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load out and gate
    out_val = tl.load(out_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    if USE_MASK:
        # each hidden entry belongs to a row = (b,i,j); compute its row index
        row_idx = offs // hidden_dim               # floor division
        mask_val = tl.load(mask_ptr + row_idx, mask=mask, other=1.0)  # 0 or 1 in fp16
        out_val = out_val * gate_val * mask_val
    else:
        out_val = out_val * gate_val

    tl.store(out_ptr + offs, out_val, mask=mask)

# ----------------------------------------------------------------------
# Helper to launch the above kernel on a tensor
# ----------------------------------------------------------------------
def _apply_gate_and_mask(
    tensor: torch.Tensor,
    gate: torch.Tensor,
    mask_flat: torch.Tensor,
    hidden_dim: int,
    use_mask: bool,
) -> None:
    """
    In‑place element‑wise multiplication:
        tensor = tensor * gate * mask (if use_mask)
    All tensors must be contiguous fp16 and share the same device.
    """
    assert tensor.is_contiguous() and gate.is_contiguous()
    total = tensor.numel()
    BLOCK = 65536  # tunable – works well for H100
    grid = ((total + BLOCK - 1) // BLOCK,)

    # If mask is not used we pass a dummy pointer (never read)
    dummy = torch.empty((1,), dtype=torch.float16, device=tensor.device)
    mask_ptr = mask_flat if use_mask else dummy

    _gate_mul_kernel[grid](
        tensor, gate, mask_ptr,
        hidden_dim,
        total,
        USE_MASK=1 if use_mask else 0,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/ bool) – ignored if config["nomask"]=True
        - weights_dict : mapping from module name to tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]      # (dim,)
    norm_bias = weights["norm.bias"]          # (dim,)

    # LayerNorm on the last dimension (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                           # still float32
    x = x.to(torch.float16)                    # [B, N, N, dim] in fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to fp16 (once)
    # ------------------------------------------------------------------
    # Linear projection weights (bias=False in the reference)
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)   # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Output‐norm and final projection weights
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)  # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)    # (hidden_dim,)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse linear projections:   left+right  (2·hidden_dim outputs)
    # ------------------------------------------------------------------
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)   # shape (2*H, dim)
    proj_out = torch.nn.functional.linear(x, proj_weight)       # [B,N,N,2*H] fp16
    left, right = torch.split(proj_out, hidden_dim, dim=-1)    # each [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Fuse gating linear layers: left_gate, right_gate, out_gate (3·H outputs)
    # ------------------------------------------------------------------
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*H, dim)
    gate_out = torch.nn.functional.linear(x, gate_weight)                    # [B,N,N,3*H] fp16
    left_gate, right_gate, out_gate = torch.split(gate_out, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask (if any) and gating with a single Triton kernel per tensor
    # ------------------------------------------------------------------
    if not nomask:
        # Prepare mask: [B,N,N] → fp16 → flatten across rows (no hidden dim)
        # `use_mask` case expects a 1‑D mask of length B·N·N.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)          # shape (B*N*N,)
        use_mask = True
    else:
        mask_flat = torch.empty((1,), dtype=torch.float16, device=device)  # dummy
        use_mask = False

    # Ensure tensors are contiguous before Triton writes
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    # Fuse gating (and mask if present)
    _apply_gate_and_mask(left, left_gate, mask_flat, hidden_dim, use_mask)
    _apply_gate_and_mask(right, right_gate, mask_flat, hidden_dim, use_mask)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – batched GEMM (cuBLAS / Tensor‑cores)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to [B·H, N, N] for batched matrix multiplication
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_flat[b*h, i, j] = Σₖ left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout → [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) → element‑wise out‑gate → final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                           # still fp16
    out = out * out_gate                         # element‑wise gate (fp16)

    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 (output format required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
