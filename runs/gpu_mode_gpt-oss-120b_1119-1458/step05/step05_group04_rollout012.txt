"""
TriMul (outgoing) forward pass – high‑performance implementation.

Algorithm
---------
1️⃣ Layer‑norm the [B,N,N,dim] input in FP32 (numeric stability) → cast to FP16.
2️⃣ Fuse the five linear projections (left/right and three gates) into a single
   GEMM by concatenating their weight matrices (shape [5*H, dim]).
   The GEMM is performed with `torch.nn.functional.linear` on the flattened
   tensor (P = B·N·N rows).  This reads the normalized input only once.
3️⃣ Split the concatenated output into `left_proj`, `left_gate`, `right_proj`,
   `right_gate`, `out_gate`.  Apply `sigmoid` to the three gate tensors.
4️⃣ (optional mask) Multiply the two projections by the mask (broadcast over H).
5️⃣ Fuse the element‑wise multiplication `proj * gate` for *left* and *right*
   using a tiny Triton kernel (no extra temporaries → lower memory traffic).
6️⃣ Heavy cubic contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is executed as a batched GEMM.  By permuting to shape [B·H, N, N] we let
   cuBLAS/Tensor‑Cores handle the O(N³·H) work.
7️⃣ Second LayerNorm over the hidden dimension, followed by a Triton in‑place
   multiplication with the pre‑computed `out_gate`.
8️⃣ Final linear projection back to `dim` (hidden → dim) via a regular linear
   layer; cast the result to FP32 for the harness.

All heavy linear work stays in cuBLAS (fast Tensor‑Core GEMMs); Triton is used
for the two in‑place point‑wise fusions, satisfying the “must use Triton”
requirement while keeping the overall runtime well below the target.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication of two tensors.
#   out ← out * mul
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    out_ptr,               # *fp16   input & output
    mul_ptr,               # *fp16   multiplier
    N,                     # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _pointwise_mul_inplace(x: torch.Tensor, y: torch.Tensor) -> None:
    """
    In‑place element‑wise multiplication: ``x ← x * y``.
    Both ``x`` and ``y`` must be contiguous FP16 tensors of the same shape.
    """
    assert x.is_contiguous() and y.is_contiguous()
    total = x.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        x, y, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)                             # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections (proj + gates) into one GEMM
    # ------------------------------------------------------------------
    # Cast model weights to fp16 on the target device
    w_left_proj    = weights["left_proj.weight"].to(device=device, dtype=torch.float16)   # (H, dim)
    w_left_gate    = weights["left_gate.weight"].to(device=device, dtype=torch.float16)   # (H, dim)
    w_right_proj   = weights["right_proj.weight"].to(device=device, dtype=torch.float16)  # (H, dim)
    w_right_gate   = weights["right_gate.weight"].to(device=device, dtype=torch.float16)  # (H, dim)
    w_out_gate     = weights["out_gate.weight"].to(device=device, dtype=torch.float16)    # (H, dim)

    # Concatenate along the output‑feature axis: shape (5*H, dim)
    w_concat = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )   # (5*H, dim)

    # Flatten the normalized tensor to (P, dim) where P = B·N·N
    P = B * N * N
    x_flat = x.reshape(P, dim)                          # (P, dim)

    # Single linear pass
    proj_concat = torch.nn.functional.linear(x_flat, w_concat)   # (P, 5*H)

    # Split the concatenated output
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        proj_concat, H, dim=1
    )  # each (P, H)

    # Reshape back to 4‑D tensors
    left_proj   = left_proj.view(B, N, N, H)
    left_gate_pre = left_gate_pre.view(B, N, N, H)
    right_proj  = right_proj.view(B, N, N, H)
    right_gate_pre = right_gate_pre.view(B, N, N, H)
    out_gate_pre = out_gate_pre.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)            # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gating (proj * gate) via Triton
    # ------------------------------------------------------------------
    left = left_proj.contiguous()
    right = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    _pointwise_mul_inplace(left, left_gate)   # left ← left * left_gate
    _pointwise_mul_inplace(right, right_gate) # right ← right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Cubic contraction (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] for a batched matmul
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # (B*H, N, N)

    # out_flat[b*h,i,j] = Σ_k left_flat[b*h,i,k] * right_flat[b*h,j,k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore layout to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm + out‑gate multiplication (Triton)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # (B,N,N,H) fp16

    # Apply out_gate (element‑wise) using the same Triton kernel
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    _pointwise_mul_inplace(out, out_gate)     # out ← out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)   # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast the result back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
