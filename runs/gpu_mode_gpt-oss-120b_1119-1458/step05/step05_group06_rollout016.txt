"""
Outgoing TriMul (AlphaFold‑3) – fused linear/gate and fast contraction.

Algorithm
---------
1.  Input ‑‑> LayerNorm (float32 for numerical stability).
2.  Cast to fp16 and compute **all** five linear maps (left_proj,
    left_gate, right_proj, right_gate, out_gate) in a *single* GEMM
    by concatenating the five weight matrices (shape (5·H, D)).
3.  Slice the result, apply sigmoid to the three gate tensors,
    optionally mask the two projection tensors, and multiply the
    projections by their gates.
4.  The heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM C = A @ Bᵀ where the hidden
    dimension H is treated as the batch axis (M = B·H).  A compact
    Triton kernel computes this with fp16 inputs and fp32 accumulation.
    For very long sequences we fall back to torch.bmm (cuBLAS), which
    is optimal on H100 for N > 256.
5.  Second LayerNorm (fp16) → element‑wise multiplication with the
    out‑gate → final Linear (hidden_dim → D) → cast back to float32.

All heavy linear passes are merged, reducing the number of GEMMs from
six to two and cutting the overall FLOP count dramatically.  The
contraction is the dominant cost and is handled by a highly tuned
Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply kernel for the TriMul contraction.

    Computes for each batch‑hidden slice (pid_z):
        C[i, j] = Σₖ A[i, k] * B[j, k]   (i, j over the sequence dimension).
    """
    pid_m = tl.program_id(0)   # tile along i (output rows)
    pid_n = tl.program_id(1)   # tile along j (output cols)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------------------------------------------------------
    # 1️⃣ tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (more accurate)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2️⃣ loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i, k]   → shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am               # batch*hidden offset
            + offs_i[:, None] * stride_ai     # i offset
            + offs_k[None, :] * stride_ak     # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load Bᵀ[j, k] (transposed view)   → shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm               # batch*hidden offset
            + offs_k[:, None] * stride_bk     # k offset as first index
            + offs_j[None, :] * stride_bj     # j offset as second index
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 3️⃣ store result C[i, j]
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the O(N³) contraction using the Triton batched‑GEMM kernel.
    Input tensors must be fp16 and have shape [B, N, N, H].
    Returns a tensor of the same shape (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H  (each hidden slice is a separate batch)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in element (not byte) units
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # tile size – 64×64 works well on H100 for fp16 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Fallback O(N³) contraction using cuBLAS (torch.bmm).
    Operates in fp16 with fp32 accumulation (the default for torch.bmm on GPU).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    if not nomask:
        # mask may be bool or float; convert to fp16 and broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply gates (element‑wise)
    left = left_proj * left_gate          # shape [B, N, N, hidden_dim]
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ heavy O(N³) contraction (batched GEMM)
    # --------------------------------------------------------------
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16 [B, N, N, hidden_dim]
    else:
        contracted = _batched_matmul_torch(left, right)    # fp16

    # --------------------------------------------------------------
    # 7️⃣ second LayerNorm (fp16) → out‑gate → final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    contracted = F.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    contracted = contracted * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(contracted, to_out_w)                    # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 8️⃣ cast back to float32 (required by test harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
