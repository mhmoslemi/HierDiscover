"""
Optimized Triton+PyTorch implementation of the *outgoing* TriMul operator
(used in AlphaFold‑3, Chai, Protenix …).

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → cast once to FP16.
2. Fuse the five linear layers (left/right projections, three gates) into a
   single FP16 mat‑mul by stacking their weights.  Split the result back
   into the five tensors and apply a sigmoid to the three gates.
3. If a mask is present, fuse the point‑wise multiplication
      proj * gate * mask
   into a tiny Triton kernel (one launch per tensor).  When there is no mask
   we just use an in‑place ``mul_``.
4. The O(N³) contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (`torch.bmm`).  By reshaping the tensors to
   [B·H, N, N] we obtain a single Strided‑Batched Tensor‑Core call, which is
   the fastest path on an H100.
5. Apply the second Layer‑Norm (FP16), multiply by the out‑gate, and a final
   linear projection back to the original channel dimension.
6. Return the result in float32.

At least one custom Triton kernel is used (mask‑gate fusion).  The code
handles both masked and un‑masked inputs, arbitrary batch sizes (1‑2), sequence
lengths up to 1024 and hidden dimensions 128‑384.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) – mask is broadcast over
# the hidden dimension `hidden_dim`.  The kernel works on a flat view of
# the 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,             # i32   : total number of elements in `x` (B*N*N*hidden_dim)
    hidden_dim,       # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask = offs < size                         # boundary mask

    # Load data (half)
    x_val   = tl.load(x_ptr   + offs, mask=mask, other=tl.float16(0.0))
    g_val   = tl.load(gate_ptr+ offs, mask=mask, other=tl.float16(0.0))

    # Broadcast mask: every `hidden_dim` consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    m_val   = tl.load(mask_ptr + mask_offs, mask=mask, other=tl.float16(1.0))

    # Fused element‑wise multiplication
    out = x_val * g_val * m_val

    # Store the result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the heavy O(N³) contraction.
# Uses cuBLAS via torch.bmm (Tensor‑Core FP16 fast path).
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    for all batches and hidden channels.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (FP16)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]  – we want A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM (FP16 inputs → FP32 accumulation inside cuBLAS)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]          # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask     = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                 # [B, N, N, dim]  float32
    x = x_norm.half()                # cast once to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Stack all five linear weights and perform a single fused mat‑mul
    # ------------------------------------------------------------------
    # All linear layers are bias‑free, shape: (hidden_dim, dim)
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()

    # Concatenate along the output‑feature axis
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Single FP16 linear projection
    fused = torch.nn.functional.linear(x, fused_weight)  # [B, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor & apply gates (sigmoid)
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        fused, hidden_dim, dim=-1
    )
    left_gate  = torch.sigmoid(left_gate_pre)   # [B,N,N,hidden_dim] FP16
    right_gate = torch.sigmoid(right_gate_pre)  # FP16
    out_gate   = torch.sigmoid(out_gate_pre)    # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) and combine with gates.
    #    The operation performed is:
    #        left  = left_proj  * left_gate  * mask
    #        right = right_proj * right_gate * mask
    # ------------------------------------------------------------------
    if nomask:
        # In‑place fused multiplication – no extra kernel launch
        left  = left_proj.mul_(left_gate)
        right = right_proj.mul_(right_gate)
    else:
        # Prepare mask as FP16, flattened to (B*N*N,)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elements = left_proj.numel()               # B * N * N * hidden_dim
        BLOCK = 1024                                      # ≤ max CUDA threads / block
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask  (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask  (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left  = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction (batched GEMM) – cuBLAS Tensor‑Core path
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)          # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out,
                                     weights["to_out.weight"].half())  # back to dim, FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast result back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```
