Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
* **Fuse linear layers** – left_proj & left_gate, right_proj & right_gate are
  combined into two “double‑width” Linear calls. This reduces five GEMM passes
  (left, right, left_gate, right_gate, out_gate) to three (left+gate, right+gate,
  out_gate) and cuts memory traffic by ~30 %.
* **Mask‑gate fusion** – a tiny Triton kernel multiplies a projection, its gate
  (already passed through sigmoid) and the mask (broadcast over the hidden
  dimension).  When masks are disabled the operation falls back to a cheap
  element‑wise multiplication.
* **Batched GEMM for the O(N³) contraction** – the heavy “triangle” contraction
  `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]` is expressed as a
  batched matrix‑multiply `C = A @ Bᵀ`.  The batch dimension is `B·H`,
  so cuBLAS/Tensor‑Cores perform all `H` independent matrix multiplies in a
  single kernel launch.
* **Mixed precision** – everything after the first LayerNorm runs in FP16
  (Tensor‑Core friendly).  The final result is cast back to FP32 for the
  harness.

The implementation follows the reference PyTorch module exactly (no bias
terms, same order of operations) while keeping the same functional
behaviour.

    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """ out = proj * gate * mask  (mask broadcast over hidden dim H). """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # i‑, j‑coordinates for this block
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # out‑of‑bounds masks (as booleans)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – will be broadcast over H later
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16 scalar broadcast later

    # --------------------------------------------------------------
    # Loop over hidden dimension tiles (BLOCK_H)
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask (mask broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store the result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H] (fp16)
    mask       : [B, N, N]   (fp16)

    Returns out = proj * gate * mask (mask broadcast over H) as fp16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides in *elements* (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    # Tile dimensions – empirically good for H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction:  out = left @ rightᵀ
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a strided‑batched GEMM (fp16 inputs, fp16 output).

    left, right : [B, N, N, H] (contiguous, fp16)
    out         : [B, N, N, H] (contiguous, fp16)
    """
    B, N, _, H = left.shape
    # Rearrange to [B·H, N, N]  (batch dimension = B * H)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM C = A @ Bᵀ   (Tensor‑Core friendly)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Back‑to original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                         # [B, N, N, dim]  float32
    x = x.to(torch.float16)   # now fp16, ready for high‑throughput kernels

    # ------------------------------------------------------------------
    # 2️⃣ Prepare FP16 model weights (move to device & cast once)
    # ------------------------------------------------------------------
    # Fuse (proj + gate) for left and right streams
    left_proj_w = weights["left_proj.weight"]   # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"]   # (hidden_dim, dim)
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=torch.float16)

    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=torch.float16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)   # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear projections (fp16)
    # ------------------------------------------------------------------
    # left_concat: [B, N, N, 2*hidden_dim]
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # proj + gate packed
    left_proj = left_concat[..., :hidden_dim]                  # [B, N, N, hidden_dim]
    left_gate_raw = left_concat[..., hidden_dim:]              # [B, N, N, hidden_dim]

    right_concat = torch.nn.functional.linear(x, right_concat_w)
    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # out_gate (no bias) – sigmoid afterwards
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid to gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – fuse mask, projection and gate with Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask once to half‑precision
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)   # fp16
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # No mask → simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate + final linear projection back to model dimension
    # ------------------------------------------------------------------
    out = out * out_gate                       # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (the harness expects FP32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
