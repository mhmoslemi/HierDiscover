import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16: left or right tensor to be multiplied
    gate_ptr,       # *fp16: gate tensor (same shape)
    mask_ptr,       # *fp16: mask tensor of shape (B*N*N)
    hidden_dim,     # i32: hidden dimension of x
    size,           # i32: total number of elements in x (B*N*N*hidden)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    # mask index
    mask_offs = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Steps
    -----
    1. Layer‑norm the input (float32) and cast to fp16.
    2. Fuse all five per‑position linear projections (left, right, left_gate,
       right_gate, out_gate) into a single matmul.
    3. Sigmoid the three gate tensors.  Fuse element‑wise mask*gate*proj for
       left and right using a Triton kernel (if a mask is supplied).
    4. Compute the O(N³) contraction:
           out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       as a batched GEMM (cuBLAS) where each hidden channel forms an
       independent batch entry.
    5. Second layer‑norm over the hidden dimension (fp16).
    6. Multiply by the out‑gate (element‑wise) and apply the final linear
       projection back to the model dimension.
    7. Cast back to float32 for the output.

    The implementation keeps the heavy data movements to a minimum:
    * The five linear projections are performed in a single matmul,
      dramatically reducing reads of the large input tensor.
    * Masking and gating of the left/right tensors are fused in a
      Triton kernel to avoid extra passes.
    * The large N³ contraction is handled by a single batched cuBLAS GEMM.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm on the raw (float32) tensor.
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape: [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the bulk of the work.
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections into one GEMM.
    # -----------------------------------------------------------------
    # Gather the five weight matrices (each has shape [hidden_dim, dim]).
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output dimension.
    # Resulting weight shape: [5*hidden_dim, dim]
    proj_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*H, dim)

    B, N, _, _ = x.shape
    # Collapse batch and pairwise dimensions: (B*N*N, dim)
    xv = x.reshape(-1, dim)  # (M, dim) where M = B*N*N

    # Single matmul: (M, dim) @ (dim, 5*H)^T -> (M, 5*H)
    projected = torch.nn.functional.linear(xv, proj_weight)  # (M, 5*H), fp16

    # Reshape back to [B, N, N, 5*H]
    projected = projected.view(B, N, N, 5 * hidden_dim)

    # Slice out the individual tensors.
    left      = projected[...,               :hidden_dim]                # (B,N,N,H)
    right     = projected[..., hidden_dim   :2*hidden_dim]               # (B,N,N,H)
    left_gate = projected[..., 2*hidden_dim :3*hidden_dim]               # (B,N,N,H)
    right_gate= projected[..., 3*hidden_dim :4*hidden_dim]               # (B,N,N,H)
    out_gate  = projected[..., 4*hidden_dim :5*hidden_dim]               # (B,N,N,H)

    # Apply sigmoid to gate tensors (in‑place to save a temporary).
    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – fuse mask * gate * proj with Triton.
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16 and flatten.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elements = left.numel()  # B*N*N*H
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)
        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple elementwise multiplication (in‑place).
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ O(N³) contraction – batched GEMM.
    # -----------------------------------------------------------------
    # We need left @ right^T per hidden channel.
    # Bring hidden dimension into the batch axis.
    # left shape: (B, N, N, H) -> (B, H, N, N)
    # right shape: (B, N, N, H) -> (B, H, N, N) with the last two dims swapped.
    a = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    b = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    # batched matrix multiply: each batch entry is N×N multiplied by N×N
    out = torch.bmm(a, b)   # (B*H, N, N)

    # Restore the original layout.
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16) over the hidden dimension.
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Out‑gate (element‑wise) and final projection.
    # -----------------------------------------------------------------
    out = out * out_gate  # element‑wise gating

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 and return.
    # -----------------------------------------------------------------
    return out.to(torch.float32)

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) and cast to fp16.
2️⃣  Fuse the five per‑position linear projections (left, right, left_gate,
    right_gate, out_gate) into a single half‑precision GEMM.  This cuts the
    number of passes over the large input tensor from five to one.
3️⃣  Apply sigmoid to the three gate tensors.
4️⃣  If a mask is present, fuse mask, gate and projection for the left/right
    tensors in a Triton kernel (element‑wise `proj * gate * mask`);
    otherwise just multiply by the gate.
5️⃣  Heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM using cuBLAS: each hidden channel becomes
    a separate batch entry, i.e. `left → (B·H,N,N)` and `right → (B·H,N,N)`.
6️⃣  Second layer‑norm (fp16) over the hidden dimension.
7️⃣  Multiply by the out‑gate, then project back to the model dimension.
8️⃣  Cast back to float32 and return.

Only the mask‑gate fusion is written in Triton; the rest uses highly‑optimized
PyTorch kernels (cuBLAS/Tensor‑Cores).  This layout minimizes memory traffic
and keeps the dominant O(N³) step in a single batched GEMM, giving a large
speed‑up over the naïve implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  : tensor to be updated (left or right)
    gate_ptr,       # *fp16  : gate tensor (same shape as x)
    mask_ptr,       # *fp16  : mask tensor (flattened B·N·N)
    hidden_dim,     # i32    : hidden dimension H
    size,           # i32    : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,
):
    """Fused element‑wise multiplication:  out = x * gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                # linear offset
    mask = offs < size                                       # guard
    # each hidden dimension is contiguous, so divide by H to get mask index
    mask_offs = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)  # default = 1
    tl.store(x_ptr + offs, x * g * m, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) or None
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable on the raw input.
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent work.
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections into one GEMM.
    # -----------------------------------------------------------------
    # Gather weight tensors (all are bias‑free Linear layers).
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (H, dim)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (H, dim)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)    # (H, dim)

    # Concatenate along the output dimension: shape (5*H, dim).
    proj_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # (5*H, dim)

    B, N, _, _ = x.shape
    # Collapse batch & pairwise axes: (B*N*N, dim)
    x_flat = x.reshape(-1, dim)  # fp16, shape (M, dim) with M = B·N·N

    # Perform the single large matmul: (M, dim) @ (dim, 5*H)^T -> (M, 5*H)
    proj = torch.nn.functional.linear(x_flat, proj_weight)  # fp16

    # Reshape back to the 4‑D layout.
    proj = proj.view(B, N, N, 5 * hidden_dim)

    # -----------------------------------------------------------------
    # 4️⃣ Slice the fused projection into its components.
    # -----------------------------------------------------------------
    left      = proj[..., 0 : hidden_dim]                                 # (B,N,N,H)
    right     = proj[..., hidden_dim : 2 * hidden_dim]                    # (B,N,N,H)
    left_gate = proj[..., 2 * hidden_dim : 3 * hidden_dim]                # (B,N,N,H)
    right_gate= proj[..., 3 * hidden_dim : 4 * hidden_dim]                # (B,N,N,H)
    out_gate  = proj[..., 4 * hidden_dim : 5 * hidden_dim]                # (B,N,N,H)

    # Apply sigmoid to the gate tensors (in‑place to save memory).
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fuse mask, gate and projection for left/right.
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to half‑precision and flatten.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()                     # B·N·N·H
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple elementwise multiplication.
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via a batched GEMM.
    # -----------------------------------------------------------------
    # Rearrange tensors so hidden channel becomes part of the batch dimension.
    # left:  (B,N,N,H) -> (B·H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right: (B,N,N,H) -> (B·H, N, N) with the inner two axes transposed.
    b = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiply: each hidden channel computes left @ right^T.
    out = torch.bmm(a, b)  # (B·H, N, N), fp16

    # Restore original layout (B,N,N,H).
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (fp16).
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Out‑gate (element‑wise) and final projection back to `dim`.
    # -----------------------------------------------------------------
    out = out * out_gate  # Elementwise gating.
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)       # (B,N,N,dim), fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected by the harness) and return.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
