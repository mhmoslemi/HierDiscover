"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold‑3 style).

Algorithm
---------
1️⃣  First LayerNorm in FP32 (stable) → cast to FP16 for the heavy part.
2️⃣  Fuse the four linear projections (left_proj, left_gate, right_proj, right_gate)
    into a single matmul: out = x @ W_concat.T  (W_concat shape (4·H, C)).
    Split the result to obtain the four tensors.
3️⃣  Apply σ (sigmoid) to the gate tensors.
4️⃣  Fuse element‑wise multiplication of projections, gates and optional mask
    with a tiny Triton kernel (in‑place update).
5️⃣  Heavy O(N³) contraction is a batched GEMM:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
    Implemented via torch.bmm on tensors reshaped to (B·H,N,N).
6️⃣  Second LayerNorm (FP16) → element‑wise multiply by out_gate.
7️⃣  Final linear projection back to the original feature dimension.
8️⃣  Cast the result to FP32 (the harness expects float32).

The kernel `fused_gate_mask_kernel` is the only Triton kernel used;
all matmul‑heavy work is delegated to cuBLAS‑TensorCore (torch.bmm) which
is already optimal on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden
# dimension `hidden_dim`.  `total_elts` = B * N * N * hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,              # *fp16 : gate tensor, same shape as x
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate
    hidden_dim: tl.int32,  # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,   # compile‑time block size (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_elts

    # Compute mask index (mask repeats every hidden_dim entries)
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM via torch.bmm (cuBLAS).  Core O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] in FP16.
    Arguments:
        left, right : [B, N, N, H] (FP16)
    Returns:
        out : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # bring hidden dim to batch dimension: [B*H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # compute batched matmul: (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool)
        - weights_dict  : name → torch.Tensor (float32)
        - config_dict   : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the computation
    # --------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Prepare concatenated projection & gate weights (FP16)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    # concatenate in the order: left_proj, left_gate, right_proj, right_gate
    concat_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
        ],
        dim=0,
    ).to(torch.float16)  # shape (4*hidden_dim, dim)

    # --------------------------------------------------------------
    # 4️⃣ Single fused linear mapping to obtain projections & raw gates
    # --------------------------------------------------------------
    # combined shape: [B, N, N, 4*hidden_dim] (FP16)
    combined = torch.nn.functional.linear(x, concat_w)  # no bias

    # Split the concatenated tensor (all views are contiguous after split)
    left_proj = combined[..., 0:hidden_dim]                       # [B,N,N,hd]
    left_gate_raw = combined[..., hidden_dim:2 * hidden_dim]      # [B,N,N,hd]
    right_proj = combined[..., 2 * hidden_dim:3 * hidden_dim]    # [B,N,N,hd]
    right_gate_raw = combined[..., 3 * hidden_dim:4 * hidden_dim]  # [B,N,N,hd]

    # --------------------------------------------------------------
    # 5️⃣ Apply sigmoid to gate tensors (FP16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # --------------------------------------------------------------
    # 6️⃣ Compute out_gate (separate linear) – still in FP16
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 7️⃣ Optional mask – fused multiplication (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # Ensure mask is FP16 and contiguous for Triton
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                    # B * N * N * hidden_dim
        BLOCK = 1024                                       # threads per block (≤1024)
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel, left_proj and right_proj already contain the gated&masked values.
        left = left_proj
        right = right_proj
    else:
        # Simple element‑wise multiplication (no mask)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core)
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] (FP16)

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 🔟 Gating with out_gate and final projection (FP16)
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)       # -> [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
