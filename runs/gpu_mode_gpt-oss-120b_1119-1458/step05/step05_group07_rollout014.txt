def batched_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Compute global offset for batch and hidden
    batch = pid_bh // H
    hidden = pid_bh % H

    # Compute row start, col start
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Create offsets for i, j
    offs_i = i_start + tl.arange(0, BLOCK_M)  # shape block_m
    offs_j = j_start + tl.arange(0, BLOCK_N)  # shape block_n

    # mask to keep within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # load left and right tiles per k iteration
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # accumulate after sum across k for hidden

    # Loop over K dimension in BLOCK_K sized tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

@triton.jit
def out_gemm_kernel(
    left_ptr, right_ptr,
    out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)
    batch = pid_bh // H
    hidden = pid_bh % H

    # coordinate cells
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch*stride_lb + offs_i[:,None]*stride_li + offs_k[None,:]*stride_lk + hidden*stride_lh
        left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

        # Load right tile: shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_ptr + batch*stride_rb + offs_j[:,None]*stride_rj + offs_k[None,:]*stride_rk + hidden*stride_rh
        right = tl.load(right_ptrs, mask=mask_j[:,None] & mask_k[None,:], other=0.0)

        # Compute dot: left (M,K) * right^T (K,N)
        # needed right_T: shape (BLOCK_K, BLOCK_N)
        right_T = tl.trans(right)
        acc += tl.dot(left, right_T, out_dtype=tl.float32)
    # store output
    out_ptrs = out_ptr + batch*stride_ob + offs_i[:,None]*stride_oi + offs_j[None,:]*stride_oj + hidden*stride_oh
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:,None] & mask_j[None,:])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # Grab weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm (float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # 2. Cast to FP16
    x = x.half()

    # 3. Cast weights to FP16
    left_proj_w = left_proj_weight.half()
    right_proj_w = right_proj_weight.half()
    left_gate_w = left_gate_weight.half()
    right_gate_w = right_gate_weight.half()
    out_gate_w = out_gate_weight.half()
    to_out_norm_w = to_out_norm_weight.half()
    to_out_norm_b = to_out_norm_bias.half()
    to_out_w = to_out_weight.half()
    
    # 4. Linear projections
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # 5. Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # 6. Mask + gate fuse
    if not nomask:
        # mask broadcast to hidden dim, FP16.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elements = left.numel()  # B*N*N*H
        BLOCK = 1024
        grid_mask = ( (total_elements + BLOCK - 1) // BLOCK, )
        fused_gate_mask_kernel[grid_mask](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        fused_gate_mask_kernel[grid_mask](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # 7. O(N^3) contraction via custom Triton kernel
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # FP16

    # define block sizes
    BLOCK_M = 64  # tunable
    BLOCK_N = 64
    BLOCK_K = 32

    # compute stride sizes for left, right, out
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)
    stride_rk = right.stride(2)
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    # grid dims
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B * H)
    # Launch
    out_gemm_kernel[grid](

        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    return out.to(torch.float32)

"""
Fast “outgoing” TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 → cast to FP16.
2️⃣  Linear projections (left/right) and three gates (left‑gate, right‑gate,
    out‑gate) are computed in FP16.
3️⃣  If a mask is present we fuse the three element‑wise multiplies
    (tensor * gate * mask) with a tiny Triton kernel.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed with a custom Triton GEMM that treats each hidden channel
    as an independent batch element (B·H kernels).  The kernel tiles the
    i‑ and j‑axes (BLOCK_M × BLOCK_N) and accumulates over k in blocks of
    size BLOCK_K using Tensor‑Core‑compatible FP16 loads and FP32 accumulation.
5️⃣  Second Layer‑norm (FP16) → multiply by the out‑gate.
6️⃣  Final linear projection back to the original feature dimension.
7️⃣  Cast the result to FP32 for the evaluation harness.

Only the forward pass is implemented; gradients are omitted.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: (tensor * gate * mask) in‑place.
#   x:   [B, N, N, H]   (output of a linear / gate)
#   g:   [B, N, N, H]   (gate tensor)
#   m:   [B, N, N]      (mask, broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                     # *fp16
    g_ptr,                     # *fp16
    mask_ptr,                  # *fp16
    total_elts: tl.int32,      # B·N·N·H
    hidden_dim: tl.int32,      # H
    BLOCK: tl.constexpr,       # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element offset
    mask = offs < total_elts

    # mask is broadcast over the hidden dimension → one mask element per H entries
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(g_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#   left/right/out: [B, N, N, H] in fp16.
#   The kernel processes each (b, h) pair as an independent batch entry.
# ----------------------------------------------------------------------
@triton.jit
def out_gemm_kernel(
    left_ptr,     # *fp16  [B, N, N, H]
    right_ptr,    # *fp16  [B, N, N, H]
    out_ptr,      # *fp16  [B, N, N, H] (output)
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    # strides of left
    stride_lb: tl.int32, stride_li: tl.int32, stride_lk: tl.int32, stride_lh: tl.int32,
    # strides of right
    stride_rb: tl.int32, stride_rj: tl.int32, stride_rk: tl.int32, stride_rh: tl.int32,
    # strides of out
    stride_ob: tl.int32, stride_oi: tl.int32, stride_oj: tl.int32, stride_oh: tl.int32,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Program IDs
    # --------------------------------------------------------------
    pid_bh = tl.program_id(2)                 # combined batch × hidden index
    pid_i  = tl.program_id(0)                 # tile‑row (i)
    pid_j  = tl.program_id(1)                 # tile‑col (j)

    batch = pid_bh // H                       # actual batch index
    hidden = pid_bh % H                       # hidden‑channel index

    # --------------------------------------------------------------
    # Offsets for the i‑ and j‑tiles
    # --------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Accumulator (FP32 for precision)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction axis k in tiles of size BLOCK_K
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        k0 = tk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left tile (M×K) --------
        left_ptrs = left_ptr \
            + batch * stride_lb \
            + offs_i[:, None] * stride_li \
            + offs_k[None, :] * stride_lk \
            + hidden * stride_lh
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # -------- right tile (N×K) --------
        right_ptrs = right_ptr \
            + batch * stride_rb \
            + offs_j[:, None] * stride_rj \
            + offs_k[None, :] * stride_rk \
            + hidden * stride_rh
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # transpose right to (K×N) and perform the dot‑product
        # tl.dot promotes the FP16 multiplication to FP32 accumulation.
        acc += tl.dot(left, tl.trans(right), out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Store the result (cast back to FP16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr \
        + batch * stride_ob \
        + offs_i[:, None] * stride_oi \
        + offs_j[None, :] * stride_oj \
        + hidden * stride_oh
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])



# ----------------------------------------------------------------------
# Entry point for the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict of model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; may contain ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # Load weights (FP32 → will be cast to FP16 when needed)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # -> [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the computation
    # --------------------------------------------------------------
    x = x.half()   # FP16

    # --------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 once
    # --------------------------------------------------------------
    left_proj_w   = left_proj_w.half()
    right_proj_w  = right_proj_w.half()
    left_gate_w   = left_gate_w.half()
    right_gate_w  = right_gate_w.half()
    out_gate_w    = out_gate_w.half()
    to_out_norm_w = to_out_norm_w.half()
    to_out_norm_b = to_out_norm_b.half()
    to_out_w      = to_out_w.half()

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (half‑precision)
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5️⃣ Gating signals (half‑precision)
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6️⃣ Optional mask – fuse multiplication (tensor * gate * mask)
    # --------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 (broadcasted over hidden_dim)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid_mask = ((total_elts + BLOCK - 1) // BLOCK,)

        # fuse left = left * left_gate * mask
        fused_gate_mask_kernel[grid_mask](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # fuse right = right * right_gate * mask
        fused_gate_mask_kernel[grid_mask](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via custom Triton GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    out_tensor = torch.empty_like(left)   # FP16 output of shape [B, N, N, H]

    # Tile sizes – tuned for H100 Tensor‑Core usage
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides (in elements) for the three tensors
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_tensor.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,                 # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,                 # j‑tiles
        B * H                                         # batch × hidden
    )

    out_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_tensor.data_ptr(),
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16) → gating → final linear projection
    # --------------------------------------------------------------
    out_tensor = torch.nn.functional.layer_norm(
        out_tensor,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out_tensor = out_tensor * out_gate
    out_tensor = torch.nn.functional.linear(out_tensor, to_out_w)   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out_tensor.to(torch.float32)
