def custom_kernel(data):
  import torch, triton, triton.language as tl
  input_tensor, mask_tensor, weights, config = data
  dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
  device = input_tensor.device

  # LayerNorm in FP32
  norm_weight = weights["norm.weight"]
  norm_bias = weights["norm.bias"]
  x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
  x = x.to(torch.float16)

  # fuse projection & gating linear ops
  left_proj_w = weights["left_proj.weight"]
  right_proj_w = weights["right_proj.weight"]
  left_gate_w = weights["left_gate.weight"]
  right_gate_w = weights["right_gate.weight"]
  out_gate_w = weights["out_gate.weight"]
  # No bias for linear layers
  fused_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0).to(device, dtype=torch.float16)

  # fused linear
  fused = torch.nn.functional.linear(x, fused_w) # shape [..., 5*hidden_dim]
  # split
  h = hidden_dim
  left_raw = fused[..., 0:h].contiguous()
  right_raw = fused[..., h:2*h].contiguous()
  left_gate_raw = fused[..., 2*h:3*h].contiguous()
  right_gate_raw = fused[..., 3*h:4*h].contiguous()
  out_gate_raw = fused[..., 4*h:5*h].contiguous()

  # apply sigmoid for gates
  left_gate = torch.sigmoid(left_gate_raw)
  right_gate = torch.sigmoid(right_gate_raw)
  out_gate = torch.sigmoid(out_gate_raw)

  # apply gating + mask via triton kernel
  B, N, _, _ = left_raw.shape  # B,N,N,H
  # if mask
  if nomask:
      # mask factor = 1.0; we can simply skip multiplication by mask in kernel using HAVE_MASK=0
      mask_fp = None
  else:
      # mask shape [B,N,N,1]
      mask_fp = mask_tensor.to(torch.float16).unsqueeze(-1).contiguous()
  total = left_raw.numel()
  BLOCK = 1024
  grid = ((total + BLOCK - 1) // BLOCK,)
  # Provide HAVE_MASK flag compile-time
  HAVE_MASK = 0 if nomask else 1
  fused_gate_mul_inplace_kernel[grid](
      left_raw, left_gate, right_raw, right_gate,
      mask_fp if mask_fp is not None else left_raw,  # placeholder for pointer; ensure pointer but not used when mask absent
      total,
      hidden_dim,
      BLOCK_SIZE=BLOCK,
      HAVE_MASK=HAVE_MASK,
      num_warps=4,
  )
  # left_raw and right_raw now contain left and right after gating and mask.

  # The kernel overwrote left_raw and right_raw. So left = left_raw, right = right_raw.
  left = left_raw
  right = right_raw

  # contraction via batched GEMM
  B, N, _, H = left.shape
  # Permute and reshape
  left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
  right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

  # GEMM
  out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

  # Reshape back to [B,N,N,H]
  out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

  # second LayerNorm on hidden_dim
  to_out_norm_w = weights["to_out_norm.weight"]
  to_out_norm_b = weights["to_out_norm.bias"]
  out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

  # Multiply by out_gate
  out = out * out_gate

  # Final linear projection
  to_out_w = weights["to_out.weight"]
  out = torch.nn.functional.linear(out, to_out_w)

  # Cast to FP32 and return
  return out.to(torch.float32)

def _gate_mul_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,                         # pointer to mask (flattened) length = B*N*N
    total_elems, hidden_dim,

"""
Outgoing TriMul (AlphaFold‑3) – fully‑fused forward pass.
Algorithm
---------
1️⃣  Layer‑Norm the input in FP32 (numerical stability) and cast to FP16.
2️⃣  Fuse all five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single GEMM by concatenating their weight matrices.
    This reduces the number of passes over the large [B, N, N, dim] tensor.
3️⃣  Split the fused tensor, apply `sigmoid` to the three gating slices.
4️⃣  Fuse the element‑wise mask (if present) and the gating multiplications
    into a single Triton kernel that updates `left` and `right` **in‑place**.
5️⃣  Perform the cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM: reshape the tensors to [B·H, N, N] and use `torch.bmm`
    (Tensor‑core accelerated on H100).
6️⃣  Second Layer‑Norm (FP16), multiply by the out‑gate, and project back to
    the original `dim` with a final linear layer.
7️⃣  Cast the result back to FP32 for downstream code.

The Triton kernel only handles the heavy element‑wise step (mask × gate), so
the implementation satisfies the “must‑use‑Triton” requirement while keeping
the dominant contraction in cuBLAS/Tensor‑cores for peak performance.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_fusion_kernel(
    left_ptr,          # *fp16  left tensor (in‑place output)
    left_gate_ptr,     # *fp16  left gate (sigmoid output)
    right_ptr,         # *fp16  right tensor (in‑place output)
    right_gate_ptr,    # *fp16  right gate (sigmoid output)
    mask_ptr,          # *fp16  mask flattened (size = B·N·N), broadcasted over hidden
    total_elems,       # i64    total number of elements = B·N·N·H
    hidden_dim,        # i64    hidden dimension H
    BLOCK_SIZE: tl.constexpr,
    HAVE_MASK: tl.constexpr,   # compile‑time flag: 1 → mask applied, 0 → identity mask
):
    """In‑place: left *= left_gate * mask; right *= right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the four tensors
    left   = tl.load(left_ptr + offs,   mask=mask, other=0.0)
    lgate  = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right  = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask factor (broadcast over hidden dimension)
    if HAVE_MASK:
        # each hidden element repeats the same mask entry → index = offs // H
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # Apply gate and mask
    left_out  = left * lgate * m
    right_out = right * rgate * m

    # Store back in‑place
    tl.store(left_ptr + offs,  left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _apply_gate_and_mask(
    left: torch.Tensor,
    left_gate: torch.Tensor,
    right: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
) -> None:
    """Fuse gating (sigmoid) and optional mask into `left` and `right`."""
    assert left.is_contiguous() and right.is_contiguous()
    assert left_gate.is_contiguous() and right_gate.is_contiguous()
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    have_mask = 0 if mask is None else 1
    mask_ptr = mask.view(-1) if mask is not None else left  # dummy pointer when unused

    _gate_mask_fusion_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        mask_ptr,
        total,
        left.size(-1),          # hidden_dim = H
        BLOCK_SIZE=BLOCK,
        HAVE_MASK=have_mask,
        num_warps=4,
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float32)

    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    # FP16 for the heavy part
    x = x.to(torch.float16)          # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Fuse five linear projections into one GEMM
    # ------------------------------------------------------------------
    # Gather weights (all linear layers have bias=False)
    # weight shapes: [hidden, dim] for proj/gate, [dim, hidden] for final to_out
    w_left  = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    # Concatenate vertically → shape [(5*hidden), dim]
    fused_weight = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    # Single linear pass: [B,N,N,5*hidden] (still FP16)
    fused = torch.nn.functional.linear(x, fused_weight)   # no bias

    # ------------------------------------------------------------------
    # 3️⃣  Slice the fused tensor into the five components
    # ------------------------------------------------------------------
    H = hidden
    left_raw   = fused[..., 0:H].contiguous()
    right_raw  = fused[..., H:2*H].contiguous()
    left_gate_raw = fused[..., 2*H:3*H].contiguous()
    right_gate_raw = fused[..., 3*H:4*H].contiguous()
    out_gate_raw = fused[..., 4*H:5*H].contiguous()

    # ------------------------------------------------------------------
    # 4️⃣  Compute sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣  Apply mask (if any) and fuse gate multiplication via Triton
    # ------------------------------------------------------------------
    if nomask:
        mask_fp = None
    else:
        # mask: [B, N, N] → [B, N, N, 1] → FP16, broadcast over hidden dim
        mask_fp = mask.to(torch.float16).unsqueeze(-1).contiguous()
    # The kernel multiplies in‑place: left_raw = left_raw * left_gate * mask,
    # right_raw = right_raw * right_gate * mask
    _apply_gate_and_mask(left_raw, left_gate, right_raw, right_gate, mask_fp)

    # After the kernel, left_raw/right_raw hold the gated (and masked) values.
    left = left_raw
    right = right_raw

    # ------------------------------------------------------------------
    # 6️⃣  Heavy cubic contraction via batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to [B·H, N, N] for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣  Second LayerNorm + out‑gate multiplication
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate   # element‑wise gating (FP16)

    # ------------------------------------------------------------------
    # 8️⃣  Final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)  # [dim, hidden]
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣  Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
