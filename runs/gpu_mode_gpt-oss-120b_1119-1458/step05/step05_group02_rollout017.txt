```
@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    b = pid_z // H
    h = pid_z % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    batch_offset_l = b * stride_l_b
    batch_offset_r = b * stride_r_b
    batch_offset_o = b * stride_o_b

    hidden_offset_l = h * stride_l_h
    hidden_offset_r = h * stride_r_h
    hidden_offset_o = h * stride_o_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, N, BLOCK_K):
        # define offsets
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_j = j_start + tl.arange(0, BLOCK_N)
        offs_k = kk + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # left tile (M,K)
        left_ptrs = left_ptr + batch_offset_l + hidden_offset_l \
                   + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # right tile (K,N)
        right_ptrs = right_ptr + batch_offset_r + hidden_offset_r \
                    + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = right.to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)    # (M,N)

    # store result
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    out_ptrs = out_ptr + batch_offset_o + hidden_offset_o \
               + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def _trmul_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    out = torch.empty_like(left) # shape [B,N,N,H], dtype fp16

    # fetch strides in elements
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1. Fuse the five input‑side linear layers (left/right projections and their gates,
   plus the out‑gate) into a single GEMM by concatenating the weight matrices.
   This reduces kernel launch overhead and cuts the number of costly
   matrix‑multiply‑adds that read the large input tensor.
2. Perform the N³ contraction (sum_k left[...,i,k,h] * right[...,j,k,h])
   with a custom Triton kernel that works directly on the original
   [B, N, N, H] layout – removing the expensive permute+contiguous copies.
3. Keep a tiny Triton kernel for the element‑wise mask‑gate fusion.
4. All heavy compute is done in fp16 (Tensor‑Core friendly); the final
   output is cast back to fp32 as required by the harness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1️⃣ Mask‑gate fusion (left/right *= gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  : tensor to be updated (left or right)
    gate_ptr,       # *fp16  : gate tensor (same shape as x)
    mask_ptr,       # *fp16  : mask (flattened B·N·N)
    hidden_dim,     # i32    : size of hidden dimension H
    size,           # i32    : total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    active = offs < size

    # mask is broadcast over the hidden dimension
    mask_offs = offs // hidden_dim                    # integer division → mask index

    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# 2️⃣ Batched O(N³) contraction (the heavy part)
#    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs: i‑block, j‑block, batch*hidden index
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # split pid_z into batch index b and hidden channel h
    b = pid_z // H
    h = pid_z % H

    # start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # constant offsets for the current batch and hidden channel
    batch_off_l = b * stride_l_b
    batch_off_r = b * stride_r_b
    batch_off_o = b * stride_o_b
    hidden_off_l = h * stride_l_h
    hidden_off_r = h * stride_r_h
    hidden_off_o = h * stride_o_h

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for kk in range(0, N, BLOCK_K):
        # tile offsets
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_j = j_start + tl.arange(0, BLOCK_N)
        offs_k = kk    + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # ----- load left tile: shape (M, K) -----
        left_ptrs = left_ptr + batch_off_l + hidden_off_l \
                    + offs_i[:, None] * stride_l_i \
                    + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- load right tile (transposed for GEMM): shape (K, N) -----
        right_ptrs = right_ptr + batch_off_r + hidden_off_r \
                     + offs_k[:, None] * stride_r_k \
                     + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----- accumulate -----
        acc += tl.dot(left, right)   # (M, N)

    # ----- write the result -----
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    out_ptrs = out_ptr + batch_off_o + hidden_off_o \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _contract_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction using the Triton kernel defined above.
    left/right: [B, N, N, H] (fp16, contiguous)
    Returns:   out [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape

    # output tensor (in‑place layout matches the inputs)
    out = torch.empty_like(left)

    # strides in number of elements (PyTorch gives int64)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # tile sizes – tuned for fp16 Tensor‑Core use on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    trmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lsb,
        lsi,
        lsk,
        lsh,
        rsb,
        rsi,
        rsk,
        rsh,
        osb,
        osi,
        osj,
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – applied before any fp16 work.
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation.
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the five *input* linear layers into a single GEMM.
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    # concatenate along the output‑feature dimension (0)
    fused_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)   # shape [5*hidden_dim, dim]

    # single large linear projection
    fused = torch.nn.functional.linear(x, fused_weight)  # [B, N, N, 5*hidden_dim]

    # split into the five logical tensors
    split = torch.split(fused, hidden_dim, dim=-1)
    left_proj, left_gate, right_proj, right_gate, out_gate = split  # each [B,N,N,hidden]

    # ----------------------------------------------
    # 4️⃣ Activations for the gates
    # ----------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask – fused element‑wise multiplication.
    #    left = left_proj * left_gate * mask
    #    right = right_proj * right_gate * mask
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is broadcast over the hidden dim inside the kernel
        mask_fp16 = mask.to(torch.float16).contiguous()
        total = left_proj.numel()               # B·N·N·hidden
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left  = left_proj
        right = right_proj
    else:
        # No mask – simple elementwise fuse
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via the custom Triton kernel.
    # --------------------------------------------------------------
    # ensure tensors are contiguous (they already are, but be explicit)
    left = left.contiguous()
    right = right.contiguous()
    out = _contract_triton(left, right)          # [B, N, N, hidden] fp16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (over hidden dimension)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8️⃣ Gating with out_gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to the original dim.
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden]
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness and return.
    # --------------------------------------------------------------
    return out.to(torch.float32)
