"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → cast to fp16 for the heavy part.
2. Linear projections (left/right) and three gating signals (left_gate,
   right_gate, out_gate) are computed in fp16.
3. If a mask is present we fuse element‑wise multiplication
   (tensor * gate * mask) with a tiny Triton kernel (`mask_gate_fuse_kernel`).
   This satisfies the “use Triton” requirement with negligible overhead.
4. O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM `C = A @ Bᵀ` via cuBLAS (`torch.bmm`),
   after reshaping to [B*H, N, N] where H = hidden_dim.
5. Second layer‑norm (fp16) → element‑wise out‑gate → final linear projection
   back to the original channel dimension.
6. Cast the result back to float32 (the harness expects float32).

All weights are cast to fp16 once, and masks are handled in fp16.
Only the forward pass is provided – gradients are not required.
"""

import math
from typing import Tuple, Dict, Any

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused multiplication of tensor * gate * mask
# Used only when a mask is supplied (ensures at least one custom kernel).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 → tensor to be updated (flattened)
    gate_ptr,       # *fp16 → gate tensor (same flattening as x)
    mask_ptr,       # *fp16 → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → size of the hidden dimension (H)
    size,           # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    """Fuse element‑wise multiplication x = x * gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the three operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0).to(tl.float16)

    # Mask does not have the hidden dimension – compute its index.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0).to(tl.float16)  # default: 1

    # Fused multiplication
    out = x * g * m

    # Write back in‑place
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper – batched GEMM via cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single batched matrix‑multiply (A @ Bᵀ).

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (fp16, contiguous).

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and perform A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                 # [B*H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Load model weights once and cast to fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w) # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (fp16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → fp16, flattened for kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)               # shape [B*N*N]
        size = left.numel()                          # B * N * N * hidden_dim
        BLOCK = 1024  # threads per block (≤1024)
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – normal element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – batched GEMM via cuBLAS
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim]  fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                       # element‑wise out‑gate
    out = torch.nn.functional.linear(out, to_out_w)  # -> [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
