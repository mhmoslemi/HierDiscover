"""
Optimized outgoing TriMul kernel (AlphaFold‑3)

Algorithm:
1. LayerNorm on the input (float32) → cast to FP16.
2. Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single matmul: `combined = x @ W_combined.T`.
3. Split `combined` into the individual tensors and apply sigmoid to the gates.
4. If a mask is present, fuse mask multiplication, left‑gate and right‑gate
   together in a Triton kernel (in‑place).  Otherwise just multiply the gates.
5. Heavy O(N³) contraction performed as a batched GEMM:
      out = (left @ rightᵀ)  (batch dim = B·H)
   The tensors are re‑interpreted as contiguous (B·H,N,N) matrices.
6. Second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result to float32 (expected by the harness).

All large‑scale arithmetic is done in FP16 to exploit H100 Tensor‑Cores.
Only the mask‑gate fusion is implemented in a custom Triton kernel, satisfying
the requirement to “implement part of the operation in a kernel”.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left and right gates.
#   left  ← left  * mask * left_gate
#   right ← right * mask * right_gate
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in-place)
    right_ptr,         # *fp16  [B,N,N,H] (in-place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcast)
    total_elements,    # int  total number of elements in left/right tensors
    H,                 # int  hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # flat index
    mask = offs < total_elements

    # Compute index for mask (broadcast over H)
    mask_offs = offs // H          # floor division

    # Load tensors (masked loads zero out‑of‑bounds)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask (float16 0.0 or 1.0).  Use 1.0 for OOB to keep values unchanged.
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fusion
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ on FP16 tensors where
        left, right : (B, N, N, H)
    Returns:
        out : (B, N, N, H)  (FP16)
    """
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) – contiguously laid out for cuBLAS.
    # The permute makes H the second axis; a .contiguous() copy is required.
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))

    # Restore original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor   : torch.Tensor [B, N, N]      (bool/float) – may be omitted
        - weights_dict  : name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``;
                          optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)          # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – all heavy ops use half precision
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Gather and cast model weights to FP16 (single device copy)
    # ------------------------------------------------------------------
    # Combine the five linear projections (left, right, left_gate,
    # right_gate, out_gate) into one weight matrix.
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    # Each weight is (hidden_dim, dim); stack along the first axis.
    combined_weight = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)  # shape (5*hidden_dim, dim)

    w_to_out_norm = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    b_to_out_norm = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    w_to_out = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Fused linear projection: produce left/right + all gates at once
    #    Shape after linear: (B, N, N, 5*hidden_dim)
    # ------------------------------------------------------------------
    combined = F.linear(x, combined_weight)  # FP16

    # Slice the combined tensor
    H = hidden_dim
    left       = combined[..., 0:H]                # (B,N,N,H)
    right      = combined[..., H:2*H]              # (B,N,N,H)
    left_gate  = combined[..., 2*H:3*H]            # (B,N,N,H)
    right_gate = combined[..., 3*H:4*H]            # (B,N,N,H)
    out_gate   = combined[..., 4*H:5*H]            # (B,N,N,H)

    # Apply sigmoid to gates (in‑place not possible, create new tensors)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask + gate fusion
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure tensors are contiguous for Triton kernel
        left       = left.contiguous()
        right      = right.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16  = mask_tensor.to(dtype_fp16).contiguous()  # [B,N,N]

        total = left.numel()
        BLOCK = 4096  # ~4K threads per block – good occupancy
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
            num_stages=1,
        )
        # Release temporary tensors
        del left_gate, right_gate, mask_fp16
    else:
        # No mask – simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched FP16 GEMM
    # ------------------------------------------------------------------
    # out = Σ_k left[...,k,:] * right[...,k,:]  (per‑hidden GEMM)
    out = _batched_matmul_fp16(left, right)   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (over hidden dimension) + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=w_to_out_norm,
                       bias=b_to_out_norm)          # FP16

    # Apply out_gate (element‑wise)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)               # (B,N,N,dim) FP16

"""
Optimized outgoing TriMul kernel (AlphaFold‑3)

Algorithm
---------
1️⃣ First LayerNorm on the input (float32) → cast to FP16.
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single matmul: `combined = x @ W_combined.T`.
3️⃣ Split `combined` into the individual tensors and apply sigmoid to the gates.
4️⃣ If a mask is present, fuse mask multiplication, left‑gate and right‑gate
   together in a Triton kernel (in‑place).  Otherwise just multiply the gates.
5️⃣ Heavy O(N³) contraction performed as a batched GEMM:
      out = (left @ rightᵀ)  (batch dim = B·H)
   The tensors are re‑interpreted as contiguous (B·H,N,N) matrices.
6️⃣ Second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7️⃣ Cast the result to float32 (expected by the harness).

All large‑scale arithmetic is done in FP16 to exploit H100 Tensor‑Cores.
Only the mask‑gate fusion is implemented in a custom Triton kernel, satisfying
the requirement to “implement part of the operation in a kernel”.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left and right gates.
#   left  ← left  * mask * left_gate
#   right ← right * mask * right_gate
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in‑place)
    right_ptr,         # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcast)
    total_elements,    # int   total number of elements in left/right
    H,                 # int   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # flat index
    mask = offs < total_elements

    # Broadcast mask over the hidden dimension
    mask_offs = offs // H

    # Load values (out‑of‑bounds reads return 0)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask (0/1).  OOB positions get 1.0 so they don’t affect the product.
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse mask & gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ on FP16 tensors where
        left, right : (B, N, N, H)
    Returns:
        out : (B, N, N, H)  (FP16)
    """
    B, N, _, H = left.shape
    # (B,N,N,H) → (B·H,N,N) with contiguous layout for cuBLAS
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B·H,N,N) @ (B·H,N,N)ᵀ → (B·H,N,N)
    C = torch.bmm(A, B_.transpose(1, 2))

    # Restore (B,N,N,H) view
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor   : torch.Tensor [B, N, N]      (bool/float) – may be omitted
        - weights_dict  : name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``;
                          optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)          # (B,N,N,dim) float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – all heavy ops use half precision
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Gather FP16 weights (single device copy)
    #    Fuse left/right projections + three gates into one weight matrix.
    # ------------------------------------------------------------------
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    # Stack along the output‑feature dimension → shape (5*H, dim)
    combined_weight = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).to(device=device, dtype=dtype_fp16)

    w_to_out_norm = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    b_to_out_norm = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    w_to_out = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Unified linear projection → (B,N,N,5*H)
    # ------------------------------------------------------------------
    combined = F.linear(x, combined_weight)   # FP16

    H = hidden_dim
    left       = combined[..., 0:H]                # (B,N,N,H)
    right      = combined[..., H:2*H]              # (B,N,N,H)
    left_gate  = combined[..., 2*H:3*H]            # (B,N,N,H)
    right_gate = combined[..., 3*H:4*H]            # (B,N,N,H)
    out_gate   = combined[..., 4*H:5*H]            # (B,N,N,H)

    # Gating activations
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure contiguous layout for the Triton kernel
        left       = left.contiguous()
        right      = right.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16  = mask_tensor.to(dtype_fp16).contiguous()   # [B,N,N]

        total = left.numel()
        BLOCK = 4096                # ≈4 K threads per block → good occupancy
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
            num_stages=1,
        )
        # Release temporary tensors
        del left_gate, right_gate, mask_fp16
    else:
        # No mask → simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched FP16 GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (over hidden dimension) + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=w_to_out_norm,
                       bias=b_to_out_norm)          # FP16
    out = out * out_gate                                 # element‑wise

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)                       # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 – the harness expects FP32.
    # ------------------------------------------------------------------
    return out.to(torch.float32)
