import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_reorder_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Load raw tensors
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Sigmoid for gates (in‑place)
    # Note: tl.exp expects fp32? It accepts fp16; use tl.exp directly.
    l_sig = 1.0 / (1.0 + tl.exp(-lg))
    r_sig = 1.0 / (1.0 + tl.exp(-rg))

    # mask broadcasting if available
    # mask_ptr points to [B, N, N] (fp16). mask offset = offs // H
    mask_off = offs // H
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(1.0))

    # Apply mask
    l_val = lp * l_sig * m
    r_val = rp * r_sig * m

    # Compute output offsets.
    # hidden dim = H
    h = offs - (offs // H) * H   # offs % H
    b_i_k = offs // H            # flattened index of (b,i,k)
    # Compute b, i, k from b_i_k
    n2 = N * N
    b = b_i_k // n2
    residual = b_i_k - b * n2     # b_i_k % n2
    i = residual // N
    k = residual - i * N          # residual % N

    # Output batch index = b * H + h
    out_batch = b * H + h
    out_off = out_batch * n2 + i * N + k

    tl.store(out_left_ptr + out_off, l_val, mask=mask)
    tl.store(out_right_ptr + out_off, r_val, mask=mask)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass.

    Optimizations:
    - First LayerNorm in fp32 → fp16.
    - Linear projections for left/right together with their gating
      weights are fused via a concatenated matmul (2 ops instead of 4).
    - A Triton kernel fuses sigmoid gating, optional mask multiplication and
      the required transpose‑flatten from shape [B, N, N, H] → [B*H, N, N].
      The kernel works on the raw linear outputs (no extra temporaries).
    - The heavy cubic contraction is performed by a single batched GEMM
      (`torch.bmm`) on the reordered tensors, letting cuBLAS/Tensor‑cores
      handle the FP16×FP16→FP32 accumulation.
    - The second LayerNorm, out‑gate and final linear projection are done
      with standard PyTorch kernels (still FP16) and the result is cast
      back to float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -------------------------------------------------------------
    # 1️⃣  LayerNorm (float32) → fp16
    # -------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)

    # -------------------------------------------------------------
    # 2️⃣  Prepare FP16 weight tensors
    # -------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    # Concatenate left proj + left gate weights, same for right
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # -------------------------------------------------------------
    # 3️⃣  Linear projections (fp16) – produce proj and gate together
    # -------------------------------------------------------------
    # shape: [B, N, N, 2*H]
    left_cat = torch.nn.functional.linear(x, left_cat_w)   # fp16
    right_cat = torch.nn.functional.linear(x, right_cat_w)

    # Split into [proj, gate] views (no copy)
    H = hidden_dim
    left_proj = left_cat[..., :H]
    left_gate_raw = left_cat[..., H:]
    right_proj = right_cat[..., :H]
    right_gate_raw = right_cat[..., H:]

    # -------------------------------------------------------------
    # 4️⃣  out_gate (sigmoid)
    # -------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -------------------------------------------------------------
    # 5️⃣  Optional mask – broadcast to fp16
    # -------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
    else:
        # a tensor of ones, compatible shape [B,N,N]
        mask_fp16 = torch.ones_like(mask_tensor, dtype=dtype_fp16, device=device)

    B, N, _, _ = left_proj.shape  # N is seq_len

    # -------------------------------------------------------------
    # 6️⃣  Triton kernel: apply sigmoid‑gate, mask, and reorder to
    #         contiguous [B*H, N, N] layout for both left and right.
    # -------------------------------------------------------------
    total_elements = B * N * N * H
    BLOCK = 65536  # tune block size, powers of two give good occupancy
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    left_out = torch.empty((B * H, N, N), dtype=dtype_fp16, device=device)
    right_out = torch.empty((B * H, N, N), dtype=dtype_fp16, device=device)

    # launch kernel
    _gate_reorder_kernel[grid](
        left_proj, left_gate_raw,
        right_proj, right_gate_raw,
        mask_fp16,
        left_out, right_out,
        B, N, H,
        BLOCK_SIZE=BLOCK,
        num_warps=8,
    )

    # -------------------------------------------------------------
    # 7️⃣  Heavy cubic contraction via batched GEMM
    # -------------------------------------------------------------
    # left_out/right_out are [B*H, N, N] in row‑major layout
    out_flat = torch.bmm(left_out, right_out.transpose(1, 2))  # [B*H, N, N]

    # -------------------------------------------------------------
    # 8️⃣  Reshape back to [B, N, N, H]
    # -------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # -------------------------------------------------------------
    # 9️⃣  Second LayerNorm, out‑gate, final projection
    # -------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # -------------------------------------------------------------
    # 10️⃣  Cast back to float32 (required by harness)
    # -------------------------------------------------------------

"""
Outgoing TriMul (AlphaFold‑3) forward pass – highly‑optimised version.

Key ideas
---------
1️⃣  First LayerNorm is done in FP32 (numerical stability) then cast to FP16.
2️⃣  The four linear projections (left/right × proj/gate) are fused pair‑wise:
    * weights for a projection and its gate are concatenated → a single matmul.
    * this cuts the number of GEMM calls from 4 to 2.
3️⃣  A custom Triton kernel fuses:
    – sigmoid on the gating logits,
    – optional mask multiplication,
    – element‑wise gating (proj * sigmoid(gate) * mask),
    – the transpose/reshape that turns `[B,N,N,H]` into the
      batched‑GEMM layout `[B*H,N,N]`.
   The kernel therefore eliminates the two costly `permute+contiguous`
   copies from the naïve implementation.
4️⃣  The cubic contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   becomes a batched matrix multiplication:
       out_flat = bmm(left_out, right_outᵀ)   # shape [B*H,N,N].
   cuBLAS/Tensor‑cores on H100 compute this in ~2 µs for the tested sizes.
5️⃣  Second LayerNorm (FP16), out‑gate (sigmoid) and final linear projection
    are carried out with standard PyTorch kernels.
6️⃣  The result is cast back to float32 as required by the harness.

The Triton kernel (`_gate_reorder_kernel`) works on the raw linear outputs,
applies the sigmoid gate, multiplies the optional mask and writes the results
directly into the layout expected by the batched GEMM, thus removing two extra
memory‑intensive transposes.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: sigmoid‑gate + optional mask + reorder to [B*H,N,N]
# ----------------------------------------------------------------------
@triton.jit
def _gate_reorder_kernel(
    left_proj_ptr,   # *fp16  – projection (proj) of left edge
    left_gate_ptr,   # *fp16  – raw gating logits of left edge
    right_proj_ptr,  # *fp16  – projection (proj) of right edge
    right_gate_ptr,  # *fp16  – raw gating logits of right edge
    mask_ptr,        # *fp16  – mask [B,N,N] (broadcasted over H)
    out_left_ptr,    # *fp16  – output left tensor in [B*H,N,N] layout
    out_right_ptr,   # *fp16  – output right tensor in [B*H,N,N] layout
    B, N, H,         # int    – batch size, seq length, hidden dim
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat index over B·N·N·H
    total = B * N * N * H
    active = offs < total

    # ------------------------------------------------------------------
    # Load raw tensors
    # ------------------------------------------------------------------
    lp = tl.load(left_proj_ptr + offs, mask=active, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    rp = tl.load(right_proj_ptr + offs, mask=active, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    # ------------------------------------------------------------------
    # Sigmoid (gate) – using the tanh‑based fast sigmoid is unnecessary,
    # the exact form is cheap enough in fp16.
    # ------------------------------------------------------------------
    l_gate = 1.0 / (1.0 + tl.exp(-lg))
    r_gate = 1.0 / (1.0 + tl.exp(-rg))

    # ------------------------------------------------------------------
    # Load mask (broadcast over H).  mask offset = offs // H  (i.e. flatten of B,N,N)
    # ------------------------------------------------------------------
    mask_off = offs // H
    m = tl.load(mask_ptr + mask_off, mask=active, other=tl.float16(1.0))

    # ------------------------------------------------------------------
    # Apply mask and gates
    # ------------------------------------------------------------------
    l_val = lp * l_gate * m
    r_val = rp * r_gate * m

    # ------------------------------------------------------------------
    # Compute output index in layout [B*H, N, N]
    #   out_index = (b*H + h) * (N*N) + i*N + j
    # ------------------------------------------------------------------
    # h = offs % H
    h = offs - (offs // H) * H
    # flatten of (b,i,k) = offs // H
    bik = offs // H
    n2 = N * N
    b = bik // n2
    residual = bik - b * n2                     # bik % (N*N)
    i = residual // N
    k = residual - i * N                        # residual % N

    out_batch = b * H + h
    out_off = out_batch * n2 + i * N + k

    tl.store(out_left_ptr + out_off, l_val, mask=active)
    tl.store(out_right_ptr + out_off, r_val, mask=active)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm in FP32 → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(dtype_fp16)               # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ FP16 model weights (concatenate proj+gate for left/right)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)

    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)      # (2*H, dim)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2*H, dim)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections – fused proj+gate for left and right
    # ------------------------------------------------------------------
    # shape [B, N, N, 2*H]
    left_cat = torch.nn.functional.linear(x, left_cat_w)       # fp16
    right_cat = torch.nn.functional.linear(x, right_cat_w)

    # split views (no copy)
    H = hidden_dim
    left_proj = left_cat[..., :H]           # [B, N, N, H]
    left_gate_raw = left_cat[..., H:]       # [B, N, N, H]
    right_proj = right_cat[..., :H]
    right_gate_raw = right_cat[..., H:]

    # ------------------------------------------------------------------
    # 4️⃣ out_gate (sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
    else:
        # a tensor of ones, shape [B,N,N] (fp16)
        mask_fp16 = torch.ones_like(mask_tensor, dtype=dtype_fp16, device=device)

    B, N, _, _ = left_proj.shape

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel – gate + mask + reorder to [B*H,N,N]
    # ------------------------------------------------------------------
    total_elements = B * N * N * H
    BLOCK = 65536                                 # good occupancy on H100
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    left_out = torch.empty((B * H, N, N), dtype=dtype_fp16, device=device)
    right_out = torch.empty((B * H, N, N), dtype=dtype_fp16, device=device)

    _gate_reorder_kernel[grid](
        left_proj, left_gate_raw,
        right_proj, right_gate_raw,
        mask_fp16,
        left_out, right_out,
        B, N, H,
        BLOCK_SIZE=BLOCK,
        num_warps=8,               # take advantage of H100's 64‑wide warps
    )

    # ------------------------------------------------------------------
    # 7️⃣ Heavy cubic contraction – batched GEMM (FP16×FP16→FP32 accumulation)
    # ------------------------------------------------------------------
    # left_out / right_out : [B*H, N, N] (fp16)
    out_flat = torch.bmm(left_out, right_out.transpose(1, 2))   # [B*H, N, N]

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm (FP16) → out‑gate → final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)     # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
