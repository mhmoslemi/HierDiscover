  input_tensor, mask_tensor, weights, config = data
  dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
  device = input_tensor.device; dtype_fp16 = torch.float16
  
  # First layernorm fp32
  norm_w = weights["norm.weight"]; norm_b = weights["norm.bias"]
  # Cast weights to dtype float32 for LN (OK) and fp16 later
  x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
  
  # Convert to FP16 for rest
  x = x.to(dtype_fp16)
  
  # Gather and convert model weights for linear projections & gates to FP16
  left_proj_w = weights["left_proj.weight"].to(device, dtype_fp16)
  right_proj_w = weights["right_proj.weight"].to(device, dtype_fp16)
  left_gate_w = weights["left_gate.weight"].to(device, dtype_fp16)
  right_gate_w = weights["right_gate.weight"].to(device, dtype_fp16)
  out_gate_w = weights["out_gate.weight"].to(device, dtype_fp16)
  to_out_w = weights["to_out.weight"].to(device, dtype_fp16)
  to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype_fp16)
  to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype_fp16)
  
  # Fuse first linear transforms: concat all weight matrices into one big weight
  concat_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)  # (5*hidden_dim, dim)
  
  # Compute combined projection in one go
  proj = torch.nn.functional.linear(x, concat_weight)  # shape [B,N,N,5*hidden_dim] FP16
  
  # Split into respective tensors
  left, right, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(proj, hidden_dim, dim=-1)
  left_gate = torch.sigmoid(left_gate_raw)
  right_gate = torch.sigmoid(right_gate_raw)
  out_gate = torch.sigmoid(out_gate_raw)
  
  # Optionally mask + gate in Triton kernel to compute left = left * left_gate * mask, right = right * right_gate * mask.
  if not nomask:
      mask_fp16 = mask_tensor.to(dtype_fp16)  # shape [B,N,N]
      left = left.contiguous()
      right = right.contiguous()
      left_gate = left_gate.contiguous()
      right_gate = right_gate.contiguous()
      mask_fp16 = mask_fp16.contiguous()
      total_elements = left.numel()
      BLOCK = 1024
      grid = ( (total_elements + BLOCK - 1) // BLOCK, )
      _mask_gate_fusion_kernel[grid](
          left_ptr=left,
          right_ptr=right,
          left_gate_ptr=left_gate,
          right_gate_ptr=right_gate,

          H=hidden_dim,

"""
Optimized outgoing TriMul kernel (AlphaFold‑3).

Algorithm (forward only)
------------------------
1. Layer‑norm the input (FP32 → FP16) – this is the only FP32 pass.
2. Fuse the five linear projections
       left_proj, right_proj,
       left_gate, right_gate, out_gate
   into a *single* FP16 linear operation using a concatenated weight matrix.
   This reduces five passes over the large [B, N, N, dim] tensor to one.
3. Split the fused projection, apply sigmoid to the three gate tensors.
4. Optional mask + gate fusion in a tiny Triton kernel:
       left  ← left  * left_gate * mask
       right ← right * right_gate * mask
   (no‑mask case simply multiplies by the gates).
5. Outgoing multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,k,j,h]
   Implemented as a batched GEMM (left @ right) after swapping the
   two sequence dimensions of `right`.  This uses torch.bmm on FP16
   matrices with shape (B·H, N, N) – the fastest path on H100.
6. Second LayerNorm over the hidden dimension (FP16).
7. Apply the out‑gate, then a final FP16 linear projection back to `dim`.
8. Cast the result back to FP32 (the harness expects float32).

Only the mask‑gate fusion is done in a custom Triton kernel; the heavy
contraction relies on cuBLAS (via torch.bmm) which already exploits
Tensor‑cores.  The fusion of the five projections is the main source of
runtime reduction compared to a naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * left_gate * mask
#   right ← right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted over H)
    total_elements,    # i64   total number of scalar elements in the tensors
    H,                 # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat indices
    mask = offs < total_elements                                 # guard OOB

    # Determine the index for the mask (same for all hidden channels)
    mask_offs = offs // H

    # Load operands (masked load returns 0.0 / 1.0 off‑tensor when OOB)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask (float16 0.0 or 1.0).  For OOB we use neutral 1.0.
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Apply mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_outgoing_fp16(left: torch.Tensor,
                                  right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication for the *outgoing* TriMul update.

    left : [B, N, N, H] (FP16)
    right: [B, N, N, H] (FP16)

    Returns:
        out  : [B, N, N, H] (FP16) where
               out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,k,j,h]
    """
    B, N, _, H = left.shape
    # (B, H, N, N) with row‑major layout per head
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose the two N dimensions of `right` before flattening
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Batched GEMM (FP16, Tensor‑core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)               # (B·H, N, N)

    # Restore original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool or float) – ignored
                         when ``config["nomask"]`` is True.
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)          # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Prepare FP16 model weights (single‑pass linear concat)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=fp16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=fp16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=fp16)

    # Concatenate all projection weights: shape (5·hidden_dim, dim)
    concat_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*hidden_dim, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Fused linear projection (one pass over x)
    # ------------------------------------------------------------------
    # Result shape: [B, N, N, 5·hidden_dim] (FP16)
    fused = F.linear(x, concat_weight)

    # ------------------------------------------------------------------
    # 5️⃣ Split and apply gate non‑linearities
    # ------------------------------------------------------------------
    (left,
     right,
     left_gate_raw,
     right_gate_raw,
     out_gate_raw) = torch.split(fused, hidden_dim, dim=-1)

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        # Mask is broadcasted over the hidden dimension, cast to FP16.
        mask_fp16 = mask_tensor.to(fp16)           # [B, N, N]
        # Ensure tensors are contiguous for the kernel.
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates and mask have been fused in‑place.
    else:
        # No mask → just gate the projections.
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Outgoing O(N³) contraction via batched GEMM (left @ right)
    # ------------------------------------------------------------------
    out = _batched_matmul_outgoing_fp16(left, right)   # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=fp16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)   # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by out‑gate and project back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                                # element‑wise (FP16)
    to_out_w = weights["to_out.weight"].to(device, dtype=fp16)
    out = F.linear(out, to_out_w)                      # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
