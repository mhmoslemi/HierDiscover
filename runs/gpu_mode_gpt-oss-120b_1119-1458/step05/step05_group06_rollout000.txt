"""
Outgoing TriMul (AlphaFold‑3) – fast forward only.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float32).
2. Cast to half‑precision for the rest of the pipeline.
3. Linear projections (left/right) and gating (left/right/out) – all
   fp16, bias‑free as in the reference module.
4. Apply optional pairwise mask (broadcasted over the hidden dimension).
5. Fuse gates with the projected tensors.
6. Heavy O(N³) contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply C = A @ Bᵀ where each hidden
   channel d is treated as an independent batch (M = B·hidden_dim).
   For N ≤ 256 we launch a custom Triton kernel that performs the
   batched GEMM in fp16 with fp32 accumulation; for larger N we fall
   back to cuBLAS via ``torch.bmm`` (the fastest path on H100).
7. Second LayerNorm (fp16) → out‑gate → final linear projection.
8. Cast the result back to float32 for the harness.

The Triton kernel tiles the matrix multiply with 64×64×64 blocks,
uses 8 warps and 2 stages, and stores the fp32 accumulator back to
fp16.  The kernel is launched with a 3‑D grid:
    (i‑tile, j‑tile, batch·hidden)   →   M = B·hidden_dim,
    N = seq_len, K = seq_len.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,  # *fp16, [M, N, K]
    B_ptr,  # *fp16, [M, N, K]
    C_ptr,  # *fp16, [M, N, N]
    M, N, K,                         # i32 sizes
    stride_am, stride_ai, stride_ak, # strides for A (M, i, k)
    stride_bm, stride_bj, stride_bk, # strides for B (M, j, k)
    stride_cm, stride_ci, stride_cj, # strides for C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for the TriMul contraction (fp16 x fp16 → fp32)."""
    pid_m = tl.program_id(0)   # tile coordinate along i (row)
    pid_n = tl.program_id(1)   # tile coordinate along j (col)
    pid_z = tl.program_id(2)   # batch * hidden index

    # ----------- compute tile offsets ---------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # zero accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- loop over K tiles ------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i, k]  (shape BLOCK_M x BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am               # jump to the correct batch*hidden slice
            + offs_i[:, None] * stride_ai     # i dimension
            + offs_k[None, :] * stride_ak     # k dimension
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load Bᵀ[j, k] = B[k, j]  (shape BLOCK_K x BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm               # batch*hidden offset
            + offs_k[:, None] * stride_bk     # k dimension (first)
            + offs_j[None, :] * stride_bj     # j dimension (second)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ----------- write back result ------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using a Triton GEMM.
    Returns a tensor of shape [B, N, N, H] in fp16.
    """
    B, N, _, H = left.shape   # left.shape == right.shape

    # reshape to [M, N, N] where M = B * H (each hidden channel is a batch)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output buffer

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile size – 64×64 works well for N up to a few hundred on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,          # M (not actually used inside kernel; kept for API compatibility)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using cuBLAS (torch.bmm). Operates in fp16 with fp32 accumulation.
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 2️⃣ first layer‑norm (float32) – numerical stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ cast model weights to fp16 once (biases are absent for linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(dtype_fp16)

    left_gate_w   = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(dtype_fp16)

    # ------------------------------------------------------------------
    # 5️⃣ linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ optional mask – broadcast over hidden_dim
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; convert to fp16.
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Fuse gates in‑place – reduces memory pressure.
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ heavy O(N³) contraction
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    # Use Triton for modest seq_len; otherwise fall back to torch.bmm (highly tuned).
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16
    else:
        contracted = _batched_matmul_torch(left, right)    # fp16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (fp16) → out‑gate → final projection
    # ------------------------------------------------------------------
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    contracted = contracted * out_gate
    out_fp16 = F.linear(contracted, to_out_w)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 (the harness expects float32)
    # ------------------------------------------------------------------
    out = out_fp16.to(dtype_fp32)
    return out
