    hidden_dim = config["hidden_dim"]

    to_out_weight = weights["to_out.weight"]

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = left_proj_weight.to(torch.float16)
    right_proj_w = right_proj_weight.to(torch.float16)
    left_gate_w = left_gate_weight.to(torch.float16)
    right_gate_w = right_gate_weight.to(torch.float16)
    out_gate_w = out_gate_weight.to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # Gating linear and sigmoid
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

        total_elems = left.numel()

            right, right_gate, mask_fp16,

            BLOCK=BLOCK,
            num_warps=4

        right = right * right_gate

    # use custom Triton kernel rather than torch.bmm to avoid copies
    # allocate output

    out = torch.empty_like(left)  # same shape and dtype
    # compute strides

    stride_left_h = left.stride(3)  # hidden stride (should be 1)
    stride_right_b = right.stride(0)

    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)
    # out strides

    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)
    # Choose tile sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # compute grid dims

        hidden_dim=H,
        num_warps=8,   # try maximize occupancy
    )

    # second LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out gating
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_w)

    return out.to(torch.float32)

@triton.jit
def tri_mul_bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_bz, stride_left_i, stride_left_k, stride_left_h,
    stride_right_bz, stride_right_i, stride_right_k, stride_right_h,
    stride_out_bz, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden channel
    batch = pid_batch // H
    h = pid_batch % H

    # base pointer offsets
    left_ptrs = left_ptr + batch * stride_left_bz + h * stride_left_h
    right_ptrs = right_ptr + batch * stride_right_bz + h * stride_right_h
    out_ptrs = out_ptr + batch * stride_out_bz + h * stride_out_h

    # tile start
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # index ranges
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    # boundary masks
    mask_i = i_range < N
    mask_j = j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left tile
        left_offsets = left_ptrs + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        left_tile = tl.load(left_offsets, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        right_offsets = right_ptrs + k_range[:, None] * stride_right_k + j_range[None, :] * stride_right_i

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3).

1. FP32 LayerNorm on the input.
2. Cast to FP16 and apply linear projections + gating.
3. Optional mask gating fused via a tiny Triton kernel.
4. A custom batched GEMM kernel computes the O(N³) contraction
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] without any
   memory‑intensive reshape/permute copies.
5. FP16 LayerNorm on the hidden dimension, out‑gate gating and final linear.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16
    gate_ptr,       # *fp16
    mask_ptr,       # *fp16   (flattened B·N·N)
    hidden_dim,     # i32
    size,           # i32  total number of elements in x
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # mask is broadcast over the hidden dimension
    mask_offs = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


@triton.jit
def tri_mul_bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """
    Batched matrix multiplication for TriMul:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Input tensors are stored as [B, N, N, H] (row‑major, H innermost).
    The kernel treats the (b, h) pair as a combined batch dimension.
    """
    pid_batch = tl.program_id(0)   # combined batch index (b * H + h)
    pid_i = tl.program_id(1)       # tile index in i direction
    pid_j = tl.program_id(2)       # tile index in j direction

    # decode batch and hidden channel
    batch = pid_batch // HIDDEN_DIM
    hidden = pid_batch % HIDDEN_DIM

    # base pointers for this (batch, hidden) slice
    left_base = left_ptr + batch * stride_left_b + hidden * stride_left_h
    right_base = right_ptr + batch * stride_right_b + hidden * stride_right_h
    out_base = out_ptr + batch * stride_out_b + hidden * stride_out_h

    # tile start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ranges within this tile
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # boundary guards
    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator in FP32 for higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ----- left tile (M x K) -----
        # left[b, i, k, h]
        left_offs = left_base + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        left_tile = tl.load(left_offs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # ----- right tile transposed (K x N) -----
        # we need right[b, j, k, h] but as a (K, N) matrix.
        # right[b, j, k, h] -> offset = right_base + j * stride_right_i + k * stride_right_k
        # Transpose on‑the‑fly: rows = k, cols = j
        right_offs = right_base + k_range[:, None] * stride_right_k + j_range[None, :] * stride_right_i
        right_tile = tl.load(right_offs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # Accumulate the partial product
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # write back
    out_offs = out_base + i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j
    out_tile = acc.to(tl.float16)
    tl.store(out_offs, out_tile, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : dict of model parameters (float32 tensors)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable on the raw input.
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part.
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (one‑time cost).
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (half‑precision).
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 5️⃣ Gating tensors (sigmoid) – also FP16.
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fused element‑wise multiplication using Triton.
    # -----------------------------------------------------------------
    if not nomask:
        # mask broadcast over the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ O(N³) contraction – custom Triton batched GEMM, no data copies.
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # shape [B, N, N, H], dtype fp16

    # stride information
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # kernel tile sizes – tuned for H100 (fp16 tensor cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_bmm_kernel[grid](
        left, right, out,
        B, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HIDDEN_DIM=H,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm on hidden dimension (FP16).
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Gating and final projection back to original dimension.
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3).

Algorithm
---------
1️⃣  FP32 LayerNorm on the input (stable, no overflow).
2️⃣  Cast to FP16 – the bulk of the computation runs in half‑precision.
3️⃣  Linear projections (left/right) and three gate projections (no bias).
4️⃣  Optional mask‑gating fused in a tiny Triton kernel:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
5️⃣  Heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom batched‑GEMM Triton kernel that works on the
    original [B,N,N,H] layout – no costly permute + contiguous copies.
6️⃣  Second LayerNorm (FP16) over the hidden dimension.
7️⃣  Out‑gate (sigmoid) and final linear projection back to the original
    channel dimension.
8️⃣  Cast the result back to FP32 (the harness expects float32).

The implementation follows the reference PyTorch module exactly (including
the order of operations) but eliminates the expensive tensor reshapes and
adds a fused mask‑gate kernel.  It runs entirely forward‑only – no gradients
are required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1: fuse mask, gate and elementwise multiplication.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (flattened B·N·N)
    hidden_dim,     # i32   : size of the hidden dimension H
    size,           # i32   : total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,
):
    pid  = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    active = offs < size                               # bounds

    # Map element index -> mask index (broadcast over H)
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)   # mask = 1 outside

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication for the O(N³) contraction.
#          out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    # left strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # right strides
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # out strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """
    Batched GEMM kernel implementing the TriMul contraction.
    All tensors are stored in row‑major order with the hidden dimension
    as the innermost axis: [B, N, N, H].
    The (b, h) pair is treated as a combined batch index.
    """
    pid_batch = tl.program_id(0)   # combined batch index = b * H + h
    pid_i = tl.program_id(1)       # tile index along i (row) dimension
    pid_j = tl.program_id(2)       # tile index along j (col) dimension

    # decode batch and hidden channel
    batch  = pid_batch // HIDDEN_DIM
    hidden = pid_batch % HIDDEN_DIM

    # base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_left_b  + hidden * stride_left_h
    right_base = right_ptr + batch * stride_right_b + hidden * stride_right_h
    out_base   = out_ptr   + batch * stride_out_b   + hidden * stride_out_h

    # start indices for the current tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator (FP32 for better precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # -------- left tile (M × K) --------
        # left[b, i, k, h]
        left_offs = left_base + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        left_tile = tl.load(left_offs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # -------- right tile transposed (K × N) --------
        # we need right[b, j, k, h] but as a (K, N) matrix.
        # Transpose on‑the‑fly: rows = k, cols = j
        right_offs = right_base + k_range[:, None] * stride_right_k + j_range[None, :] * stride_right_i
        right_tile = tl.load(right_offs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # accumulate partial product
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # write the result back (cast to FP16)
    out_offs = out_base + i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j
    tl.store(out_offs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point: forward pass of the outgoing TriMul.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : dict of model parameters (float32 tensors)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (True → no mask)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – applied to the raw input.
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the bulk of the work.
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (one‑time cost).
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (half‑precision).
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 5️⃣ Gating tensors (sigmoid) – also FP16.
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fused element‑wise multiplication using Triton.
    # -----------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dimension inside the kernel.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple elementwise multiplication.
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – custom Triton batched GEMM.
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # shape [B, N, N, H], dtype fp16

    # strides for addressing
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # tiled GEMM parameters (tuned for H100 FP16 Tensor‑Cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_bmm_kernel[grid](
        left, right, out,
        B, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=H,
        num_warps=8,          # more warps → better occupancy for large N
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16) over the hidden dimension.
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Output gating and final projection back to original dim.
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
