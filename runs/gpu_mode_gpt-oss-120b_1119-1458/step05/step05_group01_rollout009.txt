Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm (FP16 for the bulk of the work)
-----------------------------------------
1️⃣  Layer‑norm the input in FP32 → cast to FP16.
2️⃣  Fuse the two projection linear layers (left & right) into a single
    GEMM by concatenating their weight matrices.
3️⃣  Fuse the two gate linear layers (left_gate & right_gate) similarly.
4️⃣  Compute `out_gate = sigmoid(x @ out_gate_weight)`.
5️⃣  Apply mask (if present) and fuse projection·gate·mask in a tiny
    Triton kernel (`_mask_gate_kernel`).  When there is no mask the
    element‑wise multiplication is done in PyTorch.
6️⃣  Heavy O(N³) contraction: `out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]`
    is expressed as a batched GEMM `C = A @ Bᵀ` (A,B shaped (B·D,N,N)).
    This uses torch.bmm (cuBLAS tensor‑core kernels).
7️⃣  Layer‑norm on the hidden dimension, element‑wise out‑gate scaling,
    and the final linear projection back to `dim`.
8️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is written in Triton; all GEMMs use
cuBLAS which gives the best possible throughput on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                # *fp16  [B, N, N, H]
    gate_ptr,                # *fp16  [B, N, N, H]
    mask_ptr,                # *fp16  [B, N, N]
    out_ptr,                 # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)      # batch index
    pid_i = tl.program_id(1)      # i‑tile
    pid_j = tl.program_id(2)      # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load mask (no hidden dim – broadcast later)
    mask_ptrs = (mask_ptr
                 + pid_b * stride_mask_b
                 + offs_i[:, None] * stride_mask_i
                 + offs_j[None, :] * stride_mask_j)
    mask_tile = tl.load(mask_ptrs,
                         mask=mask_i[:, None] & mask_j[None, :],
                         other=0.0)          # fp16

    # Loop over hidden‑dim tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (proj_ptr
                     + pid_b * stride_proj_b
                     + offs_i[:, None, None] * stride_proj_i
                     + offs_j[None, :, None] * stride_proj_j
                     + offs_h[None, None, :] * stride_proj_h)
        proj_tile = tl.load(proj_ptrs,
                            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                            other=0.0)      # fp16

        # Load gate tile
        gate_ptrs = (gate_ptr
                     + pid_b * stride_gate_b
                     + offs_i[:, None, None] * stride_gate_i
                     + offs_j[None, :, None] * stride_gate_j
                     + offs_h[None, None, :] * stride_gate_h)
        gate_tile = tl.load(gate_ptrs,
                            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                            other=0.0)      # fp16

        # Fusion: proj * gate * mask (mask is broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (out_ptr
                    + pid_b * stride_out_b
                    + offs_i[:, None, None] * stride_out_i
                    + offs_j[None, :, None] * stride_out_j
                    + offs_h[None, None, :] * stride_out_h)
        tl.store(out_ptrs,
                 out_tile,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors are FP16 with shapes:
        proj, gate : [B, N, N, H]
        mask       : [B, N, N] (already FP16)
    Returns a new tensor of shape [B, N, N, H] (FP16).
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out

# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction (left @ rightᵀ)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    using a strided batched GEMM (FP16 inputs/outputs).

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] in FP16 (contiguous).

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # Merge batch and head dimensions: shape (B·H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Compute C = A @ Bᵀ   (cuBLAS Tensor‑core kernel)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Entry point for the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16 for speed
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                      # [B, N, N, dim]  FP32
    x = x.to(torch.float16)  # now FP16

    # -------------------------------------------------
    # 2️⃣ Cast all model weights to FP16 (once)
    # -------------------------------------------------
    # Projection weights (left & right)
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    proj_concat_w = torch.cat([left_proj_w, right_proj_w], dim=0)   # (2·H, dim)

    # Gate weights (left_gate & right_gate)
    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    gate_concat_w = torch.cat([left_gate_w, right_gate_w], dim=0)   # (2·H, dim)

    # Out‑gate weight
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Final linear & its layer‑norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias" ].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"    ].to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (concatenated GEMMs)
    # -------------------------------------------------
    # proj_concat : [B, N, N, 2·H]
    proj_concat = torch.nn.functional.linear(x, proj_concat_w)      # no bias
    left_proj = proj_concat[..., :hidden_dim]
    right_proj = proj_concat[..., hidden_dim:]

    # -------------------------------------------------
    # 4️⃣ Gate linear (concatenated) + sigmoid
    # -------------------------------------------------
    gate_concat = torch.nn.functional.linear(x, gate_concat_w)    # no bias
    left_gate = torch.sigmoid(gate_concat[..., :hidden_dim])
    right_gate = torch.sigmoid(gate_concat[..., hidden_dim:])

    # Out‑gate (separate linear)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -------------------------------------------------
    # 5️⃣ Apply mask & element‑wise gating (Triton kernel if mask exists)
    # -------------------------------------------------
    if not nomask:
        # Convert mask to FP16 and ensure shape [B, N, N]
        mask_fp16 = mask_tensor.to(torch.float16)   # already on device
        left  = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (left @ rightᵀ)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] FP16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden dimension)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Out‑gate scaling + final linear back to model dim
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)       # [B, N, N, dim] FP16

    # -------------------------------------------------
    # 9️⃣ Cast back to FP32 (harness expects FP32)
    # -------------------------------------------------
    return out.to(torch.float32)
