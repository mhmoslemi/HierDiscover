import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    The kernel fuses:
      * optional mask multiplication
      * left/right gating via sigmoid and element‑wise scaling

    into a single Triton kernel, then computes the N³ contraction
    with a batched FP16 GEMM (torch.bmm).  All heavy arithmetic is
    performed in FP16 for Tensor‑core utilization; the final result
    is cast back to FP32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # 1️⃣ LayerNorm on the input (FP32 for numeric stability)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # 2️⃣ Cast to FP16 for the rest of the pipeline
    x = x.to(dtype_fp16)

    # 3️⃣ Cast all weights to FP16 on the same device
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # 4️⃣ Linear projections → (B,N,N,hidden_dim)
    left  = F.linear(x, left_proj_w)    # FP16
    right = F.linear(x, right_proj_w)   # FP16

    # 5️⃣ Gating projections → sigmoid(linear)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # 6️⃣ Fuse mask (if any) and gating using a Triton kernel.
    #    The kernel writes the gated values in‑place into left/right.
    if not nomask:
        # Convert mask to FP16 (0/1) and make everything contiguous.
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()

        total = left.numel()
        BLOCK = 1024    # ~1K threads per block; enough to keep the GPU busy.
        grid = ( (total + BLOCK - 1) // BLOCK, )

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # release intermediate tensors early
        del left_gate, right_gate, mask_fp16
    else:
        # In‑place gating without mask – avoids an extra allocation.
        left.mul_(left_gate)
        right.mul_(right_gate)
        del left_gate, right_gate

    # 7️⃣ O(N³) contraction via batched matrix multiplication (A @ Bᵀ).
    #    Collapse batch and head dimension to a single leading dimension.
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)      # (B·H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)     # (B·H, N, N)

    # Compute A @ Bᵀ in FP16 (Tensor‑core accelerated)
    C = torch.bmm(A, B_.transpose(1, 2))

    # Restore original layout: (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8️⃣ Second LayerNorm over hidden dimension (still FP16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # 9️⃣ Apply out‑gate and final projection back to `dim`
    out = out * out_gate
    out = F.linear(out, to_out_w)    # (B,N,N,dim) FP16

    # 10️⃣ Cast back to FP32 for the caller.
    return out.to(torch.float32)


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in‑place)
    right_ptr,         # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcasted, 0/1)
    total_elements,    # int
    H,                 # int hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask (if present) and gating into left/right tensors.

    The kernel operates on flat indices.  ``mask_ptr`` is broadcast
    across the hidden dimension, i.e. each mask element corresponds to
    ``H`` consecutive elements in the ``left``/``right`` tensors.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # flat index
    mask = offs < total_elements

    # Each hidden entry shares the mask of its (b,i,k) location.
    mask_offs = offs // H

    # Load inputs (zero‑fill OOB positions)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask; for OOB we provide neutral multiplier 1.0 (no effect)
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)

The outgoing *TriMul* operator is a three‑tensor contraction  

\[
\text{out}_{b,i,j,d}= \sum_{k} \text{left}_{b,i,k,d}\;\cdot\;
                     \text{right}_{b,j,k,d},
\qquad
\text{left}=Wx,\;\;\text{right}=Wx,
\]

followed by a second LayerNorm, an output‑gate, and a final linear
projection.  

Only the forward pass is required.  The expensive \(N^{3}\) contraction is
implemented as a batched FP16 GEMM (`torch.bmm`).  The mask handling and
the two element‑wise gates are fused into a single Triton kernel to
reduce memory traffic – this satisfies the “kernel‑implementation” rule.
All heavy arithmetic is done in FP16 to exploit H100 Tensor‑cores; the
final result is returned in FP32.

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with the per‑head gates.
#   left  <- left  * left_gate  * mask
#   right <- right * right_gate * mask
# All tensors are contiguous FP16.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # int   (B·N·N·H)
    H,                 # int   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    """
    Apply mask (if any) and the sigmoid gates to the left/right
    projections in a single pass.  The mask is broadcast over the hidden
    dimension, i.e. each mask element corresponds to `H` consecutive
    entries in the left/right tensors.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat index
    valid = offs < total_elements

    # Index of the mask entry for this hidden‑dim element
    mask_offs = offs // H

    # Load values (zero for OOB)
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    # Load mask (neutral 1.0 for OOB)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)

        - mask_tensor  : torch.Tensor [B, N, N]      (bool / 0‑1 float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;

    Returns

    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 → stable statistics)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)                 # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Convert to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast all model weights to FP16 (same device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)

    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # FP16
    right = F.linear(x, right_proj_w)   # FP16

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask (+ optional) and gates in‑place with Triton
    # ------------------------------------------------------------------
    if nomask:

        left.mul_(left_gate)

        mask_fp16 = mask_tensor.to(fp16).contiguous()
        left      = left.contiguous()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim

    C = torch.bmm(A, B_.transpose(1, 2))
