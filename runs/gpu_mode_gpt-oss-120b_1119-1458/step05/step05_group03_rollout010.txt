"""
TriMul (outgoing) – fast forward pass.

The kernel implements the following stages:
1️⃣ First LayerNorm (FP32 → FP16).
2️⃣ Linear projections (left/right) + three gating streams (FP16).
3️⃣ Optional mask‑gate fusion in a tiny Triton kernel (mask broadcast over the
   hidden dimension).  When no mask is used the multiplication is performed
   in‑place with a single torch call.
4️⃣ Heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed by a custom Triton batched‑GEMM kernel.  The tensors are first
   reshaped to [B, H, N, N] (contiguous) so that each (b, h) pair is treated
   as an independent matrix‑multiply A @ Bᵀ with A,B ∈ ℝ^{N×N} (FP16 inputs,
   FP32 accumulation).  BLOCK_M = BLOCK_N = BLOCK_K = 64 gives good occupancy
   on an H100.
5️⃣ Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection
   back to the original channel dimension.
6️⃣ Result is cast to float32 (the harness expects fp32).

Only the forward pass is needed – no gradients are computed.
At least one Triton kernel (mask‑gate fusion) and the custom batched GEMM are
implemented.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over
# hidden_dim.  The kernel works on a flat view of the 4‑D tensor
# `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,        # *fp16 : tensor to be updated (size = B*N*N*H)
    gate_ptr,     # *fp16 : gate tensor (same shape as x)
    mask_ptr,     # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,         # i32   : total number of elements in x
    hidden_dim,   # i32   : width of the hidden dimension (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear block id
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask = offs < size

    # Load the data and gate
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Every `hidden_dim` consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse multiplication (x * g * m)
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# After permutation to (B, H, N, N) the operation reduces to a set of
# independent mat‑muls C = A @ Bᵀ for each (b,h).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_contract_kernel(
    left_ptr,       # *fp16 : left tensor  [B, H, N, N] (contiguous)
    right_ptr,      # *fp16 : right tensor [B, H, N, N] (contiguous)
    out_ptr,        # *fp16 : output tensor [B, H, N, N] (contiguous)
    B, H, N,        # i32   : batch, hidden, sequence length
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel.  Each program instance works on one (b, h) pair
    and a BLOCK_M × BLOCK_N tile of the output matrix."""
    pid_m = tl.program_id(0)          # tile row index
    pid_n = tl.program_id(1)          # tile column index
    pid_bh = tl.program_id(2)         # combined batch & head index

    # Decode batch and head from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # Offsets of the output tile
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N
    i_offs = i_base + tl.arange(0, BLOCK_M)
    j_offs = j_base + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_i = i_offs < N
    mask_j = j_offs < N

    # Accumulator in FP32 for numerical stability (Tensor‑Core accumulation is FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        k_offs = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # -----------------------------
        # Load A tile: left[b, h, i, k]
        # -----------------------------
        a_ptrs = (left_ptr
                  + b * stride_left_b
                  + h * stride_left_h
                  + i_offs[:, None] * stride_left_i
                  + k_offs[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)  # fp16

        # -------------------------------------------------
        # Load Bᵀ tile: we need right[b, h, j, k] transposed
        # Bᵀ[k, j] = right[b, h, j, k]
        # -------------------------------------------------
        b_ptrs = (right_ptr
                  + b * stride_right_b
                  + h * stride_right_h
                  + k_offs[:, None] * stride_right_k    # row dimension = k  (stride_k = 1)
                  + j_offs[None, :] * stride_right_i)   # col dimension = j  (stride_i = N)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)  # fp16

        # -----------------------------
        # Dot product over K dimension
        # -----------------------------
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)  (FP32 accumulation)

    # -----------------------------
    # Store the result tile
    # -----------------------------
    out_ptrs = (out_ptr
                + b * stride_out_b
                + h * stride_out_h
                + i_offs[:, None] * stride_out_i
                + j_offs[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point – the harness calls custom_kernel(data)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim]  (fp32)
        - mask_tensor  : torch.Tensor   [B, N, N]      (fp32/bool) – ignored if
                           config['nomask'] is True
        - weights_dict : mapping name → torch.Tensor (fp32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (fp32, same device as input).
    """
    # -------------------------------------------------
    # 0️⃣ Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]       # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  fp32
    x = x_norm.half()  # cast once to fp16

    # -------------------------------------------------
    # 2️⃣ Cast weight tensors to fp16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left  = torch.nn.functional.linear(x, left_proj_w)   # left_proj
    right = torch.nn.functional.linear(x, right_proj_w) # right_proj

    # -------------------------------------------------
    # 4️⃣ Gating streams (sigmoid, fp16)
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -------------------------------------------------
    # 5️⃣ Apply mask (if any) and fuse multiplication with gating
    # -------------------------------------------------
    if nomask:
        # Simple in‑place multiplication
        left = left.mul_(left_gate)
        right = right.mul_(right_gate)
    else:
        # Broadcast mask over hidden_dim inside the Triton kernel.
        # Flatten mask to (B*N*N,) and promote to fp16.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        total_elements = left.numel()           # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # gate tensors are no longer needed
        del left_gate, right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy contraction: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # -------------------------------------------------
    # Rearrange tensors to [B, H, N, N] (contiguous) for the Triton GEMM.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, N, _, _ = input_tensor.shape  # original bs, seq_len
    # Allocate output in the same layout
    out_perm = torch.empty_like(left_perm)  # [B, H, N, N] fp16

    # Retrieve strides (they are standard after .contiguous())
    stride_left_b, stride_left_h, stride_left_i, stride_left_k = left_perm.stride()
    stride_right_b, stride_right_h, stride_right_i, stride_right_k = right_perm.stride()
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out_perm.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_contract_kernel[grid](
        left_ptr=left_perm,
        right_ptr=right_perm,
        out_ptr=out_perm,
        B=B,
        H=hidden_dim,
        N=N,
        stride_left_b=stride_left_b,
        stride_left_h=stride_left_h,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_right_b=stride_right_b,
        stride_right_h=stride_right_h,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # 8 warps gives good occupancy on H100
    )

    # -------------------------------------------------
    # 7️⃣ Reshape to the original layout [B, N, N, H] for the second LN
    # -------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim] (fp16)

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate                     # element‑wise gating (fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # -------------------------------------------------
    # 🔟 Cast back to float32 for the benchmark harness
    # -------------------------------------------------
    return out.float()
```
