import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,         # *fp16
    gate_ptr,      # *fp16
    mask_ptr,      # *fp16 (flattened, size = B*seq_len*seq_len)
    hidden_dim: tl.constexpr,   # compile-time? Actually hidden_dim may be runtime but we can mark as constexpr causing compile for each value.
    size: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = offs < size

    # Load
    x = tl.load(x_ptr + offs, mask=mask_valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)

    # compute mask index
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask_valid)

def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the core TriMul contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Implemented via torch.bmm on a reshaped view.
    """
    B, N, _, H = left.shape
    # bring hidden dim to batch axis and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # batched matrix multiplication: A @ B^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).

    Steps:
    1. Layer‑norm the 4‑D input (FP32) and cast to FP16.
    2. Fuse left/right projections with their gates in a single linear call per side.
    3. Fuse element‑wise multiplication (projection * gate * optional mask) using a Triton kernel.
    4. Compute the heavy O(N³) contraction with a batched GEMM (torch.bmm).
    5. Second Layer‑norm over hidden dimension, element‑wise out‑gate, and final linear projection.
    6. Return FP32 output.
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] FP32

    # Cast to fp16 for heavy work
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Load model weights and cast to fp16
    # -----------------------------------------------------------------
    # Linear projection + gate weights (concatenated)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden_dim, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden_dim, dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections + gate logits (single call per side)
    # -----------------------------------------------------------------
    # Shapes after linear: [B, N, N, 2*hidden_dim]
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # fp16
    right_concat = torch.nn.functional.linear(x, right_concat_w)  # fp16

    # Split projection and gate parts
    left_proj = left_concat[..., :hidden_dim]                     # fp16
    left_gate_logit = left_concat[..., hidden_dim:]               # fp16
    right_proj = right_concat[..., :hidden_dim]
    right_gate_logit = right_concat[..., hidden_dim:]

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_logit)   # fp16
    right_gate = torch.sigmoid(right_gate_logit) # fp16

    # Out gate (different linear)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4️⃣ Fuse mask (if any) and element‑wise multiplication using Triton
    # -----------------------------------------------------------------
    if nomask:
        # create a dummy mask of ones (fp16)
        mask_fp16 = torch.ones(
            input_tensor.shape[0],
            input_tensor.shape[1],
            input_tensor.shape[2],
            dtype=torch.float16,
            device=device,
        )
    else:
        mask_fp16 = mask_tensor.to(torch.float16)

    # flatten mask
    mask_flat = mask_fp16.view(-1)  # [B*N*N]

    # total number of elements in the tensors to be masked
    size = left_proj.numel()  # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(size, BLOCK),)

    # left = left_proj * left_gate * mask   (in‑place)
    mask_gate_fuse_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        hidden_dim,
        size,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right = right_proj * right_gate * mask   (in‑place)
    mask_gate_fuse_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        hidden_dim,
        size,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # Now left_proj and right_proj contain the gated & masked tensors.
    left = left_proj
    right = right_proj

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (batched GEMM)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim] fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Final gating and projection
    # -----------------------------------------------------------------
    out = out * out_gate  # element‑wise gating (broadcasted)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) multiplication.
# Used for both left‑ and right‑side tensors.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 – tensor to be updated (flattened)
    gate_ptr,       # *fp16 – gate tensor (same flattening as x)
    mask_ptr,       # *fp16 – mask tensor flattened, shape [B*seq*seq]
    hidden_dim,     # i32   – size of the hidden dimension (H)
    size,           # i32   – total number of elements in x (B*seq*seq*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_ok = offs < size

    # Load operands
    x   = tl.load(x_ptr   + offs, mask=mask_ok, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=mask_ok, other=0.0)

    # one mask value per (b,i,k) – each hidden entry re‑uses it
    mask_idx = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_idx, mask=mask_ok, other=1.0)   # default mask = 1

    # Fused multiplication
    out = x * g * m

    # Write result back in‑place
    tl.store(x_ptr + offs, out, mask=mask_ok)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Operates on FP16 tensors of shape [B, N, N, H].
    Returns a tensor of the same shape (FP16) containing the contracted result.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each (b,h) pair becomes an independent GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat   = left_perm.view(B * H, N, N)
    right_mat  = right_perm.view(B * H, N, N)

    # A @ Bᵀ  (the contraction uses the transpose of right)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore [B, N, N, H] layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point – the forward pass of the *outgoing* TriMul operator.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input  [B, N, N, dim] (float32)
        torch.Tensor,               # mask   [B, N, N]    (float32 / bool) – optional
        Dict[str, torch.Tensor],    # weights dictionary (float32)
        Dict[str, Any],             # config dict – must contain dim & hidden_dim
    ]
) -> torch.Tensor:
    """
    Outgoing TriMul forward (no autograd).

    Algorithm
    ----------
    1. Layer‑norm the 4‑D input (float32) → cast to fp16.
    2. Concatenate the projection and gate weight matrices for the left and
       right sides and perform a *single* linear call per side.
    3. Split the result, apply sigmoid to the gate part and fuse
       (proj * gate * optional mask) with a tiny Triton kernel.
    4. Perform the heavy O(N³) contraction via a batched GEMM
       (torch.bmm on a reshaped view).
    5. Second Layer‑norm over the hidden dimension (fp16), element‑wise out‑gate,
       and final linear projection back to the original channel dimension.
    6. Cast the result to float32 for the harness.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable before casting down
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – float32

    # ---------------------------------
    # Cast to fp16 for the bulk work
    # ---------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Load model weights (cast once to fp16)
    # -----------------------------------------------------------------
    # Left side: (proj | gate) – shape (2*hidden_dim, dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    # Right side: (proj | gate) – shape (2*hidden_dim, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    out_gate_w      = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w   = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b   = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w        = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projection + gate logits (single call per side)
    # -----------------------------------------------------------------
    # Shape after linear: [B, N, N, 2*hidden_dim]
    left_concat  = torch.nn.functional.linear(x, left_concat_w)   # fp16
    right_concat = torch.nn.functional.linear(x, right_concat_w)  # fp16

    # Split projection and gate parts
    left_proj        = left_concat[..., :hidden_dim]          # fp16
    left_gate_logit  = left_concat[..., hidden_dim:]          # fp16
    right_proj       = right_concat[..., :hidden_dim]         # fp16
    right_gate_logit = right_concat[..., hidden_dim:]         # fp16

    # Sigmoid on gates
    left_gate  = torch.sigmoid(left_gate_logit)   # fp16
    right_gate = torch.sigmoid(right_gate_logit)  # fp16

    # Out‑gate (different linear)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # fp16

    # -----------------------------------------------------------------
    # 4️⃣ Fuse mask (if any) and element‑wise multiply (proj * gate)
    # -----------------------------------------------------------------
    if nomask:
        # Dummy mask of ones – still runs through the Triton kernel
        mask_fp16 = torch.ones(
            input_tensor.shape[0],
            input_tensor.shape[1],
            input_tensor.shape[2],
            dtype=torch.float16,
            device=device,
        )
    else:
        mask_fp16 = mask_tensor.to(torch.float16)

    mask_flat = mask_fp16.view(-1)          # [B * N * N]
    total_elems = left_proj.numel()         # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    # left = left_proj * left_gate * mask  (in‑place)
    mask_gate_fuse_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right = right_proj * right_gate * mask  (in‑place)
    mask_gate_fuse_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # After the kernel left_proj/right_proj contain the gated & masked tensors
    left  = left_proj
    right = right_proj

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (batched GEMM)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim] – fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Final gating & projection
    # -----------------------------------------------------------------
    out = out * out_gate                     # element‑wise out‑gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] – fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
