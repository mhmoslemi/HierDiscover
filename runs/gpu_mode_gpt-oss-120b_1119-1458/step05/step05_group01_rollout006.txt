"""
Fast outgoing TriMul forward pass (AlphaFold‑3 style).

Algorithm
---------
1️⃣ First layer‑norm (FP32) → cast to FP16.
2️⃣ Fuse the two linear projections *and* their respective gating linear layers:
       [proj, gate] = Linear(x, concat(proj_W, gate_W))
   Split the result, apply sigmoid to the gate branch.
3️⃣ If a mask is present, apply mask, projection and gate together in a tiny
   Triton kernel (proj * gate * mask).  Otherwise just multiply the gate.
4️⃣ O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented with a single `torch.einsum` (half‑precision) – this avoids
   the costly permute‑+‑contiguous copy required for a batched GEMM.
5️⃣ Second LayerNorm (FP16) → out‑gate (sigmoid) → element‑wise multiply.
6️⃣ Final linear projection back to model dimension.
7️⃣ Cast back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is written in Triton; the heavy O(N³) part uses a
high‑performance Torch einsum which internally maps to a fused kernel on
H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                # *fp16  [B, N, N, H]
    gate_ptr,                # *fp16  [B, N, N, H]
    mask_ptr,                # *fp16  [B, N, N]
    out_ptr,                 # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute: out = proj * gate * mask   (mask broadcast over hidden dim H).
    """
    pid_b = tl.program_id(0)   # batch block
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles (BLOCK_H)
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fuse: proj * gate * mask (mask broadcast on H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and mask: out = proj * gate * mask.
    All tensors are FP16; `mask` has shape [B,N,N] and is broadcast
    over the hidden dimension.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides in elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # FP32
    x = x.to(torch.float16)  # FP16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2️⃣ Combine projection + gating linear layers (FP16)
    # ------------------------------------------------------------------
    # Left side: (proj, gate)
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # [2*H, dim]

    # Right side: (proj, gate)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # [2*H, dim]

    # Out‑gate (produces hidden_dim values)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Final linear back to model dim
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear + gate splits (single matmul per side)
    # ------------------------------------------------------------------
    # left_comb: [B,N,N,2*H]  -> split into proj & raw_gate
    left_comb = torch.nn.functional.linear(x, left_comb_w)  # (B,N,N,2*H)
    left_proj, left_gate_raw = torch.split(left_comb, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_raw)  # (B,N,N,H) FP16

    # right_comb: same pattern
    right_comb = torch.nn.functional.linear(x, right_comb_w)
    right_proj, right_gate_raw = torch.split(right_comb, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask & gate fusion
    # ------------------------------------------------------------------
    if not nomask:
        # Cast mask to FP16 once
        mask_fp16 = mask_tensor.to(dtype=torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (FP16) via a single einsum
    # ------------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate + final projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # (B,N,N,H)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
