    proj_ptr, gate_ptr, mask_ptr, out_ptr, N, has_mask: tl.constexpr, BLOCK_SIZE: tl.constexpr):

        out = proj * gate
    tl.store(out_ptr + offs, out, mask=mask)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32 for stability) → FP16.
2️⃣ Fuse the two linear projections (proj + gate) for the left and right
   branches by concatenating the weight matrices and calling a single
   `torch.nn.functional.linear`.  This reduces the number of GEMM calls
   from 5 → 3.
3️⃣ Apply the sigmoid to the gate tensors.
4️⃣ In‑place Triton kernel fuses the element‑wise product
      proj ← proj * gate * mask
   (mask is either the provided mask or a tensor of ones).  One launch per
   branch only.
5️⃣ Heavy O(N³) contraction is performed as a batched GEMM:
      out[b·h, i, j] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   using `torch.bmm` after reshaping to [B·H, N, N].
6️⃣ Second LayerNorm, out‑gate (sigmoid) and final projection back to `dim`.
7️⃣ Cast the result back to float32.

All heavy element‑wise work is done by a tiny Triton kernel; the large
matrix multiplications rely on cuBLAS/Tensor‑cores (fast on H100).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: proj ← proj * gate * mask  (in‑place)
# --------------------------------------------------------------
@triton.jit
def _gate_mask_mul_kernel(
    proj_ptr,          # *fp16 – writable
    gate_ptr,          # *fp16 – read‑only
    mask_ptr,          # *fp16 – read‑only (broadcasted)
    total_elements,    # int64 – total number of elements in proj/gate
    hidden_dim,        # int64 – size of the last dimension (gate repeats every hidden_dim)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # Load the three tensors
    proj = tl.load(proj_ptr + offs, mask=active, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=active, other=tl.float16(0.0))

    # mask is stored without the hidden dimension – repeat every hidden_dim entries
    mask_idx = offs // hidden_dim            # integer division
    mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(1.0))

    # Fuse the three multiplications
    out = proj * gate * mask_val

    tl.store(proj_ptr + offs, out, mask=active)


def _apply_gate_and_mask_inplace(proj: torch.Tensor,
                                 gate: torch.Tensor,
                                 mask: torch.Tensor,
                                 hidden_dim: int) -> None:
    """
    In‑place: proj ← proj * gate * mask.
    All tensors must be contiguous FP16 tensors.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _gate_mask_mul_kernel[grid](
        proj,
        gate,
        mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float32) – may be None
        - weights_dict : mapping from weight name → torch.Tensor (float32)
        - config_dict  : must contain “dim” and “hidden_dim”; optional key
                         “nomask” (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)               # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Weight preparation (FP16, on‑device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear: (proj + gate) for left and right branches
    # ------------------------------------------------------------------
    # concatenate proj & gate weights: shape [2*hidden_dim, dim]
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # [2*H, dim]
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Linear -> [B, N, N, 2*hidden_dim]
    left_cat = torch.nn.functional.linear(x, left_cat_w)   # fp16
    right_cat = torch.nn.functional.linear(x, right_cat_w)

    # Split into proj / gate and apply sigmoid to gates
    left_proj = left_cat[..., :hidden_dim]                # [B,N,N,H]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])  # [B,N,N,H]

    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    if nomask:
        # tensor of ones, shape [B, N, N]
        mask_fp16 = torch.ones_like(mask_tensor, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16)  # already [B,N,N]
    mask_fp16 = mask_fp16.contiguous()            # ensure contiguous for Triton

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gating & optional mask via Triton (in‑place)
    # ------------------------------------------------------------------
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()

    _apply_gate_and_mask_inplace(left_proj, left_gate, mask_fp16, hidden_dim)
    _apply_gate_and_mask_inplace(right_proj, right_gate, mask_fp16, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate (computed from the normalized input, before contraction)
    # ------------------------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N, hidden_dim]
    out_gate = torch.sigmoid(out_gate_pre)                   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cubic contraction: batched GEMM (B·H, N, N) × (B·H, N, N)^T
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape  # H == hidden_dim
    # reshape to [B·H, N, N]
    left_flat  = left_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_flat[b·h, i, j] = Σₖ left_flat[b·h, i, k] * right_flat[b·h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                     # element‑wise product (fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Return in float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
