4️⃣  If a mask is supplied, fuse ``tensor * gate * mask`` with one Triton
    kernel per side (no extra memory traffic).
5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    performed as a batched GEMM using torch.bmm (cuBLAS/Tensor‑Core).
6️⃣  Second LayerNorm (fp16) → multiply by the out‑gate.
7️⃣  Final projection back to the original feature dimension.
8️⃣  Cast the result to float32 (required by the test harness).

Only the forward pass is needed – gradients are omitted.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden dim.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,              # *fp16 : gate tensor, same shape as x
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate (=B*N*N*H)
    hidden_dim: tl.int32,  # size of the hidden dimension (last axis of x)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elts

    # every `hidden_dim` entries belong to the same mask element
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, H] (fp16)
    Returns    : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] (each hidden channel becomes a batch item)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout -> [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict  : name → torch.Tensor (float32)
        - config_dict   : contains ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]          # (dim,)
    norm_bias   = weights["norm.bias"]            # (dim,)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  FP32

    x = x_norm.to(torch.float16)                  # cast once for all heavy work

    # ------------------------------------------------------------------
    # 2️⃣ Fuse proj + gate for left & right (single GEMM each)
    # ------------------------------------------------------------------
    # retrieve and cast weights
    left_proj_w  = weights["left_proj.weight"]    # (hidden_dim, dim)
    left_gate_w  = weights["left_gate.weight"]    # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]    # (hidden_dim, dim)

    # Concatenate proj & gate weights for a single matmul per side
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16) # (2*H, dim)

    # Flatten to (B*N*N, dim) – view, no copy
    B, N, _, _ = input_tensor.shape
    x_flat = x.view(-1, dim)   # shape (B*N*N, dim)

    # Linear → (B*N*N, 2*hidden_dim)
    left_comb = torch.nn.functional.linear(x_flat, left_comb_w)   # FP16
    right_comb = torch.nn.functional.linear(x_flat, right_comb_w)

    # Split proj/gate halves and reshape back to 4‑D
    left = left_comb[:, :hidden_dim].view(B, N, N, hidden_dim)            # projection
    left_gate_raw = left_comb[:, hidden_dim:].view(B, N, N, hidden_dim)   # pre‑sigmoid
    right = right_comb[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate_raw = right_comb[:, hidden_dim:].view(B, N, N, hidden_dim)

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 3️⃣ Compute out‑gate (from the same normalized input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w.to(torch.float16)))  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – fuse with the gating scalars via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask -> fp16 and make contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()                      # B*N*N*hidden_dim
        BLOCK = 1024                                    # ≤ 1024 threads / block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiply
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑Core)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # -> [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
