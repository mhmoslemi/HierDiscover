"""
TriMul (outgoing) – fast forward pass implemented with Triton + PyTorch.

Algorithm
---------
1. Layer‑norm the input (float32) → cast to fp16 for the heavy part.
2. Fuse linear projections and gating:
   * left  = x @ W_left_proj          (dim → hidden)
   * left_gate = sigmoid(x @ W_left_gate)
   * out_gate  = sigmoid(x @ W_out_gate)
   * right = x @ W_right_proj
   * right_gate = sigmoid(x @ W_right_gate)
   The three left‑side linear layers are merged in a single matmul
   (weights are concatenated) to avoid repeated reads of the large
   tensor *x*.  The same is done for the right side.
3. If a mask is present, fuse `tensor * gate * mask` for *left* and
   *right* with a tiny Triton kernel (`mask_gate_fuse_kernel`).  For
   the “nomask” case we simply multiply by the gates.
4. Compute the O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM using cuBLAS (torch.bmm) on a view
   (B*hidden, N, N).  This avoids any Python loops.
5. Second LayerNorm (fp16) → element‑wise multiplication by *out_gate*.
6. Final linear projection back to the original channel dimension.
7. Cast the result back to float32 (the harness expects float32).

All linear weights are cast to fp16 once; the mask and gates are kept
in fp16 throughout.  The only Triton kernel is the mask‑gate fuse,
which satisfies the “use Triton” requirement while adding negligible
overhead.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused multiplication of {tensor * gate * mask}
# Used only when a mask is supplied (ensures at least one custom kernel).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → size of the hidden dimension (H)
    size,           # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # One mask element per hidden‑dim block
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default mask=1

    # Fused multiplication
    out = x * g * m

    # Write back in‑place
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper – batched GEMM via cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single batched matrix‑multiply (cuBLAS).

    Tensors are expected to be fp16 and contiguous.
    """
    B, N, _, H = left.shape  # left.shape == (B, N, N, H)
    # reshape to (B*H, N, N) and perform A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # torch.bmm computes (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout: (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Concatenate linear weights and cast them once to fp16
    # -----------------------------------------------------------------
    # left side: proj + gate + out_gate (3 * hidden_dim rows)
    left_concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (3*H, dim)

    # right side: proj + gate (2 * hidden_dim rows)
    right_concat_w = torch.cat(
        [
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (2*H, dim)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fused)
    # -----------------------------------------------------------------
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # [B,N,N,3*H]
    right_concat = torch.nn.functional.linear(x, right_concat_w) # [B,N,N,2*H]

    # -----------------------------------------------------------------
    # 5️⃣ Slice and apply sigmoid to gating tensors
    # -----------------------------------------------------------------
    # left side
    left = left_concat[..., :hidden_dim]                        # proj
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:2 * hidden_dim])
    out_gate = torch.sigmoid(left_concat[..., 2 * hidden_dim:])  # shape [B,N,N,H]

    # right side
    right = right_concat[..., :hidden_dim]                      # proj
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:]) # shape [B,N,N,H]

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)          # shape [B*N*N]
        size = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just apply the gates
        left = torch.mul(left, left_gate)   # in‑place could be used via out=left
        right = torch.mul(right, right_gate)

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – batched GEMM via cuBLAS
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                         # element‑wise out‑gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
