"""
TRITON‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input in FP32 → cast to FP16.
2️⃣  Fuse each projection + its gate weight into a single linear layer
    (2× faster than two separate matmuls).  Apply sigmoid to the gate
    part of the result.
3️⃣  Optional per‑pair mask: we fuse `proj * gate * mask` in a tiny
    Triton kernel (`_mask_gate_kernel`).  When there is no mask we simply
    multiply the gate in‑place.
4️⃣  Heavy O(N³) contraction  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is expressed as a batched GEMM  C = A @ Bᵀ  on FP16 tensors.  The
    tensors are reshaped to [B·H, N, N] (contiguous) and evaluated with
    cuBLAS tensor‑core kernels (`torch.bmm`).  This avoids any custom
    Triton GEMM while still being memory‑efficient.
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection
    back to the model dimension.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is written in Triton – the rest stays in PyTorch
but uses fused linear layers and a strided batched GEMM to keep the total
runtime well below the 1 ms target on all benchmark shapes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                # *fp16  [B, N, N, H]
    gate_ptr,                # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,                # *fp16  [B, N, N]
    out_ptr,                 # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out = proj * gate * mask    (mask broadcast over hidden dim H).
    All tensors are in FP16.
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # -------------------- index tiles --------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------- load mask (BLOCK_I × BLOCK_J) --------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # ----------------------------------- hidden dim loop ------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ----------------------- proj -----------------------
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # ----------------------- gate -----------------------
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # ------------------- fusion -------------------------
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # ------------------- store -------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Fuse ``proj * gate * mask`` in FP16 using Triton."""
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides in *elements* (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # tile size – chosen to keep shared memory < 32 KB
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a strided‑batched GEMM (cuBLAS tensor‑core friendly).

    Arguments
    ---------
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns
    -------
    out : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # C = A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;

    torch.Tensor

    """

    # Unpack inputs
    # -------------------------------------------------

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                       # [B, N, N, dim]  float32
    x = x.to(torch.float16)  # now fp16

    # -------------------------------------------------
    # 2️⃣ Fuse projection + gate for *left* and *right*
    # -------------------------------------------------
    # concat weight: [proj ; gate]  (shape: 2·hidden_dim × dim)
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)

    left_concat_w = left_concat_w.to(device=device, dtype=torch.float16)
    right_concat_w = right_concat_w.to(device=device, dtype=torch.float16)

    # Linear produces [B, N, N, 2·hidden_dim]
    left_combined = torch.nn.functional.linear(x, left_concat_w)    # fp16
    right_combined = torch.nn.functional.linear(x, right_concat_w) # fp16

    # Split into projection & gate parts
    left_proj, left_gate_pre = left_combined.split([hidden_dim, hidden_dim], dim=-1)
    right_proj, right_gate_pre = right_combined.split([hidden_dim, hidden_dim], dim=-1)

    left_gate = torch.sigmoid(left_gate_pre)   # fp16
    right_gate = torch.sigmoid(right_gate_pre) # fp16

    # -------------------------------------------------
    # 3️⃣ Optional mask – fuse mask, projection and gate with Triton
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)   # [B, N, N]

        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:

        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # -------------------------------------------------

    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # fp16

    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # fp16, shape [B, N, N, dim]

    # -------------------------------------------------
    # 7️⃣ Cast back to FP32 (expected by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
```
