"""
Fast outgoing TriMul implementation (AlphaFold‑3) – fused projection/gating/mask.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input in FP32 (numerical stability) → cast to FP16.
2️⃣ Fuse **all** per‑position linear projections (left‑proj, left‑gate,
   right‑proj, right‑gate, out‑gate) into a **single** matmul by concatenating the
   weights.  This reads the input tensor only once and reduces memory traffic.
3️⃣ Slice the concatenated result, apply sigmoid to the three gate tensors.
4️⃣ Fuse the element‑wise product *gate* × *mask* (if a mask is present) with a
   tiny Triton kernel (`fused_gate_mask_kernel`).  The mask is broadcast over the
   hidden dimension.
5️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (`torch.bmm`).  The tensors are reshaped to
   `[B*H, N, N]` with a contiguous copy – this layout matches cuBLAS/Tensor‑Core
   expectations and yields maximal throughput on H100.
6️⃣ Second Layer‑Norm (FP16) → element‑wise multiplication with the *out‑gate*.
7️⃣ Final linear projection back to the original feature dimension.
8️⃣ Cast the result back to FP32 for the harness.

Only the forward pass is required; gradients are omitted for speed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,              # *fp16 : gate tensor, same shape as x
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate
    hidden_dim: tl.int32,  # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,   # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element indices
    valid = offs < total_elts

    # mask is broadcast over the hidden dimension:
    mask_idx = offs // hidden_dim                      # index into [B,N,N] mask

    x   = tl.load(x_ptr + offs,    mask=valid, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM using PyTorch (cuBLAS/TensorCore)
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right: [B, N, N, H] (FP16, contiguous)
    Returns:    [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel becomes its own batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM: (B*H, N, N) @ (B*H, N, N).T  → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) – ignored if config["nomask"] is True
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                               # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part of the computation
    # --------------------------------------------------------------
    x = x_norm.to(torch.float16)    # [B, N, N, dim]  (fp16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse all per‑position linear projections (proj + gates)
    #    into a single matmul to reduce memory traffic.
    # --------------------------------------------------------------
    # Gather the five weight matrices (biases are all zero in the reference).
    left_proj_w  = weights["left_proj.weight"]   # (hidden, dim)
    left_gate_w  = weights["left_gate.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden, dim)
    right_gate_w = weights["right_gate.weight"]  # (hidden, dim)
    out_gate_w   = weights["out_gate.weight"]    # (hidden, dim)

    # Concatenate along the output‑channel dimension: 5 * hidden_dim rows
    # Shape -> (5*hidden_dim, dim)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(torch.float16)

    # Single fused linear pass
    # Result shape: [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_w)

    # --------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor and compute sigmoid gates.
    #    Splits (all in FP16):
    #      0:hidden_dim               → left_proj
    #      hidden_dim:2*hidden_dim    → left_gate_raw
    #      2*hidden_dim:3*hidden_dim → right_proj
    #      3*hidden_dim:4*hidden_dim → right_gate_raw
    #      4*hidden_dim:5*hidden_dim → out_gate_raw
    # --------------------------------------------------------------
    hs = hidden_dim
    left_proj   = combined[..., 0:hs]
    left_gate_r = combined[..., hs:2*hs]
    right_proj  = combined[..., 2*hs:3*hs]
    right_gate_r= combined[..., 3*hs:4*hs]
    out_gate_r  = combined[..., 4*hs:5*hs]

    left_gate   = torch.sigmoid(left_gate_r)
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask – fuse element‑wise multiplication via Triton.
    #    The mask is broadcast over the hidden dimension.
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                      # B*N*N*hidden_dim
        BLOCK = 1024                                          # ≤ 1024 threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # Fuse left = left_proj * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right = right_proj * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Now left_proj/right_proj hold the final gated tensors.
        left = left_proj
        right = right_proj
    else:
        # No mask – simple element‑wise multiplication.
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core).
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim]  (fp16)

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16), out‑gate scaling and final projection.
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                           # still fp16

    out = out * out_gate                         # broadcasted gate (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] (fp16)

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the testing harness.
    # --------------------------------------------------------------
    return out.to(torch.float32)
