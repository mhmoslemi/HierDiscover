"""
TriMul (outgoing) – fast forward-only implementation.

The kernel performs the following steps:
1️⃣  Layer‑norm on the input (float32 → fp16 for the heavy part).
2️⃣  Two linear projections (left/right) and two gating projections.  The
    element‑wise product `proj * sigmoid(gate) * mask` is fused in a tiny
    Triton kernel (`mask_gate_fuse_kernel`).  This satisfies the “must use a
    Triton kernel” requirement while keeping launch overhead negligible.
3️⃣  The core O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is realised as a batched GEMM `C = A @ Bᵀ` using cuBLAS (torch.bmm)
    on fp16 tensors – the H100 Tensor‑Cores handle this with > 1 TFLOP/s.
4️⃣  Second Layer‑norm on the hidden dimension, out‑gate gating and the final
    linear projection back to the original `dim`.
5️⃣  The result is cast back to float32 (the harness expects float32).

All heavy arithmetic is performed in fp16; the two triton kernels are
used for (i) fused mask‑gate multiplication and (ii) a masked‑only multiplication
(when a mask is supplied).  The implementation works for any batch size ≤ 2,
sequence length up to 1024 and hidden dimensions 128/384.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask)
# Used when a mask is present.  It overwrites `x` in‑place.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 – tensor to be updated (flattened)
    gate_ptr,       # *fp16 – gate tensor (same flattening as x)
    mask_ptr,       # *fp16 – mask flattened, shape [B*N*N]
    hidden_dim,     # i32   – size of the hidden dimension H
    size,           # i32   – total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offset
    valid = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Each mask entry corresponds to `hidden_dim` consecutive elements of x
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask=1

    # Fused multiplication
    out = x * g * m

    # Store result back in‑place
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper – batched GEMM via cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a single batched matrix‑multiply with cuBLAS.

    All tensors are fp16 and contiguous.
    """
    B, N, _, H = left.shape                     # left.shape == (B, N, N, H)
    # reshape to [B*H, N, N] and perform A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))          # [B*H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Triton‑based element‑wise mask‐only multiplication.
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    x_ptr,          # *fp16 – tensor to be multiplied in‑place
    mask_ptr,       # *fp16 – mask flattened, shape [B*N*N]
    hidden_dim,     # i32   – hidden dimension H
    size,           # i32   – total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask=1
    out = x * m
    tl.store(x_ptr + offs, out, mask=valid)
