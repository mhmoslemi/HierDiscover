```python
def custom_kernel(data):
   # unpack
   input_tensor, mask_tensor, weights, config = data
   dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
   # layer norm (float32)
   norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
   x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

   # cast to fp16
   x = x.to(torch.float16)

   # prefetch all weight matrices as fp16
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(torch.float16)
   fused_weight = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0)

   # fused linear projection (B*N*N, dim) -> (B,N,N,5*hidden)
   y = torch.nn.functional.linear(x, fused_weight)  # shape [B,N,N,5*hidden]
   # split
   left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = torch.split(y, hidden_dim, dim=-1)

   # apply sigmoid to gates
   left_gate = torch.sigmoid(left_gate_raw)
   right_gate = torch.sigmoid(right_gate_raw)
   out_gate = torch.sigmoid(out_gate_raw)

   # apply gating
   left = left_proj * left_gate
   right = right_proj * right_gate

   # optional mask
   if not nomask:
       mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # broadcast
       left = left * mask_fp16
       right = right * mask_fp16

   # contract
   seq_len = input_tensor.shape[1]
   if seq_len <= 256:
       contracted = _batched_matmul_triton(left, right) # [B,N,N,hidden] fp16
   else:
       contracted = _batched_matmul_torch(left, right)

   # second LayerNorm with to_out_norm weights
   to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
   to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
   contracted = torch.nn.functional.layer_norm(contracted, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

   # apply out gate
   contracted = contracted * out_gate

   # final projection
   to_out_w = weights["to_out.weight"].to(torch.float16)
   out = torch.nn.functional.linear(contracted, to_out_w)  # [B,N,N,dim] fp16

   # cast back to float32
   return out.to(torch.float32)
```

"""
Outgoing TriMul (AlphaFold‑3) – mixed‑precision, fused linear, Triton mat‑mul.
The implementation follows the reference PyTorch module but
* fuses the five linear layers (two projections + three gates) into a single
  matrix‑multiply, reducing the memory traffic of the huge input tensor.
* performs all heavy arithmetic (the O(N³) contraction) with a custom
  Triton batched GEMM that computes
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
  using fp16 inputs, fp32 accumulation and tensor‑core dot products.
* keeps the two LayerNorms in full precision (float‑32) for numerical stability
  while the rest of the pipeline runs in fp16. The final output is cast back
  to float‑32 as required by the test harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile along i (output rows)
    pid_n = tl.program_id(1)   # tile along j (output cols)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------------------------------------------------------
    # Compute the absolute offsets for the current tile.
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator (higher precision for the reduction)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A[i, k]   (shape BLOCK_M × BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am               # offset of the current hidden slice
            + offs_i[:, None] * stride_ai     # i offset
            + offs_k[None, :] * stride_ak     # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # --------------------------------------------------------------
        # Load Bᵀ[j, k]   (shape BLOCK_N × BLOCK_K)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm               # hidden slice offset
            + offs_k[:, None] * stride_bk     # k offset (first dimension)
            + offs_j[None, :] * stride_bj     # j offset (second dimension)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the heavy O(N³) contraction using the Triton GEMM kernel.
    Input tensors must be fp16 with shape [B, N, N, H].
    Returns a tensor with the same shape (fp16).
    """
    B, N, _, H = left.shape
    # Re‑order to [M, N, N] where M = B*H  (contiguous layout)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – 64×64×64 works well on H100 for fp16 MMA.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Fallback: use cuBLAS via torch.bmm (still fp16 with fp32 accumulation).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float, may be None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32) – stability
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the rest of the pipeline
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Fuse the five independent linear layers:
    #    left_proj, left_gate, right_proj, right_gate, out_gate
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate the five weight matrices along the output‑feature dimension.
    fused_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape [(5*hidden_dim), dim]

    # Single large matmul: [B,N,N,dim] @ [(5*hidden_dim), dim]ᵀ → [B,N,N,5*hidden_dim]
    fused_out = torch.nn.functional.linear(x, fused_w)  # fp16, shape [B,N,N,5*hidden_dim]

    # Split the fused result back into its constituent parts.
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = torch.split(
        fused_out, hidden_dim, dim=-1
    )  # each shape [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 5️⃣ Apply sigmoids to the gate tensors
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)      # will be used after the second LayerNorm

    # --------------------------------------------------------------
    # 6️⃣ Apply gates (and optional mask) to obtain gated tensors
    # --------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → [B, N, N, 1] (fp16) for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction (batched GEMM)
    # --------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16, shape [B, N, N, hidden_dim]
    else:
        contracted = _batched_matmul_torch(left, right)    # fp16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # Cast contraction to float32, apply LayerNorm, then cast back
    contracted_fp32 = contracted.to(torch.float32)
    contracted = torch.nn.functional.layer_norm(
        contracted_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # back to fp16 for the final projection
    contracted = contracted.to(torch.float16)

    # --------------------------------------------------------------
    # 9️⃣ Apply the out‑gate and final linear projection
    # --------------------------------------------------------------
    contracted = contracted * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(contracted, to_out_w)  # fp16, shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 🔟 Cast back to float32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
