    left_fused_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # shape (2*hidden_dim, dim)
    right_fused_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)
    # linear projection + gate pre
    left_fused = torch.nn.functional.linear(x, left_fused_weight)  # shape [B,N,N,2*hidden_dim]
    right_fused = torch.nn.functional.linear(x, right_fused_weight)
    # split
    left_proj = left_fused[..., :hidden_dim]
    left_gate_pre = left_fused[..., hidden_dim:]
    right_proj = right_fused[..., :hidden_dim]
    right_gate_pre = right_fused[..., hidden_dim:]
    # gate
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # mask gating fuse or mul
    if not nomask:
        # mask fp16 flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # shape B*N*N
        total_elements = left_proj.numel()
        BLOCK = 1024
        grid = ( (total_elements + BLOCK - 1) // BLOCK, )

        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # release gates
        del left_gate, right_gate
    else:
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # left and right = left_proj, right_proj
    left = left_proj
    right = right_proj

    # out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # heavy contraction
    out = _batched_gemm_fp16(left, right)   # shape [B,N,N,hidden_dim]

    # second layernorm
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias)
    # out_gate multiplication
    out = out * out_gate

    # final projection
    out = torch.nn.functional.linear(out, to_out_weight)

    # cast back to fp32
    return out.float()

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (FP32) → cast to FP16.
2. Fuse each projection (left/right) with its gating linear map by concatenating the
   two weight matrices.  A single `torch.nn.functional.linear` produces both the
   projected values and the pre‑gate values.
3. Apply the sigmoid gating and, if a mask is given, fuse the element‑wise
   multiplication `proj * gate * mask` with a tiny Triton kernel (`mask_gate_fuse_kernel`).
   When there is no mask we simply multiply in‑place (`proj.mul_(gate)`).
4. The costly contraction  
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  
   is performed as a batched GEMM: for each head `h` we compute  
      C = A @ Bᵀ   (A/B shaped [B*H, N, N]).
   This leverages cuBLAS Tensor‑Core kernels on H100.
5. Second LayerNorm (FP16) → element‑wise out‑gate (sigmoid) → final linear
   projection back to the original channel dimension.
6. Cast result back to FP32 (the harness expects FP32).

At least one Triton kernel is used (the mask‑gate fusion).  The remaining
operations stay in PyTorch but are kept in FP16 to reduce memory traffic.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over
# the hidden dimension `hidden_dim`. The kernel works on a flat view of the
# 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,             # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,       # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask = offs < size                         # boundary mask

    # Load operands
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask: every `hidden_dim` consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Perform fused multiplication
    out = x * g * m

    # Store the result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single batched GEMM (FP16 inputs → FP32 accumulation).

    Arguments
    ---------
    left, right : torch.Tensor of shape [B, N, N, H] (dtype = float16)

    Returns
    -------
    out : torch.Tensor of shape [B, N, N, H] (dtype = float16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – we will do A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM (FP16 inputs, FP32 accumulation inside cuBLAS)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]       # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],  # both FP32
    )                                           # [B, N, N, dim]  float32
    x = x_norm.to(torch.float16)                # FP16 view for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (once)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()    # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ Fused Linear + Gate (one matmul per side)
    # ------------------------------------------------------------------
    # Concatenate proj and gate weights so that a single linear produces both.
    left_fused_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*hidden_dim, dim)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Project + pre‑gate (shape [..., 2*hidden_dim])
    left_fused  = torch.nn.functional.linear(x, left_fused_w)    # [B, N, N, 2*hidden_dim]
    right_fused = torch.nn.functional.linear(x, right_fused_w)

    # Split into projection and gate pre‑activations
    left_proj      = left_fused[..., :hidden_dim]                # [B,N,N,hidden_dim]
    left_gate_pre  = left_fused[..., hidden_dim:]                # [B,N,N,hidden_dim]
    right_proj     = right_fused[..., :hidden_dim]
    right_gate_pre = right_fused[..., hidden_dim:]

    # Apply sigmoid to obtain gates
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – fused element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if nomask:
        # In‑place multiplication (no extra memory allocation)
        left = left_proj.mul_(left_gate)
        right = right_proj.mul_(right_gate)
    else:
        # Flatten mask to (B*N*N,) and cast to fp16 for the Triton kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elements = left_proj.numel()          # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # No longer need the separate gate tensors
        del left_gate, right_gate
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Compute out‑gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction (batched GEMM) – cuBLAS implementation
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # stays FP16

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # back to `dim` (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast result back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
