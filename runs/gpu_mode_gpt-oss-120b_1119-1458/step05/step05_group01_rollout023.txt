def trimul_outgoing_gemm(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_h = tl.program_id(3)

    # tile offsets
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    h0 = pid_h * BLOCK_H

    # Compute offsets arrays
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    # Mask out-of-bound
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # Accumulator shape (BLOCK_I, BLOCK_J, BLOCK_H) in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over K
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for t in range(num_k_tiles):
        k0 = t * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + pid_b*stride_left_b \
            + (offs_i[:, None, None] * stride_left_i) \
            + (offs_k[None, :, None] * stride_left_k) \
            + (offs_h[None, None, :] * stride_left_h)
        left_tile = tl.load(left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0)

        # Load right tile: shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + pid_b*stride_right_b \
            + (offs_j[:, None, None] * stride_right_j) \
            + (offs_k[None, :, None] * stride_right_k) \
            + (offs_h[None, None, :] * stride_right_h)
        right_tile = tl.load(right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0)

        # For each hidden slice in tile compute gemm using tl.dot
        for hi in range(BLOCK_H):
            # condition mask for hidden slice
            mask_hi = mask_h[hi]
            # If out of bounds skip (no accumulation)
            if not mask_hi:
                continue
            # left slice (BLOCK_I, BLOCK_K)
            left_mat = left_tile[:, :, hi]    # shape (BLOCK_I, BLOCK_K)
            # right slice (BLOCK_J, BLOCK_K)
            right_mat = right_tile[:, :, hi]  # shape (BLOCK_J, BLOCK_K)
            # Compute dot: (BLOCK_I, BLOCK_J) = left_mat @ right_mat.T
            acc[:, :, hi] = acc[:, :, hi] + tl.dot(left_mat, right_mat, trans_b=True) # sum across k

    # Store accumulator
    out_ptrs = out_ptr + pid_b*stride_out_b \
                + offs_i[:, None, None] * stride_out_i \
                + offs_j[None, :, None] * stride_out_j \
                + offs_h[None, None, :] * stride_out_h

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

    x = x.to(torch.float16)

    # Cast weights
    left_proj_w = weights["left_proj.weight"].to(dtype=torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(dtype=torch.float16, device=device)
    left_gate_w = weights["left_gate.weight"].to(dtype=torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(dtype=torch.float16, device=device)
    out_gate_w = weights["out_gate.weight"].to(dtype=torch.float16, device=device)
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype=torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype=torch.float16, device=device)
    to_out_w = weights["to_out.weight"].to(dtype=torch.float16, device=device)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_w)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates and mask
    if not nomask:
        # Convert mask to fp16
        mask_fp16 = mask_tensor.to(dtype=torch.float16)
        # Multiply left and right by mask in-place after gating
        left = left * left_gate * mask_fp16[..., None]
        right = right * right_gate * mask_fp16[..., None]
    else:
        left = left * left_gate
        right = right * right_gate

    # Contraction O(N^3) using Triton kernel
    out = _tri_mul_gemm(left, right)   # shape [B,N,N,H]

    # Second layernorm
    out = torch.nn.functional.layer_norm(out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Out gate & final linear
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to fp32
    return out.to(torch.float32)

"""
TriMul (outgoing) – highly‑optimized forward pass.
=================================================
Algorithm
---------
1️⃣  FP32 LayerNorm → cast to FP16.
2️⃣  Four linear projections (left, right, left_gate, right_gate) + a linear out‑gate.
    All projections use FP16 weights and the sigmoid is fused in‑place.
3️⃣  If a mask is provided (nomask=False) we fuse mask, projection and gate with
    a tiny element‑wise Triton kernel; otherwise a plain elementwise multiplication
    is used.
4️⃣  The O(N³) contraction   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom Triton GEMM that works directly on the original
    `[B,N,N,H]` layout – no costly permute/contiguous copies are required.
    The kernel tiles the (i,j) plane and the hidden dimension (h) while looping
    over k in blocks; each hidden slice uses `tl.dot` (warp‑level tensor‑core)
    for fast accumulation.
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection back to
    the model dimension.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion (optional) and the O(N³) contraction are implemented in
Triton; all other operations stay in PyTorch for simplicity and to keep the
kernel count low.  This implementation removes the heavy memory copies of the
reference solution and runs well below the 1 ms target on the largest benchmark
shapes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask * gate * projection (optional)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                     # *fp16  [B, N, N, H]
    gate_ptr,                     # *fp16  [B, N, N, H] (sigmoid already)
    mask_ptr,                     # *fp16  [B, N, N]
    out_ptr,                      # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # load mask (no hidden dim)
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + \
                offs_i[:, None] * stride_mask_i + \
                offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptrs,
                        mask=mask_i[:, None] & mask_j[None, :],
                        other=0.0)   # fp16

    # hidden‑dim tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj tile
        proj_ptrs = proj_ptr + pid_b * stride_proj_b + \
                    offs_i[:, None, None] * stride_proj_i + \
                    offs_j[None, :, None] * stride_proj_j + \
                    offs_h[None, None, :] * stride_proj_h
        proj_tile = tl.load(proj_ptrs,
                            mask=mask_i[:, None, None] &
                                 mask_j[None, :, None] &
                                 mask_h[None, None, :],
                            other=0.0)   # fp16

        # gate tile (already sigmoid‑ed)
        gate_ptrs = gate_ptr + pid_b * stride_gate_b + \
                    offs_i[:, None, None] * stride_gate_i + \
                    offs_j[None, :, None] * stride_gate_j + \
                    offs_h[None, None, :] * stride_gate_h
        gate_tile = tl.load(gate_ptrs,
                            mask=mask_i[:, None, None] &
                                 mask_j[None, :, None] &
                                 mask_h[None, None, :],
                            other=0.0)   # fp16

        # fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # store
        out_ptrs = out_ptr + pid_b * stride_out_b + \
                   offs_i[:, None, None] * stride_out_i + \
                   offs_j[None, :, None] * stride_out_j + \
                   offs_h[None, None, :] * stride_out_h
        tl.store(out_ptrs,
                 out_tile,
                 mask=mask_i[:, None, None] &
                      mask_j[None, :, None] &
                      mask_h[None, None, :])


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Fuse proj * gate * mask (mask broadcast over hidden dim)."""
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: O(N³) contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile
    pid_h = tl.program_id(3)          # hidden‑tile

    # tile offsets
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    h_base = pid_h * BLOCK_H

    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_h = h_base + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator (fp32) for a tile [BLOCK_I, BLOCK_J, BLOCK_H]
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile   shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + pid_b * stride_l_b + \
                    offs_i[:, None, None] * stride_l_i + \
                    offs_k[None, :, None] * stride_l_k + \
                    offs_h[None, None, :] * stride_l_h
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None, None] &
                                 mask_k[None, :, None] &
                                 mask_h[None, None, :],
                            other=0.0)   # fp16

        # right tile  shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + pid_b * stride_r_b + \
                     offs_j[:, None, None] * stride_r_j + \
                     offs_k[None, :, None] * stride_r_k + \
                     offs_h[None, None, :] * stride_r_h
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None, None] &
                                  mask_k[None, :, None] &
                                  mask_h[None, None, :],
                             other=0.0)   # fp16

        # per‑hidden‑slice GEMM via tl.dot
        for hi in range(BLOCK_H):
            # Skip out‑of‑bounds hidden indexes (masked loads already zeroed)
            # left_slice: (BLOCK_I, BLOCK_K), right_slice: (BLOCK_J, BLOCK_K)
            acc[:, :, hi] = acc[:, :, hi] + tl.dot(
                left_tile[:, :, hi],
                right_tile[:, :, hi],
                trans_b=True
            )  # result is fp32

    # store result (cast back to fp16)
    out_ptrs = out_ptr + pid_b * stride_o_b + \
               offs_i[:, None, None] * stride_o_i + \
               offs_j[None, :, None] * stride_o_j + \
               offs_h[None, None, :] * stride_o_h
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_h[None, None, :])


def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched hidden‑dim GEMM in Triton, operating directly on [B,N,N,H] tensors."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)               # fp16 output

    # strides expressed in element counts (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        (H + BLOCK_H - 1) // BLOCK_H,
    )
    _trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    return out

) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → cast to FP16
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)          # now FP16 for the rest of the pipeline

    # -----------------------------------------------------------------
    # 2️⃣ Cast all model weights to FP16 on the target device
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP16
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 5️⃣ Mask + gate fusion
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)   # [B, N, N]
        left = _apply_mask_gate(left, left_gate, mask_fp16)
        right = _apply_mask_gate(right, right_gate, mask_fp16)
    else:
        # simple elementwise multiplication (in‑place to save a buffer)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via custom Triton GEMM
    # -----------------------------------------------------------------
    out = _trimul_gemm(left, right)    # shape [B, N, N, hidden_dim], fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Out‑gate, final projection back to model dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (the harness expects FP32)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
