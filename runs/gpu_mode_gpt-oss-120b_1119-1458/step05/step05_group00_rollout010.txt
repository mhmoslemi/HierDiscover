def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor):
    B,N,_,H = left.shape
    # Flatten left to shape (B*H, N, N)
    left_permuted = left.permute(0,3,1,2).contiguous()
    left_mat = left_permuted.view(B*H, N, N)
    # Right transposed (swap inner N dims)
    right_permuted = right.permute(0,3,2,1).contiguous()
    right_mat = right_permuted.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat)  # (B*H,N,N)
    # Reshape back
    out = out.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

"""
Optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → cast to FP16.
2️⃣  Fuse the four linear projections + three gating projections into a
    single GEMM:
        Y = x @ W_concat^T   (W_concat = [W_left, W_right,
                                         W_left_gate, W_right_gate,
                                         W_out_gate] )
    Y is split into left_proj, right_proj, left_gate_pre, right_gate_pre,
    out_gate_pre (all FP16).

3️⃣  Compute the out‑gate (=sigmoid(out_gate_pre)) in FP32 → FP16.
4️⃣  If a mask is present:
        * Launch a Triton kernel that, for each hidden element,
          computes     left  = left_proj  * sigmoid(left_gate_pre)  * mask,
                       right = right_proj * sigmoid(right_gate_pre) * mask.
      Otherwise multiply by the sigmoid gates only (pure PyTorch).

5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = A @ Bᵀ where A,B ∈
   ℝ^{(B·H)×N×N}.  We reshape the tensors and call ``torch.bmm`` which
   on an H100 uses Tensor‑core‑accelerated FP16 GEMM.

6️⃣  Second LayerNorm over the hidden dimension (FP16) → multiply by the
    previously computed out‑gate.

7️⃣  Final linear projection back to ``dim`` (FP16) and cast the result to
    float32 (expected by the harness).

The implementation must include at least one Triton kernel – the mask &
gate‑fusion kernel – while the remaining operations use highly‑optimized
PyTorch kernels in mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid gates and element‑wise multiplication
#   left  ← left_proj  * sigmoid(left_gate_pre)  * mask
#   right ← right_proj * sigmoid(right_gate_pre) * mask
#   All tensors are contiguous FP16 with shape [B,N,N,H] for left/right
#   and [B,N,N] for mask (broadcasted across H).
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,               # *fp16  [B,N,N,H] (in‑place)
    right_ptr,              # *fp16  [B,N,N,H] (in‑place)
    left_gate_pre_ptr,      # *fp16  [B,N,N,H] (pre‑sigmoid)
    right_gate_pre_ptr,     # *fp16  [B,N,N,H] (pre‑sigmoid)
    mask_ptr,               # *fp16  [B,N,N]   (0/1)
    total_elements,         # i32   total number of fp16 elements (B·N·N·H)
    H,                      # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # flat indices
    mask = offs < total_elements

    # Compute mask index (mask does not have the hidden dimension)
    mask_idx = offs // H

    # Load operands; out‑of‑bounds loads get a neutral value.
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gpre = tl.load(left_gate_pre_ptr  + offs, mask=mask, other=0.0)
    right_gpre= tl.load(right_gate_pre_ptr + offs, mask=mask, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Sigmoid (in‑place, FP16)
    left_gate  = 1.0 / (1.0 + tl.exp(-left_gpre))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gpre))

    # Fuse mask, gate and linear projection
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store results back in‑place
    tl.store(left_ptr  + offs, left_out ,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ where A and B are [B, N, N, H] (FP16).
    Returns C in the same shape and dtype (FP16).
    """
    B, N, _, H = left.shape
    # (B, H, N, N) – flatten (B·H) as batch dimension
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Right needs to be transposed on the two N dimensions before flattening
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Batched GEMM on FP16 (Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat)        # (B·H, N, N)

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float)
        - weights_dict : name → torch.Tensor (float32) – contains all model weights
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32, for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)                     # [B, N, N, dim] (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # All projection weights have shape [hidden_dim, dim] and bias=False.
    # Build a concatenated weight matrix of shape [(5·hidden_dim), dim].
    w_left      = weights["left_proj.weight"]
    w_right     = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate= weights["right_gate.weight"]
    w_out_gate  = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension and move to FP16.
    w_concat = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    ).to(device=device, dtype=dtype_fp16)            # [(5·H), dim]

    # Single linear transform → shape [B, N, N, 5·hidden_dim]
    combined = F.linear(x, w_concat)                  # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Slice the combined projection
    # ------------------------------------------------------------------
    H = hidden_dim
    # Slicing creates view tensors; we make them contiguous because the
    # Triton kernel expects a flat contiguous layout.
    left_proj   = combined[..., 0:H].contiguous()
    right_proj  = combined[..., H:2*H].contiguous()
    left_gate_pre  = combined[..., 2*H:3*H].contiguous()
    right_gate_pre = combined[..., 3*H:4*H].contiguous()
    out_gate_pre   = combined[..., 4*H:5*H].contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Compute the out‑gate (sigmoid) – needed after the second LN
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_pre)            # FP16, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply mask (if any) and the left/right sigmoid gates
    # ------------------------------------------------------------------
    if nomask:

        left_gate  = torch.sigmoid(left_gate_pre)    # [B,N,N,H]
        right_gate = torch.sigmoid(right_gate_pre)
        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Fuse mask, sigmoid gates and element‑wise multiplication in Triton.
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()   # [B,N,N]

        total = left_proj.numel()                     # B·N·N·H
        BLOCK = 1024                                   # ~1K threads / block
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left_proj,
            right_ptr=right_proj,
            left_gate_pre_ptr=left_gate_pre,
            right_gate_pre_ptr=right_gate_pre,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=H,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # After the kernel left_proj/right_proj now contain the gated & masked values
        left = left_proj
        right = right_proj
        # The pre‑gate tensors are no longer needed
        del left_gate_pre, right_gate_pre, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched FP16 GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)          # [B, N, N, H] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device=device, dtype=dtype_fp16)

    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)         # FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply the out_gate and project back to the original dim
    # ------------------------------------------------------------------
    out = out * out_gate                               # element‑wise (FP16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)
    out = F.linear(out, to_out_w)                      # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (as expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
