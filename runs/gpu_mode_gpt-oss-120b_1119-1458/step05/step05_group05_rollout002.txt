"""TriMul outgoing module forward pass (no autograd) with Triton fused mask gating.

The implementation follows the reference PyTorch version:
1. LayerNorm on [B, N, N, dim] (float32).
2. FP16: linear projections left/right, gating projections left_gate/right_gate/out_gate.
3. Optional mask is fused with element‑wise multiplication (left*gate*mask) via a
   custom Triton kernel (`mask_gate_fuse_kernel`).
4. Heavy contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]` is performed as
   a batched matrix multiplication (A @ Bᵀ) using cuBLAS (`torch.bmm`).  Tensors are
   reshaped to (B*H, N, N) where H = hidden_dim.
5. Second LayerNorm on hidden dimension, multiply by out_gate, and final linear
   projection back to the original channel dimension.
6. Output returned as float32 (same device as input).

All major compute steps run in FP16 to exploit TensorCores on an H100.  The Triton
kernel is only used for the mask‑gate fusion, satisfying the requirement of using
Triton while keeping overhead minimal.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate, and tensor multiplication.
# Input tensors are flattened FP16 arrays.  mask has no hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16, tensor to be updated (flattened)
    gate_ptr,       # *fp16, gate tensor (flattened)
    mask_ptr,       # *fp16, mask tensor (flattened, shape B*N*N)
    hidden_dim,     # i32, size of hidden dimension H
    size,           # i32, total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (<=1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load values
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Compute corresponding mask index (broadcast over H)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fuse multiplication
    out = x * g * m

    # Store back in place
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the core O(N³) contraction.
# Operates in fp16 and relies on cuBLAS via torch.bmm.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left, right: fp16 tensors of shape [B, N, N, H]
    Returns fp16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

# Input tensors are flattened fp16 arrays; mask has no hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → size of the hidden dimension (H)
    size,           # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    mask_idx = offs // hidden_dim

    # Fuse multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=valid)
