"""
TriMul (outgoing) – fast forward pass for AlphaFold‑3
----------------------------------------------------
* Layer‑Norm on the raw input (float32) → cast to fp16.
* Fuse the three “project‑and‑gate” linear layers per side into a single
  matmul:
      left  = Linear(x, W_left_proj‖W_left_gate‖W_out_gate)   # 3·H out‑features
      right = Linear(x, W_right_proj‖W_right_gate)            # 2·H out‑features
  After the matmul we split the results into the three tensors needed
  (proj, gate, out‑gate) and apply a sigmoid to the gate tensors.
* Optional mask: a tiny Triton kernel fuses the three element‑wise
  multiplications  proj * gate * mask, broadcasting the mask over the
  hidden dimension.
* Heavy O(N³) contraction is performed as a batched GEMM using cuBLAS:
      out_{b,i,j,d} = Σ_k left_{b,i,k,d} · right_{b,j,k,d}
  which is exactly a batched matrix multiplication
  `left @ rightᵀ` for each hidden channel.
* Second LayerNorm (fp16), gating by the previously‑computed out‑gate,
  and a final linear projection back to the original dimension.
* The result is returned as float32 (required by the Harness).

Only the forward pass is implemented – no autograd is needed.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel that fuses:   x = proj          (fp16)
#                              g = gate (sigmoid) (fp16)
#                              m = mask (fp16)   (broadcast)
# into a single in‑place multiply:  x = x * g * m
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,      # *fp16 : projected tensor (mutated in‑place)
    gate_ptr,      # *fp16 : gate tensor   (sigmoid already applied)
    mask_ptr,      # *fp16 : mask, shape (B·N·N)
    hidden_dim,    # i32   : hidden dimension H
    n_elements,    # i32   : total number of elements in proj/gate (B·N·N·H)
    BLOCK: tl.constexpr,       # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offset of this block
    mask = offs < n_elements                           # guard for the tail
    # map the linear offset to the corresponding mask element (mask is broadcast over H)
    mask_off = offs // hidden_dim
    # load tensors
    x = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)   # 1.0 = no‑mask elsewhere
    # fused multiplication
    out = x * g * m
    tl.store(proj_ptr + offs, out, mask=mask)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using a batched GEMM.
    left / right : [B, N, N, H] (fp16)
    returns      : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] – each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # bmm with the second operand transposed implements the einsum in the reference code
    c = torch.bmm(a, b.transpose(1, 2))               # [B·H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N]       (float32/bool or None)
        - weights : dict mapping weight names to torch.Tensor (float32)
        - config  : dict containing at least ``dim`` and ``hidden_dim``;
                    optional ``nomask`` (bool) indicating whether a mask is present.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable on the raw input.
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                       # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation.
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the three projected/gated linear layers.
    # -----------------------------------------------------------------
    # Left side: (proj, gate, out_gate) → 3·hidden_dim output features
    left_concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (3*H, dim)

    # Right side: (proj, gate) → 2·hidden_dim output features
    right_concat_weight = torch.cat(
        [
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (2*H, dim)

    # Linear projections (single GEMM per side)
    left_concat = torch.nn.functional.linear(x_fp16, left_concat_weight)     # [B,N,N,3H]
    right_concat = torch.nn.functional.linear(x_fp16, right_concat_weight)   # [B,N,N,2H]

    # -----------------------------------------------------------------
    # 4️⃣ Slice concatenated tensors and apply sigmoids for gates.
    # -----------------------------------------------------------------
    # left side splits
    left_proj_raw   = left_concat[..., :hidden_dim]                       # [B,N,N,H]
    left_gate_raw   = left_concat[..., hidden_dim:2 * hidden_dim]        # [B,N,N,H]
    out_gate_raw    = left_concat[..., 2 * hidden_dim:]                  # [B,N,N,H]

    left_gate   = torch.sigmoid(left_gate_raw)   # fp16
    out_gate    = torch.sigmoid(out_gate_raw)    # fp16

    # right side splits
    right_proj_raw  = right_concat[..., :hidden_dim]                     # [B,N,N,H]
    right_gate_raw  = right_concat[..., hidden_dim:]                     # [B,N,N,H]
    right_gate  = torch.sigmoid(right_gate_raw)   # fp16

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fuse proj * gate * mask using Triton.
    # -----------------------------------------------------------------
    if not nomask:
        # Cast mask to fp16 and make it contiguous.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj_raw.numel()        # B·N·N·H
        BLOCK = 1024                                 # ≤ 1024 threads per block
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj_raw * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj_raw,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj_raw * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj_raw,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left  = left_proj_raw   # now contains the fused result
        right = right_proj_raw
    else:
        # No mask -> simple element‑wise multiplication.
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS / Tensor‑Core).
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)      # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) over the hidden dimension.
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                             # fp16

    # -----------------------------------------------------------------
    # 8️⃣ Gating with the previously computed out‑gate and final projection.
    # -----------------------------------------------------------------
    out = out * out_gate            # element‑wise fp16 gating
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness) and return.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
