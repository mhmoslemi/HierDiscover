@triton.jit
def outgoing_trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N, # dim of squares
    batch, # total number of (B*H)
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute indexes
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)

    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guard, char lengths
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute offsets for K loops
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # batch offsets:
    batch_off = batch_id * stride_l_batch  # assume same for right and out

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for k
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers
        a_ptrs = left_ptr + batch_off + (offs_m[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        b_ptrs = right_ptr + batch_off + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_j)

        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_ptr + batch_off + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)

    B,N,_ , H = left.shape
    BH = B*H
    left_c = left.permute(0,3,1,2).contiguous() # shape B, H, N, N
    right_c = right.permute(0,3,1,2).contiguous()
    left_flat = left_c.view(B*H, N, N)
    right_flat = right_c.view(B*H, N, N)
    # allocate output

    # compute strides
    stride_batch = N * N
    stride_i = N
    stride_k = 1
    # for right: same stride_k 1; stride_j = N
    stride_k_right = 1
    stride_j = N
    # for out: same as left
    stride_out_i = N
    stride_out_j = 1

    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    )
    # reshape out back

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‐norm the pairwise input (FP32 → FP16).
2️⃣  Four FP16 linear projections (left/right) and three FP16 gating
    projections.  Mask + gate fusion is performed in a tiny Triton
    element‑wise kernel (in‑place → no extra memory traffic).
3️⃣  Heavy O(N³) contraction:

   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   The contraction is a batched GEMM C = A @ Bᵀ where
   A,B ∈ ℝ^{(B·H)×N×N}.  A custom Triton kernel computes all
   B·H independent GEMMs in a single launch, fully exploiting
   Tensor‑cores (FP16 ⊙ FP16 → FP32 accumulator → FP16 output).

4️⃣  Second LayerNorm (over the hidden dimension H) → element‑wise
    multiplication with `out_gate` (sigmoid) → final linear projection
    back to the original dimensionality (`dim`).

5️⃣  Cast the result back to FP32 (the harness expects FP32).

All heavy operations are fused in Triton kernels; the remaining
lightweight ops use fast PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion.
#   left  ← left  * left_gate  * mask
#   right ← right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i64   total number of fp16 elements in left/right
    H,                 # i64   hidden dimension (used for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # mask index = flat_index // H   (broadcast across hidden dim)
    mask_offs = offs // H

    # Load tensors
    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_out , mask=active)
    tl.store(right_ptr + offs, right_out, mask=active)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
#   C = A @ Bᵀ   where   A, B ∈ (BH, N, N)    (BH = B * hidden_dim)
#
#   A is the "left" tensor,   B is the "right" tensor.
#   A[i,k] * B[j,k] → C[i,j] per batch.
#
#   All tensors are FP16 (input) → accumulator FP32 → stored as FP16.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_batched_gemm_kernel(
    a_ptr,                # *fp16  [BH, N, N]
    b_ptr,                # *fp16  [BH, N, N]   (j,k layout)
    c_ptr,                # *fp16  [BH, N, N]   (output)
    N,                    # i64    matrix side length
    BH,                   # i64    total batch count (B * hidden_dim)
    stride_batch,         # i64    stride to next batch (N*N)
    stride_i,             # i64    stride for rows   (N)
    stride_k,             # i64    stride for cols   (1)
    stride_j,             # i64    stride for cols of B (N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1‑D grid layout:  (batch, block_row, block_col)
    # ------------------------------------------------------------------
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Batch offset (same for a, b, c because the layout is identical)
    batch_offset = batch_id * stride_batch

    # Accumulator (FP32 to retain precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load tiles of A (shape: BLOCK_M x BLOCK_K) and B (shape: BLOCK_K x BLOCK_N)
        # B is accessed as B[k, j] so that the multiplication implements A @ Bᵀ
        # ------------------------------------------------------------------
        a_ptrs = a_ptr + batch_offset + (offs_m[:, None] * stride_i) + (offs_k[None, :] * stride_k)
        b_ptrs = b_ptr + batch_offset + (offs_k[:, None] * stride_k) + (offs_n[None, :] * stride_j)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # ------------------------------------------------------------------
        # GEMM accumulate (Tensor‑core fast path: FP16 × FP16 → FP32)
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write result back (cast to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_offset + (offs_m[:, None] * stride_i) + (offs_n[None, :] * stride_j)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(left: torch.Tensor,
                         right: torch.Tensor,
                         block_m: int = 64,
                         block_n: int = 64,
                         block_k: int = 64) -> torch.Tensor:
    """
    Perform   out = left @ rightᵀ   for tensors of shape [B,N,N,H].
    Returns a tensor of the same shape and dtype (fp16).
    """
    B, N, _, H = left.shape
    BH = B * H

    # ------------------------------------------------------------------
    # Rearrange so that hidden dimension becomes part of the batch dim:
    #   (B, N, N, H) → (BH, N, N)   (contiguous)
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous().view(BH, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(BH, N, N)

    out_t = torch.empty_like(left_t)

    # Strides for the contiguous (BH, N, N) layout (row‑major)
    stride_batch = N * N                      # distance to next batch
    stride_i = N                              # row stride
    stride_k = 1                              # column stride (for A)
    stride_j = N                              # column stride (for B, using j‑dimension)

    # ------------------------------------------------------------------
    # Grid configuration
    # ------------------------------------------------------------------
    num_blocks_m = (N + block_m - 1) // block_m
    num_blocks_n = (N + block_n - 1) // block_n
    total_blocks = BH * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _trimul_batched_gemm_kernel[grid](
        a_ptr=left_t,
        b_ptr=right_t,
        c_ptr=out_t,
        N=N,
        BH=BH,
        stride_batch=stride_batch,
        stride_i=stride_i,
        stride_k=stride_k,
        stride_j=stride_j,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,          # enough warps to keep the tensor‑core pipelines busy
        num_stages=3,         # double‑buffering for better L2 utilisation
    )

    # Reshape back to the original layout
    out = out_t.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    return out

    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w) # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Gating projections (FP16) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # used later

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (single Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – cast to half‑precision for the kernel
        mask_fp16 = mask.to(dtype_fp16).contiguous()
        # Ensure contiguity for the tensors fed to Triton
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_elems = left.numel()
        BLOCK = 1024  # 1 K threads per program – good for latency hiding
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total_elems,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates have been fused; we can free them early
        del left_gate, right_gate, mask_fp16
    else:
        # No mask → only apply the gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via the custom Triton GEMM
    # ------------------------------------------------------------------
    out = _batched_gemm_triton(left, right,
                               block_m=64,
                               block_n=64,
                               block_k=64)   # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)        # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise (fp16)
    out = F.linear(out, to_out_w)            # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 🔟 Cast back to FP32 (expected by the outer harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
