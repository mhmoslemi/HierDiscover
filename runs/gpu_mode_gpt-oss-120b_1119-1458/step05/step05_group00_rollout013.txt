    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    pid_m = tl.program_id(1)

    h = batch_h_idx % H

    offset_b = b * stride_left_b
    offset_h = h * stride_left_h            # stride_left_h=1 thus just h

    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # We'll use pointers: base + b_offset + h_offset + j_offset[None,:] * stride_right_i + k_offset[:, None] * stride_right_k

    # We'll set accumulator

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ids
    batch_h_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute batch and hidden indices
    b = batch_h_idx // H
    h = batch_h_idx % H

    # offsets
    offset_b = b * stride_left_b
    offset_h = h * stride_left_h   # expecting stride =1
    # compute global offsets for i and j
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension

        mask_k = k_off < N

"""
TriMul (outgoing) – fast forward‐only implementation.

Algorithm
---------
1.  Layer‑norm on the 4‑D input (FP32) → cast to FP16.
2.  Four linear projections (left/right) and three gating projections (FP16).
3.  If a mask is present, fuse mask, left‑gate and right‑gate in a tiny
    Triton element‑wise kernel (in‑place update of `left` and `right`).
4.  Heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    which is a batched GEMM (A @ Bᵀ) for each hidden dimension.
    Implemented with a custom Triton kernel that works directly on the
    original `[B,N,N,H]` layout – no costly `permute/contiguous` copy.
5.  Second “to‑out” LayerNorm (FP16) → element‑wise out‑gate (sigmoid).
6.  Final linear projection back to `dim`.
7.  Cast output back to FP32 (the harness expects float32).

All heavy work is in fp16 to exploit Tensor‑cores on H100.  The kernel
launch grid is three‑dimensional:
    (B*H, ceil(N/BM), ceil(N/BN))
so each program processes one `(i,j)` tile for a fixed batch‑hidden pair.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask + gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted over H)
    total_elements,    # i64   total number of fp16 elements in left/right
    hidden_dim,        # i64   `H`
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # each hidden element shares the same mask entry → divide by H
    mask_offs = offs // hidden_dim

    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g   = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)   # 0/1 mask

    left_out  = left  * left_g  * m
    right_out = right * right_g * m

    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched MatMul for the O(N³) contraction.
#    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16  [B,N,N,H]
    right_ptr,         # *fp16  [B,N,N,H]
    out_ptr,           # *fp16  [B,N,N,H]
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs: (batch_h, block_i, block_j)
    batch_h_idx = tl.program_id(0)
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # decode batch and hidden dimension
    b = batch_h_idx // H
    h = batch_h_idx % H

    # base offsets for this (b,h) pair
    off_b = b * stride_left_b
    off_h = h * stride_left_h     # stride_left_h == 1 for contiguous layout

    # tile start coordinates
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in float32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k_blocks):
        k_off = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile: left[b,i,k,h]
        a = tl.load(
            left_ptr
            + off_b + off_h
            + i_off[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile transposed: we need right[b,j,k,h] as (k,j)
        b_tile = tl.load(
            right_ptr
            + off_b + off_h
            + k_off[:, None] * stride_right_k
            + j_off[None, :] * stride_right_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b_tile)   # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N) → (BLOCK_M,BLOCK_N)

    # write back (FP16)
    out_tile = acc.to(tl.float16)
    out_ptrs = (
        out_ptr
        + off_b + off_h
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
    )
    tl.store(out_ptrs, out_tile, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`
    using the custom Triton kernel defined above.
    """
    B, N, _, H = left.shape
    # allocate output (FP16)
    out = torch.empty_like(left)

    # block size configuration – chosen to fit H100 Tensor‑core tiles
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles along j
    )

    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        # num_stages=2  # optional, can enable for deeper pipelines
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast everything to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (same device as inputs)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Gating projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # later used after to_out_norm

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → element‑wise gate multiplication in‑place
        left  = left * left_gate
        right = right * right_gate
    else:
        # Convert mask to fp16 (0/1) and fuse mask+gate
        mask_fp16 = mask_tensor.to(dtype_fp16)          # [B,N,N]
        # Ensure contiguity for Triton kernels
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid_mask = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid_mask](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            hidden_dim=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Release temporary tensors early
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via custom Triton GEMM
    # ------------------------------------------------------------------
    # left / right now are [B, N, N, hidden_dim] (FP16)
    out = _tri_mul_triton(left, right)   # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (on hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply final out‑gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                 # element‑wise (FP16)
    out = F.linear(out, to_out_w)        # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
