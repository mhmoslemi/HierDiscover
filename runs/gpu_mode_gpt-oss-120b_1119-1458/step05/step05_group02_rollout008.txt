def trmul_gemm_kernel(a_ptr, b_ptr, c_ptr,
                      M, N, K,
                      stride_am, stride_ak, stride_ab,
                      stride_bk, stride_bn, stride_bb,
                      stride_cm, stride_cn, stride_cb,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(1)
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + offs_k[None,:] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bb + offs_k[:, None] * stride_bk + offs_n[None,:] * stride_bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # a shape (BLOCK_M, BLOCK_K), b shape (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

        # increment pointer offsets for next iteration
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Cast to fp16 and store

    # compute c_ptrs

"""
Triton‑accelerated **outgoing** TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1.  Layer‑Norm the 4‑D input (float32) → cast to fp16.
2.  Linear projections (left/right) and three gate‑projections (all fp16).
3.  If a mask is present, fuse element‑wise multiplication
    left = left * left_gate * mask   and   right = right * right_gate * mask
    with a tiny Triton kernel that broadcasts the mask over the hidden dim.
4.  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom batched GEMM Triton kernel.  The hidden dimension
    is folded into the batch dimension, i.e. we compute B·H independent
    matrix‑multiplications A @ Bᵀ where A,B∈ℝ^{N×N}.
5.  2nd LayerNorm (fp16) over the hidden dimension, out‑gate (sigmoid) and
    element‑wise gating.
6.  Final linear projection back to the original dimension and cast to float32.

Only the forward pass is implemented – gradients are not required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse element‑wise multiplication with an (optional) mask.
# x, gate have shape (B, N, N, H) → flatten = B·N·N·H.
# mask has shape (B, N, N) and is broadcast over H.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,           # *fp16 → tensor to be updated (left or right)
    gate_ptr,        # *fp16 → gate tensor (same shape as x)
    mask_ptr,        # *fp16 → mask tensor (flattened B·N·N)
    hidden_dim,      # i32 → size of the hidden dimension H
    size,            # i32 → total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                     # linear indices of this block
    active = offs < size                                          # boundary guard

    # mask index = floor(offs / H) because H is the innermost stride
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)   # default = 1 (no masking)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
# A : [BATCH, M, K]   (left)
# B : [BATCH, K, N]   (rightᵀ – we treat the original right tensor as transposed)
# C : [BATCH, M, N]   (output)
# The kernel loops over K tiles and accumulates into FP32.
# ----------------------------------------------------------------------
@triton.jit
def trmul_gemm_kernel(
    a_ptr,          # *fp16 → left (BATCH, M, K)
    b_ptr,          # *fp16 → right (BATCH, K, N)   (original right treated as transposed)
    c_ptr,          # *fp16 → output (BATCH, M, N)

    M, N, K,        # i32   → matrix dimensions
    stride_am,      # i32   → A row stride (M‑axis)
    stride_ak,      # i32   → A column stride (K‑axis)
    stride_ab,      # i32   → A batch stride

    stride_bk,      # i32   → B row stride (K‑axis) – original column stride
    stride_bn,      # i32   → B column stride (N‑axis) – original row stride
    stride_bb,      # i32   → B batch stride

    stride_cm,      # i32   → C row stride (M‑axis)
    stride_cn,      # i32   → C column stride (N‑axis)
    stride_cb,      # i32   → C batch stride

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ----- program IDs -------------------------------------------------
    batch_id = tl.program_id(1)                     # BATCH dimension
    pid = tl.program_id(0)                          # (M‑block, N‑block) linear ID

    # number of blocks in each dimension
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n                        # block row index
    pid_n = pid % num_pid_n                         # block column index

    # ----- offsets for this block --------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                     # [BLOCK_K]

    # Guard against out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- pointers ----------------------------------------------------
    # A: (batch, M, K)
    a_ptrs = a_ptr + batch_id * stride_ab \
                   + offs_m[:, None] * stride_am \
                   + offs_k[None, :] * stride_ak
    # B: (batch, K, N) – note that B is stored as the original right tensor,
    #                     therefore its K‑stride = original column stride
    b_ptrs = b_ptr + batch_id * stride_bb \
                   + offs_k[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # accumulator in FP32 (higher precision, avoids overflow)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main K‑loop -------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # load tiles with appropriate masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # block‑wise matrix multiplication (A_tile @ B_tile)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----- write back --------------------------------------------------
    # cast accumulator back to fp16 (Tensor‑Core format)
    c = acc.to(tl.float16)

    # C pointer (output)
    c_ptrs = c_ptr + batch_id * stride_cb \
                   + offs_m[:, None] * stride_cm \
                   + offs_n[None, :] * stride_cn

    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper: batched GEMM using the Triton kernel above.
# ----------------------------------------------------------------------
def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    via a Triton kernel.  ``left`` and ``right`` must have shape
    (B, N, N, H) and be fp16 & contiguous.
    """
    B, N, _, H = left.shape
    # ----------------------------------------------------
    # 1) Collapse batch and hidden dimensions → (B⋅H, N, N)
    # ----------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # ----------------------------------------------------
    # 2) Allocate output buffer (same shape, fp16)
    # ----------------------------------------------------
    out_flat = torch.empty_like(left_flat)

    # ----------------------------------------------------
    # 3) Compute stride arguments (in elements)
    # ----------------------------------------------------
    # A = left_flat; shape (batch, M, K) = (batch, N, N)
    stride_ab = left_flat.stride(0)   # batch stride
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)

    # B = right_flat (treated as transposed inside kernel);
    # shape (batch, K, N) where K = original N, N = original N
    stride_bb = right_flat.stride(0)  # batch stride
    stride_bk = right_flat.stride(2)  # stride over original K (column) = 1
    stride_bn = right_flat.stride(1)  # stride over original N (row)   = N

    # C = out_flat
    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # ----------------------------------------------------
    # 4) Choose tile sizes – 64×64×64 works well for N up to 1024.
    # ----------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m * num_pid_n, B * H)   # (M‑N tiles, batch)

    trmul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                        # M, N, K
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # ----------------------------------------------------
    # 5) Reshape back to (B, N, N, H)
    # ----------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Main entry‑point: forward pass of the outgoing TriMul.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim]        (float32)
        - mask_tensor   : torch.Tensor   [B, N, N] (float32/bool) or None
        - weights_dict  : dict name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``;
                           optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – applied on the raw input.
    # ----------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                # [B, N, N, dim]  (float32)

    # ----------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy lifting.
    # ----------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------
    # 3️⃣ Cast model weights once (fp16) – bias terms are absent.
    # ----------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ----------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ----------------------------------------------------
    # 5️⃣ Gate tensors (sigmoid)
    # ----------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ----------------------------------------------------
    # 6️⃣ Optional mask – fused element‑wise multiplication.
    # ----------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16, ensure contiguous layout.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024                              # ≤ 1024 threads per block
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication.
        left = left * left_gate
        right = right * right_gate

    # ----------------------------------------------------
    # 7️⃣ O(N³) contraction via a batched GEMM Triton kernel.
    # ----------------------------------------------------
    out = _batched_gemm_triton(left, right)   # [B, N, N, H]  (fp16)

    # ----------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over the hidden dimension.
    # ----------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------
    # 9️⃣ Gating and final projection back to the original dimension.
    # ----------------------------------------------------
    out = out * out_gate                       # element‑wise out‑gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (fp16)

    # ----------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return.
    # ----------------------------------------------------
    return out.to(torch.float32)
