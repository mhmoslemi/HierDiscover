    stride_c_h: tl.constexpr integer,

    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_j = offs_j < N

def trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_j, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    HIDDEN_DIM: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # combined batch (b*H + h)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # compute batch and hidden indices
    b = pid // HIDDEN_DIM
    h = pid - b * HIDDEN_DIM
    
    # compute base pointers for this batch+head
    a_base = a_ptr + b * stride_a_batch + h
    b_base = b_ptr + b * stride_b_batch + h
    c_base = c_ptr + b * stride_c_batch + h
    
    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        
        # compute a and b pointer offsets
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_j
        
        # load data
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        
        # accumulate dot product
        acc += tl.dot(a, b)
    
    # store result
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

import torch

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    hidden_dim,
    size,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default mask = 1
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

@triton.jit
def tri_mul_outgoing_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    HIDDEN_DIM: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    K_ITER: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    tile_i = tl.program_id(1)
    tile_j = tl.program_id(2)

    # Decode batch and head index
    b = batch_idx // HIDDEN_DIM
    h = batch_idx % HIDDEN_DIM

    # Base pointers for the current (b, h) slice
    a_base = a_ptr + b * stride_a_batch + h
    b_base = b_ptr + b * stride_b_batch + h
    c_base = c_ptr + b * stride_c_batch + h

    # Offsets within the tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32 for higher accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K (= sequence length)
    for kb in range(K_ITER):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load a tile of A and B (with transposed B)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + k_offset[None, :] * stride_a_k
        b_ptrs = b_base + k_offset[:, None] * stride_b_k + offs_j[None, :] * stride_b_j

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write out the result
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Fast forward pass of the outgoing TriMul operation.

    The implementation:
     * Performs the first LayerNorm in fp32, then casts to fp16.
     * Executes the five linear layers (2 projections + 3 gates) as separate fp16 ops.
     * Optionally fuses mask, projection and gate multiplication via a tiny Triton kernel.
     * Computes the O(N³) contraction using a custom batched‑GEMM Triton kernel
       (no extra permute or copy – we work directly on the [B,N,N,H] layout).
     * Applies the second LayerNorm, out‑gate and final linear projection.
    All heavy compute runs in fp16, and the final result is returned as fp32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable mean/var
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent ops
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Load linear weights (fp16)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16) – produce tensors of shape [B,N,N,hidden_dim]
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (fp16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Apply mask (if any) and combine with gates
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N]  (float16)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # [B*N*N]
        size = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via a custom Triton batched GEMM
    # -----------------------------------------------------------------
    batch, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty_like(left)  # [B,N,N,H] fp16

    # Strides (in element counts)
    stride_a_batch = left.stride(0)
    stride_a_i     = left.stride(1)
    stride_a_k     = left.stride(2)
    stride_b_batch = right.stride(0)
    stride_b_k     = right.stride(2)
    stride_b_j     = right.stride(1)
    stride_c_batch = out.stride(0)
    stride_c_i     = out.stride(1)
    stride_c_j     = out.stride(2)

    # Tiling parameters – chosen to fit nicely into H100 shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    K_ITER = (N + BLOCK_K - 1) // BLOCK_K
    batch_count = batch * hidden_dim

    grid = (batch_count, num_tiles_i, num_tiles_j)

    tri_mul_outgoing_gemm_kernel[grid](
        left,
        right,
        out,
        N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_j,
        stride_c_batch, stride_c_i, stride_c_j,
        HIDDEN_DIM=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        K_ITER=K_ITER,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Gating and final projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # -> [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)

   for both left and right streams.
4. The O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom batched‑GEMM Triton kernel that works directly on the
   [B,N,N,H] layout (no permute / copy).  The kernel treats each (batch,head)
   pair as a separate GEMM and accumulates in float32 before storing fp16.
5. Second Layer‑norm (fp16) → element‑wise out_gate → final linear projection
   back to the original channel dimension.
6. Cast the result back to float32 (the harness expects float32).

All heavy compute runs in fp16, only the final output is upcast.
A Triton kernel is used for mask‑gate fusion (when a mask exists) and for the
core N³ contraction, ensuring a fully GPU‑resident implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: fused element‑wise (tensor * gate * mask)
# --------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be updated (flattened)
    gate_ptr,       # *fp16  → gate tensor (same flattening as x)
    mask_ptr,       # *fp16  → mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32   → hidden dimension H
    size,           # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the three operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Map each group of `hidden_dim` consecutive x‑elements to one mask entry
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default mask=1

    # Fused multiplication
    out = x * g * m

    # In‑place store
    tl.store(x_ptr + offs, out, mask=valid)


# --------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # *fp16 pointers to left, right, out
    N,                                   # i32 – sequence length
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    HIDDEN_DIM: tl.constexpr,            # compile‑time hidden dimension (head)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    K_ITER: tl.constexpr,                # compile‑time # of K‑tiles
):
    """
    Compute for each (batch, head):
        C = A @ Bᵀ   where A, B ∈ ℝ^{N×N} (fp16) and C ∈ ℝ^{N×N}
    Shapes:
        A[b, h, i, k] = left[b, i, k, h]
        B[b, h, j, k] = right[b, j, k, h]   (transposed inside the kernel)
        C[b, h, i, j] = out[b, i, j, h]
    """
    batch_idx = tl.program_id(0)   # 0‑axis → combined (batch, head)
    tile_i   = tl.program_id(1)    # 1‑axis → block row   (i)
    tile_j   = tl.program_id(2)    # 2‑axis → block col   (j)

    # Decode batch and head indices
    b = batch_idx // HIDDEN_DIM
    h = batch_idx % HIDDEN_DIM

    # Base pointers for the selected (b, h) slice (hidden‑dim stride == 1)
    a_base = a_ptr + b * stride_a_batch + h
    b_base = b_ptr + b * stride_b_batch + h
    c_base = c_ptr + b * stride_c_batch + h

    # Offsets inside the tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (fp32 for better numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (the reduction axis)
    for k_tile in range(K_ITER):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to the current sub‑tiles
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_j

        # Load tiles with masking at the boundaries
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate dot product of the two tiles
        acc += tl.dot(a, b)                     # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # Write the result back (fp16) with edge masking
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Entry point used by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) or any placeholder
        - weights_dict : model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable statistics
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)                     # [B, N, N, dim] – fp16

    # -----------------------------------------------------------------
    # 3️⃣ Load linear weights into fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (fp16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask: fuse mask * gate * projection via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                            # [B*N*N]
        size = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – plain element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via custom Triton batched GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                 # H == hidden_dim
    out = torch.empty_like(left)            # [B, N, N, hidden_dim] fp16

    # Strides expressed in number of elements (not bytes)
    stride_a_batch = left.stride(0)   # B stride = N*N*hidden_dim
    stride_a_i     = left.stride(1)   # i stride = N*hidden_dim
    stride_a_k     = left.stride(2)   # k stride = hidden_dim

    stride_b_batch = right.stride(0)
    stride_b_k     = right.stride(2)      # same as left.stride(2)
    stride_b_j     = right.stride(1)      # j stride = N*hidden_dim

    stride_c_batch = out.stride(0)
    stride_c_i     = out.stride(1)
    stride_c_j     = out.stride(2)

    # Tiling parameters – chosen to keep threads ≤ 1024 per block
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    K_ITER = (N + BLOCK_K - 1) // BLOCK_K
    batch_count = B * hidden_dim

    grid = (batch_count, num_tiles_i, num_tiles_j)

    tri_mul_outgoing_gemm_kernel[grid](
        left,
        right,
        out,
        N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_j,
        stride_c_batch, stride_c_i, stride_c_j,
        HIDDEN_DIM=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        K_ITER=K_ITER,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, hidden_dim] – fp16

    # -----------------------------------------------------------------
    # 9️⃣ Gating + final linear projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                                 # element‑wise out‑gate
    out = torch.nn.functional.linear(out, to_out_w)      # [B, N, N, dim] – fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
