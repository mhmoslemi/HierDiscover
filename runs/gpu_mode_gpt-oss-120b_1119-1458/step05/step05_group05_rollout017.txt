"""
Highly‑optimized forward pass for the “outgoing” TriMul operation (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32 → keep numerical stability) and cast to fp16.
2️⃣ Compute three linear projections (left/right) and three gate projections
   (left_gate, right_gate, out_gate) using the supplied FP32 weights that are
   cast once to FP16.
3️⃣ Fuse the element‑wise mask‑*‑gate multiplication for *both* left and right
   tensors in a single Triton kernel (executed only when a mask is present).
4️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed with a single torch.einsum call.  This avoids the two
   ``permute+contiguous`` copies required by the naïve batched‑GEMM approach
   while still exploiting the highly‑optimized CUTLASS kernels used under the
   hood of ``einsum`` for this pattern.
5️⃣ Apply a second LayerNorm (over the hidden dimension) → element‑wise out‑gate.
6️⃣ Final linear projection back to the model dimension.
7️⃣ Cast the result back to float32 (the harness expects float32).

Only the mask‑gate fusion requires a custom Triton kernel – everything else
remains in PyTorch to keep the implementation short and portable.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def mask_gate_fuse_two_tensors(
    left_ptr,               # *fp16 – left tensor (flattened)
    left_gate_ptr,          # *fp16 – left gate (flattened)
    right_ptr,              # *fp16 – right tensor (flattened)
    right_gate_ptr,         # *fp16 – right gate (flattened)
    mask_ptr,               # *fp16 – mask tensor (flattened, shape B·N·N)
    hidden_dim,             # i32 – size of the hidden dimension (H)
    size,                   # i32 – total number of elements (B·N·N·H)
    BLOCK: tl.constexpr,    # compile‑time tile size (≤1024)
):
    """
    Fuse   left  = left  * left_gate  * mask
           right = right * right_gate * mask
    with one kernel launch.  The mask has no hidden dimension, therefore
    ``mask_idx = element_index // hidden_dim``.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands
    l_val = tl.load(left_ptr + offs, mask=valid, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=valid, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Compute mask index (each mask entry corresponds to hidden_dim consecutive values)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fused updates
    l_out = l_val * l_gate * m
    r_out = r_val * r_gate * m

    # Write back in‑place
    tl.store(left_ptr + offs, l_out, mask=valid)
    tl.store(right_ptr + offs, r_out, mask=valid)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])
    nomask = bool(config.get("nomask", True))

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast all model weights to fp16 (once)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Gate activations (fp16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fuse element‑wise multiplication via a single Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and flatten it: shape [B * N * N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)

        # All tensors are already contiguous after the linear/gate ops
        size = left.numel()                     # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        mask_gate_fuse_two_tensors[grid](
            left, left_gate,            # left & its gate
            right, right_gate,          # right & its gate
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → plain element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – use Einstein summation which internally
    #     reduces to a batched GEMM without the extra permute+contiguous ops.
    # -----------------------------------------------------------------
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over the hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                         # element‑wise out‑gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] (fp16)

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
