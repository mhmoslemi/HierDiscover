```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
Optimized outgoing TriMul kernel (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) → cast to FP16.
2. Four linear projections (left/right) + three gating linear layers (FP16).
3. If a mask is present, fuse mask‑multiplication and gating with a tiny
   Triton element‑wise kernel (in‑place update of left/right tensors).
4. Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication C = A @ Bᵀ with
   A,B ∈ ℝ^{(B·H)×N×N}.  We perform it with `torch.bmm`,
   which on H100 uses Tensor‑core‑accelerated FP16 GEMM.
5. Second LayerNorm (FP16) → multiply by out_gate (sigmoid) → final linear
   projection back to the original `dim`.
6. Cast the result back to FP32 (the harness expects FP32).

A minimal Triton kernel is used to apply the optional mask together with the
gates in a single pass, reducing memory traffic and satisfying the
“implement part of the op in a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * mask * left_gate
#   right ← right * mask * right_gate
# All tensors are contiguous FP16 with shape [B, N, N, H] (left/right/gates)
# and mask has shape [B, N, N] (broadcast along H).
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # int  (B·N·N·H)
    H,                 # int  hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat indices
    mask = offs < total_elements                                 # OOB guard

    # mask index = flat_index // H  (broadcast across hidden dim)
    mask_offs = offs // H

    # Load inputs (masked loads guarantee zero for OOB)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask (float16, 0/1).  For OOB positions we use a neutral value 1.0.
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store results back in‑place
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform C = A @ Bᵀ where A and B are [B, N, N, H] (FP16).
    Returns C with the same shape and dtype.
    """
    B, N, _, H = left.shape
    # Collapse batch & hidden into a single leading dimension
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B·H, N, N)

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))

    # Restore original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()      # (B, N, N, H)
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)          # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (same device as inputs)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Gating projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # used later

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (Triton kernel)
    #    – if ``nomask`` is False we fuse mask, left_gate and right_gate.
    #    – otherwise we simply multiply gates in‑place.
    # ------------------------------------------------------------------
    if nomask:
        # No mask → just gate the projections
        left  = left * left_gate
        right = right * right_gate
    else:
        # Prepare mask as FP16 (broadcasted on hidden dim)
        mask_fp16 = mask_tensor.to(dtype_fp16)                # [B,N,N]
        # Ensure all tensors are contiguous for the kernel
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()
        BLOCK = 1024  # ~1K threads per block; good occupancy
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates are now fused; we can free the gate tensors early
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched FP16 GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # left / right are now [B,N,N,H]; we need C = left @ rightᵀ per (B,H)
    out = _batched_matmul_fp16(left, right)   # [B, N, N, H] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16) over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply final out‑gate and project back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise (FP16)
    out = F.linear(out, to_out_w)            # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
