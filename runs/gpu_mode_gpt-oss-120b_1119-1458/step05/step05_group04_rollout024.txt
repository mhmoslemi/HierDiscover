def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', True)

    device = input_tensor.device

    # Step 1: LayerNorm
    norm_weight = weights['norm.weight']; norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # Move weights to fp16
    left_proj_weight = weights['left_proj.weight'].to(dtype=torch.float16)
    left_gate_weight = weights['left_gate.weight'].to(dtype=torch.float16)
    right_proj_weight = weights['right_proj.weight'].to(dtype=torch.float16)
    right_gate_weight = weights['right_gate.weight'].to(dtype=torch.float16)
    out_gate_weight = weights['out_gate.weight'].to(dtype=torch.float16)
    to_out_norm_weight = weights['to_out_norm.weight'].to(dtype=torch.float16)
    to_out_norm_bias = weights['to_out_norm.bias'].to(dtype=torch.float16)
    to_out_weight = weights['to_out.weight'].to(dtype=torch.float16)

    # Combine projection+gate weights for left and right
    left_combined_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0) # shape (2*hidden_dim, dim)
    right_combined_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)

    # Linear projections
    left_combined = torch.nn.functional.linear(x, left_combined_weight) # shape [B,N,N,2*hidden_dim]
    right_combined = torch.nn.functional.linear(x, right_combined_weight)

    # Split into projection and gates
    left_proj = left_combined[..., :hidden_dim] # shape [B,N,N,hidden_dim]
    left_gate_logits = left_combined[..., hidden_dim:] # shape [B,N,N,hidden_dim]
    left_gate = torch.sigmoid(left_gate_logits)

    right_proj = right_combined[..., :hidden_dim]
    right_gate_logits = right_combined[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_logits)

    # If mask needed combine mask
    if not nomask:
        # Convert mask to fp16 and unsqueeze for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
    else:
        mask_fp16 = None

    # Apply fused gating and mask (if any) using Triton kernel
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    if not nomask:
        # apply mask and gate to projection to obtain left
        # left = left_proj * left_gate * mask
        # Using Triton kernel
        _apply_gate_mask_inplace(left_proj, left_gate, mask_fp16)
    else:
        _apply_gate_mask_inplace(left_proj, left_gate, None)

    # Similarly for right
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()
    _apply_gate_mask_inplace(right_proj, right_gate, mask_fp16)

    # Now left_proj and right_proj contain final left and right after gating and mask

    left = left_proj
    right = right_proj

    # Contraction: batched matmul
    B, N, _, H = left.shape
    left_perm = left.permute(0,3,1,2).contiguous() # shape (B,H,N,N)
    right_perm = right.permute(0,3,1,2).contiguous()
    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2)) # shape (B*H, N, N)
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out_gate linear and sigmoid
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gating via kernel
    out = out * out_gate  # can fuse multiply using kernel if we want
    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    # Cast to float32
    return out.to(torch.float32)
```

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for fused gating and optional mask multiplication
@triton.jit
def _gate_mask_fuse_kernel(
    out_ptr,        # *fp16 output (in-place)
    proj_ptr,       # *fp16 projection (input)
    gate_ptr,       # *fp16 gate (sigmoid output)
    mask_ptr,       # *fp16 mask (for broadcast), may be nullptr
    total_elems,    # total number of elements in projection and gate (== out elements)
    hidden_dim,     # size of the last dim (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # mask broadcast over hidden_dim
        mask_off = offs // hidden_dim
        m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(1.0))
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_mask_inplace(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor | None):
    """
    In‑place fusion of projection, sigmoid gate and optional mask.
    proj/gate are fp16 tensors of identical shape (contiguous).
    mask (if not None) is a fp16 tensor of shape [B, N, N] (flat view is fine).
    """
    assert proj.is_contiguous() and gate.is_contiguous()
    total = proj.numel()
    BLOCK = 1024
    has_mask = mask is not None
    grid = ((total + BLOCK - 1) // BLOCK,)
    mask_ptr = mask if has_mask else torch.empty((0,), dtype=torch.float16, device=proj.device)
    _gate_mask_fuse_kernel[grid](
        proj,        # out_ptr
        proj,        # proj_ptr (read)
        gate,        # gate_ptr
        mask_ptr,    # mask_ptr
        total,
        proj.shape[-1],
        BLOCK_SIZE=BLOCK,
        HAS_MASK=1 if has_mask else 0,
        num_warps=8,
    )
    # No return, operation modifies proj in-place

@triton.jit
def _pointwise_mul_kernel(
    out_ptr, a_ptr, b_ptr, total, BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, a * b, mask=mask)

def _pointwise_mul_inplace(a: torch.Tensor, b: torch.Tensor):
    """In‑place element‑wise multiplication a ← a * b (both fp16, contiguous)."""
    assert a.is_contiguous() and b.is_contiguous()
    total = a.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        a, a, b, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (AlphaFold‑3 style).
    • FP32 LayerNorm → FP16.
    • Fuse left/right projection + gate linear layers.
    • Fuse projection‑gate‑mask multiplication via Triton.
    • Batched GEMM (fp16×fp16→fp32) implements the O(N³) contraction.
    • Second FP16 LayerNorm → out‑gate → final linear.
    Returns float32 output of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32) -> FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x_norm.to(torch.float16)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Load model weights in FP16 on right device
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w = _to_fp16(weights["left_proj.weight"])
    left_gate_w = _to_fp16(weights["left_gate.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _to_fp16(weights["to_out_norm.bias"])
    to_out_w = _to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Fuse projection + gate for left & right using one Linear each
    # ------------------------------------------------------------------
    # Combine weight matrices: [proj;gate] -> shape (2*hidden_dim, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*hdim, dim)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Linear: result shape [B, N, N, 2*hidden_dim]
    left_combined = torch.nn.functional.linear(x, left_combined_w)
    right_combined = torch.nn.functional.linear(x, right_combined_w)

    # Slice projection and gate logits
    left_proj = left_combined[..., :hidden_dim]          # [B,N,N,hdim]
    left_gate_logit = left_combined[..., hidden_dim:]    # [B,N,N,hdim]
    right_proj = right_combined[..., :hidden_dim]
    right_gate_logit = right_combined[..., hidden_dim:]

    # sigmoid on gate logits (fp16)
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)

    # ------------------------------------------------------------------
    # 4️⃣  Apply mask (if any) and fuse multiplication (proj * gate * mask)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → fp16 → flat view
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)   # shape B*N*N
    else:
        mask_flat = None

    # Ensure tensors are contiguous for Triton
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()

    _apply_gate_mask_inplace(left_proj, left_gate, mask_flat)
    _apply_gate_mask_inplace(right_proj, right_gate, mask_flat)

    # At this point left_proj and right_proj hold the gated, masked tensors
    left = left_proj
    right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣  Heavy cubic contraction using batched GEMM (fp16×fp16 → fp32)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    # reshape to [B*H, N, N] for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[i,k] * right_flat[j,k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Compute out_gate (sigmoid of linear(x))

    M, N, K,

):

    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row index
    pid_n = tl.program_id(1)   # tile column index
    pid_z = tl.program_id(2)   # batch / head index

    # Compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute offsets for current batch
    batch_off_a = pid_z * stride_batch_a
    batch_off_b = pid_z * stride_batch_b
    batch_off_c = pid_z * stride_batch_c

    # Pointers for block of A and B
    a_ptrs = a_ptr + batch_off_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_off_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=tl.float16(0.0))
        # Multiply-accumulate (tensor core)
        acc += tl.dot(a, b)   # acc shape (BLOCK_M, BLOCK_N)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Convert to fp16
    c = acc.to(tl.float16)

    # Compute output pointers (with mask)
    c_ptrs = c_ptr + batch_off_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # Store output with mask
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑fused implementation
----------------------------------------------------------------
Algorithm (FP16 for the heavy work, FP32 for the first LayerNorm):
1️⃣  LayerNorm the [B,N,N,dim] input in FP32, cast to FP16.
2️⃣  Fuse each projection + gate linear layer (proj + sigmoid‑gate) into a
    single matmul of shape (2·hidden_dim, dim).  This halves the memory
    traffic for the large input tensor.
3️⃣  Fuse the element‑wise projection * gate * optional mask into a
    single Triton kernel (≈1 k elements per thread block).  No extra
    temporaries are allocated.
4️⃣  Compute the cubic contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM (fp16×fp16 → fp32 accumulation) via a custom
    Triton kernel that processes all (B·hidden_dim) batches in one launch.
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → elementwise multiply.
6️⃣  Final Linear back to the original `dim`.
7️⃣  Cast the result to float32.

All heavy linear & elementwise work is performed with Triton, while the
batched GEMM uses Tensor‑Core‑accelerated fp16 matmul.  The implementation
matches the reference forward pass (no gradients required) and runs fully on
the GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse proj * gate * (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    out_ptr,                 # *fp16  output (in‑place)
    proj_ptr,                # *fp16  projection (left/right)
    gate_ptr,                # *fp16  sigmoid‑gate
    mask_ptr,                # *fp16  mask (flattened), ignored if HAS_MASK=0
    total_elems,             # total number of elements in proj/gate (B·N·N·H)
    hidden_dim,              # size of the last dimension (H)
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=valid, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcast over hidden_dim: mask_index = element_index // hidden_dim
        mask_index = offs // hidden_dim
        m = tl.load(mask_ptr + mask_index, mask=valid, other=tl.float16(1.0))
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_mask_inplace(proj: torch.Tensor,
                             gate: torch.Tensor,
                             mask: torch.Tensor | None):
    """In‑place fusion of proj * sigmoid(gate) * optional mask.
    All tensors must be fp16 and contiguous.
    """
    assert proj.is_contiguous() and gate.is_contiguous()
    total = proj.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy = torch.empty((0,), dtype=torch.float16, device=proj.device)
        _gate_mask_fuse_kernel[grid](
            proj,               # out_ptr
            proj,               # proj_ptr
            gate,               # gate_ptr
            dummy,              # mask_ptr (unused)
            total,
            proj.shape[-1],
            BLOCK_SIZE=BLOCK,
            HAS_MASK=0,
            num_warps=8,
        )
    else:
        mask_flat = mask.view(-1)  # shape B·N·N
        _gate_mask_fuse_kernel[grid](
            proj,
            proj,
            gate,
            mask_flat,
            total,
            proj.shape[-1],
            BLOCK_SIZE=BLOCK,
            HAS_MASK=1,
            num_warps=8,
        )


# ----------------------------------------------------------------------
# Triton kernel: batched fp16 GEMM (fp16×fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K) strides
    stride_bk, stride_bn,          # B: (batch, K, N) strides (transposed operand)
    stride_cm, stride_cn,          # C: (batch, M, N) strides
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # batch index (B·H)

    # ----- offsets inside the current tile -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ----- base pointers for this batch -----
    batch_off_a = pid_z * stride_batch_a
    batch_off_b = pid_z * stride_batch_b
    batch_off_c = pid_z * stride_batch_c

    # pointers to the beginning of the current tile
    a_ptrs = a_ptr + batch_off_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_off_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # mask out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=mask_b, other=tl.float16(0.0))

        # Tensor‑core dot (fp16×fp16→fp32)
        acc += tl.dot(a, b)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back result (FP16)
    c = acc.to(tl.float16)

    c_ptrs = c_ptr + batch_off_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask_c)


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for fp16 matrices:
        a : (B, M, K)  fp16
        b : (B, K, N)  fp16   (already transposed in the caller)
    Returns
        c : (B, M, N)  fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, Kb, N = b.shape
    assert K == Kb

    # output tensor (fp16)
    c = torch.empty_like(a)  # same shape as a (B, M, N)

    # strides (in elements)
    stride_batch_a = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    stride_batch_b = b.stride(0)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)

    stride_batch_c = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    # tiling parameters (tuned for H100 FP16 Tensor‑cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (M + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, B)

    _batched_matmul_fp16_kernel[grid](
        a, b, c,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_batch_a, stride_batch_b, stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return c

    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))

def _elemwise_mul_inplace(x: torch.Tensor, y: torch.Tensor):

    else:
        mask_fp16 = None

    # Ensure tensors are contiguous (they already are, but make it explicit)
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()

    _apply_gate_mask_inplace(left_proj, left_gate, mask_fp16)
    _apply_gate_mask_inplace(right_proj, right_gate, mask_fp16)

    # At this point `left_proj` and `right_proj` hold the gated, masked tensors
    left = left_proj
    right = right_proj

    # --------------------------------------------------------------
    # 5️⃣ Heavy N³ contraction – batched GEMM (fp16×fp16→fp32)
    # --------------------------------------------------------------
    B, N_seq, _, H = left.shape                     # H == hidden_dim
    # reshape to [B·H, N, N] for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N_seq, N_seq)          # (B·H, N, N)
    # transpose the second operand: (B·H, K, N) where K=N
    right_T = right_perm.transpose(2, 3).contiguous()      # (B, H, N, N)
    right_flat_T = right_T.view(B * H, N_seq, N_seq)        # (B·H, N, N)

    # custom batched fp16 GEMM
    out_flat = _batched_matmul_fp16(left_flat, right_flat_T)   # (B·H, N, N)

    # restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # out‑gate (sigmoid)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # fuse out * out_gate
    _elemwise_mul_inplace(out, out_gate)

    # final projection back to original dimensionality
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  fp16

    # --------------------------------------------------------------
    # 7️⃣ Cast to float32 (required by the harness) and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
