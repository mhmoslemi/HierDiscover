"""
TriMul (outgoing) – fast forward pass in half precision.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣  Fuse the *projection* and *gate* linear layers of the left and right path
    by stacking their weights (2·hidden_dim × dim) and performing a single
    mat‑mul per side.  The gate pre‑activations are then passed through sigmoid.
3️⃣  If a mask is supplied, fuse the three element‑wise factors
    (proj, gate, mask) with a tiny Triton kernel (≈ 1 µs).  When `nomask=True`
    we simply multiply proj and gate in FP16.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is exactly a batched GEMM `C = A @ Bᵀ`.  We reshape the tensors to
    (B·hidden_dim, N, N) and call `torch.bmm`, which on H100 runs on Tensor
    Cores (FP16) and dominates the runtime.
5️⃣  Second LayerNorm (FP16) on the result, element‑wise out‑gate,
    and final linear projection back to `dim`.
6️⃣  Cast the output back to FP32 (the harness expects float32).

All weights are loaded once and converted to FP16.  The only custom Triton
kernel performs the mask‑gate‑project fusion, satisfying the “must‑use‑Triton”
requirement while keeping launch overhead negligible.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused multiplication of {proj * gate * mask}
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,                     # *fp16 – tensor to be updated (flattened)
    gate_ptr,                     # *fp16 – gating tensor (same flattening)
    mask_ptr,                     # *fp16 – mask tensor, shape [B*N*N]
    hidden_dim,                   # i32   – size of the hidden dimension (H)
    total_elems,                  # i32   – total number of elements in proj (B*N*N*H)
    BLOCK: tl.constexpr,          # compile‑time tile size (≤ 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load projection and gate
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Each contiguous block of `hidden_dim` elements shares the same mask entry
    mask_idx = offs // hidden_dim
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fused element‑wise multiplication
    out = proj * gate * mask

    tl.store(proj_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper – batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single batched matrix‑multiply (A @ Bᵀ).

    All tensors must be FP16 and contiguous.
    """
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] and perform A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                 # [B·H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        # Right side
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ----------------------- 7️⃣ heavy O(N³) contraction ---------------
    # left_proj and right_proj now hold the gated & (masked) tensors
    # They are already FP16 and contiguous.
    out = _batched_gemm_fp16(left_proj, right_proj)   # [B,N,N,hidden_dim] FP16

    # ----------------------- 8️⃣ second LayerNorm ----------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # FP16

    # ----------------------- 9️⃣ out‑gate + final projection -----------
    out = out * out_gate                              # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # ----------------------- 10️⃣ cast back to FP32 --------------------
    return out.to(torch.float32)
```
