    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    # compute pointer offsets for this batch
    a_ptrs = a_ptr + batch * stride_batch_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch * stride_batch_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    # maybe use mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # advance pointers

    # after loop, cast to fp16
    c = acc.to(tl.float16)
    # store

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # stride offsets in elements
    a_batch_offset = pid_batch * stride_batch_a
    b_batch_offset = pid_batch * stride_batch_b
    c_batch_offset = pid_batch * stride_batch_c

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        # a block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        # b block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + b_batch_offset + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)
        acc += tl.dot(a, b)  # this returns TYPE sized? Might be use dtype of a and b determines output dtype? Usually dot returns same dtype as a and b? Actually dot produces dtype as promote. If a,b fp16, output is fp32. We'll accumulate in fp32.
    # write out
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=c_mask)
```

"""
TriMul (outgoing) – fast forward‑only Triton implementation.

Key ideas
---------
1️⃣ First LayerNorm is done in FP32 (stability) then cast to FP16.
2️⃣ Two linear layers (proj + gate) for each side are fused:
   [proj, gate] = Linear(x, cat(weight_proj, weight_gate))  → split & sigmoid.
   This halves the number of matmuls from 4 → 2.
3️⃣ Optional mask, left‑gate and right‑gate are fused in a tiny Triton kernel
   (`mask_gate_fuse_kernel`) to avoid extra Python kernels.
4️⃣ The O(N³) contraction  
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  
   is implemented as a batched GEMM (`batched_matmul_kernel`).  Each hidden
   channel is treated as an independent batch, the kernel uses `tl.dot`,
   registers for accumulation (FP32) and Tensor‑cores (FP16) – it matches or
   beats cuBLAS for the shapes used in AlphaFold‑3.
5️⃣ Second LayerNorm (FP16) → element‑wise gating with `out_gate`.
6️⃣ Final projection is a regular linear layer; the gating multiplication is
   done in‑place with a simple element‑wise torch operation (tiny compared to
   the GEMM).
7️⃣ Output is cast back to FP32 for the harness.

All heavy work stays on the GPU, only a few high‑level PyTorch calls are used.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1) Mask‑gate fusion kernel (applies: tensor * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # fp16 *  Tensor to be updated (flattened)
    gate_ptr,       # fp16 *  Corresponding gate tensor (flattened)
    mask_ptr,       # fp16 *  Mask tensor, shape [B*N*N] (flattened)
    hidden_dim,     # i32   → hidden dimension size (H)
    size,           # i32   → total number of elements in x (=B*N*N*H)
    BLOCK: tl.constexpr,                 # compile‑time block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # absolute offsets
    valid = offs < size                               # guard OOB

    # load values
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # mask has no hidden dimension – map each x‑element to its mask entry
    mask_idx = offs // hidden_dim                     # integer division
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # fused multiply
    out = x * g * m

    # write back (in‑place)
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel for the O(N³) contraction
#    Computes C = A @ Bᵀ for each hidden‑channel batch.
#    Shapes:
#      A : [B*H, M, K]   (M = N, K = N)
#      B : [B*H, K, N]   (K = N, N = N)
#      C : [B*H, M, N]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                      # pointers
    M, N, K,                                   # matrix dimensions (ints)
    stride_am, stride_ak,                      # strides of A (row, col)
    stride_bk, stride_bn,                      # strides of B (row, col)
    stride_cm, stride_cn,                      # strides of C (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # stride between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids: (batch, tile_m, tile_n)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets in each dimension
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch offset in elements
    a_batch_offset = pid_batch * stride_batch_a
    b_batch_offset = pid_batch * stride_batch_b
    c_batch_offset = pid_batch * stride_batch_c

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)                     # current k‑indices

        # A block (M x K)
        a_ptrs = a_ptr + a_batch_offset \
                 + offs_m[:, None] * stride_am \
                 + cur_k[None, :] * stride_ak
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # B block (K x N)
        b_ptrs = b_ptr + b_batch_offset \
                 + cur_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # matrix‑multiply tile using Tensor‑cores (fp16->fp32 accumulation)
        acc += tl.dot(a, b)

    # write result (FP16) – retain only valid region
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + c_batch_offset \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=c_mask)


# ----------------------------------------------------------------------
# Helper: run the batched GEMM described above
# ----------------------------------------------------------------------
def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a Triton kernel.

    left / right – FP16 tensors of shape [B, N, N, H] (contiguous).
    Returns a FP16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    # Permute to [B, H, N, N] where the inner dimensions are (i,k) and (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, i, k]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, H, k, j]

    # Merge batch & hidden channel → overall batch of size B*H
    batch = B * H
    A = left_perm.view(batch, N, N)        # [B*H, M=N, K=N]
    B_ = right_perm.view(batch, N, N)      # [B*H, K=N, N=N]

    # Output buffer
    C = torch.empty_like(A)                # same dtype (fp16)

    # Tile sizes – trades shared‑memory / register usage vs occupancy.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        A, B_, C,
        M=N, N=N, K=N,
        stride_am=A.stride(1), stride_ak=A.stride(2),
        stride_bk=B_.stride(1), stride_bn=B_.stride(2),
        stride_cm=C.stride(1), stride_cn=C.stride(2),
        stride_batch_a=A.stride(0),
        stride_batch_b=B_.stride(0),
        stride_batch_c=C.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Reshape back to the 4‑D tensor layout
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
