"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

We fuse the five linear projections (left_proj, left_gate, right_proj,
right_gate, out_gate) into a single matrix‑multiply (shape
[batch*seq_len*seq_len, dim] @ [dim, 5*hidden_dim]).  Afterwards we
separate the result, apply the optional mask, sigmoid‑gates and obtain
the gated left/right tensors.  The O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed as a batched GEMM C = A @ Bᵀ where the batch dimension is
`batch*hidden_dim`.  For small sequences (≤256) we launch a custom
Triton kernel that does mixed‑precision fp16·fp16→fp32 accumulation with
Tensor‑Cores (64×64×64 tiles).  For larger N we fall back to
torch.bmm (cuBLAS), which is faster on current GPUs for those sizes.
All LayerNorms are executed in float‑32 for numerical stability; the
rest of the pipeline uses float‑16 to keep memory bandwidth low.  The
final output is cast back to float‑32 as required by the harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]   (M = batch*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply for the triangular update."""
    pid_m = tl.program_id(0)   # tile along i (output rows)
    pid_n = tl.program_id(1)   # tile along j (output cols)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ---------------------------------------------------------------
    # tile offsets
    # ---------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (Tensor‑Core MMA accumulates in FP32 for FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # loop over K‑tiles
    # ---------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i, k]
        a_ptrs = (
            A_ptr
            + pid_z * stride_am               # batch*hidden offset
            + offs_i[:, None] * stride_ai     # i offset
            + offs_k[None, :] * stride_ak     # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load Bᵀ[j, k]  (i.e. B[k, j])
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm               # batch*hidden offset
            + offs_k[:, None] * stride_bk     # k offset first
            + offs_j[None, :] * stride_bj     # j offset second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # store C[i, j]
    # ---------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using a custom Triton GEMM.
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # tile sizes – 64×64 tiles with K‑tile of 64 work well on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using cuBLAS (torch.bmm). Operates in fp16
    with fp32 accumulation (default for torch.bmm on GPU).
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # A @ Bᵀ  (batched GEMM)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool)
                         – may be all‑ones (nomask=True)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fuse five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Convert weight tensors once to fp16 on the target device
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # shape of each: (hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*hidden_dim, dim)

    # fused linear: [B, N, N, 5*hidden_dim] (fp16)
    combined = F.linear(x, combined_weight)  # no bias as per module definition

    # ------------------------------------------------------------------
    # 5️⃣ split fused projection
    # ------------------------------------------------------------------
    # reshape to expose the 5 “heads” – (B, N, N, 5, hidden_dim)
    combined = combined.view(B, N, N, 5, hidden_dim)
    left      = combined[..., 0, :]  # [B, N, N, hidden_dim]
    left_gate = combined[..., 1, :]
    right     = combined[..., 2, :]
    right_gate= combined[..., 3, :]
    out_gate  = combined[..., 4, :]

    # ------------------------------------------------------------------
    # 6️⃣ optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left  * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ sigmoid gating (fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 8️⃣ apply gates
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 9️⃣ heavy contraction (O(N³))
    # ------------------------------------------------------------------
    # Choose kernel based on sequence length
    if N <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16
    else:
        contracted = _batched_matmul_torch(left, right)   # fp16

    # ------------------------------------------------------------------
    # 🔟 second LayerNorm (fp32 for stats, weight/bias are fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ apply output gate and final projection
    # ------------------------------------------------------------------
    contracted = contracted * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(contracted, to_out_w)   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ cast to float32 for the test harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
