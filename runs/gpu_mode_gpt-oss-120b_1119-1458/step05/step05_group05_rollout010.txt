"""
Highly‑optimized Triton + PyTorch forward pass for the outgoing TriMul
operator (AlphaFold‑3).  The implementation fuses the mask‑gate step in a
custom Triton kernel, reduces the number of dense linear layers by
concatenating projection+gate matrices, and uses a single strided‑batched
GEMM (cuBLAS) for the N³ contraction.  All heavy work is performed in
FP16; only the initial LayerNorm runs in FP32 for numerical stability.
The final result is cast back to FP32 as required by the harness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused multiplication with optional mask
#   out = (x * gate) * mask   (mask defaults to 1.0 when not provided)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,           # *fp16  – tensor to be updated (flattened)
    gate_ptr,        # *fp16  – gate tensor (same flattening as x)
    mask_ptr,        # *fp16  – mask tensor (flattened), length = B*L*L
    hidden_dim,      # i32    – hidden dimension H
    size,            # i32    – total number of elements in x (B*L*L*H)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Compute mask index: each block of `hidden_dim` entries shares one mask value
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Element‑wise fused multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: Strided‑batched GEMM for the O(N³) contraction
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the N³ contraction using a single cuBLAS strided‑batched GEMM.
    All tensors are expected to be (B, N, N, H) and fp16.
    """
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) – the view is contiguous because we
    # permute the hidden channel to batch position before .contiguous().
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM with transposed right-hand side
    c = torch.bmm(a, b.transpose(1, 2))                     # (B*H, N, N)
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the computation
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Load and concatenate linear weights (fp16)
    #    - left_proj + left_gate
    #    - right_proj + right_gate
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # concatenate proj/gate weights for left and right sides
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # -----------------------------------------------------------------
    # 4️⃣ Combined linear projections (fp16)
    #    Result shape: [B, N, N, 2*H]
    # -----------------------------------------------------------------
    left_comb = torch.nn.functional.linear(x, left_comb_w)   # (B,N,N,2H)
    right_comb = torch.nn.functional.linear(x, right_comb_w) # (B,N,N,2H)

    # -----------------------------------------------------------------
    # 5️⃣ Split projection / gate tensors and apply sigmoid to gates
    # -----------------------------------------------------------------
    left      = left_comb[..., :hidden_dim]                     # (B,N,N,H)
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])      # (B,N,N,H)

    right      = right_comb[..., :hidden_dim]                   # (B,N,N,H)
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])    # (B,N,N,H)

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask gate fusion (Triton kernel)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: [B, N, N] → fp16 and flatten for kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)            # length = B*N*N

        size = left.numel()                       # B * N * N * H
        BLOCK = 1024                               # ≤ 1024 threads / block
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – plain element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Out‑gate (separate linear + sigmoid) – still in fp16
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via strided‑batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # (B,N,N,H)  fp16

    # -----------------------------------------------------------------
    # 9️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 🔟 Gating + final projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 📏 Cast back to float32 (expected by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
