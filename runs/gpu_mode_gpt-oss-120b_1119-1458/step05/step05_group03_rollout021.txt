"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1. Layer‑norm the input (FP32) → cast to FP16.
2. Fuse each projection + its gate in a single matmul by concatenating
   the two weight matrices (shape [2·hidden, dim]).
3. Apply the sigmoid on the gate part, then element‑wise multiply
   proj·gate (and optional mask).  All of these are cheap FP16 kernels.
4. Heavy contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton kernel that directly computes
   `C = A @ Bᵀ` for every head `h`.  The kernel works on the original
   layout `[B,N,N,H]` → no extra transposes or copies are needed.
5. Second Layer‑norm (FP16) → out‑gate (sigmoid) → final linear
   projection back to the original channel dimension.
6. Cast the result back to FP32 for the harness.

Only the forward pass is required; gradients are omitted.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
#   left  : [B, N, N, H]   (row index i, shared dim k)
#   right : [B, N, N, H]   (row index j, shared dim k)
#   out   : [B, N, N, H]   (C[i,j] = Σ_k left[i,k] * right[j,k])
# Memory layout is the default PyTorch one: strides (B, i, k, h) = (N*N*H, N*H, H, 1).
# The kernel loops over tiles of size BLOCK_M×BLOCK_N and BLOCK_K,
# using fp16 math with fp32 accumulation.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # *fp16 global pointers
    B, N, hidden_dim,                           # i32 scalar sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # i32 strides for left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # i32 strides for right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # i32 strides for out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs: (pid_m, pid_n, pid_batch_head)
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)               # block row (i dimension)
    pid_n = tl.program_id(1)               # block col (j dimension)
    pid_batch = tl.program_id(2)           # flattened (batch * hidden)

    # --------------------------------------------------------------
    # Recover batch index b and hidden index h from the flattened ID.
    # --------------------------------------------------------------
    b = pid_batch // hidden_dim
    h = pid_batch % hidden_dim

    # --------------------------------------------------------------
    # Starting offsets for the i‑ and j‑tiles.
    # --------------------------------------------------------------
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Base pointers for this (b, h) slice.
    # --------------------------------------------------------------
    base_l = left_ptr + b * stride_l_b + h * stride_l_h
    base_r = right_ptr + b * stride_r_b + h * stride_r_h
    base_o = out_ptr + b * stride_o_b + h * stride_o_h

    # --------------------------------------------------------------
    # Accumulator in fp32.
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the shared dimension k.
    # --------------------------------------------------------------
    n_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(n_k_blocks):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        mask_a = mask_i[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # Load a tile of rightᵀ: rows = k, cols = j  → shape (BLOCK_K, BLOCK_N)
        b_ptrs = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        mask_b = mask_k[:, None] & mask_j[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate dot product (fp16 -> fp32)
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write result back (cast to fp16).
    # --------------------------------------------------------------
    out_ptrs = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    mask_o = mask_i[:, None] & mask_j[:, None]   # both i and j must be valid
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_o)

# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]

    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]      (float32/bool) – ignored if nomask=True
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).half()   # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate weight matrices (cast to FP16 once)
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # (2*H, dim)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # (2*H, dim)

    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias = weights["to_out_norm.bias"].half()
    to_out_weight = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ Combined linear for left and right (produces proj & gate together)
    # ------------------------------------------------------------------
    left_comb = torch.nn.functional.linear(x, left_comb_weight)   # [B,N,N,2*H] FP16
    right_comb = torch.nn.functional.linear(x, right_comb_weight)

    # ------------------------------------------------------------------
    # 4️⃣ Split & apply sigmoid to gate part
    # ------------------------------------------------------------------
    left_proj = left_comb[..., :hidden_dim]                     # [B,N,N,H]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])      # [B,N,N,H]

    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 5️⃣ Compute out‑gate (also from the normalized input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 6️⃣ Apply gating (and optional mask) → left/right ready for contraction
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate                 # in‑place not needed, new tensor
        right = right_proj * right_gate
    else:
        # mask broadcasted over the hidden dimension
        mask_exp = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp

    # ------------------------------------------------------------------
    # 7️⃣ Heavy contraction via custom Triton kernel
    # ------------------------------------------------------------------
    Bsz, N, _, _ = left.shape
    out = torch.empty_like(left)   # [B,N,N,H] FP16 output buffer

    # Strides (in elements)
    slb, sli, slk, slh = left.stride()
    srb, srj, srk, srh = right.stride()
    sob, soi, soj, soh = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        Bsz * hidden_dim,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        Bsz,
        N,
        hidden_dim,
        slb,
        sli,
        slk,
        slh,
        srb,
        srj,
        srk,
        srh,
        sob,
        soi,
        soj,
        soh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on the contracted result (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Modulate with out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 🔟 Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.float()
```
