"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 → float32 for stability).
2. Cast to fp16 for the bulk of the computation.
3. Fuse the two linear projections (proj + gate) for the left/right branches:
   * For each side concatenate the projection and gate weight matrices,
     perform one half‑precision matmul, split the result and apply the sigmoid
     to the gate part.
   * Apply the optional mask *before* the gate multiplication.
4. Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM C = A @ Bᵀ, where A/B are shaped
   (B·hidden_dim, N, N).  For N ≤ 512 we launch a custom Triton kernel
   (128×128×64 tiles, fp16×fp16 → fp32 accumulation); larger N fall back to
   torch.bmm which internally uses cuBLAS.
5. Apply a second LayerNorm, the output‑gate (sigmoid), and the final linear
   projection back to the original embedding dimension.
6. Cast the result back to float32.

Both the heavy contraction and the fused projection/gate steps are performed
in half‑precision to maximise throughput on H100 while keeping numerical
stability via fp32 accumulation inside the Triton kernel and fp32
LayerNorms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 φ-> fp16, accumulator fp32)
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # *fp16  [M, N, K]
    B_ptr,                     # *fp16  [M, N, K]
    C_ptr,                     # *fp16  [M, N, N]
    M, N, K,                   # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # strides of A (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B (M, j, k)
    stride_cm, stride_ci, stride_cj,   # strides of C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile index over output rows (i)
    pid_n = tl.program_id(1)   # tile index over output columns (j)
    pid_z = tl.program_id(2)   # index over the batch*hidden dimension (M)

    # ---------------------------------------------------------------
    # Offsets for the current tile
    # ---------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over the K‑dimension in tiles
    # ---------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load a block of A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am               # jump to the correct M‑slice
            + offs_i[:, None] * stride_ai     # i‑offset
            + offs_k[None, :] * stride_ak     # k‑offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )

        # Load a block of Bᵀ: shape (BLOCK_K, BLOCK_N)
        # (B stored as [M, N, K]; we need B(k, j))
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm               # jump to the correct M‑slice
            + offs_k[:, None] * stride_bk     # k‑offset (first)
            + offs_j[None, :] * stride_bj     # j‑offset (second)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )

        # Mixed‑precision dot: fp16 × fp16 → fp32
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # Write the result back to C
    # ---------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left/right tensors of shape [B, N, N, H] via a custom Triton GEMM.
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides expressed in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile size – 128×128 works well on H100 for fp16 mma
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over cols (j)
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,          # M
        N,
        N,              # K == N
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Same contraction as above but via torch.bmm (cuBLAS). Operates in fp16
    with fp32 accumulation (torch.bmm default on GPUs).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack + config
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32 → fp32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 3️⃣ cast everything to fp16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ prepare fused projection + gate weight matrices (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]            # (hidden_dim, dim)

    # concatenate proj + gate weights for each side
    left_weights_cat = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)   # (2*H, dim)
    right_weights_cat = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16) # (2*H, dim)

    # ------------------------------------------------------------------
    # 5️⃣ fused linear + gating for the left branch
    # ------------------------------------------------------------------
    # result shape: [B, N, N, 2*hidden_dim]
    left_cat = F.linear(x, left_weights_cat)  # bias=False by construction
    # split projection and gate pre‑sigmoid parts
    left_proj_raw, left_gate_pre = torch.split(left_cat, hidden_dim, dim=-1)

    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] (fp16)
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj_raw = left_proj_raw * mask_fp16

    left_gate = torch.sigmoid(left_gate_pre)        # fp16
    left = left_proj_raw * left_gate                # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ fused linear + gating for the right branch
    # ------------------------------------------------------------------
    right_cat = F.linear(x, right_weights_cat)
    right_proj_raw, right_gate_pre = torch.split(right_cat, hidden_dim, dim=-1)

    if not nomask:
        right_proj_raw = right_proj_raw * mask_fp16

    right_gate = torch.sigmoid(right_gate_pre)
    right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ output gating (independent of the mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w.to(torch.float16)))  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 8️⃣ heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 512:
        contracted = _batched_matmul_triton(left, right)  # fp16
    else:
        contracted = _batched_matmul_torch(left, right)   # fp16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (fp32 accumulation) → out‑gate → final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    contracted = F.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # still fp16, LayerNorm uses fp32 internally for stats

    contracted = contracted * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(contracted, to_out_w)                    # [B, N, N, dim], fp16

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 for the test harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
