@triton.jit
def gated_mask_kernel(x_ptr, gate_pre_ptr, mask_ptr, out_ptr, total_elts, hidden_dim, BLOCK):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = offs < total_elts

    # mask index per hidden
    mask_idx = offs // hidden_dim
    # load
    x = tl.load(x_ptr + offs, mask=mask_valid, other=0.0)
    g = tl.load(gate_pre_ptr + offs, mask=mask_valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    # compute sigmoid(g)
    # Use stable formulation: inv = tl.exp(-g)
    # g is half, but compute with float32? tl.exp works on fp16 returns fp16 approx.
    s = 1.0 / (1.0 + tl.exp(-g))
    result = x * s * m
    tl.store(out_ptr + offs, result, mask=mask_valid)
```

"""
Fast outgoing TriMul forward pass (AlphaFold‑3 style)

Algorithm
---------
1️⃣  FP32 LayerNorm on the input (stable statistics) → cast to FP16.
2️⃣  Fuse all five linear projections (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single FP16 matmul by concatenating the
    weight matrices.  The result is split into the five tensors
    `[B, N, N, hidden_dim]`.
3️⃣  Fuse the element‑wise multiplication of left/right projections with
    their sigmoid gates and (optional) binary mask using a Triton kernel.
    This eliminates the intermediate gate tensors and two element‑wise
    passes.
4️⃣  Compute the heavy O(N³) contraction with a batched half‑precision GEMM:
       out[b, i, j, h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a reshaped `torch.bmm`, which on an H100 maps to a
    Tensor‑Core kernel.
5️⃣  FP16 LayerNorm on the hidden dimension.
6️⃣  Apply the out‑gate (sigmoid) and the final linear projection back
    to the original ``dim``.
7️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the forward pass is implemented – no autograd needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (x * sigmoid(g) * mask)
# Broadcasted mask has shape (B, N, N) and is repeated every
# `hidden_dim` elements of the flattened tensor.
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_kernel(
    x_ptr,          # *fp16: tensor to be updated (left or right)
    g_ptr,          # *fp16: pre‑gate tensor (same shape as x)
    mask_ptr,       # *fp16: mask tensor (no hidden dim)
    total_elts: tl.int32,   # B * N * N * hidden_dim
    hidden_dim: tl.int32,   # size of the hidden dimension
    BLOCK: tl.constexpr,    # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear indices
    mask = offs < total_elts

    # Index in the broadcasted mask (mask repeats every `hidden_dim` entries)
    mask_idx = offs // hidden_dim

    # Load tensors
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(g_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Sigmoid: 1 / (1 + exp(-g))
    sig = 1.0 / (1.0 + tl.exp(-g))

    # Write back the gated & masked result
    tl.store(x_ptr + offs, x * sig * m, mask=mask)


# ----------------------------------------------------------------------
# Batched half‑precision GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (FP16)

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) blocks
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))               # (B*H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(

) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability) → FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # Cast to FP16 for the expensive part
    x = x_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections into a single matmul.
    #   Shapes:
    #     left_proj, left_gate, right_proj, right_gate, out_gate
    #   each : [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    # Concatenate weight matrices (all bias=False in the reference)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*hidden_dim, dim)
    ).to(torch.float16)

    # Combined linear projection
    # Result shape: [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_weight)

    # Split the concatenated tensor back into the five components
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        combined, hidden_dim, dim=-1
    )  # each [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Gating + (optional) mask – fused via Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        # Prepare mask in FP16 (broadcasted later across hidden_dim)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()                     # B * N * N * hidden_dim
        BLOCK = 1024                                        # ≤ 1024 threads per block
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # In‑place fusion: left_proj = left_proj * sigmoid(left_gate_pre) * mask
        gated_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate_pre.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Same for the right side
        gated_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate_pre.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel left_proj/right_proj already contain the gated values.
        left = left_proj
        right = right_proj
        # out_gate is not masked – compute its sigmoid the normal way
        out_gate = torch.sigmoid(out_gate_pre)
    else:

        left_gate = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)
        left = left_proj * left_gate
        right = right_proj * right_gate
        out_gate = torch.sigmoid(out_gate_pre)

    # --------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched half‑precision GEMM
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] (FP16)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # → [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (expected by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
