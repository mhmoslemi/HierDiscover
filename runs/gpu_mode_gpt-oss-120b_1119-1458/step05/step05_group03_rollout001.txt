"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. **Layer‑norm** the input (FP32 → FP16) – a single torch.nn.functional.layer_norm.
2. **Fused projection + gating**:
   * Concatenate the five weight matrices (left_proj, left_gate,
     right_proj, right_gate, out_gate) into one large matrix of shape
     (5*hidden_dim, dim).
   * One half‑precision GEMM (`torch.nn.functional.linear`) produces a
     tensor of shape `[B, N, N, 5*hidden_dim]`.
   * Split it into the five streams.  Apply `sigmoid` to the three
     gating streams.
3. **Mask‑gate fusion** (Triton):
   * If a mask is present the element‑wise product
     `proj * gate * mask` is performed in‑place for `left` and `right`
     by a tiny Triton kernel that broadcasts the mask over the hidden
     dimension.
   * If no mask, a simple `*=` suffices.
4. **Heavy contraction** (`out = Σ_k left[...,k,:] * right[...,k,:]`):
   * Treat each hidden dimension as an independent batch.
   * Reshape `left`/`right` to `(B*hidden_dim, N, N)` and invoke
     `torch.bmm` (cuBLAS Tensor‑Core GEMM) which is the fastest path on
     an H100.
5. **Second layer‑norm** over the hidden dimension (FP16).
6. **Out‑gate** multiplication (FP16) and final linear projection back
   to the original channel dimension (FP16).
7. Cast the result to `float32` for the harness.

Only the mask‑gate fusion is implemented in Triton; all dense linear
operations use cuBLAS which already delivers peak Tensor‑Core throughput.
Memory traffic is reduced dramatically by fusing the five linear layers
into a single GEMM.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # *fp16  : projection tensor (size = total_elems)
    gate_ptr,          # *fp16  : gate tensor (same shape as proj)
    mask_ptr,          # *fp16  : mask vector of shape (B*N*N,)
    total_elems,       # i32    : number of elements = B*N*N*hidden_dim
    hidden_dim,        # i32    : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    """proj = proj * gate * mask   (mask broadcast over hidden_dim)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load projection and gate
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute corresponding mask offset (broadcast over hidden_dim)
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(proj_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the O(N³) contraction.
    left/right : [B, N, N, H] (FP16)
    Returns    : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each head is an independent batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM: (B*H, N, N) @ (B*H, N, N)^T  →  (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]       (float32/bool) – ignored if `nomask` is True
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (default True).

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 -> FP16)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32
    x = x_norm.half()                     # FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear projections (5 * hidden_dim output channels)
    # ------------------------------------------------------------------
    # Gather weight matrices (out_features, in_features) and cast once
    left_proj_w   = weights["left_proj.weight"].half()   # (hidden_dim, dim)
    left_gate_w   = weights["left_gate.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()

    # Concatenate along the out_features dimension
    fused_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Single GEMM: (B*N*N, dim) @ (dim, 5*hidden_dim)ᵀ → (B*N*N, 5*hidden_dim)
    fused = torch.nn.functional.linear(x, fused_weight)  # no bias, FP16
    # Split into the 5 streams (views, no copy)
    split = torch.split(fused, hidden_dim, dim=-1)
    left_proj = split[0]   # [B,N,N,hidden_dim]
    left_gate = split[1]
    right_proj = split[2]
    right_gate = split[3]
    out_gate = split[4]

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid to gating streams (still FP16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 4️⃣ Mask‑gate fusion (Triton) for left/right streams
    # --------------------------------------------------------------
    if not nomask:
        # Prepare mask broadcasted over hidden_dim
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        total_elems = left_proj.numel()          # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Discard gates (they are no longer needed)
        del left_gate, right_gate
    else:
        # No mask → simple in‑place multiplication
        left_proj = left_proj.mul_(left_gate)
        right_proj = right_proj.mul_(right_gate)
        del left_gate, right_gate

    # Re‑use variable names expected by later stages
    left = left_proj   # [B,N,N,hidden_dim]
    right = right_proj

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction via batched GEMM (Tensor‑Core)
    # --------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim]  FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (FP16)

    # Final linear projection (hidden_dim -> dim)
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].half(),
        bias=None,
    )  # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 8️⃣ Cast to float32 (expected by the harness)
    # --------------------------------------------------------------
    return out.float()
