def _batched_matmul_triton(left, right):
    B,N,N_,H = left.shape
    assert N == N_
    A = left.permute(0,3,1,2).contiguous()  # shape (B,H,N,N)
    B_ = right.permute(0,3,1,2).contiguous()
    # view to (B*H,N,N)
    A_flat = A.view(B*H, N, N)
    B_flat = B_.view(B*H, N, N)

    C_flat = torch.empty_like(A_flat)  # allocate fp16

    # strides
    stride_am, stride_ai, stride_ak = A_flat.stride()
    stride_bm, stride_bj, stride_bk = B_flat.stride()
    stride_cm, stride_ci, stride_cj = C_flat.stride()
    # The first stride (M) is N*N because contiguous in memory after flattening.

    # tile sizes based on seq_len
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
        num_stages = 2
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8
        num_stages = 3

    grid = ( (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N, B*H)
    _trimul_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        B*H,
        N,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=num_stages,
    )
    # reshape back
    C = C_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C
```

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

The module consists of two LayerNorms, three linear‐gate
projections, the O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
and a final linear projection.  The heavy contraction is
implemented as a batched GEMM C = A @ Bᵀ using a custom
Triton kernel that works on fp16 data with fp32 accumulation.
All other ops stay in PyTorch but are performed in the smallest
precision that does not affect numerical stability (first /
last LayerNorms stay in fp32).  The kernel adapts its tile size
to the sequence length (64‑tile for N≤256, 128‑tile otherwise)
and therefore runs well on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ (fp16 → fp16, accumulator fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # fp16 *[M, N, K]
    B_ptr,               # fp16 *[M, N, K] (we will read it transposed)
    C_ptr,               # fp16 *[M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides of A (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B (M, j, k)
    stride_cm, stride_ci, stride_cj,   # strides of C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile along the output‑row axis (i)
    pid_n = tl.program_id(1)   # tile along the output‑col axis (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # fp32 accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ------- load A[i,k] ------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ------- load Bᵀ[j,k]  (read B in transposed view) -----------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # mixed‑precision dot: fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write back the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction left @ rightᵀ implemented with the Triton kernel above.
    Input tensors have shape [B, N, N, H] (fp16) and the returned tensor has the
    same shape and dtype.
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in **elements** (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()       # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()      # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()       # (N*N, N, 1)

    # Choose tile size depending on N (better occupancy for larger N)
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
        num_stages = 2
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8
        num_stages = 3

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,                # M
        N,                    # N (rows)
        N,                    # K (cols of A / rows of Bᵀ)
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=num_stages,
    )

    # reshape back to the original [B, N, N, H] layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]     (bool/float, may be all‑ones)
        - weights       : dict of torch.Tensor containing all model weights
        - config        : dict with at least ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – this matches the reference implementation
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ compute all three gates in float32 (high‑precision, cheap)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ fp16 linear projections (bias=False as per definition)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)      # [B,N,N,hidden_dim] fp16
    right = F.linear(x_fp16, right_proj_w)   # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 6️⃣ optional mask (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to fp16 and add a trailing dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ fuse gates  (multiplication is in‑place to reduce memory traffic)
    # ------------------------------------------------------------------
    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 8️⃣ heavy O(N³) contraction via Triton
    # ------------------------------------------------------------------
    contracted_fp16 = _batched_matmul_triton(left, right)   # -> [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (fp32) → out‑gate → final projection
    # ------------------------------------------------------------------
    # Cast to fp32 for a numerically stable LN
    contracted_fp32 = contracted_fp16.to(torch.float32)

    # LayerNorm parameters are stored as float32 already
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    contracted_fp32 = torch.nn.functional.layer_norm(
        contracted_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply the output gate (still in fp32)
    contracted_fp32 = contracted_fp32 * out_gate

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]          # shape (dim, hidden_dim), float32
    out = F.linear(contracted_fp32, to_out_w)    # [B,N,N,dim] float32

    # ------------------------------------------------------------------
    # 🔟 return result (float32)
    # ------------------------------------------------------------------
    return out
```
