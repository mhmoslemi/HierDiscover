    # left_gate:

    wlg = tl.load(wlg_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)
    # right_proj:
    wrp_ptr = right_proj_ptr + k_offs[:, None] * weight_row_stride + col_offs[None, :]
    wrp = tl.load(wrp_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

    wrg = tl.load(wrg_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)
    # out_gate:

    wg = tl.load(wg_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

    left_proj_acc += tl.dot(a, wlp)   # a: (M,K), wlp: (K,N) -> (M,N)
    left_gate_acc += tl.dot(a, wlg)
    right_proj_acc += tl.dot(a, wrp)
    right_gate_acc += tl.dot(a, wrg)
    out_gate_acc += tl.dot(a, wg)  # out_gate is linear projection (no gating inside)

  # Apply sigmoid to gate accumulators and out_gate
  left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate_acc))
  right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate_acc))
  out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate_acc))

  # Apply gating and optional mask
  left_out = left_proj_acc * left_gate_sig
  right_out = right_proj_acc * right_gate_sig
  if HAS_MASK:
      # Load mask per row:
      mask_row_offset = row_offs
      mask_vals = tl.load(mask_ptr + mask_row_offset, mask=valid_rows, other=1.0)
      # Broadcast across hidden_dim columns
      mask_vals = mask_vals[:, None]  # shape (BLOCK_M,1)
      left_out = left_out * mask_vals
      right_out = right_out * mask_vals

  # Store left_out and right_out (if we have out_gate, store)
  # Pointers for left_out and right_out:
  left_out_ptrs = left_out_ptr + row_offs[:, None] * hidden_dim + col_offs[None, :]
  right_out_ptrs = right_out_ptr + row_offs[:, None] * hidden_dim + col_offs[None, :]
  # Store using mask rows and cols
  tl.store(left_out_ptrs, left_out, mask=valid_rows[:, None] & valid_cols[None, :])
  tl.store(right_out_ptrs, right_out, mask=valid_rows[:, None] & valid_cols[None, :])

  # Also store out_gate_sig to out_gate_out_ptr, same pointer offsets (M*hidden_dim)
  out_gate_out_ptrs = out_gate_out_ptr + row_offs[:, None] * hidden_dim + col_offs[None, :]
  tl.store(out_gate_out_ptrs, out_gate_sig, mask=valid_rows[:, None] & valid_cols[None, :])

@triton.jit
def final_linear_fuse_kernel(
    norm_ptr,
    gate_ptr,
    w_ptr,
    out_ptr,
    M, hidden_dim, dim,
    stride_norm_hdim,
    stride_gate_hdim,
    stride_w_hidden,
    BLOCK_M: tl.constexpr, BLOCK_O: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)   # output dim block

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    row_offs = row_start + tl.arange(0, BLOCK_M)
    col_offs = col_start + tl.arange(0, BLOCK_O)

    valid_rows = row_offs < M
    valid_cols = col_offs < dim

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float16)

    # loop over hidden_dim
    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offs = k_start + tl.arange(0, BLOCK_K)
        valid_k = k_offs < hidden_dim

        # load norm slice
        a_ptr = norm_ptr + row_offs[:, None] * stride_norm_hdim + k_offs[None, :]
        a = tl.load(a_ptr, mask=valid_rows[:, None] & valid_k[None, :], other=0.0)

        # load gate slice
        b_ptr = gate_ptr + row_offs[:, None] * stride_gate_hdim + k_offs[None, :]
        b = tl.load(b_ptr, mask=valid_rows[:, None] & valid_k[None, :], other=0.0)

        # elementwise multiplication: gated = a * b
        gated = a * b

        # load weight slice (hidden_dim x dim)
        w_ptr_tile = w_ptr + k_offs[:, None] * stride_w_hidden + col_offs[None, :] # shape (BLOCK_K, BLOCK_O)
        w = tl.load(w_ptr_tile, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

        # acc += dot(gated, w)
        acc += tl.dot(gated, w)

    # store result
    out_ptrs = out_ptr + row_offs[:, None] * dim + col_offs[None, :]
    tl.store(out_ptrs, acc, mask=valid_rows[:, None] & valid_cols[None, :])

```python
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,            # *fp16, shape (M, dim)
    left_proj_ptr,    # *fp16, shape (dim, hidden_dim)
    left_gate_ptr,
    right_proj_ptr,
    right_gate_ptr,
    out_gate_ptr,
    mask_ptr,         # *fp16, shape (M,) - may be unused
    left_out_ptr,     # *fp16, shape (M, hidden_dim)
    right_out_ptr,
    out_gate_out_ptr,
    M, dim, hidden_dim,
    stride_x_row,     # int - dim (row stride)
    stride_weight_row, # int - hidden_dim (as row stride for weight_T)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_N

    row_offs = row_start + tl.arange(0, BLOCK_M)
    col_offs = col_start + tl.arange(0, BLOCK_N)
    valid_rows = row_offs < M
    valid_cols = col_offs < hidden_dim

    # accumulators
    proj_left_acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    gate_left_acc  = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    proj_right_acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    gate_right_acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    out_gate_acc   = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # K-loop
    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offs = k_start + tl.arange(0, BLOCK_K)
        valid_k = k_offs < dim

        # Load A tile (M x K)
        a_ptr = x_ptr + row_offs[:, None] * stride_x_row + k_offs[None, :]
        a = tl.load(a_ptr, mask=valid_rows[:, None] & valid_k[None, :], other=0.0)

        # Load weight tiles (K x N)
        wlp_ptr = left_proj_ptr + k_offs[:, None] * stride_weight_row + col_offs[None, :]
        wlp = tl.load(wlp_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

        wlg_ptr = left_gate_ptr + k_offs[:, None] * stride_weight_row + col_offs[None, :]
        wlg = tl.load(wlg_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

        wrp_ptr = right_proj_ptr + k_offs[:, None] * stride_weight_row + col_offs[None, :]
        wrp = tl.load(wrp_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

        wrg_ptr = right_gate_ptr + k_offs[:, None] * stride_weight_row + col_offs[None, :]
        wrg = tl.load(wrg_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

        wg_ptr = out_gate_ptr + k_offs[:, None] * stride_weight_row + col_offs[None, :]
        wg = tl.load(wg_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

        # Dot products
        proj_left_acc += tl.dot(a, wlp)
        gate_left_acc  += tl.dot(a, wlg)
        proj_right_acc += tl.dot(a, wrp)
        gate_right_acc += tl.dot(a, wrg)
        out_gate_acc   += tl.dot(a, wg)

    # Sigmoid gating
    gate_left_sig = 1.0 / (1.0 + tl.exp(-gate_left_acc))
    gate_right_sig = 1.0 / (1.0 + tl.exp(-gate_right_acc))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate_acc))

    # Apply gating
    left = proj_left_acc * gate_left_sig
    right = proj_right_acc * gate_right_sig

    # Apply mask if present
    if HAS_MASK:
        mask_vals = tl.load(mask_ptr + row_offs, mask=valid_rows, other=1.0)
        mask_vals = mask_vals[:, None]
        left = left * mask_vals
        right = right * mask_vals

    # Store outputs
    left_ptrs = left_out_ptr + row_offs[:, None] * hidden_dim + col_offs[None, :]
    right_ptrs = right_out_ptr + row_offs[:, None] * hidden_dim + col_offs[None, :]
    out_gate_ptrs = out_gate_out_ptr + row_offs[:, None] * hidden_dim + col_offs[None, :]
    tl.store(left_ptrs, left, mask=valid_rows[:, None] & valid_cols[None, :])
    tl.store(right_ptrs, right, mask=valid_rows[:, None] & valid_cols[None, :])
    tl.store(out_gate_ptrs, out_gate_sig, mask=valid_rows[:, None] & valid_cols[None, :])
```

```python
@triton.jit
def final_linear_fuse_kernel(
    norm_ptr,   # *fp16 shape (M, hidden_dim)
    gate_ptr,   # *fp16 shape (M, hidden_dim)
    weight_ptr, # *fp16 shape (hidden_dim, dim) (i.e., transposed weight)
    out_ptr,    # *fp16 shape (M, dim)
    M, hidden_dim, dim,
    stride_norm_row,  # hidden_dim
    stride_gate_row,
    stride_weight_row, # dim
    BLOCK_M: tl.constexpr,
    BLOCK_O: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)
    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    row_offs = row_start + tl.arange(0, BLOCK_M)
    col_offs = col_start + tl.arange(0, BLOCK_O)

    valid_rows = row_offs < M
    valid_cols = col_offs < dim

    acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float16)

    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offs = k_start + tl.arange(0, BLOCK_K)
        valid_k = k_offs < hidden_dim

        # Load norm and gate tiles
        norm_tile_ptr = norm_ptr + row_offs[:, None] * stride_norm_row + k_offs[None, :]
        norm_tile = tl.load(norm_tile_ptr, mask=valid_rows[:, None] & valid_k[None, :], other=0.0)

        gate_tile_ptr = gate_ptr + row_offs[:, None] * stride_gate_row + k_offs[None, :]
        gate_tile = tl.load(gate_tile_ptr, mask=valid_rows[:, None] & valid_k[None, :], other=0.0)

        # Elementwise product
        gated_tile = norm_tile * gate_tile

        # Load weight tile
        weight_tile_ptr = weight_ptr + k_offs[:, None] * stride_weight_row + col_offs[None, :]
        weight_tile = tl.load(weight_tile_ptr, mask=valid_k[:, None] & valid_cols[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(gated_tile, weight_tile)

    # Store result
    out_ptrs = out_ptr + row_offs[:, None] * dim + col_offs[None, :]
    tl.store(out_ptrs, acc, mask=valid_rows[:, None] & valid_cols[None, :])

"""
TriMul (outgoing) forward – high‑performance Triton+PyTorch implementation.

Algorithm
---------
1️⃣  Layer‑norm the input tensor (float32) → cast to fp16.
2️⃣  Fuse *all* linear projections, gating (sigmoid) and optional mask into a single
    Triton kernel that reads the input only once.  The kernel produces:
        left   = (x·W_left_proj)   * sigmoid(x·W_left_gate)   * mask
        right  = (x·W_right_proj)  * sigmoid(x·W_right_gate)  * mask
        out_gate = sigmoid(x·W_out_gate)
3️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is implemented as a batched GEMM (`torch.bmm`) on fp16 tensors.
4️⃣  Apply the second Layer‑norm (fp16) on the contracted result.
5️⃣  Fuse the Element‑wise gate (out_gate) with the final linear projection
    (hidden_dim → dim) into a second Triton kernel:
        out_final = ((out_norm * out_gate) @ W_to_outᵀ)
6️⃣  Reshape and cast back to float32.

All heavy matrix multiplications are performed in fp16 (Tensor‑Core friendly).
The two custom Triton kernels eliminate the four separate linear passes of the
reference implementation, dramatically reducing memory traffic.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: fused projections + gating + optional mask
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    # pointers
    x_ptr,                # *fp16, shape (M, dim)
    left_proj_ptr,        # *fp16, shape (dim, hidden_dim)   = (dim, H)   (transposed)
    left_gate_ptr,        # *fp16, shape (dim, hidden_dim)
    right_proj_ptr,       # *fp16, shape (dim, hidden_dim)
    right_gate_ptr,       # *fp16, shape (dim, hidden_dim)
    out_gate_ptr,         # *fp16, shape (dim, hidden_dim)
    mask_ptr,             # *fp16, shape (M,) – only read if HAS_MASK==1
    left_out_ptr,         # *fp16, shape (M, hidden_dim)
    right_out_ptr,        # *fp16, shape (M, hidden_dim)
    out_gate_out_ptr,     # *fp16, shape (M, hidden_dim)
    # sizes / strides
    M, dim, hidden_dim,
    stride_x_row,         # = dim  (row stride of x)
    stride_w_row,         # = hidden_dim (row stride of the transposed weight matrices)
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)      # row tile
    pid_n = tl.program_id(1)      # hidden‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    row_offs = row_start + tl.arange(0, BLOCK_M)
    col_offs = col_start + tl.arange(0, BLOCK_N)

    valid_rows = row_offs < M
    valid_cols = col_offs < hidden_dim

    # accumulators for the five matmuls
    acc_lproj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    acc_lgate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    acc_rproj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    acc_rgate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    acc_outg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over the dim‑axis (K dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offs = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offs < dim

        # ---- load a tile of X -------------------------------------------------
        a_ptr = x_ptr + row_offs[:, None] * stride_x_row + k_offs[None, :]
        a = tl.load(a_ptr,
                    mask=valid_rows[:, None] & mask_k[None, :],
                    other=0.0)

        # ---- load weight tiles -------------------------------------------------
        # layout of the transposed weights: (dim, hidden_dim)
        # row stride = hidden_dim, column stride = 1
        wp_lproj_ptr = left_proj_ptr + k_offs[:, None] * stride_w_row + col_offs[None, :]
        wp_lgate_ptr = left_gate_ptr + k_offs[:, None] * stride_w_row + col_offs[None, :]
        wp_rproj_ptr = right_proj_ptr + k_offs[:, None] * stride_w_row + col_offs[None, :]
        wp_rgate_ptr = right_gate_ptr + k_offs[:, None] * stride_w_row + col_offs[None, :]
        wp_outg_ptr  = out_gate_ptr   + k_offs[:, None] * stride_w_row + col_offs[None, :]

        wp_lproj = tl.load(wp_lproj_ptr,
                           mask=mask_k[:, None] & valid_cols[None, :],
                           other=0.0)
        wp_lgate = tl.load(wp_lgate_ptr,
                           mask=mask_k[:, None] & valid_cols[None, :],
                           other=0.0)
        wp_rproj = tl.load(wp_rproj_ptr,
                           mask=mask_k[:, None] & valid_cols[None, :],
                           other=0.0)
        wp_rgate = tl.load(wp_rgate_ptr,
                           mask=mask_k[:, None] & valid_cols[None, :],
                           other=0.0)
        wp_outg  = tl.load(wp_outg_ptr,
                           mask=mask_k[:, None] & valid_cols[None, :],
                           other=0.0)

        # ---- accumulate the five matmul results --------------------------------
        acc_lproj += tl.dot(a, wp_lproj)   # (M, K) × (K, H) -> (M, H)
        acc_lgate += tl.dot(a, wp_lgate)
        acc_rproj += tl.dot(a, wp_rproj)
        acc_rgate += tl.dot(a, wp_rgate)
        acc_outg  += tl.dot(a, wp_outg)

    # ------------------------------------------------------------------
    # Apply sigmoid to gate results
    # ------------------------------------------------------------------
    gate_l = 1.0 / (1.0 + tl.exp(-acc_lgate))
    gate_r = 1.0 / (1.0 + tl.exp(-acc_rgate))
    outg   = 1.0 / (1.0 + tl.exp(-acc_outg))

    # Apply gating
    left  = acc_lproj * gate_l
    right = acc_rproj * gate_r

    # Optional mask multiplication (mask is broadcast over the hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + row_offs,
                           mask=valid_rows,
                           other=1.0)                # shape (BLOCK_M,)
        mask_val = mask_val[:, None]                     # broadcast to (BLOCK_M, 1)
        left  = left  * mask_val
        right = right * mask_val

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    out_l_ptr = left_out_ptr  + row_offs[:, None] * hidden_dim + col_offs[None, :]
    out_r_ptr = right_out_ptr + row_offs[:, None] * hidden_dim + col_offs[None, :]
    out_g_ptr = out_gate_out_ptr + row_offs[:, None] * hidden_dim + col_offs[None, :]

    store_mask = valid_rows[:, None] & valid_cols[None, :]

    tl.store(out_l_ptr, left, mask=store_mask)
    tl.store(out_r_ptr, right, mask=store_mask)
    tl.store(out_g_ptr, outg, mask=store_mask)


# ----------------------------------------------------------------------
# Kernel 2: final projection (layer‑norm output * out_gate) @ W_to_outᵀ
# ----------------------------------------------------------------------
@triton.jit
def final_proj_kernel(
    norm_ptr,          # *fp16, shape (M, hidden_dim) – result of second LN
    gate_ptr,          # *fp16, shape (M, hidden_dim) – out_gate from kernel 1
    weight_ptr,        # *fp16, shape (hidden_dim, dim) – transposed to_out weight
    out_ptr,           # *fp16, shape (M, dim)
    # sizes / strides
    M, hidden_dim, dim,
    stride_norm,       # row stride of norm/gate tensors (= hidden_dim)
    stride_gate,
    stride_weight,     # row stride of weight matrix (= dim)
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_O: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # row tile
    pid_o = tl.program_id(1)      # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_o * BLOCK_O

    row_offs = row_start + tl.arange(0, BLOCK_M)
    col_offs = col_start + tl.arange(0, BLOCK_O)

    valid_rows = row_offs < M
    valid_cols = col_offs < dim

    # accumulator for the final matrix product
    acc = tl.zeros((BLOCK_M, BLOCK_O), dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension (K)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offs = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offs < hidden_dim

        # (M, K) tile from the normalized tensor
        n_ptr = norm_ptr + row_offs[:, None] * stride_norm + k_offs[None, :]
        n_tile = tl.load(n_ptr,
                         mask=valid_rows[:, None] & mask_k[None, :],
                         other=0.0)

        # (M, K) tile from the gate tensor
        g_ptr = gate_ptr + row_offs[:, None] * stride_gate + k_offs[None, :]
        g_tile = tl.load(g_ptr,
                         mask=valid_rows[:, None] & mask_k[None, :],
                         other=0.0)

        # element‑wise gated tensor
        gated = n_tile * g_tile          # shape (BLOCK_M, BLOCK_K)

        # weight tile: (K, O), row stride = dim, column stride = 1
        w_ptr = weight_ptr + k_offs[:, None] * stride_weight + col_offs[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & valid_cols[None, :],
                         other=0.0)

        # acc += gated @ w_tile
        acc += tl.dot(gated, w_tile)

    # ------------------------------------------------------------------
    # Store the final result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + row_offs[:, None] * dim + col_offs[None, :]
    store_mask = valid_rows[:, None] & valid_cols[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Helper – batched GEMM (left @ rightᵀ) on fp16 tensors
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a single batched matrix multiplication (torch.bmm).

    Shapes:
        left, right : [B, N, N, H]   (fp16)
    Returns:
        out : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]  (A as i×k, B as j×k)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # BMM with transpose on the second operand
    c = torch.bmm(a, b.transpose(1, 2))          # [B*H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    TriMul (outgoing) forward pass – no autograd.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : dict name → torch.Tensor  (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    M = B * N * N                                          # total number of pairwise positions

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Flatten to (M, dim) for the fused Triton kernel
    # -----------------------------------------------------------------
    x_flat = x.view(M, dim).contiguous()

    # -----------------------------------------------------------------
    # 3️⃣ Prepare transposed weight matrices (dim, hidden_dim) in fp16
    # -----------------------------------------------------------------
    left_proj_T   = weights["left_proj.weight"].t().contiguous().to(torch.float16)
    left_gate_T   = weights["left_gate.weight"].t().contiguous().to(torch.float16)
    right_proj_T  = weights["right_proj.weight"].t().contiguous().to(torch.float16)
    right_gate_T  = weights["right_gate.weight"].t().contiguous().to(torch.float16)
    out_gate_T    = weights["out_gate.weight"].t().contiguous().to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Allocate output buffers for left, right and out_gate (fp16)
    # -----------------------------------------------------------------
    left   = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
    right  = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask (fp16, flattened)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        mask_ptr = mask_fp16.data_ptr()
        HAS_MASK = 1
    else:
        # dummy tensor – never read because HAS_MASK==0
        mask_dummy = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = mask_dummy.data_ptr()
        HAS_MASK = 0

    # -----------------------------------------------------------------
    # 6️⃣ Launch fused projection+gate+mask kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 128
    BLOCK_K = 32

    grid_x = (M + BLOCK_M - 1) // BLOCK_M
    grid_y = (hidden_dim + BLOCK_N - 1) // BLOCK_N
    grid = (grid_x, grid_y)

    fused_proj_gate_kernel[grid](
        x_flat.data_ptr(),
        left_proj_T.data_ptr(),
        left_gate_T.data_ptr(),
        right_proj_T.data_ptr(),
        right_gate_T.data_ptr(),
        out_gate_T.data_ptr(),
        mask_ptr,
        left.data_ptr(),
        right.data_ptr(),
        out_gate.data_ptr(),
        M,
        dim,
        hidden_dim,
        dim,               # stride_x_row
        hidden_dim,        # stride_w_row (row stride of transposed weights)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Reshape back to 4‑D tensors
    # -----------------------------------------------------------------
    left   = left.view(B, N, N, hidden_dim)
    right  = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)   # used later after second LN

    # -----------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # -----------------------------------------------------------------
    contracted = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 9️⃣ Second LayerNorm (fp16) – to_out_norm
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 10️⃣ Final fused projection: (out_norm * out_gate) @ W_to_outᵀ
    # -----------------------------------------------------------------
    # Flatten tensors for the kernel
    out_norm_flat = out_norm.view(M, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(M, hidden_dim).contiguous()

    # Transposed final weight: (hidden_dim, dim)
    to_out_weight_T = weights["to_out.weight"].t().contiguous().to(torch.float16)

    final_out = torch.empty((M, dim), dtype=torch.float16, device=device)

    BLOCK_M2 = 64
    BLOCK_O  = 128
    BLOCK_K2 = 32

    grid_x2 = (M + BLOCK_M2 - 1) // BLOCK_M2
    grid_y2 = (dim + BLOCK_O - 1) // BLOCK_O
    grid2 = (grid_x2, grid_y2)

    final_proj_kernel[grid2](
        out_norm_flat.data_ptr(),
        out_gate_flat.data_ptr(),
        to_out_weight_T.data_ptr(),
        final_out.data_ptr(),
        M,
        hidden_dim,
        dim,
        hidden_dim,    # stride_norm (row stride of out_norm / out_gate)
        hidden_dim,    # stride_gate
        dim,           # stride_weight (row stride of weight_T)
        BLOCK_M=BLOCK_M2,
        BLOCK_O=BLOCK_O,
        BLOCK_K=BLOCK_K2,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 11️⃣ Reshape to [B, N, N, dim] and cast back to float32
    # -----------------------------------------------------------------
    out = final_out.view(B, N, N, dim).to(torch.float32)
    return out
