import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _trimul_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ai, stride_ak,
                          stride_bm, stride_bj, stride_bk,
                          stride_cm, stride_ci, stride_cj,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    # M = B * hidden_dim, N = seq_len, K = seq_len
    pid_m = tl.program_id(0)  # tile in i dimension
    pid_n = tl.program_id(1)  # tile in j dimension
    pid_z = tl.program_id(2)  # batch * hidden

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k-tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # load A[i,k]
        a_ptrs = (A_ptr
                  + pid_z * stride_am
                  + offs_i[:, None] * stride_ai
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & (offs_k[None, :] < K),
                    other=0.0)   # fp16

        # load Bᵀ[j,k] -> B[k,j]
        b_ptrs = (B_ptr
                  + pid_z * stride_bm
                  + offs_k[:, None] * stride_bk    # k dimension first
                  + offs_j[None, :] * stride_bj)   # then j
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_j[None, :],
                    other=0.0)   # fp16

        acc += tl.dot(a, b)

    # store output
    c_ptrs = (C_ptr
              + pid_z * stride_cm
              + offs_i[:, None] * stride_ci
              + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C_{b,i,j,h} = sum_k left_{b,i,k,h} * right_{b,j,k,h}
    using a Triton batched GEMM (A @ Bᵀ) where each hidden channel h is a separate batch.
    Input tensors should be fp16 and have shape [B, N, N, H].
    Returns tensor of same shape and dtype.
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H, each matrix corresponds to a (batch, hidden) pair
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16

    # strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Heuristic block sizes – 64 works well for N up to 256,
    # larger N can benefit from 128×128 tiles
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Fallback GEMM using torch.bmm (cuBLAS)."""
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (AlphaFold‑3).  The implementation fuses the two
    projection Linear layers with their corresponding gating Linear layers, uses
    a fused mask*gate*proj multiplication, and accelerates the O(N³) contraction
    with a custom Triton batched GEMM.  All heavy arithmetic is performed in
    fp16 with fp32 accumulation; the first LayerNorm runs in fp32 for stability,
    and the final result is returned as float32.
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gating for left and right branches
    # ------------------------------------------------------------------
    # combine projection and gate weights (out_features, in_features)
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)    # shape (2*hidden_dim, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # Linear forward – produces [B, N, N, 2*hidden_dim]
    left_combined = F.linear(x, left_combined_w)   # fp16
    right_combined = F.linear(x, right_combined_w) # fp16

    # split projection and gate logits
    left_proj = left_combined[..., :hidden_dim]
    left_gate_logits = left_combined[..., hidden_dim:]
    right_proj = right_combined[..., :hidden_dim]
    right_gate_logits = right_combined[..., hidden_dim:]

    # gates after sigmoid (fp16)
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    # Optional mask – broadcast to hidden_dim
    if not nomask:
        # mask already on same device; ensure dtype matches
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply gates (in‑place multiplication to reduce temporary tensors)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Compute out‑gate (shared across hidden_dim)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – batched GEMM left @ rightᵀ
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16
    else:
        contracted = _batched_matmul_torch(left, right)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # apply out‑gate
    contracted = contracted * out_gate

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(contracted, to_out_w)   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 (required by test harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) forward pass.

The implementation fuses the two projection Linear layers with their
corresponding gating Linear layers, applies the optional mask in‑place,
and accelerates the O(N³) contraction with a custom Triton batched GEMM
(A @ Bᵀ) where each hidden channel is treated as an independent batch.
All heavy arithmetic runs in fp16 (fp32 accumulation) while the first
LayerNorm stays in fp32 for stability.  The final output is cast back to
float32 as required by the test harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ (fp16 × fp16 → fp32 accum)
# A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
# C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM kernel used by TriMul."""
    pid_m = tl.program_id(0)   # output‑row tile index
    pid_n = tl.program_id(1)   # output‑col tile index
    pid_z = tl.program_id(2)   # batch*hidden index

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A[i, k]
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Bᵀ[j, k] → B[k, j]
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # Store C[i, j] (as fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using a Triton batched GEMM.
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # Heuristic tile size – 64×64 for modest N, upscale for larger sequences
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using cuBLAS (torch.bmm). Operates in fp16 with fp32
    accumulation (default for torch.bmm on recent GPUs).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
                         – may be all‑ones if ``nomask`` is True.
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection + gating for left and right branches
    # ------------------------------------------------------------------
    # combine proj‑weight & gate‑weight (out_features, in_features)
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)   # (2*hidden_dim, dim)

    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)   # (2*hidden_dim, dim)

    # Linear forward → [B, N, N, 2*hidden_dim] (fp16)
    left_combined = F.linear(x, left_combined_w)
    right_combined = F.linear(x, right_combined_w)

    # Split projection and gate logits
    left_proj = left_combined[..., :hidden_dim]                     # (B,N,N,H)
    left_gate_logits = left_combined[..., hidden_dim:]              # (B,N,N,H)

    right_proj = right_combined[..., :hidden_dim]
    right_gate_logits = right_combined[..., hidden_dim:]

    # Gating (sigmoid) – fp16
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcasted to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Fuse gate multiplication (in‑place to keep memory pressure low)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate (shared across hidden_dim)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – batched GEMM left @ rightᵀ
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16
    else:
        contracted = _batched_matmul_torch(left, right)   # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    contracted = contracted * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(contracted, to_out_w)   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
