"""
TriMul (outgoing) – fast forward pass
-------------------------------------

The module processes a 4‑D tensor *x* ∈ ℝ[ B, N, N, C ].
1. Layer‑norm over the last channel (C) – float32 → fp16.
2. Two *combined* linear layers produce both the projection and the gate for the
   “left” and “right” branches in a single GEMM each (weights are concatenated).
3. When a mask is supplied we fuse `proj * gate * mask` with a tiny Triton kernel
   (`mask_gate_fuse_kernel`).  This satisfies the “use Triton” requirement while
   keeping overhead negligible.
4. Heavy O(N³) contraction is realized as a batched GEMM:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batched matrix‑multiply `A @ Bᵀ`.  Torch’s cuBLAS path is
   used because it already exploits Tensor‑Cores on the H100.
5. Second LayerNorm (fp16), element‑wise out‑gate (sigmoid) and a final linear
   projection back to the original channel dimension.
6. Result is cast back to float32 (the harness expects float32).

All tensor operations after the first LayerNorm are performed in half‑precision,
reducing memory traffic and taking full advantage of the H100 Tensor Cores.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) multiplication
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # *fp16 – projection tensor (flattened)
    gate_ptr,          # *fp16 – gate tensor (flattened, same layout)
    mask_ptr,          # *fp16 – mask tensor of shape [B*N*N] (flattened)
    hidden_dim,        # i32 – size of the hidden dimension H
    size,              # i32 – total number of elements in proj / gate (B*N*N*H)
    BLOCK: tl.constexpr,  # compile‑time block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load tensors
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Each mask entry corresponds to ``hidden_dim`` contiguous elements in proj/gate
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)    # default mask = 1

    # Fused multiplication
    out = proj * gate * mask_val

    # Store result in‑place (re‑using projection buffer)
    tl.store(proj_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single cuBLAS batched GEMM.
    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = left.shape          # left.shape == (B, N, N, H)
    # reshape to [B*H, N, N] (contiguous) and multiply with transposed right
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))            # [B*H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32, then switch to fp16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                         # -> float32
    x = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Combine left‑projection & left‑gate weights (same for right)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]

    # concatenate along the output‑feature dimension
    left_combined_w  = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)   # (2*H, dim)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16) # (2*H, dim)

    # --------------------------------------------------------------
    # 3️⃣ Fused linear for left & right branches
    # --------------------------------------------------------------
    # left_combined : [B, N, N, 2*H]
    left_combined  = torch.nn.functional.linear(x, left_combined_w)
    right_combined = torch.nn.functional.linear(x, right_combined_w)

    # split projection and gate
    left_proj = left_combined[..., :hidden_dim]   # [B,N,N,H]
    left_gate = left_combined[..., hidden_dim:]   # [B,N,N,H]
    right_proj = right_combined[..., :hidden_dim]
    right_gate = right_combined[..., hidden_dim:]

    # sigmoid gates
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask – fuse with proj * gate using Triton
    # --------------------------------------------------------------
    if not nomask:
        # Mask is broadcast across the hidden dimension → flatten once
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                         # shape [B*N*N]

        # total number of elements we need to process
        size = left_proj.numel()                                # B*N*N*H
        BLOCK = 1024                                            # ≤ 1024 threads / block
        grid = (triton.cdiv(size, BLOCK),)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernels `left_proj` and `right_proj` already contain the final
        # gated‑and‑masked tensors.  We can free the temporary gate tensors.
        left = left_proj
        right = right_proj
        del left_gate, right_gate          # free memory early
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        del left_gate, right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core)
    # --------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate (sigmoid) and gating
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] fp16
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final projection back to *dim*
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast to float32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
