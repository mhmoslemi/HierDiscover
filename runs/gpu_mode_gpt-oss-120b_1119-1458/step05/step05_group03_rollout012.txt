"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. **Input layer‑norm** (FP32) → cast to FP16.
2. **Fused linear projections**  
   All five linear maps (left/right projections, three gates) are
   concatenated into a single weight matrix of shape (5·H, C) and
   evaluated with one `torch.nn.functional.linear` call.  This reduces
   kernel launch overhead and improves weight locality.
3. **Gate & mask fusion** – a tiny Triton kernel fuses the element‑wise
   product `proj * sigmoid(gate) * mask` (mask is broadcast over the
   hidden dimension).  When `nomask=True` we simply multiply by the
   sigmoid without the extra launch.
4. **Heavy contraction** – compute  
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`  
   as a batched GEMM: reshape to `[B·H, N, N]` and call `torch.bmm`,
   which dispatches to cuBLAS Tensor‑Core kernels (the fastest path on
   H100).
5. **Second layer‑norm**, **out‑gate**, and **final projection**.  
   The out‑gate is applied element‑wise after the second norm, and the
   result is linearly projected back to the original channel dimension.
6. Cast the output back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is performed in Triton; all heavy matrix
operations remain in highly‑optimized cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over
# the hidden dimension `hidden_dim`.  Works on a flat view of the
# 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,             # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,       # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load operands
    x   = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask: each hidden_dim values share the same mask entry
    mask_offs = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store result in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
#    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, H] (fp16)
    Returns    : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – treat each head as a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM A @ Bᵀ  (fp16 inputs → fp32 accumulation)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]          # input channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask     = config.get("nomask", True)

    device = input_tensor.device
    dtype  = input_tensor.dtype          # expected torch.float32

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                           # [B, N, N, dim], float32
    x = x_norm.half()           # cast once to FP16 for the rest

    B, N, _, _ = x.shape
    M = B * N * N                # flattened number of (pair) positions

    # --------------------------------------------------------------
    # 2️⃣ Gather all linear weights and cast to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # Concatenate the five weight matrices (no bias for any of them)
    # Shape: (5*hidden_dim, dim)
    weight_concat = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )   # fp16, on same device

    # --------------------------------------------------------------
    # 3️⃣ Fused linear projection for left/right & gates
    # --------------------------------------------------------------
    # Flatten input to (M, dim) and apply one big GEMM
    x_flat = x.view(M, dim)               # [M, dim], fp16
    proj_flat = F.linear(x_flat, weight_concat)   # [M, 5*hidden_dim], fp16
    # Reshape back to 4‑D tensor
    proj = proj_flat.view(B, N, N, 5 * hidden_dim)   # [B,N,N,5*H]

    # Slice the concatenated result (views, not copies)
    left_proj_raw   = proj[..., 0 : hidden_dim]                     # [B,N,N,H]
    right_proj_raw  = proj[..., hidden_dim : 2 * hidden_dim]
    left_gate_raw   = proj[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate_raw  = proj[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate_raw    = proj[..., 4 * hidden_dim : 5 * hidden_dim]

    # --------------------------------------------------------------
    # 4️⃣ Gate activations (sigmoid) – produced as FP16 tensors
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)   # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)    # used later (no mask)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # --------------------------------------------------------------
    if nomask:
        # Simple element‑wise multiplication (no extra launch)
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # Prepare mask in FP16 and flatten to (B*N*N,)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elems = left_proj_raw.numel()          # B*N*N*H
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # Ensure contiguity for the tensors we will modify in‑place
        left  = left_proj_raw.contiguous()
        right = right_proj_raw.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Free temporaries we no longer need
        del left_gate, right_gate, mask_fp16, proj

    # --------------------------------------------------------------
    # 6️⃣ Heavy contraction (batched GEMM) – cuBLAS tensor‑core path
    # --------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H], fp16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # still fp16

    # --------------------------------------------------------------
    # 8️⃣ Apply the output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate                     # element‑wise gate (fp16)
    out = F.linear(out, to_out_w)            # [B,N,N,dim], fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast result back to float32 (expected by the harness)
    # --------------------------------------------------------------
    return out.float()
```
