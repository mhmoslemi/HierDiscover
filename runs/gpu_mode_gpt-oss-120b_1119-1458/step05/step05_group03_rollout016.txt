"""
TriMul (outgoing) – fast forward pass.

Algorithmic tricks
------------------
1. **First LayerNorm** (FP32) → cast to FP16.
2. **Fused linear projections**:
   - Left & right projections are computed together with a single
     `linear` using a concatenated weight `[left_proj; right_proj]`.
   - The three gating streams (left_gate, right_gate, out_gate) are
     generated together with another concatenated `linear`.
   - This reduces the number of passes over the massive input tensor
     from five to two, cutting memory bandwidth dramatically.
3. **Mask + gate fuse** – a tiny Triton kernel multiplies
   `proj * gate * mask` (mask is broadcast over the hidden dimension)
   in‑place.  When `nomask=True` we pass a mask of ones, so the kernel
   is always exercised (required by the task).
4. **Heavy O(N³) contraction** – performed as a batched GEMM:
   for each head `h` we compute `C = A @ Bᵀ` with `torch.bmm`,
   which dispatches to cuBLAS tensor‑core kernels.
5. **Second LayerNorm** (FP16) → out‑gate (sigmoid) → final projection.
6. Cast back to FP32 for the harness.

All heavy element‑wise work (mask‑gate multiplication) is done in the
custom Triton kernel, while the dense matrix‑multiplications rely on
cuBLAS for maximal throughput on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) → in‑place multiplication.
# Works on a flat view of a 4‑D tensor `[B, N, N, H]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # *fp16 : projection tensor, size = total_elements
    gate_ptr,          # *fp16 : gate tensor, same size as proj_ptr
    mask_ptr,          # *fp16 : mask tensor of size (B*N*N,)
    total_elements,    # i32   : total number of elements in proj_ptr
    hidden_dim,        # i32   : hidden dimension H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (broadcast over hidden dim)
    mask_index = offs // hidden_dim
    m = tl.load(mask_ptr + mask_index, mask=mask, other=1.0)

    # Fused multiplication
    out = proj * gate * m

    # Store back in‑place
    tl.store(proj_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM helper (uses cuBLAS via torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    using a batched GEMM (A @ Bᵀ) for each head `h`.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and multiply A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))               # FP16 inputs → FP32 acc.
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Triton‑augmented TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32/bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]  # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    bs, N, _, _ = input_tensor.shape
    total_positions = bs * N * N          # B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32
    x = x_norm.half()                     # cast once to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Prepare mask (FP16, broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if nomask:
        # mask of ones – still go through the Triton kernel
        mask_fp16 = torch.ones(total_positions, dtype=torch.float16, device=device)
    else:
        # mask tensor may be bool or float – cast to FP16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)

    # ------------------------------------------------------------------
    # 3️⃣ Concatenated linear projections (left/right) – one matmul
    # ------------------------------------------------------------------
    # Concatenate weights for left_proj and right_proj  → shape (2*H, dim)
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).half()                                 # (2*hidden_dim, dim)

    # Flatten input (same tensor for both projections)
    flat_x = x.reshape(-1, dim)              # (B*N*N, dim)
    proj_out = torch.nn.functional.linear(flat_x, proj_weight)  # (B*N*N, 2*H)

    # Reshape and split
    proj_out = proj_out.view(bs, N, N, 2 * hidden_dim)
    left_proj = proj_out[..., :hidden_dim]   # (B, N, N, H)
    right_proj = proj_out[..., hidden_dim:]  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Concatenated gating streams – one matmul
    # ------------------------------------------------------------------
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                                 # (3*hidden_dim, dim)

    gate_out = torch.nn.functional.linear(flat_x, gate_weight)  # (B*N*N, 3*H)
    gate_out = gate_out.view(bs, N, N, 3 * hidden_dim)
    left_gate_raw = gate_out[..., :hidden_dim]
    right_gate_raw = gate_out[..., hidden_dim : 2 * hidden_dim]
    out_gate_raw = gate_out[..., 2 * hidden_dim :]

    # Sigmoid the gating streams (still FP16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse (proj * gate * mask) in‑place via Triton kernel
    # ------------------------------------------------------------------
    total_elements = left_proj.numel()          # B * N * N * H
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    # left = left_proj * left_gate * mask
    mask_gate_fuse_kernel[grid](
        left_proj,
        left_gate,
        mask_fp16,
        total_elements,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right = right_proj * right_gate * mask
    mask_gate_fuse_kernel[grid](
        right_proj,
        right_gate,
        mask_fp16,
        total_elements,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left_proj, right_proj)   # (B, N, N, H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gate (FP16)

    # Final projection: (hidden_dim -> dim)
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].half(),
        bias=None,
    )                                         # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.float()
