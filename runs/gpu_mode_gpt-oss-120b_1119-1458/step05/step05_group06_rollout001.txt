"""
Outgoing TriMul (AlphaFold‑3) – fused forward pass.

The heavy O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched GEMM (C = A @ Bᵀ) where each hidden channel h of each batch
is treated as an independent GEMM batch.  A custom Triton kernel performs the batched
matrix multiply in fp16 with an fp32 accumulator.  All other operations (LayerNorm,
Linear projections, gating, optional masking) are carried out with PyTorch and
mixed‑precision (fp16 for the linear work, fp32 for LayerNorms).  The final result
is returned in float32, matching the reference PyTorch implementation.

The kernel is invoked for seq_len ≤ 256 (where Triton is faster); for larger
sequences we fall back to torch.bmm (cuBLAS), which is the fastest on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel for batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K] (M = batch*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides for B (M, j, k)
    stride_cm, stride_ci, stride_cj,   # strides for C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel used by the outgoing TriMul."""
    pid_i = tl.program_id(0)   # tile index over output rows   (i)
    pid_j = tl.program_id(1)   # tile index over output cols   (j)
    pid_z = tl.program_id(2)   # batch*hidden index           (m)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator – essential for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----------- load A(i, k) ---------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am                # batch*hidden offset
            + offs_i[:, None] * stride_ai      # i offset
            + offs_k[None, :] * stride_ak      # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----------- load Bᵀ(j, k) = B(m, j, k) (no transpose) -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm                # batch*hidden offset
            + offs_k[:, None] * stride_bk      # k offset first
            + offs_j[None, :] * stride_bj      # j offset second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ---------- store result ---------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using the Triton kernel.
    left/right: [B, N, N, H] (fp16)
    Returns:      [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [M, N, K] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile configuration – 64×64 is a good default on H100 for fp16 mma
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,  # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,  # j‑tiles
        B * H,                         # number of independent GEMMs
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Fallback contraction using torch.bmm (cuBLAS).  Operates in fp16 with fp32
    accumulation (the default for torch.bmm on recent GPUs).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # A @ Bᵀ  => shape (B*H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))

    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool or float) –
                         may be all‑ones.
        - weights_dict : dict with keys matching the module's weight names.
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as input.
    """
    # -------------------------------------------------------------
    # 1️⃣ unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32 for stability)
    # -------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # -------------------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------------------
    # 4️⃣ cast linear weights to fp16 (biases are None for proj/gates)
    # -------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------------------
    # 5️⃣ linear projections (no bias, fp16)
    # -------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # -------------------------------------------------------------
    # 6️⃣ optional mask (broadcast across hidden)
    # -------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – convert to fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------------------
    # 7️⃣ gating (sigmoid) – fp16
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # fuse gates in‑place to reduce memory traffic
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 8️⃣ heavy contraction (O(N³))
    # -------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16
    else:
        contracted = _batched_matmul_torch(left, right)   # fp16

    # -------------------------------------------------------------
    # 9️⃣ second LayerNorm + out‑gate + final projection
    # -------------------------------------------------------------
    contracted = F.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    contracted = contracted * out_gate
    out = F.linear(contracted, to_out_w)   # [B, N, N, dim] fp16

    # -------------------------------------------------------------
    # 🔟 cast back to float32 for the harness
    # -------------------------------------------------------------
    return out.to(torch.float32)
