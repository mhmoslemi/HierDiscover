"""
Optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑Norm on the raw input (float32) – this stabilises the
    statistics before we cast to fp16.
2️⃣  Cast to fp16 for the heavy part of the computation.
3️⃣  Linear projections (`left_proj`, `right_proj`) and three gating
    projections are performed with fp16 weights.
4️⃣  When a mask is present we fuse the three element‑wise products
       left = left * left_gate * mask
       right = right * right_gate * mask
    in a single Triton kernel (`mask_gate_fuse_kernel`).  This avoids
    multiple memory passes.
5️⃣  The O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is realised as a batched GEMM (`torch.bmm`) after reshaping the
    tensors to [B·H, N, N].  This exploits cuBLAS Tensor‑Core kernels
    on the H100.
6️⃣  Second Layer‑Norm (fp16) over the hidden dimension.
7️⃣  Multiply by the output gate, then a final linear projection back
    to the original dimension.
8️⃣  Cast the result back to float32 (required by the harness).

Only the forward pass is implemented – no autograd is needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel to fuse masking and gating (mask is broadcast over the
# hidden dimension).  `x` and `gate` have shape (B, N, N, H), `mask`
# has shape (B, N, N).  The kernel works on the flattened view.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (left or right)
    gate_ptr,       # *fp16 : gate tensor (same shape as x)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : size of hidden dimension H
    size,           # i32   : total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)    # linear indices for this block
    active = offs < size                         # guard for out‑of‑bounds

    # The mask index is obtained by integer division because the hidden
    # dimension is contiguous in memory (innermost stride).
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)  # mask=1 outside valid range

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the heavy O(N³) contraction using a batched GEMM.
    Both `left` and `right` have shape [B, N, N, H] (fp16).
    Returns a tensor of the same shape.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))        # [B*H, N, N]

    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 (one‑time cost)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (half‑precision)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 5️⃣ Gating tensors (sigmoid) – also fp16
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fused element‑wise multiplication using Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and contiguous.  Mask may be bool or float.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()                # B·N·N·H
        BLOCK = 1024                               # ≤ max threads per block (H100)
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # Fuse: left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse: right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple elementwise multiplication (in‑place to reduce temporaries)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑Core)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over the hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating and projection back to the original dimension
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
