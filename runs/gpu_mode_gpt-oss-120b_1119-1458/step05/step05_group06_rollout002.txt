    x_flat = x_h.view(-1, dim)  # shape (B*N*N, dim)

    # combine projection weights
    # left_proj and right_proj
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0)  # shape (2*hidden_dim, dim)

    proj_out = torch.nn.functional.linear(x_flat, proj_w)  # shape (B*N*N, 2*hidden_dim)
    # slice
    left_proj = proj_out[:, :hidden_dim]
    right_proj = proj_out[:, hidden_dim:]

    # reshape to original shape
    left = left_proj.view(B, N, N, hidden_dim)
    right = right_proj.view(B, N, N, hidden_dim)

    # combine gates weights
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    gate_w = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)   # shape (3*hidden_dim, dim)

    gate_out = torch.nn.functional.linear(x_flat, gate_w)  # shape (B*N*N, 3*hidden_dim)
    # apply sigmoid
    gate_out = torch.sigmoid(gate_out)
    left_gate = gate_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = gate_out[:, hidden_dim:2*hidden_dim].view(B, N, N, hidden_dim)
    out_gate = gate_out[:, 2*hidden_dim:].view(B, N, N, hidden_dim)

    # gating product
    left = left * left_gate
    right = right * right_gate

    # mask
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # contraction
    # flatten left and right to [B*hidden_dim, N, N]
    # First we permute: left -> [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*hidden_dim, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # compute contraction with kernel fallback to torch
    if N <= 256:
        # Use triton kernel
        contracted = _batched_matmul_triton(left_mat, right_mat)  # output [B*H,N,N] half
    else:
        contracted = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H,N,N] half

    # reshape back to [B, N, N, hidden_dim]
    contracted = contracted.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # second layernorm on hidden_dim dimension
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )
    # gating with out_gate
    contracted = contracted * out_gate
    # final projection
    out = torch.nn.functional.linear(contracted, weights["to_out.weight"].half())
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fused linear projections + batched GEMM.

Algorithm
---------
1️⃣ Layer‑norm the input (float32 for numerical stability).
2️⃣ Cast to FP16 and flatten the [B,N,N] dimensions → (B·N·N, dim).
3️⃣ Fuse the two projection linear layers (left/right) into a single
   mat‑mul with a weight matrix of shape (2·hidden_dim, dim).
4️⃣ Fuse the three gate linear layers (left_gate, right_gate, out_gate)
   into a single mat‑mul followed by a sigmoid → shape (3·hidden_dim, dim).
5️⃣ Apply gates and optional mask (all in FP16, in‑place to save memory).
6️⃣ Contract `left` and `right` with a batched matrix‑multiply:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  =  left @ rightᵀ .
   • For seq_len ≤ 256 we launch a custom Triton GEMM kernel (64×64×64 tiles,
     fp16 × fp16 → fp32 accumulation, 8 warps, 2 stages).
   • For larger seq_len we fallback to torch.bmm (cuBLAS) which is optimal on H100.
7️⃣ Second LayerNorm on the hidden dimension (fp16 weights / bias) → out_gate.
8️⃣ Multiply by out_gate, apply the final linear projection (hidden_dim → dim)
   and cast the result back to fp32.

The fusion of linear layers reduces the number of GEMM calls from 5 to 2,
dramatically cutting memory traffic and kernel launch overhead while keeping
the heavy O(N³) contraction fully fused in a high‑performance Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM  (C = A @ Bᵀ)
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (fp16 × fp16 → fp32 accumulation)."""
    pid_m = tl.program_id(0)   # tile along i (output row)
    pid_n = tl.program_id(1)   # tile along j (output col)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator in fp32 for precision
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i,k] slice (M,BLOCK_M,BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load Bᵀ[j,k] slice (M,BLOCK_K,BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` (both [B, N, N, H]) using a
    Triton GEMM.  Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # tile size – 64×64 works well on H100 for fp16 mma
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – may be all‑ones
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ FP16 cast – the rest of the pipeline works in half precision
    # ------------------------------------------------------------------
    x_fp16 = x.half()                         # [B, N, N, dim]  fp16
    B, N, _, _ = x_fp16.shape

    # ------------------------------------------------------------------
    # 4️⃣ flatten to (B·N·N, dim) for batched linear layers
    # ------------------------------------------------------------------
    x_flat = x_fp16.view(-1, dim)               # [B·N·N, dim]

    # ------------------------------------------------------------------
    # 5️⃣ fused projection: left & right (2·hidden_dim → hidden_dim each)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()     # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].half()   # (hidden_dim, dim)
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0)   # (2*hidden_dim, dim)

    proj_out = F.linear(x_flat, proj_w)               # [B·N·N, 2*hidden_dim]
    left = proj_out[:, :hidden_dim].view(B, N, N, hidden_dim)   # (B,N,N,H)
    right = proj_out[:, hidden_dim:].view(B, N, N, hidden_dim) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ fused gates: left_gate, right_gate, out_gate (3·hidden_dim)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    gate_w = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)   # (3*H, dim)

    gate_raw = F.linear(x_flat, gate_w)               # [B·N·N, 3*hidden_dim]
    gate = torch.sigmoid(gate_raw)                    # sigmoid in fp16
    left_gate = gate[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = gate[:, hidden_dim:2 * hidden_dim].view(B, N, N, hidden_dim)
    out_gate = gate[:, 2 * hidden_dim:].view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣ apply gates (in‑place to keep memory low)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ optional mask – broadcast over the hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 9️⃣ heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    if N <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16, shape [B,N,N,H]
    else:
        # fallback to cuBLAS – batched matmul (A @ Bᵀ)
        # reshape to (B*H, N, N) and use torch.bmm
        left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        contracted = torch.bmm(left_mat, right_mat.transpose(1, 2))
        contracted = contracted.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 🔟 second LayerNorm (fp16 weights) → out_gate → final projection
    # ------------------------------------------------------------------
    contracted = F.layer_norm(
        contracted,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )
    contracted = contracted * out_gate

    out = F.linear(contracted, weights["to_out.weight"].half())   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to FP32 for the test harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
