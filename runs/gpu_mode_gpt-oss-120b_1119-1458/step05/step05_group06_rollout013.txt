"""
Outgoing TriMul (AlphaFold‑3) – fused, mixed‑precision implementation.

The kernel follows the reference PyTorch implementation but
* performs the two projection+gate linear layers in a *single* fused
  linear call per side (left/right),
* keeps the heavy O(N³) contraction in fp16 with an FP32 accumulator
  via a small Triton batched‑GEMM kernel,
* runs all LayerNorms and the small linear layers in fp32 for numerical
  stability,
* falls back to torch.bmm for very large sequences where cuBLAS is
  faster than the Triton kernel.

The overall flow is:
    1. Layer‑norm the input (float32 → float32)
    2. Cast to fp16, compute left/right together with their gates
    3. Apply the optional mask
    4. Batched contraction `out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]`
       (implemented as a batched GEMM `A @ Bᵀ`)
    5. Layer‑norm, out‑gate, and final output projection
    6. Return a float32 tensor of shape [B, N, N, dim]
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile along output‑row (i)
    pid_n = tl.program_id(1)   # tile along output‑col (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------- tile offsets -------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----------- load A(i, k) -------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am               # batch*hidden offset
            + offs_i[:, None] * stride_ai     # i offset
            + offs_k[None, :] * stride_ak     # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----------- load Bᵀ(j, k)  (transposed view) -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm               # batch*hidden offset
            + offs_k[:, None] * stride_bk     # k offset first
            + offs_j[None, :] * stride_bj     # j offset second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------- store result -------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` (both [B, N, N, H]) using a
    Triton GEMM. Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # Choose tile sizes based on seq_len (use larger tiles for bigger N)
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 8
    else:  # 257 <= N <= 512  (this path is used only for N up to 512)
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` using cuBLAS (torch.bmm). Operates in fp16
    with fp32 accumulation. Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # compute A @ Bᵀ  (batched GEMM)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 / bool) – may be all‑ones
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – numerical stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the heavy pipeline
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fused linear projections (proj + gate) for left and right sides
    # ------------------------------------------------------------------
    # left side: concat proj and gate weights → shape [2*hidden, dim]
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    left_combined = torch.nn.functional.linear(x_fp16, left_combined_w)  # [B,N,N,2*H]

    left_proj = left_combined[..., :hidden_dim]
    left_gate_pre = left_combined[..., hidden_dim:]          # [B,N,N,H]

    # right side: concat proj and gate weights
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    right_combined = torch.nn.functional.linear(x_fp16, right_combined_w)  # [B,N,N,2*H]

    right_proj = right_combined[..., :hidden_dim]
    right_gate_pre = right_combined[..., hidden_dim:]

    # apply sigmoid to gate pre‑activations
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 5️⃣ optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * mask_fp16 * left_gate
        right = right_proj * mask_fp16 * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout before the contraction
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    # Use Triton for N <= 512, otherwise cuBLAS (torch.bmm)
    if seq_len <= 512:
        contracted = _batched_matmul_triton(left, right)   # fp16
    else:
        contracted = _batched_matmul_torch(left, right)    # fp16

    # ------------------------------------------------------------------
    # 7️⃣ second LayerNorm (float32), out‑gate, final projection
    # ------------------------------------------------------------------
    # Cast to float32 for the second normalization (as in reference)
    contracted_fp32 = contracted.to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"]  # float32
    to_out_norm_bias = weights["to_out_norm.bias"]      # float32
    contracted_norm = torch.nn.functional.layer_norm(
        contracted_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # out‑gate (computed from the same normalized x)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_weight))  # fp16
    out_gate_fp32 = out_gate.to(torch.float32)

    # Apply gating
    gated = contracted_norm * out_gate_fp32

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]  # float32, shape (dim, hidden_dim)
    out = torch.nn.functional.linear(gated, to_out_weight)  # float32, shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 8️⃣ return float32 result
    # ------------------------------------------------------------------
    return out
