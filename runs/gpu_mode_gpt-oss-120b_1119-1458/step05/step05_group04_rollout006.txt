# ----------------------------------------------------------------------
# Triton kernel: fused sigmoid, mask and element‑wise multiplication.
#    proj ← proj * sigmoid(gate_raw) * mask    (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_and_mask_kernel(
    proj_ptr,                 # *fp16  (output & input)
    gate_ptr,                 # *fp16  raw gate pre‑sigmoid
    mask_ptr,                 # *fp16  flattened mask (optional)
    total_elts,               # total number of elements in proj (B·N·N·H)
    hidden_dim,               # H (size of the hidden channel)
    HAS_MASK: tl.constexpr,   # compile‑time flag: 0 = no mask, 1 = mask present
    BLOCK_SIZE: tl.constexpr, # #elements processed per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    # Load projection and raw gate values
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate_raw = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Sigmoid (approximate but identical to torch.sigmoid in fp16)
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    if HAS_MASK:
        # mask is stored as a flat tensor of shape [B·N·N]
        mask_idx = offs // hidden_dim          # integer division
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        out = proj * gate * mask_val
    else:
        out = proj * gate

    tl.store(proj_ptr + offs, out, mask=mask)


def _apply_gate_and_mask_inplace(
    proj: torch.Tensor,
    gate_raw: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """
    In‑place: proj ← proj * sigmoid(gate_raw) * mask (if provided).

    All tensors must be contiguous FP16 and of the same logical shape
    (mask may be a flat [B·N·N] tensor; it will be broadcast across the hidden dim).
    """
    assert proj.is_contiguous() and gate_raw.is_contiguous()
    total = proj.numel()
    BLOCK = 8192  # provides good occupancy on H100
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        # dummy pointer (won’t be read)
        dummy_ptr = proj
        _apply_gate_and_mask_kernel[grid](
            proj, gate_raw, dummy_ptr, total, hidden_dim,
            HAS_MASK=0,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # mask must be a flat FP16 tensor of shape [B·N·N]
        mask_flat = mask.contiguous().view(-1)
        _apply_gate_and_mask_kernel[grid](
            proj, gate_raw, mask_flat, total, hidden_dim,
            HAS_MASK=1,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input tensor [B, N, N, D]   (float32)
        torch.Tensor,                # mask tensor  [B, N, N]    (float32 or bool)
        Dict[str, torch.Tensor],    # weight dict (float32)
        Dict[str, Any],             # config dict with 'dim', 'hidden_dim', optionally 'nomask'
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 for stability) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the last dimension (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)   # [B, N, N, D] in fp16

    # ------------------------------------------------------------------
    # 2️⃣ Prepare concatenated linear weights (proj + gate) and cast to fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # Cast all model weights to fp16 on the target device once
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=torch.float16)
    out_gate_w_fp16 = out_gate_w.to(device=device, dtype=torch.float16)
    to_out_norm_w_fp16 = to_out_norm_w.to(device=device, dtype=torch.float16)
    to_out_norm_b_fp16 = to_out_norm_b.to(device=device, dtype=torch.float16)
    to_out_w_fp16 = to_out_w.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Concatenated linear projections (proj+gate) for left & right
    # ------------------------------------------------------------------
    # left_concat: [B, N, N, 2*H]; first H entries = proj, second H = gate_raw
    left_concat = torch.nn.functional.linear(x, left_comb_w)     # bias=False
    right_concat = torch.nn.functional.linear(x, right_comb_w)   # bias=False

    # Split into projection & raw gate tensors (all are contiguous)
    left_proj_raw = left_concat[..., :hidden_dim]
    left_gate_raw = left_concat[..., hidden_dim:]
    right_proj_raw = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask handling – flatten for fast indexing inside Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16, device=device).contiguous()
        # Flatten to 1‑D for indexing inside the kernel (shape B·N·N)
        mask_flat = mask_fp16.view(-1)
    else:
        mask_fp16 = None
        mask_flat = None  # not used

    # ------------------------------------------------------------------
    # 5️⃣ Fuse sigmoid(gate), optional mask, and projection multiplication
    # ------------------------------------------------------------------
    # In‑place: left_proj_raw = left_proj_raw * sigmoid(left_gate_raw) * mask
    _apply_gate_and_mask_inplace(
        proj=left_proj_raw,
        gate_raw=left_gate_raw,
        mask=mask_flat,
        hidden_dim=hidden_dim,
    )
    # Same for the right side
    _apply_gate_and_mask_inplace(
        proj=right_proj_raw,
        gate_raw=right_gate_raw,
        mask=mask_flat,
        hidden_dim=hidden_dim,
    )
    # At this point `left_proj_raw` and `right_proj_raw` hold the gated tensors

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑core)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj_raw.shape  # hidden_dim already known
    # reshape to [B·H, N, N] for batched matrix multiplication
    left_flat = left_proj_raw.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_proj_raw.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) → out_gate (sigmoid) → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w_fp16,
        bias=to_out_norm_b_fp16,
    )

    # Compute the out‑gate from the *normalized* input `x` (same as in the reference)
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w_fp16)  # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(out_gate_raw)                         # fp16

    out = out * out_gate  # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w_fp16)  # back to original dim

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
