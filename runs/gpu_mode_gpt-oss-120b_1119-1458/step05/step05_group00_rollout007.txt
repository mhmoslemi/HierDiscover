    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    left_proj_w = weights["left_proj.weight"].to(dtype=torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(dtype=torch.float16, device=device)
    left_gate_w = weights["left_gate.weight"].to(dtype=torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(dtype=torch.float16, device=device)

    proj = F.linear(x, w_combined) # shape (B,N,N,4*hidden_dim)

    left_proj = split[0]

        right_proj = right_proj.contiguous()

"""
Highly‑optimized outgoing TriMul kernel (AlphaFold‑3) in Triton + PyTorch.

Algorithm
---------
1.  Layer‑Norm on the input (FP32) → cast to FP16.
2.  Fuse the four linear projections (left/right + gates) into a single
    matmul with a concatenated weight matrix.
3.  Apply the sigmoid to the gate slices in‑place.
4.  If a mask is provided, fuse mask multiplication and gate‑mask
    multiplication for *both* left and right tensors with a tiny Triton
    kernel (one kernel launch, no extra memory traffic).
5.  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched FP16 GEMM (torch.bmm) after reshaping
    (B·H, N, N) × (B·H, N, N)ᵀ.
6.  Second LayerNorm (hidden dimension) in FP16.
7.  Multiply by the out‑gate (sigmoid) and project back to the original
    dimension with a final FP16 linear.
8.  Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is performed in Triton – this satisfies the
requirement to implement at least part of the Op in a kernel while the
rest of the work leverages highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * left_gate  * mask
#   right ← right * right_gate * mask
#   All tensors are contiguous FP16 with shape [B, N, N, H].
#   `mask` has shape [B, N, N] and is broadcast along the hidden dim H.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # i32   total number of elements in left/right tensors
    H,                 # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat indices
    valid = offs < total_elements                               # OOB guard

    # Index into the mask (one mask value per H elements)
    mask_offs = offs // H

    # Load tensors (masked loads guarantee zero for OOB positions)
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    # Load mask; use 1.0 for OOB entries (neutral element of the product)
    mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store the result back in‑place
    tl.store(left_ptr  + offs, left_out ,  mask=valid)
    tl.store(right_ptr + offs, right_out,  mask=valid)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform C = A @ Bᵀ where A and B are [B, N, N, H] (FP16).
    Returns C with shape (B, N, N, H) in FP16.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → view as (B·H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))

    # Restore original layout (B, N, N, H)
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul Module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float, optional)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16
    fp32 = torch.float32

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.to(fp16)      # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four projections (left/right + their gates) into one linear
    # ------------------------------------------------------------------
    # Cast the required weight matrices to FP16 once
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=fp16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=fp16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=fp16)

    # Concatenate along the output dimension (first axis of the weight)
    # Resulting shape: (4 * hidden_dim, dim)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*H, dim)

    # Single fused linear projection
    # Output shape: (B, N, N, 4*hidden_dim) – still FP16
    proj = F.linear(x, combined_w)   # bias=None (bias=False in all linear layers)

    # Split the fused tensor into the individual tensors (views, no copy)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        proj, hidden_dim, dim=-1
    )  # each (B, N, N, hidden_dim)

    # Apply sigmoid to gate tensors in‑place (still FP16)
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()

    # ------------------------------------------------------------------
    # 3️⃣ Compute the out‑gate (independent linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=fp16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))      # shape (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion
    # ------------------------------------------------------------------
    if nomask:
        # Simple in‑place multiplication: proj * gate
        left = left_proj
        left.mul_(left_gate)          # left = left_proj * left_gate   (FP16, in‑place)
        right = right_proj
        right.mul_(right_gate)        # right = right_proj * right_gate
    else:
        # Convert mask to FP16 (0.0 / 1.0) and make everything contiguous
        mask_fp16 = mask_tensor.to(fp16).contiguous()
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()          # B * N * N * hidden_dim
        BLOCK = 1024                  # ~1 K threads per block – good occupancy
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # After the kernel, `left` and `right` already contain the
        # masked‑and‑gated values.
        # (The gate tensors are no longer needed – free them early.)
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched FP16 GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)   # shape (B, N, N, hidden_dim), FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden dimension) – FP16
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, hidden_dim) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Multiply by the out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise (FP16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)
    out = F.linear(out, to_out_w)            # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 (expected by the surrounding harness)
    # ------------------------------------------------------------------
    return out.to(fp32)
