"""
Optimized Triton implementation of the outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1. **Fuse the five per‑position linear layers** (left_proj, left_gate,
   right_proj, right_gate, out_gate) into a single matrix multiplication.
   This reduces the number of times the huge input tensor `[B,N,N,dim]` is
   read from memory and cuts kernel‑launch overhead.

2. **Mask‑gate fusion** – if a mask is supplied we fuse `tensor * gate *
   mask` into one tiny Triton kernel (`fused_gate_mask_kernel`).  The mask
   is broadcast over the hidden dimension, so each element reads the mask
   once.

3. **Heavy O(N³) contraction** – performed as a batched GEMM with
   `torch.bmm`.  Each hidden channel becomes an independent batch element,
   which lets cuBLAS/Tensor‑Core kernels run at near‑peak throughput.

4. **Mixed‑precision** – all heavy arithmetic (linear layers, GEMM,
   final projection) runs in FP16, while the first LayerNorm uses FP32 for
   numerical stability.  The result is cast back to FP32 for the harness.

The function returns a tensor of shape `[B, N, N, dim]` (float32) and can
handle the “nomask” case or a broadcast mask of shape `[B, N, N]`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) – in‑place update.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated (size = total_elts)
    gate_ptr,              # *fp16 : gate tensor (same shape as x)
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x / gate
    hidden_dim: tl.int32,  # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # linear indices
    mask = offs < total_elts

    # mask repeats every `hidden_dim` entries (broadcast over the hidden dim)
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM used for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a batched half‑precision GEMM.

    left, right : [B, N, N, H] (FP16)
    Returns:
        out : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) – each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ  → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=torch.float32)

    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # -> [B, N, N, dim] float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy parts of the computation
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Stack all five linear weight matrices (proj + gate) into one
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    stacked_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape: (5 * hidden_dim, dim)

    # --------------------------------------------------------------
    # 4️⃣ Single fused linear projection → obtain all raw tensors at once
    # --------------------------------------------------------------
    # (B,N,N,dim) @ (dim,5*hidden_dim)ᵀ → (B,N,N,5*hidden_dim)
    combined = torch.nn.functional.linear(x, stacked_w)  # FP16

    H = hidden_dim
    # Slice the concatenated result
    left_raw      = combined[..., 0:H]
    left_gate_raw = combined[..., H:2 * H]
    right_raw     = combined[..., 2 * H:3 * H]
    right_gate_raw = combined[..., 3 * H:4 * H]
    out_gate_raw  = combined[..., 4 * H:5 * H]

    # --------------------------------------------------------------
    # 5️⃣ Apply sigmoids to gate tensors
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # --------------------------------------------------------------
    # 6️⃣ Optional mask – fuse (proj * gate * mask) with a tiny Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_raw.numel()          # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left = left_raw * left_gate * mask
        fused_gate_mask_kernel[grid](
            left_raw.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            H,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_raw * right_gate * mask
        fused_gate_mask_kernel[grid](
            right_raw.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            H,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_raw          # already multiplied by gate & mask
        right = right_raw
    else:
        # No mask – just multiply by the gates
        left = left_raw * left_gate
        right = right_raw * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑Cores)
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] FP16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16) → multiply by the out‑gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to original dim (FP16)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] FP16

    # --------------------------------------------------------------
    # 🔟 Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
