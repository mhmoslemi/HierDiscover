                              stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                              stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                              stride_mask_b, stride_mask_i, stride_mask_j,
                              BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:,None] * stride_mask_i + offs_j[None,:] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:,None] & mask_j[None,:], other=0.0) # fp16
    
    # Loop over hidden dimension tiles

        h_mask = offs_h < H

        proj_ptrs = (proj_ptr + pid_b * stride_proj_b + offs_i[:,None,None] * stride_proj_i +
                     offs_j[None,:,None] * stride_proj_j + offs_h[None,None,:] * stride_proj_h)

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input in FP16 (the PyTorch implementation uses FP32
    accumulation internally).  The result stays in FP16 for the rest
    of the pipeline.

2️⃣  Fuse the two linear projections (proj + gate) for the left and right
    pair‑wise tensors:
        [B, N, N, dim] @ (2·hidden_dim × dim) → [B, N, N, 2·hidden_dim]
    The first half are the raw projections, the second half are the
    pre‑sigmoid gate values.

3️⃣  The optional mask is fused together with the gate sigmoid in a tiny
    Triton kernel:
        proj = proj * sigmoid(gate) * mask      (mask is broadcast over H)

    When there is no mask we simply do
        proj = proj * sigmoid(gate)                     (in‑place).

4️⃣  The heavy O(N³) contraction is performed as a batched GEMM
    (C = A @ Bᵀ) for each hidden channel, using cuBLAS tensor‑core
    kernels:
        left  = left_proj   (shape B,N,N,H)
        right = right_proj  (shape B,N,N,H)
        out   = Σₖ left[...,k,:] * right[...,k,:]ᵀ
    This is realised by reshaping the tensors to [B·H, N, N] and calling
    ``torch.bmm``.

5️⃣  Second LayerNorm (over the hidden dimension) → out‑gate (sigmoid) →
    final linear projection back to model dimension.

6️⃣  Cast the result back to FP32 (the harness expects FP32).

All heavy tensor operations stay in FP16, only the final result is cast
up.  The custom Triton kernel fuses mask, gate‑sigmoid and the
projection, saving two passes over the massive ``[B,N,N,H]`` tensors.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj = proj * sigmoid(gate) * mask   (mask ‑‑> broadcast H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_sigmoid_inplace_kernel(
    proj_ptr,                     # *fp16  [B, N, N, H]
    gate_ptr,                     # *fp16  [B, N, N, H]   (pre‑sigmoid)
    mask_ptr,                     # *fp16  [B, N, N]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    In‑place fusion: proj = proj * sigmoid(gate) * mask.
    The mask is broadcast across the hidden dimension.
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # tile along i (seq‑len)
    pid_j = tl.program_id(2)          # tile along j (seq‑len)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Load a 2‑D mask tile – this is reused for every H‑tile
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=tl.float16(0.0),
    )   # [BLOCK_I, BLOCK_J]

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        h_mask = offs_h < H

        # proj tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & h_mask[None, None, :],
            other=tl.float16(0.0),
        )

        # gate tile (pre‑sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile_raw = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & h_mask[None, None, :],
            other=tl.float16(0.0),
        )
        # sigmoid (1/(1+exp(-x))) – stay in FP16
        gate_tile = 1.0 / (1.0 + tl.exp(-gate_tile_raw))

        # fused multiplication
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # write back in‑place
        tl.store(
            proj_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & h_mask[None, None, :],
        )
    # end for h_tile
# ----------------------------------------------------------------------


def _apply_mask_gate_sigmoid_inplace(proj: torch.Tensor,
                                     gate: torch.Tensor,
                                     mask: torch.Tensor) -> torch.Tensor:
    """
    In‑place ``proj = proj * sigmoid(gate) * mask``.
    All tensors are expected to be fp16 on the same CUDA device.
    Returns the modified ``proj`` tensor (the operation is done in‑place).
    """
    B, N, _, H = proj.shape
    # Triton works with strides expressed in *elements* (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()

    # modest tile sizes – keep register pressure low and stay below 32 KB
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_sigmoid_inplace_kernel[grid](
        proj,
        gate,
        mask,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,          # more warps ⇒ higher occupancy on H100
        num_stages=2,         # double‑buffering of loads
    )
    return proj


def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a strided batched GEMM (C = A @ Bᵀ) with half‑precision.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → reshape to (B·H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # A @ Bᵀ  (B·H independent GEMMs)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP16, weight & bias are also FP16)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float16)

    x = input_tensor.to(torch.float16)                     # [B,N,N,dim] fp16
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2️⃣  Combine left‑proj & left‑gate weights (same for the right side)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2·H, dim)

    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2·H, dim)

    # -----------------------------------------------------------------
    # 3️⃣  Linear projections (single GEMM for each side)
    # -----------------------------------------------------------------
    left_combined = torch.nn.functional.linear(x, left_combined_w)   # [B,N,N,2·H]
    right_combined = torch.nn.functional.linear(x, right_combined_w)

    # split into proj / gate (pre‑sigmoid) halves
    left_proj_raw = left_combined[..., :hidden_dim]      # [B,N,N,H]
    left_gate_raw = left_combined[..., hidden_dim:]      # [B,N,N,H]

    right_proj_raw = right_combined[..., :hidden_dim]
    right_gate_raw = right_combined[..., hidden_dim:]

    # -----------------------------------------------------------------
    # 4️⃣  out‑gate (linear + sigmoid) – needed after the second LayerNorm
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_raw)                     # fp16

    # -----------------------------------------------------------------
    # 5️⃣  Mask + Gate fusion (in‑place) – only if a mask is supplied
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – cast to fp16 and keep [B,N,N]
        mask_fp16 = mask_tensor.to(torch.float16)
        # left side
        left_proj_raw = _apply_mask_gate_sigmoid_inplace(left_proj_raw,
                                                         left_gate_raw,
                                                         mask_fp16)
        # right side
        right_proj_raw = _apply_mask_gate_sigmoid_inplace(right_proj_raw,
                                                          right_gate_raw,
                                                          mask_fp16)
    else:
        # No mask → just sigmoid‑gate the projections (element‑wise)
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left_proj_raw = left_proj_raw * left_gate
        right_proj_raw = right_proj_raw * right_gate

    # -----------------------------------------------------------------
    # 6️⃣  Heavy O(N³) contraction via batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_proj_raw, right_proj_raw)   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 7️⃣  Second LayerNorm (over hidden_dim) – FP16 weights/bias
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=to_out_norm_w,
                                         bias=to_out_norm_b)

    # -----------------------------------------------------------------
    # 8️⃣  Multiply by the out‑gate and final linear projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣  Cast back to FP32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
