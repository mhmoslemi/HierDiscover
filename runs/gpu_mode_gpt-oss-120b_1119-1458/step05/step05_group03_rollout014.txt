Implementation of fused kernel:

```
@triton.jit
def fuse_proj_gate_mask_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    total_elements, hidden_dim,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # compute mask index (global element index / hidden_dim)
    mask_index = offs // hidden_dim
    m = tl.load(mask_ptr + mask_index, mask=mask, other=1.0)
    left_out = left * left_gate * m
    right_out = right * right_gate * m
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)
```

def mask_gate_fuse_left_right_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    numel: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    # compute mask index
    mask_idx = offs // hidden_dim
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    right_out = right * right_gate * m
    tl.store(left_ptr + offs, left_out, mask=mask)

```python
"""
TriMul (outgoing) forward pass – Triton‑accelerated variant.
Algorithm
---------
1. First LayerNorm (float32) → cast to FP16.
2. All four projections (left_proj, left_gate, right_proj, right_gate) are fused
   into a *single* linear layer with a weight matrix of shape (4*hidden_dim, dim).
   This halves the number of memory reads of the large input tensor.
3. Sigmoid gates are applied to the three gating streams.
4. If a mask is present we fuse the three element‑wise multiplications
   `proj = proj * gate * mask` for **both** left and right tensors with one
   Triton kernel (`mask_gate_fuse_two_tensor_kernel`). No extra memory
   allocation is needed.
5. Heavy O(N³) contraction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   is performed as a batched GEMM using `torch.bmm`.  The tensor is first
   reshaped to `[B*hidden_dim, N, N]` and the matrix multiplication
   `A @ Bᵀ` is dispatched to cuBLAS Tensor‑Core kernels (the fastest path on
   an H100).
6. Second LayerNorm on the hidden dimension, followed by the out‑gate
   (sigmoid) and the final linear projection to the original dimension.
   These steps stay in FP16 and the result is cast back to FP32.

Only the forward pass is required.  The kernel `mask_gate_fuse_two_tensor_kernel`
is the custom Triton kernel used.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) for left and right tensors.
# The mask is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_two_tensor_kernel(
    proj_a_ptr,    # *fp16    : left projection tensor
    gate_a_ptr,    # *fp16    : left gate tensor
    proj_b_ptr,    # *fp16    : right projection tensor
    gate_b_ptr,    # *fp16    : right gate tensor
    mask_ptr,      # *fp16    : mask tensor of shape (B*N*N,)
    numel: tl.int32,   # total number of elements in a projection tensor
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < numel

    # Load data
    proj_a = tl.load(proj_a_ptr + offs, mask=valid, other=0.0)
    gate_a = tl.load(gate_a_ptr + offs, mask=valid, other=0.0)
    proj_b = tl.load(proj_b_ptr + offs, mask=valid, other=0.0)
    gate_b = tl.load(gate_b_ptr + offs, mask=valid, other=0.0)

    # Broadcast mask (index repeats every hidden_dim entries)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fused operation
    proj_a = proj_a * gate_a * m
    proj_b = proj_b * gate_b * m

    # Store back in‑place
    tl.store(proj_a_ptr + offs, proj_a, mask=valid)
    tl.store(proj_b_ptr + offs, proj_b, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched matrix multiplication using torch.bmm (leverages cuBLAS/Tensor‑Core).
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a batched GEMM.
    Both left and right have shape [B, N, N, H] (FP16).

    Returns:
        Tensor of shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where the inner dimension K = N
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM: A @ Bᵀ
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main public entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool). Ignored when `nomask` is True.
        - weights_dict  : mapping name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``;
                          optional key ``nomask`` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32) → cast to FP16 for the rest
    # ------------------------------------------------------------------
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] FP32
    x = x_norm_fp32.half()  # FP16

    # ------------------------------------------------------------------
    # 2️⃣  Prepare weight matrices (FP16).  Fuse the four linear layers
    #     (left_proj, left_gate, right_proj, right_gate) into a *single*
    #     GEMM with weight shape (4*hidden_dim, dim).
    # ------------------------------------------------------------------
    # Cast all weights to FP16 once
    left_proj_w = weights["left_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w = weights["to_out.weight"].half()

    # Stack the four projection weights vertically: order = Lproj, Lgate, Rproj, Rgate
    stacked_weights = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape (4*hidden_dim, dim)
    # Linear projection: x (B,N,N,dim) @ stacked_weights.T -> (B,N,N,4*hidden_dim)
    proj_concat = torch.nn.functional.linear(x, stacked_weights)  # FP16

    # Split concatenated tensor
    # Note: the hidden_dim dimension is the last one.
    total_hidden = 4 * hidden_dim
    # sanity check
    assert proj_concat.shape[-1] == total_hidden
    # Use view slicing:
    left_proj = proj_concat[..., 0:hidden_dim]
    left_gate = proj_concat[..., hidden_dim:2 * hidden_dim]
    right_proj = proj_concat[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = proj_concat[..., 3 * hidden_dim:4 * hidden_dim]

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 3️⃣  Mask + gate fusion (if mask is present)
    # ------------------------------------------------------------------
    if nomask:
        # Simple in‑place multiplication – no extra memory traffic
        left = left_proj.mul_(left_gate)
        right = right_proj.mul_(right_gate)
    else:
        # Ensure mask is in half‑precision and flattened (B*N*N,)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elements = left_proj.numel()  # B * N * N * hidden_dim
        # Triton kernel launch parameters
        BLOCK = 1024  # Tunable – adjust according to GPU
        grid = ((total_elements + BLOCK - 1) // BLOCK,)
        # In‑place fused update for both left and right tensors
        mask_gate_fuse_two_tensor_kernel[grid](
            left_proj,
            left_gate,
            right_proj,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Keep the fused tensors as the final left/right values
        left = left_proj
        right = right_proj
        # Gates are no longer needed – free memory
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 4️⃣  Heavy contraction – batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    out_hidden = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    # ------------------------------------------------------------------
    # 6️⃣  Out‑gate (sigmoid) and final projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden_dim] FP16
    # Element‑wise gating
    gated = out_norm * out_gate  # FP16

    # Final linear projection back to original channel dimension
    # Using torch.nn.functional.linear (FP16 input, FP16 weight)
    out = torch.nn.functional.linear(gated, to_out_w)  # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (the harness expects float32)
    # ------------------------------------------------------------------
    return out.float()
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated variant.
================================================================
The operator works on a 4‑D pairwise tensor [B, N, N, C] and
computes a heavy O(N³) contraction.  The implementation follows the
reference PyTorch code but aggressively fuses operations with Triton
to reduce memory traffic and launch overhead.

Algorithmic steps
-----------------
1️⃣  Layer‑norm the input (float32) and cast to FP16.
2️⃣  Fuse the four linear layers (left_proj, left_gate,
    right_proj, right_gate) into a *single* GEMM:
    the weight matrix has shape (4·hidden_dim, dim) and the output
    is split into the four required streams.  This halves the number
    of passes over the large input tensor.
3️⃣  Apply the sigmoid to the two gate streams.
4️⃣  If a mask is present we fuse the three element‑wise multiplications
    `proj = proj * gate * mask` for **both** left and right tensors with
    one Triton kernel (`mask_gate_fuse_two_tensor_kernel`).  The mask,
    originally [B, N, N], is broadcast over the hidden dimension inside
    the kernel, so only a single launch per tensor is required.
5️⃣  The O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM using `torch.bmm`, which on an
    H100 maps to cuBLAS Tensor‑Core kernels (the most efficient path).
6️⃣  Second Layer‑norm on the hidden dimension (still FP16).
7️⃣  Compute the out‑gate (sigmoid), multiply it element‑wise, and
    apply the final linear projection back to the original channel
    dimension.
8️⃣  Cast the result back to float32 for the harness.

Only the mask‑gate fusion kernel is written in Triton; all other
operations use PyTorch but stay in half‑precision where possible to
maximise throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) for the left and right streams.
# The mask is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_two_tensor_kernel(
    proj_a_ptr,    # *fp16 : left projection tensor  (B,N,N,hidden_dim)
    gate_a_ptr,    # *fp16 : left gate tensor
    proj_b_ptr,    # *fp16 : right projection tensor
    gate_b_ptr,    # *fp16 : right gate tensor
    mask_ptr,      # *fp16 : mask tensor of shape (B*N*N,)
    numel: tl.int32,      # total number of elements in a projection tensor
    hidden_dim: tl.int32, # width of the hidden dimension
    BLOCK: tl.constexpr,  # block size (launch‑time constant)
):
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < numel

    # Load the four operand tensors
    proj_a = tl.load(proj_a_ptr + offs, mask=valid, other=0.0)
    gate_a = tl.load(gate_a_ptr + offs, mask=valid, other=0.0)
    proj_b = tl.load(proj_b_ptr + offs, mask=valid, other=0.0)
    gate_b = tl.load(gate_b_ptr + offs, mask=valid, other=0.0)

    # The mask repeats every `hidden_dim` entries
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fuse the element‑wise multiplications
    proj_a = proj_a * gate_a * m
    proj_b = proj_b * gate_b * m

    # Store the results back in‑place
    tl.store(proj_a_ptr + offs, proj_a, mask=valid)
    tl.store(proj_b_ptr + offs, proj_b, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM using torch.bmm (leverages cuBLAS/Tensor‑Core).
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a
    batched matrix multiplication.  ``left`` and ``right`` have shape
    [B, N, N, H] and are FP16.  The result also has shape [B, N, N, H] FP16.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where each batch element is a standard GEMM
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))  # A @ Bᵀ
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional key ``nomask`` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32
    x = x_norm_fp32.half()  # cast once to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four projection/gate linear layers:
    #    (left_proj, left_gate, right_proj, right_gate) → one GEMM
    # ------------------------------------------------------------------
    # Cast all weights to FP16
    left_proj_w   = weights["left_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # Stack the four weight matrices vertically: Lproj | Lgate | Rproj | Rgate
    stacked_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape (4*hidden_dim, dim)

    # Single linear projection → (B, N, N, 4*hidden_dim)
    proj_concat = torch.nn.functional.linear(x, stacked_w)  # FP16

    # Split the concatenated tensor into four streams
    # Order: left_proj, left_gate, right_proj, right_gate
    left_proj = proj_concat[..., 0:hidden_dim]
    left_gate = torch.sigmoid(proj_concat[..., hidden_dim:2 * hidden_dim])
    right_proj = proj_concat[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = torch.sigmoid(proj_concat[..., 3 * hidden_dim:4 * hidden_dim])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – fuse (proj * gate * mask) for both tensors
    # ------------------------------------------------------------------
    if nomask:
        # In‑place multiplication when no mask is required
        left = left_proj.mul_(left_gate)
        right = right_proj.mul_(right_gate)
    else:
        # Prepare mask in FP16 and flatten it (B*N*N,)
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elems = left_proj.numel()          # B * N * N * hidden_dim
        BLOCK = 1024                               # launch‑time constant
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # In‑place fused update for left and right streams
        mask_gate_fuse_two_tensor_kernel[grid](
            left_proj,                # proj_a_ptr
            left_gate,                # gate_a_ptr
            right_proj,               # proj_b_ptr
            right_gate,               # gate_b_ptr
            mask_fp16,                # mask_ptr
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
        # Gates are no longer needed after fusion
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via a batched GEMM
    # ------------------------------------------------------------------
    out_hidden = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm on the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate (sigmoid) and final linear projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # FP16
    gated = out_norm * out_gate   # element‑wise gating, FP16
    out = torch.nn.functional.linear(gated, to_out_w)  # FP16, shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
