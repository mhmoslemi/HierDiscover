"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32 → FP16 for the heavy part).
2️⃣ Four linear projections (left/right) and three gating linear layers.
3️⃣ If a mask is present, fuse mask‑multiplication with the two gates in a
   single Triton element‑wise kernel (in‑place update of `left` and `right`).
   When `nomask` is True we simply multiply the gates.
4️⃣ Outgoing triangular multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = A @ Bᵀ with A,B ∈
   ℝ^{(B·H)×N×N}; we reuse Torch’s `torch.bmm` on FP16 tensors which
   invokes Tensor‑core‑accelerated GEMM on H100.
5️⃣ Second LayerNorm (FP16) over the hidden dimension.
6️⃣ Apply the final sigmoid gate (`out_gate`) and a linear projection back to
   the original `dim`.
7️⃣ Cast the result back to float32 (the harness expects FP32).

Only the mask‑gate fusion is performed with Triton – this satisfies the
requirement of using a custom kernel while leaving the O(N³) contraction to
the highly‑optimized cuBLAS implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * left_gate  * mask
#   right ← right * right_gate * mask
# Tensors are contiguous FP16 with shape [B, N, N, H] and mask is [B, N, N].
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in‑place)
    right_ptr,         # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcast over H)
    total_elements,    # i32   total number of elements = B*N*N*H
    H,                 # i32   hidden dimension (size of last axis)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat indices
    mask = offs < total_elements                                 # OOB guard

    # Compute mask index (broadcast across hidden dim)
    mask_offs = offs // H

    # Load operands (masked loads guarantee zero for OOB elements)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask value; for OOB we use neutral value 1.0 so it does not affect product
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C = left @ rightᵀ for tensors of shape [B,N,N,H] (FP16).
    The operation is performed as a batched GEMM on (B·H) matrices of size N×N.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B·H, N, N)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ
    C = torch.bmm(A, B_.transpose(1, 2))
    # Restore original layout (B, N, N, H)
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16 for subsequent ops)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim] float32

    # Cast to half‑precision for the heavy part
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 on the correct device
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections → [B, N, N, H]  (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Gating projections (FP16) → [B, N, N, H]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # used after second LN

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple in‑place gating
        left.mul_(left_gate)
        right.mul_(right_gate)
    else:
        # Ensure tensors are contiguous for the kernel
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()

        # Convert mask to FP16 (0.0 / 1.0) and ensure contiguity
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()

        total = left.numel()                         # B·N·N·H
        BLOCK = 1024                                 # elements processed per program instance
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates are fused into left/right; they can be released
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Outgoing triangular multiplication (batched GEMM)
    # ------------------------------------------------------------------
    # out = Σ_k left[..., i, k, :] * right[..., j, k, :]  (FP16)
    out = _batched_matmul_fp16(left, right)   # shape (B, N, N, hidden_dim), FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B, N, N, hidden_dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Apply final gate and project back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                   # element‑wise gating (FP16)
    out = F.linear(out, to_out_w)          # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 as required by the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
