    norm_b = weights["norm.bias"]

        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x_norm.to(torch.float16)

    # Load and cast weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    combined_weight = torch.cat(

    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    combined = torch.nn.functional.linear(x, combined_weight)  # FP16

    H = hidden_dim
    left_proj = combined[..., :H]        # (B,N,N,H)
    right_proj = combined[..., H:2*H]
    left_gate_raw = combined[..., 2*H:3*H]
    right_gate_raw = combined[..., 3*H:4*H]
    out_gate_raw = combined[..., 4*H:5*H]  # raw out_gate (pre-sigmoid)

    # 3. Fuse gating, mask and out_gate sigmoid
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()  # B*N*N*H
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # allocate output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)
        out_gate = torch.empty_like(out_gate_raw)  # to store sigmoid

        fused_gates_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate_raw.data_ptr(),
            right_proj.data_ptr(),
            right_gate_raw.data_ptr(),
            out_gate_raw.data_ptr(),
            mask_fp16.data_ptr(),
            left.data_ptr(),
            right.data_ptr(),
            out_gate.data_ptr(),
            total_elts,
            H,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # no mask: simple elementwise
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj * left_gate
        right = right_proj * right_gate
        out_gate = torch.sigmoid(out_gate_raw)

    # 4. Heavy O(N³) contraction via batched GEMM
    out = _batched_matmul_torch(left, right)

    # 5. Second LayerNorm (FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 6. Multiply by out_gate and final linear projection
    out = out * out_gate   # elementwise FP16
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to FP32 for output
    return out.to(torch.float32)

"""
Highly‑optimized "outgoing" TriMul forward pass (AlphaFold‑3).

Key ideas
---------
1️⃣  First LayerNorm in FP32 (numerical stability) → cast to FP16.
2️⃣  Fuse the five linear projections (left/right + three gates) into **one**
    matrix‑multiply by concatenating the weight matrices.
3️⃣  Fuse the three sigmoids, the optional binary mask and the output‑gate
    sigmoid into a single Triton kernel.  The kernel reads the four projection
    tensors and the mask once and writes the gated tensors (left, right) and the
    sigmoid‑ed out‑gate.
4️⃣  The heavy O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM (`C = A @ Bᵀ`) using a stride‑batched
    `torch.bmm`.  This leverages H100 Tensor‑Cores without extra memory copies.
5️⃣  Second LayerNorm (FP16) → multiply by the out‑gate → final linear projection.
6️⃣  The final result is cast back to FP32 as required by the harness.

Only the forward pass is implemented; no autograd bookkeeping is needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right projections, gates, optional mask,
# and compute the sigmoid for the out‑gate.
# ----------------------------------------------------------------------
@triton.jit
def fused_gates_mask_kernel(
    left_proj_ptr,       # *fp16 : left projection (B,N,N,H)
    left_gate_ptr,       # *fp16 : left‑gate pre‑sigmoid (B,N,N,H)
    right_proj_ptr,      # *fp16 : right projection (B,N,N,H)
    right_gate_ptr,      # *fp16 : right‑gate pre‑sigmoid (B,N,N,H)
    out_gate_ptr,        # *fp16 : out‑gate pre‑sigmoid (B,N,N,H)
    mask_ptr,            # *fp16 : mask (B,N,N) – broadcast over H
    left_out_ptr,        # *fp16 : gated left output (B,N,N,H)
    right_out_ptr,       # *fp16 : gated right output (B,N,N,H)
    out_gate_out_ptr,    # *fp16 : sigmoid(out_gate) (B,N,N,H)
    total_elts: tl.int32,    # B*N*N*H
    hidden_dim: tl.int32,    # H
    BLOCK: tl.constexpr,     # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element id
    mask = offs < total_elts

    # ------------------------------------------------------------------
    # Load data (masked loads return 0 for invalid entries)
    # ------------------------------------------------------------------
    lp = tl.load(left_proj_ptr + offs,   mask=mask, other=0.0)
    lg_raw = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs,  mask=mask, other=0.0)
    rg_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    og_raw = tl.load(out_gate_ptr + offs, mask=mask, other=0.0)

    # ------------------------------------------------------------------
    # Sigmoid – fully in fp16
    # ------------------------------------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg_raw))
    rg = 1.0 / (1.0 + tl.exp(-rg_raw))
    og = 1.0 / (1.0 + tl.exp(-og_raw))

    # ------------------------------------------------------------------
    # Broadcast mask (mask repeats every hidden_dim elements)
    # ------------------------------------------------------------------
    mask_idx = offs // hidden_dim               # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # ------------------------------------------------------------------
    # Apply mask & gates
    # ------------------------------------------------------------------
    left_out  = lp * lg * m
    right_out = rp * rg * m
    out_gate_out = og

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
    tl.store(out_gate_out_ptr + offs, out_gate_out, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM via torch.bmm (cuBLAS / Tensor‑Core)
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right: [B, N, N, H] (fp16)
    Returns   : [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel becomes a batch element
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) – ignored if config["nomask"]==True
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → cast to fp16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # -> [B, N, N, dim] float32
    x = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse all five linear layers into a single matmul
    # --------------------------------------------------------------
    # gather & cast weights (out_features = hidden_dim, in_features = dim)
    w_left   = weights["left_proj.weight"]   # (H, dim)
    w_right  = weights["right_proj.weight"]  # (H, dim)
    w_lgate  = weights["left_gate.weight"]   # (H, dim)
    w_rgate  = weights["right_gate.weight"]  # (H, dim)
    w_ogate  = weights["out_gate.weight"]    # (H, dim)

    # concatenated weight: (5*H, dim)
    combined_weight = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_ogate], dim=0
    ).to(torch.float16)

    # Linear projection (single GEMM)
    combined = torch.nn.functional.linear(x, combined_weight)  # [B,N,N,5*H] fp16

    H = hidden_dim
    left_proj     = combined[..., :H]           # (B,N,N,H)
    right_proj    = combined[..., H:2*H]        # (B,N,N,H)
    left_gate_raw = combined[..., 2*H:3*H]      # (B,N,N,H)
    right_gate_raw= combined[..., 3*H:4*H]      # (B,N,N,H)
    out_gate_raw  = combined[..., 4*H:5*H]      # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Fuse gates, optional mask, and out‑gate sigmoid
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → fp16 broadcast across hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left_proj.numel()               # B * N * N * H
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # allocate destination tensors
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)
        out_gate = torch.empty_like(out_gate_raw)

        fused_gates_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate_raw.data_ptr(),
            right_proj.data_ptr(),
            right_gate_raw.data_ptr(),
            out_gate_raw.data_ptr(),
            mask_fp16.data_ptr(),
            left.data_ptr(),
            right.data_ptr(),
            out_gate.data_ptr(),
            total_elts,
            H,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # no mask → simple element‑wise ops (fp16)
        left_gate  = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left  = left_proj * left_gate
        right = right_proj * right_gate
        out_gate = torch.sigmoid(out_gate_raw)

    # --------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction (batched GEMM)
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16)
    # --------------------------------------------------------------

        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (fp16)
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # → [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
