def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # layer norm first
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B,N,N,dim], float32

    # cast to half for heavy compute
    x = x.to(torch.float16)

    # combine all projection + gate weights (including out_gate)
    # Order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape [5*hidden_dim, dim]

    # linear projection for all
    fused = torch.nn.functional.linear(x, combined_weight)  # [B,N,N,5*hidden_dim]
    # reshape to separate
    B, N, _, _ = fused.shape
    fused = fused.view(B, N, N, 5, hidden_dim)

    # slices
    left_proj = fused[..., 0, :]  # [B,N,N,hidden_dim]
    left_gate_raw = fused[..., 1, :]
    right_proj = fused[..., 2, :]
    right_gate_raw = fused[..., 3, :]
    out_gate_raw = fused[..., 4, :]  # for gating after second norm

    # allocate output tensors for left and right (in-place use proj memory)
    left = left_proj
    right = right_proj

    # Apply gate (sigmoid) and optional mask
    total_elts = left.numel()  # B * N * N * hidden_dim
    BLOCK = 1024
    grid = ( (total_elts + BLOCK - 1) // BLOCK, )
    if not nomask:
        # mask must be fp16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        # left
        gated_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate_raw.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right
        gated_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate_raw.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # no mask
        gated_kernel[grid](
            left_proj.data_ptr(),
            left_gate_raw.data_ptr(),
            total_elts,
            BLOCK=BLOCK,
            num_warps=4,
        )
        gated_kernel[grid](
            right_proj.data_ptr(),
            right_gate_raw.data_ptr(),
            total_elts,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj

    # finalize out_gate (sigmoid) – elementwise multiplication after norm later
    out_gate = torch.sigmoid(out_gate_raw)  # fp16

    # allocate output placeholder for contraction
    out = torch.empty_like(left)  # [B,N,N,hidden_dim]

    # Strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()  # note: out shape same as left

    # Compute batched matrix multiplication using Triton
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid_matmul = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    batched_matmul_kernel[grid_matmul](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N, N, N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # At this point out = contraction result

    # Second layer norm over hidden_dim
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Apply out gate
    out = out * out_gate

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim], fp16

    return out.to(torch.float32)

"""
Highly‑optimized “outgoing” TriMul forward pass (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Layer‑Norm the input (float32 → FP16 for speed).
2️⃣  Fuse **all** linear projections (left‑proj, left‑gate, right‑proj,
    right‑gate, out‑gate) into a *single* `torch.nn.functional.linear`
    call by concatenating the weight matrices.  This reduces the number
    of heavy GEMM passes from five to **one**.
3️⃣  Slice the concatenated result into the four projection tensors and
    the out‑gate tensor.
4️⃣  Apply the sigmoid‑gate (and optional mask) **in‑place** with a tiny
    Triton kernel: `proj = proj * sigmoid(gate) * mask`.  This fuses three
    element‑wise ops into a single kernel launch.
5️⃣  Perform the O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a *batched* GEMM using a custom Triton kernel that works directly
    on the original memory layout (no costly permute‑/contiguous copies).
    The kernel loops over K in tiles and uses `tl.dot` (Tensor‑Core
    friendly) with FP16 → FP32 accumulation.
6️⃣  Second Layer‑Norm (FP16) → element‑wise multiply by the out‑gate.
7️⃣  Final linear projection back to the original feature dimension.
8️⃣  Cast the result to FP32 (required by the evaluation harness).

Only the forward pass is implemented – no autograd support is needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1: apply sigmoid‑gate (and optional mask) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_kernel(
    proj_ptr,          # *fp16 : projection tensor, shape = total_elts
    gate_ptr,          # *fp16 : raw gate tensor (same shape as proj)
    mask_ptr,          # *fp16 : mask tensor, shape = total_elts // hidden_dim
    total_elts: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # load projection and gate
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # broadcast mask (mask repeats every hidden_dim entries)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # sigmoid = 1 / (1 + exp(-x))
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # store the gated & masked result back into proj_ptr (in‑place)
    tl.store(proj_ptr + offs, proj * gate_sig * m, mask=mask)


@triton.jit
def gated_kernel(
    proj_ptr,          # *fp16 : projection tensor, shape = total_elts
    gate_ptr,          # *fp16 : raw gate tensor (same shape as proj)
    total_elts: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    gate_sig = 1.0 / (1.0 + tl.exp(-gate))
    tl.store(proj_ptr + offs, proj * gate_sig, mask=mask)

# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the O(N³) contraction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # *fp16 pointers
    M, N, K,                           # int dimensions (M=N and K=N here)
    stride_Ab, stride_Ai, stride_Ak,   # strides for A (batch, i, k)
    stride_Bb, stride_Bi, stride_Bk,   # strides for B (batch, j, k)
    stride_Cb, stride_Ci, stride_Cj,   # strides for C (batch, i, j)
    hidden_dim,                        # hidden dimension (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # row block i
    pid_n = tl.program_id(1)      # col block j
    pid_batch = tl.program_id(2)  # combined batch*hidden index

    # derive original batch (b) and hidden slice (h)
    b = pid_batch // hidden_dim
    h = pid_batch - b * hidden_dim

    # base pointers for this (b,h)
    A_base = A_ptr + b * stride_Ab + h          # stride over hidden dim = 1
    B_base = B_ptr + b * stride_Bb + h
    C_base = C_ptr + b * stride_Cb + h

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (M × K)
        a_ptrs = A_base + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block: we need Bᵀ (K × N) → read B[j,k] with j from N, k from K
        b_ptrs = B_base + offs_n[None, :] * stride_Bi + offs_k[:, None] * stride_Bk
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store C block (M × N) – hidden index already baked into the pointer
    c_ptrs = C_base + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy ops
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse all linear projections + gates into a single Linear
    # --------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # shape [5*H, dim]

    fused = torch.nn.functional.linear(x, combined_weight)   # [B,N,N,5*H]
    # reshape to separate the five parts
    B, N, _, _ = fused.shape
    fused = fused.view(B, N, N, 5, hidden_dim)              # [B,N,N,5,H]

    left_proj   = fused[..., 0, :]   # [B,N,N,H]
    left_gate   = fused[..., 1, :]   # raw gate, before sigmoid
    right_proj  = fused[..., 2, :]   # [B,N,N,H]
    right_gate  = fused[..., 3, :]   # raw gate
    out_gate_raw = fused[..., 4, :]  # raw out‑gate (sigmoid later)

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gate (and optional mask) in‑place via Triton
    # --------------------------------------------------------------
    total_elts = left_proj.numel()            # B*N*N*H
    BLOCK = 1024
    grid = ((total_elts + BLOCK - 1) // BLOCK,)

    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # left
        gated_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right
        gated_mask_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # no mask → only sigmoid‑gate
        gated_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            total_elts,
            BLOCK=BLOCK,
            num_warps=4,
        )
        gated_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            total_elts,
            BLOCK=BLOCK,
            num_warps=4,
        )

    # After the kernels `left_proj` and `right_proj` contain the
    # already‑gated tensors.
    left  = left_proj
    right = right_proj

    # --------------------------------------------------------------
    # 5️⃣ Compute out_gate (sigmoid) – used after the second LN
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)    # [B,N,N,H] (FP16)

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via a custom Triton batched GEMM
    # --------------------------------------------------------------
    # Allocate output tensor
    out = torch.empty_like(left)   # shape [B,N,N,H], FP16

    # Strides for left/right/out tensors (batch, i, k, h) layout
    stride_a_batch, stride_a_i, stride_a_k, _ = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, _ = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, _ = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid_matmul = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    batched_matmul_kernel[grid_matmul](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N, N, N,                                      # M, N, K
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) → out_gate gating → final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # -> [B,N,N,dim] FP16

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
