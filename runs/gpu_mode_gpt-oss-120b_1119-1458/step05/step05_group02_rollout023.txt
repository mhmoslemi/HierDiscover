@triton.jit
def left_right_gate_kernel(
    left_proj_ptr,
    left_gate_raw_ptr,
    right_proj_ptr,
    right_gate_raw_ptr,
    mask_ptr,
    hidden_dim,
    total_elements,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements
    mask_offs = offs // hidden_dim  # index into mask

    left_proj = tl.load(left_proj_ptr + offs, mask=active, other=0.0)
    left_gate_raw = tl.load(left_gate_raw_ptr + offs, mask=active, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    right_gate_raw = tl.load(right_gate_raw_ptr + offs, mask=active, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # compute sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    left_out = left_proj * mask_val * left_gate
    right_out = right_proj * mask_val * right_gate

    tl.store(left_proj_ptr + offs, left_out, mask=active)
    tl.store(right_proj_ptr + offs, right_out, mask=active)
```

"""
Implementation of the outgoing TriMul operator using Triton and fused linear layers.

Algorithm:
1. LayerNorm on the 4‑D input (float32) → cast to fp16.
2. Fuse the five linear transformations (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single matmul.
3. In‑place fuse mask, sigmoid gating and element‑wise multiplication for the
   left and right branches using a Triton kernel.  If `nomask` the mask part is
   omitted.
4. Compute the O(N³) contraction with a batched GEMM (torch.bmm) on tensors
   reshaped to [B·H, N, N].
5. Layer‑norm on the hidden dimension, gate with `out_gate`, and final linear
   projection back to the original channel dimension.
All heavy operations are performed in FP16; the final result is cast to FP32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for mask + sigmoid gating for both left and right projections
@triton.jit
def mask_gate_fuse2_kernel(
    left_proj_ptr,   # *fp16
    left_gate_ptr,   # *fp16
    right_proj_ptr,  # *fp16
    right_gate_ptr,  # *fp16
    mask_ptr,        # *fp16, broadcast over hidden_dim
    hidden_dim,      # i32
    total_elements, # i64
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements
    # mask is indexed by element // hidden_dim (broadcast)
    mask_offs = offs // hidden_dim

    # Load
    lp = tl.load(left_proj_ptr + offs, mask=active, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # Sigmoid
    lgate = 1.0 / (1.0 + tl.exp(-lg))
    rgate = 1.0 / (1.0 + tl.exp(-rg))

    # Apply mask and gating
    lout = lp * m * lgate
    rout = rp * m * rgate

    # Store results back in place (overwrites proj tensors)
    tl.store(left_proj_ptr + offs, lout, mask=active)
    tl.store(right_proj_ptr + offs, rout, mask=active)


# Triton kernel for sigmoid gating without mask (nomask case)
@triton.jit
def gate_fuse_kernel(
    proj_ptr,        # *fp16
    gate_ptr,        # *fp16
    hidden_dim,      # i32
    total_elements, # i64
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    p = tl.load(proj_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)

    gate = 1.0 / (1.0 + tl.exp(-g))
    out = p * gate

    tl.store(proj_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Args:
        left, right: tensors of shape [B, N, N, H], dtype fp16.

    Returns:
        out: tensor of same shape, dtype fp16.
    """
    B, N, _, H = left.shape
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool) or None
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1️⃣ First LayerNorm (float32) – stable on the raw input.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # Cast to fp16 for the bulk of the computation.
    x = x.to(torch.float16)

    # 2️⃣ Fuse the five linear layers (left_proj, left_gate, right_proj,
    #     right_gate, out_gate) into a single matmul.
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Stack weights along the output‑feature axis.
    combined_weight = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)  # shape [5 * hidden_dim, dim]

    # Single large linear projection.
    combined = torch.nn.functional.linear(x, combined_weight)  # [B,N,N,5*hidden]
    # Ensure contiguous layout for slicing.
    combined = combined.contiguous()

    # Slice the combined result.
    left_proj   = combined[..., :hidden_dim]                     # [B,N,N,hidden]
    left_gate_r = combined[..., hidden_dim:2*hidden_dim]        # raw gate (pre‑sigmoid)
    right_proj  = combined[..., 2*hidden_dim:3*hidden_dim]
    right_gate_r= combined[..., 3*hidden_dim:4*hidden_dim]
    out_gate_r  = combined[..., 4*hidden_dim:]                  # raw out‑gate

    # 3️⃣ Apply mask + sigmoid gating (fused) or only sigmoid gating.
    if not nomask:
        # Convert mask to fp16 and make it contiguous.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total = left_proj.numel()
        BLOCK = 1024  # ≤ 1024 threads per block for good occupancy.
        grid = ((total + BLOCK - 1) // BLOCK,)

        # Fuse mask, sigmoid and element‑wise multiplication for both sides.
        mask_gate_fuse2_kernel[grid](
            left_proj,
            left_gate_r,
            right_proj,
            right_gate_r,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Compute out_gate (no mask needed) – use torch.sigmoid on the raw tensor.
        out_gate = torch.sigmoid(out_gate_r)
    else:
        # No mask – only sigmoid gating for left/right.
        total = left_proj.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        gate_fuse_kernel[grid](
            left_proj,
            left_gate_r,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        gate_fuse_kernel[grid](
            right_proj,
            right_gate_r,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        out_gate = torch.sigmoid(out_gate_r)

    # At this point:
    #   left_proj   = left   (masked & gated)
    #   right_proj  = right  (masked & gated)
    left = left_proj
    right = right_proj

    # 4️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑Core).
    out = _batched_matmul_torch(left, right)   # [B,N,N,hidden] fp16

    # 5️⃣ Second LayerNorm (fp16) over the hidden dimension.
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 6️⃣ Final gating with out_gate and final linear projection.
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # Cast back to float32 (required by the harness) and return.
    return out.to(torch.float32)

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1️⃣ Fuse the five linear layers (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single matmul – the input tensor is read
   only once.

2️⃣ Fuse mask multiplication, sigmoid gating and element‑wise multiplication
   for the left/right branches in a single Triton kernel (two kernels for the
   masked / nomask cases). This removes an extra pass over the large tensors.

3️⃣ The heavy O(N³) contraction is performed with a batched GEMM
   (torch.bmm) on tensors reshaped to [B·H, N, N]; this fully exploits
   Tensor‑Core performance on H100.

4️⃣ The second LayerNorm, out‑gate multiplication and final projection are
   kept in FP16 for speed, and the result is cast back to FP32 at the end.

Only the forward pass is implemented – no autograd is required.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse2_kernel(
    left_proj_ptr,   # *fp16 left projection (output in‑place)
    left_gate_ptr,   # *fp16 raw left gate (pre‑sigmoid)
    right_proj_ptr,  # *fp16 right projection (output in‑place)
    right_gate_ptr,  # *fp16 raw right gate (pre‑sigmoid)
    mask_ptr,        # *fp16 mask broadcast over the hidden dim
    hidden_dim,      # i32
    total_elements,  # i64  = B·N·N·hidden_dim
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements
    # index into mask = element index // hidden_dim  (broadcast)
    mask_offs = offs // hidden_dim

    # Load tensors
    lp = tl.load(left_proj_ptr + offs, mask=active, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    m  = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # Sigmoid (1/(1+exp(-x))) – uses the fast `tl.exp`
    lgate = 1.0 / (1.0 + tl.exp(-lg))
    rgate = 1.0 / (1.0 + tl.exp(-rg))

    # Apply mask and gating
    lout = lp * m * lgate
    rout = rp * m * rgate

    # Store back in‑place (overwrite the projection tensors)
    tl.store(left_proj_ptr + offs, lout, mask=active)
    tl.store(right_proj_ptr + offs, rout, mask=active)


@triton.jit
def gate_fuse_kernel(
    proj_ptr,       # *fp16 projection tensor (output in‑place)
    gate_ptr,       # *fp16 raw gate (pre‑sigmoid)
    hidden_dim,     # i32   (unused, kept for signature symmetry)
    total_elements, # i64
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    p = tl.load(proj_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)

    gate = 1.0 / (1.0 + tl.exp(-g))
    out = p * gate

    tl.store(proj_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Helper for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Args:
        left, right: tensors of shape [B, N, N, H] (fp16)

    Returns:
        out: tensor of same shape (fp16)
    """
    B, N, _, H = left.shape
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # 0️⃣ Unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ Initial LayerNorm (float32) – keeps numerical stability.
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline.
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single matmul.
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape [5*H, dim]

    # Single (large) linear transformation on the normalized input.
    combined = torch.nn.functional.linear(x, combined_weight)   # [B,N,N,5*H]
    combined = combined.contiguous()  # make sure slicing yields contiguous views

    # -----------------------------------------------------------------
    # 4️⃣ Slice the fused result.
    # -----------------------------------------------------------------
    left_proj   = combined[..., :hidden_dim]
    left_gate_r = combined[..., hidden_dim:2 * hidden_dim]
    right_proj  = combined[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate_r= combined[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate_r  = combined[..., 4 * hidden_dim:]  # raw out‑gate (pre‑sigmoid)

    # -----------------------------------------------------------------
    # 5️⃣ Apply mask + sigmoid gating (fused) or just sigmoid gating.
    # -----------------------------------------------------------------
    total = left_proj.numel()               # B·N·N·H
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    if not nomask:
        # mask is expected to be broadcastable over the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_gate_fuse2_kernel[grid](
            left_proj,
            left_gate_r,
            right_proj,
            right_gate_r,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        out_gate = torch.sigmoid(out_gate_r)
    else:
        # No mask – only sigmoid gating.
        gate_fuse_kernel[grid](
            left_proj,
            left_gate_r,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        gate_fuse_kernel[grid](
            right_proj,
            right_gate_r,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        out_gate = torch.sigmoid(out_gate_r)

    left  = left_proj      # now already masked & gated
    right = right_proj

    # -----------------------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM.
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B,N,N,hidden] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) over the hidden dimension.
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication & final projection.
    # -----------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
