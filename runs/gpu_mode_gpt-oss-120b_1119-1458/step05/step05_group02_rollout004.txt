"""
TriMul (outgoing) – fast forward kernel.

Algorithm
---------
1.  Layer‑Norm the input tensor (float32) → keep computation in float32 for
    numerical stability.
2.  Cast the normalized tensor to FP16.
3.  Fuse the five linear projections (left‑proj, right‑proj, left‑gate,
    right‑gate, out‑gate) into a *single* GEMM by concatenating the weight
    matrices.  This reduces kernel launch overhead and memory traffic.
4.  Slice the concatenated result into the five tensors and apply the
    sigmoid to the three gates.
5.  If a mask is provided, a small Triton kernel fuses
       X = X * G * mask
    for both the left and right side, eliminating three separate element‑wise
    launches.  When no mask is present we fall back to a plain PyTorch mul.
6.  Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is implemented as a batched GEMM (torch.bmm) on reshaped tensors
    [B·H, N, N] → uses cuBLAS Tensor‑Core kernels on H100.
7.  Second Layer‑Norm over the hidden dimension (FP16) → out‑gate (sigmoid) →
    element‑wise gating.
8.  Final linear projection back to the original `dim` and cast to float32.

All operations are performed without autograd – only the forward pass is
implemented.  The only Triton kernel used is the mask‑gate fusion (step 5);
the remaining compute relies on highly‑optimised cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel – fuse element‑wise mask, gate and input multiplication.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  : pointer to tensor to be updated (left/right)
    gate_ptr,       # *fp16  : pointer to gate tensor (same shape as x)
    mask_ptr,       # *fp16  : mask tensor (flattened B·N·N)
    hidden_dim,     # i32    : size of hidden dimension (H)
    size,           # i32    : total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time block size (≤1024)
):
    """
    In‑place computation:
        x = x * gate * mask
    where mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    active = offs < size                               # guard against overflow

    # The mask index is offset // hidden_dim because H is the innermost stride.
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)

    tl.store(x_ptr + offs, x * g * m, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using a batched GEMM.
    left / right : [B, N, N, H] (FP16)
    Returns out  : [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # Rearrange to [B·H, N, N] – each hidden channel becomes an independent batch.
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ
    c = torch.bmm(a, b.transpose(1, 2))
    # Restore original layout.
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass – “outgoing” TriMul.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32 / bool) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor : output [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)   # True -> ignore mask

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the work
    # -----------------------------------------------------------------
    x = x.to(dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single GEMM
    #    (left_proj, right_proj, left_gate, right_gate, out_gate)
    # -----------------------------------------------------------------
    # Cast weight tensors to FP16 once.
    left_proj_w   = weights["left_proj.weight"].to(dtype=torch.float16, device=device)
    right_proj_w  = weights["right_proj.weight"].to(dtype=torch.float16, device=device)
    left_gate_w   = weights["left_gate.weight"].to(dtype=torch.float16, device=device)
    right_gate_w  = weights["right_gate.weight"].to(dtype=torch.float16, device=device)
    out_gate_w    = weights["out_gate.weight"].to(dtype=torch.float16, device=device)

    # Concatenate along the output‑feature axis (0)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )   # shape (5*hidden_dim, dim), FP16

    # Single linear operation → shape [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, combined_weight)

    # -----------------------------------------------------------------
    # 4️⃣ Slice the concatenated tensor and apply sigmoid to gates
    # -----------------------------------------------------------------
    left, right, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each: [B, N, N, hidden_dim]

    # Ensure tensors are contiguous before any in‑place kernel work.
    left      = left.contiguous()
    right     = right.contiguous()
    left_gate = torch.sigmoid(left_gate.contiguous())
    right_gate= torch.sigmoid(right_gate.contiguous())
    out_gate  = torch.sigmoid(out_gate.contiguous())

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fused element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; broadcast over hidden dimension.
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device).contiguous()
        total_elements = left.numel()                     # B·N·N·H
        BLOCK = 1024                                      # ≤ 1024 threads per block
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=8,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – simple elementwise gating.
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (batched GEMM)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) over the hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype=torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias" ].to(dtype=torch.float16, device=device)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # still FP16

    # -----------------------------------------------------------------
    # 8️⃣ Gating with out_gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(dtype=torch.float32)
