"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass using
  • FP16 arithmetic for the heavy path
  • ONE fused Triton kernel that combines mask, projection and gate
  • cuBLAS batched GEMM (torch.matmul) with no intermediate copies.

The O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is expressed as a batch of matrix multiplications
    left_permuted[b,h,i,k] @ right_permuted[b,h,k,j]^T
via torch.matmul, which on H100 executes a full‑tensor‑core GEMM
(~0.1 TFLOP for N=1024, hidden=128) in a few hundred microseconds.

Only the mask‑gate fusion is implemented in Triton; all linear layers,
layer‑norms and the final projection run in native PyTorch kernels.
The result is returned in float32 as required by the harness.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel:  proj * gate * mask   (mask is broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,          # *fp16  [B, N, N, H]
    gate_ptr,          # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,          # *fp16  [B, N, N]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # tile in I dimension
    pid_j = tl.program_id(2)          # tile in J dimension

    # ---- I,J coordinates of the current tile --------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---- Load a (BLOCK_I × BLOCK_J) mask tile --------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16

    # ---- Loop over the hidden‑dimension tiles ---------------------------------
    n_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(n_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj[b,i,j,h]
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # gate[b,i,j,h]   (gate already passed through sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Fusion: proj * gate * mask    (mask broadcast over H)
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H]  (fp16, contiguous)
    mask       : [B, N, N]    (fp16)
    Returns out = proj * gate * mask   (mask broadcast over H)   as fp16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides are expressed in elements, not bytes
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # tile sizes – chosen to fit comfortably in ~32 KB shared memory per SM
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 16

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out

# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Args
    ----
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor   : [B, N, N, dim]  (float32, on CUDA)
        mask_tensor    : [B, N, N]       (float32/bool) – optional
        weights_dict   : model parameters (float32)
        config_dict    : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (default True).

    Returns
    -------
    torch.Tensor  : [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → FP16 for the heavy path
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x.to(torch.float16)  # FP16 for everything that follows

    # -----------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (one‑time cost)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # -----------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) – also FP16
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 5️⃣ Mask‑gate fusion (if required)
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool – cast to fp16 for the kernel
        mask_fp16 = mask_tensor.to(dtype=torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)   # fp16, contiguous
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ O(N³) contraction: batched GEMM via torch.matmul (no …contiguous copy)
    #     left  : [B, N, N, H]
    #     right : [B, N, N, H]
    #   after permute → [B, H, N, N]
    #   out = left @ rightᵀ → shape [B, H, N, N]
    # -----------------------------------------------------------------
    # (permute only changes strides – no memory moves)
    left_perm = left.permute(0, 3, 1, 2)      # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)    # [B, H, N, N]

    # (B, H, N, N) @ (B, H, N, N)ᵀ  →  (B, H, N, N)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # Back to original layout (still a view, no copy)
    out = out.permute(0, 2, 3, 1)            # [B, N, N, H]

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Output‑gate and final linear projection back to model dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  fp16

    # -----------------------------------------------------------------
    # 9️⃣ Return FP32 tensor (the evaluation harness expects float32)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
