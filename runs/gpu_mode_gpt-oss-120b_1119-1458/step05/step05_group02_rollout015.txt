@triton.jit

def trmul_kernel(

    A_ptr, B_ptr, C_ptr,

    B, N, H,

    stride_A_batch, stride_A_i, stride_A_k,

    stride_B_batch, stride_B_i, stride_B_k,

    stride_C_batch, stride_C_i, stride_C_k,

    BLOCK_M: tl.constexpr,

    BLOCK_N: tl.constexpr,

    BLOCK_K: tl.constexpr

):

    # __program_id axis 0: batch_head idx

    bh = tl.program_id(0)

    batch = bh // H

    head = bh % H

    # compute block indices for M (i) and N (j)

    pid_mn = tl.program_id(1)  # flatten (or axis 1?) Actually we can use axis 1 for M tiles; axis 2 for N tiles.

    # Use tl.program_id(1) for M tile index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for row and col blocks
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

    # compute pointers base offsets

    a_ptr = A_ptr + batch * stride_A_batch + head * stride_A_h # stride_A_h = 1 (since hidden dim innermost)
    b_ptr = B_ptr + batch * stride_B_batch + head * stride_B_h
    c_ptr = C_ptr + batch * stride_C_batch + head * stride_C_h

    # Compute offsets for accessing rows
    # For A (left): element A[i,k] => offset i*stride_i + k*stride_k
    # For B (right): B[j,k] => offset j*stride_i + k*stride_k (since we treat as B_T)

    # Load K tiles in loop:

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # range over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    mask_i = i_offs < N
    mask_j = j_offs < N

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Layer‑Norm the raw 4‑D tensor (float32) → half‑precision.
2️⃣  Fuse the four linear projections (left/right and two gates) into a single
    matrix‑multiply with a concatenated weight matrix.
3️⃣  Split the result, apply sigmoid to the two gate vectors and (optionally)
    mask them.  This yields the tensors `left` and `right` of shape
    [B, N, N, H] (H == hidden_dim) in FP16.
4️⃣  Core O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a custom Triton kernel that treats the hidden dimension as
    an extra batch axis, thereby avoiding the costly permute/contiguous
    reshapes used by the naïve torch.bmm approach.
5️⃣  Second Layer‑Norm (half‑precision) over the hidden dimension,
    element‑wise gating with `out_gate` (sigmoid of a linear on the normalized
    input) and a final linear projection back to the original feature dimension.
6️⃣  Cast the result back to float32 (required by the harness) and return.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimensions up to 768.  All heavy work is done in FP16; only the two
LayerNorms use the supplied weight/bias tensors (also cast to FP16).  The
implementation requires Triton ≥ 3.3.1 and runs on an H100 GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM that directly computes the
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# on tensors stored as [B, N, N, H] (row‑major, hidden dimension innermost).
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    A_ptr,                 # *fp16 : left tensor (B,N,N,H)
    B_ptr,                 # *fp16 : right tensor (B,N,N,H)
    C_ptr,                 # *fp16 : output tensor (B,N,N,H)
    B,                     # i32   : batch size
    N,                     # i32   : sequence length
    H,                     # i32   : hidden dimension
    batch_stride,          # i32   : stride to advance one batch (elements)
    row_stride,            # i32   : stride to advance one row (i‑index)
    col_stride,            # i32   : stride to advance one column (k‑index)
    BLOCK_M: tl.constexpr, # compile‑time tile size for i
    BLOCK_N: tl.constexpr, # compile‑time tile size for j
    BLOCK_K: tl.constexpr, # compile‑time tile size for reduction over k
):
    # ------------------------------------------------------------------
    # 1️⃣  Identify (batch, hidden) pair processed by THIS program.
    # ------------------------------------------------------------------
    bh = tl.program_id(0)                      # index over B*H programs
    batch = bh // H
    head  = bh % H                              # position inside hidden dim

    # ------------------------------------------------------------------
    # 2️⃣  Compute block coordinates for the output matrix (i,j).
    # ------------------------------------------------------------------
    pid_m = tl.program_id(1)                    # block row index (i)
    pid_n = tl.program_id(2)                    # block column index (j)

    # Offsets for the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑indices
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑indices

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 3️⃣  Pointer to the start of the (batch, head) slice.
    # ------------------------------------------------------------------
    a_ptr = A_ptr + batch * batch_stride + head        # hidden index = head
    b_ptr = B_ptr + batch * batch_stride + head
    c_ptr = C_ptr + batch * batch_stride + head

    # ------------------------------------------------------------------
    # 4️⃣  Accumulator (float32 for reduced rounding error)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 5️⃣  Loop over the reduction dimension k.
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)   # k‑indices
        mask_k = offs_k < N

        # Load A[i,k]  → shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + offs_i[:, None] * row_stride + offs_k[None, :] * col_stride,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B_T[k,j] (i.e. right[j,k]) → shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr + offs_k[:, None] * col_stride + offs_j[None, :] * row_stride,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Multiply‑accumulate
        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_N)   FP32 accumulator

    # ------------------------------------------------------------------
    # 6️⃣  Write the result back (cast to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + offs_i[:, None] * row_stride + offs_j[None, :] * col_stride,
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _trmul_fused_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the outgoing TriMul contraction using the Triton kernel above.
    left / right : FP16 tensors of shape [B, N, N, H] (contiguous)
    Returns      : FP16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # Allocate output
    out = torch.empty_like(left)

    # Strides in *elements* (not bytes)
    batch_stride = left.stride(0)        # B → N*N*H
    row_stride   = left.stride(1)        # i → N*H
    col_stride   = left.stride(2)        # k → H

    # Tile sizes – empirically good on H100 for fp16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                   # axis 0 : batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,            # axis 1 : rows
        (N + BLOCK_N - 1) // BLOCK_N,            # axis 2 : columns
    )

    trmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        batch_stride,
        row_stride,
        col_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # 8 warps give good occupancy for 64×64 tiles
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]

        - weights      : dict name → torch.Tensor (float32)
        - config       : contains ``dim`` and ``hidden_dim`` and optional ``nomask``
    Returns

    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 (one‑time cost)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse the four linear projections (left/right + two gates)
    # ------------------------------------------------------------------
    # shape of each: (hidden_dim, dim)
    # combined shape: (4*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )  # [4*H, dim]

    # One big linear (no bias)
    combined = torch.nn.functional.linear(x, combined_weight)  # [B,N,N,4H] fp16

    # Split
    h = hidden_dim
    left_proj_raw   = combined[..., :h]                      # [B,N,N,H]
    right_proj_raw  = combined[..., h : 2 * h]
    left_gate_raw   = combined[..., 2 * h : 3 * h]
    right_gate_raw  = combined[..., 3 * h :]

    # --------------------------------------------------------------
    # 5️⃣ Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # --------------------------------------------------------------
    # 6️⃣ Apply mask (if any) and combine projections with gates
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: (B, N, N) → broadcast over hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left_proj_raw * left_gate * mask_fp16
        right = right_proj_raw * right_gate * mask_fp16
    else:
        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # Ensure contiguous layout before feeding into the Triton kernel
    left  = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate for the final output (computed on the original normed x)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 8️⃣ Core O(N³) contraction using the Triton kernel
    # --------------------------------------------------------------
    out = _trmul_fused_gemm(left, right)  # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm (over hidden dimension)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # --------------------------------------------------------------
    # 🔟 Gating + final projection back to original dimension
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 11️⃣ Cast back to float32 (required by the harness) and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
