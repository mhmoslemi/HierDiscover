@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr, raw_gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    strides...
    BLOCK_I, BLOCK_J, BLOCK_H:

    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    off_i = pid_i*BLOCK_I + tl.arange(0,BLOCK_I)
    off_j = pid_j*BLOCK_J + tl.arange(0,BLOCK_J)

    mask_i = off_i < N
    mask_j = off_j < N

    # load mask tile (scalar)
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + off_i[:,None] * stride_mask_i + off_j[None,:] * stride_mask_j
    mask_tile = tl.load(mask_ptrs, mask=mask_i[:,None] & mask_j[None,:], other=0.0)  # shape [BLOCK_I, BLOCK_J]

    # iterate over hidden dimension tiles
    num_h = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h):
        off_h = h_tile*BLOCK_H + tl.arange(0,BLOCK_H)
        mask_h = off_h < H

        proj_ptrs = proj_ptr + pid_b * stride_proj_b + off_i[:,None,None] * stride_proj_i + off_j[None,:,None] * stride_proj_j + off_h[None,None,:] * stride_proj_h
        proj_tile = tl.load(proj_ptrs,
                            mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:],
                            other=0.0)

        raw_gate_ptrs = raw_gate_ptr + pid_b * stride_gate_b + off_i[:,None,None] * stride_gate_i + off_j[None,:,None] * stride_gate_j + off_h[None,None,:] * stride_gate_h
        raw_gate_tile = tl.load(raw_gate_ptrs,
                                mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:],
                                other=0.0)

        # compute sigmoid: 1/(1+exp(-x))
        # Use tl.exp for exponent.
        gate_tile = 1.0 / (1.0 + tl.exp(-raw_gate_tile))

        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        out_ptrs = out_ptr + pid_b * stride_out_b + off_i[:,None,None] * stride_out_i + off_j[None,:,None] * stride_out_j + off_h[None,None,:] * stride_out_h
        tl.store(out_ptrs, out_tile,
                 mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:])

"""
TriMul (outgoing) forward‐pass, heavily fused for speed.

Algorithm
---------
1️⃣  Layer‑norm the input (FP32) → cast to FP16.
2️⃣  Fuse the five linear transforms (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a *single* matmul with a stacked weight
    matrix → shape [B, N, N, 5·hidden_dim] (FP16).
3️⃣  Slice the result; apply sigmoid to the three gate slices.
4️⃣  If a mask is present, fuse mask, sigmoid‑gate and projection in a
    Triton kernel (`_proj_gate_mask_kernel`).  Otherwise multiply by the
    gates in‑place (still FP16).
5️⃣  Heavy O(N³) contraction  out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]
    is performed with a single `torch.einsum`, which avoids the
    costly permute‑+‑contiguous reshapes of the naïve batched‑GEMM.
6️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → elementwise product.
7️⃣  Final linear projection back to the model dimension (FP16).
8️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is written in Triton; all other heavy
operations stay in PyTorch where cuBLAS/Tensor‑cores are already optimal.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: (proj * sigmoid(gate) * mask) – mask broadcast over H
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,            # *fp16  [B, N, N, H]   (projection)
    raw_gate_ptr,        # *fp16  [B, N, N, H]   (pre‑sigmoid gate)
    mask_ptr,            # *fp16  [B, N, N]      (binary mask)
    out_ptr,             # *fp16  [B, N, N, H]   (output)
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute: out = proj * sigmoid(raw_gate) * mask,
    with mask broadcast over the hidden dimension H.
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # ------------------------------------------------------------------
    # Index computation
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast later over H
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # shape [BLOCK_I, BLOCK_J]  (fp16)

    # --------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ------------------------------------------------------------------
        # Load projection tile
        # ------------------------------------------------------------------
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Load raw gate tile
        # ------------------------------------------------------------------
        gate_ptrs = (
            raw_gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        raw_gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Sigmoid on the gate (1 / (1 + exp(-x))) – fp16 arithmetic
        # ------------------------------------------------------------------
        gate_tile = 1.0 / (1.0 + tl.exp(-raw_gate_tile))

        # ------------------------------------------------------------------
        # Fuse: proj * gate * mask
        # ------------------------------------------------------------------
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # ------------------------------------------------------------------
        # Store result back into the projection buffer (in‑place)
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: apply mask + sigmoid‑gate fusion using Triton.
# ----------------------------------------------------------------------
def _apply_proj_gate_mask(proj: torch.Tensor, raw_gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, raw_gate : [B, N, N, H]  (fp16)
    mask           : [B, N, N]    (fp16)

    Returns out = proj * sigmoid(raw_gate) * mask (fp16),
    written *in‑place* into ``proj`` and also returned for convenience.
    """
    B, N, _, H = proj.shape
    # Cast mask to fp16 if not already – it will be broadcast inside the kernel.
    mask = mask.to(dtype=torch.float16)

    # Determine strides in *elements* (not bytes)
    s_proj = proj.stride()
    s_gate = raw_gate.stride()
    s_mask = mask.stride()
    s_out = proj.stride()   # we write back into proj (same strides)

    # Tile sizes – trade‑off between register pressure and occupancy.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _proj_gate_mask_kernel[grid](
        proj,
        raw_gate,
        mask,
        proj,                     # in‑place write‑back
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return proj  # now contains the gated & masked result


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – FP32
    x = x.to(dtype=torch.float16)  # now fp16

    # -------------------------------------------------
    # 2️⃣ Stack the five weight matrices into one big linear
    # -------------------------------------------------
    # All Linear layers have bias=False ⇒ just concat the weight tensors.
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    combined_weight = torch.cat(
        [
            left_proj_w,          # (hid, dim)
            left_gate_w,          # (hid, dim)
            right_proj_w,         # (hid, dim)
            right_gate_w,         # (hid, dim)
            out_gate_w,           # (hid, dim)
        ],
        dim=0,  # (5*hid, dim)
    ).to(device=device, dtype=torch.float16)  # fp16

    # -------------------------------------------------
    # 3️⃣ Single fused Linear → shape [B, N, N, 5·hid]
    # -------------------------------------------------
    combined = torch.nn.functional.linear(x, combined_weight)  # FP16
    # -------------------------------------------------
    # 4️⃣ Slice the projections & raw gates
    # -------------------------------------------------
    # NOTE: Slicing yields views – no extra memory allocation.
    h = hidden_dim
    left_proj = combined[..., 0 * h : 1 * h]                     # [B, N, N, h]
    left_gate_raw = combined[..., 1 * h : 2 * h]                # [B, N, N, h]
    right_proj = combined[..., 2 * h : 3 * h]                   # [B, N, N, h]
    right_gate_raw = combined[..., 3 * h : 4 * h]               # [B, N, N, h]
    out_gate_raw = combined[..., 4 * h : 5 * h]                 # [B, N, N, h]

    # -------------------------------------------------
    # 5️⃣ Apply mask + sigmoid‑gate (fused via Triton) or plain gating
    # -------------------------------------------------
    if not nomask:
        # mask may be bool or float – convert to fp16 (0.0/1.0)
        mask_fp16 = mask_tensor.to(dtype=torch.float16)
        left = _apply_proj_gate_mask(left_proj, left_gate_raw, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate_raw, mask_fp16)
    else:
        # No mask → simple sigmoid + element‑wise product (all fp16)
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction.
    #    Use torch.einsum to avoid the costly permute‑contiguous shuffle.
    # -------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum("bikd,bjkd->bijd", left, right)   # fp16 output

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) over hidden dimension
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # -------------------------------------------------
    # 8️⃣ Out‑gate and final linear projection back to model dim
    # -------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)          # fp16
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)  # fp16 → [B,N,N,dim]

    # -------------------------------------------------
    # 9️⃣ Cast back to FP32 (the harness expects FP32)
    # -------------------------------------------------
    return out.to(dtype=torch.float32)
