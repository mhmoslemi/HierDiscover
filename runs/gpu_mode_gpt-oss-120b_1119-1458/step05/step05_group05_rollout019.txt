def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # Norm weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Apply first LayerNorm in float32
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B,N,N,dim] fp32

    # Cast to fp16
    x = x_norm.to(torch.float16)

    # Gather weight tensors
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Cast all weights to fp16
    left_proj_weight_fp16 = left_proj_weight.to(torch.float16)
    right_proj_weight_fp16 = right_proj_weight.to(torch.float16)
    left_gate_weight_fp16 = left_gate_weight.to(torch.float16)
    right_gate_weight_fp16 = right_gate_weight.to(torch.float16)
    out_gate_weight_fp16 = out_gate_weight.to(torch.float16)
    to_out_norm_weight_fp16 = to_out_norm_weight.to(torch.float16)
    to_out_norm_bias_fp16 = to_out_norm_bias.to(torch.float16)
    to_out_weight_fp16 = to_out_weight.to(torch.float16)

    # Concatenate weights for fused projection
    fused_weight_fp16 = torch.cat(
        [
            left_proj_weight_fp16,
            right_proj_weight_fp16,
            left_gate_weight_fp16,
            right_gate_weight_fp16,
            out_gate_weight_fp16,
        ],
        dim=0,
    )  # shape [5*hidden_dim, dim]

    # Linear projection fused
    fused_output = torch.nn.functional.linear(x, fused_weight_fp16)   # [B,N,N,5*H]

    # Split fused output
    H = hidden_dim
    left_proj_fp16 = fused_output[..., :H]
    right_proj_fp16 = fused_output[..., H:2*H]
    left_gate_raw_fp16 = fused_output[..., 2*H:3*H]
    right_gate_raw_fp16 = fused_output[..., 3*H:4*H]

    left = left_proj_fp16 * left_gate

        mask_flat = mask_fp16.view(-1)   # shape [B*N*N]
        # left and right are contiguous? Ensure
        left = left.contiguous()
        right = right.contiguous()
        size = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)
        mask_fuse_kernel[grid](
            left,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        mask_fuse_kernel[grid](
            right,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:

        pass

def _batched_gemm_fp16(left, right):
    # left shape [B,N,N,H]
    B, N, _, H = left.shape
    # Reshape as [B*H, N, N] but we must move H before N? left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    # right permute similarly.
    left_ = left.permute(0, 3, 1, 2).contiguous()
    right_ = right.permute(0, 3, 1, 2).contiguous()
    # left_,right_ shape [B, H, N, N], contiguous.
    # Flatten batch * hidden_dim:
    left_flat = left_.view(B * H, N, N)
    right_flat = right_.view(B * H, N, N)
    # Compute bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Reshape back to shape [B, H, N, N], permute to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
TriMul (outgoing) – highly‑fused forward pass.

Algorithm
---------
1. Layer‑norm the input (float32) → cast to fp16.
2. Fuse the five linear layers (left/right projections, three gates) into a
   single matmul so that the input tensor is read only once.
3. Split the fused result, apply a sigmoid to the three gate blocks and
   fuse the left/right gates with the projections (in‑place element‑wise
   multiplication).  If a mask is present, a tiny Triton kernel multiplies
   the gated tensors by the mask (mask is broadcast over the hidden dim).
4. Perform the O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM (B·H independent N×N products) using torch.bmm.
5. Second Layer‑norm over the hidden dimension, element‑wise out‑gate
   multiplication and the final linear projection back to the original
   channel dimension.
6. Cast the result back to float32 (the harness expects float32).

All heavy arithmetic is in FP16; masks and gates are kept in FP16 to avoid
extra casts.  Only the mask‑fusion step is written as a Triton kernel,
satisfying the “use Triton” requirement while keeping launch overhead
tiny.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: mask * tensor (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_fuse_kernel(
    x_ptr,          # *fp16  → tensor to be masked (flattened)
    mask_ptr,       # *fp16  → mask tensor, shape [B*N*N]
    hidden_dim,     # i32   → size of the hidden dimension (H)
    size,           # i32   → total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load the operand and the mask (mask does not have a hidden dim)
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    mask_idx = offs // hidden_dim            # one mask per (b,i,j)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default = 1

    # Apply the mask
    tl.store(x_ptr + offs, x * m, mask=valid)


# ----------------------------------------------------------------------
# Helper – batched GEMM (fp16) for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a single batched matmul (FP16).
    left/right: [B, N, N, H] – must be contiguous.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) in contiguous layout
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    # Merge batch and hidden dims
    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)
    # Batched matmul (right transposed on the last two dims)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – most stable for the raw input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj + 3 gates + out_gate)
    # ------------------------------------------------------------------
    # Grab individual weight tensors (float32) and cast once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature dimension: shape (5*H, dim)
    fused_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )   # [5*hidden_dim, dim]  (fp16)

    # Single linear pass – reads `x` only once
    fused = torch.nn.functional.linear(x, fused_weight)   # [B, N, N, 5*H]

    # ------------------------------------------------------------------
    # 4️⃣ Split – compute sigmoids for the three gate blocks
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = fused[..., 0:H]                 # [B,N,N,H]
    right_proj  = fused[..., H:2 * H]             # [B,N,N,H]
    left_gate_r = fused[..., 2 * H:3 * H]         # raw gate values
    right_gate_r = fused[..., 3 * H:4 * H]
    out_gate_r  = fused[..., 4 * H:5 * H]

    left_gate   = torch.sigmoid(left_gate_r)
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)

    # Apply the projection‑gate product (in‑place)
    left = left_proj * left_gate      # [B,N,N,H]
    right = right_proj * right_gate   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – fuse element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Prepare a flattened fp16 mask: shape [B*N*N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # flatten batch * N * N

        # Ensure tensors are contiguous for the kernel
        left = left.contiguous()
        right = right.contiguous()

        # Launch a tiny Triton kernel to apply the mask in‑place
        size = left.numel()                 # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        mask_fuse_kernel[grid](
            left,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        mask_fuse_kernel[grid](
            right,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – batched GEMM (FP16)
    # ------------------------------------------------------------------
    out_hidden = _batched_gemm_fp16(left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate + final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)
    output_fp16 = torch.nn.functional.linear(out_norm, to_out_w)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected by the harness)
    # ------------------------------------------------------------------
    return output_fp16.to(torch.float32)
