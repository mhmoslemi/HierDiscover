def _batched_matmul_fp16(left, right):
    B,N,_,H = left.shape
    # shape (B,H,N,N)
    left_permuted = left.permute(0, 3, 1, 2).contiguous()
    right_permuted = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)
    # view as (B*H,N,N)
    A = left_permuted.view(B*H, N, N)
    B_ = right_permuted.view(B*H, N, N)
    C = torch.bmm(A, B_.transpose(1,2)).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

"""
custom_kernel – highly‑optimised “outgoing” TriMul for AlphaFold‑3.

Algorithm
---------
1. Layer‑Norm the input (FP32) → cast to FP16 for the heavy part.
2. Fuse the two projection matrices (proj + gate) for the left and right sides:
   *   W_left  = [left_proj; left_gate]      (2 · hidden_dim × dim)
   *   W_right = [right_proj; right_gate]    (2 · hidden_dim × dim)
   One `torch.nn.functional.linear` per side produces
   `[proj , gate_pre]` in a single GEMM → `sigmoid` on the gate part.
3. (Optional) fuse mask, gate and the element‑wise multiplication for *both*
   sides with a tiny Triton kernel – this removes an extra pass over memory.
4. Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `C = A @ Bᵀ` where
   `A,B ∈ ℝ^{(B·H)×N×N}`.  The implementation uses `torch.bmm`,
   which on H100 maps to a Tensor‑Core‑accelerated GEMM.
5. Second LayerNorm over the hidden dimension.
6. Apply the output gate (`sigmoid`) and the final linear projection back
   to `dim`.  All these last three ops are performed in FP16.
7. Cast the result back to FP32 (the harness expects float32).

At least one Triton kernel is used (step 3).  The rest of the work relies on
high‑performance Torch kernels and FP16 Tensor‑Core acceleration.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and element‑wise multiplication for
# left and right tensors (in‑place).  All tensors are contiguous FP16.
#   left  ← left  * left_gate  * mask
#   right ← right * right_gate * mask
# mask has shape [B,N,N] and is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in‑place)
    right_ptr,         # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcasted)
    total_elements,    # i32   total number of elements in left/right (B·N·N·H)
    H,                 # i32   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    # flat index of the element this thread works on
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # index of the mask value (broadcast over H)
    mask_off = offs // H

    # Load everything (out‑of‑bounds loads return 0/1 as requested)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # mask is 0/1 (FP16).  Use 1.0 for OOB entries to keep the multiplication
    # neutral – those values are later discarded by the “mask” guard.
    mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Write back (in‑place)
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out = Σ_k left[...,k,:] * right[...,k,:]``  (outer product over k)
    for tensors of shape [B, N, N, H] using a batched GEMM.
    Returns a tensor of the same shape and dtype (FP16).
    """
    B, N, _, H = left.shape
    # (B, H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    # Collapse (B·H) batch dimension
    A = left_t.view(B * H, N, N)
    B_ = right_t.view(B * H, N, N)
    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    # Restore original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (bool/float, ignored if nomask)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16 later)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (all on the same device)
    # ------------------------------------------------------------------
    # Concatenate projection & gate matrices for left and right sides
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)          # (2·H, dim)

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)          # (2·H, dim)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Left side: one fused linear → split → sigmoid gate
    # ------------------------------------------------------------------
    left_cat = F.linear(x, left_concat_w)                       # [B,N,N,2·H]
    left_raw = left_cat[..., :hidden_dim].contiguous()
    left_gate_pre = left_cat[..., hidden_dim:].contiguous()
    left_gate = torch.sigmoid(left_gate_pre)                    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Right side: same pattern
    # ------------------------------------------------------------------
    right_cat = F.linear(x, right_concat_w)                     # [B,N,N,2·H]
    right_raw = right_cat[..., :hidden_dim].contiguous()
    right_gate_pre = right_cat[..., hidden_dim:].contiguous()
    right_gate = torch.sigmoid(right_gate_pre)                  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (used after the second LayerNorm)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))          # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Mask + gate fusion (Triton) or simple element‑wise multiply
    # ------------------------------------------------------------------
    if nomask:
        left = left_raw * left_gate
        right = right_raw * right_gate
    else:
        # Ensure all tensors are contiguous
        left_raw = left_raw.contiguous()
        right_raw = right_raw.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_tensor.to(dtype=dtype_fp16).contiguous()   # [B,N,N]

        total = left_raw.numel()
        BLOCK = 1024  # max threads per block on H100
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left_raw,
            right_ptr=right_raw,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # In‑place modification performed by the kernel
        left = left_raw
        right = right_raw

    # ------------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction via batched FP16 GEMM
    # ------------------------------------------------------------------
    # out: [B, N, N, H] (FP16)
    out = _batched_matmul_fp16(left, right)

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 🔟 Apply output gate and final projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate                        # element‑wise (FP16)
    out = F.linear(out, to_out_w)               # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to FP32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
