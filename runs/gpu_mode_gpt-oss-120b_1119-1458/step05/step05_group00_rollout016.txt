def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H] (in‑place)
    right_ptr,         # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N] (broadcasted on H)
    total_elements,    # i64   total number of fp16 elements in left/right
    H,                 # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)      # flat index
    mask = offs < total_elements                               # OOB guard

    # mask index = flat_index // H  (broadcast along hidden)
    mask_offs = offs // H

    left      = tl.load(left_ptr      + offs, mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr     + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=tl.float16(0.0))
    mask_val  = tl.load(mask_ptr      + mask_offs,
                        mask=mask, other=tl.float16(1.0))

    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_ptr  + offs, left_out ,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#   left / right  : [B,N,N,H] (fp16)
#   out           : [B,N,N,H] (fp16) – written in‑place
#   Hidden dimension H is fused into the batch dimension (batch_h = B*H).
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,        # *fp16  [B,N,N,H]
    right_ptr,       # *fp16  [B,N,N,H]
    out_ptr,         # *fp16  [B,N,N,H]
    B, N, H,
    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # -------------------------------------------------
    # Decode program id → (batch_h, block_i, block_j)
    # -------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_i * num_blocks_j

    batch_h = pid // blocks_per_batch                # (0 .. B*H-1)
    block_idx = pid % blocks_per_batch

    pid_i = block_idx // num_blocks_j
    pid_j = block_idx % num_blocks_j

    # Recover original batch and hidden indices
    b = batch_h // H
    h = batch_h % H

    # -------------------------------------------------
    # Offsets for the output tile
    # -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the beginning of the (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over the k‑dimension in tiles
    # -------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ---- Load a tile of left (shape BLOCK_M × BLOCK_K) ----
        left_ptrs = left_base + (offs_i[:, None] * stride_left_i) \
                               + (cur_k[None, :] * stride_left_k)
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0)).to(tl.float32)

        # ---- Load a tile of right (shape BLOCK_N × BLOCK_K) ----
        # note: right is indexed as (b, j, k, h) – shape (N, N, H)
        right_ptrs = right_base + (offs_j[:, None] * stride_right_i) \
                                 + (cur_k[None, :] * stride_right_k)
        b_mat = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=tl.float16(0.0)).to(tl.float32)

        # ---- Multiply‑accumulate across K ----
        # tl.dot expects (M,K) and (N,K) and returns (M,N)
        acc += tl.dot(a, b_mat)

    # -------------------------------------------------
    # Write the result back (as FP16)
    # -------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) \
                         + (offs_j[None, :] * stride_out_j)

    c = acc.to(tl.float16)
    tl.store(out_ptrs,
             c,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# ---------------------------- Entry point ------------------------------
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float, optional)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)          # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 (the heavy part works in FP16)
    # --------------------------------------------------------------
    x = x.to(dtype_fp16)

    # --------------------------------------------------------------
    # 3️⃣ Cast all linear weights to FP16 on the same device
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (FP16) → [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Gating projections (FP16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (Triton)
    # --------------------------------------------------------------
    if nomask:
        # No mask → just apply the gates
        left = left * left_gate
        right = right * right_gate
    else:
        # Fuse mask, left_gate and right_gate in a single kernel
        # mask is broadcasted on the hidden dimension
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_elements = left.numel()
        BLOCK = 1024  # ~1K threads per block; empirically a good balance
        n_blocks = (total_elements + BLOCK - 1) // BLOCK
        _mask_gate_fusion_kernel[(n_blocks,)](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total_elements,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates have been fused; we can free the temporaries
        del left_gate, right_gate, mask_fp16

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via custom Triton kernel
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    out = torch.empty_like(left, dtype=dtype_fp16)

    # Get strides (in elements, not bytes) – they are needed by the kernel
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile size – tuned for H100 Tensor‑cores and 128‑dim hidden size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Number of output tiles per (b,h) slice
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_i * num_j

    # Grid encodes (batch_h * tiles_per_batch) programs
    total_batch_h = B * H
    grid = (total_batch_h * tiles_per_batch,)

    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,   # *j* dimension for right tensor
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # --------------------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate                 # element‑wise gate (FP16)
    out = F.linear(out, to_out_w)        # (B,N,N,dim)  FP16

    # --------------------------------------------------------------
    # 🔟 Cast back to FP32 (expected by the runner)
    # --------------------------------------------------------------
    return out.to(torch.float32)
