def mask_gate_fusion_sigmoid_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements, H,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    # indices for mask: for each hidden dimension index h, the mask offset is off // H
    mask_idx = offs // H
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr+ offs, mask=mask, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    # compute sigmoid
    left_gate = tl.where(left_gate_raw >= 0,
        1.0 / (1.0 + tl.exp(-left_gate_raw)),
        tl.exp(left_gate_raw) / (1.0 + tl.exp(left_gate_raw))) # using stable variant maybe?

    right_gate = tl.where(right_gate_raw >= 0,
        1.0 / (1.0 + tl.exp(-right_gate_raw)),
        tl.exp(right_gate_raw) / (1.0 + tl.exp(right_gate_raw)))
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)
```

"""
custom_kernel: outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1.  Layer‑norm the input (float32) -> cast to FP16.
2.  Fuse the five linear projections (left, right and three gates) into a
    single matmul:
        y = x_flat @ W_combined.T   (W_combined = concat of the 5 weight matrices)
    and split the result into
        left_proj, right_proj, left_gate_logit, right_gate_logit, out_gate_logit.
3.  Apply sigmoid to the three gating tensors.
4.  If a mask is present, fuse mask, left‑gate and right‑gate in a Triton
    elementwise kernel (in‑place update of left/right).  Otherwise multiply
    the projections by their gates directly.
5.  Heavy O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    which is a batched matrix multiplication C = A @ Bᵀ with
        A = left.permute(0,3,1,2).view(B*H,N,N)
        B = right.permute(0,3,1,2).view(B*H,N,N)
    performed with torch.bmm (FP16 Tensor‑core GEMM).
6.  Second LayerNorm over the hidden dimension (still FP16).
7.  Multiply by the final out‑gate and apply the output linear projection.
8.  Cast the result back to float32 (the harness expects fp32).

All heavy elementwise work (mask × gate) is executed by a custom Triton
kernel; the large N³ contraction uses a single batched GEMM (torch.bmm) which
is highly optimized on H100.  The implementation works for any
B∈{1,2}, N∈{128,256,512,1024}, dim∈{128,384} and hidden_dim=128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * left_gate * mask
#   right ← right * right_gate * mask
# All tensors are contiguous FP16 with shape [B, N, N, H] and mask has shape
# [B, N, N] (broadcast along H).  The kernel processes the flat memory layout.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,                # *fp16  [B,N,N,H] (in‑place)
    right_ptr,               # *fp16  [B,N,N,H] (in‑place)
    left_gate_ptr,           # *fp16  [B,N,N,H]
    right_gate_ptr,          # *fp16  [B,N,N,H]
    mask_ptr,                # *fp16  [B,N,N]   (broadcasted)
    total_elements,          # i64   total number of elements in left/right tensors
    H,                       # i64   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)           # flat offsets
    mask_oob = offs < total_elements                             # guard

    # mask index = flat_offset // H  (mask does not have the hidden dim)
    mask_offs = offs // H

    # Load data (out‑of‑bounds loads return 0/1 which are neutral for the op)
    left      = tl.load(left_ptr        + offs, mask=mask_oob, other=0.0)
    right     = tl.load(right_ptr       + offs, mask=mask_oob, other=0.0)
    left_gate = tl.load(left_gate_ptr   + offs, mask=mask_oob, other=0.0)
    right_gate= tl.load(right_gate_ptr  + offs, mask=mask_oob, other=0.0)

    # Load binary mask (0/1).  For OOB indices use mask value 1.0 (neutral)
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask_oob, other=1.0)

    # Fuse gating and masking
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store the results back in‑place
    tl.store(left_ptr  + offs, left_out , mask=mask_oob)
    tl.store(right_ptr + offs, right_out, mask=mask_oob)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H] (fp16)

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # bring hidden dimension to the batch axis
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)    # (B·H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ -> (B·H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))

    # reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()      # (B, N, N, H)
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32, then cast to fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the channel dimension `dim`
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)                    # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy parts
    # ------------------------------------------------------------------
    x = x.to(fp16)                                      # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections into a single matmul
    #    (left_proj, right_proj, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # Gather the five weight matrices (shape: [hidden_dim, dim] each)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Concatenate them along the output‑feature axis → shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device=device, dtype=fp16)                    # FP16 weights

    B, N, _, _ = x.shape
    # Flatten the spatial dimensions for a single GEMM
    x_flat = x.view(-1, dim)                            # (B·N·N, dim)

    # Compute all five projections in one go
    combined_out = F.linear(x_flat, combined_weight)    # (B·N·N, 5*hidden_dim)

    # Restore the four‑dimensional layout
    combined_out = combined_out.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Split the combined tensor
    # ------------------------------------------------------------------
    d0 = hidden_dim
    left_proj   = combined_out[..., :d0]                # [B,N,N,H]
    right_proj  = combined_out[..., d0:2*d0]            # [B,N,N,H]
    left_gate_l = combined_out[..., 2*d0:3*d0]          # logits
    right_gate_l = combined_out[..., 3*d0:4*d0]         # logits
    out_gate_l   = combined_out[..., 4*d0:]             # logits

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid to gating logits (still FP16)
    # ------------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_l)            # [B,N,N,H] FP16
    right_gate  = torch.sigmoid(right_gate_l)           # [B,N,N,H] FP16
    out_gate    = torch.sigmoid(out_gate_l)             # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Mask + gate fusion (Triton kernel) – executed only if a mask is present
    # ------------------------------------------------------------------
    if nomask:
        # No mask → just multiply with the gates
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Prepare mask as FP16 (broadcast over hidden dimension)
        mask_fp16 = mask_tensor.to(fp16)                # [B,N,N] 0/1

        # Ensure tensors are contiguous for the kernel
        left = left_proj.contiguous()
        right = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()          # B * N * N * H
        BLOCK = 1024                  # max threads per block
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # The kernel writes the gated & masked tensors back in‑place.
        # Gate tensors can be released early.
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched FP16 GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)            # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=fp16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)        # FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply final out‑gate and project back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                                 # element‑wise FP16
    to_out_w = weights["to_out.weight"].to(device=device, dtype=fp16)  # [dim, H]
    out = F.linear(out, to_out_w)                        # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
