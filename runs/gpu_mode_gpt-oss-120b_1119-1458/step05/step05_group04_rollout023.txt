@triton.jit
def batched_gemm_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Compute batch b and hidden h indices
    b = pid_batch // H
    h = pid_batch % H

    # Compute base pointer for this batch+hidden
    # base offset includes stride for batch and hidden offset
    a_base = a_ptr + b * stride_a_batch + h
    b_base = b_ptr + b * stride_b_batch + h
    c_base = c_ptr + b * stride_c_batch + h

    # Block offset for i and j
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension of size N
    # We'll compute number of KBLOCKs
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute k offsets
        k_offsets = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # pointers to A tile (i,k)
        a_ptrs = a_base + i_offsets[:, None] * stride_a_i + k_offsets[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None,:], other=0.0).to(tl.float32)

        # pointers to B tile (k,j) -> note B in original layout is (b, j, k, h). We treat B_T with row=k, col=j.
        b_ptrs = b_base + k_offsets[:, None] * stride_b_k + j_offsets[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None,:], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)
        # Note: we can avoid pointer increments using recompute each iteration

    # Store result
    c_ptrs = c_base + i_offsets[:, None] * stride_c_i + j_offsets[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

So stride for 'j' dimension = stride[1] = N*H = stride_b_i (previous variable). stride for 'k' dimension = stride[2] = H = stride_b_k. Good.

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def batched_tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    pid_batch = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # decompose batch and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # base pointers for this (b, h) slice
    a_base = a_ptr + b * stride_a_batch + h
    b_base = b_ptr + b * stride_b_batch + h
    c_base = c_ptr + b * stride_c_batch + h

    # tile offsets
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # A tile: rows i, cols k
        a_ptrs = a_base + i_off[:, None] * stride_a_i + k_off[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # B tile (transposed): rows k, cols j
        b_ptrs = b_base + k_off[:, None] * stride_b_k + j_off[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result tile
    c_ptrs = c_base + i_off[:, None] * stride_c_i + j_off[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass.
    Implements the heavy cubic contraction via a custom Triton batched‑GEMM kernel,
    fuses left/right linear projections with gating, and reduces memory copies.
    Operates in fp16 for speed and returns fp32 output.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Prepare weights (fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)

    # combine (proj + gate) weights for left and right side to halve the linear calls
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=torch.float16)

    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=torch.float16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (fp16) – normalize over the last dim
    # ------------------------------------------------------------------
    # Cast input to fp16 first, then apply LN with fp16 parameters
    x = input_tensor.to(torch.float16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣  Linear combined proj+gate (fp16) → split & sigmoid gate
    # ------------------------------------------------------------------
    # Left side
    left_combined = torch.nn.functional.linear(x, left_combined_w)   # [B,N,N,2*H]
    left_proj, left_gate_raw = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_raw)

    # Right side
    right_combined = torch.nn.functional.linear(x, right_combined_w)
    right_proj, right_gate_raw = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 3️⃣  Apply gating (and optional mask) – element‑wise multiply
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 4️⃣  Heavy cubic contraction – custom batched GEMM in Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B,N,N,H

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton batched‑GEMM kernel for the cubic contraction
# Computes, for every (b, h):
#   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
# Input tensors are stored as [B, N, N, H] with the hidden dimension
# as the innermost (contiguous) axis. No extra permute / copy is needed.
# ----------------------------------------------------------------------
@triton.jit
def batched_tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # combined batch*hidden index
    pid_i     = tl.program_id(0)   # tile row
    pid_j     = tl.program_id(1)   # tile column

    # De‑compose the packed index into batch and hidden indices
    b = pid_batch // H
    h = pid_batch % H

    # Base pointers for the (b, h) slice – note hidden stride == 1, so we just add h
    a_base = a_ptr + b * stride_a_batch + h
    b_base = b_ptr + b * stride_b_batch + h
    c_base = c_ptr + b * stride_c_batch + h

    # Offsets inside the tile
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # Accumulator in FP32 (Tensor‑core dot accumulates to FP32 automatically)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the contraction dimension k (size N)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ---- Load A tile: shape (BLOCK_M, BLOCK_K) ----
        # A layout: (i, k, h)  → strides (stride_a_i, stride_a_k, 1)
        a_ptrs = a_base + i_off[:, None] * stride_a_i + k_off[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)               # fp16

        # ---- Load B tile (transposed) : shape (BLOCK_K, BLOCK_N) ----
        # Original B layout: (j, k, h). We need Bᵀ so rows correspond to k.
        b_ptrs = b_base + k_off[:, None] * stride_b_k + j_off[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)               # fp16

        # ---- Matrix multiply over the K‑block ----
        acc += tl.dot(a, b)               # fp16·fp16 → fp32 accumulation

    # ---- Store the result tile ------------------------------------------------
    c_ptrs = c_base + i_off[:, None] * stride_c_i + j_off[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Algorithm
    ---------
    1️⃣  Layer‑Norm the 4‑D input in fp16.
    2️⃣  Fuse the left/right linear projections with their gates by concatenating
        the weight matrices → only two linear calls instead of four.
    3️⃣  Apply sigmoid gates (and optional mask) element‑wise.
    4️⃣  Compute the cubic contraction with a custom Triton batched‑GEMM kernel
        that works directly on the native [B, N, N, H] layout (no permute/copy).
    5️⃣  Second Layer‑Norm, out‑gate (sigmoid) and final linear projection.
    6️⃣  Cast the result back to float32 for the caller.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Cast all model parameters to fp16 on target device
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=torch.float16)

    # combine left projection + left gate weights (dim → 2·hidden)
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    # combine right projection + right gate weights
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (fp16) over the channel dimension `dim`
    # ------------------------------------------------------------------
    # Cast input to fp16 first – loss of a few bits is acceptable for speed
    x = input_tensor.to(torch.float16)
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # 2️⃣  Fused projection + gate (fp16) – split + sigmoid
    # ------------------------------------------------------------------
    # Left side: produces [B, N, N, 2·hidden]
    left_combined = torch.nn.functional.linear(x, left_combined_w)
    left_proj, left_gate_raw = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_raw)

    # Right side
    right_combined = torch.nn.functional.linear(x, right_combined_w)
    right_proj, right_gate_raw = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 3️⃣  Apply element‑wise gating (and optional mask)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 4️⃣  Cubic contraction via custom Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                         # H == hidden_dim
    out = torch.empty_like(left)                    # fp16, [B, N, N, H]

    # Strides for the native layout [B, N, N, H]
    stride_a_batch = left.stride(0)   # N*N*H
    stride_a_i     = left.stride(1)   # N*H
    stride_a_k     = left.stride(2)   # H
    stride_b_batch = right.stride(0)
    stride_b_i     = right.stride(1)
    stride_b_k     = right.stride(2)
    stride_c_batch = out.stride(0)
    stride_c_i     = out.stride(1)
    stride_c_j     = out.stride(2)

    # Tunable tile sizes – give good occupancy for all benchmark sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j
        B * H,                          # one program per (batch, hidden)
    )

    batched_tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                    # 4 warps → 128 threads per program (good for H100)
    )

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # 6️⃣  Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
