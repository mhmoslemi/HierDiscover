def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. First LayerNorm (FP32) with weight and bias
    x_normed = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # 2. FP16 conversion
    x = x_normed.to(torch.float16)

    device = input_tensor.device

    # 3. Build combined weight (5*hidden_dim, dim) for linear projections & gates
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0
    ).to(device=device, dtype=torch.float16)

    # 4. Single fused linear
    # Input shape [B, N, N, dim]; output shape [B, N, N, 5*hidden_dim]
    fused = torch.nn.functional.linear(x, combined_weight)  # no bias

    h = hidden_dim
    left_proj = fused[..., :h]
    right_proj = fused[..., h:2*h]
    left_gate = torch.sigmoid(fused[..., 2*h:3*h])
    right_gate = torch.sigmoid(fused[..., 3*h:4*h])
    out_gate = torch.sigmoid(fused[..., 4*h:5*h])

    # 5. Apply gates (and optional mask) using Triton kernel for mask case
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6. O(N^3) contraction: batched matmul
    out = _batched_matmul_transpose_fp16(left, right)

    # 7. Second LayerNorm (FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Out‑gate scaling and final linear projection
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # shape [B,N,N,dim] fp16

    # 9. Cast back to FP32
    return out.to(torch.float32)

"""
Highly‑optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm the input in FP32 (numerical stability) and cast to FP16.
2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into a single matrix multiplication.  This reduces the number
    of passes over the huge [B, N, N, dim] tensor from five to one while
    keeping all operations in FP16 → Tensor‑Core friendly.
3️⃣  Apply sigmoid to the three gate slices.
4️⃣  If a mask is present, fuse ``proj * gate * mask`` in a tiny Triton kernel
    (otherwise a simple element‑wise multiplication is used).
5️⃣  Perform the O(N³) contraction
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    as a batched GEMM C = A @ Bᵀ using cuBLAS (torch.bmm).  The batch dimension
    is ``B * hidden_dim`` which lets cuBLAS run many 1024×1024 matmuls in
    parallel.
6️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final projection back to
    the model dimension.
7️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is written in Triton, satisfying the requirement
to include a custom kernel while all heavy linear algebra is delegated to
highly‑optimized cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden dim H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                # *fp16  [B, N, N, H]
    gate_ptr,                # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,                # *fp16  [B, N, N]
    out_ptr,                 # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute:  out = proj * gate * mask   (mask broadcast over hidden dim H)."""
    pid_b = tl.program_id(0)      # batch index
    pid_i = tl.program_id(1)      # i‑tile
    pid_j = tl.program_id(2)      # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – will be broadcast over H later
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Loop over hidden‑dimension tiles (BLOCK_H)
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store the result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse proj, gate and mask (all fp16) into ``proj * gate * mask``.
    ``proj`` and ``gate`` are of shape [B, N, N, H].
    ``mask`` is of shape [B, N, N] and is broadcast over H.
    Returns an fp16 tensor with the same shape as ``proj``.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a strided‑batched GEMM (fp16 inputs → fp16 output).

    Arguments
    ---------
    left, right : [B, N, N, H] (fp16, contiguous)

    Returns
    -------
    out : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # Merge the hidden dimension into the batch dimension for a single cudaBMM call.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM:  C = A @ Bᵀ   (cuBLAS tensor‑cores)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # now fp16

    # -------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # shape: [5*hidden_dim, dim]

    # One large FP16 GEMM: (B·N·N, dim) @ (dim, 5*hidden_dim).T → (B·N·N, 5*hidden_dim)
    fused = torch.nn.functional.linear(x, combined_weight)   # [B, N, N, 5*hidden_dim]

    h = hidden_dim
    left_proj   = fused[..., :h]
    right_proj  = fused[..., h:2 * h]
    left_gate   = torch.sigmoid(fused[..., 2 * h:3 * h])
    right_gate  = torch.sigmoid(fused[..., 3 * h:4 * h])
    out_gate    = torch.sigmoid(fused[..., 4 * h:5 * h])

    # -------------------------------------------------
    # 3️⃣ Apply gates (and optional mask) – Triton for masked case
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)   # [B, N, N]
        left  = _apply_mask_gate(left_proj,  left_gate,  mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # -------------------------------------------------
    # 6️⃣ Out‑gate scaling + final projection back to model dim
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),
    )   # [B, N, N, dim] (fp16)

    # -------------------------------------------------
    # 7️⃣ Cast back to FP32 (the harness expects FP32)
    # -------------------------------------------------
    return out.to(torch.float32)
