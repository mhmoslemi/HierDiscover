The implementation follows these steps:
1️⃣ Layer‑norm the input (float32) for numerical stability.
2️⃣ Cast to fp16 and apply the four linear projections (left, right,
   left_gate, right_gate, out_gate) with bias‑free weights.
3️⃣ Optionally mask the projected tensors.
4️⃣ Fuse the gates into `left` and `right` via in‑place multiplication.
5️⃣ Perform the O(N³) contraction with a custom Triton kernel that
   avoids any data layout permutation.
6️⃣ Apply the second LayerNorm, the output gate, and the final linear
   projection back to the original `dim`.
7️⃣ Return the result as float32.

The kernel is tuned for H100:
* Tile sizes (BLOCK_M, BLOCK_N, BLOCK_K) are chosen at runtime based on
  the sequence length (64×64×64 for N≤256, 128×128×64 for 256<N≤512,
  128×128×128 for larger N).
* 8 warps and 2 stages give a good balance between latency hiding
  and shared‑memory pressure.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (fp16 × fp16 → fp32 accumulate)
#   A, B, C : [B, N, N, H]   (original layout, no permutation)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,          # *fp16   [B, N, N, H]
    right_ptr,         # *fp16   [B, N, N, H]
    out_ptr,           # *fp16   [B, N, N, H]   (output)
    B, N, H,           # i32 sizes
    # left strides
    stride_lb, stride_li, stride_lk, stride_lh,
    # right strides
    stride_rb, stride_ri, stride_rk, stride_rh,
    # out strides
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix‑multiply kernel for the outgoing TriMul contraction.
    Each program instance works on one (batch, hidden) pair and a tile
    of the output (i, j).  The K‑dimension is summed in a loop.
    """
    pid_i = tl.program_id(0)        # tile along output row dimension i
    pid_j = tl.program_id(1)        # tile along output column dimension j
    pid_z = tl.program_id(2)        # combined batch*hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden index from pid_z
    # --------------------------------------------------------------
    batch = pid_z // H
    hidden = pid_z % H

    # --------------------------------------------------------------
    # Tile offsets for i and j dimensions
    # --------------------------------------------------------------
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # --------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # --------------------------------------------------------------
    left_base  = left_ptr  + batch * stride_lb + hidden * stride_lh
    right_base = right_ptr + batch * stride_rb + hidden * stride_rh
    out_base   = out_ptr   + batch * stride_ob + hidden * stride_oh

    # --------------------------------------------------------------
    # Accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K‑tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)

        # Load A = left[batch, i, k, hidden]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + \
                 i_off[:, None] * stride_li + \
                 k_off[None, :] * stride_lk
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (k_off[None, :] < N),
            other=0.0,
        )  # fp16

        # Load B = right[batch, j, k, hidden]   shape (BLOCK_N, BLOCK_K)
        b_ptrs = right_base + \
                 j_off[:, None] * stride_ri + \
                 k_off[None, :] * stride_rk
        b = tl.load(
            b_ptrs,
            mask=mask_j[:, None] & (k_off[None, :] < N),
            other=0.0,
        )  # fp16

        # Mixed‑precision dot (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result  (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = out_base + \
             i_off[:, None] * stride_oi + \
             j_off[None, :] * stride_oj
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based contraction for left/right of shape [B, N, N, H].
    Returns a tensor of shape [B, N, N, H] (fp16) without any layout copy.
    """
    B, N, _, H = left.shape

    # Allocate output tensor (fp16)
    out = torch.empty_like(left)

    # Strides in elements (not bytes)
    lsb, lsi, lsk, lsh = left.stride()   # (B, i, k, h)
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # Choose tile sizes based on sequence length
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Cast model weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)    # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)   # (H, dim)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)    # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)   # (H, dim)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)     # (H, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (dim, H)

    # ------------------------------------------------------------------
    # 5️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x, right_proj_w)  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask broadcasted over hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B, N, N, H] fp16
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # [B, N, N, H] fp16

    # Fuse gates in‑place to reduce memory traffic
    left.mul_(left_gate)
    right.mul_(right_gate)

    # ------------------------------------------------------------------
    # 8️⃣ Heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    contracted = _batched_matmul_triton(left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    contracted = F.layer_norm(
        contracted,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    contracted = contracted * out_gate
    out = F.linear(contracted, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the test harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
