@triton.jit
def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_a_batch, stride_a_row, stride_a_col,
                        stride_b_batch, stride_b_row, stride_b_col,
                        stride_c_batch, stride_c_row, stride_c_col,
                        BLOCK_M: tl.constexpr,
                        BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)    # batch dimension will be axis0
    pid_i = tl.program_id(1)       # i tile
    pid_j = tl.program_id(2)       # j tile
    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for boundaries
    mask_i = offs_i < M
    mask_j = offs_j < N
    # base pointers for this batch
    a_ptrs = A_ptr + pid_batch * stride_a_batch
    b_ptrs = B_ptr + pid_batch * stride_b_batch
    c_ptrs = C_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # load A tile
        a = tl.load(a_ptrs + (offs_i[:, None] * stride_a_row
                              + offs_k[None, :] * stride_a_col),
                    mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # load B tile (B is B_T: shape (K, N))
        b = tl.load(b_ptrs + (offs_k[:, None] * stride_b_row
                              + offs_j[None, :] * stride_b_col),
                    mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        acc += tl.dot(a, b)   # yield fp32 accumulation
    # writeback
    # cast to output dtype (fp16)
    c = acc.to(tl.float16)
    tl.store(c_ptrs + (offs_i[:, None] * stride_c_row + offs_j[None, :] * stride_c_col),
                c,
                mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_ai, stride_aj,  # batch, i, k strides for a
    stride_bz, stride_bi, stride_bj,  # batch, k, j strides for b (b is B_T)
    stride_cz, stride_ci, stride_cj,  # batch, i, j strides for c
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)  # batch idx
    pid_y = tl.program_id(1)  # tile row (i)
    pid_x = tl.program_id(2)  # tile col (j)

    # Compute base pointers for this batch
    a_batch_ptr = a_ptr + pid_z * stride_az
    b_batch_ptr = b_ptr + pid_z * stride_bz
    c_batch_ptr = c_ptr + pid_z * stride_cz

    # Offsets within this batch
    offs_i = pid_y * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_x * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a = tl.load(a_batch_ptr + (offs_i[:, None] * stride_ai + offs_k[None, :] * stride_aj), mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load B tile (K x N) because B is transposed (original shape K,N)
        b = tl.load(b_batch_ptr + (offs_k[:, None] * stride_bi + offs_j[None, :] * stride_bj), mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    tl.store(c_batch_ptr + (offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj), c, mask=mask_i[:, None] & mask_j[None, :])
```

The strides maps are:

- stride_aj: stride between columns in A: A.stride(2) = 1.

- stride_bi: stride between rows in B_T: B_T.stride(1) = N (since moving along K dimension? So stride of dimension 1 = N (element count along j). Actually B_T is (K,N). In contiguous memory, the stride for dimension 1 (K) is N? Let's compute: B_T.stride(0) (batch) = K*N; B_T.stride(1) (K) = N; B_T.stride(2) (N) = 1. So stride_bi = B_T.stride(1) = N. And stride_bj = B_T.stride(2) = 1.

- stride_cz: output batch stride = C.stride(0) = M*N.

- stride_ci: output row stride = M? Actually shape (batch,M,N). Output C shape (batch, M, N). Constrain: stride_cz = M*N, stride_ci = N, stride_cj = 1.
```

"""
Custom Triton forward pass for TriMul (outgoing) operation.

The implementation fuses:
- A single linear projection for all projection and gate weights via a concatenated weight matrix.
- Optional mask and gate multiplication using a custom Triton kernel.
- A batched matrix multiplication (A @ B^T) executed by a custom Triton GEMM kernel,
  which replaces the O(N^3) Einstein summation.
All heavy parts use FP16 for Tensor‑Core acceleration; final output is cast back to FP32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    hidden_dim, total_size,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_size

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # mask index = floor(offs/hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_ai, stride_aj,
    stride_bz, stride_bi, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program identifiers
    pid_z = tl.program_id(0)   # batch index (B*hidden_dim)
    pid_y = tl.program_id(1)   # tile row (i)
    pid_x = tl.program_id(2)   # tile col (j)

    # Compute base pointers for this batch
    a_batch = a_ptr + pid_z * stride_az
    b_batch = b_ptr + pid_z * stride_bz
    c_batch = c_ptr + pid_z * stride_cz

    # Compute tile offsets
    offs_i = pid_y * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_x * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_batch + (offs_i[:, None] * stride_ai + offs_k[None, :] * stride_aj),
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N) – note B is already transposed (K,N)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bi + offs_j[None, :] * stride_bj),
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj),
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) First LayerNorm in FP32 → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] fp32
    x = x_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Concatenated linear projection (proj + gates) → fp16
    # ------------------------------------------------------------------
    # Gather and concatenate linear weight matrices (biases are absent)
    # order: left_proj, right_proj, left_gate, right_gate, out_gate
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Single linear pass
    combined = torch.nn.functional.linear(x, concat_weight)  # [B, N, N, 5*hidden_dim]

    # Split combined tensor
    sep = hidden_dim
    left = combined[..., 0 * sep : 1 * sep]
    right = combined[..., 1 * sep : 2 * sep]
    left_gate_raw = combined[..., 2 * sep : 3 * sep]
    right_gate_raw = combined[..., 3 * sep : 4 * sep]
    out_gate_raw = combined[..., 4 * sep : 5 * sep]

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 3) Fuse mask (if present) and gate into left/right tensors
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → fp16 flattened
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # [B*N*N]
        # total number of elements in left/right
        total_size = left.numel()  # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_size, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            total_size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            total_size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # left_gate, right_gate are no longer needed
    else:
        # Simple elementwise multiplication (no mask)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Heavy O(N³) contraction via custom batched GEMM (fp16)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange tensors to (batch * hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden_dim
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Transpose right for GEMM
    right_T = right_mat.transpose(1, 2).contiguous()

    # Allocate output tensor
    out_mat = torch.empty_like(left_mat)

    # Strides (int64)
    stride_az = left_mat.stride(0)
    stride_ai = left_mat.stride(1)
    stride_aj = left_mat.stride(2)

    stride_bz = right_T.stride(0)
    stride_bi = right_T.stride(1)
    stride_bj = right_T.stride(2)

    stride_cz = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑Norm the 4‑D input (float32) and cast to FP16.
2. Fuse the five linear projections (left/right and three gates) into a single
   matmul by concatenating the weight matrices along the output dimension.
   This reduces memory traffic because the input tensor is read only once.
3. Split the joint result, apply sigmoid to the three gate tensors.
4. Optional mask: a tiny Triton kernel fuses *gate* and *mask* element‑wise
   multiplication (`x ← x * gate * mask`) for both left and right tensors.
5. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom batched GEMM kernel that computes
   `C = A @ Bᵀ` for each hidden‑dim channel using Tensor‑Cores (FP16 inputs,
   FP32 accumulation).  The kernel tiles the M×N output and loops over K.
6. Second Layer‑Norm over the hidden dimension, element‑wise gating by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to float32 (the harness expects FP32).

Only the forward pass is implemented – gradients are not required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: fused multiplication of {tensor * gate * mask}
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  – tensor to be updated   (flattened)
    gate_ptr,       # *fp16  – gate tensor (same flattening)
    mask_ptr,       # *fp16  – mask tensor flattened, shape [B*N*N]
    hidden_dim,     # i32    – hidden dimension size (H)
    total_size,     # i32    – total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_size

    # Load operands
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # One mask element per hidden‑dim group
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fused multiply
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the N³ contraction
#   C[bh,i,j] = Σ_k A[bh,i,k] * B_T[bh,k,j]
# A : (batch_hidden, M, K)  – left tensor
# B_T : (batch_hidden, K, N) – right transposed
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_ai, stride_aj,
    stride_bz, stride_bi, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_z = tl.program_id(0)   # batch index (B*hidden_dim)
    pid_y = tl.program_id(1)   # tile row (i)
    pid_x = tl.program_id(2)   # tile col (j)

    # Base pointers for this batch
    a_batch = a_ptr + pid_z * stride_az
    b_batch = b_ptr + pid_z * stride_bz
    c_batch = c_ptr + pid_z * stride_cz

    # Tile offsets
    offs_i = pid_y * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_x * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator in FP32 (Tensor‑Core MMA)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a = tl.load(
            a_batch + (offs_i[:, None] * stride_ai + offs_k[None, :] * stride_aj),
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K × N) – B is already transposed
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bi + offs_j[None, :] * stride_bj),
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Dot‑product + accumulate
        acc += tl.dot(a, b)

    # Write back (cast to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj),
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data :
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask_tensor  : [B, N, N]       (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] FP32
    x = x_norm_fp32.to(torch.float16)   # FP16 for the heavy lifts

    # -----------------------------------------------------------------
    # 2️⃣ Single concatenated linear projection for all proj/gate weights
    # -----------------------------------------------------------------
    # Weight order: left_proj, right_proj, left_gate, right_gate, out_gate
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5*hidden_dim, dim)

    # One huge matmul: (B*N*N, dim) @ (dim, 5*hidden_dim) → (B,N,N,5*hidden_dim)
    combined = torch.nn.functional.linear(x, concat_weight)

    # -----------------------------------------------------------------
    # 3️⃣ Split and apply sigmoid to gate tensors
    # -----------------------------------------------------------------
    H = hidden_dim
    left          = combined[..., 0 * H : 1 * H]   # proj left
    right         = combined[..., 1 * H : 2 * H]   # proj right
    left_gate_raw = combined[..., 2 * H : 3 * H]   # left gate pre‑sigmoid
    right_gate_raw= combined[..., 3 * H : 4 * H]   # right gate pre‑sigmoid
    out_gate_raw  = combined[..., 4 * H : 5 * H]   # out gate  pre‑sigmoid

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 4️⃣ Fuse mask (if present) with gating
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → fp16 flattened
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                       # [B*N*N]
        total_elements = left.numel()                        # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # (left_gate/right_gate no longer needed)
    else:
        # Simple element‑wise multiplication (no mask)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ O(N³) contraction via custom batched GEMM (FP16 inputs)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension to the batch axis for easy tiling
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Collapse batch & hidden dimension → (B*H, N, N)
    left_mat  = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Transpose right for A @ Bᵀ
    right_T = right_mat.transpose(1, 2).contiguous()

    # Output placeholder
    out_mat = torch.empty_like(left_mat)

    # Strides (int64) needed by the Triton kernel
    stride_az = left_mat.stride(0)   # batch stride for A
    stride_ai = left_mat.stride(1)   # row stride (i) for A
    stride_aj = left_mat.stride(2)   # col stride (k) for A

    stride_bz = right_T.stride(0)    # batch stride for B_T
    stride_bi = right_T.stride(1)    # row stride (k) for B_T
    stride_bj = right_T.stride(2)    # col stride (j) for B_T

    stride_cz = out_mat.stride(0)    # batch stride for C
    stride_ci = out_mat.stride(1)    # row stride (i) for C
    stride_cj = out_mat.stride(2)    # col stride (j) for C

    # Tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # batch dimension (bh)
        (N + BLOCK_M - 1) // BLOCK_M,           # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,           # tiles along N (j)
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_T,
        out_mat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_az,
        stride_ai,
        stride_aj,
        stride_bz,
        stride_bi,
        stride_bj,
        stride_cz,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply out_gate and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate                              # element‑wise gating (FP16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
