"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input tensor and cast it to FP16.
2️⃣  Four linear projections (left / right) and three gating projections
    (left_gate, right_gate, out_gate) – all in FP16.
3️⃣  If a mask is supplied, fuse mask‑multiplication with the gating in a tiny
    Triton kernel (in‑place).  When there is no mask the gates are applied
    directly with an in‑place multiplication.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is expressed as a single `torch.einsum`.  This avoids the costly
    permute+contiguous copies required by a manual batched GEMM while still
    leveraging cuBLAS/Tensor‑cores.
5️⃣  A second LayerNorm (FP16) over the hidden dimension, element‑wise
    multiplication by the out‑gate, and a final linear projection back to the
    original model dimension.
6️⃣  Cast the result back to float32 (the harness expects FP32).

The implementation uses at least one Triton kernel (mask‑gate fusion) and
keeps the full forward pass in mixed‑precision FP16 for best memory‑bandwidth
and Tensor‑core utilisation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for `left` and `right`.
#   left  ← left  * mask * left_gate
#   right ← right * mask * right_gate
# All tensors are contiguous FP16 with shape [B, N, N, H] (left/right/gates)
# and mask has shape [B, N, N] (broadcast along H).
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (in‑place)
    right_ptr,         # *fp16  [B,N,N,H]  (in‑place)
    left_gate_ptr,     # *fp16  [B,N,N,H]
    right_gate_ptr,    # *fp16  [B,N,N,H]
    mask_ptr,          # *fp16  [B,N,N]    (broadcasted)
    total_elements,    # int   total number of elements in a tensor
    H,                 # int   hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat indices
    mask = offs < total_elements                                # OOB guard

    # Each element belongs to a (b,i,k) coordinate; the mask index is the
    # flat index divided by the hidden dimension H (broadcast over H).
    mask_offs = offs // H

    # Load tensors (out‑of‑bounds loads return the “other” value)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # Load mask (float16 0.0/1.0).  For OOB we use neutral value 1.0.
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fuse mask and gates
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # Store results back in‑place
    tl.store(left_ptr  + offs, left_out , mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradient computation).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack and cast inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    fp16 = torch.float16

    # ------------------------------------------------------------------
    # Cast model weights to fp16 once (they are stored as float32)
    # ------------------------------------------------------------------
    norm_w   = weights["norm.weight"].to(device=device, dtype=fp16)
    norm_b   = weights["norm.bias"  ].to(device=device, dtype=fp16)

    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=fp16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=fp16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=fp16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=fp16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=fp16)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device=device, dtype=fp16)

    to_out_w      = weights["to_out.weight"].to(device=device, dtype=fp16)

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm – keep everything in fp16 to save bandwidth
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(fp16)                     # [B,N,N,dim] FP16
    x = F.layer_norm(
        x_fp16,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )                                                   # still FP16

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣  Gating projections (fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4️⃣  Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if nomask:
        # In‑place multiplication avoids an extra allocation
        left.mul_(left_gate)
        right.mul_(right_gate)
    else:
        # Prepare mask as FP16 (0/1) – broadcast over the hidden dimension
        mask_fp16 = mask_tensor.to(fp16)          # [B,N,N]
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()
        BLOCK = 1024  # ~1 K threads per block – good occupancy on H100
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fusion_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            total_elements=total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates are no longer needed
        del left_gate, right_gate, mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣  O(N³) contraction (batched GEMM) via a single einsum.
    #     Einsum automatically reduces to a strided‑batched matmul and
    #     avoids the explicit permute/contiguous copies used previously.
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]  <--  Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('bikd,bjkd->bijd', left, right)

    # Free tensors we no longer need to keep the memory budget low.
    del left, right, left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm (FP16) over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                                    # still FP16

    # ------------------------------------------------------------------
    # 7️⃣  Apply out‑gate (in‑place) and final linear projection
    # ------------------------------------------------------------------
    out.mul_(out_gate)                      # element‑wise gate
    out = F.linear(out, to_out_w)           # → [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣  Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
