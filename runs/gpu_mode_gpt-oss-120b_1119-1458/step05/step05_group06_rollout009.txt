"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

The implementation fuses the two linear projections (proj + gate) for the
“left” and “right” branches, applies the optional binary mask, performs a
batched matrix‑multiply C = A @ Aᵀ for each hidden channel using a custom
Triton kernel (fp16 math, fp32 accumulation), runs a second LayerNorm,
applies the output gate and final linear projection, and finally returns
the result in float32.  All heavy work (the O(N³) contraction) is executed
inside the Triton kernel; all other ops use PyTorch FP16 kernels for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel for C = A @ Bᵀ  (batched GEMM, fp16 → fp16, fp32 accumulation)
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile over output rows (i)
    pid_n = tl.program_id(1)   # tile over output cols (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load a block of A (shape BLOCK_M x BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am               # batch*hidden offset
            + offs_i[:, None] * stride_ai     # i offset
            + offs_k[None, :] * stride_ak     # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load a block of Bᵀ (shape BLOCK_K x BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm               # batch*hidden offset
            + offs_k[:, None] * stride_bk     # k offset first
            + offs_j[None, :] * stride_bj     # j offset second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot product (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write out the result
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` (both [B, N, N, H]) using a Triton GEMM.
    Returns a tensor of shape [B, N, N, H] in fp16.
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)   # fp16 output

    # strides in elements (not bytes) for contiguous tensors
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # tile size – 64×64 works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,          # M (unused inside kernel but kept for signature)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : mapping name → torch.Tensor (float32, already on device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to FP16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ merge linear + gate weights (FP16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # concatenate proj and gate weights (shape: [2*hidden_dim, dim])
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)

    out_gate_w = out_gate_w.to(torch.float16)
    to_out_norm_w = to_out_norm_w.to(torch.float16)
    to_out_norm_b = to_out_norm_b.to(torch.float16)
    to_out_w = to_out_w.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ fused proj+gate for left and right (FP16)
    # ------------------------------------------------------------------
    # left branch: [B, N, N, 2*H] → split into proj & gate
    left_concat = F.linear(x, left_concat_w)          # [B, N, N, 2*H]
    left_proj = left_concat[..., :hidden_dim]         # [B, N, N, H]
    left_gate_raw = left_concat[..., hidden_dim:]     # [B, N, N, H]

    # right branch: same pattern
    right_concat = F.linear(x, right_concat_w)
    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 6️⃣ optional mask (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; convert to FP16 and add trailing dim.
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ gating (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # fuse projection with gating
    left = left_proj * left_gate       # [B, N, N, H]
    right = right_proj * right_gate    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ heavy contraction (batched GEMM C = A @ Bᵀ)
    # ------------------------------------------------------------------
    contracted = _batched_matmul_triton(left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (still in FP16) → out‑gate → final projection
    # ------------------------------------------------------------------
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H] fp16

    # out‑gate (computed from original normalized x)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, H] fp16
    contracted = contracted * out_gate

    # final linear projection back to dim
    out = F.linear(contracted, to_out_w)                # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
