def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    # 1. LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)
    x_norm_fp16 = x_norm.to(torch.float16)

    # 2. fuse left proj+gate
    left_concat = torch.nn.functional.linear(x_norm_fp16,
                                           torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16))
    # left_concat shape [B,N,N,2H]
    left_proj, left_gate_pre = left_concat.split(hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
    else:
        left = left_proj * left_gate

    # similarly right
    right_concat = torch.nn.functional.linear(x_norm_fp16,
                                          torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16))
    right_proj, right_gate_pre = right_concat.split(hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)
    if not nomask:
        right = right_proj * right_gate * mask_fp16
    else:
        right = right_proj * right_gate

    # 3. contraction: out_hidden = batched matmul (B*H, N, N)
    out = _batched_matmul_torch(left, right)

    # 4. second LN
    out = torch.nn.functional.layer_norm(out,
                                        (hidden_dim,),
                                        weight=to_out_norm_weight,
                                        bias=to_out_norm_bias)

    # 5. out_gate
    # We need out_gate (sigmoid of linear transformation from x_norm): compute earlier maybe.

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H,N,N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    # compute pairwise multiplication and sum across k dimension: left @ right^T
    out = torch.bmm(left_flat, right_flat.transpose(1,2))
    # reshape to original shape
    out = out.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1st LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias) # [B,N,N,dim] float32

    # Save copy for out_gate computing after gating.
    # out_gate to be computed later using same x.

    # Cast to fp16
    x_fp16 = x.to(torch.float16).contiguous()

    # Prepare concatenated weights for left and right projections
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)

    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)

    # Linear projections (left+gate, right+gate)
    left_concat = torch.nn.functional.linear(x_fp16, left_cat_w)   # shape [B,N,N,2*H]
    right_concat = torch.nn.functional.linear(x_fp16, right_cat_w)

    # Split and make contiguous
    left_proj = left_concat[..., :hidden_dim].contiguous()
    left_gate_pre = left_concat[..., hidden_dim:].contiguous()
    del left_concat  # free memory

    right_proj = right_concat[..., :hidden_dim].contiguous()
    right_gate_pre = right_concat[..., hidden_dim:].contiguous()
    del right_concat

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    # free pre gate storage
    del left_gate_pre, right_gate_pre

    # Apply mask and gating
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)
        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Now left_proj and right_proj hold final values
        left = left_proj
        right = right_proj
    else:
        # No mask
        left = left_proj * left_gate
        right = right_proj * right_gate

    # free gate tensors after use
    del left_proj, left_gate, right_proj, right_gate

    # Batched matrix multiplication (O(N^3) contraction)
    out = _batched_matmul_torch(left, right)   # shape [B,N,N,H] fp16
    del left, right

    # Second LayerNorm (over hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Compute out_gate (sigmoid of linear on x after first norm) in fp32 for accuracy
    out_gate_fp32 = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))  # float32
    out_gate = out_gate_fp32.to(torch.float16)  # cast for multiplication

    # Apply out gate
    out = out * out_gate
    del out_gate

    # Final projection back to dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)    # [B,N,N,dim] fp16

    # Cast to float32 for final output
    return out.to(torch.float32)

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

The kernel fuses the mask, gate and linear projections for efficiency,
uses a batched GEMM for the O(N³) contraction, and runs the remaining
operations in half precision. The final result is returned as fp32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel for fused multiplication with broadcasting mask.
# It computes: out = x * gate * mask where `x` and `gate` have shape
# (B, N, N, H) (flattened) and `mask` has shape (B, N, N) broadcast
# over the hidden dimension H.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16: tensor to be updated (left or right)
    gate_ptr,       # *fp16: gate tensor (same shape as x)
    mask_ptr,       # *fp16: mask tensor (shape B·N·N)
    hidden_dim,     # i32: hidden dimension H
    size,           # i32: total number of elements in x (B·N·N·H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear indices of this block
    active = offs < size

    # Index in mask = floor(offs / H) because mask is broadcasted over hidden dim
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)          # fp16
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)          # fp16
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)    # fp16 (1 = no mask)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication implementing the core O(N³) contraction:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    Both `left` and `right` are tensors of shape [B, N, N, H] (fp16).
    The function returns a tensor of the same shape.
    """
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) where the hidden channel becomes a batch dimension
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Matrix multiply left * rightᵀ for each batch entry
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32/bool, may be ignored)
        - weights : dict name → torch.Tensor (float32)
        - config  : contains ``dim`` and ``hidden_dim``; may also contain ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)   # [B,N,N,dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation.
    # -----------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16).contiguous()

    # -----------------------------------------------------------------
    # 3️⃣ Fuse left/right projections & their gates (two linear calls instead of four).
    # -----------------------------------------------------------------
    # Concatenate projection and gate weights for the “left” side.
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)   # (2*H, dim)

    # Same for the “right” side.
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # (2*H, dim)

    # Linear projection produces both the raw values and the gate pre‑activations.
    left_concat  = F.linear(x_fp16, left_cat_w)   # [B,N,N,2*H]
    right_concat = F.linear(x_fp16, right_cat_w)  # [B,N,N,2*H]

    # Split and make contiguous (required for the Triton kernel).
    left_proj = left_concat[..., :hidden_dim].contiguous()
    left_gate_pre = left_concat[..., hidden_dim:].contiguous()
    del left_concat

    right_proj = right_concat[..., :hidden_dim].contiguous()
    right_gate_pre = right_concat[..., hidden_dim:].contiguous()
    del right_concat

    # Sigmoid on gate pre‑activations.
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    del left_gate_pre, right_gate_pre

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – fused elementwise multiplication using Triton.
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to fp16 and make it contiguous.
        mask_fp16 = mask.to(torch.float16).contiguous()
        total = left_proj.numel()
        BLOCK = 1024                     # ≤ 1024 threads per block
        grid  = ((total + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left  = left_proj
        right = right_proj
    else:
        # No mask – simple elementwise multiply.
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Release intermediate tensors early.
    del left_proj, left_gate, right_proj, right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑Core).
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H] fp16
    del left, right

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16) over the hidden dimension.
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # -----------------------------------------------------------------
    # 7️⃣ Out‑gate (computed in float32 for accuracy, then cast).
    # -----------------------------------------------------------------
    out_gate_fp32 = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # float32
    out_gate = out_gate_fp32.to(torch.float16)

    # Apply gating.
    out = out * out_gate
    del out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimension.
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness) and return.
    # -----------------------------------------------------------------
    return out.to(torch.float32)

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

The kernel fuses mask‑gate multiplication, uses a batched GEMM for the
O(N³) pairwise contraction, and runs the bulk of the computation in
half‑precision. Only the forward pass is implemented (no autograd).

Algorithm
---------
1️⃣ Layer‑norm the raw input (float32) → `x_norm`.
2️⃣ Cast `x_norm` to fp16.
3️⃣ Fuse the left/right linear projections with their gates by concatenating
   the two weight matrices and applying a single `F.linear` per side.
4️⃣ Apply sigmoid to the gate pre‑activations.
5️⃣ If a mask is present, a tiny Triton kernel performs the in‑place
   multiplication `proj * gate * mask` (broadcast over the hidden dim).
   When no mask is provided we fall back to a simple element‑wise mul.
6️⃣ Perform the heavy contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM (`torch.bmm`) on tensors reshaped to
   [B·H, N, N].
7️⃣ Second LayerNorm (fp16) over the hidden dimension.
8️⃣ Compute the out‑gate in float32 for numerical stability,
   cast it to fp16 and multiply element‑wise.
9️⃣ Final linear layer back to `dim`.
🔟 Cast the result to float32 and return.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel for fused elementwise multiplication with broadcasting mask.
# The tensors `x` and `gate` have shape (B, N, N, H) flattened to a 1‑D
# view, while `mask` has shape (B, N, N) and a single entry is reused
# for the whole hidden slice.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : projection tensor (to be overwritten)
    gate_ptr,       # *fp16 : gate tensor (same shape as `x`)
    mask_ptr,       # *fp16 : mask tensor (shape B·N·N)
    hidden_dim,     # i32   : hidden size H
    size,           # i32   : total number of elements in `x` (B·N·N·H)
    BLOCK: tl.constexpr,          # compile‑time block size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear indices of this block
    active = offs < size                       # boundary guard

    # The mask is broadcast over H, therefore each block of `hidden_dim`
    # contiguous entries shares the same mask index.
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)   # fp16
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)   # fp16
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)  # fp16 (1 = no mask)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Implements the core O(N³) contraction via a batched GEMM.
    `left`/`right`: [B, N, N, H] (fp16).
    Returns:         [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # contract over the third axis (k) → matrix multiplication with transpose
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32/bool, may be ignored)
        - weights      : dict name → torch.Tensor (float32)
        - config       : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy work
    # -----------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16).contiguous()

    # -----------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left and right sides
    # -----------------------------------------------------------------
    # Concatenate projection and gate weights → (2*H, dim)
    left_cat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_cat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # One linear call per side produces both the projection and its gate pre‑activation
    left_concat  = F.linear(x_fp16, left_cat_w)   # [B,N,N,2*H]
    right_concat = F.linear(x_fp16, right_cat_w)  # [B,N,N,2*H]

    # Split and make the sub‑tensors contiguous (required for the Triton kernel)
    left_proj = left_concat[..., :hidden_dim].contiguous()
    left_gate_pre = left_concat[..., hidden_dim:].contiguous()
    del left_concat

    right_proj = right_concat[..., :hidden_dim].contiguous()
    right_gate_pre = right_concat[..., hidden_dim:].contiguous()
    del right_concat

    # Sigmoid on gate pre‑activations
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    del left_gate_pre, right_gate_pre

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – fused multiplication using Triton
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).contiguous()
        total = left_proj.numel()           # B·N·N·H
        BLOCK = 1024                        # ≤ 1024 threads per block
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
    else:
        # No mask – plain elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release temporaries early
    del left_proj, left_gate, right_proj, right_gate

    # -----------------------------------------------------------------
    # 5️⃣ O(N³) contraction using a batched GEMM
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B,N,N,H] fp16
    del left, right

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # -----------------------------------------------------------------
    # 7️⃣ Out‑gate (computed in float32 for numerical stability)
    # -----------------------------------------------------------------
    out_gate_fp32 = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # float32
    out_gate = out_gate_fp32.to(torch.float16)

    # Apply out‑gate
    out = out * out_gate
    del out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast to float32 (required by the harness) and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
