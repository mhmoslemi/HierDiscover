"""
Triton‑accelerated **outgoing** TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑Norm on the input tensor (float32) – stable on the raw data.
2. Cast to fp16, apply the 5 linear layers (left/right projection &
   three gates).  The three gate‑projections are passed through a
   sigmoid.  The mask (if present) and the gates are fused in a tiny
   Triton kernel to avoid temporaries.
3. Heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a custom Triton GEMM kernel that works directly
   on the original layout `[B,N,N,H]`.  This removes the costly
   `permute → contiguous → view` dance used in a naïve torch
   implementation.
4. Layer‑Norm over the hidden dimension, gating and final projection
   back to the original channel dimension (all in fp16).
5. Cast the result back to float32 (required by the harness).

Only the forward pass is needed – no autograd logic.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1. Fuse mask, gate and elementwise multiplication
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16: tensor to be gated (left or right)
    gate_ptr,       # *fp16: corresponding gate tensor (same shape as x)
    mask_ptr,       # *fp16: broadcast mask, shape = B * N * N
    hidden_dim,     # i32: size of hidden dim H
    size,           # i32: total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,              # compile‑time tile size (<=1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)        # linear indices
    active = offs < size

    # The mask is broadcast over the hidden dimension.  One mask element
    # corresponds to BLOCK_H consecutive entries in the flattened x.
    mask_offs = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)  # 1 = keep

    tl.store(x_ptr + offs, x * g * m, mask=active)


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel that directly consumes tensors of shape
#    [B, N, N, H] (fp16) and writes out the contracted result of the
#    same shape.  The hidden dimension H is fused into the batch
#    dimension (B*H independent GEMMs).
# ----------------------------------------------------------------------
@triton.jit
def trmul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,            # *fp16 pointers
    B, N, H,                                 # i32 dimensions
    # stride arguments (in number of elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    with the hidden dimension folded into the batch axis.
    """
    pid_bh = tl.program_id(0)      # combined batch‑hidden index
    pid_i  = tl.program_id(1)      # tile row (i)
    pid_j  = tl.program_id(2)      # tile col (j)

    # --------------------------------------------------------------
    # decode batch `b` and hidden channel `h` from the combined id
    # --------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh %  H

    # base pointers that already contain the batch & hidden offsets
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # --------------------------------------------------------------
    # tile origin in the (i,j) output space
    # --------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # accumulator in fp32 (to keep precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension K (= N)
    # --------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)

        # ------------------------------------------------------------------
        # build index vectors (clamped later by masks)
        # ------------------------------------------------------------------
        i_idx = i_start + tl.arange(0, BLOCK_M)
        j_idx = j_start + tl.arange(0, BLOCK_N)
        k_idx = k_start + tl.arange(0, BLOCK_K)

        i_mask = i_idx < N
        j_mask = j_idx < N
        k_mask = k_idx < N

        # ------------------------------------------------------------------
        # load a block of `left`   -> shape (BLOCK_M, BLOCK_K)
        #   left[b,i,k,h]   (stride_i = N*H, stride_k = H)
        # ------------------------------------------------------------------
        a_ptrs = left_base + i_idx[:, None] * stride_li + k_idx[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask = i_mask[:, None] & k_mask[None, :],
                    other = 0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # load a block of `right` transposed -> shape (BLOCK_K, BLOCK_N)
        #   right[b,j,k,h]  (we need B_T[k,j] = right[b,j,k,h])
        #   stride_k = H, stride_j = N*H
        # ------------------------------------------------------------------
        b_ptrs = right_base + k_idx[:, None] * stride_rk + j_idx[None, :] * stride_ri
        b = tl.load(b_ptrs,
                    mask = k_mask[:, None] & j_mask[None, :],
                    other = 0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # matrix multiplication of the two tiles
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)                     # (M,K)·(K,N) → (M,N)

    # ----------------------------------------------------------------------
    # write back the computed tile
    # ----------------------------------------------------------------------
    i_mask = i_start + tl.arange(0, BLOCK_M) < N
    j_mask = j_start + tl.arange(0, BLOCK_N) < N
    c_ptrs = out_base + (i_start + tl.arange(0, BLOCK_M))[:, None] * stride_oi + \
                        (j_start + tl.arange(0, BLOCK_N))[None, :] * stride_oj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask = i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Helper that wraps the Triton GEMM kernel
# ----------------------------------------------------------------------
def _trmul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using the Triton kernel.
    left/right: fp16 tensors of shape [B, N, N, H]
    Returns a fp16 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16)

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                      # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    trmul_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool, may be None)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : ``dim`` and ``hidden_dim`` required; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1️⃣ First LayerNorm (float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # 2️⃣ Cast to fp16 for the heavy work
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5️⃣ Compute gates (sigmoid) – still fp16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6️⃣ Optional mask – fused element‑wise multiplication (Triton)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left.numel()               # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – fused batched GEMM (Triton)
    # --------------------------------------------------------------
    out = _trmul_gemm(left, right)        # [B,N,N,hidden_dim] fp16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden_dim) – fp16
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Gating & final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 🔟 Cast back to float32 for the downstream code
    # --------------------------------------------------------------
    return out.to(torch.float32)
