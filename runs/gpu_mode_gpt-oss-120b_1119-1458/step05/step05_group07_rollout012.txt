"""
Fast outgoing TriMul implementation (AlphaFold‑3) – fused linear projections.

Algorithm
---------
1️⃣  First LayerNorm (float32) → cast to float16.
2️⃣  Pack the five linear layers (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single weight matrix of shape
    (5·hidden_dim, dim).  One large half‑precision GEMM produces all
    intermediate tensors at once, halving the number of times the large
    input tensor is streamed from memory.
3️⃣  Split the GEMM output, apply σ (sigmoid) to the three gate tensors.
4️⃣  Fuse element‑wise multiplication (proj·gate·mask) for *left* and
    *right* using a tiny Triton kernel (mask broadcast over the hidden
    dimension).  When ``nomask=True`` a dummy all‑ones mask is used.
5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM ``C = A @ Bᵀ`` (cuBLAS/Tensor‑Core).
6️⃣  Second LayerNorm (fp16) → multiply by ``out_gate``.
7️⃣  Final linear projection back to the original feature dimension.
8️⃣  Cast the result to float32 (required by the harness).

Only the forward pass is implemented – no autograd graphs are built.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) in‑place.
# --------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # *fp16 : projection tensor, size = total_elts
    gate_ptr,          # *fp16 : sigmoid‑gated tensor, same shape as proj
    mask_ptr,          # *fp16 : mask tensor (no hidden dim), size = B·N·N
    total_elts: tl.int32,   # B·N·N·hidden_dim
    hidden_dim: tl.int32,   # hidden dimension
    BLOCK: tl.constexpr,    # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elts

    # which mask entry does this offset correspond to?
    mask_idx = offs // hidden_dim                      # broadcast over hidden dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(proj_ptr + offs, proj * gate * m, mask=mask)


# --------------------------------------------------------------
# Batched GEMM (FP16) for the O(N³) contraction.
# --------------------------------------------------------------
def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a batched half‑precision GEMM.

    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H] (fp16), already multiplied by gates/mask.

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) then use torch.bmm
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))               # (B*H, N, N)
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# --------------------------------------------------------------
# Entry point used by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                          optional ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to half‑precision
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                               # [B,N,N,dim]  float32
    x = x_norm.to(torch.float16)                    # -> fp16 for the heavy part

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the 5 linear layers (project & gate) into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate the weight matrices along the out‑feature dimension.
    # All weight tensors are stored as (out_features, in_features) in PyTorch.
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # shape (5*hidden_dim, dim)

    # Perform the single large linear projection.
    x_flat = x.reshape(-1, dim)                    # (B*N*N, dim)
    proj_all = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*H)
    proj_all = proj_all.view(B, N, N, 5 * hidden_dim)               # (B,N,N,5*H)

    # ------------------------------------------------------------------
    # 3️⃣ Split and apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = torch.split(
        proj_all, hidden_dim, dim=-1
    )  # each: (B,N,N,H)

    left_gate   = torch.sigmoid(left_gate_raw)
    right_gate  = torch.sigmoid(right_gate_raw)
    out_gate    = torch.sigmoid(out_gate_raw)     # will be used after 2nd LN

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask and gate (proj * gate * mask) via Triton kernel
    # ------------------------------------------------------------------
    # mask broadcasted over the hidden dimension
    if nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        # mask may be bool or float; ensure fp16 values (1.0 for True, 0.0 for False)
        mask_fp16 = mask_tensor.to(torch.float16)

    # flatten for kernel (linear indexing)
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_flat = mask_fp16.contiguous().view(-1)   # (B*N*N,)

    total_elts = left_proj.numel()                # B * N * N * H
    BLOCK = 1024                                   # threads per block (≤1024)
    grid = ((total_elts + BLOCK - 1) // BLOCK,)

    # left = left_proj * left_gate * mask
    apply_gate_mask_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        mask_flat.data_ptr(),
        total_elts,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right = right_proj * right_gate * mask
    apply_gate_mask_kernel[grid](
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_flat.data_ptr(),
        total_elts,
        hidden_dim,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # After the kernel, left_proj/right_proj already contain the final tensors
    left  = left_proj   # shape [B,N,N,H] (fp16)
    right = right_proj  # shape [B,N,N,H] (fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)       # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )                                            # still fp16

    # ------------------------------------------------------------------
    # 7️⃣ Final gating (out_gate) and projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                           # broadcast (B,N,N,H)
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)       # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
