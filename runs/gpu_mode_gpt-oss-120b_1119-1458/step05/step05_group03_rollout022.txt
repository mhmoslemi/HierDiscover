@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)
    b = bh // hidden_dim
    h = bh % hidden_dim

    # Compute base offsets for this batch head
    batch_offset_left = b * stride_left_batch + h * stride_left_h
    batch_offset_right = b * stride_right_batch + h * stride_right_h
    batch_offset_out = b * stride_out_batch + h * stride_out_h

    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    # Compute i and j offsets
    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    i_offs = i_start + tl.arange(0, BLOCK_M)
    j_offs = j_start + tl.arange(0, BLOCK_N)

    # Masks for boundary conditions
    mask_i = i_offs < N
    mask_j = j_offs < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        k_offs = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # Load A block (left)
        a_ptrs = left_ptr + batch_offset_left + i_offs[:, None] * stride_left_i + k_offs[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B block (right) transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + batch_offset_right + k_offs[:, None] * stride_right_k + j_offs[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product
        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # Store result
    c_ptrs = out_ptr + batch_offset_out + i_offs[:, None] * stride_out_i + j_offs[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – fully fused forward pass.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32, then cast to FP16.
2️⃣  Fuse the five linear transformations (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single matrix multiplication:
       [B,N,N,dim] → (B·N·N, dim) × (5·hidden_dim, dim)ᵀ
   The result is reshaped to [B,N,N,5·hidden_dim] and sliced.
3️⃣  Apply the three sigmoid gates.  If a mask is present we fuse
    `proj * gate * mask` with a tiny Triton kernel (`mask_gate_fuse_kernel`);
    otherwise a simple in‑place multiplication is used.
4️⃣  Heavy contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton batched‑GEMM kernel (`trimul_outgoing_kernel`);
   this avoids the costly permute+contiguous copy that the reference uses.
5️⃣  Second Layer‑norm (FP16) → out‑gate gating → final linear projection
    back to the original channel dimension.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

All heavy tensor‑core work runs in FP16 with FP32 accumulation.
A custom Triton kernel is used for both mask‑gate fusion and the O(N³) contraction.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: element‑wise (tensor * gate * mask) – mask is
# broadcast over the hidden dimension `hidden_dim`.
# --------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, size = total elements
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask vector of shape (B * N * N)
    size,           # i32   : total number of elements in x
    hidden_dim,     # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # load data
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # broadcast mask: every `hidden_dim` elements share the same mask entry
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # fused multiply‑accumulate
    out = x * g * m

    # store back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# --------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,        # *fp16 : left tensor  [B,N,N,H]
    right_ptr,       # *fp16 : right tensor [B,N,N,H]
    out_ptr,         # *fp16 : output tensor [B,N,N,H]
    batch_head,      # i32   : B * hidden_dim
    N,               # i32   : sequence length
    hidden_dim,      # i32   : hidden dimension
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 0️⃣ deduce batch (=b) and head (=h) from the flattened batch_head index
    # ------------------------------------------------------------------
    bh = tl.program_id(0)                     # 0 ≤ bh < B*H
    b = bh // hidden_dim                      # batch index
    h = bh % hidden_dim                       # head index

    # base offsets for this (b,h) slice
    base_left = b * stride_left_b + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out = b * stride_out_b + h * stride_out_h

    # ------------------------------------------------------------------
    # 1️⃣ block coordinates for the output matrix (i‑tile, j‑tile)
    # ------------------------------------------------------------------
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    i_offs = i_start + tl.arange(0, BLOCK_M)
    j_offs = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_offs < N
    mask_j = j_offs < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2️⃣ loop over reduction axis k
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        k_offs = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + base_left + i_offs[:, None] * stride_left_i + k_offs[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + base_right + k_offs[:, None] * stride_right_k + j_offs[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # 3️⃣ write result
    # ------------------------------------------------------------------
    c_ptrs = out_ptr + base_out + i_offs[:, None] * stride_out_i + j_offs[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Entrypoint for the evaluation harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`,
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]  # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 0️⃣ First LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_norm.to(dtype_fp16)  # [B, N, N, dim] in FP16

    # ------------------------------------------------------------------
    # 1️⃣ Fuse all five linear maps (proj + three gates) into one matmul
    # ------------------------------------------------------------------
    # flatten to (B·N·N, dim)
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # shape (M, dim) where M = B*N*N

    # concatenate weight matrices: each weight is (hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"].to(dtype_fp16)
    right_proj_w = weights["right_proj.weight"].to(dtype_fp16)
    left_gate_w = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_w = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w = weights["out_gate.weight"].to(dtype_fp16)

    combined_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,  # (5*hidden_dim, dim)
    ).contiguous()                               # ensure contiguous for cuBLAS

    # Linear: (M, dim) × (dim, 5*hidden_dim)ᵀ → (M, 5*hidden_dim)
    proj = torch.nn.functional.linear(x_flat, combined_weight)   # FP16, shape (M, 5*H)

    # reshape back to [B, N, N, 5*hidden_dim]
    proj = proj.view(B, N, N, hidden_dim * 5)

    # split
    h0 = hidden_dim
    left_proj   = proj[..., 0:h0]                     # [B,N,N,H]
    right_proj  = proj[..., h0:2 * h0]
    left_gate_r = proj[..., 2 * h0:3 * h0]
    right_gate_r = proj[..., 3 * h0:4 * h0]
    out_gate_r  = proj[..., 4 * h0:5 * h0]

    # --------------------------------------------------------------
    # 2️⃣ Apply sigmoid to gating tensors
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_r)
    right_gate = torch.sigmoid(right_gate_r)
    out_gate = torch.sigmoid(out_gate_r)

    # Free the raw gate tensors early
    del left_gate_r, right_gate_r, out_gate_r

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and combine projection with gate
    # --------------------------------------------------------------
    if nomask:
        # simple in‑place multiplication
        left = left_proj.mul_(left_gate)      # [B,N,N,H]
        right = right_proj.mul_(right_gate)
    else:
        # mask → FP16, flatten to (B*N*N,)
        mask_fp16 = mask_tensor.to(dtype_fp16).contiguous().view(-1)
        total_elems = left_proj.numel()        # B * N * N * H
        BLOCK = 1024
        grid_mask = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid_mask](
            left_proj,
            left_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid_mask](
            right_proj,
            right_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_proj
        right = right_proj
        # Gates are no longer needed
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy contraction using a Triton batched GEMM kernel
    # ------------------------------------------------------------------
    # pre‑allocate output tensor
    out = torch.empty_like(left)

    # stride information (elements, not bytes)
    st_left_b = left.stride(0)
    st_left_i = left.stride(1)
    st_left_k = left.stride(2)
    st_left_h = left.stride(3)

    st_right_b = right.stride(0)
    st_right_j = right.stride(1)   # j uses the same stride as i in the tensor layout
    st_right_k = right.stride(2)
    st_right_h = right.stride(3)

    st_out_b = out.stride(0)
    st_out_i = out.stride(1)
    st_out_j = out.stride(2)
    st_out_h = out.stride(3)

    batch_head = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_head,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        batch_head,
        N,
        hidden_dim,
        st_left_b,
        st_left_i,
        st_left_k,
        st_left_h,
        st_right_b,
        st_right_j,
        st_right_k,
        st_right_h,
        st_out_b,
        st_out_i,
        st_out_j,
        st_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # free the projected tensors early
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Out‑gate multiplication + final projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(dtype_fp16)
    out = torch.nn.functional.linear(out, to_out_w)   # -> [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # ------------------------------------------------------------------
    return out.to(dtype_fp32)
