```
def _batched_matmul_torch(left, right):
    B,N,_,H = left.shape
    # reshape to [B*H, N, N] -> [B*H, N, N] for left and right
    left_ = left.permute(0,3,1,2).reshape(B*H,N,N)
    right_ = right.permute(0,3,1,2).reshape(B*H,N,N)
    out_ = torch.bmm(left_, right_.transpose(1,2))

"""
Outgoing TriMul (AlphaFold‑3) – fast forward implementation.

Key ideas
---------
1️⃣  Layer‑norm the input in FP32 (numerical stability) and cast to FP16.
2️⃣  The five linear layers (left/right projections + three gates) share the same
    input tensor.  By concatenating their weight matrices we can compute all
    five outputs with a *single* half‑precision GEMM, i.e. we replace five
    matrix‑multiplies by one.
3️⃣  Gate tensors are turned into sigmoids in‑place (`.sigmoid_()`), avoiding
    extra allocations.
4️⃣  If a mask is present we fuse the element‑wise product
    `tensor * gate * mask` with a tiny Triton kernel that updates the tensor
    in‑place (one kernel launch per side).  When there is no mask we simply
    multiply the tensor by the gate in‑place.
5️⃣  The heavy O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched half‑precision GEMM (`torch.bmm`), which on an H100
    maps to a Tensor‑Core kernel.
6️⃣  A second LayerNorm (FP16) is applied, followed by element‑wise multiplication
    with the `out_gate`, a final linear projection back to the original feature
    dimension and a cast to FP32.

The function `custom_kernel` follows the exact signature required by the
evaluation harness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused (tensor * gate * mask) in‑place.
# The mask has shape (B, N, N) and is broadcast across the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    x_ptr,                 # *fp16 : tensor to be updated, size = total_elts
    gate_ptr,              # *fp16 : gate tensor, same shape as x
    mask_ptr,              # *fp16 : mask tensor (no hidden dim)
    total_elts: tl.int32,  # total number of elements in x/gate
    hidden_dim: tl.int32,  # hidden dimension (size of last axis of x)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear indices
    mask = offs < total_elts                     # validity mask

    # Every `hidden_dim` elements belong to the same mask entry.
    mask_idx = offs // hidden_dim

    x   = tl.load(x_ptr + offs,    mask=mask, other=0.0)
    g   = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # In‑place update: x ← x * g * m
    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
# out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]  (fp16)
    returns out  : [B, N, N, H]  (fp16)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) – each hidden channel becomes a separate batch element
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) → to FP16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  FP32

    x = x_norm.to(torch.float16)  # half‑precision for the heavy part

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single GEMM
    # --------------------------------------------------------------
    # Cast all five weight matrices to FP16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature axis (dim 0)
    # Shape: [5*hidden_dim, dim]
    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )

    # One fused linear pass – result shape: [B, N, N, 5*hidden_dim]
    proj = torch.nn.functional.linear(x, combined_w)  # FP16

    hl = hidden_dim
    # Views into the concatenated result
    left       = proj[..., 0:hl]                # [B,N,N,hl]
    right      = proj[..., hl:2 * hl]           # [B,N,N,hl]
    left_gate  = proj[..., 2 * hl:3 * hl]       # will be sigmoided in‑place
    right_gate = proj[..., 3 * hl:4 * hl]       # will be sigmoided in‑place
    out_gate   = proj[..., 4 * hl:5 * hl]       # will be sigmoided in‑place

    # In‑place sigmoid for the three gate tensors (no extra memory)
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and gate multiplication – in‑place
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elts = left.numel()                       # B * N * N * hl
        BLOCK = 1024
        grid = ((total_elts + BLOCK - 1) // BLOCK,)

        # left  ← left  * left_gate  * mask
        fused_gate_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right ← right * right_gate * mask
        fused_gate_mask_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elts,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple in‑place multiplication by the gates
        left.mul_(left_gate)
        right.mul_(right_gate)

    # --------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via batched GEMM
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hl]  FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP16) + out‑gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Element‑wise gating with out_gate (in‑place)
    out.mul_(out_gate)

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
