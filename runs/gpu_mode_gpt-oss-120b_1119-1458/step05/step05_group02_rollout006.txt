def gated_mask_kernel(

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def gated_mask_kernel(
    proj_ptr, gate_raw_ptr, mask_ptr,
    hidden_dim,
    size,
    BLOCK: tl.constexpr
):
    """
    In-place fused kernel that applies sigmoid(gate), optional mask,
    and multiplies projection by the result.
    proj_ptr   : fp16 * (projected tensor)
    gate_raw_ptr : fp16 * (gate tensor before sigmoid)
    mask_ptr   : fp16 * (mask broadcasted over hidden_dim)
    hidden_dim : integer, stride of hidden dimension.
    size       : total number of elements in proj_ptr ( = B*N*N*hidden_dim )
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < size
    # mask index: floor division by hidden_dim (since mask broadcasted across hidden dim)
    mask_offs = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_raw_ptr + offs, mask=active, other=0.0)
    mask = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)
    # sigmoid(gate)
    gate = 1.0 / (1.0 + tl.exp(-gate))
    out = proj * gate * mask
    tl.store(proj_ptr + offs, out, mask=active)

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction as a batched GEMM.
    left/right: [B,N,N,H] (fp16)
    Returns:    [B,N,N,H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each hidden channel becomes a separate batch
    # we want out = left @ right^T per hidden channel
    left_re = left.permute(0, 3, 1, 2).reshape(B*H, N, N)
    right_re = right.permute(0, 3, 1, 2).reshape(B*H, N, N)
    # bmm expects [batch, N, K] * [batch, K, M] -> [batch, N, M]
    out = torch.bmm(left_re, right_re.transpose(1, 2))
    # restore [B,N,N,H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    This implementation reduces memory traffic by:
      * Merging all linear transforms that share the same input into a single
        matrix multiplication (producing left_proj, left_gate, right_proj,
        right_gate and out_gate in one pass).
      * Using a Triton kernel to fuse the sigmoid gating, optional mask
        application and the element‑wise multiplication for the left/right
        tensors.
      * Performing the O(N³) contraction with a single batched GEMM (torch.bmm)
        on the hidden dimension.
      * Keeping the first layer‑norm in float32 for numerical stability and
        the remaining ops in fp16 for speed.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool) or None
        - weights : dict of torch.Tensor (float32)
        - config  : dict with keys 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for speed
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Combine all linear transforms that use the same input
    #    Shape of each weight: (out_features, in_features) = (hidden_dim, dim)
    #    Combined weight shape: (5*hidden_dim, dim)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*H, dim) in fp16

    # -----------------------------------------------------------------
    # 4️⃣ Single linear projection produces all intermediate tensors
    # -----------------------------------------------------------------
    # torch.nn.functional.linear works on the last dimension
    Y = torch.nn.functional.linear(x, combined_weight)  # [B, N, N, 5*H] fp16

    # Slice Y into the five needed tensors
    H = hidden_dim
    left_proj = Y[..., :H]                     # [B,N,N,H]
    left_gate_raw = Y[..., H:2*H]              # [B,N,N,H]
    right_proj = Y[..., 2*H:3*H]               # [B,N,N,H]
    right_gate_raw = Y[..., 3*H:4*H]           # [B,N,N,H]
    out_gate_raw = Y[..., 4*H:5*H]             # [B,N,N,H]

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask + sigmoid gating via a Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and contiguous.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # flatten to shape [B*N*N]
        total = left_proj.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        # left side
        gated_mask_kernel[grid](
            left_proj,
            left_gate_raw,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right side
        gated_mask_kernel[grid](
            right_proj,
            right_gate_raw,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just apply sigmoid to the gate tensors
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # If mask present, the kernel already applied sigmoid and mask, so left_proj/right_proj are ready.

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (batched GEMM)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Apply out_gate (computed from the saved raw tensor)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)  # [B,N,N,H] fp16
    out = out * out_gate

    # -----------------------------------------------------------------
    # 9️⃣ Final linear projection back to original dim
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for downstream usage
    # -----------------------------------------------------------------
    return out.to(torch.float32)

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ First LayerNorm (float32) on the raw 4‑D tensor.
2️⃣ Cast to half‑precision and concatenate *all* linear maps that share the
   same input (left_proj, left_gate, right_proj, right_gate, out_gate)
   into a single matmul:  Y = x @ W_all   →   [B,N,N,5·H] (fp16).
3️⃣ Slice Y to obtain the five required tensors.
4️⃣ If a mask is present, a Triton kernel fuses the sigmoid of the gate,
   the optional mask multiplication and the elementwise product
   `proj = proj * sigmoid(gate) * mask` for both left and right tensors,
   overwriting the projection tensors in‑place.
   When there is no mask we simply apply the sigmoid and elementwise
   multiplication in PyTorch.
5️⃣ Heavy O(N³) contraction ‑‑ the “TriMul” – is performed as a batched GEMM:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   →   torch.bmm on tensors
   reshaped to shape [B·H, N, N].
6️⃣ Second LayerNorm (fp16) over the hidden dimension.
7️⃣ Apply the out‑gate (sigmoid of the pre‑computed slice) and a final
   linear projection back to the original dimension.
8️⃣ Cast the result back to float32 for the harness.

Only the forward pass is required; gradients are omitted.
The Triton kernel is used for the mask‑gate‑fusion step, satisfying the
requirement that at least part of the computation be performed in Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel that fuses: sigmoid(gate) → *mask → *proj (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_kernel(
    proj_ptr,          # *fp16: projected tensor (left or right)
    gate_raw_ptr,      # *fp16: raw gate tensor (pre‑sigmoid)
    mask_ptr,          # *fp16: flattened mask [B·N·N]
    hidden_dim: tl.constexpr,   # int32: size of the hidden dimension (H)
    size: tl.constexpr,         # int32: total #elements in proj (B·N·N·H)
    BLOCK: tl.constexpr,        # tile size (e.g. 1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear offsets of the block
    active = offs < size

    # mask is broadcast on the hidden dim → each hidden element uses the same mask entry
    mask_offs = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate_raw = tl.load(gate_raw_ptr + offs, mask=active, other=0.0)
    mask = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # sigmoid(gate_raw)
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    out = proj * gate * mask
    tl.store(proj_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction as a batched matrix multiplication.
    left, right: [B, N, N, H] (fp16)          ->   out: [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # combine batch and hidden dimensions
    left_bh = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # [B·H, N, N]
    right_bh = right.permute(0, 3, 1, 2).reshape(B * H, N, N)    # [B·H, N, N]
    # GEMM for each hidden slice: left @ rightᵀ
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))        # [B·H, N, N]
    # restore the original layout
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the bulk of the computation
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Pack all linear transformations that share the same input into one matmul
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)      # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)     # (H, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)      # (H, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)     # (H, dim)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)       # (H, dim)

    # concatenate along the output‑feature axis → (5·H, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*H, dim)

    # single linear projection; result shape [B, N, N, 5·H] (fp16)
    Y = torch.nn.functional.linear(x, combined_weight)

    H = hidden_dim
    left_proj   = Y[..., :H]           # [B, N, N, H]
    left_gate_r = Y[..., H:2*H]        # raw gate (pre‑sigmoid)
    right_proj  = Y[..., 2*H:3*H]      # [B, N, N, H]
    right_gate_r= Y[..., 3*H:4*H]      # raw gate (pre‑sigmoid)
    out_gate_r  = Y[..., 4*H:5*H]      # raw out‑gate (pre‑sigmoid)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask + sigmoid gating (fused in Triton when mask present)
    # -----------------------------------------------------------------
    if not nomask:
        # ensure mask is fp16 and flatten to [B·N·N]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)

        total = left_proj.numel()           # B·N·N·H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left side: proj ← proj * sigmoid(gate) * mask  (in‑place)
        gated_mask_kernel[grid](
            left_proj,
            left_gate_r,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right side
        gated_mask_kernel[grid](
            right_proj,
            right_gate_r,
            mask_flat,
            hidden_dim,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → apply sigmoid and multiply normally
        left_gate = torch.sigmoid(left_gate_r)
        right_gate = torch.sigmoid(right_gate_r)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via a batched GEMM
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (still fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply out‑gate (sigmoid of the pre‑computed slice)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_r)          # [B, N, N, H] fp16
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)        # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 for downstream consumption
    # -----------------------------------------------------------------
    return out.to(torch.float32)
