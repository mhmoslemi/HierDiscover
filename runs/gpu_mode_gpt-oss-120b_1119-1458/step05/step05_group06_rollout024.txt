    C_ptr, G_ptr, out_ptr,
    ln_weight_ptr, ln_bias_ptr,
    B, N, H,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    stride_g_batch, stride_g_i, stride_g_j, stride_g_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    epsilon: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # Offsets for hidden dimension iteration
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    # Loop over hidden dim in blocks
    for off in range(0, H, BLOCK_H):
        h_idxs = off + tl.arange(0, BLOCK_H)
        mask_h = h_idxs < H

        c_ptrs = C_ptr + b*stride_c_batch + i*stride_c_i + j*stride_c_j + h_idxs*stride_c_h
        c = tl.load(c_ptrs, mask=mask_h, other=0.0).to(tl.float32)
        sum_val += tl.sum(c, axis=0)
        sumsq_val += tl.sum(c * c, axis=0)

    var = sumsq_val / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    for off in range(0, H, BLOCK_H):
        h_idxs = off + tl.arange(0, BLOCK_H)
        mask_h = h_idxs < H

        c_ptrs = C_ptr + b*stride_c_batch + i*stride_c_i + j*stride_c_j + h_idxs*stride_c_h
        g_ptrs = G_ptr + b*stride_g_batch + i*stride_g_i + j*stride_g_j + h_idxs*stride_g_h

def layernorm_gate_kernel(
    C_ptr, G_ptr, Out_ptr,
    ln_weight_ptr, ln_bias_ptr,
    B, N, H,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    stride_g_batch, stride_g_i, stride_g_j, stride_g_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    EPS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # Initialize sum and sum squares
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    # compute sum across hidden dim
    for off in range(0, H, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < H

        c_ptr = C_ptr + b * stride_c_batch + i * stride_c_i + j * stride_c_j + h * stride_c_h
        c_val = tl.load(c_ptr, mask=mask_h, other=0.0).to(tl.float32)

        sum_val += tl.sum(c_val, axis=0)
        sumsq_val += tl.sum(c_val * c_val, axis=0)

    mean = sum_val / H
    var = sumsq_val / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # compute normalized, gate and store
    for off in range(0, H, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < H

        c_ptr = C_ptr + b * stride_c_batch + i * stride_c_i + j * stride_c_j + h * stride_c_h

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for fused layernorm + out gating on contraction result
@triton.jit
def _layernorm_gate_kernel(
    C_ptr,         # *fp16 contracted: [B, N, N, H]
    G_ptr,         # *fp16 out_gate: [B, N, N, H]
    Out_ptr,       # *fp16 output hidden after LN and gate: [B,N,N,H]
    ln_weight_ptr, # *fp16 (H,)
    ln_bias_ptr,   # *fp16 (H,)
    B, N, H,       # dimensions, i32
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,   # int64 strides
    stride_g_batch, stride_g_i, stride_g_j, stride_g_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    EPS: tl.constexpr,            # layernorm epsilon
    BLOCK_H: tl.constexpr,        # e.g. 64
):
    # program ids map to (b,i,j)
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # ---- First pass: compute mean and variance across hidden dimension ----
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    for offs_h in range(0, H, BLOCK_H):
        h = offs_h + tl.arange(0, BLOCK_H)
        mask_h = h < H

        c_ptr = C_ptr + b*stride_c_batch + i*stride_c_i + j*stride_c_j + h*stride_c_h
        c = tl.load(c_ptr, mask=mask_h, other=0.0).to(tl.float32)  # fp16 -> fp32

        sum_val += tl.sum(c, axis=0)
        sumsq_val += tl.sum(c * c, axis=0)

    mean = sum_val / H
    var = sumsq_val / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ---- Second pass: normalize, apply LN weight/bias, apply out_gate, write out ----
    for offs_h in range(0, H, BLOCK_H):
        h = offs_h + tl.arange(0, BLOCK_H)
        mask_h = h < H

        # load tensors
        c_ptr = C_ptr + b*stride_c_batch + i*stride_c_i + j*stride_c_j + h*stride_c_h
        c = tl.load(c_ptr, mask=mask_h, other=0.0).to(tl.float32)

        g_ptr = G_ptr + b*stride_g_batch + i*stride_g_i + j*stride_g_j + h*stride_g_h
        g = tl.load(g_ptr, mask=mask_h, other=0.0).to(tl.float32)

        w = tl.load(ln_weight_ptr + h, mask=mask_h, other=0.0).to(tl.float32)
        bnorm = tl.load(ln_bias_ptr + h, mask=mask_h, other=0.0).to(tl.float32)

        # layer‑norm
        norm = (c - mean) * inv_std
        norm = norm * w + bnorm

        # out‑gate multiplication
        out = norm * g

        # store result (fp16)
        out_ptr = Out_ptr + b*stride_o_batch + i*stride_o_i + j*stride_o_j + h*stride_o_h
        tl.store(out_ptr, out.to(tl.float16), mask=mask_h)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd) optimized for a H100.

    The implementation fuses:
      * first LayerNorm (float32),
      * left/right projection+gate using a single Linear per side,
      * optional mask multiplication,
      * O(N³) contraction performed with batched cuBLAS (torch.bmm),
      * second LayerNorm + out‐gate applied by a custom Triton kernel,
      * final projection from hidden → model dimension.

    The heavy O(N³) item uses torch.bmm which is faster than the previous
    custom Triton matmul for the tested sizes. The second LayerNorm + gating
    is fused into a Triton kernel to avoid extra memory traversals.
    """
    # ------------------------------------------------------------------
    # unpack input
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5  # LayerNorm epsilon

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right projection & gate linear layers
    # ------------------------------------------------------------------
    # combine projection and gate weights for left side
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # [2*H, dim]

    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # [2*H, dim]

    # combined linear transforms
    left_comb = F.linear(x, left_comb_weight)  # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_weight)  # same shape

    # split into proj and gate
    left_proj = left_comb[..., :hidden_dim]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])

    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Mask (if any) applied only to the *projected* tensors
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates → final left/right tensors used for contraction
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # [B, N, N, H] fp16
    right = right_proj * right_gate       # same

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # batched matmul: left @ rightᵀ
    contracted = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to [B, N, N, H]
    contracted = contracted.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ out_gate = sigmoid(linear(x, out_gate_weight))
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm + out_gate, fused in Triton
    # ------------------------------------------------------------------
    # Cast layer‑norm parameters to fp16
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    # allocate output of the fused LN+gate stage (same shape as contracted)
    hidden_processed = torch.empty_like(contracted)

    # strides (in elements)
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = contracted.stride()
    stride_g_batch, stride_g_i, stride_g_j, stride_g_h = out_gate.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = hidden_processed.stride()
    # strides for LN weight/bias are just 1 (contiguous 1‑D)
    # launch grid: one program per (b,i,j) location
    grid = (B, N, N)

    # Choose a block size that covers hidden_dim efficiently
    # 64 works for hidden_dim up to 384 (two blocks for the max)
    BLOCK_H = 64

    _layernorm_gate_kernel[grid](
        contracted,
        out_gate,
        hidden_processed,
        to_out_norm_weight,
        to_out_norm_bias,
        B,
        N,
        hidden_dim,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        stride_g_batch,
        stride_g_i,
        stride_g_j,
        stride_g_h,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        EPS=eps,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final projection (fp16)
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(hidden_processed, to_out_weight)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for external use
    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul forward (no gradients) – highly‑optimized implementation.

Key ideas
---------
1. First LayerNorm (float32) is performed with torch.nn.functional.layer_norm.
2. The two projection linear layers *and* their gates are fused per side:
   a single Linear produces a concatenated tensor of shape (..., 2*hidden_dim)
   that is split into (proj, gate).  This cuts the number of GEMMs from 4 to 2.
3. Optional mask is applied only to the projected tensors (as in the reference).
4. The expensive O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is executed with a batched cuBLAS GEMM via torch.bmm – this beats the
   handcrafted Triton version for the sequence lengths in the test suite.
5. The second LayerNorm, the out‑gate multiplication, and the affine part
   (weight + bias) of the LayerNorm are fused into a custom Triton kernel.
   The kernel works per (batch, i, j) location, computes mean/variance in
   fp32, normalises the hidden vector, applies the LayerNorm affine parameters
   and finally multiplies by the out‑gate.
6. The final projection from hidden → model dimension is performed with a
   standard fp16 Linear (cuBLAS) and the result is cast back to float32.

The only Triton kernel used is the fused LayerNorm + gate kernel; all other
heavy work relies on cuBLAS which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------

    C_ptr,        # *fp16  contracted Tensor   [B, N, N, H]
    G_ptr,        # *fp16  out_gate Tensor    [B, N, N, H]
    Out_ptr,      # *fp16  destination       [B, N, N, H]
    ln_w_ptr,     # *fp16  LayerNorm weight   [H]
    ln_b_ptr,     # *fp16  LayerNorm bias    [H]
    B, N, H,      # i32 scalar dimensions
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,   # strides of C
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,   # strides of G
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # strides of Out
    EPS: tl.constexpr,                               # LayerNorm epsilon
    BLOCK_H: tl.constexpr,                           # tile size over hidden dim
):
    # program ids map to (batch, i, j)
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # --------------------------------------------------------------
    # 1️⃣ first pass: compute mean & variance across hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    for start_h in range(0, H, BLOCK_H):
        h = start_h + tl.arange(0, BLOCK_H)          # hidden indices for this tile
        mask_h = h < H

        c_ptr = (
            C_ptr
            + b * stride_c_b
            + i * stride_c_i
            + j * stride_c_j
            + h * stride_c_h
        )
        c = tl.load(c_ptr, mask=mask_h, other=0.0).to(tl.float32)  # fp16 → fp32

        sum_val += tl.sum(c, axis=0)
        sumsq_val += tl.sum(c * c, axis=0)

    mean = sum_val / H
    var = sumsq_val / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # --------------------------------------------------------------
    # 2️⃣ second pass: normalise, apply LayerNorm weight/bias,
    #                multiply by out_gate and store result
    # --------------------------------------------------------------
    for start_h in range(0, H, BLOCK_H):
        h = start_h + tl.arange(0, BLOCK_H)
        mask_h = h < H

        # load contracted value
        c_ptr = (
            C_ptr
            + b * stride_c_b
            + i * stride_c_i
            + j * stride_c_j
            + h * stride_c_h
        )
        c = tl.load(c_ptr, mask=mask_h, other=0.0).to(tl.float32)

        # load out_gate value
        g_ptr = (
            G_ptr
            + b * stride_g_b
            + i * stride_g_i
            + j * stride_g_j
            + h * stride_g_h
        )
        g = tl.load(g_ptr, mask=mask_h, other=0.0).to(tl.float32)

        # load LayerNorm affine parameters
        w = tl.load(ln_w_ptr + h, mask=mask_h, other=0.0).to(tl.float32)
        bnorm = tl.load(ln_b_ptr + h, mask=mask_h, other=0.0).to(tl.float32)

        # normalisation
        norm = (c - mean) * inv_std
        norm = norm * w + bnorm

        # apply out_gate
        out = norm * g

        # store fp16 result
        out_ptr = (
            Out_ptr
            + b * stride_o_b
            + i * stride_o_i
            + j * stride_o_j
            + h * stride_o_h
        )
        tl.store(out_ptr, out.to(tl.float16), mask=mask_h)


# --------------------------------------------------------------
# Entry point used by the evaluator
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be all‑ones
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``. Optional ``nomask`` bool.

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, on same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5  # same epsilon used by torch.nn.LayerNorm

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32) – numeric stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2️⃣ cast to fp16 for the bulk of the computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ fuse projection + gate for left and right sides
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # [2*H, dim]
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # [2*H, dim]

    # combined linear transforms (batch‑first, fp16)
    left_comb = F.linear(x, left_comb_weight)   # [B, N, N, 2H]
    right_comb = F.linear(x, right_comb_weight) # same shape

    # split into projection and gate
    left_proj = left_comb[..., :hidden_dim]          # [B, N, N, H]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])   # [B, N, N, H]

    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ optional mask (applied only to the *projected* tensors)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ apply gates → final left / right tensors used for contraction
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # [B, N, N, H]
    right = right_proj * right_gate       # same

    # ------------------------------------------------------------------
    # 6️⃣ heavy O(N³) contraction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where each hidden slice is a separate batch entry
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: left @ rightᵀ (Tensor C = A @ Bᵀ)
    contracted = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to [B, N, N, H]
    contracted = contracted.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ out_gate = sigmoid(linear(x, out_gate_weight))
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ fused LayerNorm + out_gate (Triton)
    # ------------------------------------------------------------------
    ln_weight = weights["to_out_norm.weight"].to(torch.float16)   # [H]
    ln_bias = weights["to_out_norm.bias"].to(torch.float16)      # [H]

    hidden_processed = torch.empty_like(contracted)  # fp16 buffer for LN‑gated activations

    # strides in elements (not bytes)
    stride_c_b, stride_c_i, stride_c_j, stride_c_h = contracted.stride()
    stride_g_b, stride_g_i, stride_g_j, stride_g_h = out_gate.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = hidden_processed.stride()

    # launch Grid: one program per (b, i, j) element
    grid = (B, N, N)
    BLOCK_H = 64   # works for hidden_dim up to 384 (two blocks at most)

    _layernorm_gate_kernel[grid](
        contracted,
        out_gate,
        hidden_processed,
        ln_weight,
        ln_bias,
        B,
        N,
        hidden_dim,
        stride_c_b,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        stride_g_b,
        stride_g_i,
        stride_g_j,
        stride_g_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        EPS=eps,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9️⃣ final projection from hidden → model dimension
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    out = F.linear(hidden_processed, to_out_weight)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 for the test harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
