        y = y + b
    # store as fp16

    tl.store(out_ptr + row * D + offs, y, mask=mask)

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  First LayerNorm (float32) → cast to fp16.
2️⃣  Fuse left/right projection *and* their gates into a single
    linear layer each (weights concatenated → one GEMM per side).
3️⃣  Apply the optional mask, then fuse the element‑wise gate multiplication
    with an in‑place Triton kernel (proj *= gate).
4️⃣  Heavy O(N³) contraction is performed as a strided‑batched GEMM
    (torch.bmm) on the tensor layout [B·H, N, N] which lets cuBLAS/Tensor‑cores
    handle the work.
5️⃣  Second LayerNorm (fp16) → out_gate (sigmoid) → in‑place Triton kernel
    (out *= out_gate).
6️⃣  Final linear projection back to `dim`.
7️⃣  Cast the result back to float32.

The Triton kernel is used for the two element‑wise multiplications
(left/right gating) and for the final out‑gate multiplication,
slashing temporary memory traffic and kernel launch overhead.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    out_ptr,                # *fp16   in‑place output
    mul_ptr,                # *fp16   multiplier
    N,                      # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _pointwise_mul_inplace(x: torch.Tensor, y: torch.Tensor) -> None:
    """
    In‑place element‑wise multiplication:  x ← x * y
    Both tensors must be contiguous and have the same shape (FP16).
    """
    assert x.is_contiguous() and y.is_contiguous()
    total = x.numel()
    BLOCK = 131072  # 128 Ki elements per program (adjusted for occupancy)
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _pointwise_mul_kernel[grid](
        x, y, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool) – optional
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional

    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5  # LayerNorm epsilon

    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.nn.functional.layer_norm works on float32; we keep it that way for stability.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0

    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction: batched GEMM (cuBLAS/Tensor‑core)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to [B·H, N, N] – each hidden slice becomes a batch entry
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # out_flat[b·H, i, j] = Σ_k left_flat[b·H, i, k] * right_flat[b·H, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) → out_gate multiplication → final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    # Apply out_gate (in‑place Triton kernel)
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    _pointwise_mul_inplace(out, out_gate)

    # Final projection back to `dim`
    out = torch.nn.functional.linear(out, to_out_w)   # bias=False

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
