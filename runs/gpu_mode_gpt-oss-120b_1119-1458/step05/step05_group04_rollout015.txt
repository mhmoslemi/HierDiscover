def _apply_mask_and_gate_kernel(
    left_ptr, right_ptr,

    mask_ptr,
    N, # total elements

    right = tl.load(right_ptr + offs, mask=mask_ok, other=0.0)

        right = right * mask * rgate

        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask_ok)
    tl.store(right_ptr + offs, right, mask=mask_ok)

"""
TriMul – outgoing version (AlphaFold‑3) forward pass
===================================================

The TriMul module updates a pairwise tensor ``x`` of shape
``[B, N, N, C]`` by a “cubic” contraction:

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where ``left`` and ``right`` are linear projections of ``x`` gated by
sigmoids (and optionally masked).  The implementation follows the
reference PyTorch version but makes three key performance‑oriented
changes:

1. **Projection & gating fusion** – the two linear layers that produce a
   projection and its gate are merged into a single ``nn.functional.linear``
   call (weights are concatenated).  This halves the number of kernel launches
   and reduces weight‑load traffic.

2. **Fused mask + gate kernel** – a custom Triton kernel applies the mask
   (if present) and the sigmoid gates to ``left`` and ``right`` in‑place.
   The kernel processes both tensors in a single pass, eliminating intermediate
   temporaries and extra memory traffic.

3. **Batched GEMM via cuBLAS** – after gating the tensors are reshaped to
   ``[B·H, N, N]`` (``H`` = hidden dimension) and multiplied with a single
   ``torch.bmm`` call, which on an H100 exploits Tensor‑cores (FP16×FP16→FP32)
   for the heavy ``O(N³·H)`` work.

All other operations (LayerNorm, final gating, output projection) are kept
in FP16 for speed and cast back to FP32 at the end, matching the harness
requirements.  The kernel is compatible with the test configurations
(``bs∈{1,2}``, ``N∈{128,256,512,1024}``, ``C∈{128}``, ``hidden∈{128}``).

The forward pass returns a tensor of shape ``[B, N, N, C]`` (float32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place gated multiplication (optionally with mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr,                # *fp16 left  (input & output)
    right_ptr,               # *fp16 right (input & output)
    left_gate_ptr,           # *fp16 left gate (sigmoid output)
    right_gate_ptr,          # *fp16 right gate (sigmoid output)
    mask_ptr,                # *fp16 mask (broadcasted over hidden dim) – ignored if HAS_MASK == 0
    N,                       # total number of scalar elements (B·N·N·H)
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,  # compile‑time flag
):
    pid = tl.program_id(0)                     # 1‑D grid over total elements
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < N

    # load operands (mask loads a dummy pointer when not used)
    lhs = tl.load(left_ptr + offs,   mask=valid, other=0.0)
    rhs = tl.load(right_ptr + offs, mask=valid, other=0.0)

    lgate = tl.load(left_gate_ptr + offs,   mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs,  mask=valid, other=0.0)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=valid, other=1.0)   # mask is 0/1 in FP16
        lhs = lhs * lgate * m
        rhs = rhs * rgate * m
    else:
        lhs = lhs * lgate
        rhs = rhs * rgate

    tl.store(left_ptr + offs,  lhs, mask=valid)
    tl.store(right_ptr + offs, rhs, mask=valid)


def _apply_gate_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
) -> None:
    """
    In‑place fusion of mask (if provided) and sigmoid gates onto ``left`` and ``right``.
    All tensors must be contiguous, same shape, and FP16.
    """
    total = left.numel()
    BLOCK = 8192                         # works well for the sizes in the test suite
    grid = ((total + BLOCK - 1) // BLOCK,)

    has_mask = mask is not None
    # Triton requires a pointer for every argument; when no mask we pass ``left`` as a dummy.
    mask_ptr = mask if has_mask else left

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=1 if has_mask else 0,
        num_warps=4,
    )

    ]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation (no gradients).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same device as ``input_tensor``.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(dtype_fp16)                     # [B, N, N, dim] in fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights once to fp16 on target device
    # ------------------------------------------------------------------
    # projection + gate weights are concatenated to halve the number of matmuls
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)   # (2·H, dim)

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)   # (2·H, dim)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (fp16) – fused projection & gate
    # ------------------------------------------------------------------
    # left_proj + left_gate
    left_cat = torch.nn.functional.linear(x, left_concat_w)     # [B,N,N,2·H]
    left, left_gate = torch.split(left_cat, hidden_dim, dim=-1)  # each [B,N,N,H]

    # right_proj + right_gate
    right_cat = torch.nn.functional.linear(x, right_concat_w)   # [B,N,N,2·H]
    right, right_gate = torch.split(right_cat, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    mask_fp16 = None
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → fp16 → broadcast over hidden dim
        mask_fp16 = mask_tensor.to(dtype=dtype_fp16).unsqueeze(-1)

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates (fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask (if any) & gates with a Triton kernel
    # ------------------------------------------------------------------
    # Ensure all tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if mask_fp16 is not None:
        mask_fp16 = mask_fp16.contiguous()
    _apply_gate_and_mask(left, right, left_gate, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – batched GEMM (Tensor‑Core accelerated)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                # H == hidden_dim
    # shape to [B·H, N, N] – this layout is optimal for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B·H, N, N]
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # [B·H, N, N]

    # out_flat[b·H,i,j] = Σ_k left_flat[b·H,i,k] * right_flat[b·H,j,k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Element‑wise product with out_gate (already fp16)
    out = out * out_gate

    # Final linear back to original ``dim`` (fp16)
    out = torch.nn.functional.linear(out, to_out_w)          # [B,N,N,dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
