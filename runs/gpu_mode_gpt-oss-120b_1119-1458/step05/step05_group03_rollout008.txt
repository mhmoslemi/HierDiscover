"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1.  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2.  Fuse the four linear projections (left/right value + left/right gate)
    into a single matmul with weight of shape (4·hidden_dim, dim).  Slice the
    result into the four tensors (proj/value & gate‑pre‑sigmoid).
3.  Compute the out‑gate (sigmoid(linear(x))) separately.
4.  If a mask is present, fuse element‑wise masking, sigmoid(gate) and
    multiplication `proj * gate * mask` with a tiny Triton kernel
    (`apply_gate_sigmoid_mask_kernel`).  When there is no mask we simply
    multiply by the sigmoid‑gate using PyTorch (in‑place).
5.  Heavy contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    performed as a batched GEMM (B·hidden_dim independent matrices of
    size N×N) using `torch.bmm`.  The data layout is made contiguous so
    cuBLAS Tensor‑Core kernels are used.
6.  Second Layer‑norm (FP16) → out‑gate multiplication (element‑wise).
7.  Final linear projection back to the original channel dimension.
8.  Cast the result back to float32 (the harness expects FP32).

Only the forward pass is required; gradients are not computed.
A Triton kernel is used for the mask‑gate‑sigmoid fusion.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * sigmoid(gate) * mask) – mask is broadcast
# over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_sigmoid_mask_kernel(
    proj_ptr,          # *fp16 : projection tensor to be updated, shape (size,)
    gate_ptr,          # *fp16 : pre‑sigmoid gate tensor, same shape as proj
    mask_ptr,          # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,              # i32   : total number of elements in proj (B*N*N*H)
    hidden_dim,        # i32   : width of the hidden dimension (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    active = offs < size                        # boundary mask

    # Load data
    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)

    # Sigmoid(gate) = 1 / (1 + exp(-gate))
    gate = 1.0 / (1.0 + tl.exp(-gate))

    # Broadcast mask: every hidden_dim consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    mask = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)

    # Fuse multiplication
    out = proj * gate * mask

    # Store result back in‑place (overwrites the projection tensor)
    tl.store(proj_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a batched GEMM.
    Both left and right are FP16 tensors of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] with inner dimension (k) contiguous
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM (A @ Bᵀ) – use cuBLAS Tensor‑Core kernels (FP16 inputs → FP32 acc.)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]      (float32/bool) – may be ignored if `nomask=True`
        - weights_dict : dict[str, torch.Tensor] – FP32 model weights
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # input channel dim (c_z)
    hidden_dim = config["hidden_dim"]  # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # -> [B, N, N, dim]  float32
    x = x_norm.half()      # cast once to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right value and gate linear projections (FP16)
    # ------------------------------------------------------------------
    # Cast all relevant weights to FP16 once
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # Combine the four weight matrices into a single (4*hidden, dim) matrix
    # Order: left_proj | left_gate | right_proj | right_gate
    combined_lr_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape (4*hidden_dim, dim)

    # Single linear projection for all four streams
    combined = torch.nn.functional.linear(x, combined_lr_weight)  # [B, N, N, 4*H]

    # Slice the concatenated result – no copy, just view
    left_proj, left_gate_pre, right_proj, right_gate_pre = torch.split(
        combined, hidden_dim, dim=-1
    )
    # At this point:
    #   left_proj, right_proj    : [B, N, N, H]  (value streams)
    #   left_gate_pre, right_gate_pre : [B, N, N, H] (pre‑sigmoid gates)

    # ------------------------------------------------------------------
    # 3️⃣ Out‑gate (sigmoid) – separate linear on the same x
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Apply masking + sigmoid‑gate (fused Triton kernel) or plain gating
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple sigmoid + element‑wise multiplication (in‑place)
        left_gate = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)
        left = left_proj.mul_(left_gate)   # FP16 in‑place
        right = right_proj.mul_(right_gate)
        # Release intermediate tensors
        del left_gate, right_gate, left_gate_pre, right_gate_pre
    else:
        # Mask is present → fuse sigmoid, mask, and multiplication.
        # The mask is broadcast over the hidden dimension.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # shape (B*N*N,)

        total_elements = left_proj.numel()               # B * N * N * H
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left_proj * sigmoid(left_gate_pre) * mask   (in‑place)
        apply_gate_sigmoid_mask_kernel[grid](
            left_proj,
            left_gate_pre,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * sigmoid(right_gate_pre) * mask   (in‑place)
        apply_gate_sigmoid_mask_kernel[grid](
            right_proj,
            right_gate_pre,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # The gate_pre tensors are no longer needed
        del left_gate_pre, right_gate_pre

        # After the kernel `left_proj` holds the gated+masked left tensor,
        # and `right_proj` holds the gated+masked right tensor.
        left = left_proj
        right = right_proj

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, H]   FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16) on the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate               # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # back to `dim` (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the external harness and return
    # ------------------------------------------------------------------
    return out.float()
