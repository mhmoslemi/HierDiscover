"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

The implementation follows the reference PyTorch module but removes all
intermediate materialisations of the projected tensors and runs the
N³ contraction with a single‐purpose Triton GEMM that works for any
sequence length.  The 4‑D tensor X is first LayerNorm‑normalised in
float32, then cast to fp16 for all linear / gating operations.  The
heavy contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

is performed as a batched matrix‑multiply C = A @ Bᵀ where A and B are
reshaped to (B·H, N, N).  The kernel uses 128×128 tiles (64×64 for very
short sequences) with fp16 arithmetic and fp32 accumulation,
prefetching through `num_stages=2`.  After the contraction a second
LayerNorm, the output‑gate and a final linear projection are applied
with standard PyTorch kernels.  The final tensor is returned in float32.

The code works for all configurations listed in the problem statement,
including optional masking (nomask=False).  No gradients are required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides of A (M,i,k)
    stride_bm, stride_bj, stride_bk,   # strides of B (M,j,k)
    stride_cm, stride_ci, stride_cj,   # strides of C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM used for the TriMul contraction.

    A and B are already the masked‐and‑gated tensors of shape
    (B*H, N, N).  The kernel computes C = A @ Bᵀ with fp16 inputs and
    fp32 accumulation.
    """
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ---------------------------------------------------------------
    # tile offsets
    # ---------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------- load A(i,k) -------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am                # batch*hidden offset
            + offs_i[:, None] * stride_ai      # i offset
            + offs_k[None, :] * stride_ak      # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # --------------------- load Bᵀ(j,k) ------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm                # batch*hidden offset
            + offs_k[:, None] * stride_bk      # k offset (first)
            + offs_j[None, :] * stride_bj      # j offset (second)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # write back C[i,j]
    # ---------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` (both [B, N, N, H]) using a fused
    Triton GEMM.  The result has the same shape and dtype (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()    # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()     # (N*N, N, 1)

    # Choose tile size based on sequence length.
    # 64×64 works well for N ≤ 256, 128×128 is faster for larger N.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_gemm_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) – may be all‑ones
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the bulk of the work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ cast all model weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)       # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)     # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ optional mask (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ gating (sigmoid) — fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Fuse gates in‑place to keep memory pressure low
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ heavy contraction (O(N³)) – always use the Triton GEMM
    # ------------------------------------------------------------------
    contracted = _batched_matmul_triton(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (fp16) → out‑gate → final projection
    # ------------------------------------------------------------------
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    contracted = contracted * out_gate
    out = F.linear(contracted, to_out_w)                # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
