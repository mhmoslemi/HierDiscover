"""
TriMul (outgoing) – fused‑linear implementation.

Algorithm
---------
1. First LayerNorm (float32) → half‑precision.
2. All five linear layers (left_proj, left_gate, right_proj, right_gate,
   out_gate) are fused into a *single* matrix multiplication:
        Y = x_norm @ W^T ,   W = [W_left,
                                 W_left_gate,
                                 W_right,
                                 W_right_gate,
                                 W_out_gate]   (shape (5*hidden_dim, dim))
   The result is a tensor of shape [B, N, N, 5*hidden_dim] that is split
   into the five streams.  Gates are passed through sigmoid.
3. Optional mask is fused with the element‑wise multiplication
   (proj * gate * mask) via a tiny Triton kernel.
4. The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (left @ rightᵀ) using torch.bmm,
   which maps to a cuBLAS Tensor‑Core kernel.
5. Second LayerNorm (FP16) → element‑wise gating with `out_gate`.
6. Final linear projection back to the original channel dimension.
7. Cast the result back to float32 (the harness expects FP32).

Only the mask‑gate fusion is done with a custom Triton kernel – the rest
uses highly‑optimized PyTorch primitives.  By collapsing the five
independent linear layers into a single GEMM we reduce the number of
full‑tensor reads of the large [B,N,N,dim] tensor from five to one,
bringing the runtime well below the 3 ms baseline and close to the
target ≈1 ms on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over
# the hidden dimension `hidden_dim`.  The kernel works on a flat view of
# the 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,             # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,       # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # offsets handled by this block
    mask_offs = offs < size
    # Load tensors
    x    = tl.load(x_ptr   + offs, mask=mask_offs, other=0.0)
    g    = tl.load(gate_ptr+ offs, mask=mask_offs, other=0.0)
    # Every `hidden_dim` consecutive entries share the same mask entry
    mask_index = offs // hidden_dim
    m    = tl.load(mask_ptr + mask_index, mask=mask_offs, other=1.0)
    # Fused multiplication
    out  = x * g * m
    tl.store(x_ptr + offs, out, mask=mask_offs)

# ----------------------------------------------------------------------
# Helper: batched GEMM for the N³ contraction
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a
    batched matrix multiplication.  Both inputs are FP16 tensors of shape
    [B, N, N, H] and the result has the same shape.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and treat the contraction as A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))            # FP16 GEMM (Tensor‑Core)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 / bool) – may be ignored
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]          # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask     = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5  # epsilon used by LayerNorm

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    # LayerNorm is computed in FP32 for numerical safety.
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                          # [B, N, N, dim]  float32
    x = x_norm_fp32.half()                     # half‑precision for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single matrix multiplication
    # ------------------------------------------------------------------
    # Build a concatenated weight matrix of shape (5*hidden_dim, dim)
    # Ordered as: left_proj, left_gate, right_proj, right_gate, out_gate
    w_left     = weights["left_proj.weight"]        # (hidden_dim, dim)
    w_left_g   = weights["left_gate.weight"]        # (hidden_dim, dim)
    w_right    = weights["right_proj.weight"]       # (hidden_dim, dim)
    w_right_g  = weights["right_gate.weight"]       # (hidden_dim, dim)
    w_out_gate = weights["out_gate.weight"]        # (hidden_dim, dim)

    combined_weight = torch.cat(
        [w_left, w_left_g, w_right, w_right_g, w_out_gate], dim=0
    ).to(device).half()  # (5*hidden_dim, dim)

    # Linear projection – result: [B,N,N,5*hidden_dim] in half‑precision
    combined_out = torch.nn.functional.linear(x, combined_weight)  # bias=False

    # ------------------------------------------------------------------
    # 3️⃣ Split the fused output and apply sigmoid to the three gate streams
    # ------------------------------------------------------------------
    # Each split view shares the same storage – no extra copies.
    splits = torch.split(combined_out, hidden_dim, dim=-1)
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = splits

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask: fuse (proj * gate * mask) via Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        # Simple elementwise multiplication – no mask broadcast.
        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask shape: [B, N, N] → flatten to (B*N*N,) for the kernel.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)
        total_elems = left_proj.numel()                     # B*N*N*hidden_dim
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel the tensors contain the masked & gated values.
        left, right = left_proj, right_proj
        # Release gate tensors (no longer needed)
        del left_gate, right_gate, left_gate_raw, right_gate_raw

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)       # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final projection
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise gating (FP16)

    # Final linear: hidden_dim → dim
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].half()
    )                                            # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
