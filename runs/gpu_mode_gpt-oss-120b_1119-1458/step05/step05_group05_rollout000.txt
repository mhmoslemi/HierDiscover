import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    hidden_dim, size,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load tensors
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Get index into mask (no hidden_dim dimension)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left/right: (B, N, N, H) fp16
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    # Compute A @ B^T for each batch
    c = torch.bmm(a, b.transpose(1, 2))
    # Restore shape (B, N, N, H)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass, fused mask‑gate multiplication with Triton.
    The heavy N³ contraction is performed as a batched GEMM via cuBLAS
    (torch.bmm). All heavy computation is in fp16; LayerNorms are applied
    in fp32 for stability. The final result is returned as fp32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. First LayerNorm (float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # 2. Cast to fp16 for all subsequent heavy ops
    x = x.to(torch.float16)

    # 3. Load fp16 weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections (fp16)
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w) # (B,N,N,hidden_dim)

    # 5. Compute gating signals (fp16)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # 6. Fuse mask, gate and mul via Triton (if mask present)
    if not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)          # (B*N*N,)
        size = left.numel()                     # B*N*N*hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_flat,
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # 7. O(N³) contraction via batched GEMM (fp16)
    out = _batched_gemm_fp16(left, right)  # (B,N,N,hidden_dim) fp16

    # 8. Second LayerNorm (fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 9. Apply out_gate and final projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim) fp16

    # 10. Cast back to float32 for the harness
    return out.to(torch.float32)

import torch

from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask·gate·tensor multiplication
# Used only when a mask is supplied (ensures at least one custom kernel
# implementation).  The kernel works on a flattened view of the tensor
# of shape [B,N,N,hidden_dim] and a flattened mask of shape [B,N,N].
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  tensor to be updated (flattened)
    gate_ptr,       # *fp16  gate tensor (same shape as x)
    mask_ptr,       # *fp16  mask tensor (flattened, shape [B*N*N])
    hidden_dim,     # i32   hidden dimension (H)
    size,           # i32   total number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load operands (they are fp16 because the pointers point to fp16 tensors)
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Broadcast mask across the hidden dimension.
    # Each mask entry corresponds to `hidden_dim` contiguous elements in x.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default mask = 1

    # Fuse multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM (fp16) performing the O(N³) contraction.
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H) in fp16
    Returns:
        out: (B, N, N, H) fp16
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and multiply A @ Bᵀ
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))                # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
