def proj_gate_mask_fuse_kernel(
    proj_ptr, gate_pre_ptr, mask_ptr, hidden_dim, size,
    BLOCK: tl.constexpr, 
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < size

    # compute mask index
    mask_offs = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate_pre = tl.load(gate_pre_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        mask = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)
        out = proj * gate * mask
    else:
        out = proj * gate

    tl.store(proj_ptr + offs, out, mask=active)

Implementation:

```
@triton.jit
def proj_gate_mask_fuse_kernel(
    proj_ptr, gate_pre_ptr, mask_ptr,
    hidden_dim,
    size,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < size
    mask_offs = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate_pre = tl.load(gate_pre_ptr + offs, mask=active, other=0.0)
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))
    if HAS_MASK:
        m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)
        out = proj * gate * m
    else:
        out = proj * gate
    tl.store(proj_ptr + offs, out, mask=active)

"""
TriMul (outgoing) forward pass – highly optimized Triton‐augmented implementation.

Algorithm
---------
1️⃣  Layer‑Norm the input tensor (float32) → cast to float16.
2️⃣  Fuse **all** linear projections (left/right and all gates) into a single
    matmul:   out = x @ W_all.T   where W_all contains 5·hidden_dim rows
    (left_proj, right_proj, left_gate, right_gate, out_gate).
3️⃣  Split the fused result into its five components.
4️⃣  Fuse element‑wise gating, sigmoid, and optional mask multiplication for the
    left and right branches in a tiny Triton kernel (computes
    proj * sigmoid(gate) * mask).  No mask → mask is omitted at compile‑time.
5️⃣  Heavy O(N³) contraction:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    implemented as a batched GEMM (torch.bmm) on tensors reshaped to
    [B·H, N, N]; this uses cuBLAS Tensor‑Core kernels on H100.
6️⃣  Second Layer‑Norm over the hidden dimension (float16).
7️⃣  Apply the out‑gate (sigmoid) and the final linear projection back to `dim`.
8️⃣  Cast the result back to float32 (required by the harness).

Only the forward pass is provided; no autograd is needed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
#  Triton kernel: proj * sigmoid(gate) * (optional) mask
#  The kernel works on a flattened view of the tensor [B, N, N, H].
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_mask_fuse_kernel(
    proj_ptr,          # *fp16 : tensor to be updated (left_proj or right_proj)
    gate_pre_ptr,      # *fp16 : pre‑sigmoid gate tensor (same shape as proj)
    mask_ptr,          # *fp16 : mask tensor (shape B·N·N), broadcasted over H
    hidden_dim,        # i32   : size of hidden dimension H
    size,              # i32   : total number of elements in proj (B·N·N·H)
    BLOCK: tl.constexpr,            # compile‑time tile size (≤1024)
    HAS_MASK: tl.constexpr,         # 1 → apply mask, 0 → skip mask
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear indices [pid*B …]
    active = offs < size                                 # guard out‑of‑bounds

    # Index of the mask element that corresponds to this proj element.
    # Since hidden_dim is the innermost stride we can obtain the mask index
    # by integer division (the mask does not have a hidden dimension).
    mask_offs = offs // hidden_dim

    # Load the tensors.
    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate_pre = tl.load(gate_pre_ptr + offs, mask=active, other=0.0)

    # Sigmoid(gate) = 1 / (1 + exp(-x))
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))

    if HAS_MASK:
        m = tl.load(mask_ptr + mask_offs, mask=active, other=1.0)
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(proj_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
#  Batched GEMM implementing the O(N³) contraction.
#  left/right: [B, N, N, H] (fp16)
#  returns:    [B, N, N, H] (fp16)
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # out[i,j] = Σ_k a[i,k] * b[j,k]  ==  a @ bᵀ
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out  # fp16


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32/bool, may be all‑ones)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable on the raw input.
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the computationally heavy part.
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse *all* linear projections (left, right and the three gates)
    #    into a single matmul to reduce memory traffic.
    # -----------------------------------------------------------------
    # Convert weights once to fp16.
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate the five weight matrices along the output‑feature axis.
    # Shape after concat: (5*hidden_dim, dim)
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # fp16

    # Flatten x to (B·N·N, dim) and perform the fused linear.
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # [B*N*N,5*H]
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)           # [B,N,N,5*H]

    # -----------------------------------------------------------------
    # 4️⃣ Split the fused tensor into its components.
    # -----------------------------------------------------------------
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        fused_out, hidden_dim, dim=-1
    )  # each: [B,N,N,H]

    # -----------------------------------------------------------------
    # 5️⃣ Fuse sigmoid(gate) + (optional) mask multiplication using Triton.
    # -----------------------------------------------------------------
    total_elements = left_proj.numel()  # B·N·N·H
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    if not nomask:
        # Convert mask to fp16, flatten it, and keep contiguous storage.
        mask_fp16 = mask_tensor.to(device=device, dtype=torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)   # shape B·N·N
        HAS_MASK = 1
    else:
        # Dummy tensor – will never be read because HAS_MASK=0.
        mask_flat = torch.empty(0, device=device, dtype=torch.float16)
        HAS_MASK = 0

    # Left branch: proj * sigmoid(gate) * mask
    proj_gate_mask_fuse_kernel[grid](
        left_proj,
        left_gate_pre,
        mask_flat,
        hidden_dim,
        total_elements,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )
    # Right branch
    proj_gate_mask_fuse_kernel[grid](
        right_proj,
        right_gate_pre,
        mask_flat,
        hidden_dim,
        total_elements,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )
    # The kernel overwrites the proj tensors in‑place, so they now contain
    # the gated and masked values.
    left = left_proj
    right = right_proj

    # ---------------------------------------------------------------
    # 6️⃣ Compute the out_gate (sigmoid) – used after the second norm.
    # ---------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_pre)   # [B,N,N,H] fp16

    # ---------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS/Tensor‑Core).
    # ---------------------------------------------------------------
    out = _batched_matmul_torch(left, right)  # [B,N,N,H] fp16

    # ---------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16) over the hidden dimension.
    # ---------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ---------------------------------------------------------------
    # 9️⃣ Apply out_gate and final linear projection back to `dim`.
    # ---------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
    )  # [B,N,N,dim] fp16

    # ---------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness) and return.
    # ---------------------------------------------------------------
    return out.to(torch.float32)
