"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (FP32) and cast to FP16.
2️⃣  Linear projections (left/right) and three gating streams are computed in FP16.
3️⃣  If a mask is present we fuse ``proj * gate * mask`` with a tiny Triton kernel
    (`mask_gate_fuse_kernel`).  The mask is broadcast over the hidden dimension,
    so only a single launch per tensor is needed.
4️⃣  Heavy contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is implemented as a custom batched GEMM in Triton (`tri_mul_kernel`).  The
    kernel works on the original layout `[B, N, N, H]` and avoids the extra
    contiguity/permute copies required by ``torch.bmm``.
5️⃣  Second Layer‑norm (FP16) → out‑gate (sigmoid) → final linear projection
    back to the original channel dimension.
6️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the forward pass is needed; no gradients are computed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over
# the hidden dimension `hidden_dim`.  Operates on a flattened view of
# the 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,)
    size,             # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,       # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask = offs < size                         # boundary mask

    # Load tensor and corresponding gate
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask: every `hidden_dim` consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched Tri‑Mul contraction
#
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#
# Operates directly on tensors with layout [B, N, N, H] (FP16) and
# writes the result into `out` with the same layout.  No permutes or
# copies are required.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16 : left tensor  [B, N, N, H]
    right_ptr,         # *fp16 : right tensor [B, N, N, H]
    out_ptr,           # *fp16 : output tensor [B, N, N, H]
    B, N, H,           # i32   : dimensions
    # strides of the left tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides of the right tensor
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides of the output tensor
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1️⃣ Identify the (batch, head) pair handled by this block
    # ------------------------------------------------------------------
    pid_bh = tl.program_id(0)               # 0‑th axis: batch * head
    pid_i  = tl.program_id(1)               # 1‑st axis: tile row (i dimension)
    pid_j  = tl.program_id(2)               # 2‑nd axis: tile column (j dimension)

    # Derive batch index `b` and hidden dimension index `h`
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # 2️⃣ Base pointers for this (b, h) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # ------------------------------------------------------------------
    # 3️⃣ Row / column offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 4️⃣ Accumulator (FP32 for better numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 5️⃣ Loop over the reduction dimension `k`
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left tile (i, k) ---------------------------------
        left_offs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_offs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # ------- load right tile transposed (k, j) ----------------------
        # right_T[k, j] = right[b, j, k, h]
        right_offs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_tile = tl.load(right_offs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # ------- multiply‑accumulate -----------------------------------
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # 6️⃣ Write back the result (cast back to FP16)
    # ------------------------------------------------------------------
    out_offs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_offs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])



# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (bool / float)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]          # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask     = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # [B, N, N, dim]  float32
    x = x_norm.half()      # cast once to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Gating streams (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Mask + gate fusion (in‑place) – Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        # no mask → simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate
    else:
        # mask shape: [B, N, N] → flatten to (B*N*N,) and keep fp16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)

        total_elements = left.numel()               # B * N * N * H
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed – free for memory
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via custom Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # [B, N, N, H] FP16

    # Strides for the three tensors (all are contiguous in the same layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – tuned for H100 tensor‑core performance
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        B * H,                                   # batch * head dimension
        (N + BLOCK_M - 1) // BLOCK_M,           # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # j‑tiles
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast result back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
```
