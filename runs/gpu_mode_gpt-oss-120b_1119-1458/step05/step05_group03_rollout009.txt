"""
TriMul (outgoing) – fast forward pass.

The kernel performs the same operations as the reference PyTorch implementation
but reduces the number of launches and memory traffic:

1️⃣  First LayerNorm (FP32) → cast to FP16.
2️⃣  Fuse the five linear projections (`left`, `right`, three gates) into a
    single GEMM (`x @ W_fused.T`).  The result is split into the five tensors.
3️⃣  Compute the three gates with a sigmoid.
4️⃣  Fuse element‑wise mask·gate·proj multiplication for `left` and `right`
    with a tiny Triton kernel (`mask_gate_fuse_kernel`).  When `nomask=True`
    a simple in‑place multiplication is used.
5️⃣  The heavy O(N³) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is executed as a batched GEMM using cuBLAS (`torch.bmm`).  This is the
    fastest path on an H100.
6️⃣  Second LayerNorm (still in FP16), out‑gate multiplication and the final
    linear back to the original channel dimension.
7️⃣  Cast the result back to float32 (the harness expects FP32).

At least one custom Triton kernel is used (mask‑gate fusion).  The implementation
covers all test configurations, works with or without a mask and runs on the
H100 with Triton 3.3.1.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over the
# hidden dimension `hidden_dim`.  The kernel works on a flat view of the
# 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,        # *fp16 : tensor to be updated          (size,)
    g_ptr,        # *fp16 : gate tensor (same shape)     (size,)
    m_ptr,        # *fp16 : mask tensor (B*N*N)          (mask_size,)
    size,         # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,   # i32   : hidden dimension width
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load tensors
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(g_ptr + offs, mask=mask, other=0.0)

    # Each `hidden_dim` consecutive entries share the same mask entry
    mask_idx = offs // hidden_dim
    m = tl.load(m_ptr + mask_idx, mask=mask, other=1.0)

    # Fused multiplication
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a batched GEMM (cuBLAS).  Inputs are FP16 tensors of shape
    [B, N, N, H] and the result has the same shape.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) – make contiguous to give cuBLAS a dense layout
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # A @ Bᵀ for every hidden head
    c = torch.bmm(a, b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 0️⃣ Bring model weights to FP16 once (except the first LayerNorm)
    # ------------------------------------------------------------------
    # Weights for the fused projection (5 × hidden_dim × dim)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    fused_proj_w = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).half()  # (5*hidden_dim, dim)

    # Weights / bias for the second LayerNorm
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()

    # Final projection back to `dim`
    to_out_w = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32
    x = x_norm.half()  # cast once to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # shape: [B, N, N, 5*hidden_dim]
    fused = torch.nn.functional.linear(x, fused_proj_w)  # FP16

    # Split the concatenated result
    (
        left,
        right,
        left_gate_pre,
        right_gate_pre,
        out_gate_pre,
    ) = torch.split(fused, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 3️⃣ Gating signals (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask and gating (fused via Triton when a mask exists)
    # ------------------------------------------------------------------
    if nomask:
        # In‑place element‑wise multiplication (no extra kernel launch)
        left = left.mul_(left_gate)
        right = right.mul_(right_gate)
    else:
        # Prepare mask (FP16) – broadcasted over the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        total_elements = left.numel()  # B * N * N * hidden_dim
        BLOCK = 65536  # tuneable – fits well into H100 shared memory
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # we no longer need the intermediate gate tensors
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim]  (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication + final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)  # back to `dim` (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the evaluation harness
    # ------------------------------------------------------------------
    return out.float()
