def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    device = input_tensor.device
    # 1) first layernorm using float32 weight and bias
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x_norm.to(torch.float16)
    
    # Pack projection weights for left,right,gates
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # Concatenate
    prepacked_weight = torch.cat([w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate], dim=0).to(torch.float16) # shape (5*hidden_dim, dim)
    # Linear
    proj = torch.nn.functional.linear(x, prepacked_weight)  # shape [B,N,N,5*hidden_dim]
    # Slice
    left_proj = proj[..., :hidden_dim]
    left_gate = proj[..., hidden_dim:2*hidden_dim]
    right_proj = proj[..., 2*hidden_dim:3*hidden_dim]
    right_gate = proj[..., 3*hidden_dim:4*hidden_dim]
    out_gate = proj[..., 4*hidden_dim:5*hidden_dim]
    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)
    
    # Apply mask before gating multiplication if mask present (nomask flag)
    if not nomask:
        mask_fp = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp
    # Multiply gating
    left = left_proj * left_gate
    right = right_proj * right_gate
    
    # O(N^3) heavy contraction
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)
    else:

"""
Outgoing TriMul (AlphaFold‑3) – fused linear‐projections + fast batched GEMM.

The forward pass consists of:
  1️⃣ LayerNorm on the input (float32 for numerical stability)
  2️⃣ One fused linear layer that simultaneously computes
        - left projection
        - left gating
        - right projection
        - right gating
        - out gating
     (concatenated weight matrix → a single matmul)
  3️⃣ Sigmoid on the three gate tensors.
  4️⃣ Optional masking of the two projection tensors.
  5️⃣ Element‑wise multiplication of each projection by its gate.
  6️⃣ The heavy O(N³) contraction  out[i,j,h] = Σₖ left[i,k,h]·right[j,k,h]
     is performed as a batched GEMM (A @ Bᵀ) where the batch dimension
     is B·hidden_dim.  For short sequences (≤256) we launch a custom
     Triton kernel with 64×64×64 tiles; longer sequences fall back to
     torch.bmm (cuBLAS) which is optimal on H100.
  7️⃣ LayerNorm over the hidden dimension, multiplication by the out‑gate,
     and a final linear projection back to the original feature dimension.
  8️⃣ Cast the result back to float32 (required by the test harness).

All heavy linear work (steps 2‑5) is done in fp16 with fp32 accumulation.
The batched GEMM also accumulates in fp32.  This reduces memory traffic
by collapsing five separate matmul‑gate calls into a single matmul,
greatly lowering runtime while preserving exact functional behaviour.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (batch, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 × fp16 → fp32 GEMM with fp16 output (C = A @ Bᵀ)."""
    pid_m = tl.program_id(0)   # tile along i (output rows)
    pid_n = tl.program_id(1)   # tile along j (output cols)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A[i, k]   (shape BLOCK_M × BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am               # batch*hidden offset
            + offs_i[:, None] * stride_ai     # i offset
            + offs_k[None, :] * stride_ak     # k offset
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # --------------------------------------------------------------
        # Load Bᵀ[j, k]   (shape BLOCK_K × BLOCK_N)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm               # batch*hidden offset
            + offs_k[:, None] * stride_bk     # k offset (first)
            + offs_j[None, :] * stride_bj     # j offset (second)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # --------------------------------------------------------------
        # Mixed‑precision dot: fp16 × fp16 → fp32
        # --------------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store result (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using the Triton GEMM.
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()     # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()      # (N*N, N, 1)

    # Tile size – 64×64×64 works well on H100 for fp16 mma
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,          # M
        N,              # N
        N,              # K
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using cuBLAS (torch.bmm). Operates in fp16
    with fp32 accumulation (default for torch.bmm on GPU).
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]

    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) – may be all‑ones
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """

    # 1️⃣ unpack
    # ------------------------------------------------------------------

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------

        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # shape (5*hidden_dim, dim)

    # shape: [B, N, N, 5*hidden_dim] (fp16)
    fused = torch.nn.functional.linear(x, packed_weight)

    right_gate_raw = fused[..., 3 * h:4 * h]   # [B, N, N, h]
    out_gate_raw = fused[..., 4 * h:5 * h]    # [B, N, N, h]

    # 6️⃣ sigmoid gates (fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 7️⃣ optional mask – broadcast to hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 8️⃣ apply gates (in‑place to reduce memory pressure)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 9️⃣ heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        contracted = _batched_matmul_triton(left, right)   # fp16, [B, N, N, h]
    else:
        contracted = _batched_matmul_torch(left, right)    # fp16

    # ------------------------------------------------------------------
    # 🔟 second LayerNorm (fp16) → out‑gate → final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    contracted = contracted * out_gate                         # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)      # (dim, hidden_dim)
    out = torch.nn.functional.linear(contracted, to_out_w)    # fp16, shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
