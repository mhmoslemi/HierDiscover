"""
Outgoing TriMul (AlphaFold‑3) forward pass – highly‑fused implementation.

Algorithm
---------
1️⃣  Layer‑norm the input in FP32 (numerical stability) and cast to FP16.
2️⃣  Fuse the five linear layers (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single matrix‑multiply:
       [B,N,N,dim] @ [dim, hidden_dim*5]  →  five tensors of shape
       [B,N,N,hidden_dim] (two projections and three gates).
3️⃣  Apply sigmoid to the three gate tensors.
4️⃣  If a mask is present, fuse mask‑multiplication, gating and the
    projections with a tiny Triton kernel:
        out = proj * gate * mask    (mask broadcast over hidden dim).
    Otherwise simply multiply projection × gate.
5️⃣  O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is expressed as a batched GEMM (C = A @ Bᵀ) by merging the batch
   and hidden dimensions, which lets cuBLAS tensor‑core kernels run at
   full speed without any data copies.
6️⃣  Second LayerNorm (still in FP16), out‑gate (sigmoid) and final
    Linear projection back to the model dimension.
7️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the mask‑gate fusion is written in Triton; all heavy linear algebra
uses the highly‑optimized cuBLAS kernels.  The implementation works for
any batch size B∈{1,2}, sequence length N∈{128,256,512,1024},
model dimension dim, hidden dimension hidden_dim and optional mask.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask   (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                # *fp16  [B, N, N, H]
    gate_ptr,                # *fp16  [B, N, N, H]  (already sigmoid‑ed)
    mask_ptr,                # *fp16  [B, N, N]
    out_ptr,                 # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute: out = proj * gate * mask   (mask broadcast over hidden dim H).
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # i‑, j‑coordinates for this block
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask flags (outside range → masked out)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load mask tile (BLOCK_I × BLOCK_J) – single element per (i,j) pair
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16, shape (BLOCK_I, BLOCK_J)

    # Loop over hidden dimension tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16, shape (BLOCK_I, BLOCK_J, BLOCK_H)

        # Load gate tile (already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask   – broadcast mask over H
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse projection, gate and binary mask (broadcast over hidden dim).

    Arguments
    ---------
    proj : torch.Tensor [B, N, N, H]   (fp16)
    gate : torch.Tensor [B, N, N, H]   (fp16, already sigmoid‑ed)
    mask : torch.Tensor [B, N, N]      (fp16)

    Returns
    -------
    out : torch.Tensor [B, N, N, H]    (fp16)
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides are in elements (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out  = out.stride()

    # Tune these tile sizes for H100 – they fit easily in registers/shared memory
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8   # 8 × hidden_dim tiles → 128/8 = 16 iterations for hidden_dim=128

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0],  s_out[1],  s_out[2],  s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction:  out = left @ rightᵀ
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a strided batched GEMM (fp16 inputs → fp16 output).

    Arguments
    ---------
    left, right : torch.Tensor [B, N, N, H] (fp16, contiguous)

    Returns
    -------
    out : torch.Tensor [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # Merge batch and hidden dims → shape (B*H, N, N)
    # We purposely avoid .contiguous() after permute to keep memory
    # bandwidth low; torch.bmm requires contiguity, so we reshape after
    # the cheap view/permute that does *no* data copy.
    left_perm = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2) # [B, H, N, N]

    # The view below is on a contiguous tensor because permute does not
    # change the underlying memory layout; the resulting view is a stride‑
    # aware reshape that does not copy data.
    left_flat = left_perm.reshape(B * H, N, N)   # [B*H, N, N]
    right_flat = right_perm.reshape(B * H, N, N) # [B*H, N, N]

    # Batched GEMM C = A @ Bᵀ   (TensorCore fp16 kernels)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    B, N, _, _ = x.shape
    x_flat = x.view(B * N * N, dim)          # [B·N·N, dim]
    proj_gate_flat = torch.nn.functional.linear(x_flat, weight_cat)  # [B·N·N, 5*hidden_dim]
    # Reshape back to 4‑D
    proj_gate = proj_gate_flat.view(B, N, N, hidden_dim * 5)

    # Split the concatenated tensor
    split_t = torch.split(proj_gate, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = split_t

    # -----------------------------------------------------------------
    # 3️⃣  Gates -> sigmoid (still fp16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # -----------------------------------------------------------------
    # 4️⃣  Optional mask gating – fusion via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and has a batch dimension matching inputs
        mask_fp16 = mask_tensor.to(dtype_fp16)   # [B, N, N]
        left  = _apply_mask_gate(left_proj,  left_gate,  mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        # No mask → just multiply by gates
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣  Heavy O(N³) contraction via batched GEMM (cuBLAS Tensor‑Core)
    # -----------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # -----------------------------------------------------------------
    # 6️⃣  Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16),
    )

    # -----------------------------------------------------------------
    # 7️⃣  Out‑gate & final projection back to model dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=dtype_fp16),
    )   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 8️⃣  Cast back to FP32 (the harness expects FP32)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
