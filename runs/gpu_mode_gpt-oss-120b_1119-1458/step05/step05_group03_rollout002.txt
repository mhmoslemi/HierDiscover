"""
TriMul (outgoing) – fast forward pass.

Algorithm Overview
------------------
1️⃣ **First LayerNorm** (FP32 → FP16) to normalize the 4‑D pairwise tensor.
2️⃣ **Fused projection + gating**: 5 linear transforms (left_proj,
   right_proj, left_gate, right_gate, out_gate) are merged into a single
   half‑precision GEMM (`torch.nn.functional.linear`).  The result is
   reshaped to ``[B, N, N, 5, hidden_dim]`` and split.
3️⃣ **Sigmoid gates** are applied to the three gating streams.
4️⃣ **Mask‑gate fusion** – if a mask is present we fuse ``proj * gate *
   mask`` for ``left`` and ``right`` using a tiny Triton kernel that
   broadcasts the mask over the hidden dimension.
5️⃣ **Heavy contraction** – compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] *
   right[b,j,k,h]`` as a batched GEMM via ``torch.bmm`` (cuBLAS/Tensor‑Core).
6️⃣ **Second LayerNorm** (FP16) on the contracted tensor.
7️⃣ **Output gating** (element‑wise multiplication with ``out_gate``).
8️⃣ **Final linear projection** back to the original channel dimension.
9️⃣ Cast the result to FP32 (the harness expects float32).

Only the mask‑gate fusion is performed in a custom Triton kernel; the
remaining work leverages highly‑optimized cuBLAS GEMMs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over the
# hidden dimension `hidden_dim`.  The kernel works on a flat view of the
# 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,       # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,    # *fp16 : gate tensor, same shape as x
    mask_ptr,    # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,        # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,  # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask = offs < size                         # boundary guard

    # Load data from global memory
    x   = tl.load(x_ptr   + offs, mask=mask, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=mask, other=0.0)

    # Compute mask offset (one mask value per hidden_dim consecutive elements)
    mask_offs = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM (FP16 inputs → FP32 accumulation)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Computes the contraction:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    Parameters
    ----------
    left, right : torch.Tensor
        Half‑precision tensors of shape [B, N, N, hidden_dim].

    Returns
    -------
    torch.Tensor
        Half‑precision tensor of shape [B, N, N, hidden_dim].
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]; GEMM will compute A @ Bᵀ for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))  # (B*H, N, N)  – uses cuBLAS Tensor‑Core
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,  # input tensor [B, N, N, dim] (float32)
        torch.Tensor,  # mask tensor [B, N, N] (float32/bool) – may be ignored
        Dict[str, torch.Tensor],  # model weights (float32)
        Dict[str, Any],           # config dict, must contain dim & hidden_dim
    ]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32
    x = x_norm.half()  # cast once to FP16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2️⃣ Build a single combined weight for the five linear layers
    #    (left_proj, right_proj, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [hidden_dim, dim]
            weights["right_proj.weight"],  # [hidden_dim, dim]
            weights["left_gate.weight"],   # [hidden_dim, dim]
            weights["right_gate.weight"],  # [hidden_dim, dim]
            weights["out_gate.weight"],    # [hidden_dim, dim]
        ],
        dim=0,
    ).half()  # shape (5*hidden_dim, dim)

    # ------------------------------------------------------------------
    # 3️⃣ Single fused GEMM: x @ combined_weightᵀ  → [B, N, N, 5*hidden_dim]
    # ------------------------------------------------------------------
    y = torch.nn.functional.linear(x, combined_weight)  # FP16
    # reshape to separate the 5 streams
    B, N, _, _ = y.shape
    y = y.view(B, N, N, 5, hidden_dim)  # [B, N, N, 5, hidden_dim]

    left_raw   = y[..., 0, :]  # [B, N, N, hidden_dim]
    right_raw  = y[..., 1, :]
    left_gate_raw = y[..., 2, :]
    right_gate_raw = y[..., 3, :]
    out_gate_raw  = y[..., 4, :]

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask and gating.
    #     We fuse ``proj * gate * mask`` for left/right via a small Triton
    #     kernel when a mask is present; otherwise we do an in‑place mul.
    # ------------------------------------------------------------------
    if nomask:
        # No mask → simple element‑wise multiplication
        left = left_raw.mul_(left_gate)
        right = right_raw.mul_(right_gate)
    else:
        # Flatten mask to (B*N*N,) and cast to FP16
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        total_elements = left_raw.numel()  # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Fuse left = left_raw * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_raw,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Fuse right = right_raw * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_raw,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_raw
        right = right_raw
        # Gate tensors can be freed early
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (batched GEMM) – cuBLAS/Tensor‑Core.
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)  # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim` (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].half(),
    )  # → [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
