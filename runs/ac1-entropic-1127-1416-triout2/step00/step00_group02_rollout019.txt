def score(seq):
    arr = np.array(seq, dtype=np.float64)
    fft_len = 1 << (len(seq)*2 - 1).bit_length()
    # Actually need convolution of length 2*n -1 => we can compute using FFT length >=2*n
    f = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:2*len(seq)-1]
    max_c = np.max(conv)
    sum_a = np.sum(arr)
    if sum_a < 0.01: return np.inf
    return 2 * len(seq) * max_c / (sum_a * sum_a)

        # conv_rev gradient

assistantfinal**Solution Explanation**

The score of a sequence `a = (a₀,…,aₙ₋₁)` is  

\[
\text{score}(a)=
\frac{2\,n\;\max\_{k}(a*a)_k}{\bigl(\sum_{i}a_i\bigr)^2},
\qquad
(a*a)_k=\sum_{i+j=k}a_i\,a_j .
\]

Because the formula is *scale invariant* (multiplying all entries by a
constant does not change the value) we may keep the sequence normalised
with `sum(a)=1`. Then the problem becomes

```
 minimise   max_k (a*a)_k
 subject to  a_i ≥ 0 ,  Σ a_i = 1 .
```

The optimal shape is not known analytically, but a classical solution is
the **inverse‑square‑root** sequence  

\[
a_i \propto \frac{1}{\sqrt{i+1}}\; .
\]

With this shape the convolution is almost constant and the score is
≈ π/2 ≈ 1.57, already a big improvement over the uniform sequence (score = 2).

To push the score even lower we perform a projected gradient descent on a
*smoothed* version of the objective.
For a given `a` we compute the full convolution `c = a*a`.  
The supremum `max(c)` is approximated by a soft‑max

\[
w_k=\exp\bigl(\lambda\,(c_k-\max c)\bigr) ,
\qquad
\lambda>0 .
\]

When `λ` is large the weights concentrate on the indices where the
convolution is maximal, but the function stays smooth and differentiable.
The (sub)‑gradient of the smoothed objective is

\[
\frac{\partial}{\partial a_i}
\sum_k w_k (a*a)_k
=2\sum_k w_k\,a_{k-i}
   =2\;[(w\ \text{reversed})*a]_{\,2n-2-i},
\]

which can be evaluated with one additional FFT‑convolution.
A projected step

```
a ← a − η·gradient          (η : step size)
a ← max(a,0)                (non‑negativity)
a ← a / sum(a)              (renormalise to Σa=1)
```

keeps the constraints satisfied.
The parameters `η` and `λ` are slowly annealed, and occasional random
perturbations help to escape possible plateaus.

The whole optimisation runs until the time budget expires (≤ 1000 s) and
the best sequence found is returned.

The algorithm uses only NumPy and the supplied `evaluate_sequence`
function, runs on a single CPU and respects all competition rules.

import time
import numpy as np

def _conv_fft(a, L):
    """
    Convolution of `a` with itself using FFT of length L.
    Returns the linear convolution (length 2*len(a)-1).
    """
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * len(a) - 1]
    return conv, A


def _gradient_via_weights(a, A, w, L):
    """
    Computes the gradient of the soft‑max weighted convolution.
    a : current sequence (len=n)
    A : rfft of a (length L)
    w : weight vector (len=2n-1)
    Returns a vector of length n.
    """
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L)
    # convolution a * w_rev  (length n + (2n-1) - 1 = 3n-2)
    conv_rev = np.fft.irfft(A * W_rev, n=L)[: 3 * len(a) - 2]
    # gradient_i = 2 * conv_rev[2n-2-i]
    n = len(a)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_rev[idx]
    return grad


def propose_candidate(seed=0, budget_s=1000):

    Searches for a step‑function (a non‑negative sequence) that minimises
    `evaluate_sequence`.  The optimisation starts from the inverse‑square‑root
    shape a_i ∝ 1/√(i+1) and performs a projected gradient descent on a
    soft‑max approximation of the L∞ norm of the self‑convolution.
    The best sequence seen before the time limit is returned.
    """
    np.random.seed(seed)

    # length of the sequence – use the length of the best known construction
    # if it exists, otherwise fall back to a reasonable size.
    try:
        n = len(height_sequence_1)   # provided by the runner
    except NameError:
        n = 3000

    # ----- initial shape -------------------------------------------------
    a = np.array([1.0 / np.sqrt(i + 1) for i in range(n)], dtype=np.float64)
    a /= a.sum()                     # normalise to Σa = 1

    # best solution so far
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"[init] length={n}   score={best_score:.6f}")

    # FFT length large enough for both convolutions (a*a) and (a*w_rev)
    L = 1 << ((3 * n - 2).bit_length())   # next power of two >= 3n-2

    # optimisation parameters
    eta = 0.02            # initial step size (L1‑change per iteration)
    lam = 1.0             # initial soft‑max sharpness
    max_l1_change = 0.01  # we never change the total mass by more than 1%
    no_improve = 0
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    while time.time() < deadline:
        # ----- convolution and weights ------------------------------------
        conv, A = _conv_fft(a, L)               # a*a
        max_c = conv.max()

        # ----- projected gradient step ------------------------------------

        # limit the L1 norm of the step

            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            no_improve = 0

    print(f"[final] best score = {best_score:.6f}")
    return best.tolist()
