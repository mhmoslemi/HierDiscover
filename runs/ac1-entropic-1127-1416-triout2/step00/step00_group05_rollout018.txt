def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = np.inf

    lengths = [8,10,12,14,16,18,20,24,30,40,50,60] # can adjust
    while time.time() < deadline:
        n = np.random.choice(lengths)
        # random init dirichlet
        seq = np.random.rand(n)
        seq = seq / seq.sum()
        # ensure scale not too small: fine
        cur_score = evaluate_sequence(seq.tolist())
        # annealing
        T = 1.0
        while time.time() < deadline and T > 1e-4:
            # propose modification
            i,j = np.random.randint(n), np.random.randint(n)
            if i == j:
                continue
            # compute epsilon based on current values
            amount = (np.random.rand() - 0.5) * 0.2 * seq[i]  # up to 10% shift
            # move from i to j
            new_seq = seq.copy()
            if new_seq[i] - amount < 0:
                amount = new_seq[i]  # move all
            new_seq[i] -= amount
            new_seq[j] += amount
            # evaluate
            new_score = evaluate_sequence(new_seq.tolist())
            delta = new_score - cur_score
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                seq = new_seq
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.tolist()
                    print(f"New best {best_score:.6f} n={n}")
            # update temperature schedule
            T *= 0.9999  # geometric cooling
        # perform occasional random restart
    return best_seq

    # Project onto simplex {x >=0, sum x =1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

"""
propose_candidate
-----------------
A simulated‑annealing / gradient‑descent hybrid that searches for a
non‑negative step‑function (a list of heights) minimizing the
evaluation function

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm works for many different lengths (n from 8 to 120).  For each
length it starts from a random point on the probability simplex
(sum = 1, all entries ≥0) and performs:

* a Metropolis‑style random “mass‑transfer” move (preserves the sum);
* occasional projected gradient steps that try to lower the current
  maximal convolution value;
* a geometric cooling schedule.

All candidates are kept, and the best sequence found until the time
budget expires is returned.  If nothing better than the existing
`height_sequence_1` is found, that sequence is returned as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """
    Projects a real vector v onto the simplex {x | x_i >= 0, sum(x) = 1}.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)            # 1‑based index in the paper
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core search routine
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000):
    """
    Returns a list of non‑negative floats (the height sequence) that
    minimizes `evaluate_sequence`.  The function runs for at most
    `budget_s` seconds and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a little safety margin

    # ------------------------------------------------------------------
    # start from the existing construction if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        best_seq = list(height_sequence_1)          # global variable provided by the environment
        best_score = evaluate_sequence(best_seq)
        print(f"[init] using height_sequence_1 – score = {best_score:.6f}")
    except Exception:
        best_seq = None
        best_score = np.inf

    # ------------------------------------------------------------------
    # search parameters
    # ------------------------------------------------------------------
    lengths = [8, 10, 12, 14, 16, 20, 24, 30, 40, 50, 60, 80, 100, 120]
    # inner loop limits – these are cheap for the modest n we use
    max_inner_iters = 3000          # per length
    grad_interval = 200             # perform a gradient step every this many iterations
    # ------------------------------------------------------------------

    while time.time() < deadline:
        n = int(np.random.choice(lengths))
        # random start on the simplex (sum = 1)
        cur_seq = np.random.rand(n)
        cur_seq = cur_seq / cur_seq.sum()
        cur_score = evaluate_sequence(cur_seq.tolist())
        # keep the best seen so far for this length
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.tolist()
            print(f"[new‑global] score={best_score:.6f}  n={n}")

        # simulated annealing parameters
        T = max(0.5 * cur_score, 0.1)      # initial temperature
        cooling = 0.9995                  # geometric cooling factor

        for it in range(max_inner_iters):
            if time.time() >= deadline:
                break

            # ----- random mass‑transfer move (preserves sum) -----
            i, j = np.random.choice(n, 2, replace=False)
            # move a small fraction of mass from i to j
            alpha = (np.random.rand() - 0.5) * 0.4        # in [-0.2, 0.2]
            delta = alpha * cur_seq[i]                  # amount to shift
            # clip to keep entries non‑negative
            if delta > cur_seq[i]:
                delta = cur_seq[i]
            elif -delta > cur_seq[j]:
                delta = -cur_seq[j]
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

            new_score = evaluate_sequence(new_seq.tolist())
            dE = new_score - cur_score
            if dE < 0 or np.random.rand() < np.exp(-dE / max(T, 1e-12)):
                cur_seq, cur_score = new_seq, new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.tolist()
                    print(f"[global‑update] score={best_score:.6f} n={n} it={it}")

            # ----- occasional projected gradient step -----
            if (it + 1) % grad_interval == 0:
                # compute full convolution once
                conv = np.convolve(cur_seq, cur_seq)
                t_max = int(np.argmax(conv))
                # gradient of the maximal convolution term
                grad = np.zeros_like(cur_seq)
                idx = t_max - np.arange(n)
                mask = (idx >= 0) & (idx < n)
                grad[mask] = 2.0 * cur_seq[idx[mask]]
                # step size (scaled with problem size)
                lr = 0.05 / n
                cand = cur_seq - lr * grad
                cand = _project_onto_simplex(cand)   # keep feasible
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < cur_score:
                    cur_seq, cur_score = cand, cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = cur_seq.tolist()
                        print(f"[grad‑update] score={best_score:.6f} n={n} it={it}")

            # ----- temperature cooling -----
            T *= cooling
            if T < 1e-8:
                T = 1e-8   # avoid division by zero

    # ------------------------------------------------------------------
    # final safety: make sure we return a list of floats
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen but guards against empty result
        best_seq = [1.0]
    print(f"[finished] best score = {best_score:.6f} length = {len(best_seq)}")
    return best_seq
