    # v is array of floats, maybe not sum 1.
    # returns w with sum=1, w_i>=0.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

def project_simplex(v):
    """Project onto probability simplex (sum=1, nonnegative)."""
    # v can be numpy array
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

            best = cand

"""
Propose a step‐function with a very low evaluation score.

The evaluation function is
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv(a,a)` is the aperiodic convolution of the coefficient vector
with itself.  For a non‑negative vector with sum `S` the theoretical lower
bound is `2*n/(2*n-1) ≈ 1`.  This can be approached by making the
autoconvolution as flat as possible.  A simple way to obtain an almost
flat convolution is to use a random 0/1 pattern with roughly half of the
entries equal to 1.  With `m ≈ n/2` ones we have

    sum(a) = m,
    average(conv) = sum(a)^2 / (2*n-1) ≈ n/8,
    max(conv) ≈ average(conv) (fluctuations are only O(√n)),

giving a score close to 1.  The function below builds such a binary vector,
does a short hill‑climbing phase (random swaps of a 1 and a 0) that only
keeps improvements, and finally returns the best sequence found before
the time budget expires.

The algorithm runs well within the 1000 s budget, uses only a single CPU
core and a few hundred evaluations of `evaluate_sequence`.
"""
import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    deadline = time.time() + budget_s - 5.0   # leave a tiny safety margin
    iteration = 0
    # a very small step size for the hill‑climbing: we only accept strict
    # improvements, so there is no need for a temperature schedule.
    while time.time() < deadline:
        # pick a random 1 and a random 0 to swap
        if ones_pos.size == 0 or zeros_pos.size == 0:
            break
        i = np.random.choice(ones_pos)
        j = np.random.choice(zeros_pos)

        # propose the swap
        cand = best_a.copy()
        cand[i] = 0.0
        cand[j] = 1.0

        # evaluate the candidate
        cand_score = evaluate_sequence(cand.tolist())

        # accept only if the score improves
        if cand_score < best_score:
            best_a = cand
            best_score = cand_score
            # update the index lists efficiently
            ones_pos = np.flatnonzero(best_a == 1.0)
            zeros_pos = np.flatnonzero(best_a == 0.0)
            print(f"[iter {iteration:06d}] improved score={best_score:.6f}")

        iteration += 1
        # optional early stop if we already hit a near‑optimal bound
        if best_score <= 1.02:          # the theoretical limit is ~1
            break

    print(f"[finished] best score={best_score:.6f} after {iteration} iterations")
    return best_a.tolist()
