def simplex_projection(v):
    """Return projection of vector v onto simplex {x: x >=0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def conv_fft(a):
    size = 2*len(a) - 1
    # Next power of 2 for efficiency
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[:size]
    return conv

def project_simplex(v):
    """Project vector v onto the probability simplex (non-negative sum=1)."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    condition = u - cssv / ind > 0
    rho = ind[condition].max()
    theta = cssv[condition].min() / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        # all non-positive
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def perturb(a, step):
    n = len(a)
    # Choose two distinct indices
    i, j = np.random.choice(n, 2, replace=False)
    delta = step * np.random.randn()
    # Transfer delta from i to j while staying non-negative
    a_new = a.copy()
    if delta >= 0:
        # move from i to j
        delta = min(delta, a_new[i])  # cannot exceed a_i
        a_new[i] -= delta
        a_new[j] += delta
    else:

"""
Simple simulated‑annealing search with occasional LP refinements.

The goal is to minimise

    2·n·max(conv(a, a)) / (∑ a)²

for a non‑negative vector ``a`` (the step heights).  The problem is
non‑convex because of the ``max`` over the convolution.  The algorithm
works on the probability simplex (∑ a = 1, a_i ≥ 0) because the
objective is scale‑invariant.

* A temperature‑controlled random walk (simulated annealing) performs
  cheap “mass‑transfer’’ moves: a small amount of weight is moved from
  one coordinate to another, keeping the ℓ₁‑norm exactly constant.
  Moves that improve the objective are always accepted; otherwise they
  are accepted with probability exp(−Δ/ T) where the temperature T
  decreases linearly with the remaining time.

* Every few hundred iterations a deterministic refinement is attempted:
  the current vector ``a`` is used as the fixed factor in the linear
  programme

        maximise   ∑ b_j
        subject to  conv(a, b) ≤ M   ,   b ≥ 0

  where ``M = max(conv(a, a))``.  The LP solution ``b`` is mixed into the
  current vector (step size 0.1) and then projected back onto the simplex.
  This is exactly the “direction’’ used by the state‑of‑the‑art method and
  provides a reliable descent direction.

The loop runs until the allotted time budget (default 1000 s) is exhausted.
The best sequence found is returned as a plain ``list[float]``.  Only
NumPy, SciPy and the supplied ``evaluate_sequence`` are used.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_max_fft(a: np.ndarray) -> float:
    """
    Compute max(conv(a, a)) using an FFT based convolution.
    ``a`` is assumed to be a one‑dimensional real array.
    """
    n = a.shape[0]
    size = 2 * n - 1
    N = _next_pow2(size)
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[:size]
    return float(np.max(conv))


def eval_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective used by the optimiser.
    The function is scale‑invariant; the vector is not altered.
    """
    sum_a = np.sum(a)
    if sum_a < 0.01:                     # same guard as evaluate_sequence
        return float("inf")
    max_b = conv_max_fft(a)
    n = a.shape[0]
    return 2.0 * n * max_b / (sum_a * sum_a)


def perturb(a: np.ndarray, step: float) -> np.ndarray:
    """
    Produce a new candidate by moving a small amount of mass from one
    coordinate to another.  The ℓ₁‑norm stays exactly constant.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = step * np.random.randn()

    # Transfer |delta| from the source to the destination,
    # never making a component negative.
    a_new = a.copy()
    if delta >= 0.0:
        delta = min(delta, a_new[i])
        a_new[i] -= delta
        a_new[j] += delta
    else:
        delta = min(-delta, a_new[j])
        a_new[j] -= delta
        a_new[i] += delta
    return a_new


# ----------------------------------------------------------------------
# LP based refinement (the same idea as the reference implementation)
# ----------------------------------------------------------------------


def solve_convolution_lp(f: np.ndarray, rhs: float):
    """
    Linear programme:
        maximise    ∑ b_j
        subject to   conv(f, b)[k] ≤ rhs   for all k
                     b ≥ 0
    Returns the optimal ``b`` as a NumPy array, or ``None`` if the LP fails.
    """
    n = f.shape[0]
    # objective: minimise -∑ b   (linprog always minimises)
    c = -np.ones(n)

    # Build A_ub x ≤ b_ub for every convolution index.
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv(f, b)[k] = Σ_i f[i] * b[k - i]   (where the second index is in range)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f[i]
        rows.append(row)
        rhs_vec.append(rhs)

    A_ub = np.vstack([rows, -np.eye(n)])            # also enforce b ≥ 0  →  -b ≤ 0
    b_ub = np.concatenate([rhs_vec, np.zeros(n)])

    # Use the robust HiGHS solver
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None


def get_lp_improved(seq: np.ndarray) -> np.ndarray | None:
    """
    Compute a LP‑based improvement direction similar to the reference
    ``get_good_direction_to_move_into`` but with a larger mixing factor.
    Returns a new candidate (already normalised to sum = 1) or ``None``.
    """
    n = seq.shape[0]
    total = np.sum(seq)
    if total == 0:
        return None

    # Normalise to the classical magic sum √(2n) used in the paper.
    scale = np.sqrt(2.0 * n) / total
    f_norm = seq * scale

    rhs = conv_max_fft(f_norm)     # max(conv(f_norm, f_norm))
    b = solve_convolution_lp(f_norm, rhs)
    if b is None:
        return None

    # Rescale the LP solution to the same magic sum.
    sum_b = np.sum(b)
    if sum_b == 0:
        return None
    b_rescaled = b * (np.sqrt(2.0 * n) / sum_b)

    # Mix the direction into the current vector (mixing factor t).
    t = 0.1
    new_seq = (1.0 - t) * seq + t * b_rescaled
    new_seq = np.maximum(new_seq, 0.0)
    s = np.sum(new_seq)
    if s == 0:
        return None
    new_seq = new_seq / s
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑height sequence with a low evaluation score.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence discovered.  A simple simulated‑annealing loop explores the
    simplex of non‑negative vectors, while every few hundred iterations an
    LP based refinement is performed to obtain a reliable descent direction.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # small safety margin

    # Keep the globally best sequence across all lengths.
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Lengths to try; larger n gives a better theoretical bound but costs
    # more per iteration.  Adjust as time permits.
    lengths = [200, 400, 800]

    start_time = time.time()

    for n in lengths:
        # ----- initialisation -------------------------------------------------
        # Start from a random point on the simplex.
        cur = np.random.exponential(1.0, size=n)
        cur = cur / cur.sum()
        cur_val = eval_score(cur)

        # Record the very first uniform sequence as a fallback.
        if best_seq is None or cur_val < best_val:
            best_seq = cur.copy()
            best_val = cur_val

        # ----- annealing parameters -----------------------------------------
        # Temperature decreases linearly with the remaining time.
        # Step size for the random perturbation is tied to the current mean.
        iter_count = 0
        lp_interval = 300            # try LP refinement every N iterations

        while time.time() < deadline:
            # linear cooling schedule
            remaining = deadline - time.time()
            fraction = max(0.0, remaining / budget_s)   # 1 → start, 0 → end
            temperature = max(1e-6, 0.5 * fraction)     # scale of accepted uphill moves
            step_size = max(1e-4, 0.05 * np.mean(cur) * fraction)

            # ----- propose a new point --------------------------------------
            cand = perturb(cur, step_size)
            cand_val = eval_score(cand)

            # ----- Metropolis acceptance --------------------------------------
            if cand_val < cur_val or np.random.rand() < np.exp(-(cand_val - cur_val) / temperature):
                cur, cur_val = cand, cand_val
                if cur_val < best_val:
                    best_seq = cur.copy()
                    best_val = cur_val

            # ----- occasional LP refinement -----------------------------------
            if iter_count % lp_interval == 0:
                refined = get_lp_improved(cur)
                if refined is not None:
                    ref_val = eval_score(refined)
                    if ref_val < cur_val:
                        cur, cur_val = refined, ref_val
                        if ref_val < best_val:
                            best_seq = cur.copy()
                            best_val = ref_val

            # occasional logging (does not affect performance heavily)
            if iter_count % 2000 == 0:
                print(
                    f"[{time.time() - start_time:6.2f}s] n={n}, it={iter_count:,}, "
                    f"best={best_val:.6f}"
                )

            # early exit if we already passed the target
            if best_val < 1.503:
                break

            iter_count += 1

        if best_val < 1.503:
            break
        # If time remains, move to the next length.

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # should never happen, fallback to a single height 1.0
        best_seq = np.array([1.0])

    final = [float(v) for v in best_seq.tolist()]
    print(f"Search finished. Best score {best_val:.6f} with length {len(final)}.")
    return final
