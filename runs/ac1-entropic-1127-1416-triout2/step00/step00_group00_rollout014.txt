"""
Optimisation of the self‑convolution bound.

The algorithm explores a few low‑dimensional families of non‑negative
step‑functions (exponential, symmetric exponential and power‑law “U”
shapes) and keeps the best candidate found.  The best solution is then
refined with a simple sub‑gradient descent that directly reduces the
largest convolution entry.  All work is done under the time budget
provided by the caller.

No external data are read; the global variable ``height_sequence_1`` is
used only as a possible starting point.
"""

import numpy as np
import time
import random
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_score(seq: np.ndarray) -> float:
    """
    Compute the same value as ``evaluate_sequence`` but without the
    security wrapper.  This is used for speed.
    """
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    n = seq.shape[0]
    return 2.0 * n * max_b / (sum_a * sum_a)


def exponential_seq(n: int, r: float) -> np.ndarray:
    """a_i = r**i   (r in (0,1)).  Normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    seq = r ** i
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= total
    return seq


def symmetric_exponential_seq(n: int, r: float) -> np.ndarray:
    """a_i = r**i + r**(n-1-i)  (U‑shaped) – normalised."""
    i = np.arange(n, dtype=np.float64)
    seq = r ** i + r ** (n - 1 - i)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= total
    return seq


def power_seq(n: int, gamma: float) -> np.ndarray:
    """
    a_i = ((i+1)*(n-i))**(-gamma)   (edges heavy, centre light).
    When gamma==0 the sequence is uniform.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    # product (i+1)*(n-i) is never zero for 1≤i≤n
    prod = idx * (n + 1 - idx)
    # guard against overflow / under‑flow
    with np.errstate(divide='ignore', invalid='ignore'):
        seq = np.power(prod, -gamma)
    seq = np.nan_to_num(seq, nan=0.0, posinf=0.0, neginf=0.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= total
    return seq


def subgradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    One sub‑gradient step that tries to reduce the maximal convolution
    entry.  ``step`` is a positive scalar.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k_max = int(conv.argmax())

    # gradient of the selected convolution entry:
    grad = np.zeros_like(seq)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_range = k_max - i_range
    grad[i_range] = 2.0 * seq[j_range]

    new_seq = seq - step * grad
    new_seq = np.where(new_seq < 0.0, 0.0, new_seq)
    tot = new_seq.sum()
    if tot < 1e-12:                     # avoid division by zero
        return seq
    new_seq /= tot                       # renormalise (scaling does not matter)
    return new_seq

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    The routine runs for ``budget_s`` seconds and returns the best
    sequence (as a python list) found so far.

    The search consists of:
      * sampling fast parametric families (exponential, symmetric
        exponential, power‑law U‑shapes);
      * refining the current best with a few sub‑gradient iterations;
      * occasional random perturbations to escape local minima.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence we will work with.
    # Prefer the length of the existing construction if it exists.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size == 0:
            raise ValueError
        n = int(base.size)
        # start from the provided sequence (normalised to sum 1)
        cur_seq = base.copy()
        cur_seq = cur_seq.clip(min=0.0)      # just in case
        cur_seq_sum = cur_seq.sum()
        if cur_seq_sum == 0.0:
            cur_seq = np.full(n, 1.0 / n)
        else:
            cur_seq = cur_seq / cur_seq_sum
    except Exception:
        # fall back to a moderate default size
        n = 5000
        cur_seq = np.full(n, 1.0 / n)

    best_seq = cur_seq.copy()
    best_score = compute_score(best_seq)

    print(f"[init] length={n}, start score={best_score:.6f}")

    deadline = time.time() + float(budget_s) - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ---- pick a parametric family ---------------------------------
        family = random.choice(["exp", "symexp", "power"])
        if family == "exp":
            # pick c = n*(1-r) in a reasonable range
            c = random.uniform(0.5, 5.0)          # empirical range
            r = 1.0 - c / n
            r = max(min(r, 0.999999), 0.1)        # keep inside (0,1)
            cand = exponential_seq(n, r)

        elif family == "symexp":
            c = random.uniform(0.5, 5.0)
            r = 1.0 - c / n
            r = max(min(r, 0.999999), 0.1)
            cand = symmetric_exponential_seq(n, r)

        else:  # power
            gamma = random.uniform(0.0, 2.5)      # 0 → uniform, >0 → U‑shape
            cand = power_seq(n, gamma)

        # ---- evaluate -------------------------------------------------
        score = compute_score(cand)

        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[{iteration:06d}] New best {best_score:.6f} "
                  f"by {family}")

        # ---- occasional local refinement -------------------------------
        if random.random() < 0.07:  # roughly every 15 iterations
            # start from the current best
            refined = best_seq.copy()
            step = 0.5
            for _ in range(12):
                refined = subgradient_step(refined, step)
                step *= 0.9
                sc = compute_score(refined)
                if sc < best_score:
                    best_score = sc
                    best_seq = refined.copy()
                    print(f"[{iteration:06d}] Sub‑grad improvement {best_score:.6f}")
                    break

        # ---- random jitter to avoid stagnation ------------------------
        if random.random() < 0.03:
            idx = random.randrange(n)
            delta = best_seq[idx] * (random.uniform(-0.2, 0.2))
            new_seq = best_seq.copy()
            new_seq[idx] = max(0.0, new_seq[idx] + delta)
            new_seq_sum = new_seq.sum()
            if new_seq_sum > 0:
                new_seq /= new_seq_sum
                sc = compute_score(new_seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = new_seq.copy()
                    print(f"[{iteration:06d}] Jitter improvement {best_score:.6f}")

    print(f"[done] Best score after {iteration} iterations: {best_score:.6f}")
    return best_seq.tolist()
