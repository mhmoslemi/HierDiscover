# Optional: a small helper that can be used later for a more aggressive
# LP based refinement (not required for the basic search).
def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """
    Solve:
        max   sum(g)
        s.t.  (f_seq * g)[k] <= rhs  for each k = 0..2n‑2
              g >= 0
    Returns the optimal g as a NumPy array or None on failure.
    """
    from scipy.optimize import linprog

    n = len(f_seq)
    # objective: maximise sum(g)  -> minimise -sum(g)
    c = -np.ones(n)

    # Build the matrix A_ub where row k contains f_seq[i] when i+j=k
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start … i_end, j = k-i
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, rhs, dtype=float)

    bounds = [(0, None)] * n

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit},
        )
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    lengths = [300, 600, 1000, 1500]           # different n to explore
    c_grid = np.linspace(0.1, 4.0, 40)        # exponent‑decay parameter
    p_grid = [0.5, 1.0, 2.0, 3.0, 4.0]        # power‑law exponent

    while time.time() < deadline:
        # Choose a length randomly (helps to diversify the search)
        n = int(random.choice(lengths))

        # ---- exponential decay a[i] = λ**i  (λ = exp(-c/n)) ----
        for c in c_grid:
            lam = math.exp(-c / n)
            # a[i] = lam**i  (i = 0 … n‑1)
            seq = (lam ** np.arange(n)).tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq

        # ---- power‑law decay a[i] = (1 - i/(n‑1))**p ----
        for p in p_grid:
            # avoid division by zero when n==1 (never happens here)
            seq = ((1 - np.arange(n) / (n - 1)) ** p).tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq

        # ------------------------------------------------------------------
        # 2.  Small random perturbations of the current best sequence
        # ------------------------------------------------------------------
        if best_seq is not None:
            # make a copy and perturb a few entries
            seq = best_seq.copy()
            for _ in range(5):
                i = random.randint(0, n - 1)
                # add a tiny Gaussian noise; keep non‑negative
                seq[i] = max(0.0, seq[i] + random.gauss(0.0, 0.05))
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq

        # ------------------------------------------------------------------
        # 3.  Optional aggressive refinement using an LP step.
        #     This is cheap for n ≤ 800 and can lower the bound further.
        # ------------------------------------------------------------------
        if best_seq is not None and n <= 800:
            # Normalise the current best to the standard sum √(2n) – the
            # scaling does not affect the evaluation but makes the LP
            # constraints numerically well‑behaved.
            target_sum = math.sqrt(2.0 * n)
            cur_sum = sum(best_seq)
            if cur_sum > 0:
                f_norm = [x * target_sum / cur_sum for x in best_seq]
                # current maximal convolution
                M = max(np.convolve(f_norm, f_norm))
                # solve LP to get a direction that can increase the sum
                g = solve_convolution_lp(f_norm, M, time_limit=2.0)
                if g is not None:
                    sum_g = float(g.sum())
                    # Scale down g so that the bound stays valid (scale ≤ 1)
                    if sum_g > 0:
                        scale = min(1.0, target_sum / sum_g)
                        g_scaled = (g * scale).tolist()
                        # Try several mixing coefficients
                        for t in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                            mixed = [(1 - t) * x + t * y for x, y in zip(f_norm, g_scaled)]
                            score = evaluate_sequence(mixed)
                            if score < best_score:
                                best_score = score
                                best_seq = mixed
                                break

        # stop early if we have already reached a very good bound
        if best_score < 1.6:
            break

    # Fallback – never happens, but keeps the function safe.
    if best_seq is None:
        best_seq = [1.0] * 500

    print(f"Best evaluation score: {best_score:.6f} (length = {len(best_seq)})")
    return best_seq
