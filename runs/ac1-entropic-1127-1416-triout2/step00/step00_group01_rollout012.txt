def propose_candidate(seed=42, budget_s=1000, n=None):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    if n is None:
        # choose length similar to existing best, maybe length of height_sequence_1
        n = len(height_sequence_1)
    # Ensure n at least 100, else set.
    # Optionally try a few lengths.
    best_seq = None
    best_score = float('inf')
    # try a few restarts
    while time.time() < deadline:
        # random initialization
        a = np.random.rand(n)
        a = a / a.sum()   # normalize sum = 1
        # optional small jitter if we start from height_sequence_1
        if np.random.rand() < 0.1 and height_sequence_1:
            a = np.array(height_sequence_1[:n], dtype=float)
            a = a / a.sum()
        # local descent
        # set parameters
        it = 0
        max_iter = 5000
        # learning rate schedule
        base_lr = 0.5
        while it < max_iter and time.time() < deadline:
            conv = np.convolve(a, a)
            max_val = conv.max()
            # compute all indices within 0.999999 of max
            # using tolerance relative
            idxs = np.where(conv >= max_val * (1 - 1e-12))[0]
            # compute gradient: sum of contributions for each idx
            grad = np.zeros_like(a)
            for k in idxs:
                start = max(0, k - (n - 1))
                end = min(k, n - 1)
                # i runs over start..end
                # j = k - i
                # grad[i] += 2 * a[j]
                # For efficiency vectorized:
                i_range = np.arange(start, end + 1)
                j_range = k - i_range
                grad[i_range] += 2.0 * a[j_range]
            # update rates
            lr = base_lr / np.sqrt(it + 1)
            # multiplicative update
            a = a * np.exp(-lr * grad)
            # renormalize to sum 1
            total = a.sum()
            if total <= 0:
                # reinitialize
                a = np.random.rand(n)
                a = a / a.sum()
                continue
            a = a / total
            # evaluate
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best {best_score:.6f} at iter {it}")
            it += 1
        # after local descent, maybe perform a few random perturbations for diversification
    # after time expires, return best_seq as list
    return best_seq.tolist() if best_seq is not None else list(height_sequence_1)

"""Propose candidate step heights using an iterated LP direction and optimal mixing."""
import time
import numpy as np
from scipy.optimize import linprog

def solve_convolution_lp(f, rhs):
    """Solve LP max sum(b) s.t. conv(f,b) <= rhs, b >= 0."""
    n = len(f)
    # Build convolution constraint matrix
    # Represent conv(f,b)[k] = sum_{i} f[i] * b[k-i]
    # For each k, set appropriate coefficients.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        A[k, j_range] = f[i_range]
    c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
    b = np.full(2 * n - 1, rhs, dtype=np.float64)
    # Use high-performance HiGHS solver
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'time_limit': 5})
    if res.success:
        return res.x
    return None

def optimal_mix(t_f, t_g, conv_ff, conv_gg, conv_fg):
    """Return t in [0,1] that minimizes max conv of (1-t)f + t g."""
    lo, hi = 0.0, 1.0
    def max_after(t):
        h = (1 - t) ** 2 * conv_ff + t ** 2 * conv_gg + 2 * t * (1 - t) * conv_fg
        return h.max()
    for _ in range(30):

        m2 = max_after(t2)
        if m1 < m2:
            hi = t2
        else:
            lo = t1

Propose a step‑function (height) sequence that minimises the provided
``evaluate_sequence``.  

The search is based on the classic “LP‑direction + optimal mixing’’ scheme
described in the prompt, with two practical enhancements:

* **Exact 1‑D line search** – instead of the tiny fixed step t=0.01 a
  ternary search finds the mixing coefficient that gives the smallest
  maximal convolution of the combined sequence.
* **Random restarts & jitter** – occasional re‑initialisation of the
  sequence and tiny random perturbations keep the optimisation from
  getting stuck in a local plateau.

The algorithm works on a normalised version of the sequence:
the L1‑norm is forced to ``sqrt(2*n)`` (the convention used in the
reference paper).  This does not affect the objective because the ratio
computed by ``evaluate_sequence`` is scale‑invariant.

The linear programme solves  

    maximise   Σ b_j
    subject to conv(f, b) ≤ M     (M = current max‑convolution of f)
               b ≥ 0

which is linear in ``b``.  The programme is assembled as a dense matrix
because the problem size (≈1500 – 2000 entries) comfortably fits into
memory and the HiGHS solver handles it very quickly.

The best sequence (converted back to a probability vector) is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _build_conv_matrix(f):
    """Return the dense matrix A (shape (2n‑1, n)) such that A @ b = conv(f,b)."""
    n = f.shape[0]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # each row k contains f[i] at column j = k‑i whenever 0≤i,j<n
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        A[k, j_range] = f[i_range]
    return A


def _solve_convolution_lp(f, rhs):
    """
    Solve the LP:
        maximise   Σ b_j
        subject to conv(f, b) ≤ rhs   (for every lag)
                 b ≥ 0
    Returns the optimal b as a 1‑D ``np.ndarray`` or ``None`` on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                # (2n‑1) × n
    c = -np.ones(n, dtype=np.float64)        # minimise –Σb  == maximise Σb
    b = np.full(2 * n - 1, rhs, dtype=np.float64)

    # The bounds enforce b ≥ 0, so we do not need extra inequality rows.
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                  method='highs', options={'time_limit': 5})
    if res.success:

    return None

        h = (1 - t) ** 2 * conv_ff + t ** 2 * conv_gg + 2 * t * (1 - t) * conv_fg

        m2 = _max_after(t2)
        if m1 < m2:

        else:
            lo = t1
    return (lo + hi) / 2.0
