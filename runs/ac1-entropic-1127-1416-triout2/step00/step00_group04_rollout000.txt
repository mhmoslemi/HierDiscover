Search algorithm for step function heights.

Goal: minimize evaluate_sequence which computes 2*n*max(conv(a,a))/(sum a)^2.
Uniform distribution yields minimal possible value 2 (by Cauchy–Schwarz).
The algorithm performs a multi‑start stochastic search with occasional
LP‑based direction moves.  If no improvement is found the uniform
distribution is returned.
"""

import time
import numpy as np
import math
import random

try:
    from scipy import optimize
    linprog = optimize.linprog
except Exception:
    linprog = None

def solve_convolution_lp(f_seq, rhs):
    """Linear program: maximize sum(b) s.t. conv(f_seq, b) <= rhs, b >= 0.
    Returns the optimal b as a numpy array or None on failure."""
    if linprog is None:
        return None
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum(b)
    # Build convolution constraints for each outcome index
    A = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # sum_i f_i * b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)
    # non‑negativity constraints b >= 0 -> -b <= 0
    A.append(-np.eye(n))
    b.append(np.zeros(n))
    A_ub = np.vstack(A)
    b_ub = np.hstack(b)
    # Solve using the GLOP/simplex solver with a short timeout.
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=(0, None),
                      method='highs',
                      options={'time_limit': 5.0, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def get_lp_step(sequence):
    """Return a refined sequence moving towards a better direction using the LP."""
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return None
    # Normalise to satisfy the usual normalisation sqrt(2n) used in the original
    # algorithm; this is only required for the LP formulation.
    norm_factor = math.sqrt(2.0 * n) / sum_seq
    f_norm = [x * norm_factor for x in sequence]
    rhs = max(np.convolve(f_norm, f_norm))

    t = 0.02  # modest step size to ensure stability
    new_seq = [(1 - t) * a + t * b for a, b in zip(sequence, g_rescaled)]
    return new_seq

"""
Multi‑start stochastic optimiser for the step‑function evaluation.

The evaluation function is

    E(a) = 2 * n * max(conv(a,a)) / (sum(a))**2

where `conv` is the ordinary discrete convolution.  By Cauchy–Schwarz
`E(a) >= 2` for any non‑negative vector, and the uniform distribution attains
the bound.  The optimiser therefore tries to find a sequence with the
smallest possible value (the theoretical optimum is 2).  It combines three
simple strategies:

1. Initialise from the historic construction ``height_sequence_1`` (if
   available), from a clean uniform sequence, and from a few short random
   seeds.
2. Periodically compute an LP‑derived direction (the same idea used in the
   reference SOTA algorithm) that attempts to increase the total mass while
   keeping the convolution bound unchanged.
3. Otherwise perform a tiny random perturbation of a single coordinate.

All candidates are kept inside the range `[0, 1000]` (the same range that
`evaluate_sequence` internally clamps to).  The routine runs for the given
budget (default 1000 s) and returns the best sequence discovered.
"""

import time
import random
import math
import numpy as np

# Linear programming utilities are optional – the code works without SciPy.
try:
    from scipy import optimize
    _linprog = optimize.linprog
except Exception:          # pragma: no cover
    _linprog = None


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the linear programme

        maximise   Σ b_j
        subject to max_k Σ_i f_i * b_{k-i} ≤ rhs
                   b_j ≥ 0 .

    Returns the optimal b as a NumPy array or ``None`` if LP solving fails.
    """
    if _linprog is None:
        return None

    n = len(f_seq)
    # objective – we maximise Σb, so we minimise –Σb
    c = -np.ones(n)

    # Build convolution constraints: one for each possible shift k.
    A = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)

    # non‑negativity b ≥ 0  →  –b ≤ 0
    A.append(-np.eye(n))
    b.append(np.zeros(n))

    A_ub = np.vstack(A)
    b_ub = np.hstack(b)

    # Use the high‑performance “highs” solver with a tiny time limit.
    try:
        res = _linprog(c,
                       A_ub=A_ub,
                       b_ub=b_ub,
                       bounds=(0, None),
                       method='highs',
                       options={'time_limit': 5.0, 'presolve': True})
        if res.success:
            return res.x
    except Exception:       # pragma: no cover
        pass
    return None


def get_lp_step(sequence):
    """
    Produce a refined sequence by moving a small amount towards the LP
    solution that keeps the convolution bound unchanged.
    """
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return None

    # Normalise to the convention used in the reference (Σa = √(2n)).
    scale = math.sqrt(2.0 * n) / sum_seq
    f_norm = [x * scale for x in sequence]

    # Current maximal convolution value.
    rhs = max(np.convolve(f_norm, f_norm))

    # Solve the LP for a new vector g.
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None

    t = 0.02
