def project_simplex(v):
    # ensure v sums to 1 and v_i >= 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    max_val = np.max(c)
    # get indices where close to max
    mask = np.isclose(c, max_val, atol=1e-12)
    grads = np.zeros_like(a)
    # for each k in max indices:
    for k in np.where(mask)[0]:
        # gradient term for each i such that 0 <= k - i < n
        # add 2 * a[k - i] to grads[i]
        # we can vectorized: idx_i = np.arange(n)
        # idx_j = k - idx_i
        # valid mask: (0 <= idx_j < n)
        i = np.arange(len(a))

        valid = (j >= 0) & (j < len(a))
        grads[i[valid]] += 2 * a[j[valid]]

    grads = grads / np.sum(mask)  # average across max indices

    # project onto simplex
    a_new = project_simplex(a_new)

def project_simplex(v):
    # projecting onto simplex of sum 1
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
projected subgradient search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works on the probability simplex (sum == 1).  The objective
becomes 2·n·max(conv) and is invariant under scaling, so we optimise the
shape only.  Starting from many random Dirichlet draws we perform a
projected sub‑gradient descent on the function f(a)=max_k (a*a)[k].
The gradient at the current maximal index k* is

    g_i = 2 * a_{k* - i}   (if the index is inside the vector, else 0)

which is the sub‑gradient of the max‑convolution.  After a step we project
back onto the simplex using the O(n log n) Duchi projection.  The step size
decays like 1/√t and we accept any improvement; otherwise a small random
perturbation is applied to escape plateaus.  Multiple restarts are performed
until the time budget expires.  The best sequence found is finally scaled
so that its L1‑norm equals √(2n) (the normalisation used in the original
code).  The returned list can be directly fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex using the algorithm of Duchi et al."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i > (cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full(n, 1.0 / n, dtype=float)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine
def _subgradient_descent(
    a: np.ndarray,
    n: int,
    max_iter: int,
    time_deadline: float,
    global_best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a projected sub‑gradient descent starting from a (which is already on the simplex).
    Returns the best vector found in this run and its score.
    """
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max()   # internal (scale‑invariant) score
    step0 = 1.0                         # initial step magnitude

    for it in range(max_iter):
        if time.time() >= time_deadline:
            break

        # current convolution and maximal index(es)
        conv = np.convolve(a, a)
        max_val = conv.max()
        max_idx = np.where(np.isclose(conv, max_val, atol=1e-12))[0]

        # sub‑gradient of the max‑convolution (averaged over all maximal indices)
        g = np.zeros_like(a)
        for k in max_idx:
            # i ranges where both i and k-i are valid
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i = np.arange(i_min, i_max + 1, dtype=int)
            j = k - i
            g[i] += 2.0 * a[j]                 # contribution 2*a_{k-i}
        g /= len(max_idx)                       # average if several maximisers

        # direction that preserves the sum (sum(direction) == 0)
        direction = -g + np.mean(g)

        # decreasing step size
        step = step0 / math.sqrt(it + 1.0)

        # try to move; if no improvement shrink step a few times
        improved = False
        for _ in range(5):
            a_candidate = _project_simplex(a + step * direction)
            cand_score = 2.0 * n * np.convolve(a_candidate, a_candidate).max()
            if cand_score < best_score:
                a = a_candidate
                best_score = cand_score
                best_a = a_candidate.copy()
                improved = True
                break
            else:
                step *= 0.5  # reduce step and retry

        if not improved:
            # occasional random jitter to escape possible plateaux
            jitter = np.random.randn(n) * 0.005
            a = _project_simplex(a + jitter)

        # keep track of the global best (used for early stopping)
        if best_score < global_best_score:
            # early exit signal – caller will notice via the returned score
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that yields a low evaluation score.
    Uses many random restarts of a projected sub‑gradient descent on the L∞‑norm
    of the self‑convolution (the quantity inside ``evaluate_sequence``).
    The final sequence is rescaled so that its L1‑norm equals √(2·n) as in the
    reference implementation.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a few seconds safety margin

    # Length of the sequence – fall back to a modest size if the global variable is missing
    try:
        n = len(height_sequence_1)            # provided by the environment
    except Exception:
        n = 800                               # a reasonable default

    # Global best across all restarts
    global_best_seq = None
    global_best_score = float("inf")

    restart_counter = 0
    while time.time() < deadline:
        restart_counter += 1

        # --- random start on the simplex ---------------------------------
        # Dirichlet with concentration < 1 encourages a variety of shapes
        alpha = np.full(n, 0.4) + np.random.rand() * 0.1   # slight randomness per restart
        a = np.random.dirichlet(alpha)

        # Run a limited number of inner iterations (adjustable)
        max_inner = 20000
        a_best_local, score_local = _subgradient_descent(
            a, n, max_inner, deadline, global_best_score
        )

        # Update the global best if we improved
        if score_local < global_best_score:
            global_best_score = score_local
            global_best_seq = a_best_local.copy()
            elapsed = time.time() - start_time
            print(
                f"[{elapsed:6.1f}s] Restart {restart_counter:3d} – new best score: {global_best_score:.6f}"
            )
            # Early termination if we already beat the target bound (≈1.503)
            if global_best_score <= 1.55:
                # small margin to allow a quick final scaling pass
                break

    # In the unlikely event that nothing was found, fall back to a uniform vector
    if global_best_seq is None:
        global_best_seq = np.full(n, 1.0 / n, dtype=float)

    # ---------------------------------------------------------------
    # Final scaling to the normalisation used in ``evaluate_sequence``:
    # sum must be sqrt(2·n).  Scaling does not change the evaluation value,
    # but keeps the sequence in a familiar range.
    target_sum = math.sqrt(2.0 * n)
    current_sum = np.sum(global_best_seq)
    scale = target_sum / current_sum
    final_seq = (global_best_seq * scale).tolist()

    # Optional sanity check (does not affect the returned sequence)
    try:
        val = evaluate_sequence(final_seq)
        print(f"Final evaluated score (using evaluate_sequence): {val:.6f}")
    except Exception as e:
        # Should never happen – evaluate_sequence guards against bad inputs.
        print("Evaluation failed:", e)

    return final_seq
