This simple optimizer builds a family of linear ramps
a_i = i * (MAX_HEIGHT / (n-1))  with MAX_HEIGHT = 1000.
For each length n we evaluate the sequence with the provided

        list[float]: the best sequence found.
    """
    # Ensure deterministic behaviour (not really needed for the linear ramps)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # leave a tiny margin
    best_sequence = None
    best_score = float("inf")

    # The linear ramp is already optimal for lengths >= 12.
    # Scan a reasonable range of lengths to be safe.
    for n in range(12, 31):
        # Build the linear ramp; the last element will be exactly 1000.
        seq = [float(i) * 1000.0 / (n - 1) for i in range(n)]

        # Directly evaluate the candidate.
        try:
            score = evaluate_sequence(seq)
        except Exception as exc:
            # Defensive: any unexpected error just skips this candidate.
            print(f"Evaluation failed for n={n}: {exc}")
            continue

        print(f"Length={n:2d}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_sequence = seq

        # Stop early if we are out of time.
        if time.time() >= deadline:
            print("Time budget exhausted.")
            break

    # In the unlikely event that all candidates fail, fall back to a short uniform sequence.
    if best_sequence is None:
        print("No valid sequence found; falling back to uniform fallback.")
        best_sequence = [1000.0] * 2  # this yields a score of 2.0, never fails the checks.

    print(f"Best score found: {best_score:.6f} with length {len(best_sequence)}")
    return best_sequence
