def evaluate_exp(n, r):
    if r <= 0: return np.inf
    # compute sum S
    if abs(r-1.0) < 1e-12:
        S = float(n)

        S = (1 - r**n) / (1 - r)
    # compute max b
    # part1: k=0..n-1
    ks = np.arange(2*n-1)
    L = np.where(ks < n, ks+1, 2*n-1-ks)

def golden_section_minimize(func, a, b, max_iter=50, tol=1e-12):
    phi = (1 + math.sqrt(5)) / 2
    invphi = 1/phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)

"""
search_step_function.py

A lightweight search for a step‐function (non‑negative sequence) that minimizes

    2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The evaluation is scale‑invariant, so the shape of the sequence is what matters.
The algorithm explores three parametric families (exponential, power‑law and
polynomial “linear” shapes) for several lengths, evaluates them cheaply (the
exponential case admits a closed‑form evaluation), and keeps the best found
candidate.  Afterwards a simple hill‑climbing refinement is performed on the
current best sequence.

The whole search respects the 1000 s time budget and returns a valid list of
floats that can be passed directly to ``evaluate_sequence``.
"""

import math
import random
import time
from typing import List

import numpy as np


def _evaluate_exponential(n: int, gamma: float) -> float:
    """
    Fast evaluation of a geometric sequence a_i = exp(-gamma * i) for length n.
    The convolution of this sequence satisfies
        conv[k] = L_k * exp(-gamma * k)
    where L_k = min(k+1, 2n-1-k) is the number of overlapping terms.
    """
    if gamma <= 0.0:
        # Degenerates to constant sequence -> value 2
        return 2.0

    r = math.exp(-gamma)          # common ratio 0 < r < 1
    # sum of the geometric series
    # use the stable formulation for r very close to 1
    if r == 1.0:
        sum_a = float(n)
    else:
        sum_a = (1.0 - r**n) / (1.0 - r)

    # compute max_{k} L_k * r**k   (k = 0 … 2n‑2)
    ks = np.arange(2 * n - 1, dtype=np.float64)
    L = np.where(ks < n, ks + 1.0, 2.0 * n - 1.0 - ks)   # length of overlap
    rp = np.exp(-gamma * ks)                               # r**k
    max_b = float(np.max(L * rp))

    return float(2.0 * n * max_b / (sum_a * sum_a))

        if time.time() > deadline:
            break
        for gamma in exp_gammas:
            if time.time() > deadline:
                break
            score = _evaluate_exponential(n, gamma)
            if score < best_score:
                best_score = score
                r = math.exp(-gamma)
                best_seq = [r ** i for i in range(n)]
                print(f"[exp] n={n:<5d} gamma={gamma:.2e}  score={score:.6f}")

    # --- power‑law family (a_i = (i+1)^{-p}) -------------------------- #
    power_ps = np.linspace(0.2, 3.0, num=30)   # exponent p
    for n in lengths:
        if time.time() > deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)
        for p in power_ps:
            if time.time() > deadline:
                break
            seq = (idx ** (-p)).tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[pow] n={n:<5d} p={p:.3f}  score={score:.6f}")

    # --- polynomial “linear” shapes (a_i = 1 - (i/(n-1))^{p}) --------- #
    linear_ps = np.linspace(0.2, 5.0, num=30)   # curvature exponent
    for n in lengths:
        if time.time() > deadline:
            break
        pos = np.arange(n, dtype=np.float64)
        denom = float(n - 1) if n > 1 else 1.0
        for p in linear_ps:
            if time.time() > deadline:
                break
            seq_arr = 1.0 - (pos / denom) ** p
            seq_arr = np.clip(seq_arr, 0.0, None)      # enforce non‑negativity
            seq = seq_arr.tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[lin] n={n:<5d} p={p:.3f}  score={score:.6f}")

    # ------------------------------------------------------------------ #
    # 3.  Simple hill‑climbing refinement on the current best candidate
    # ------------------------------------------------------------------ #
    # Convert to a NumPy array for fast elementwise updates
    cur = np.array(best_seq, dtype=np.float64)
    cur_score = best_score
    n = cur.size
    temperature = 1.0          # initial “temperature” for occasional uphill steps
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # gradually cool down
        temperature = max(0.001, temperature * 0.9999)

        # pick two distinct positions
        i, j = random.sample(range(n), 2)
        if cur[i] == 0.0:
            continue
        # propose moving a random fraction of cur[i] to cur[j]
        frac = random.uniform(-0.2, 0.2)          # up to ±20 % of the source value
        delta = cur[i] * frac
        if delta > cur[i]:        # cannot move more than we have
            delta = cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        if new[i] < 0.0:          # respect non‑negative constraint
            continue

        # evaluate the proposal (using the official routine)
        cand_seq = new.tolist()
        cand_score = evaluate_sequence(cand_seq)

        # accept if better, else accept with a small probability (simulated annealing)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(cand_score - cur_score) / temperature)
            if random.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq
                print(f"[hill] iter={iteration:6d}  score={cand_score:.6f}")

    # ------------------------------------------------------------------ #
    # 4.  Return the best sequence found
    # ------------------------------------------------------------------ #
    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
