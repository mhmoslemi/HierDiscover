   dead = time.time() + budget_s - 5

   best_seq = list(height_sequence_1)  # fallback start
   best_score = compute_score(best_seq)

   # define lengths to explore
   lengths = [200, 500, 1000, 1500, 2000]  # maybe limited by compute.

   for n in lengths:
       # random init series
       a = np.random.rand(n)
       a /= a.sum()
       # optionally start from uniform + noise
       # iterate for limited steps
       step = 0.05
       while time.time() < dead:
           # compute conv
           conv = np.real(np.fft.ifft(np.fft.fft(a, 2*n) * np.fft.fft(a, 2*n)))[:2*n-1]
           M = conv.max()
           # compute gradient
           k_max = conv.argmax()
           # compute g
           g = np.zeros_like(a)
           low = max(0, k_max-(n-1))
           high = min(n-1, k_max)
           idx = np.arange(low, high+1)
           g[idx] = a[k_max - idx]
           # gradient of f
           # sum is 1
           grad_f = (2 * n) * (g - 2 * M * np.ones_like(a))  # because sum=1 , S=1

           # perhaps scaling
           # step size adaptation: compute step*grad_f and adjust if new score not improved
           # try candidate step sizes
           new_a = a - step * grad_f
           # project onto simplex
           new_a = project_simplex(new_a)

           # compute new score
           new_score = compute_score(new_a)
           # If improved accept with step maybe increase
           if new_score < best_score:
               best_score = new_score
               best_seq = new_a.copy()
               print(f'Improved score {best_score:.6f} n={n}')
               a = new_a

                   a /= a.sum()

   # After loops return best_seq as list

Implementation: Duchi et al algorithm.

def project_simplex(v):
    """Project vector v onto simplex {x >=0, sum x =1}."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function sequence.

The evaluation metric after normalising the sum of the coefficients to 1 is

    score = 2 * n * max_k (p * p)[k]                     (1)

where *p* is the normalised non‑negative vector and (p*p) denotes the
auto‑convolution.  Hence the problem reduces to minimising the maximum
entry of the self‑convolution of a probability vector.
The algorithm performs a projected sub‑gradient descent on the quantity

    M(p) = max_k (p * p)[k] .

* Sub‑gradient of M w.r.t. p at the active index k* is
      g_i = p_{k*-i}   (zero outside the support) .
* A step is taken in the direction –g, the vector is projected onto the
  probability simplex (non‑negative, unit‑sum) and the new maximum
  convolution value is evaluated.
* An adaptive step size (increase on improvement, decrease on failure)
  together with occasional random restarts is used.
* The procedure is executed for several problem sizes (n) within the
  allowed time budget; the best sequence found overall is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Small utility: next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0          # cumulative sum minus 1
    ind = np.arange(1, n + 1)
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(p: np.ndarray) -> float:
    """
    Compute the maximum entry of the auto‑convolution p * p.
    Uses an FFT of length power‑of‑two for speed.
    """
    n = p.shape[0]
    size = _next_pow_two(2 * n - 1)
    # real FFT (more efficient than complex FFT)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fp * fp, size)[: 2 * n - 1]
    return float(conv.max())


def _subgradient_step(p: np.ndarray, step: float) -> (np.ndarray, float):
    """
    Perform one sub‑gradient step for the objective max(p*p).
    Returns the new probability vector and the corresponding max‑convolution.
    """
    n = p.shape[0]
    size = _next_pow_two(2 * n - 1)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fp * fp, size)[: 2 * n - 1]
    M = float(conv.max())
    k = int(conv.argmax())

    # sub‑gradient of the active entry
    g = np.zeros_like(p)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = p[k - idx]

    # move opposite to the gradient, then project back onto the simplex
    p_new = p - step * g
    p_new = _project_simplex(p_new)
    # recompute the objective for the new point
    new_M = _conv_max(p_new)
    return p_new, new_M


def _score_from_p(p: np.ndarray) -> float:
    """
    Direct score according to evaluate_sequence for a normalised vector.
    Since sum(p) = 1, the denominator is 1 and the score is simply 2*n*M.
    """
    n = p.shape[0]
    M = _conv_max(p)
    return 2.0 * n * M
