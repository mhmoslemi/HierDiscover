def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # initialize
    if 'height_sequence_1' in globals():
        seq = np.array(list(height_sequence_1), dtype=float)
    else:
        # random length between 500 and 2000, values uniform in [0,1]
        n = np.random.randint(500, 2000)
        seq = np.random.rand(n)
    # optionally normalize? Not needed.
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # functions for gradient step
    def gradient_step(a):
        n_len = len(a)
        b = np.convolve(a, a)
        max_b = b.max()
        kstar = b.argmax()
        sum_a = a.sum()
        if sum_a == 0:
            return a
        C = 2.0 * max_b / sum_a
        # build u vector: a shifted by kstar
        u = np.zeros_like(a)
        # indices i where 0 <= kstar - i < n_len => i in [kstar - (n_len-1), kstar]
        i_start = max(0, kstar - (n_len - 1))
        i_end = min(n_len - 1, kstar)
        # compute u[i] = a[kstar - i] for these i
        # Could vectorize:
        idx = np.arange(i_start, i_end + 1)
        u[idx] = a[kstar - idx]
        # compute descent direction
        d = - (a + u) + C
        return d
    # main loop
    iter_count = 0
    while time.time() < deadline:
        # compute gradient direction
        d = gradient_step(seq)
        # line search for step size
        # try a few alphas
        best_local_seq = seq
        best_local_score = best_score
        # maybe compute candidate alphas logspace: [0, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0] but cap not exceed
        for alpha in [0.0, 0.005, 0.01, 0.025, 0.05, 0.1, 0.2, 0.5]:
            if alpha == 0.0:
                cand = seq.copy()
            else:
                cand = seq + alpha * d

def simplex_projection(v):
    # v is np.ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Simple yet effective search for a step‑function with a low 
2·n·max(conv)/sum² value.

The algorithm combines three ideas:

1. **Random start** – initialise a non‑negative vector with the
   required L¹ norm (√(2·n)).  A uniform vector gives the
   classical bound 2.0, but random starts allow the optimiser to
   explore other shapes.

2. **Maximum‑convolution flattening** – at each iteration the shift
   `k*` where the convolution attains its maximum is identified.
   The pairs (i, j) that contribute the most to this peak are
   slightly weakened while the removed mass is redistributed to the
   smallest entries.  This “mass‑shifting’’ step strives to equalise
   the convolution values and therefore to reduce the overall bound.

3. **Occasional linear‑programming boost** – the LP used in the
   reference implementation is called only rarely (every few hundred
   iterations).  It produces a direction `g` that can increase the
   L¹ norm without increasing the current peak `M`.  A short line‑search
   over mixing parameters `t` then tries to improve the objective.

The procedure runs until the time budget expires (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def solve_convolution_lp(f_seq: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    LP: maximise Σ b_j   s.t.   (f_seq * b)[k] ≤ rhs  for all k,
    and b ≥ 0.  Returns the optimal b (as a NumPy array) or None.
    """
    n = f_seq.shape[0]
    c = -np.ones(n)                     # maximise sum(b)
    # build A_ub, b_ub for the convolution constraints
    a_rows = []
    b_vals = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # b_j appears in (f*b)[k] iff there exists i with i+j = k
        # i runs over indices where 0 ≤ i < n and 0 ≤ k-i < n
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # For every valid i we have f[i] * b[k-i]
        # Therefore the coefficient of b[j] is f[k-j] if that index exists.
        # We fill the row by looping over i (fast enough for n≈5k).
        for i in range(i_low, i_high + 1):
            j = k - i
            row[j] = f_seq[i]
        a_rows.append(row)
        b_vals.append(rhs)
    # non‑negativity constraints  -b ≤ 0
    a_rows.append(-np.eye(n))
    b_vals.append(np.zeros(n))
    A_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_vals)

    # solve the LP – a very short time limit prevents hanging
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=(0, None),
        method='highs',
        options={'time_limit': 8.0, 'presolve': True},
    )
    if res.success:
        return np.clip(res.x, 0, 1000.0)   # respect the global caps
    return None


def flatten_step(a: np.ndarray, k_max: int, η: float = 0.01) -> np.ndarray:
    """
    Reduce the contribution to the current maximal convolution shift `k_max`
    by moving a small fraction `η` of the mass from the largest contributing
    pairs to the currently smallest entries.
    """
    n = a.shape[0]
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)

    i = np.arange(i_start, i_end + 1, dtype=int)
    j = k_max - i

    prod = a[i] * a[j]                     # contribution a_i * a_j
    # number of pairs to modify – at least one, up to 1 % of all pairs
    n_pairs = i.shape[0]
    n_adj = max(1, int(0.01 * n_pairs))

    # indices of the largest contributions
    top_idx = np.argpartition(prod, -n_adj)[-n_adj:]

    # total mass that will be taken away
    removed = 0.0

    # weaken the selected pairs
    for idx in top_idx:
        ii = i[idx]
        jj = j[idx]
        if ii == jj:                       # diagonal element (i=j)
            delta = a[ii] * η
            a[ii] -= delta
            removed += delta
        else:
            delta_i = a[ii] * η
            delta_j = a[jj] * η
            a[ii] -= delta_i
            a[jj] -= delta_j
            removed += delta_i + delta_j

    # distribute the removed mass to the smallest entries
    if removed > 0:
        low_idx = np.argpartition(a, int(0.05 * n))[:int(0.05 * n)]
        inc = removed / low_idx.size
        a[low_idx] += inc

    # keep everything in the allowed range
    a = np.clip(a, 0, 1000.0)
    return a

            best_seq = a.copy()
            print(f"[update] iter={iter_cnt}, new best={cur_score:.6f}")

        # --- (f) occasional random shake‑up to escape plateaus -------------------
        if iter_cnt % 150 == 0 and iter_cnt > 0:
            # pick a small random subset and perturb it
            shake_idx = np.random.choice(n, size=int(0.02 * n), replace=False)
            a[shake_idx] = np.random.rand(shake_idx.size) * np.max(a)
            a = a / a.sum() * np.sqrt(2.0 * n)

        iter_cnt += 1

    print(f"[finished] best score={best_score:.6f}, length={best_seq.shape[0]}")
    return best_seq.tolist()
