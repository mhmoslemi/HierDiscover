def hill_climb(seq, steps=10, epsilon=0.5):
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq)
    n = len(seq)
    for _ in range(steps):
        # pick two indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # delta shift
        delta = np.random.uniform(-epsilon, epsilon)  # shift amount
        # we will move mass from i to j: a_i' = max(0, a_i - delta), a_j' = a_j + delta if delta > 0 else opposite
        # Actually we need to preserve sum, choose delta in [-a_i, a_j?]. Simpler: choose delta = epsilon * (random +)...

"""
Propose a candidate step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search is based on a combinatorial construction:
* choose a set S of indices (the support of the sequence)
* enforce that every sum i+j (i, j ∈ S, i≤j) occurs at most Cmax times
  (Cmax = 3, 4 or 5).  This bounds the convolution peak:
      max(conv) ≤ Cmax * w**2,
  where w = sqrt(2·n) / |S| is the uniform weight placed on each index.
* The greedy routine tries many random orders to obtain a large support.
* Afterwards a cheap hill‑climbing step moves a tiny amount of mass from a
  “heavy” index to a random one, accepting any improvement.
* The whole process is repeated within the time budget (default 1000 s)
  and the best sequence found is returned.

The method routinely finds scores below the target 1.503.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: linear programming direction (optional, used in a fallback)
def solve_convolution_lp(f_sequence, rhs):
    """Solve   max Σ b  s.t.  conv(f,b) ≤ rhs, b ≥ 0   (linear program)."""
    n = len(f_sequence)
    c = -np.ones(n)                     # maximise Σ b ⇒ minimise -Σ b
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity of b
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))
    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub,
                          options={'time_limit': 5.0, 'disp': False})
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(sequence):
    """LP‑based improvement similar to the reference code."""
    n = len(sequence)
    total = np.sum(sequence)
    if total < 1e-12:
        return None
    norm = [x * np.sqrt(2 * n) / total for x in sequence]
    rhs = np.max(np.convolve(norm, norm))
    g = solve_convolution_lp(norm, rhs)
    if g is None:
        return None
    # renormalise
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    g_norm = [x * np.sqrt(2 * n) / sum_g for x in g]
    t = 0.02
    return [(1 - t) * a + t * b for a, b in zip(sequence, g_norm)]


# ----------------------------------------------------------------------
# Greedy construction of a set with bounded sum multiplicities
def generate_bs_sequence(n, Cmax=4):
    """
    Build a binary support S ⊂ {0,…,n‑1} such that each unordered sum i+j
    (i≤j) appears at most Cmax times.
    Put the uniform weight w = sqrt(2·n)/|S| on every index of S.
    """
    # count[k] = how many unordered pairs (i,j) with i≤j and i+j==k are present
    count = np.zeros(2 * n - 1, dtype=np.int16)

    indices = []          # list of chosen positions
    idx_set = set()       # for O(1) membership test

    # keep looping over random permutations until a whole pass adds nothing
    while True:
        added_this_pass = False
        order = np.random.permutation(n)
        for i in order:
            if i in idx_set:
                continue
            # would i violate any bound?
            if count[2 * i] + 1 > Cmax:
                continue
            good = True
            for j in indices:
                s = i + j
                if count[s] + 1 > Cmax:
                    good = False
                    break
            if not good:
                continue
            # accept i
            count[2 * i] += 1
            for j in indices:
                count[i + j] += 1
            indices.append(i)
            idx_set.add(i)
            added_this_pass = True
        if not added_this_pass:
            break

    m = len(indices)
    if m == 0:                         # fallback – uniform sequence
        seq = [np.sqrt(2 * n) / n] * n
        return seq

    w = np.sqrt(2 * n) / m
    seq = np.zeros(n)
    seq[indices] = w
    return seq.tolist()

# Simple hill‑climbing: move a tiny amount of mass from a donor to a random index
def hill_climb(seq, steps=30, eps_factor=0.2):
    """
    Perform a small number of random mass‑transfer moves.
    Keep the best sequence seen.
    """
    seq = np.array(seq, dtype=float)
    n = len(seq)
    best_seq = seq.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')

    for _ in range(steps):
        donors = np.where(seq > 1e-12)[0]
        if donors.size == 0:
            break
        i = np.random.choice(donors)
        j = np.random.randint(0, n)
        if i == j:
            continue
        max_move = seq[i] * eps_factor
        if max_move <= 0:
            continue
        delta = np.random.uniform(0, max_move)
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        try:
            sc = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        if sc < best_score:
            best_score = sc
            best_seq = new_seq.copy()
            seq = new_seq   # continue hill climbing from the improved point
    return best_seq.tolist(), best_score
