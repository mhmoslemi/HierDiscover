def perturb(seq):
    new = seq.copy()
    # choose operation
    op = np.random.choice(['transfer', 'redistribute', 'swap', 'scale'])
    if op == 'transfer':
        i, j = np.random.choice(len(seq), size=2, replace=False)
        delta = step_frac * new[i]
        if delta < 1e-12: return new
        new[i] = max(0, new[i] - delta)
        new[j] += delta
    elif op == 'redistribute':
        k = np.random.randint(2, 6) # how many indices
        idx = np.random.choice(len(seq), size=k, replace=False)
        total = new[idx].sum()
        if total == 0:
            # set random Dirichlet
            new[idx] = np.random.dirichlet(np.ones(k))
        else:
            # generate new random proportions
            props = np.random.dirichlet(np.ones(k))
            new[idx] = total * props
    elif op == 'swap':
        i, j = np.random.choice(len(seq), size=2, replace=False)
        new[i], new[j] = new[j], new[i]
    elif op == 'scale':
        idx = np.random.choice(len(seq))
        factor = 1 + (np.random.rand() - 0.5) * step_frac  # small scaling
        new[idx] = max(0, new[idx] * factor)
        # optionally renormalize full seq to keep sum=1.
    # ensure sum>0
    return new

def compute_score(seq):
    a = np.array(seq, dtype=np.float64)
    # clip to bounds
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()

    # convolution via FFT
    n = len(a)

    Fa = np.fft.rfft(a, n=size)

    max_b = conv.max()
    return float(2 * n * max_b)

"""
Fast heuristic for minimizing 2 * n * max(conv(a,a)) / (sum(a) ** 2).

The algorithm works in three stages:
1.  Generate a few simple parametric families (uniform, half‑sine,
    exponential, beta‑like) and keep the best.
2.  Run a short “LP‑direction” loop: given the current sequence f,
    solve a linear program that finds a non‑negative sequence g with a
    larger ℓ₁‑norm while keeping the convolution f * g bounded by the
    current maximum.  The new sequence is a convex combination
    (1‑t)·f + t·g.  Several values of t are tried and the best one is
    kept.
3.  If the LP fails or no improvement is found, apply a tiny random
    perturbation (preserving non‑negativity) to escape a possible
    local minimum.

All heavy work (convolution, evaluation) is performed with an FFT
based implementation that is equivalent to `evaluate_sequence`.  The
function `propose_candidate` respects the given time budget (default
≈1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: fast evaluation (identical to evaluate_sequence but faster)
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return the objective value for a non‑negative sequence.

    This is equivalent to evaluate_sequence(seq) but uses an FFT for
    speed.  The sequence is clipped to the allowed range [0,1000] and
    a zero‑check on the sum is performed for safety.
    """
    # protect against NaNs / infinities (mirrors evaluate_sequence)
    if np.any(np.isnan(seq)) or np.any(np.isinf(seq)):
        return float('inf')
    # clip as evaluate_sequence does
    a = np.clip(seq, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:                     # enforce the same sanity check
        return float('inf')
    # normalise – scaling does not change the objective, but it avoids
    # overflow in the FFT when numbers become huge.
    a = a / s
    n = a.shape[0]

    # convolution via real FFT (length = next power of two >= 2n-1)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    # undo the normalisation (the formula is homogeneous, so we can
    # directly use the normalised values)
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Linear‑programming direction (identical to the SOTA step)
# ----------------------------------------------------------------------
def solve_convolution_lp(f_norm: np.ndarray, rhs: float) -> np.ndarray | None:
    """Linear program:
       maximise   sum(b)
       subject to  conv(f_norm, b) <= rhs ,  b >= 0
    Returns the optimal b (as a dense ndarray) or None on failure.
    """
    n = f_norm.shape[0]
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # Build the matrix for the convolution inequalities.
    # Row k (0 <= k < 2n-1) contains the coefficients of b_j that
    # appear in (f_norm * b)_k .
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices of f_norm, j = k - i for b
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For every i in [i_min, i_max] we have j = k - i in range.
        row_indices = np.arange(i_min, i_max + 1)
        j = k - row_indices
        row[j] = f_norm[row_indices]
        rows.append(row)

    # non‑negativity constraints:  -b <= 0
    rows.append(-np.eye(n))
    A_ub = np.vstack(rows)
    b_ub = np.hstack([np.full(2 * n - 1, rhs), np.zeros(n)])

    # Use the reliable HiGHS simplex solver (fast for moderate size)
    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"presolve": True, "time_limit": 10.0},
    )
    return res.x if res.success else None


def lp_direction(current: np.ndarray) -> np.ndarray | None:
    """Return a new candidate sequence obtained from one LP step.
    The returned array is not yet mixed with the current one.
    """
    n = current.shape[0]
    s = current.sum()
    if s < 0.01:
        return None

    # Normalise to the standard sum sqrt(2n) – the exact value does not
    # matter, only that the bound M is computed on the same scale.
    scale = np.sqrt(2.0 * n) / s
    f_norm = current * scale
    rhs = np.max(np.convolve(f_norm, f_norm))

    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # The LP returns a vector on the normalised scale.  Bring it back
    # to the original scale (so that the ℓ₁‑norm of g matches that of
    # the normalised f).  Afterwards the caller will mix it with the
    # unscaled current sequence.
    g = g * (s / np.sqrt(2.0 * n))
    return g

def family_half_sine(n: int) -> np.ndarray:

    x = np.exp(-rate * np.arange(n))

    i = np.arange(n) + 1
    x = (i ** (a - 1)) * ((n - i + 1) ** (b - 1))
