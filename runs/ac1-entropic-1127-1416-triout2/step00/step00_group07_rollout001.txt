        init_seq = [1.0] * n
        sum_init = n

    current = init_seq

        sum_a = np.sum(arr)
        if sum_a < 1e-6:

Algorithm:
- Start from known sequence or random.
- Normalize sum to sqrt(2n) (invariant under scaling).
- Repeatedly try two improvement moves:
   * Solve a linear program to find a promising direction that maximizes total mass under current convolution bound.
   * Mix the current sequence with the LP solution using several step sizes and keep the best improvement.
   * Perform a random mass transfer between two positions (preserve total sum) with adaptive step size; accept if it lowers the score.
- Keep track of the best sequence seen.
- Stop before the time budget expires and return the best sequence.

def compute_score(seq):
    arr = np.asarray(seq, dtype=np.float64)
    conv = np.convolve(arr, arr)
    maxb = np.max(conv)
    sum_a = np.sum(arr)
    if sum_a < 1e-12:
        return float('inf')
    n = len(seq)
    return 2.0 * n * maxb / (sum_a * sum_a)

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    rows = []
    b = []
    # convolution constraints
    for k in range(2*n - 1):
        row = np.zeros(n)
        # b_j enters where j ranges valid
        # For each i in 0..n-1, j = k - i, if 0 <= j < n then coefficient f_i
        # vector: sum_i f_i * b_{k-i}
        # So row[j] = f_i
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
        b.append(rhs)
    # nonnegative constraints: -I * b <= 0
    rows.append(-np.eye(n))
    b.append(np.zeros(n))
    A_ub = np.vstack(rows)
    b_ub = np.hstack(b)

    return None

"""
Propose a non‑negative step‑function (list of heights) that gives a low
value of

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the discrete self‑convolution.
The algorithm mixes three cheap search ideas:

1. **Linear‑programming direction** – solve a small LP that maximises the
   total mass while keeping the convolution with the current sequence
   bounded by its present maximum.  The LP solution is mixed with the
   current sequence using a few step sizes; any improvement is kept.

2. **Targeted mass transfer** – identify the index `k` where the current
   convolution is maximal, then move a little mass from a location that
   contributes heavily to that peak to a random location.  This
   directly attacks the current bottleneck.

3. **Random balanced perturbation** – move a small amount of mass from
   one random position to another while preserving the total sum.

The search is time‑limited (default 1000 s) and keeps the best sequence
found.  All numbers are kept in the allowed range [0, 1000]; the final
sequence is returned unchanged – the external grader will evaluate it
with `evaluate_sequence`.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: compute the objective value (identical to evaluate_sequence
# except for the safety‑checks and clipping, which we keep under control).
# ----------------------------------------------------------------------
def _score(seq):
    arr = np.asarray(seq, dtype=np.float64)
    conv = np.convolve(arr, arr)
    maxb = np.max(conv)
    s = np.sum(arr)
    if s < 1e-12:
        return float("inf")
    return 2.0 * len(seq) * maxb / (s * s)


# ----------------------------------------------------------------------
# Solve the LP
#   maximise  Σ b_j
#   s.t.    (f * b)_k ≤ rhs   for all k
#           b_j ≥ 0
# The LP is linear because (f * b)_k = Σ_i f_i * b_{k-i} is linear in b.
# ----------------------------------------------------------------------
def _solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    # objective: maximise sum(b) → minimise –sum(b)
    c = -np.ones(n)

    # Build convolution constraints.
    A = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)

    A = np.vstack(A)
    b = np.array(b)

    # Use the HiGHS interior‑point solver (fast for moderate sizes).
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if res.success:
        return res.x
    return None

# Main entry point required by the competition harness.
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000):
    """
    Return a list of non‑negative floats (the step heights) that tries to
    minimise the evaluation function.  The routine runs for at most
    ``budget_s`` seconds and prints occasional progress messages.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Starting sequence – use the known candidate if present,
    #    otherwise a random one.  Truncate very long sequences to keep
    #    the LP affordable.
    # ------------------------------------------------------------------
    MAX_LEN = 800  # LP works comfortably up to this size
    try:
        seq0 = list(height_sequence_1)  # provided by the grader
    except NameError:
        seq0 = list(np.random.rand(MAX_LEN))

    if len(seq0) > MAX_LEN:
        # pick a random contiguous sub‑block
        start = np.random.randint(0, len(seq0) - MAX_LEN + 1)
        seq0 = seq0[start : start + MAX_LEN]

    # Clip to the allowed numeric range and make sure we have a positive sum.
    seq0 = [max(0.0, min(1000.0, float(v))) for v in seq0]
    if sum(seq0) < 1e-8:
        seq0 = [1.0] * len(seq0)

    # Normalise the sum to √(2 n); this does not change the objective.
    n = len(seq0)
    target_sum = np.sqrt(2.0 * n)
    cur_sum = sum(seq0)
    seq0 = [v * target_sum / cur_sum for v in seq0]

    current = seq0[:]
    best_seq = current[:]
    best_val = _score(best_seq)

    # Adaptive step size for random mass transfers.
    eps_factor = 0.25  # start with 25 % of the average entry
    it = 0

    print(f"Start optimisation: n={n}, initial score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 2a. LP‑based direction (used sparingly – it is the most expensive).
        # --------------------------------------------------------------
        if n <= 800 and np.random.rand() < 0.20 and time.time() < deadline - 10:
            # Normalise current vector for the LP formulation.
            cur_sum = sum(current)
            f_norm = [v * np.sqrt(2.0 * n) / cur_sum for v in current]
            rhs = np.max(np.convolve(f_norm, f_norm))

            g = _solve_convolution_lp(f_norm, rhs)
            if g is not None:
                # Try a handful of mixing coefficients and keep the best.
                improved = False
                for t in (0.1, 0.25, 0.4, 0.6):
                    mixed = [(1 - t) * a + t * b for a, b in zip(current, g)]
                    mixed = [max(0.0, min(1000.0, v)) for v in mixed]
                    val = _score(mixed)
                    if val < best_val:
                        best_val = val
                        best_seq = mixed[:]
                        current = mixed[:]
                        improved = True
                        eps_factor = max(0.01, eps_factor * 0.9)  # fine‑tune later
                        break
                if improved:
                    continue  # move to next iteration

        # --------------------------------------------------------------
        # 2b. Targeted mass transfer – reduce the current maximal convolution.
        # --------------------------------------------------------------
        arr = np.asarray(current, dtype=np.float64)
        conv = np.convolve(arr, arr)
        k_max = int(np.argmax(conv))

        # Compute contributions a_i * a_{k_max-i} for every i.
        contrib = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = current[i] * current[j]

        if np.sum(contrib) > 0:
            probs = contrib / np.sum(contrib)
            i_src = np.random.choice(n, p=probs)
        else:
            i_src = np.random.randint(n)

        # Choose a destination index different from the source.
        j_dst = i_src
        while j_dst == i_src:
            j_dst = np.random.randint(n)

        # Move a small fraction of the source entry.
        max_move = eps_factor * (sum(current) / n)
        eps = np.random.uniform(0.0, min(max_move, current[i_src]))
        new_seq = current[:]
        new_seq[i_src] -= eps
        new_seq[j_dst] += eps

        # Clip to the safe interval.
        new_seq = [max(0.0, min(1000.0, v)) for v in new_seq]
        val = _score(new_seq)

        if val < best_val:
            best_val = val
            best_seq = new_seq[:]
            current = new_seq[:]
            eps_factor = max(0.01, eps_factor * 0.95)
            continue

        # --------------------------------------------------------------
        # 2c. Simple random balanced perturbation (fallback).
        # --------------------------------------------------------------
        i, j = np.random.choice(n, 2, replace=False)
        avg = sum(current) / n
        max_eps = eps_factor * avg
        eps = np.random.uniform(-max_eps, max_eps)

        # Prevent negativity.
        if current[i] - eps < 0.0:
            eps = current[i]

        new_seq = current[:]
        new_seq[i] -= eps
        new_seq[j] += eps
        new_seq = [max(0.0, min(1000.0, v)) for v in new_seq]
        val = _score(new_seq)

        if val < best_val:
            best_val = val
            best_seq = new_seq[:]
            current = new_seq[:]
            eps_factor = max(0.01, eps_factor * 0.97)

        # --------------------------------------------------------------
        # 2d. Logging and step‑size cooling.
        # --------------------------------------------------------------
        if it % 2000 == 0:
            print(
                f"Iter {it:6d} – best score {best_val:.6f} – eps_factor {eps_factor:.4f}"
            )
        if it % 5000 == 0:
            # slowly shrink the move size to finish with a fine scan
            eps_factor = max(0.01, eps_factor * 0.9)

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (already respecting the bounds).
    # ------------------------------------------------------------------
    final_seq = [float(v) for v in best_seq]
    final_val = _score(final_seq)
    print(
        f"Finished after {it} iterations – best score {final_val:.6f} – length {len(final_seq)}"
    )
    return final_seq
