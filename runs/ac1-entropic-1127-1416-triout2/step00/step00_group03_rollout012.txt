def proj_simplex(v):
    # v is np array
    # based on algorithm from paper (Duchi et al.)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=123, budget_s=1000, n_iter=200000):
    np.random.seed(seed)
    t_end = time.time() + budget_s - 5
    n = len(height_sequence_1)
    best = None
    best_obj = float('inf')
    # Parameters
    L = 1 << ((3*n-2).bit_length())
    beta_start = 0.5
    beta_end = 30.0
    lr_start = 1.0
    lr_end = 0.01
    iteration = 0
    # We'll run multiple restarts
    while time.time() < t_end:
        # Initialize random x
        x = np.random.rand(n)
        x = proj_simplex(x)  # sum=1
        # local optimization
        while time.time() < t_end:
            # compute conv via FFT
            fft_x = np.fft.rfft(x, n=L)
            conv = np.fft.irfft(fft_x * fft_x, n=L)[:2*n-1]
            # objective (approx max) get actual max for tracking
            max_conv = conv.max()
            obj = 2.0 * n * max_conv
            if obj < best_obj:
                best_obj = obj
                best = x.copy()
                print(f"Iter {iteration}, obj {obj:.6f}")
            # schedule parameters
            progress = iteration / 50000  # maybe 50000 total
            beta = beta_start + (beta_end - beta_start) * min(1.0, progress)
            lr = lr_start * (lr_end / lr_start) ** min(1.0, progress)
            # compute softmax
            shift = max_conv
            exp_weights = np.exp(beta * (conv - shift))
            p = exp_weights / exp_weights.sum()
            # gradient via correlation
            fft_p = np.fft.rfft(p, n=L)
            # correlation: ifft(fft(p) * conj(fft(x)))
            corr = np.fft.irfft(fft_p * np.conj(fft_x), n=L)
            grad = 2.0 * corr[n-1:n-1+n]  # extract length n region
            # gradient descent step
            x_new = x - lr * grad
            # projection onto simplex
            x = proj_simplex(x_new)
            iteration += 1
            # rejuvenate if stagnation
            if iteration % 20000 == 0:
                break
    return best.tolist()

"""
Projected gradient descent on a smooth surrogate of max(conv(f,f)).
The algorithm works on the simplex (non‑negative entries that sum to 1),
uses FFT for fast convolutions and correlation,
and performs many random restarts within the allowed time budget.
It returns the best sequence found.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]               # the last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical cleanup
    w = np.where(w < 1e-15, 0.0, w)
    # renormalise (important because of rounding)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ------------------------------------------------------------
# Core routine: one gradient descent run starting from init_x
# ------------------------------------------------------------
def _run_descent(
    init_x: np.ndarray,
    n: int,
    L: int,
    max_inner: int,
    deadline: float,
    best_obj: float,
    best_seq: np.ndarray,
):
    """
    Performs a projected gradient descent run.
    Returns possibly improved (best_obj, best_seq).
    """
    x = proj_simplex(init_x.copy())
    # schedule parameters (tuned empirically)
    beta_start = 0.5
    beta_end = 80.0
    lr_start = 0.8
    lr_end = 0.02

    for it in range(max_inner):
        if time.time() >= deadline:
            break

        # ----- convolution f*f (FFT) -----
        fx = np.fft.fft(x, n=L)
        conv_full = np.fft.ifft(fx * fx).real
        conv = conv_full[: 2 * n - 1]                     # linear convolution
        max_conv = conv.max()
        obj = 2.0 * n * max_conv                          # sum(x) == 1 → simplified score

        if obj < best_obj:
            best_obj = obj
            best_seq = x.copy()
            print(
                f"[{time.time():.1f}] new best: {best_obj:.6f}  (it={it:,})"
            )

        # ----- smoothing of max (softmax) -----
        # progress [0,1] for schedule
        prog = it / max_inner
        beta = beta_start + (beta_end - beta_start) * prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # softmax weights over the convolution values
        shift = max_conv                     # for numerical stability
        exp_w = np.exp(beta * (conv - shift))
        p = exp_w / exp_w.sum()             # probabilities, length 2n‑1

        # ----- gradient of the smooth surrogate -----
        fp = np.fft.fft(p, n=L)
        # cross‑correlation p ⋆ x (valid part length n)
        corr_full = np.fft.ifft(fp * np.conj(fx)).real
        # the valid slice starts at index n‑1
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- gradient step + projection -----
        x_new = x - lr * grad
        x = proj_simplex(x_new)

    return best_obj, best_seq
