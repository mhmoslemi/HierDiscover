def propose_candidate(...):
    import numpy as np, random, time, math
    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = float('inf')
    # fixed parameters but can vary
    while time.time() < deadline:
        # random sample of n, H, h, low_val
        n = np.random.randint(1500, 4000)
        low = np.random.uniform(0.1, 2.0)   # baseline
        H = np.random.randint(max(1, int(0.02*n)), int(0.2*n)+1)
        h = np.random.uniform(low+0.5, low+10.0)   # high value
        # ensure high <= 1000.
        h = min(1000.0, h)
        seq = np.full(n, low, dtype=float)
        indices = np.random.choice(n, H, replace=False)
        seq[indices] = h
        # optionally, if max(high-high collisions) too high? but ignore.
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"New best {best_score:.6f} with n={n}, low={low:.3f}, H={H}, h={h:.3f}")
            # early exit if below target
            if best_score < 1.5:
                break
    return best_seq

"""
Propose a very simple but effective construction.
The goal is to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

with a_i ≥ 0.  A uniform sequence gives a value of 2.
We improve it by using two levels of heights:

    * a small baseline value `low` on almost all positions,
    * a larger value `high` on a modest fraction of the indices.

With this pattern the convolution peak is roughly

    (n‑H)·low²  +  H·high·low

while the total sum is

    (n‑H)·low + H·high .

Thus the ratio  max/∑² can be made smaller than the uniform case by
choosing an appropriate proportion `p = H/n` and a sufficiently large
`high/low`.  The algorithm simply samples a few random parameters,
builds the corresponding sequence and keeps the best one found
within the allotted time (≈ 1000 s).  No external files or network
access are needed.

The code is deliberately lightweight: it only uses NumPy for fast
convolution and random sampling, and calls the supplied `evaluate_sequence`
function to obtain the true score.
"""

import time
import numpy as np

# The evaluation function `evaluate_sequence` is assumed to be already
# imported in the execution environment.


def _build_sequence(n: int, low: float, high: float, hcount: int) -> np.ndarray:
    """
    Build a sequence of length `n` where all entries are `low`
    except `hcount` randomly chosen positions set to `high`.

    Parameters
    ----------
    n : int
        Length of the sequence.
    low : float
        Baseline (non‑negative) value for most entries.
    high : float
        Value for the distinguished entries (must be ≥ low).
    hcount : int
        Number of high entries (0 < hcount ≤ n).

    Returns
    -------
    np.ndarray
        The generated sequence.
    """
    seq = np.full(n, low, dtype=float)
    if hcount > 0:
        idx = np.random.choice(n, size=hcount, replace=False)
        seq[idx] = high
    return seq


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function height sequence that beats the current
    bound (≈2.0).  Randomly samples simple two‑level constructions,
    evaluates them, and returns the best one found before the time budget
    expires.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 1000 s).
    **kwargs : dict
        Ignored, kept for compatibility with the calling framework.

    Returns
    -------
    list[float]
        A list of heights (non‑negative) achieving a low evaluation value.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a tiny safety margin

    best_seq = None
    best_score = float('inf')
    iteration = 0

    # Pre‑compute the current best known bound (uniform sequence) to stop early.
    target = 1.503  # we aim to go below this value

    # Main random search loop
    while time.time() < deadline:
        iteration += 1

        # ----- randomised hyper‑parameters -----
        # Length of the sequence (keep it moderate for speed)
        n = np.random.randint(1500, 5000)

        # Baseline height (low) – typically small but >0 to keep low‑low
        # contributions from dominating.
        low = np.random.uniform(0.1, 2.0)

        # Fraction of high entries (p) – between 2 % and 30 % of the length.
        p = np.random.uniform(0.02, 0.30)
        hcount = max(1, int(p * n))

        # Height multiplier – we make `high` noticeably larger than `low`
        # but stay comfortably below the hard clamp of 1000.
        mult = np.random.uniform(2.0, 40.0)          # ratio high/low
        high = min(1000.0, low * mult)

        # Build the candidate sequence
        seq_arr = _build_sequence(n, low, high, hcount)

        # Evaluate – the function returns np.inf for illegal inputs.
        try:
            score = evaluate_sequence(seq_arr.tolist())
        except Exception:
            # Defensive: in the very unlikely event of an error just skip.
            continue

        # Keep the best sequence seen so far.
        if score < best_score:
            best_score = score
            best_seq = seq_arr.tolist()
            print(
                f"[Iter {iteration:05d}] New best: {best_score:.6f} "
                f"(n={n}, low={low:.3f}, high={high:.3f}, H={hcount})"
            )
            # Early exit if we already beat the target.
            if best_score < target:
                print("Target achieved – exiting early.")
                break

    # Fallback: if nothing was found (should never happen) return a trivial
    # uniform sequence of length 2.
    if best_seq is None:
        best_seq = [1.0, 1.0]

    return best_seq
