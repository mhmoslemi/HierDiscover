"""
Candidate generator for the step‑function evaluation problem.

The core idea:
* The analytical optimum for a monotone decreasing shape a_i = 1/√(i+1) yields a score
  close to π/2 ≈ 1.5708.  This is already a substantial improvement over the constant
  sequence (score = 2).  Starting from this shape we perform a simple simulated‑annealing
  / hill‑climbing optimisation: repeatedly move a small amount of mass from one index
  to another (preserving the total sum) and keep the change if it improves the score
  or with a temperature‑dependent probability.  The algorithm works within the
  allotted time budget, keeps the best sequence seen and finally returns it.

The sequence always stays non‑negative and all values are far below the 1000‑limit,
so no additional clipping is required.
"""

import time
import numpy as np
import math

def _initial_power_sequence(n: int, p: float = 0.5) -> np.ndarray:
    """Return a monotone decreasing sequence a_i = (i+1)^(-p), normalized to sum 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _evaluate(a: np.ndarray) -> float:
    """Wrap evaluate_sequence (assumed to be globally available)."""
    # The function works with Python lists, but numpy works fine after conversion.
    return evaluate_sequence(a.tolist())


def _anneal_once(a: np.ndarray, cur_score: float, temperature: float, rng: np.random.Generator) -> tuple:
    """
    Propose a single random transfer of mass between two positions.
    Returns (new_array, new_score, accepted_flag).
    """
    n = a.shape[0]

    # pick two distinct indices
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)

    # do not touch a zero entry
    if a[i] <= 1e-12:
        return a, cur_score, False

    # transfer a random fraction (up to 20%) of a[i] to a[j]
    fraction = rng.random() * 0.2
    delta = fraction * a[i]

    # create candidate
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta

    # (sum is preserved, no need to renormalise)
    new_score = _evaluate(cand)

    # acceptance rule (Metropolis)
    if new_score < cur_score:
        accept = True
    else:
        # probability to accept a worsening move
        prob = math.exp((cur_score - new_score) / max(temperature, 1e-12))
        accept = rng.random() < prob

    if accept:
        return cand, new_score, True
    else:
        return a, cur_score, False


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence giving a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5  # keep a safety margin

    # length of the construction – use the provided global if present
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 2000  # fallback length

    # ----------------------------------------------------------------------
    # 1.  Initialise with the analytically motivated power‑law shape.
    # ----------------------------------------------------------------------
    current = _initial_power_sequence(n, p=0.5)  # a_i = 1 / sqrt(i+1)
    best_seq = current.copy()
    best_score = _evaluate(best_seq)
    cur_score = best_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2.  Simulated‑annealing / hill climbing loop.
    # ----------------------------------------------------------------------
    # Temperature schedule: start relatively hot and cool to near zero.
    T_initial = 0.02 * best_score  # a fraction of the initial score
    T_final = 1e-6
    total_iters = 0

    while time.time() < deadline:
        # linear cooling based on elapsed fraction
        elapsed = time.time() - (deadline - budget_s + 5)
        frac = min(1.0, elapsed / (budget_s - 5))
        temperature = T_initial * (1 - frac) + T_final * frac

        # perform a batch of proposals before recomputing temperature
        for _ in range(5):   # a few cheap proposals per outer loop
            current, cur_score, improved = _anneal_once(current, cur_score, temperature, rng)
            total_iters += 1

            if cur_score < best_score:
                best_score = cur_score
                best_seq = current.copy()
                print(f"[improve] iter={total_iters}, score={best_score:.6f}")

        # occasional random restart if we get stuck for a long time
        if total_iters % 5000 == 0 and time.time() < deadline:
            # keep the best found so far, but try a fresh random perturbation
            restart = _initial_power_sequence(n, p=0.5)
            # add a small random noise to avoid being stuck in the exact same basin
            noise = rng.normal(scale=0.01, size=n)
            restart = np.maximum(restart + noise, 0.0)
            restart /= restart.sum()
            restart_score = _evaluate(restart)
            if restart_score < cur_score:
                current = restart
                cur_score = restart_score
                print(f"[restart] iter={total_iters}, new score={cur_score:.6f}")

    # ----------------------------------------------------------------------
    # 3.  Return the best sequence discovered.
    # ----------------------------------------------------------------------
    print(f"[done] total_iters={total_iters}, best_score={best_score:.6f}")
    return best_seq.tolist()
