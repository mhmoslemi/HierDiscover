"""
Propose a candidate sequence for the step‑function minimisation task.

Algorithm
---------

1. **Initialisation**  
   * If the global variable `height_sequence_1` exists, start from it (it is
     already a good baseline).  
   * Otherwise start from a uniform distribution of length `n` chosen at
     random in the range 200 – 2000.  The uniform sequence is the known
     analytic optimum (value ≈ 2) and is a safe starting point.

2. **Simulated‑annealing local search**  
   * At each iteration a few coordinates are perturbed by a small Gaussian
     displacement (ensuring non‑negativity).  
   * The perturbed vector is re‑normalised so that its sum equals the original
     sum (the evaluation function is scale‑invariant, but keeping the sum
     close to 1 improves numerical stability).  
   * The new sequence is accepted if it improves the score or with a
     temperature‑controlled probability to allow occasional uphill moves.
   * The temperature decays geometrically.

3. **Random restarts**  
   * When the time budget allows, the whole procedure is restarted with a new
     random length and seed.  The best sequence seen over all restarts is
     returned.

All operations respect the security checks performed inside
`evaluate_sequence`.  The implementation uses only NumPy and the standard
library, runs on a single CPU core and respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _initial_sequence(seed: int) -> np.ndarray:
    """
    Return a starting sequence.
    If a global `height_sequence_1` is present we copy it,
    otherwise we generate a uniform vector of random length.
    """
    np.random.seed(seed)
    # Try to use the provided baseline if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)
        if base.ndim == 1 and base.size > 0:
            # Normalise to sum = 1 (scale invariant, improves numeric stability)
            s = base.sum()
            if s > 0:
                return base / s
    except NameError:
        pass

    # Fallback: create a uniform distribution of random length
    n = np.random.randint(200, 2000)          # moderate length – large enough for variation
    seq = np.full(n, 1.0 / n, dtype=float)    # uniform sum = 1
    return seq


def _perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a small random perturbation to a copy of ``seq``.
    Guarantees non‑negativity and re‑normalises to keep the total sum = 1.
    """
    n = seq.size
    new = seq.copy()

    # number of coordinates to touch – at least 1, at most 1% of n
    k = rng.integers(1, max(2, n // 100 + 1))
    for _ in range(k):
        i = rng.integers(0, n)
        # Gaussian step; scale chosen experimentally
        delta = rng.normal(scale=0.05)
        new[i] = max(0.0, new[i] + delta)

    # Re‑normalise (the evaluation function is homogeneous, but a sum close to 1
    # reduces floating‑point round‑off issues)
    total = new.sum()
    if total < 1e-12:
        # Degenerate case – revert to the original vector
        return seq
    new /= total
    return new


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a short simulated‑annealing run starting from ``start_seq``.
    Returns the best sequence seen and its score.
    """
    best_seq = start_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Temperature schedule
    T = 0.5
    decay = 0.9995

    iteration = 0
    while time.time() < deadline:
        cand = _perturb(best_seq, rng)
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            # Defensive – any unexpected error is ignored
            continue

        # Accept if better, otherwise with probability exp(-(Δ)/T)
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # protection against division by zero
            prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            best_seq = cand
            best_score = cand_score
            # occasional progress output
            if iteration % 200 == 0:
                print(
                    f"[{time.time():.2f}] Iter {iteration} – new best score {best_score:.6f}"
                )
        # temperature decay
        T *= decay
        iteration += 1

    return best_seq, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best known bound.

    Parameters
    ----------
    seed : int, optional
        Base seed for reproducibility.
    budget_s : int, optional
        Allowed wall‑clock time in seconds (the harness will terminate the
        process after `budget_s` seconds, but we keep a small safety margin).

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    # ------------------------------------------------------------------
    # Set up timing and random generators
    # ------------------------------------------------------------------
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 0.5
    rng = np.random.default_rng(seed)

    global_best_seq = None
    global_best_score = np.inf

    # ------------------------------------------------------------------
    # Perform several independent restarts while time permits
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # Allocate a slice of the remaining time for this restart
        remaining = deadline - time.time()
        # Give each restart at most 8 seconds; if less time left, use it all
        this_deadline = time.time() + min(8.0, remaining)

        # Initialise a fresh sequence
        init_seq = _initial_sequence(rng.integers(0, 2**31 - 1))

        # Run local search
        seq, score = _simulated_annealing(init_seq, this_deadline, rng)

        # Keep the global best
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(
                f"[{time.time():.2f}] Restart {restart} – new global best {global_best_score:.6f}"
            )
        restart += 1

    # ------------------------------------------------------------------
    # Fallback – if for any reason no sequence was produced, use a safe default
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # uniform of length 500 as a deterministic fallback
        n_fallback = 500
        global_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=float)

    # Ensure the final output satisfies the required type
    return global_best_seq.tolist()
