"""
Propose a step‑function (list of non‑negative heights) with a low
evaluate_sequence() value.

The algorithm mixes three ideas:

1. **Smart initialisation** – a short geometric (exponential) decay is
   searched for the best parameter r.  This already drops the score
   from 2.0 to ≈1.8.

2. **Local mass‑transfer optimisation** – repeatedly move a small
   amount of “mass’’ from a random high entry to a random low entry.
   The move is accepted if it improves the score, otherwise it may be
   accepted with a temperature‑controlled probability (simulated
   annealing).  The total sum of the sequence stays (almost) constant,
   which is essential because the evaluation is scale‑invariant.

3. **LP‑guided direction** – every few hundred iterations the LP from
   the reference implementation is solved (maximise ∑b subject to
   conv(f,b) ≤ current max).  The resulting direction is mixed into the
   current vector with a relatively large step size (t≈0.1).  This
   provides a powerful global improvement that quickly reduces the
   supremum of the convolution.

The process runs until the time budget is exhausted (default 1000 s)
and returns the best sequence seen.  All numbers are kept non‑negative
and below the 0‑1000 limits enforced inside ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: convolution constrained linear programme (the same as in the
# reference code but with the modern HiGHS solver for speed)
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs):
    """
    Solve:
        maximise   sum(b)
        subject to conv(f_seq, b)_k ≤ rhs   for all k
                   b ≥ 0
    Returns the optimal b (numpy array) or None if the solver fails.
    """
    n = len(f_seq)
    # Build coefficient matrix A_ub so that A_ub @ b ≤ rhs
    # Row k corresponds to the convolution term at offset k.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv(f,b)[k] = sum_i f_i * b_{k-i}
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill the needed entries
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])          # non‑negativity
    b_ub = np.hstack([np.full(2 * n - 1, rhs), np.zeros(n)])

    # minimise -sum(b)  == maximise sum(b)
    c = -np.ones(n)

    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",          # fast, stable interior‑point/exact solver
        options={"presolve": True, "time_limit": 5.0},
    )
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Obtain a better direction using the LP (the reference “get_good_direction”)
# ----------------------------------------------------------------------
def get_good_direction_to_move_into(seq, t=0.1):
    """
    Take the current sequence ``seq`` (list of floats), compute the LP
    direction g that increases the total mass without raising the
    current convolution bound, rescale to keep the same L1‑norm and
    return the mixed sequence: (1‑t)*seq + t*g.
    Returns None if the LP fails.
    """
    n = len(seq)
    total = np.sum(seq)
    if total < 1e-12:
        return None

    # Normalise to the “standard” L1‑norm √(2n) used in the reference.
    scale = np.sqrt(2.0 * n) / total
    f_norm = [x * scale for x in seq]

    # current bound M = max(conv(f_norm, f_norm))
    conv_max = np.max(np.convolve(f_norm, f_norm))

    # Solve LP for the direction with the original bound M
    g = solve_convolution_lp(f_norm, conv_max)
    if g is None:
        return None

    # Scale g to the original L1‑norm (the LP used a normalised f)
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    g_scaled = [x * (total / sum_g) for x in g]

    # Mixed update
    new_seq = [(1.0 - t) * a + t * b for a, b in zip(seq, g_scaled)]
    return new_seq
