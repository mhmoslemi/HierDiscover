def simplex_projection(v):
    # Projects onto probability simplex (sum=1, v>=0)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Simple stochastic optimizer for the step‑function evaluation problem.

The evaluation is invariant under positive scaling (2·n·max(conv)/sum²),
so we may restrict the search to probability vectors (sum = 1).
The algorithm repeatedly draws a random start, then performs a few
iterations of a smoothed “max‑convolution” gradient descent:

* For the current vector a compute b = a * a (discrete convolution).
* Approximate the max‑operator by a soft‑max with temperature α
  (grows during the run).  The resulting weights p form a probability
  distribution over the convolution entries.
* The gradient of   f(a)=∑ₖ pₖ·(a*a)[k]   is
        g_i = 2·∑ₖ pₖ·a_{k‑i}
  which can be evaluated via a single correlation.
* Take a step a ← a – η·g and project back onto the simplex
  (non‑negative entries summing to one).
* Occasionally jitter the vector with tiny Gaussian noise to escape
  plateaus.

All candidates are scored with the provided `evaluate_sequence`
function; the best one seen before the time budget expires is returned.
The method does not rely on any external file I/O and respects the
1000‑second limit.
"""

import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum x = 1}."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán (2013) – Efficient projections onto the L1‑ball
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Return soft‑max probabilities of α·b (numerically stable)."""
    max_b = b.max()
    exp_b = np.exp(alpha * (b - max_b))
    return exp_b / exp_b.sum()

def _grad_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σₖ pₖ·(a*a)[k]  with respect to a.
    Using the identity   ∂(a*a)[k]/∂a_i = 2·a_{k‑i}
    (indices outside the range are ignored).
    The operation reduces to a correlation between p and a.
    """
    # full correlation: length = len(p) + len(a) – 1
    corr = np.correlate(p, a, mode="full")
    # index (len(a)-1) corresponds to shift 0
    start = len(a) - 1
    # g_i = 2 * corr[start - i]
    g = 2.0 * corr[start - np.arange(len(a))]
    return g

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will try to finish a few
        seconds before the limit to guarantee a return value).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum is unrestricted).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper: try to initialise from the previously discovered sequence
    # ------------------------------------------------------------------
    try:
        init_global = np.asarray(height_sequence_1, dtype=float)
        have_global = True
    except Exception:   # noqa: BLE001
        have_global = False

    # ------------------------------------------------------------------
    # Main optimisation loop (multiple random restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Choose problem size – small enough for fast iterations,
        # large enough to give the algorithm room to improve.
        # ----------------------------------------------------------------
        n = 1200   # length of the candidate vector

        # --------------------------------------------------------------
        # Initialise a probability vector a (sum = 1)
        # --------------------------------------------------------------
        if have_global and np.random.rand() < 0.3:
            # use a truncated / padded version of the known construction
            if init_global.size >= n:
                a = init_global[:n].copy()
            else:
                a = np.pad(init_global, (0, n - init_global.size), "constant")
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                a = np.random.rand(n)
            a = a / a.sum()
        else:
            # fresh random start (Dirichlet‑like)
            a = np.random.rand(n)
            a = a / a.sum()

        # --------------------------------------------------------------
        # Quick evaluation of the start point
        # --------------------------------------------------------------
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()

        # --------------------------------------------------------------
        # Inner optimisation – smoothed max‑convolution descent
        # --------------------------------------------------------------
        max_iters = 800   # enough for noticeable progress, cheap per step
        alpha = 1.0       # temperature for the soft‑max (grows)
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # convolution a*a (full length = 2n‑1)
            conv = np.convolve(a, a)

            # increase the soft‑max sharpness gradually
            if (it % 150) == 0 and alpha < 80.0:
                alpha *= 1.6

            # soft‑max probabilities over the convolution entries
            probs = _softmax_weights(conv, alpha)

            # gradient of the soft‑max‑weighted objective
            grad = _grad_softmax(a, probs)

            # simple diminishing step size
            step = 0.45 / np.sqrt(it + 1.0)

            a = a - step * grad

                alpha = max(alpha, 1.0)

        # ------------------------------------------------------------------
        # If we already beat the target we can finish early
        # ------------------------------------------------------------------
        if best_score <= 1.503 and (time.time() + 2) < deadline:
            # give a tiny grace period for a final clean‑up
            break

    # ----------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # As a safety net – should never happen because the loop always
        # evaluates at least one candidate.
        best_seq = [1.0]

    return best_seq.tolist()
