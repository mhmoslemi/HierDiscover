def project_simplex(v):
    # v: np.array
    # returns projection onto simplex sum = 1, v_i >= 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    """Euclidean projection of vector v onto probability simplex (sum=1, v>=0)."""
    # Implementation from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all entries are zero or negative; return uniform maybe
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Simple projected sub‑gradient optimizer for the “max‑convolution” bound.

The evaluation function (up to scaling) is  

    E(a) = 2·n·max(a * a) / (sum(a)^2)

which is scale‑invariant.  After normalising the sequence so that sum(a)=1 we
must minimise  max(a*a).  The convex objective is tackled with a projected
sub‑gradient method:

* a is kept on the probability simplex (non‑negative, sum = 1) by an
  Euclidean projection.
* At each iteration the index k where the convolution attains its maximum
  is found; the sub‑gradient of that maximal entry is simply a shifted copy
  of a (with a doubled diagonal term when k is even).
* A diminishing step size ηt = η0/√t is used and after the step we project
  back onto the simplex.

The optimiser is run for several sequence lengths (powers of two up to a
few thousand).  The best sequence found in the allotted time is returned.
The implementation is completely self‑contained, uses only NumPy and the
provided `evaluate_sequence` for the final score, and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of the algorithm by Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # If all entries are non‑positive, the projection is the uniform vector.
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via FFT (real input, real output)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> tuple[float, int]:
    """Return (max_value, argmax) of the full convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.
    For each i, ∂c_k/∂a_i = a_{k-i} if 0 ≤ k-i < n,
    and = 2·a_i when k = 2·i (diagonal term).
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    # double the diagonal term when it exists
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def compute_score(a: np.ndarray) -> float:
    """Score corresponding to evaluate_sequence but using the scale‑invariant form."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # sum(a) is forced to 1 by the optimiser, but we keep the formula for safety.
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Initialise a sequence that mimics the continuous arcsine density:
    a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i)).
    This shape puts more weight at the ends and is known to flatten the
    self‑convolution.
    """
    i = np.arange(n, dtype=float)
    # small epsilon avoids exact zeros at the extreme ends
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def improve_sequence(initial_a: np.ndarray,
                    time_deadline: float,
                    max_iter: int = 200_000,
                    eta0: float = 0.5,
                    noise_every: int = 500,
                    noise_scale: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.

    Parameters
    ----------
    initial_a : ndarray
        Starting non‑negative vector (not necessarily normalised).
    time_deadline : float
        Absolute Unix time after which the routine must stop.
    max_iter : int
        Hard limit on iteration count.
    eta0 : float
        Initial step‑size (scaled by 1/√t).
    noise_every : int
        Every `noise_every` iterations a tiny random perturbation is added
        to avoid stagnation.
    noise_scale : float
        Standard deviation of the random perturbation.

    Returns
    -------
    best_a : ndarray
        The best sequence seen.
    best_score : float
        Corresponding evaluation score.
    """
    a = initial_a.astype(float).copy()
    a = project_to_simplex(a)                     # ensure we are on the simplex
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    iter_cnt = 0
    while time.time() < time_deadline and iter_cnt < max_iter:
        iter_cnt += 1
        # ----- current convolution and worst index -----
        max_val, k_max = max_convolution_and_index(a)

        # ----- sub‑gradient step -----
        grad = subgradient_of_max_at_index(a, k_max)

        # diminishing step size
        eta = eta0 / np.sqrt(iter_cnt)

        a_new = a - eta * grad

        # occasional tiny stochastic noise to escape flat regions
        if noise_every and (iter_cnt % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        # ----- keep the best -----
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if iter_cnt % 2000 == 0:
            print(f"  iter={iter_cnt:,d}  score={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score

# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) with a low
    `evaluate_sequence` value.  The routine runs a projected sub‑gradient
    optimiser for several sequence lengths and returns the best sequence
    discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # leave a tiny safety margin

    # ------------------------------------------------------------------
    # Keep track of the overall best candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # Try the previously known construction (if it exists) to have a safe fallback
    try:
        height_sequence_1      # may raise NameError
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore.  Powers of two give a good trade‑off between
    # speed (FFT) and asymptotic bound (larger n → lower possible score).
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\nStarting optimisation for length n = {n}")

        # Two different initialisations: arcsine‑shaped and uniform.
        inits = [arcsine_initial(n), np.full(n, 1.0 / n)]

        for init_idx, init_a in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"  init {init_idx}/{len(inits)} ({'arcsine' if init_idx == 1 else 'uniform'})")
            seq, val = improve_sequence(
                initial_a=init_a,
                time_deadline=deadline,
                max_iter=100_000,
                eta0=0.8,
                noise_every=800,
                noise_scale=5e-5,
            )
            print(f"    best score for this run: {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (slow but done only once)
    # ------------------------------------------------------------------
    if best_seq is None:
        # As a last resort, return a simple uniform sequence of modest size.
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The competition harness expects a plain Python list of floats.
    return best_seq
