"""
Propose a low‑evaluation step‑function.

The evaluation is   E(a) = 2·n·max(a*a)/(sum(a)^2) .
Uniform heights give E = 2 for any length n, and this is the
trivial bound.  A classical construction based on the
continuous function f(x)=1/√x gives a much smaller value:
for a_i ∝ 1/√(i+1) one obtains  E → π/2 ≈ 1.57 as n grows.
This sequence is used as a good starting point.

A very cheap local‑search is then applied.
At each iteration the convolution is computed, the shift k with
the largest value is identified, and one of the indices that
contribute to this shift (i such that 0≤k−i<n) is selected.
A small amount of mass is moved from that index to a random
index that does **not** participate in the maximal shift.
The move is accepted only if it decreases the evaluation.
The process runs until the allocated time budget expires and the
best sequence found is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _initial_power_law(n: int) -> np.ndarray:
    """
    Return a non‑negative vector of length n with entries
    proportional to 1/sqrt(i+1) and normalized to sum to 1.
    """
    idx = np.arange(1, n + 1, dtype=float)          # 1 … n
    seq = 1.0 / np.sqrt(idx)                        # 1/√i
    seq /= seq.sum()                                # L1‑normalisation
    return seq


def _evaluation(seq: np.ndarray) -> float:
    """
    Compute the same quantity as `evaluate_sequence` but without the
    security checks (the sequence is always non‑negative, finite,
    and its sum is > 0 in our code).
    """
    n = seq.size
    # linear convolution (zero padding at the ends)
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    total = seq.sum()
    return float(2.0 * n * max_conv / (total * total))


def _contributors_to_shift(k: int, n: int) -> np.ndarray:
    """
    Indices i that appear in the convolution term with shift k,
    i.e. 0 <= i < n and 0 <= k-i < n.
    """
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    return np.arange(lo, hi + 1, dtype=int)


# ------------------------------------------------------------
# Main entry point required by the platform
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a
    small evaluation value.

    The algorithm:
        1. Determine the length n (use the known construction if
           available, otherwise fall back to 5000).
        2. Initialise with the 1/√i distribution (theoretical constant
           π/2 ≈ 1.57).
        3. Repeatedly try tiny mass transfers from indices that
           participate in the current maximal convolution shift to
           indices that do not.  Accept the move only if the evaluation
           improves; keep the best sequence seen.
        4. Stop when the time budget is exhausted and return the best
           sequence as a plain Python list.
    """
    # -----------------------------------------------------------------
    # 0. basic set‑up
    # -----------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0      # a little slack

    # length of the sequence – use the known solution if present
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        print(f"Using length from existing construction: n = {n}")
    else:
        n = 5000
        print(f"No existing construction – using default length n = {n}")

    # -----------------------------------------------------------------
    # 1. start from the analytically good power‑law shape
    # -----------------------------------------------------------------
    current = _initial_power_law(n)
    best_seq = current.copy()
    best_score = _evaluation(best_seq)
    print(f"Initial score (power‑law start) = {best_score:.6f}")

    # parameters for the local search
    step_factor = 0.08          # relative size of the mass transfer
    iteration = 0
    no_improve = 0

    # -----------------------------------------------------------------
    # 2. local improvement loop
    # -----------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # convolution and the shift that currently dominates the score
        conv = np.convolve(current, current)
        k_max = int(conv.argmax())

        # indices that actually contribute to this shift
        contrib = _contributors_to_shift(k_max, n)

        # pick the largest contributor – moving mass away from it helps
        i = contrib[np.argmax(current[contrib])]

        # pick a target index that does *not* belong to the contributing set
        mask = np.ones(n, dtype=bool)
        mask[contrib] = False
        if mask.any():
            j = int(np.random.choice(np.where(mask)[0]))
        else:
            # extremely unlikely, but fall back to any other index
            j = int(np.random.randint(n))
            while j == i:
                j = int(np.random.randint(n))

        # propose a tiny transfer of mass from i to j
        # delta is signed; we keep the move small enough to stay non‑negative
        delta = step_factor * current[i] * (np.random.rand() - 0.5)
        delta = np.clip(delta, -current[i], current[j])
        if abs(delta) < 1e-12:
            continue

        candidate = current.copy()
        candidate[i] -= delta
        candidate[j] += delta

        cand_score = _evaluation(candidate)

        # accept only if we get a genuine improvement
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            current = candidate.copy()
            no_improve = 0
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")
        else:
            # keep exploring from the current point with a small probability
            if np.random.rand() < 0.02:
                current = candidate.copy()
            no_improve += 1

        # occasional random restart if we are stuck for a long time
        if no_improve > 1500:
            # re‑initialise from the analytically good shape
            current = _initial_power_law(n)
            no_improve = 0
            print(f"Iter {iteration:6d}: restart due to stagnation")

    # -----------------------------------------------------------------
    # 3. final reporting
    # -----------------------------------------------------------------
    final_score = _evaluation(best_seq)
    print(f"Finished after {iteration} iterations – final score = {final_score:.6f}")

    # Return a plain list – the host environment will feed it to
    # `evaluate_sequence` for the official score.
    return best_seq.tolist()
