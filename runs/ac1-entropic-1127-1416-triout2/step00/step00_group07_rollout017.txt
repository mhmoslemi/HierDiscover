def propose_candidate(seed=1234, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    # Choose length n perhaps between 2000-5000 depending on time.
    n = 2000
    # Initialize using geometric distribution with c close to optimum ~1.34.
    c = 1.34  # delta n
    delta = c / n
    r = 1 - delta
    # generate a_i = r**i
    seq = np.array([r**i for i in range(n)], dtype=float)
    # optionally multiply by (1-r) to approximate normalized; not required.
    # We'll renormalize to sum=1.
    seq = seq / np.sum(seq)

    best_seq = seq.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    print(f"Initial value = {best_val}")

    # Simulated annealing loop
    temperature = 0.1
    cooling_rate = 0.9999
    iter_count = 0

    while time.time() < deadline:
        # pick random modification
        # choose i from distribution weighted by contributions to current max
        conv = np.convolve(seq, seq)
        K = np.argmax(conv)
        # compute product contributions
        # indices i where both i and K-i in range
        indices = np.arange(max(0, K - (n - 1)), min(n - 1, K) + 1)
        # compute contributions
        contrib = seq[indices] * seq[K - indices]
        # pick i index weighted by contribution
        prob = contrib / np.sum(contrib)
        i = np.random.choice(indices, p=prob)
        # pick j index random (different)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # amount to move
        eps = np.random.exponential(scale=0.02)  # typical step
        delta_mass = eps * seq[i]  # fraction of current mass at i
        # ensure nonnegative
        delta_mass = min(delta_mass, seq[i])
        if delta_mass <= 0:
            continue
        new_seq = seq.copy()
        new_seq[i] -= delta_mass
        new_seq[j] += delta_mass
        # renormalize to sum 1
        new_seq = new_seq / np.sum(new_seq)

        val = evaluate_sequence(new_seq.tolist())
        # accept if better or with probability
        if val < best_val:
            best_val = val
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"Iter {iter_count}: new best {best_val:.6f}")
        else:
            # Metropolis accept
            accept_prob = np.exp((best_val - val) / temperature) if temperature > 0 else 0
            if np.random.rand() < accept_prob:
                seq = new_seq

        # temperature schedule
        temperature *= cooling_rate
        iter_count += 1

    print(f"Finished after {iter_count} iterations, best value = {best_val}")
    return best_seq.tolist()

            a_new = np.ones(n) / n

        # evaluate
        val = evaluate_sequence(a_new.tolist())
        if val < best_val:
            best_val = val

"""
Propose a high‑quality step‑function (non‑negative sequence) for the
evaluate_sequence() benchmark.

Algorithm
---------

1. **Initial candidate** – start from the provided `height_sequence_1` if
   present, otherwise use a uniform vector.  The candidate is normalised
   to sum = 1 (the evaluation is scale‑invariant).

2. **Random parametric search** – generate many sequences of the form  

      a_i ∝ exp(-α·i) + w·exp(-α·(n‑1‑i))   +   mix·(1/n)

   where the three parameters (α, w, mix) are sampled randomly
   (log‑uniform for α and w, uniform for mix).  Each generated vector is
   normalised and evaluated; the best one is kept.

   This family contains the pure exponential (the known analytical
   improvement) and also symmetric “U‑shaped’’ mixtures that can lower
   the peak of the self‑convolution further.

3. **Gradient‑descent refinement** – starting from the best sequence
   found, minimise a smooth approximation of the L∞‑norm of the
   convolution:

      Lβ(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

   The gradient of Lβ is

      ∇Lβ_i = 2·Σ_k w_k·a_{k‑i},
      w_k = softmax_β((a∗a)_k).

   Using this gradient we perform projected gradient steps (project onto
   the probability simplex to keep non‑negativity and unit sum).  The
   smoothing parameter β is increased slowly, turning the softmax into a
   tighter approximation of the true maximum.  A simple greedy accept‑
   reject rule is used, with occasional tiny random jitter to escape
   plateaus.

4. **Time‑bounded loop** – the whole routine respects the supplied
   `budget_s` (default 1000 s).  Progress is printed, and the best
   sequence found is returned as a plain list of floats.

The method combines a cheap global search over a low‑dimensional family
with a deterministic local optimiser, and typically drives the score
well below the uniform bound 2 (often ≈ 1.6–1.5 on lengths ≈ 1500–2000).

"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    # ------------------------------------------------------------
    # 1. Basic setup
    # ------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # a small safety margin

    # -----------------------------------------------------------------
    # 2. Choose a working length (large enough to give a good bound,
    #    small enough to keep each evaluation cheap)
    # -----------------------------------------------------------------
    n = 1500                     # can be tuned; 1‑2 k is a good trade‑off

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    # try to start from the supplied SOTA construction
    try:

        if init.size < n:
            init = np.pad(init, (0, n - init.size), constant_values=0.0)
        else:
            init = init[:n]
        if init.sum() < 1e-12:
            init = np.ones(n, dtype=float)
    except Exception:
        init = np.ones(n, dtype=float)

    # normalise to a probability vector (scale does not affect the score)
    init = np.maximum(init, 0.0)
    init = init / max(init.sum(), 1e-12)

    best_seq = init.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------
    # 4. Random exploration in a 2‑exponential family
    # ------------------------------------------------------------
    it_rand = 0
    while time.time() < deadline and it_rand < 2000:
        # sample parameters
        log_alpha = np.random.uniform(np.log(1e-5), np.log(0.01))
        alpha = np.exp(log_alpha)                     # decay per index
        w_factor = np.exp(np.random.uniform(-5.0, 5.0))  # weight of the right‑hand exponential
        mix = np.random.rand() * 0.5                  # how much pure uniform to blend in

        idx = np.arange(n, dtype=float)
        left = np.exp(-alpha * idx)                  # exp(-α i)
        right = np.exp(-alpha * (n - 1 - idx))       # exp(-α (n‑1‑i))

        seq = (1.0 - mix) * (left + w_factor * right) + mix * (1.0 / n)

        total = seq.sum()
        if total < 1e-12:        # degenerate – skip
            continue
        seq = seq / total

        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[rand {it_rand:04d}] new best = {best_score:.6f} "

        it_rand += 1

    # ------------------------------------------------------------
    # 5. Gradient‑descent refinement (soft‑max surrogate of max)
    # ------------------------------------------------------------
    a = best_seq.copy()
    prev_score = evaluate_sequence(a.tolist())
    beta = 5.0               # smoothing for the soft‑max
    lr = 0.12                # initial learning rate
    beta_mul = 1.01
    lr_decay = 0.9995
    gd_iter = 0

    while time.time() < deadline:
        # convolution a*a
        conv = np.convolve(a, a)                # length 2n‑1
        max_c = conv.max()

        # soft‑max weights (more weight on the current maximum)
        shifted = conv - max_c                  # ≤ 0  → no overflow

        w_sum = w.sum()
        if w_sum == 0.0:

        inds = L - 1 - np.arange(n)            # map i → proper slice
        grad = 2.0 * grad_full[inds]

        # projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:

        if new_score < prev_score:

        # slowly tighten the soft‑max and shrink the step‑size

            a = a / a.sum()
            prev_score = evaluate_sequence(a.tolist())

    print(f"[done] random trials = {it_rand}, gradient steps = {gd_iter}, "
          f"best score = {best_score:.6f}")
    return best_seq.tolist()
