"""
Heuristic optimizer for the step‑function evaluation problem.

The algorithm repeatedly improves a candidate sequence `a` by:
1. Normalising it so that Σa = √(2·n) (the evaluation is scale‑invariant,
   this normalisation only makes the linear program well conditioned).
2. Computing `M = max(a*a)` (self‑convolution peak).
3. Solving a linear program that finds a non‑negative vector `g` with the
   largest possible sum while respecting the cross‑convolution bound
   `a * g ≤ M`.  This is exactly the direction used in the published SOTA.
4. Mixing the current sequence with `g` for several mixing coefficients
   `α ∈ {0.01,0.05,0.1,0.2,0.4}`; after each mix the result is re‑normalised
   to maintain Σ = √(2·n).
5. Keeping the best mix (lowest `evaluate_sequence` value) as the new
   current sequence.  If the LP fails a small random perturbation is applied.
6. Occasionally restarting from a fresh random vector if no improvement has
   been seen for a while.

All operations respect the non‑negativity and clipping limits enforced by
`evaluate_sequence`.  The loop runs until the time budget (≈ 1000 s) is exhausted,
and the best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level – required by the execution environment)
# ----------------------------------------------------------------------


def _build_conv_constraint_matrix(f):
    """
    Build the (2n‑1) × n matrix A such that A @ g = conv(f, g) (valid indices only).

    For each column i the entry f[i] appears on the anti‑diagonal rows i … i+n‑1.
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i, val in enumerate(f):
        A[i:i + n, i] = val
    return A


def _solve_lp_direction(f_norm, M):
    """
    Solve the LP:
        max   Σ g_i
        s.t.  conv(f_norm, g) ≤ M    (2n‑1 linear inequalities)
              g ≥ 0
    Returns the optimal g as a NumPy array, or None if the solver fails.
    """
    n = len(f_norm)

    # Linear constraints for the cross‑convolution
    A_conv = _build_conv_constraint_matrix(f_norm)
    # Non‑negativity rows: -g ≤ 0  →  g ≥ 0
    A_nonneg = -np.eye(n, dtype=np.float64)

    A_ub = np.vstack([A_conv, A_nonneg])
    b_ub = np.hstack([np.full(2 * n - 1, M, dtype=np.float64), np.zeros(n, dtype=np.float64)])

    # Objective: maximise Σ g_i  →  minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)

    # Use the default HiGHS solver; a small time limit guards against pathological cases.
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"time_limit": 5.0, "presolve": True},
        )
    except Exception:            # pragma: no cover – robust fallback
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method="highs")

    if res.success:
        return res.x
    return None


def _random_perturb(seq, sigma=0.02):
    """Small random mass transfer keeping the total sum constant."""
    seq = seq.copy()
    n = len(seq)
    idx = np.random.randint(n)
    delta = np.random.randn() * sigma
    seq[idx] = max(0.0, seq[idx] + delta)

    # renorm to the canonical sum √(2·n)
    total = seq.sum()
    if total <= 0:
        seq[:] = 1.0 / n
        total = seq.sum()
    seq *= np.sqrt(2.0 * n) / total
    return seq


def _mix_and_normalise(f, g, alpha):
    """
    Return (1‑α)·f + α·g, renormalised so that its ℓ1‑norm equals √(2·n).
    """
    mixed = (1.0 - alpha) * f + alpha * g
    total = mixed.sum()
    if total > 0:
        mixed *= np.sqrt(2.0 * len(mixed)) / total
    else:                                   # pragma: no cover – safety guard
        mixed = np.full_like(mixed, 1.0 / len(mixed))
        mixed *= np.sqrt(2.0 * len(mixed))
    return mixed

def propose_candidate(seed=42, budget_s=1000, **kwargs):

    """

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    if "height_sequence_1" in globals() and np.random.rand() < 0.5:

    no_improve_cnt = 0
    MAX_STAGNANT = 250

    while time.time() < deadline:
        n = len(current)
        # ------------------------------------------------------------------
        # Normalise current vector (required by the LP formulation)
        # ------------------------------------------------------------------
        total = current.sum()

            total = current.sum()
        f_norm = current * (np.sqrt(2.0 * n) / total)

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        conv_f = np.convolve(f_norm, f_norm)
        M = conv_f.max()

        # --------------------------------------------------------------
        # Solve the LP to obtain a promising direction g
        # --------------------------------------------------------------
        g = _solve_lp_direction(f_norm, M)

        candidate_generated = False

        if g is not None:
            # Try several mixing ratios – keep the best improvement
            for alpha in alphas:
                mixed = _mix_and_normalise(f_norm, g, alpha)
                # Evaluate (convert to plain Python list for the external function)
                try:
                    score = evaluate_sequence(mixed.tolist())

                    score = np.inf

                    best_score = score

                # Accept the first mixing that improves the current sequence
                cur_score = evaluate_sequence(current.tolist())
                if score < cur_score:
                    current = mixed
                    candidate_generated = True
                    break

            # If none of the alphas improved the current sequence,
            # still keep the best mixed candidate as the new current vector.
            if not candidate_generated:
                # pick the mix with the smallest score (even if not better)
                scores = []
                mixes = []
                for alpha in alphas:
                    mixed = _mix_and_normalise(f_norm, g, alpha)
                    try:
                        s = evaluate_sequence(mixed.tolist())
                    except Exception:           # pragma: no cover
                        s = np.inf
                    scores.append(s)
                    mixes.append(mixed)
                best_alpha_idx = int(np.argmin(scores))
                current = mixes[best_alpha_idx]
                candidate_generated = True
        else:
            # ----------------------------------------------------------------
            # LP failed – fall back to a tiny random perturbation
            # ----------------------------------------------------------------
            current = _random_perturb(current, sigma=0.02)

        # --------------------------------------------------------------------
        # Update stagnation counter
        # --------------------------------------------------------------------
        if best_score < evaluate_sequence(current.tolist()):
            no_improve_cnt = 0
        else:
            no_improve_cnt += 1

        if no_improve_cnt >= MAX_STAGNANT:
            # restart with a fresh random vector of the same length
            current = np.random.rand(n)
            current *= np.sqrt(2.0 * n) / current.sum()
            no_improve_cnt = 0
            print(f"[{time.time() - start_time:6.1f}s] Restart due to stagnation (n={n})")

        # occasional cheap re‑normalisation to avoid drift caused by rounding
        current = np.maximum(current, 0.0)
        cur_sum = current.sum()
        if cur_sum > 0:
            current *= np.sqrt(2.0 * n) / cur_sum

    # Return the best sequence found (as plain Python list)
    return best_seq.tolist()
