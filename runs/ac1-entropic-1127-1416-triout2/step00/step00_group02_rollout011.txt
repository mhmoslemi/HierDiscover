def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2
    n = np.random.randint(500, 2000)  # choose random length.
    # start with uniform distribution plus slight random perturbation
    a = np.ones(n) / n
    a = a + (np.random.rand(n) - 0.5) * 0.001
    a = np.clip(a, 0, None)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    step = 0.1
    iteration = 0
    while time.time() < deadline and iteration < 20000:
        # compute convolution
        b = np.convolve(a, a)
        M = b.max()
        # active indices
        # tolerance relative to M to capture near max
        eps = 1e-12
        active = np.where(b >= M - eps)[0]
        I = np.zeros_like(b)
        I[active] = 1.0
        # gradient via correlation
        corr = np.correlate(a, I, mode='full')
        offset = len(I) - 1
        grad = 2 * corr[offset: offset + n]   # size n
        # update
        a_new = a - step * grad
        # project onto simplex
        a_new = project_simplex(a_new)
        # optionally add small random perturbation to escape stagnation
        if iteration % 1000 == 0:
            a_new += (np.random.rand(n) - 0.5) * 0.001
            a_new = np.clip(a_new, 0, None)
            a_new = project_simplex(a_new)
        # compute score
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            # maybe reduce step
            step = max(step * 0.9, 1e-4)
        else:
            # if not improved, maybe shrink step
            step = min(step * 1.05, 1.0)
        a = a_new
        iteration += 1
        if iteration % 200 == 0:
            print(f"Iter {iteration}, best_score {best_score:.6f}, n {n}, step {step:.6f}")
    return best_seq.tolist()

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Optimisation of the step‑function bound.

The objective for a non‑negative vector `a` of length *n* (with
`sum(a)=S`) is  

    score = 2 * n * max(conv(a,a)) / S**2 .

Since the score is invariant under scaling we normalise the vector
to `sum=1`.  The problem reduces to minimising the maximal entry of the
self‑convolution.  This is a non‑convex min‑max problem; a simple
projected sub‑gradient method works well when started from a random
point (uniform vectors are stationary and give the obvious bound `2`).

Algorithm
---------
1. Choose a random length `n` (­200 … 3000) and a random initial
   probability vector `a` (Dirichlet with concentration <1 for
   variability).
2. Repeatedly:
   * Compute the full convolution `b = a * a` (FFT based, O(n log n)).
   * Find the set of indices where `b` attains its current maximum.
   * The sub‑gradient of the max‑function w.r.t. `a` is the correlation
     of `a` with the indicator of the active set.  Using the FFT we
     obtain the gradient efficiently.
   * Update `a ← a – η·grad` with a diminishing step `η` and project back
     onto the probability simplex (non‑negative, sum = 1).
   * Keep the best vector seen so far (according to the exact
     `evaluate_sequence` routine).
   * Occasionally inject a tiny random perturbation to escape plateaus.
3. Perform many short restarts while time permits and retain the best
   sequence overall.

The whole search runs within the allotted 1000 s and returns the best
candidate found.
"""

import time
import numpy as np


def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(a, size):
    """Return max of the linear convolution a*a (full, length 2n‑1)."""
    fa = np.fft.rfft(a, size)
    conv_full = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv_full.max()

    """

    deadline = time.time() + budget_s - 0.2  # tiny safety margin

    best_score = np.inf
    best_seq = None

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    try:

                sc = evaluate_sequence(init.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = init.copy()
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Main optimisation loop: many short random restarts with sub‑grad.
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # ------- choose problem size ------------------------------------------------
        n = np.random.randint(300, 2500)               # moderate length
        # ------- random start (Dirichlet with concentration < 1 for variance) ------
        alpha = np.full(n, 0.5)                         # encourages sparsity

        size = 1 << ((2 * n - 1).bit_length())         # FFT length (next power of two)

        # ----- local best for this restart ----------------------------------------
        fa = np.fft.rfft(a, size)

        # ----- iterative projected sub‑gradient descent ----------------------------
        for it in range(3000):

            fa = np.fft.rfft(a, size)

            M = b_full.max()

            # active set (indices attaining the maximum, with a tiny tolerance)
            eps = 1e-12
            active = np.where(b_full >= M - eps)[0]
