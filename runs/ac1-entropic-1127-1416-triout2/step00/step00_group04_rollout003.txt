def fft_convolve(x, y):
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    return np.fft.irfft(fx * fy, n=size)[:len(x)+len(y)-1]

import numpy as np
import time

def fft_convolve(x, y):
    """Return the linear convolution of two 1‑D real arrays using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv

def normalize(a):
    s = np.sum(a)
    if s <= 0:
        return a
    return a / s

def project_nonneg_norm(a):
    """Project onto the simplex: non‑negative entries that sum to 1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s == 0:
        # equally distribute if everything vanished
        a = np.full_like(a, 1.0 / len(a))
    else:
        a = a / s
    return a

def gradient_step(a, target, lr):
    """One projected gradient step w.r.t. the squared error ‖a*a−target‖²."""
    n = len(a)
    # a*a
    b = fft_convolve(a, a)
    # error
    e = b - target
    # cross‑correlation between error and a
    a_rev = a[::-1]
    g_full = fft_convolve(e, a_rev)          # length = 3n‑2
    g = 4.0 * g_full[n-1 : n-1 + n]          # shape (n,)
    # gradient descent on a, then project back onto the simplex
    a_new = a - lr * g
    a_new = project_nonneg_norm(a_new)
    return a_new, b

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes ``evaluate_sequence``.
    The algorithm uses projected gradient descent on the squared error between
    the auto‑convolution and a flat target (uniform distribution).  Several
    lengths and random restarts are tried within the allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # candidate lengths – small enough for fast FFT but large enough to beat the bound
    lengths = [500, 1000, 2000, 3000]
    # optionally add the length of the known construction if it exists and is not huge
    try:
        L = len(height_sequence_1)   # noqa: F821 – global supplied by the harness
        if L <= 5000:
            lengths.append(L)
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')
    total_iters = 0

    for n in lengths:
        if time.time() >= deadline:
            break
        m = 2 * n - 1
        target = np.full(m, 1.0 / m, dtype=np.float64)   # flat target for a*a

        # initialise a – use the known sequence when lengths match, otherwise random Dirichlet
        init = None
        try:
            if len(height_sequence_1) == n:  # noqa: F821
                init = np.array(height_sequence_1, dtype=np.float64)
                init = project_nonneg_norm(init)
        except Exception:
            pass
        if init is None:
            # Dirichlet with parameter 0.5 favours a few larger entries while staying dense
            alpha = np.full(n, 0.5)
            init = np.random.dirichlet(alpha)

        a = project_nonneg_norm(init)
        lr = 0.2  # starting learning rate – will decay

        # inner loop – projected gradient descent
        while time.time() < deadline:
            total_iters += 1
            a, b = gradient_step(a, target, lr)

            # evaluate – sum(a) is 1 by construction, so the score simplifies
            max_b = np.max(b)
            score = 2.0 * n * max_b

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New best: {best_score:.6f} (n={n}, iter={total_iters})")

            # simple exponential decay of the learning rate
            lr *= 0.9997
            if lr < 1e-9:
                break

            # occasional jitter to escape shallow local minima
            if total_iters % 5000 == 0:
                jitter = np.random.randn(n) * 0.001
                a = project_nonneg_norm(a + jitter)

        # after finishing one length we do a quick random restart if time permits
        # (helps to explore different basins)
        # The restart logic is simple: we just loop again with a fresh Dirichlet,
        # but only once per length to keep the whole search manageable.
        if time.time() < deadline:
            # one extra restart
            a = project_nonneg_norm(np.random.dirichlet(np.full(n, 0.5)))
            lr = 0.2
            while time.time() < deadline:
                total_iters += 1
                a, b = gradient_step(a, target, lr)
                max_b = np.max(b)
                score = 2.0 * n * max_b
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New best (restart): {best_score:.6f} (n={n}, iter={total_iters})")
                lr *= 0.9997
                if lr < 1e-9:
                    break
                if total_iters % 5000 == 0:
                    a = project_nonneg_norm(a + np.random.randn(n) * 0.001)

    # Fallback – uniform sequence if nothing was found (should never happen)
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # convert to plain Python list of floats for the evaluator
    return [float(x) for x in best_seq]

# -*- coding: utf-8 -*-
"""
Gradient‑descent search for a non‑negative step function that minimises
``evaluate_sequence``.
The algorithm tries several sequence lengths and a few random restarts.
For a fixed length *n* we minimise the squared error between the auto‑
convolution a∗a and the flat target (a uniform distribution on
{0,…,2n‑2}).  The optimisation is performed by projected gradient descent
using FFT‑based convolutions – fast even for a few thousand coefficients.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real 1‑D arrays using an FFT of power‑of‑two size."""
    n = len(x) + len(y) - 1
    sz = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=sz)
    fy = np.fft.rfft(y, n=sz)
    conv = np.fft.irfft(fx * fy, n=sz)[:n]
    return conv


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projection onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    v = np.maximum(v, 0.0)
    s = np.sum(v)
    if s == 0.0:
        # fall back to a uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _gradient_step(a: np.ndarray, target: np.ndarray, lr: float):
    """
    One projected gradient step on the loss
        ½‖a∗a – target‖² .
    Returns the updated sequence and the current auto‑convolution.
    """
    n = len(a)

    # a∗a
    conv_aa = _fft_convolve(a, a)

    # error w.r.t. the flat target
    err = conv_aa - target

    # gradient of the squared error:
    # g_j = 4 * Σ_k err_k * a_{k‑j}
    a_rev = a[::-1]
    g_full = _fft_convolve(err, a_rev)               # length = 3n‑2
    g = 4.0 * g_full[n - 1 : n - 1 + n]               # size n

    # gradient descent step followed by projection onto the simplex
    a_new = a - lr * g
    a_new = _project_to_simplex(a_new)

    return a_new, conv_aa


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The routine respects the 1000‑second budget, tries several lengths,
    performs projected gradient descent on the squared error to a flat
    auto‑convolution, and keeps the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # Lengths we will try – a compromise between speed and theoretical bound.
    lengths = [500, 1000, 2000, 3000, 4000]
    # If the supplied construction (height_sequence_1) is not huge, also try its length.
    try:
        L = len(height_sequence_1)   # noqa: F821 – supplied by the evaluation harness
        if L <= 5000:
            lengths.append(L)
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')
    total_iters = 0

    for n in lengths:
        if time.time() >= deadline:
            break

        m = 2 * n - 1
        target = np.full(m, 1.0 / m, dtype=np.float64)   # flat target for a∗a

        # ---- initialisation for this length ----------------------------------
        init_seq = None
        try:
            if len(height_sequence_1) == n:   # noqa: F821
                init_seq = np.array(height_sequence_1, dtype=np.float64)
        except Exception:
            pass

        if init_seq is None:
            # Dirichlet(0.5) gives a fairly spread random start.
            init_seq = np.random.dirichlet(np.full(n, 0.5))

        a = _project_to_simplex(init_seq)
        lr = 0.2                     # initial learning rate
        max_iters_len = 15000        # safety bound per length
        iters_len = 0

        # ---- main optimisation loop -----------------------------------------
        while time.time() < deadline and iters_len < max_iters_len:
            total_iters += 1
            iters_len += 1

            a, conv_aa = _gradient_step(a, target, lr)

            # current score (sum(a)=1, therefore score = 2·n·max(conv_aa))
            cur_score = 2.0 * n * np.max(conv_aa)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New best = {best_score:.6f} "
                      f"(n={n}, iter={total_iters})")

            # exponential decay of the learning rate
            lr *= 0.9997
            if lr < 1e-9:
                break

            # occasional jitter to help escape shallow minima
            if total_iters % 5000 == 0:
                a = _project_to_simplex(a + np.random.randn(n) * 0.001)

        # ---- one extra random restart for the same length (if time permits) ----
        if time.time() < deadline:
            a = _project_to_simplex(np.random.dirichlet(np.full(n, 0.5)))
            lr = 0.2
            iters_len = 0
            while time.time() < deadline and iters_len < max_iters_len:
                total_iters += 1
                iters_len += 1

                a, conv_aa = _gradient_step(a, target, lr)
                cur_score = 2.0 * n * np.max(conv_aa)

                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New best (restart) = {best_score:.6f} "
                          f"(n={n}, iter={total_iters})")

                lr *= 0.9997
                if lr < 1e-9:
                    break
                if total_iters % 5000 == 0:
                    a = _project_to_simplex(a + np.random.randn(n) * 0.001)

    # Fallback – should never be needed, but guarantees a valid return value.
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # Convert to plain Python list of floats for the external evaluator.
    return [float(x) for x in best_seq]
