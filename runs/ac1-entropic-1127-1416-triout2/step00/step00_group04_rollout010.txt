def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simple baseline: return a uniform non‑negative step function.
    Since evaluate_sequence is scale‑invariant and the convolution
    contains the term ∑ a_i^2, the ratio 2·n·max(conv)/sum^2
    cannot be lower than 2 (Cauchy–Schwarz).  The uniform
    distribution attains this bound, so we return it directly.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Choose a reasonable length (e.g., 1000) and uniform height.
    length = 1000
    # height any positive number, 1.0 works.
    uniform_val = 1.0
    best_sequence = [uniform_val] * length
    best_score = evaluate_sequence(best_sequence)

    # Attempt some trivial random perturbations; they cannot improve the bound,
    # but we keep the loop for safety / to respect the time budget.
    while time.time() < deadline:
        # generate a small random jitter.
        seq = best_sequence.copy()
        idx = np.random.randint(length)
        seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_sequence = seq
            print(f"Found improved score: {best_score}")
        # break early if we see any improvement (unlikely)
        if best_score < 2.0 - 1e-12:
            break

    print(f"Returning sequence of length {len(best_sequence)} with score {best_score}")
    return best_sequence

def compute_score(seq):
    seq = np.array(seq, dtype=np.float64)
    sum_s = seq.sum()
    if sum_s < 0.01:
        return np.inf
    n = len(seq)
    # Use FFT
    fft_len = 2*n - 1
    # Next pow of 2 for speed:
    N = 1 << (fft_len-1).bit_length()
    f = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(f * f, n=N)[: 2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_s * sum_s)
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0
    n = 5000 # chosen
    seq = np.full(n, 1.0, dtype=np.float64)   # sum = n
    # optionally random perturbation to edges?
    best_seq = seq.copy()
    best_score = compute_score(best_seq)
    current_seq = seq.copy()
    current_score = best_score
    # Parameters
    temperature = 1.0
    decay = 0.9995  # per iteration
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two indices i, j distinct
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Choose delta: small fraction of sum, limited by current value at i
        max_delta = min(current_seq[i], 0.5)  # can't exceed half of current
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        # apply move
        new_seq = current_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # compute score
        new_score = compute_score(new_seq)
        # Accept if better or with probability exp(-(new_score - current_score)/temperature)
        if new_score < current_score or np.random.rand() < np.exp(-(new_score - current_score) / (temperature + 1e-12)):
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {it}, new best {best_score}")
        temperature *= decay
        # break early if near lower bound maybe
        if best_score < 1.51:
            break
    print(f"Finished after {it} iterations, best score {best_score}")
    return best_seq.tolist()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simulated‑annealing / gradient descent search for a step‑function
    whose self‑convolution is as flat as possible.
    Returns a list of non‑negative heights.
    """
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Parameter: length of the sequence
    n = 3000

    # Initial uniform distribution (any positive sum works, scaling is irrelevant)
    seq = np.full(n, 1.0, dtype=np.float64)   # sum = n

    # Quick sanity: evaluate initial score
    def score_of(arr):
        s = arr.sum()
        if s < 0.01:
            return np.inf
        conv = np.convolve(arr, arr)
        m = conv.max()
        return 2.0 * len(arr) * m / (s * s)

    best_seq = seq.copy()
    best_score = score_of(seq)

    current_seq = seq.copy()
    current_score = best_score

    # tuning constants
    min_delta = 1e-4
    max_delta = 0.5   # max amount transferred in a single step
    temperature = 1.0
    decay = 0.9999

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # compute convolution and locate the index with maximal value
        conv = np.convolve(current_seq, current_seq)
        k_max = int(np.argmax(conv))

        # gradient of the active constraint (2 * a_{k_max-i})
        grad = np.zeros_like(current_seq)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        jdx = k_max - idx
        grad[idx] = 2.0 * current_seq[jdx]

        # choose donor (large gradient, non‑zero amount) and receiver (small gradient)
        donor_candidates = np.where(current_seq > min_delta)[0]
        if donor_candidates.size == 0:
            break
        donor_grad = grad[donor_candidates]
        donor = donor_candidates[np.argmax(donor_grad)]

        receiver_candidates = np.where(current_seq < 1000.0 - min_delta)[0]
        if receiver_candidates.size == 0:
            break
        receiver_grad = grad[receiver_candidates]
        receiver = receiver_candidates[np.argmin(receiver_grad)]

        # size of the transfer
        allowable = min(current_seq[donor] - min_delta,
                         1000.0 - current_seq[receiver] - min_delta)
        if allowable <= 0:
            # cannot move, pick another pair
            continue
        delta = np.random.rand() * min(max_delta, allowable)

        # apply move
        new_seq = current_seq.copy()
        new_seq[donor] -= delta
        new_seq[receiver] += delta

        new_score = score_of(new_seq)

        # accept if improvement, otherwise maybe accept with a small probability
        accept = False
        if new_score < current_score:
            accept = True
        else:
            # simulated‑annealing acceptance probability
            prob = np.exp(-(new_score - current_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current_seq = new_seq
            current_score = new_score
            temperature *= decay
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iteration}: new best score = {best_score:.6f}")

        # occasional random shake‑up to escape flat regions

"""
Simple simulated‑annealing / gradient‑descent search for a
non‑negative step function whose self‑convolution is as flat as possible.
The evaluation metric is

    2 * n * max(conv(f,f)) / (sum(f)**2)

which we try to minimise.  The algorithm starts from a uniform (or a
previously found) sequence, repeatedly moves a small amount of mass from
indices that contribute most to the current maximal convolution entry to
indices that contribute least, and accepts the move if it improves the
score (or with a tiny probability to escape plateaus).  The whole search
stops after the allotted budget (default 1000 s) and returns the best
sequence encountered.
"""

import time
import numpy as np

def _score(seq: np.ndarray) -> float:
    """
    Exact score as defined by `evaluate_sequence`.
    seq must be a 1‑D numpy array of non‑negative floats.
    """
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    m = conv.max()
    return 2.0 * len(seq) * m / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves upon the
    current best bound of 2.0.  The routine runs for up to ``budget_s`` seconds
    and keeps the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # a tiny safety margin

    # ----------------------------------------------------------------------
    # Initial sequence
    # ----------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # start from the previously discovered construction (if present)
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.size
    else:
        # otherwise use a modest uniform start (any length works)
        n = 3000
        init = np.full(n, 1.0, dtype=np.float64)

    # Ensure the sequence respects the built‑in clipping limits
    init = np.clip(init, 0.0, 1000.0)

    current_seq = init.copy()
    current_score = _score(current_seq)

    best_seq = current_seq.copy()
    best_score = current_score

    # ----------------------------------------------------------------------
    # Hyper‑parameters
    # ----------------------------------------------------------------------
    min_delta = 1e-4          # smallest amount we ever move
    max_delta = 0.5           # upper bound on a single move
    temperature = 1.0         # initial temperature for occasional uphill steps
    decay = 0.9999            # exponential decay per accepted move

    iteration = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ---- 1. locate the currently maximal convolution entry ----------------
        conv = np.convolve(current_seq, current_seq)
        k_max = int(np.argmax(conv))

        # ---- 2. compute the gradient of that entry w.r.t. the variables -------
        #    d/d a_i  ( sum_j a_j a_{k_max-j} ) = 2 * a_{k_max-i}  (if in range)
        grad = np.zeros_like(current_seq)

        i_start = max(0, k_max - (n - 1))
        i_end   = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        jdx = k_max - idx
        grad[idx] = 2.0 * current_seq[jdx]

        # ---- 3. pick a donor (large gradient, non‑zero) and a receiver -------
        donor_candidates = np.where(current_seq > min_delta)[0]
        if donor_candidates.size == 0:
            break
        donor_grad = grad[donor_candidates]
        donor = donor_candidates[np.argmax(donor_grad)]

        receiver_candidates = np.where(current_seq < 1000.0 - min_delta)[0]
        if receiver_candidates.size == 0:
            break
        receiver_grad = grad[receiver_candidates]
        receiver = receiver_candidates[np.argmin(receiver_grad)]

        # ---- 4. decide how much to transfer -----------------------------------
        allowable = min(current_seq[donor] - min_delta,
                        1000.0 - current_seq[receiver] - min_delta)
        if allowable <= 0:
            # nothing sensible can be moved this round – try a random shake
            continue
        delta = np.random.rand() * min(max_delta, allowable)
        delta = max(delta, min_delta)

        # ---- 5. propose the new sequence --------------------------------------
        new_seq = current_seq.copy()
        new_seq[donor] -= delta
        new_seq[receiver] += delta

        new_score = _score(new_seq)

        # ---- 6. accept / reject ------------------------------------------------
        accept = False
        if new_score < current_score:
            accept = True
        else:
            # very small probability to accept a worsening move (simulated annealing)
            prob = np.exp(-(new_score - current_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current_seq = new_seq
            current_score = new_score
            temperature *= decay
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ---- 7. occasional random perturbation to avoid dead‑ends ---------------
        if iteration % 500 == 0:
            # tiny zero‑sum random jitter
            i, j = np.random.randint(n, size=2)
            eps = (np.random.rand() - 0.5) * 0.01
            # clamp the perturbation so we stay in bounds
            eps = max(- (current_seq[i] - min_delta), eps)
            eps = min(1000.0 - current_seq[i] - min_delta, eps)
            pert_seq = current_seq.copy()
            pert_seq[i] -= eps
            pert_seq[j] += eps
            pert_score = _score(pert_seq)
            if pert_score < current_score:
                current_seq = pert_seq
                current_score = pert_score

    print(f"Search finished after {iteration} iterations. "
          f"Best score = {best_score:.6f}")
    # Return a plain Python list as required by the outer harness
    return best_seq.tolist()
