import numpy as np, time, scipy.optimize as opt

def solve_lp_fixed_f(f_seq, M):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum -> minimize -sum
    # Build constraints
    rows = []
    b = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # g_j coefficient is f_{k-j}
        start = max(0, k-(n-1))
        end = min(n-1, k)
        # for j from start..end: row[j] = f_seq[k-j]
        # vectorized
        idx_j = np.arange(start, end+1)
        row[idx_j] = f_seq[k - idx_j]
        rows.append(row)
        b.append(M)
    # nonneg constraints as -I g <= 0
    rows.append(-np.eye(n))
    b.append(np.zeros(n))
    A = np.vstack(rows)
    b = np.concatenate(b)
    # solve
    res = opt.linprog(c, A_ub=A, b_ub=b, method='highs', options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    else:
        return None
```

        if iteration % 10 == 0:
            new_seq += np.random.randn(len(new_seq))*0.001

        # evaluate

        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"Iter {iteration}, new best {best_score:.6f}, t={t:.3f}")
        # accept or reject?

        if score <= best_score + 1e-9:  # improvement or approx same

"""Propose a candidate sequence.

The algorithm consists of two phases:

1. Parameterised search – we generate monotone decreasing sequences of the form
   a_i = (i + offset)^{-p} (with offset>0 to avoid the singularity at i=0).
   The parameters n (length), offset and exponent p are sampled randomly
   (and also explored locally).  For each generated sequence we evaluate the
   objective defined in ``evaluate_sequence`` and keep the best one found.

2. Local refinement – starting from the best sequence we run a few iterations
   of the LP‑based improvement used in the previous state‑of‑the‑art
   implementation (alternating convex programming).  The LP improves the
   sum under the same infinity‑norm bound on the convolution; we mix the
   new vector into the current one with a decaying step size.

Both phases respect the time budget (default 1000 s) and guarantee that a
valid non‑negative list of floats is returned.
"""
import numpy as np
import time
from scipy.optimize import linprog

def generate_power_sequence(n, offset, p):
    """Return a monotone decreasing sequence a_i = (i+offset)^(-p)."""
    i = np.arange(n, dtype=float)
    seq = (i + offset) ** (-p)
    # Scale to avoid values >1000 (does not affect the score because the
    # objective is scale‑invariant).
    max_val = seq.max()
    if max_val > 1000.0:
        seq *= 1000.0 / max_val
    return seq

def evaluate(seq):
    """Wrap the provided evaluate_sequence with a list conversion."""
    return evaluate_sequence(seq.tolist())

def solve_conv_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise sum(b)
        subject to conv(f_seq, b) <= rhs, b >= 0.
    Returns the optimal b as a NumPy array or None if infeasible.
    """
    n = len(f_seq)
    c = -np.ones(n)               # maximise sum(b)  -> minimise -sum(b)
    # Build the convolution constraints: for each k in 0..2n-2
    rows = []
    bounds = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # only indices j with 0<=j<n and 0<=k-j<n contribute
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        if low <= high:
            j = np.arange(low, high + 1)
            row[j] = f_seq[k - j]
        rows.append(row)
        bounds.append(rhs)
    # Non‑negativity constraints  -b <= 0
    rows.append(-np.eye(n))
    bounds.append(np.zeros(n))
    A = np.vstack(rows)
    b = np.concatenate(bounds)
    # Use the HIGHs solver – fast for the moderate sizes we use
    res = linprog(c, A_ub=A, b_ub=b, method='highs', options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None

def improve_with_lp(seq, max_iter, deadline):
    """
    Perform a few LP‑based improvement steps (the same idea as the SOTA
    algorithm).  ``seq`` is modified in‑place and the best version seen
    during the walk is returned.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = evaluate(best_seq)
    t = 0.5                                 # start with a relatively large mixing step
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # current bound M
        conv = np.convolve(seq, seq)
        M = float(conv.max())
        # solve the linear program to get a direction with a larger sum
        g = solve_conv_lp(seq, M)
        if g is None:
            # fallback – small random perturbation
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
            continue
        # renormalise g to have the same L1‑norm as seq (the LP already tends to increase it)
        sum_seq = seq.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g * (sum_seq / sum_g)
        # convex combination
        new_seq = (1.0 - t) * seq + t * g
        # keep within allowed numeric range
        new_seq = np.clip(new_seq, 0.0, None)
        # evaluate
        cur_score = evaluate(new_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            # print progress inside the LP loop
            print(f"   LP‑step {it}: improved score {best_score:.6f}  (t={t:.3f})")
        # accept the new point (even if not better – it usually reduces the peak)
        seq = new_seq
        # decay the step size slowly
        t = max(0.01, t * 0.995)
    return best_seq.tolist(), best_score

"""
Propose a candidate sequence.

The algorithm consists of two phases:

1️⃣  Randomised parametric search – we generate monotone decreasing
    sequences of the form a_i = (i + offset) ** (‑p) (with offset > 0 to
    avoid the singularity at i = 0).  The parameters (length n,
    offset and exponent p) are sampled many times (and also explored
    locally).  For each generated sequence we call ``evaluate_sequence``
    and keep the best one found.

2️⃣  LP‑based polishing – starting from the best sequence we perform a
    few iterations of the LP improvement used in the previous state‑of‑the‑art
    implementation.  The LP maximises the ℓ₁‑norm of a new vector g while
    keeping the convolution with the current vector bounded by the same
    infinity‑norm M.  The new vector is mixed into the current one with a
    decaying step size.  This step often reduces the peak of the
    self‑convolution further.

Both phases respect the time budget (default 1000 s) and guarantee that a
valid non‑negative list of floats is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def generate_power_sequence(n: int, offset: int, p: float) -> np.ndarray:
    """Return a monotone decreasing sequence a_i = (i + offset) ** (‑p)."""
    i = np.arange(n, dtype=float)
    seq = (i + offset) ** (-p)
    # ensure we stay below the hard 1000‑value cap (does not affect the score)
    mx = seq.max()
    if mx > 1000.0:
        seq *= 1000.0 / mx
    return seq


def evaluate(seq) -> float:
    """Convenient wrapper around the imported ``evaluate_sequence``."""
    return evaluate_sequence(seq.tolist())


def solve_conv_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve the LP
        maximise   sum(b)
        subject to  conv(f_seq, b) <= rhs
                    b >= 0
    Returns the optimal b as a NumPy array or ``None`` if the solver fails.
    """
    n = len(f_seq)
    # we minimise -sum(b)  =>  maximise sum(b)
    c = -np.ones(n)

    rows = []
    bounds = []

    # convolution constraints: for each k = 0 .. 2n‑2
    for k in range(2 * n - 1):
        row = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            row[j] = f_seq[k - j]
        rows.append(row)
        bounds.append(rhs)

    # non‑negativity  –b <= 0
    rows.append(-np.eye(n))
    bounds.append(np.zeros(n))

    A = np.vstack(rows)
    b = np.concatenate(bounds)

    # Highs is fast for the modest sizes we use (n ≲ 3000)
    res = linprog(c, A_ub=A, b_ub=b, method="highs",
                  options={"presolve": True, "time_limit": 5.0})
    if res.success:
        return res.x
    return None


def improve_with_lp(seq: np.ndarray, max_iter: int, deadline: float):
    """
    Perform a few LP‑based improvement steps (the same idea as the SOTA
    algorithm).  ``seq`` is modified in‑place; the best version seen is
    returned together with its score.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = evaluate(best_seq)
    t = 0.5  # fairly aggressive initial mixing

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = np.convolve(seq, seq)
        M = float(conv.max())

        g = solve_conv_lp(seq, M)
        if g is None:
            # fallback – tiny random tweak
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
            continue

        sum_seq = seq.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break

        # scale g to have the same ℓ₁‑norm as seq
        g = g * (sum_seq / sum_g)

        new_seq = (1.0 - t) * seq + t * g
        new_seq = np.clip(new_seq, 0.0, None)

        cur_score = evaluate(new_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            print(f"   LP step {it:03d}: improved score {best_score:.6f} (t={t:.3f})")

        seq = new_seq
        t = max(0.01, t * 0.995)  # gradual decay

    return best_seq.tolist(), best_score
