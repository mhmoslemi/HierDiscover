def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = np.inf

    # optionally start from height_sequence_1 if present:
    if 'height_sequence_1' in globals():
        seq = list(height_sequence_1)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq

    # define lengths to try
    lengths = [500, 800, 1200, 1600, 2000, 2500, 3000]
    # define parameter sampling sets
    while time.time() < deadline:
        # pick random family
        family = np.random.choice(['exp','pow','linear','mixed'])
        n = int(np.random.choice(lengths))
        if family == 'exp':
            # sample alpha in log scale
            log_alpha = np.random.uniform(np.log10(1e-4), np.log10(1))
            alpha = 10**log_alpha
            # generate sequence
            seq = np.exp(-alpha * np.arange(n))
        elif family == 'pow':
            p = np.random.uniform(0.0, 2.5)
            seq = 1.0 / ((np.arange(1, n+1))**p)
        elif family == 'linear':
            # a_i = max(0, 1 - beta * i) maybe shift so sum>0
            beta = np.random.uniform(0.0, 2.0/n) # ensures some positive tail
            seq = np.maximum(0.0, 1.0 - beta * np.arange(n))
        elif family == 'mixed':
            # mix exponential with constant
            log_alpha = np.random.uniform(np.log10(1e-4), np.log10(1))
            alpha = 10**log_alpha
            eps = np.random.uniform(0.0, 0.5)
            seq = np.exp(-alpha * np.arange(n)) + eps
        else:
            continue

        # ensure non-negative and sum>0.01 after clipping? evaluate will clip anyway.
        # But important: sum may be huge; but evaluation scaling independent.
        # However if we produce extremely large values > 1000, evaluate will cap them to 1000.
        # This may distort shape due to clipping. Avoid huge values: we can rescale sequence to max 1 before passing.
        # So we can normalize to have max <= 1 for numeric stability (scaling invariant).
        seq = seq.astype(float)
        max_val = seq.max()
        if max_val > 0:
            seq = seq / max_val  # scale max to 1
        # also ensure not all zeros
        if seq.sum() < 1e-6:
            continue

        # convert to list
        seq_list = seq.tolist()
        # evaluate
        try:
            score = evaluate_sequence(seq_list)
        except Exception as e:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq_list
            print(f"New best: {best_score:.6f} using {family} n={n}")
        # Possibly also do small local optimization: random perturb of a few entries
        # if we have best_seq, try neighbor modifications
        if best_seq is not None and time.time() < deadline:
            # random neighbor search a few times
            for _ in range(5):
                # copy best_seq
                cand = best_seq.copy()
                # modify up to 5 random indices by factor (1+delta)
                k = np.random.randint(len(cand))
                delta = np.random.normal(scale=0.1)
                cand[k] = max(0.0, cand[k] * (1 + delta))
                # re-normalize max to 1 to avoid clipping
                max_c = max(cand)
                if max_c > 0:
                    cand = [x / max_c for x in cand]
                try:
                    sc = evaluate_sequence(cand)
                except Exception:
                    continue
                if sc < best_score:
                    best_score = sc
                    best_seq = cand
                    print(f"Improved by perturb: {best_score:.6f}")
        # Also occasionally try the LP-direction method (from given code).
        # However we would need solve_convolution_lp etc., which may be heavy. Perhaps skip.

    if best_seq is None:
        # fallback: uniform
        n = 500
        best_seq = [1.0]*n
    return best_seq

"""Use parametric families and hill climbing to minimize evaluate_sequence.
We sample exponential, power‐law and linear decays of various lengths,
keep the best and perform local stochastic hill‑climbing on the current
best. The algorithm stops when the time budget is exhausted and returns
the best sequence found.
"""

import time, math, random
import numpy as np

def _score(seq):
    # Wrapper to handle errors
    try:
        return evaluate_sequence(seq)
    except Exception:
        return math.inf

def _normalize(seq):
    # Normalize to max 1 to keep scaling invariant and stay in safe range.
    seq = np.array(seq, dtype=float)
    mx = seq.max()
    if mx > 0:
        seq = seq / mx
    return seq.tolist()

def _make_exponential(n, alpha):
    seq = np.exp(-alpha * np.arange(n, dtype=float))
    return _normalize(seq)

def _make_powerlaw(n, p):
    idx = np.arange(1, n+1, dtype=float)
    seq = idx ** (-p)
    return _normalize(seq)

def _make_linear(n, beta):
    # a_i = max(0, 1 - beta*i)
    seq = np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float))
    return _normalize(seq)

def _make_mixed(n, alpha, eps):
    seq = np.exp(-alpha * np.arange(n, dtype=float)) + eps
    return _normalize(seq)

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    ``evaluate_sequence``.  The routine explores a few parametric families
    (exponential decay, power‑law decay, linear decay and a mixed exponential
    + constant) with random hyper‑parameters and lengths, keeps the best
    candidate and then applies a simple stochastic hill‑climbing refinement.
    The total wall‑clock time is limited by ``budget_s`` seconds.
    """
    import numpy as np, time, random, math

    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # leave a margin

    # ------------------------------------------------------------------
    # 1) initialise best candidate (maybe the previous best)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    # Try to use the known construction if present.
    if 'height_sequence_1' in globals():
        try:
            seq0 = _normalize(list(height_sequence_1))
            sc0 = _score(seq0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = seq0
                print(f"Starting from height_sequence_1 → score {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) Main search loop: generate parametric candidates and occasionally
    #    perform a local perturbation of the current best.
    # ------------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1500, 2000, 2500, 3000]
    # We will allocate about 80 % of the time to random sampling and the
    # remaining to hill‑climbing.
    hill_start_time = deadline - (budget_s * 0.2)

    while time.time() < deadline:
        # Decide whether we are in the hill‑climbing phase.
        now = time.time()
        if (best_seq is not None) and (now > hill_start_time):
            # Hill‑climbing step: perturb the current best.
            cand = np.array(best_seq, copy=True)
            n = len(cand)
            # number of coordinates to touch (1‑5)
            m = np.random.randint(1, min(5, n) + 1)
            idxs = np.random.choice(n, m, replace=False)
            # Apply multiplicative Gaussian perturbation.
            for i in idxs:
                factor = 1.0 + np.random.normal(scale=0.1)
                cand[i] = max(0.0, cand[i] * factor)
            # Renormalise maximal value to 1.
            mx = cand.max()
            if mx > 0:
                cand = cand / mx
            cand_list = cand.tolist()
            sc = _score(cand_list)
            if sc < best_score:
                best_score = sc
                best_seq = cand_list
                print(f"hill‑climb improvement: score {best_score:.6f}")
            continue

        # --- generate a fresh candidate from a parametric family ---
        n = int(np.random.choice(lengths))
        family = np.random.choice(['exp', 'pow', 'lin', 'mix'])

        if family == 'exp':
            # log‑uniform sampling of α ∈ [1e‑4, 5]
            log_alpha = np.random.uniform(-4, np.log10(5))
            alpha = 10 ** log_alpha
            cand_seq = _make_exponential(n, alpha)
        elif family == 'pow':
            p = np.random.uniform(0.0, 3.0)          # power‑law exponent
            cand_seq = _make_powerlaw(n, p)
        elif family == 'lin':
            # β controls how fast the linear decay goes to zero.
            # Choose β so that at least a few entries stay positive.
            beta = np.random.uniform(0.0, 2.0 / n)
            cand_seq = _make_linear(n, beta)
        else:  # mixed exponential + constant
            log_alpha = np.random.uniform(-4, np.log10(5))
            alpha = 10 ** log_alpha
            eps = np.random.uniform(0.0, 0.5)
            cand_seq = _make_mixed(n, alpha, eps)

        # Guard against pathological all‑zero vectors (should not happen)
        if sum(cand_seq) < 1e-8:
            continue

        sc = _score(cand_seq)
        if sc < best_score:
            best_score = sc
            best_seq = cand_seq
            print(f"new best ({family}) n={n} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final fallback if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500    # uniform fallback, score = 2
        best_score = _score(best_seq)

    print(f"final best score {best_score:.6f} with length {len(best_seq)}")
    return best_seq

"""Search for a low‑score step‑function.

The algorithm explores a few simple parametric families (exponential,
power‑law, linear, mixed exponential+constant, logistic and random‑decreasing
sequences) with random hyper‑parameters and lengths.  The best candidate is
kept and, once a decent solution is found, a stochastic hill‑climbing phase
perturbs the current best to try to lower the value further.  All
operations respect the non‑negative requirement; the sequence is always
renormalised so that its maximum entry equals 1 (the objective is scaling‑
invariant, this avoids overflow and keeps the numbers in the safe range of
``evaluate_sequence``).  The routine stops when the time budget expires and
returns the best sequence discovered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _score(seq):
    """Safely evaluate a candidate; return +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:               # pragma: no cover
        return math.inf

def _normalize(seq):
    """Scale so that max(seq) == 1 (no‑op if all zeros)."""
    arr = np.asarray(seq, dtype=float)
    mx = arr.max()
    if mx > 0:
        arr = arr / mx
    return arr.tolist()

def _exp_seq(n, alpha):
    """Exponential decay a_i = exp(-alpha * i)."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)))

def _pow_seq(n, p):
    """Power‑law decay a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    return _normalize(idx ** (-p))

def _lin_seq(n, beta):
    """Linear decay a_i = max(0, 1 - beta*i)."""
    return _normalize(np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float)))

def _mix_seq(n, alpha, eps):
    """Mixed exponential + constant a_i = exp(-alpha*i) + eps."""
    return _normalize(np.exp(-alpha * np.arange(n, dtype=float)) + eps)

def _logistic_seq(n, steep):
    """Logistic decay a_i = 1/(1+exp((i-mid)/steep))."""
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    seq = 1.0 / (1.0 + np.exp((i - mid) / steep))
    return _normalize(seq)

def _randdec_seq(n):
    """Random decreasing sequence (sorted random numbers)."""
    arr = np.random.rand(n)
    arr.sort()
    arr = arr[::-1]                     # descending
    return _normalize(arr)

# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine spends roughly ``budget_s`` seconds
    searching; it mixes random sampling of parametric families with a
    stochastic hill‑climbing refinement of the best candidate found so far.
    """
    np.random.seed(seed)
    random.seed(seed)

    # overall deadline (a tiny margin to avoid overshooting)
    deadline = time.time() + budget_s - 0.3

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0) start from the globally supplied construction if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = _normalize(list(height_sequence_1))
            sc = _score(init)
            if sc < best_score:
                best_seq, best_score = init, sc
                print(f"Initial reference score: {best_score:.6f}")
        except Exception:               # pragma: no cover
            pass

    # possible lengths (the search prefers a few thousand entries, but also
    # tries smaller sizes for speed)
    lengths = [400, 600, 800, 1000, 1400, 1800, 2200, 2600, 3000]

    # ------------------------------------------------------------------
    # 1) random sampling of parametric families
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Occasionally try a local perturbation of the current best.
        if best_seq is not None and random.random() < 0.15:
            # ----- hill‑climbing step -------------------------------------------------
            cand = np.array(best_seq, copy=True)
            n = len(cand)
            # modify a few random positions multiplicatively
            m = random.randint(1, min(5, n))
            idxs = random.sample(range(n), m)
            for i in idxs:
                factor = 1.0 + random.gauss(0.0, 0.08)
                cand[i] = max(0.0, cand[i] * factor)
            # renormalise max to 1 (scale does not affect the score)
            mx = cand.max()
            if mx > 0:
                cand = cand / mx
            cand = cand.tolist()
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
                print(f"Hill‑climb improvement → {best_score:.6f}")
            continue

        # ----- generate a fresh candidate -----------------------------------------
        n = int(random.choice(lengths))
        family = random.choice(['exp', 'pow', 'lin', 'mix', 'log', 'randdec'])

        if family == 'exp':
            # log‑uniform α ∈ [1e‑4, 5]
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            cand = _exp_seq(n, alpha)

        elif family == 'pow':
            p = random.uniform(0.0, 3.0)          # exponent
            cand = _pow_seq(n, p)

        elif family == 'lin':
            # β such that a few entries stay positive
            beta = random.uniform(0.0, 2.0 / n)
            cand = _lin_seq(n, beta)

        elif family == 'mix':
            log_a = random.uniform(-4, math.log10(5))
            alpha = 10 ** log_a
            eps = random.uniform(0.0, 0.5)
            cand = _mix_seq(n, alpha, eps)

        elif family == 'log':
            steep = random.uniform(0.01, n / 2.0)   # width of the logistic transition
            cand = _logistic_seq(n, steep)

        else:  # 'randdec'
            cand = _randdec_seq(n)

        # discard pathological all‑zero vectors (should not happen)
        if sum(cand) < 1e-8:
            continue

        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"New best ({family}) n={n} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) final fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500
        best_score = _score(best_seq)

    print(f"Finished search – best score {best_score:.6f} (len={len(best_seq)})")
    return best_seq
