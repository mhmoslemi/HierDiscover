                seq = seq / np.sum(seq) * target_sum
                continue

            # --- bring g to the required ℓ₁‑norm -----------------------
            sum_g = np.sum(g)
            if sum_g <= 0:
                break
            g = g * target_sum / sum_g

            # --- try convex combinations of various step sizes ---------
            improved = False
            for t in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                cand = (1.0 - t) * seq + t * g
                cand_score = _score(cand)
                if cand_score < cur_score - 1e-12:       # strict improvement
                    seq = cand
                    cur_score = cand_score
                    improved = True
                    break

            if not improved:
                # small random redistribution of mass
                stagnation += 1
                if stagnation > 80:              # give up on this restart
                    break
                i, j = np.random.choice(n, size=2, replace=False)
                delta = np.random.rand() * seq[i] * 0.05   # move ≤ 5 % of i to j
                seq[i] -= delta
                seq[j] += delta
                seq = np.maximum(seq, 0.0)
                seq = seq / np.sum(seq) * target_sum
            else:
                stagnation = 0                 # reset counter after progress

            # keep the best overall solution seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[update] n={n:4d} score={best_score:.6f}")

        # end of one restart
    # ----------------------------------------------------------------------
    # Fallback: if nothing succeeded, return a simple uniform vector.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 300
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)

    print("Final best score:", best_score)
    return best_seq.tolist()
