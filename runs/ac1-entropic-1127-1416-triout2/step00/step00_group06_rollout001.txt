"""
Gradient‐descent based optimizer for the step‑function evaluation problem.

The objective for a sequence a (length n) is
    R(a) = 2·n·max(conv(a,a)) / (sum(a)^2).

For a fixed length the sum factor cancels out if we rescale a, so the problem
reduces to flattening the autocorrelation (the convolution) while keeping all
entries non‑negative.  Starting from a symmetric uniform (or slightly noisy)
sequence we perform many iterations of a simple projected gradient step that
moves mass away from the location where the convolution attains its maximum.
The update direction is derived from the exact gradient of R and is cheap to
compute:

    dR/da_i ∝ (S·a_{k−i} – M)   (k = argmax conv, S = sum a, M = max conv)

The opposite direction (M – S·a_{k−i}) is used as a descent step.
After each step we project onto the non‑negative orthant and enforce symmetry
(a_i = a_{n−1−i}) which reduces the number of degrees of freedom and often
helps the autocorrelation become flatter.

The optimizer runs several restarts with different lengths (up to a few
thousand entries) until the time budget expires, keeping the best sequence
found.  Convolutions are computed with an FFT (linear complexity) to keep the
iteration cheap.

The function `propose_candidate` conforms to the required signature and
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real input)."""
    n = a.size
    # size must be at least 2*n-1 and a power of two for speed
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    C = np.fft.irfft(A * A, n=size)
    return C[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a given sequence."""
    n = a.size
    S = a.sum()
    if S < 0.01:
        return np.inf
    maxb = np.max(_conv_fft(a))
    return 2.0 * n * maxb / (S * S)


def _gradient_direction(a: np.ndarray, max_idx: int, maxb: float, S: float) -> np.ndarray:
    """
    Compute the descent direction (M – S·shifted) where
    shifted[i] = a[max_idx - i] if the index is in range, else 0.
    """
    n = a.size
    idx = np.arange(n)
    j = max_idx - idx
    mask = (j >= 0) & (j < n)
    shifted = np.zeros_like(a)
    shifted[mask] = a[j[mask]]
    # direction = M - S * shifted
    return maxb - S * shifted

# ----------------------------------------------------------------------

            # projection onto feasible set: non‑negative & symmetry
            a_new = np.maximum(a_new, 0.0)
            a_new = (a_new + a_new[::-1]) / 2.0

            new_score = _score(a_new)

            # accept if we made progress, otherwise shrink the step size
            if new_score < curr_score:
                a = a_new
                curr_score = new_score
                # store globally if this is the best so far
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.tolist()
                    print(f"[improve] n={n} it={it} score={new_score:.6f}")
            else:
                lr *= 0.5
                if lr < min_lr:
                    lr = 0.2   # reset to avoid stagnation

            # decay learning rate gently
            lr = max(lr * (1.0 - decay), min_lr)

            # occasional random jitter to escape flat regions
            if it % 1000 == 0 and it > 0:
                jitter = 0.001 * np.mean(a) * np.random.randn(n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = (a + a[::-1]) / 2.0

            it += 1

    # ------------------------------------------------------------------
    # final fallback – ensure we return a non‑empty list
    # ------------------------------------------------------------------
    if best_seq is None or len(best_seq) == 0:
        # simple uniform fallback (should never happen)
        n = 256
        best_seq = [np.sqrt(2.0 * n) / n] * n

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
