"""Propose a candidate step function sequence.

The algorithm performs a quick random search over a few parametric families
(exponential, power‑law, linear) to obtain a shape whose convolution peak is
lower than the uniform distribution (the current 2.0 bound).  The best
candidate is then refined by a simple simulated‑annealing move that preserves
the total ℓ₁‑norm while trying to lower the maximal convolution coefficient.
The whole process respects the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

def evaluate(seq):
    # evaluate_sequence is provided in the environment
    return evaluate_sequence(seq)

def generate_exponential(n, c):
    """Exponential decay a_i = r**i with r = 1 - c/n, normalized sum."""
    r = max(0.0, 1.0 - c / n)
    a = np.power(r, np.arange(n, dtype=float))
    # avoid all zeros
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a

def generate_powerlaw(n, p):
    """Power‑law decreasing a_i = ((n-i)/n)**p, normalized."""
    idx = np.arange(n, dtype=float)
    a = ((n - idx) / n) ** p
    a /= a.sum()
    return a

def generate_linear(n):
    """Linear decreasing a_i = n-i, normalized."""
    a = (np.arange(n, 0, -1, dtype=float))
    a /= a.sum()
    return a

def solve_convolution_lp(f_seq, bound):
    """Linear program: maximise Σ b_j   s.t.  (f * b)_k ≤ bound,  b ≥ 0.

    Returns the optimal b as a 1‑D NumPy array or None on failure.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)
    # Build the constraints (f * b)_k ≤ bound.
    a_rows = []
    b_vals = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_rows.append(row)
        b_vals.append(bound)
    # non‑negativity: -I * b ≤ 0
    a_rows.append(-np.eye(n))
    b_vals.append(np.zeros(n))
    A_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_vals)

    # use the HiGHS interior‑point solver (fast for moderate n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def lp_improve(seq):
    """Try one LP step: compute a partner g with the same (or larger) ℓ₁‑norm
    that respects the current max‑convolution bound.  If g itself improves
    the score we return it, otherwise we return a convex combination that
    reduces the score (if any).  The caller may keep the original sequence
    when the function returns None.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    bound = conv.max()
    g = solve_convolution_lp(seq, bound)
    if g is None:
        return None
    # Evaluate both candidates.
    score_g = evaluate(g.tolist())
    score_f = evaluate(seq.tolist())
    if score_g < score_f:
        return g.tolist()
    # Try a few mixing coefficients.
    for t in (0.2, 0.5, 0.8):
        mixed = (1 - t) * seq + t * g
        score_m = evaluate(mixed.tolist())
        if score_m < score_f:
            return mixed.tolist()
    return None

def anneal_refine(seq, deadline):
    """Simple simulated‑annealing style local search that preserves the total sum."""
    n = len(seq)
    best = seq.copy()
    best_score = evaluate(best.tolist())
    cur = seq.copy()
    cur_score = best_score

    # temperature schedule
    temperature = 1.0
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose moving a small amount from i to j
        delta = np.random.normal(scale=0.02) * cur[i]   # relative change
        if delta > cur[i]:
            delta = cur[i]
        if delta < -cur[j]:
            delta = -cur[j]
        if abs(delta) < 1e-12:
            continue
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # evaluate
        new_score = evaluate(new.tolist())
        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                print(f"Anneal iter {it}: new best {best_score:.6f}")
        # simple cooling
        if it % 500 == 0:
            temperature *= 0.9
    return best

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Main entry point used by the grader.

    The routine builds a candidate sequence within the allocated time budget.
    The strategy is:
        1. Randomly pick a length (between 150 and 800).
        2. Sample a handful of parametric families (exponential, power‑law,
           linear) and keep the best according to evaluate().
        3. Apply a short LP‑based improvement step.
        4. Finish with a light simulated‑annealing refinement.
    The final sequence is returned as a plain list of floats.
    """
    np.random.seed(seed)
    random = np.random

    # leave a safe margin for the final return
    deadline = time.time() + budget_s - 5.0

    # 1. pick a reasonable length – not too small (poor bound) and not too big
    n = int(random.randint(150, 800))
    best_seq = None
    best_score = float('inf')

    # 2. random search in simple families
    families = ['exp', 'pow', 'lin']
    for _ in range(120):   # about a few hundred evaluations – cheap for n≤800
        if time.time() >= deadline:
            break
        typ = random.choice(families)
        if typ == 'exp':
            c = random.uniform(0.8, 2.5)   # parameter controlling the decay
            cand = generate_exponential(n, c)
        elif typ == 'pow':
            p = random.uniform(0.3, 5.0)
            cand = generate_powerlaw(n, p)
        else:   # linear
            cand = generate_linear(n)
        # occasional random jitter to avoid exact patterns
        if random.rand() < 0.1:
            cand += 0.001 * random.rand(n)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
        score = evaluate(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"Initial family {typ} improvement: score {best_score:.6f}")

    # 3. a couple of LP improve steps (if time permits)
    for _ in range(12):
        if time.time() >= deadline:
            break
        improved = lp_improve(best_seq)
        if improved is not None:
            new_score = evaluate(improved)
            if new_score < best_score:
                best_score = new_score
                best_seq = np.array(improved)
                print(f"LP improvement: score {best_score:.6f}")

    # 4. final annealing refinement (light, respects remaining budget)
    if time.time() < deadline - 1.0:
        best_seq = anneal_refine(best_seq, deadline)

    # Return as plain list of native Python floats
    return best_seq.tolist()

"""
Search for a step‑function sequence with a low evaluation score.

The strategy builds a small “core” that already has a low
max‑convolution (by using an exponential decay) and then appends a very long
tail of tiny constant values.  The tail increases the ℓ₁‑norm far more than
the maximal convolution, thereby reducing the quantity

    2·n·max(a*a) / (sum a)²

which the grader evaluates.  Randomly varying the core size,
the exponent of the decay and the total length explores the space quickly,
while each candidate is cheap to evaluate (a single convolution).

The best sequence found before the time budget expires is returned.
"""
import time
import numpy as np
from scipy.optimize import linprog

# evaluate_sequence is provided by the grader; we call it directly.


def generate_exponential(length: int, c: float) -> np.ndarray:
    """
    Create a decreasing exponential sequence a_i = r**i with r = 1 - c/length.
    The sequence is normalised so that its sum is exactly 1.
    """
    r = max(0.0, 1.0 - c / length)
    a = np.power(r, np.arange(length, dtype=float))
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


def candidate_from_core(core: np.ndarray, total_len: int) -> np.ndarray:
    """
    Given a core (already normalised to sum 1) and a desired total length,
    construct a tail of constant value epsilon such that the tail's
    self‑convolution does not exceed the core's maximal convolution.
    """
    n_core = core.shape[0]
    if total_len <= n_core:
        # Too short – just return the core (truncated or padded later)
        return core[:total_len]

    # maximal convolution of the core (max entry of core*core)
    max_core = float(np.convolve(core, core).max())

    # length of the tail
    tail_len = total_len - n_core
    # epsilon chosen so that tail_self = tail_len * eps**2 == max_core
    eps = np.sqrt(max_core / tail_len)
    # In the rare case eps > max_core we clamp it (does not hurt)
    eps = min(eps, max_core)

    tail = np.full(tail_len, eps, dtype=float)
    seq = np.concatenate([core, tail])
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a candidate sequence within the time budget.

    The algorithm repeatedly:
        * picks a core size m (100‑300) and an exponential decay parameter c,
        * builds the core, computes its maximal convolution,
        * chooses a total length N (800‑6000),
        * appends a tail of tiny constants that keep the overall maximal
          convolution unchanged,
        * evaluates the full sequence and remembers the best one.

    The best sequence found before the deadline is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float('inf')

    # a quick warm‑up: keep a uniform sequence as a fallback
    fallback_len = 5208
    fallback = np.full(fallback_len, 1.0, dtype=float)
    fallback /= fallback.sum()          # normalise to sum = 1 (any scale works)
    best_seq = fallback.tolist()
    best_score = evaluate_sequence(best_seq)

    while time.time() < deadline:
        # --- pick random parameters for the core ---------------------------------
        core_len = np.random.randint(120, 280)          # size of the “high” part
        c_param = np.random.uniform(0.7, 2.0)           # controls the decay rate

        core = generate_exponential(core_len, c_param)

        # --- total length of the whole step function -----------------------------
        total_len = np.random.randint(1500, 6000)       # overall number of entries

        # --- build tail and complete sequence ------------------------------------
        candidate = candidate_from_core(core, total_len)

        # Small random jitter on the tail helps to avoid accidental exact ties
        # (the jitter is tiny so it does not increase the maximal convolution).
        if candidate.size > core_len and np.random.rand() < 0.3:
            tail_jitter = np.random.randn(candidate.size - core_len) * 1e-6
            candidate[core_len:] = np.maximum(0.0, candidate[core_len:] + tail_jitter)

        # --- evaluate -------------------------------------------------------------
        try:
            score = evaluate_sequence(candidate.tolist())
        except Exception:
            # any unexpected error just discards this candidate
            continue

        if score < best_score:
            best_score = score
            best_seq = candidate.tolist()
            print(f"[{time.time() - start_time:5.1f}s] "
                  f"new best = {best_score:.6f} (core={core_len}, total={total_len}, c={c_param:.3f})")

    # Return the best sequence we have discovered.
    return best_seq
