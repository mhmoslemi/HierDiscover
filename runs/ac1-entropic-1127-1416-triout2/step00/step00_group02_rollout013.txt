def fast_evaluate(seq):
    a = np.asarray(seq, dtype=float)
    n = a.shape[0]
    # Compute convolution using FFT
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

"""
Propose candidate step‑function heights.

The evaluation target is

    2 * n * max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear convolution of the non‑negative sequence `a`.
For a uniform sequence the value is exactly 2, but the optimum can be
lower because the maximal convolution entry can be made smaller than the
triangular peak of the uniform case.

The algorithm below performs a fast simulated‑annealing / hill‑climbing
search directly on the sequence.  It repeatedly

* computes the convolution using an FFT (O(n log n)),
* finds the index `k_max` where the convolution attains its current
  maximum,
* builds a simple sub‑gradient `g_i = 2 * a_{k_max-i}` (zero when the
  index is out of range) which tells which coefficients contribute most
  to the peak,
* transfers a small amount of mass from a coefficient with the largest
  gradient to a random other coefficient,
* accepts the move if it improves the objective or with a small
  probability that decays with time (simulated annealing).

All moves keep the total sum of the sequence fixed, and the sequence
remains non‑negative.  The objective is evaluated with the same formula
as `evaluate_sequence` but using a faster FFT implementation.
The best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but using FFT)
def fast_evaluate(seq: np.ndarray) -> float:
    """Return the same value as evaluate_sequence(seq) but faster."""
    n = seq.shape[0]
    # length for linear convolution = 2*n-1, choose next power of two
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft * fft, size)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    # implementation of the algorithm from
    # "Efficient Projections onto the L1-Ball for Learning in High Dimensions"
    # (Duchi et al., 2008)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    init_temp = 0.02
    temp = init_temp
    init_step = 0.02          # max mass transferred in a single move (as a fraction of total sum)
    min_step = 1e-5
    # exponential decay rates
    temp_decay = 0.9995
    step_decay = 0.9999

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- evaluate current sequence -----
        conv = np.fft.rfft(seq, 1 << ((2 * n - 1).bit_length()))
        conv = np.fft.irfft(conv * conv, 1 << ((2 * n - 1).bit_length()))[: 2 * n - 1]
        max_val = np.max(conv)
        k_max = int(np.argmax(conv))

        # ----- build simple sub‑gradient for the dominant convolution entry -----
        # g_i = 2 * a_{k_max - i} when the partner index lies in [0,n)
        idx = k_max - np.arange(n)
        in_range = (idx >= 0) & (idx < n)
        grad = np.zeros_like(seq)
        grad[in_range] = 2.0 * seq[idx[in_range]]

        # pick a source index with large gradient (contributing most to the peak)
        src = int(np.argmax(grad))
        # random destination index, different from source
        dst = int(rng.integers(0, n - 1))
        while dst == src:
            dst = int(rng.integers(0, n - 1))

        # amount to move (bounded by current value at src and step size)
        cur_step = max(min_step, init_step * (step_decay ** iteration))
        delta = rng.random() * min(cur_step, seq[src])
        if delta <= 0.0:
            continue

        # propose the new sequence
        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # keep feasibility (non‑negative, sum = 1) by a cheap projection
        # only tiny numerical drift can occur, so a single projection is enough.
        new_seq = project_onto_simplex(new_seq)

        # evaluate the proposal
        new_score = fast_evaluate(new_seq)

        # ----- acceptance test (simulated annealing) -----
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # uphill move probability
            prob = math.exp(-(new_score - best_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            seq = new_seq
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                # optional progress output
                print(
                    f"Iter {iteration:7d} | best = {best_score:.6f} | time = {time.time() - start:.1f}s"
                )
        # anneal temperature and step size
        temp *= temp_decay
        # keep temperature from vanishing completely
        if temp < 1e-6:
            temp = 1e-6

        # occasional global random restart if we stagnate
        if iteration % 20000 == 0:
            # keep the best found so far, but try a new random start
            seq = np.maximum(0.0, rng.normal(loc=1.0 / n, scale=0.05 / n, size=n))
            seq = project_onto_simplex(seq)

    # final sanity check – ensure the returned sequence is a plain list
    return best_seq.tolist()
