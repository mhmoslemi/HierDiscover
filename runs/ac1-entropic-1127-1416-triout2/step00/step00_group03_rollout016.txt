def propose_candidate(seed=12345, budget_s=1000):
    import time, numpy as np, math, random
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # Determine length; use length of height_sequence_1 if available, else fallback to 5000
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000

    # Build initial sequence using sqrt-difference shape
    i = np.arange(n, dtype=float)
    base = np.sqrt(i+1) - np.sqrt(i)   # length n
    # Scale to achieve sum sqrt(2n)
    target_sum = math.sqrt(2.0 * n)
    cur_sum = base.sum()
    scale = target_sum / cur_sum
    a = base * scale

    best_seq = a.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    print(f"Initial value: {best_val:.6f}")

    # Parameters
    step0 = 0.1   # initial step size (relative to magnitude)
    decay = 0.9995  # decay per iteration
    step = step0

    while time.time() < deadline:
        # compute convolution and max index
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # gradient for max index
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # take a small step opposite gradient
        # step factor relative to a magnitude
        a_new = a - step * grad
        # enforce nonnegativity
        a_new = np.maximum(a_new, 0.0)

        # renormalize sum to target
        s = a_new.sum()
        if s == 0:
            # reinitialize randomly if degenerate
            a_new = base * scale
            s = a_new.sum()
        a_new = a_new * (target_sum / s)

        # evaluate
        val = evaluate_sequence(a_new.tolist())
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            print(f"New best {best_val:.6f} at time {time.time()- (deadline - budget_s):.1f}s")
        # simple acceptance: keep a if improves else keep old (could also accept worse with prob)
        if val <= best_val + 1e-12:
            a = a_new
        else:
            # occasional random move to escape
            if np.random.rand() < 0.01:
                # random perturbation: pick index and add small noise
                i_rand = np.random.randint(n)

        # also optionally random restart if stuck

"""
propose_candidate
-----------------

Search for a non‑negative step function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear convolution used in ``evaluate_sequence``.
The known uniform construction gives a value of ``2.0``.  
A continuous analogue shows that the sequence

    a_i = sqrt(i+1) – sqrt(i)

(up to a normalising factor) makes the self‑convolution almost constant
and yields a value close to π/2 ≈ 1.5708, which is already a large
improvement.  

The algorithm proceeds in two phases:

1. **Initialisation** – build the “square‑root‑difference’’ sequence,
   scale it so that its L¹‑norm equals √(2 n) (the normalisation used by
   ``evaluate_sequence``) and compute its initial score.

2. **Stochastic refinement** – run a simulated‑annealing / hill‑climbing
   loop for the remaining time budget.
   * Randomly choose two positions and transfer a small amount of mass
     between them (keeping the total sum fixed).
   * Occasionally perform a **gradient step** that moves the vector
     opposite to the sub‑gradient of the current peak of the convolution
     (the index where the maximum is attained).  This directly reduces
     the worst convolution coefficient.
   * Accept moves that improve the score, otherwise accept with a
     probability that decreases with the current temperature.
   * Keep the best sequence seen so far.

The whole search runs within the provided time budget (default 1000 s)
and returns the best sequence found.  No files are read or written,
only standard ``numpy``/``math``/``time`` functions are used.
"""

import math

def _initial_sqrt_diff_sequence(n: int) -> np.ndarray:
    """Return the normalised sqrt‑difference sequence of length ``n``."""
    # a_i = sqrt(i+1) - sqrt(i)
    i = np.arange(n, dtype=float)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    # Scale so that sum = sqrt(2*n) (the normalisation used by evaluate_sequence)
    target = math.sqrt(2.0 * n)
    a *= target / a.sum()
    return a


def _grad_step(a: np.ndarray, target_sum: float, step: float) -> np.ndarray:
    """
    One sub‑gradient step: move opposite to the gradient of the
    currently maximal convolution entry.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = int(conv.argmax())
    # gradient of b_k = sum_i a_i a_{k-i} w.r.t a_j is 2*a_{k-j} (if inside range)
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    # take a step opposite to the gradient
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    # re‑normalise the sum (the step changes it slightly)
    if a_new.sum() == 0.0:
        return a  # degenerate – keep old
    a_new *= target_sum / a_new.sum()
    return a_new


def _random_transfer(a: np.ndarray, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Propose a new sequence by moving a small amount of mass from one
    index to another (sum is unchanged).  The amount is a random
    fraction of the available mass at the source.
    """
    n = a.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # maximum we can move from i to j without creating a negative entry
    max_move = a[i]
    if max_move <= 0.0:
        return a
    # transfer up to 5 % of the source value (chosen uniformly)
    delta = (rng.random() - 0.5) * 0.1 * max_move
    # enforce non‑negativity at both ends
    if delta > a[j]:
        delta = a[j]
    if -delta > a[i]:
        delta = -a[i]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise (numerical drift)
    a_new *= target_sum / a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a sequence of non‑negative heights that minimises the
    ``evaluate_sequence`` objective.  The method combines a mathematically
    motivated initial construction with a simulated‑annealing hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # leave a little head‑room

    # ------------------------------------------------------------------
    # Choose a working length.  Use the length of the previously found
    # construction if available, otherwise fall back to a medium size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 1 – deterministic start (square‑root‑difference shape)
    # ------------------------------------------------------------------
    a = _initial_sqrt_diff_sequence(n)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    cur_score = best_score
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – stochastic optimisation (simulated annealing + occasional
    # gradient steps).  Temperature decays linearly from an initial
    # value to a tiny final value.
    # ------------------------------------------------------------------
    start_time = time.time()
    start_temp = best_score * 0.05   # 5 % of the initial value
    end_temp = best_score * 1e-4    # essentially zero at the end

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # linear temperature schedule
        elapsed = time.time() - start_time
        t_frac = min(1.0, elapsed / budget_s)
        temp = start_temp + (end_temp - start_temp) * t_frac

        # --------------------------------------------------------------
        # Random transfer move (keeps the sum exactly constant)
        # --------------------------------------------------------------
        a_candidate = _random_transfer(a, target_sum, rng)

        # evaluate the candidate
        cand_score = evaluate_sequence(a_candidate.tolist())

        # Metropolis acceptance test
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{elapsed:.1f}s] new best (transfer) = {best_score:.6f}")

        # --------------------------------------------------------------
        # Occasionally perform a deterministic gradient step that
        # attacks the current maximum of the self‑convolution.
        # --------------------------------------------------------------
        if iteration % 50 == 0:
            step_size = temp * (target_sum / n)   # scale with temperature and average height
            a_grad = _grad_step(a, target_sum, step_size)
            grad_score = evaluate_sequence(a_grad.tolist())
            if grad_score < cur_score or rng.random() < math.exp(-(grad_score - cur_score) / max(temp, 1e-12)):
                a = a_grad
                cur_score = grad_score
                if grad_score < best_score:
                    best_score = grad_score
                    best_seq = a_grad.copy()
                    print(f"[{elapsed:.1f}s] new best (gradient) = {best_score:.6f}")

        # --------------------------------------------------------------
        # Small global perturbation now and then to escape plateaus.
        # --------------------------------------------------------------
        if iteration % 200 == 0 and rng.random() < 0.1:
            perturb = rng.normal(scale=0.02 * (target_sum / n), size=n)
            a_pert = np.maximum(a + perturb, 0.0)
            a_pert *= target_sum / a_pert.sum()
            pert_score = evaluate_sequence(a_pert.tolist())
            if pert_score < cur_score or rng.random() < math.exp(-(pert_score - cur_score) / max(temp, 1e-12)):
                a = a_pert
                cur_score = pert_score
                if pert_score < best_score:
                    best_score = pert_score
                    best_seq = a_pert.copy()
                    print(f"[{elapsed:.1f}s] new best (perturb) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    total_time = time.time() - start_time
    print(f"Finished after {total_time:.1f}s – best score = {best_score:.6f}")
    # Return as a plain Python list of floats
    return best_seq.tolist()
