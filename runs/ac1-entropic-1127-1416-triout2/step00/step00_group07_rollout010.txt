def project_simplex(v):
    """Project onto probability simplex (nonnegative, sum=1)"""
    # Based on Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=..., budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Choose n
    n = np.random.randint(1000, 3000)   # maybe similar to existing.
    # initialize uniform
    p = np.full(n, 1.0/n)
    # Add small random noise
    p += np.random.randn(n) * 0.001
    p = np.maximum(p, 0)
    p = p / np.sum(p)
    best_sequence = p.copy()
    best_score = evaluate_sequence(best_sequence.tolist())
    # iterative optimization
    iteration = 0
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(p, p)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]
        # compute gradient of c_k (subgradient)
        g = np.zeros_like(p)
        # for i where j = k_max - i in range
        # vectorized: create mask for indices
        # compute j array
        indices = np.arange(n)
        j = k_max - indices
        # mask valid
        mask = (j >= 0) & (j < n)
        # set g[mask] = p[j[mask]]
        g[mask] = p[j[mask]]
        # diagonal term double
        if k_max % 2 == 0:
            i_diag = k_max // 2
            if 0 <= i_diag < n:
                g[i_diag] += p[i_diag]  # extra term
        # step
        # choose step size adaptively: step = 0.1 / (iteration+1)**0.5
        step = 0.1 / np.sqrt(iteration+1)
        # update p
        p_new = p - step * g
        # project onto simplex
        p_new = project_simplex(p_new)
        p = p_new
        # maybe occasionally random perturbation to escape local minima
        if iteration % 100 == 0:
            # random noise
            noise = np.random.randn(n) * 0.01 / np.sqrt(iteration+1+1)
            p = p + noise
            p = np.maximum(p, 0)
            p = p / np.sum(p)
        # evaluate
        seq = p.tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_sequence = seq.copy()
            print(f"Iter {iteration}: new best score {best_score:.6f}")
        iteration += 1
        # break if improvement minimal
        if iteration > 2000:
            # we can also break if time left low
            pass

    # maybe final local search refine using simulated annealing
    # random moves limited by time
    t0 = time.time()
    while time.time() < deadline:
        # propose random swap
        i, j = np.random.randint(n, size=2)
        delta = (np.random.rand() - 0.5) * 0.01
        p_candidate = best_sequence.copy()
        # adjust
        p_candidate[i] = max(0, p_candidate[i] + delta)
        # to preserve sum, adjust other index
        p_candidate[j] = max(0, p_candidate[j] - delta)
        # renormalize due to rounding errors
        total = sum(p_candidate)
        if total == 0:
            continue
        p_candidate = [x / total for x in p_candidate]
        score = evaluate_sequence(p_candidate)
        if score < best_score:
            best_score = score
            best_sequence = p_candidate
            print(f"Anneal iter: new best {best_score:.6f}")

    print(f"Final best score: {best_score}")

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Heuristic optimizer for the step-function bound.
    Uses projected (sub)gradient descent on the probability simplex
    combined with random restarts and simple local perturbations.
    """
    import numpy as np, time, random, math
    
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave some margin
    
    best_seq = None
    best_score = float('inf')
    # function to evaluate and maybe update best
    def maybe_update(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best: {best_score:.6f}")
    
    # Helper: project onto simplex (sum=1, nonnegative)
    def proj_simplex(v):
        # Duchi et al. 2008
        n = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(n) + 1
        # Find rho
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # all entries negative, return uniform zero?
            theta = 0
        else:
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w
    
    # subgradient update for max convolution
    def step_subgradient(p, step):
        n = p.shape[0]
        conv = np.convolve(p, p)
        k = np.argmax(conv)
        # gradient of selected entry
        # compute g_i = p_{k-i}  + extra diag term if needed
        i_vals = np.arange(n)
        j_vals = k - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        g = np.zeros_like(p)
        g[mask] = p[j_vals[mask]]
        if k % 2 == 0:
            i_mid = k // 2
            if 0 <= i_mid < n:
                g[i_mid] += p[i_mid]   # extra because derivative of p_i^2 is 2 p_i
        # Gradient descent step (we want to decrease the selected entry)
        p_new = p - step * g
        p_new = proj_simplex(p_new)
        return p_new, conv[k]
    
    # Main loop: keep restarting until time runs out
    restart = 0
    while time.time() < deadline:
        restart += 1
        # choose length n roughly around 2000
        n = np.random.randint(1500, 2500)
        # initialize probability vector via Dirichlet (uniform prior)
        alphas = np.ones(n)
        p = np.random.dirichlet(alphas)
        # initial evaluation
        cur_score = evaluate_sequence(p.tolist())
        maybe_update(p.tolist(), cur_score)
        # subgradient iterations
        # adaptive step schedule
        iteration = 0
        while time.time() < deadline:
            iteration += 1
            step = 0.5 / np.sqrt(iteration)  # decreasing step size
            p, cur_max = step_subgradient(p, step)
            # evaluate
            try:
                cur_score = evaluate_sequence(p.tolist())
                maybe_update(p.tolist(), cur_score)
            except Exception:
                pass
            # stop inner loop after some iterations
            if iteration >= 5000:
                break
        # after inner loop, maybe try random local tweaks for a short while
        # simple hill climbing
        for _ in range(200):
            if time.time() >= deadline:
                break
            # propose small random redistribution
            i, j = np.random.randint(n, size=2)
            delta = (np.random.rand() - 0.5) * 0.02  # small transfer
            q = p.copy()
            if q[i] + delta < 0:
                delta = -q[i]  # keep nonnegative
            q[i] += delta
            q[j] -= delta
            # reproject to simplex to fix negativity and sum
            q = proj_simplex(q)
            try:
                score = evaluate_sequence(q.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = q.tolist()
                    print(f"Local tweak improved to {best_score:.6f}")
                # accept if improved for next tweaks
                if score < cur_score:
                    p = q
                    cur_score = score
            except Exception:
                pass
    
    # Ensure we return a list of floats
    if best_seq is None:
        # fallback: uniform
        best_seq = [1.0] * 2145
    print(f"Finished search. Best score {best_score:.6f}")
    return best_seq

"""
Propose a (near) optimal step‐function for the given evaluation function.

Algorithm
---------

* Work with a probability vector `p` (non‑negative, sum = 1).  
  The evaluation of such a vector is  **2·n·max(p∗p)**  because the
  scaling of the heights is irrelevant (the evaluator divides by the
  square of the sum).

* Starting points are taken from a few random Dirichlet draws and,
  when available, from the global `height_sequence_1` (the current SOTA
  construction).  All vectors are normalised to a probability vector.

* For a vector `p` we compute the full convolution `c = np.convolve(p,p)`.
  Let `k = argmax(c)`.  The sub‑gradient of `c_k` w.r.t. `p` is
  `g_i = p_{k‑i}` (zero when the index is out of range) with an extra
  term `+p_i` for the diagonal element when `k` is even (the derivative
  of `p_i²` is `2·p_i`).  Moving `p` a little **against** this gradient
  reduces the presently maximal convolution entry.

* After each gradient step we project back onto the probability simplex
  by setting negative entries to zero and renormalising (this cheap
  projection works well for our objective).

* The step size follows a simple schedule `η = 0.5 / sqrt(iter)`.  After a
  few hundred gradient steps the improvement plateaus, so we restart the
  whole process with a fresh random start.  The outer loop runs until the
  time budget (≈ 1000 s) is exhausted.

* Occasionally a tiny random redistribution between two coordinates is
  tried (simple hill‑climbing) to escape shallow plateaus.

The best vector found is returned as a plain Python list; the evaluator
will later rescale it (the objective is scale‑invariant).  The routine
prints progress messages so you can see how the bound improves over time.

"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Simple non‑negative projection: clip negatives and renormalise."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _gradient_step(p: np.ndarray, step: float) -> tuple[np.ndarray, float]:
    """
    Perform one sub‑gradient descent step on the L∞ norm of the self‑convolution.
    Returns the new probability vector and the value of the maximal convolution entry.
    """
    n = p.size
    conv = np.convolve(p, p)          # length 2n‑1
    k = int(np.argmax(conv))          # index of the current worst (largest) entry
    # construct sub‑gradient of conv[k] w.r.t. p
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(p)
    g[mask] = p[j[mask]]
    # diagonal contribution when i == j (i.e. 2*i == k)
    if k % 2 == 0:
        i_mid = k // 2
        if 0 <= i_mid < n:
            g[i_mid] += p[i_mid]       # adds the missing second p_i factor
    # move *against* the gradient (we want to reduce the peak)
    p_new = p - step * g
    p_new = _project_to_simplex(p_new)
    return p_new, conv[k]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    best_seq = None
    best_score = float('inf')

    def _maybe_update(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New best score: {best_score:.6f}")

    restart = 0
    while time.time() < deadline:
        restart += 1
        # --------------------------------------------------------------
        # choose a length (similar to the current SOTA length)
        n = np.random.randint(1500, 2500)

        # initialise a probability vector
        if restart == 1 and 'height_sequence_1' in globals():
            # try the existing construction as first start‑point
            p0 = np.array(height_sequence_1, dtype=float)
            p0 = np.maximum(p0, 0.0)
            if p0.size != n:
                # trim or pad with small values to match the chosen length
                if p0.size > n:
                    p0 = p0[:n]
                else:
                    pad = np.full(n - p0.size, 1e-12)
                    p0 = np.concatenate([p0, pad])
            p = _project_to_simplex(p0)
        else:
            # random Dirichlet (uniform prior) – already a probability vector
            p = np.random.dirichlet(np.ones(n))

        # ----------------------------------------------------------------
        # inner gradient‑descent loop
        it = 0
        while time.time() < deadline:
            it += 1
            # step size decays like 1/√it
            η = 0.5 / np.sqrt(it)
            p, cur_max = _gradient_step(p, η)

            # the true evaluation for a probability vector (sum = 1):
            cur_score = 2.0 * n * cur_max

            _maybe_update(p.tolist(), cur_score)

            # a modest cap on inner iterations – after it we go to a new restart
            if it >= 3000:
                break

        # ----------------------------------------------------------------
        # short random‑swap hill‑climbing phase to polish the current best
        for _ in range(200):
            if time.time() >= deadline:
                break
            i, j = np.random.randint(n, size=2)
            delta = (np.random.rand() - 0.5) * 0.02  # small mass transfer
            q = p.copy()
            # keep non‑negativity before projection
            q[i] += delta
            q[j] -= delta
            q = _project_to_simplex(q)
            # evaluate
            conv = np.convolve(q, q)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = q.tolist()
                print(f"[{time.time():.1f}] Local tweak improved to {best_score:.6f}")
            # keep the improvement for the next tweak
            if score < cur_score:
                p = q
                cur_score = score

    # ----------------------------------------------------------------------
    # safety fallback (should never be needed)
    if best_seq is None:
        best_seq = [1.0] * 2145

    print(f"Search finished. Best score = {best_score:.6f}")
    return best_seq
