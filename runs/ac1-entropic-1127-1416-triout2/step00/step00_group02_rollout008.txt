def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    # Choose length L
    L = np.random.randint(200, 800)   # moderate sized
    # initial sequence random uniform
    seq = np.random.rand(L)
    seq = seq / seq.sum()   # normalized sum=1
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq)
    # keep track of iteration counts
    it = 0
    while time.time() < deadline:
        # pick move
        if np.random.rand() < 0.1:
            # LP direction step
            new_seq = get_good_direction_to_move_into(seq)
            if new_seq is None:
                continue
        else:
            # random transfer
            i, j = np.random.randint(L), np.random.randint(L)
            if i == j:
                continue
            eps = np.random.exponential(scale=0.01) # small positive
            # ensure i has enough mass
            if seq[i] < eps:
                eps = seq[i] * np.random.rand()
            seq2 = seq.copy()
            seq2[i] -= eps
            seq2[j] += eps
            # clip negative due to floating errors
            seq2 = np.maximum(seq2, 0)
            # renorm to sum=1
            total = seq2.sum()
            if total < 1e-6:
                continue
            seq2 /= total
            new_seq = seq2
        # evaluate
        score = evaluate_sequence(new_seq)
        # accept if better
        if score < best_score * (1 - 1e-9):  # strictly lower
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
        else:
            # accept with probability exp(-(score - best_score)/temp)
            # temperature schedule
            # compute temperature based on iteration progress
            progress = it / 10000  # arbitrary scaling
            temp = max(0.001, (1 - progress))  # linearly decreasing
            if np.random.rand() < np.exp(-(score - best_score) / temp):
                seq = new_seq
        it += 1
    return best_seq.tolist()

def solve_convolution_lp(f, M):
    n = len(f)
    c = -np.ones(n) # maximize sum
    # Build constraints A_ub x <= b_ub
    # For each k, sum_i f[i] * x[k-i] <= M
    # This is conv(f, x) <= M
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f[i]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])  # include x >= 0 constraints as -x <= 0
    b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

    temperature = 1.0

    while time.time() < deadline:
        # generate neighbor
        # choose two indices i,j and move epsilon
        i,j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # epsilon as proportion of seq[i]
        eps = np.random.uniform(0, 0.1) * seq[i]

        seq2[j] += eps
        # ensure non-negative
        if seq2[i] < 0:

        # renormalize
        total = seq2.sum()
        if total == 0:

        seq2 = seq2 / total

"""
Propose a candidate sequence for the convolution‐inequality challenge.

Algorithm summary
-----------------
1. Choose a moderate length (200–500) – large enough to allow a flat
   convolution but small enough for fast linear‑programming.
2. Initialise the sequence with a random non‑negative vector and normalise
   (the objective is scale invariant).
3. Repeatedly try two kinds of moves while time remains:
   * **LP move** – solve the linear programme used in the SOTA method:
     maximise the ℓ1‑norm of a new vector `g` under the constraint
     `conv(f,g) ≤ M` where `M = max(conv(f,f))`.  Afterwards perform a line
     search on the convex combination `(1‑t)·f + t·g` for a few values of `t`
     and keep the best one.
   * **Random perturbation** – pick two indices and move a small random
     amount of mass from one to the other, re‑normalise and accept the
     move if it improves the score (or with a small Metropolis probability
     to escape local minima).
4. Keep the best sequence seen (lowest `evaluate_sequence` value) and
   return it when the time budget is exhausted.

The mix of deterministic LP steps (which increase the total sum
without worsening the maximal convolution) and stochastic local
search quickly drives the ratio
`2·n·max(conv)/(sum²)` well below the trivial uniform value 2.0.
The routine is self‑contained, uses only NumPy and SciPy, respects the
1 CPU limit and prints occasional progress reports.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_seq, rhs):
    """
    Linear programme:
        maximise   sum(b)
        subject to conv(f_seq, b) ≤ rhs   (elementwise)
                   b ≥ 0
    Returns the optimal vector b (as a NumPy array) or None on failure.
    """
    n = len(f_seq)
    # objective – maximise sum => minimise -sum
    c = -np.ones(n)

    # build the convolution constraints: for each possible sum k,
    #   Σ_i f_i * b_{k‑i} ≤ rhs
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # b_j appears multiplied by f_i where i + j = k
        # => j = k - i
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])          # also enforce b ≥ 0 as –b ≤ 0
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

    # solve – use the fast HiGHS interior point implementation
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs",
                      options={"presolve": True, "time_limit": 5.0})
    except Exception:      # safety net – in some environments the keyword may differ
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if res.success:
        # numerical safeguard – clip tiny negatives that may appear
        return np.clip(res.x, 0.0, None)
    return None


def lp_step(current):
    """
    Perform one LP‑based improvement step.
    Returns a new sequence (numpy array) and the corresponding score,
    or (None, None) if the LP failed.
    """
    n = len(current)
    # current maximal convolution value
    M = np.max(np.convolve(current, current))
    g = solve_convolution_lp(current, M)
    if g is None:
        return None, None

    # line‑search on the convex combination (1‑t)·current + t·g
    best_seq = current
    best_score = evaluate_sequence(current.tolist())
    # try a handful of step sizes – they are cheap to evaluate
    for t in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        cand = (1.0 - t) * current + t * g
        cand = np.maximum(cand, 0.0)      # non‑negativity
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
    return best_seq, best_score


def random_move(seq, sigma=0.2):
    """
    Produce a neighbour by moving a random amount of mass between two positions.
    The amount is drawn from a log‑normal multiplicative factor.
    Returns the new sequence (numpy array) and its score.
    """
    n = len(seq)
    i, j = np.random.randint(n), np.random.randint(n)
    if i == j:
        return seq, evaluate_sequence(seq.tolist())

    # multiplicative perturbation of entry i
    factor = np.exp(np.random.normal(scale=sigma))
    new_seq = seq.copy()
    new_seq[i] *= factor
    # keep total mass unchanged – re‑normalise to sum = 1 (any scale works)
    total = new_seq.sum()
    if total <= 0.0:
        return seq, evaluate_sequence(seq.tolist())
    new_seq /= total
    # ensure non‑negativity (already satisfied)
    score = evaluate_sequence(new_seq.tolist())
    return new_seq, score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point required by the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) yields a low
    `evaluate_sequence` value.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # spare a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest problem size – large enough to admit a flat
    #    autocorrelation but small enough for fast LP solves.
    #    We pick length in the 300‑500 range (randomised for robustness).
    # ------------------------------------------------------------------
    length = np.random.randint(300, 500)

    # 2. Initialise with a random positive vector and normalise.
    #    (The objective is scale‑invariant, so any normalisation works.)
    current = np.random.rand(length)
    current /= current.sum()
    best_seq = current.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # optional: try the provided SOTA construction if it exists
    global_vars = globals()

        if len(candidate) > 0:

                cand = np.concatenate([candidate, pad])
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()

    # 3. Simulated‑annealing / LP hybrid loop.
    # ------------------------------------------------------------------
    iteration = 0
    temperature = 1.0
    while time.time() < deadline:
        iteration += 1

        # Occasionally (≈30 % of the time) try an LP move, otherwise a random tweak.
        if np.random.rand() < 0.3:

            if new_seq is None:

                new_seq, new_score = random_move(current, sigma=0.2)
        else:

        # Metropolis acceptance – this allows occasional uphill steps.
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # probability to accept a worse move (temperature slowly cools)
            prob = np.exp(-(new_score - best_score) / max(temperature, 1e-8))
            if np.random.rand() < prob:

            # record a new global best if we improved
            if new_score < best_score:
                best_score, best_seq = new_score, new_seq.copy()

        # Simple cooling schedule
        if iteration % 1000 == 0:
            temperature *= 0.9
            # occasional random restart to diversify the search space
            if temperature < 1e-3:
                current = np.random.rand(length)
                current /= current.sum()
                temperature = 1.0

        if iteration % 2000 == 0:
            print(f"Iter {iteration:6d} | best score = {best_score:.6f} | len = {length}")

    final = np.maximum(best_seq, 0.0).tolist()

        eps = 0.02 / len(final)
        final = [(x + eps) for x in final]

    print(f"Finished after {iteration} iterations – best score = {best_score:.6f}")
    return final
