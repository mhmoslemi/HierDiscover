    # optionally short local perturbation but keep improving

        i = random.randint(0, n-1)
        old = sequence[i]
        # propose new value (Gaussian perturb)
        new = max(0.0, old + np.random.normal(scale=0.2))
        sequence[i] = new
        # recompute score (expensive but small n)
        score = evaluate_sequence(sequence)
        if score < best_score:
            best_score = score
            best_seq = sequence.copy()

        else:

    print(f'Final best score {best_score}')
