```
def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    best_seq = None
    best_score = float('inf')
    iteration = 0

    while time.time() < deadline:
        # random length for this run
        n = np.random.randint(200, 800)   # moderate length
        # Initialize sequence: uniform
        seq = np.full(n, 1.0)
        # random perturbation for diversity: small Gaussian noise, then clip negatives, renormalize to sum=1
        seq += np.random.randn(n) * 0.1
        seq = np.clip(seq, 0, None)
        if np.sum(seq) < 1e-8:
            seq = np.full(n, 1.0)
        seq = seq / np.sum(seq)   # sum=1

        cur_score = evaluate_sequence(seq.tolist())
        # simple hill climb
        temp = 0.01 * cur_score
        t0 = time.time()
        while time.time() < deadline and time.time() - t0 < 0.5: # maybe 0.5 sec per inner run
            # generate neighbor
            i, j = np.random.choice(n, 2, replace=False)
            if seq[i] == 0:
                continue
            # transfer fraction of i's mass to j
            transfer = np.random.rand() * seq[i] * 0.2 # up to 20% of source
            new_seq = seq.copy()
            new_seq[i] -= transfer
            new_seq[j] += transfer
            # evaluate
            new_score = evaluate_sequence(new_seq.tolist())
            # acceptance
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / temp):
                seq = new_seq
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.tolist()
                    # optionally print progress
                    print(f'New best {best_score:.6f} at length {n} iter {iteration}')
            # cooling
            temp *= 0.9995   # slowly reduce temperature
        iteration += 1

    # After time, return best_seq
    return best_seq
```

```
def simplex_projection(v):
    """Project v onto probability simplex (nonnegative sum 1)."""
    # if sum(v) == 1 and all(v >= 0): return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def project_simplex(v, z=1):
    # Projects onto set {x >= 0, sum x = z}
    # Implementation per Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
def project_to_simplex(v, z):
    # Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1   # zero-index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Stochastic hill‑climbing search for a low‑value step function.

The evaluation metric is
    2 * n * max(conv(f,f)) / (sum(f)^2)
where conv(f,f) is the full discrete convolution of the non‑negative
vector f with itself.  The algorithm repeatedly:

* picks a random length n (200–800 is a good trade‑off between quality
  and speed);
* creates a random start vector with the required L¹‑norm
  (sum = sqrt(2*n), the same normalisation used in the reference code);
* runs a short simulated‑annealing / hill‑climbing loop that repeatedly
  transfers a random fraction of mass from one index to another.
  A move is accepted if it reduces the objective, otherwise it may be
  accepted with a Metropolis probability that slowly cools down.
* keeps the best sequence seen over all restarts.

The objective is calculated directly (no call to the costly
`evaluate_sequence`) because it is exactly the same formula; the final
sequence is returned as a plain list so the external evaluator will
apply the same security checks.
"""

import numpy as np
import time

def _objective(seq):
    """Return the evaluation metric for a given sequence (no safety checks)."""
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.size
    s = seq.sum()
    if s < 0.01:                 # matches the guard inside evaluate_sequence
        return float('inf')
    conv = np.convolve(seq, seq)
    m = conv.max()
    return 2.0 * n * m / (s * s)


def _random_transfer(seq, max_frac=0.3):
    """Transfer a random fraction of mass from one index to another."""
    n = seq.size
    i = np.random.randint(n)
    if seq[i] <= 0.0:
        return seq                     # nothing to move
    j = np.random.randint(n)
    if i == j:
        return seq
    # amount is a random fraction of the source entry
    frac = np.random.rand() * max_frac
    delta = seq[i] * frac
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return new_seq


def _local_search(start_seq, time_limit):
    """
    Short hill‑climbing run starting from `start_seq`.
    Returns the best (seq, value) found within `time_limit` seconds.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    # temperature for the Metropolis acceptance; start relatively high
    temp = max(cur_val * 0.1, 1e-6)

    t0 = time.time()
    while time.time() - t0 < time_limit:
        cand = _random_transfer(cur, max_frac=0.3)
        cand_val = _objective(cand)

        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best_seq, best_val = cand.copy(), cand_val
        else:
            # occasional uphill moves (simulated annealing)
            prob = np.exp(-(cand_val - cur_val) / temp) if temp > 0 else 0.0
            if np.random.rand() < prob:
                cur, cur_val = cand, cand_val

        # geometric cooling
        temp *= 0.9995
    return best_seq, best_val


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation value.
    The function runs for roughly `budget_s` seconds and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # safety margin

    # keep the overall best sequence
    best_seq = None
    best_val = float('inf')
    iteration = 0

    # a quick baseline (uniform vector) – useful if the search stalls
    n0 = 500
    target_sum = np.sqrt(2.0 * n0)
    uniform = np.full(n0, target_sum / n0)
    best_seq = uniform.copy()
    best_val = _objective(best_seq)
    print(f'Initial uniform baseline: value = {best_val:.6f}')

    while time.time() < deadline:
        # random length for this restart (200‑800 is fast enough)
        n = np.random.randint(200, 801)
        target_sum = np.sqrt(2.0 * n)

        # three kinds of random starts: Dirichlet, decreasing, increasing
        mode = np.random.choice([0, 1, 2])
        if mode == 0:                          # Dirichlet (purely random)
            vec = np.random.rand(n)
            vec = vec / vec.sum() * target_sum
        elif mode == 1:                        # decreasing (1/(i+1))
            vec = 1.0 / (np.arange(n, dtype=np.float64) + 1.0)
            vec = vec / vec.sum() * target_sum
        else:                                  # increasing (linear ramp)
            vec = np.arange(1, n + 1, dtype=np.float64)
            vec = vec / vec.sum() * target_sum

        # allocate a short time slice for the local optimisation
        inner_time = min(0.6, deadline - time.time())
        if inner_time <= 0:
            break

        cand_seq, cand_val = _local_search(vec, inner_time)

        # keep the best overall
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f'Iter {iteration:04d}: new best value = {best_val:.6f} (len={n})')
        iteration += 1

    # final safety: ensure the returned list is plain Python floats
    return [float(x) for x in best_seq]
