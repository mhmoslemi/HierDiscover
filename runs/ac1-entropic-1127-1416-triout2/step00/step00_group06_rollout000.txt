"""
Propose_candidate uses a simple projected gradient descent on the
max‐auto‑correlation (max of the self‑convolution).  Starting from a
random Dirichlet distribution we repeatedly:
  * compute the self‑convolution,
  * locate the indices attaining the maximum,
  * build the sub‑gradient of the L∞‑norm,
  * move the current vector opposite to that gradient,
  * project back onto the probability simplex.
Occasional random perturbations and short restarts avoid getting stuck.
The best sequence discovered within the time budget is returned.
"""
import time
import numpy as np

def _subgradient_max_conv(a, active):
    """Return a sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = a.shape[0]
    g = np.zeros(n)
    # For each active index k, add a_{k-i} to g[i].
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # vectorized slice:
        i_range = np.arange(i_min, i_max + 1)
        j_range = k - i_range
        g[i_range] += a[j_range]
    return g

def _project_to_simplex(v):
    """Project vector v onto the probability simplex (non‑negative, sum=1)."""
    # Standard algorithm (sorting)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero -> return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    Search for a step‑function (non‑negative list) that minimises
    evaluate_sequence.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # keep a safety margin
    best_seq = None
    best_score = float("inf")
    iteration = 0

    # Target score for early stop (optional)
    target_score = 1.503

    while time.time() < deadline:
        # ---- random restart parameters ----
        n = np.random.randint(200, 801)          # length of the sequence
        # start from a Dirichlet distribution (sum = 1)
        a = np.random.dirichlet(np.ones(n))
        # tiny jitter to break symmetry
        a += 1e-6 * np.random.rand(n)
        a /= a.sum()

        # evaluate initial
        try:
            cur_score = evaluate_sequence(a.tolist())
        except Exception:
            cur_score = float("inf")

        if np.isinf(cur_score):
            continue

        # inner optimisation loop
        lr = 0.1                               # initial learning rate
        inner_deadline = time.time() + min(5.0, deadline - time.time())
        no_improve_cnt = 0

        while time.time() < inner_deadline:
            # compute convolution and max
            b = np.convolve(a, a)
            max_b = b.max()
            active = np.where(np.abs(b - max_b) < 1e-12)[0]

"""
Propose_candidate uses a simple projected sub‑gradient method to reduce the
maximum entry of the self‑convolution (the quantity that appears in the
evaluation function).  Starting from a random Dirichlet vector we repeatedly

1. compute the convolution a*a,
2. locate the index/indices where the convolution attains its maximum,
3. form a sub‑gradient of max(conv(a,a)) with respect to a,
4. move a a little opposite to that gradient,
5. project the result back onto the probability simplex (non‑negative,
   sum = 1).

Small random perturbations and short restarts avoid being trapped in
local minima.  The best sequence found before the time budget expires is
returned.  The algorithm works for any length (the code chooses a random
length between 200 and 800 for speed/quality trade‑off).
"""

import time

def _subgradient_max_conv(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    For every active index k we add a_{k‑i} to the i‑th component.
    """
    n = a.shape[0]
    g = np.zeros(n)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1)
        j_range = k - i_range
        g[i_range] += a[j_range]
    return g


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i >= 0 , sum_i x_i = 1}.
    Standard O(n log n) algorithm (see Duchi et al., ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Return uniform vector if everything would be negative.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
