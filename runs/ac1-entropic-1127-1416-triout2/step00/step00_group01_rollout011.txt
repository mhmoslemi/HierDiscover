def initialize_sequence(n, method='uniform'):
    if method == 'uniform':
        a = np.full(n, 1.0 / n)
    elif method == 'random':
        a = np.random.rand(n)
        a = a / a.sum()
    elif method == 'valley':
        i = np.arange(n)
        eps = 1e-12
        a = 1.0 / ((i + 1) * (n - i) + eps)
        a = a / a.sum()
    elif method == 'hill':
        i = np.arange(n)
        a = (i + 1) * (n - i)  # peaked at centre
        a = a / a.sum()
    elif method == 'mixed':
        # combine uniform and valley
        u = np.full(n, 1.0 / n)
        v = 1.0 / ((np.arange(n) + 1) * (n - np.arange(n)) + 1e-12)
        v = v / v.sum()
        w = np.random.rand()
        a = (1 - w) * u + w * v
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a.astype(float)
```

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
evaluation metric:

    2 * n * max(conv(a, a)) / (sum(a)^2)

The algorithm combines three ideas:

1. **Random initialisation** – many restarts with different shapes
   (uniform, random, “edge‑heavy” etc.) break symmetry of the uniform solution.

2. **Smoothed gradient descent** – minimise a smooth approximation of the
   maximum convolution using a log‑sum‑exp (soft‑max) and a simple projected
   gradient step.  This quickly reduces the peak of the convolution.

3. **LP‑based mixing** – for the current sequence `f` we solve a linear program
   that finds a direction `g` with maximal total mass while keeping the
   convolution `f * g` below the current worst convolution value `M`.  A line
   search over several mixing coefficients `t` then updates the sequence.
   This step is able to escape the stationary point of the gradient method
   (the uniform distribution) and drives the sequence towards a flatter
   autocorrelation.

The routine runs for the allotted time (default 1000 s) and keeps the best
sequence found.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def initialize_sequence(n: int, method: str = "uniform") -> np.ndarray:
    """Return a non‑negative sequence of length *n* with unit sum."""
    if method == "uniform":
        a = np.full(n, 1.0 / n)
    elif method == "random":
        a = np.random.rand(n)
        a = a / a.sum()
    elif method == "valley":
        # high at the ends, low in the centre:  a_i ∝ 1/((i+1)*(n-i))
        i = np.arange(n)
        eps = 1e-12
        a = 1.0 / ((i + 1) * (n - i) + eps)
        a = a / a.sum()
    elif method == "hill":
        # opposite of valley – peak in the centre
        i = np.arange(n)
        a = (i + 1) * (n - i)
        a = a / a.sum()
    elif method == "mixed":
        # blend uniform and valley
        u = np.full(n, 1.0 / n)
        i = np.arange(n)
        eps = 1e-12
        v = 1.0 / ((i + 1) * (n - i) + eps)
        v = v / v.sum()
        w = np.random.rand()
        a = (1 - w) * u + w * v
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)

    return a.astype(float)


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve the LP:

        maximise   sum(b)
        subject to  (f_seq * b)[k] <= rhs   for all k = 0 … 2n‑2
                    b >= 0

    Returns the optimal vector `b` (as a NumPy array) or ``None`` on failure.
    """
    n = len(f_seq)

    # Build the constraint matrix A_ub (size (2n‑1) × n)
    # Row k contains f_seq[i] for every pair i+j = k (i.e. j = k-i)
    A = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(2 * n - 1):
        # i runs over all indices such that 0 ≤ i < n and 0 ≤ k‑i < n
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # vectorised fill of the row
        idx = np.arange(i_min, i_max + 1, dtype=int)
        j = k - idx
        A[k, j] = f_seq[idx]

    b = np.full(2 * n - 1, rhs, dtype=float)

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # Bounds: b_j >= 0
    bounds = [(0, None)] * n

    try:
        res = optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 10.0},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


def gradient_step(a: np.ndarray, lr: float, alpha: float) -> np.ndarray:
    """
    One projected gradient descent step on the smooth max of the convolution.
    *lr* – learning rate,
    *alpha* – soft‑max sharpness.
    Returns the new sequence (still non‑negative and normalized to sum = 1).
    """
    # Convolution a*a
    conv = np.convolve(a, a)

    # Soft‑max weights approximating the max
    shift = conv - np.max(conv)               # for numerical stability
    w = np.exp(alpha * shift)
    w /= w.sum()

    # Gradient of  (1/α) log Σ exp(α conv)  w.r.t a
    #   ∂/∂a_i  = 2 * Σ_k w_k * a_{k-i}
    # This is exactly a cross‑correlation of w with a.
    grad = 2.0 * np.correlate(w, a, mode="valid")

    # Gradient step
    a_new = a - lr * grad

    # Projection onto the simplex (non‑negative + sum = 1)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a_new, 1.0 / len(a_new))
    else:
        a_new = a_new / s
    return a_new


def lp_mix_step(a: np.ndarray, t_candidates=None):
    """
    Perform one LP‑based mixing step.
    Returns the best new sequence (or the original one if no improvement).
    """
    if t_candidates is None:
        t_candidates = [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]

    n = len(a)
    sum_a = a.sum()
    # Normalise exactly as in the reference algorithm
    f_norm = a * np.sqrt(2.0 * n) / sum_a
    M = np.max(np.convolve(f_norm, f_norm))

    g_raw = solve_convolution_lp(f_norm, M)
    if g_raw is None:
        return a, None, None  # LP failed

    # Normalise g to have the same total mass as a (i.e. sum = 1)
    g = g_raw / g_raw.sum()

    # Try a few mixing coefficients and keep the best
    best_seq = a
    best_score = evaluate_sequence(a.tolist())
    best_t = None
    for t in t_candidates:
        cand = (1.0 - t) * a + t * g
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
            best_t = t
    return best_seq, best_score, best_t

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the given
    evaluation function.  The function runs for *budget_s* seconds (minus a
    small safety margin) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a few seconds buffer

    # Keep the globally known construction as a fallback
    try:
        base_seq = np.array(height_sequence_1, dtype=float)
        base_seq = np.maximum(base_seq, 0.0)
        base_seq /= base_seq.sum() if base_seq.sum() > 0 else 1.0
    except Exception:  # pragma: no cover
        base_seq = None

    best_seq = None
    best_score = float("inf")

    # Different lengths to try – not too large for the LP part
    length_options = [800, 1200, 1600, 2000, 2500, 3000]

    iteration = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Choose a length and an initialisation method
        # --------------------------------------------------------------
        n = int(np.random.choice(length_options))
        init_method = np.random.choice(
            ["uniform", "random", "valley", "hill", "mixed"]
        )
        a = initialize_sequence(n, method=init_method)

        # Occasionally start from the known sequence (trim / pad as needed)
        if base_seq is not None and np.random.rand() < 0.2:
            if len(base_seq) >= n:
                a = base_seq[:n].copy()
            else:
                # pad with small random noise
                pad = np.random.rand(n - len(base_seq)) * 1e-3
                a = np.concatenate([base_seq, pad])
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # --------------------------------------------------------------
        # Phase 1 – smoothed gradient descent (fast, many cheap iterations)
        # --------------------------------------------------------------
        lr = 0.2
        alpha = 25.0
        for _ in range(40):
            if time.time() >= deadline:
                break
            a = gradient_step(a, lr=lr, alpha=alpha)
            lr *= 0.97  # slowly decay learning rate
            # occasional tiny random jitter to avoid being trapped
            if np.random.rand() < 0.03:
                idx = np.random.randint(n)
                a[idx] = max(a[idx] + np.random.randn() * 0.005, 0.0)
                a /= a.sum()

        # --------------------------------------------------------------
        # Phase 2 – LP‑based mixing (more aggressive but slower)
        # --------------------------------------------------------------
        for _ in range(15):
            if time.time() >= deadline:
                break
            new_a, new_score, _ = lp_mix_step(a)
            if new_score is not None and new_score < evaluate_sequence(a.tolist()):
                a = new_a
                # keep the improvement
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.tolist()
                    print(
                        f"[{time.time():.1f}] New best score {best_score:.6f} "
                        f"len={n} method={init_method} iter={iteration}"
                    )
            else:
                # no LP improvement – fall back to a tiny random perturbation
                idx = np.random.randint(n)
                a[idx] = max(a[idx] + np.random.randn() * 0.01, 0.0)
                a /= a.sum()

        # --------------------------------------------------------------
        # Final evaluation of the current candidate
        # --------------------------------------------------------------
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(
                f"[{time.time():.1f}] Improved (final) score {best_score:.6f} "
                f"len={n} method={init_method} iter={iteration}"
            )

        # Stop early if we already beat the target (1.503)
        if best_score <= 1.503:

        iteration += 1

    # If nothing was found (unlikely) fall back to a simple uniform sequence
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
    return best_seq
