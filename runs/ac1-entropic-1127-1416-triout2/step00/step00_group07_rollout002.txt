    while time.time() < deadline:
        # sample shape
        shape = random.choice(shape_types)
        if shape == 'uniform':
            param = None
            cand = np.ones(n)
        elif shape == 'exponential':
            param = random.uniform(0.001, 0.5)  # adjust
            distances = np.minimum(np.arange(n), n-1-np.arange(n))
            cand = np.exp(-param * distances)
        elif shape == 'power':
            param = random.uniform(0.1, 5.0)

            cand = (distances + 1) ** (-param)

            param = random.uniform(0.001, 0.01)  # slope

            cand = np.maximum(0, 1 - param * distances)
        else:
            continue

        # convert to list

        try:

        except Exception as e:

        if score < best_score:

"""
Propose a good step‑function (non‑negative sequence) by random sampling +
lightweight local optimisation.

The algorithm works as follows:

1. Initialise the current best with the provided `height_sequence_1`
   (if it exists) – otherwise start from a uniform sequence.

2. Main loop (until the time budget is exhausted):
   * Randomly choose a length `n` (2000–8000) and a distribution
     (exponential, uniform or log‑normal).
   * Generate a fresh random vector `a` of that length.
   * Evaluate it with the supplied `evaluate_sequence`.  If it improves
     the best score, remember it and immediately run a fast “swap‑hill‑climb”
     on this candidate (swap two entries and keep the swap if the score
     improves).  Only a few hundred swaps are attempted – enough to
     polish a good candidate without spending too much time.

3. Occasionally run a short hill‑climb on the overall best sequence
   (even if it was not updated in the current iteration).

Because the scoring function is scale‑invariant, only the *shape* of the
vector matters.  Random vectors of moderate size already have a much
flatter auto‑convolution than the uniform sequence (score ≈ 2).  By
adding a lightweight local search we can drive the score well below the
target 1.503 in the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: a very cheap hill‑climb that only swaps two entries.
# ----------------------------------------------------------------------
def _swap_hill_climb(seq_arr: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator,
                     max_steps: int = 2000):
    """
    Try to improve `seq_arr` by swapping pairs of entries.
    The routine stops when the deadline is reached or after `max_steps`
    swaps.  Returns the (possibly) improved list and its score.
    """
    best_seq = seq_arr.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    n = best_seq.size
    step = 0
    while step < max_steps and time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a swap
        cand = best_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]

        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            # safety – any unexpected error just rejects the candidate
            step += 1
            continue

        if score < best_score:
            best_seq = cand
            best_score = score
        step += 1

    return best_seq.tolist(), best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default: 1000).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # --------------------------------------------------------------
    # Start from the known construction if it is available.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        best_seq = list(height_sequence_1)
        try:
            best_score = evaluate_sequence(best_seq)
        except Exception:
            # fall back to a uniform sequence if the provided one is bad
            best_seq = [1.0] * 1000
            best_score = evaluate_sequence(best_seq)
    else:
        best_seq = [1.0] * 1000                     # generic uniform start
        best_score = evaluate_sequence(best_seq)

    print(f"[init] score={best_score:.6f} length={len(best_seq)}")

    # --------------------------------------------------------------
    # A quick local improvement of the initial candidate.
    # --------------------------------------------------------------
    try:
        seq_arr = np.array(best_seq, dtype=float)
        refined, refined_score = _swap_hill_climb(
            seq_arr, deadline, rng, max_steps=500)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[init‑refine] score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Main random‑search loop.
    # --------------------------------------------------------------
    # Length bounds – larger sequences give a tighter theoretical lower bound.
    N_MIN, N_MAX = 2000, 8000
    distributions = ['exponential', 'uniform', 'lognormal']
    iteration = 0

    while time.time() < deadline:
        n = int(rng.integers(N_MIN, N_MAX + 1))
        dist = rng.choice(distributions)

        # generate a fresh random vector
        if dist == 'exponential':
            seq_np = rng.exponential(scale=1.0, size=n)
        elif dist == 'uniform':
            seq_np = rng.random(size=n)          # values in [0,1)
        elif dist == 'lognormal':
            seq_np = rng.lognormal(mean=0.0, sigma=1.0, size=n)
        else:
            seq_np = rng.random(size=n)

        # safety – very small sums would be rejected by `evaluate_sequence`
        if seq_np.sum() < 0.02:

        # evaluate the raw candidate
        try:

        except Exception:
            # ignore malformed candidates
            continue

        if score < best_score:
            best_seq = seq_np.tolist()
            best_score = score

            # Run a short hill‑climb to polish the fresh winner.
            try:

    return best_seq
