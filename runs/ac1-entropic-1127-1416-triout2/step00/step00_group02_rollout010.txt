def perturb(seq, step=0.01):
    n = len(seq)
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j: return seq
    # transfer amount delta from i to j
    delta = step * seq[i]  # up to step of source's mass
    # ensure not negative
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return new_seq

"""
Propose a candidate step‑function sequence.

The core idea is a simple projected gradient descent that tries to
*flatten* the auto‑convolution of the sequence.
For a probability vector a (∑a_i = 1) the convolution b = a*a has
average 1/(2n‑1).  By minimizing the L2 distance ‖b‑average‖² we push
all b_k towards the same value, which reduces the maximal entry of b
and therefore the evaluation score
    2·n·max(b)/(∑a)²  = 2·n·max(b)   (since the sum is kept =1).

The gradient of ‖b‑c‖² w.r.t. a is
    g = 4·conv(b‑c, rev(a)) .
A small step a ← a – η·g is taken, negative entries are clipped,
and the vector is projected back onto the probability simplex by simple
renormalisation.  The step size η decays slowly; occasional random
jitter is added if progress stalls.

The method works for any length (the existing `height_sequence_1` is
used if available, otherwise a fresh random start is created) and
fits comfortably inside the 1000‑second budget.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: simple projection onto the non‑negative simplex
# (renormalise after clipping – sufficient for our use case)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum 1."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ------------------------------------------------------------
# One gradient‑flattening step
def _flatten_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform a single projected gradient step that flattens the
    self‑convolution of `a`.

    Args:
        a   – current probability vector (sum = 1, a_i ≥ 0)
        eta – step size (positive scalar)

    Returns:
        Updated probability vector (still sum = 1, non‑negative).
    """
    n = a.size
    # full linear convolution a*a
    conv = np.convolve(a, a)                     # length 2n‑1
    avg = conv.mean()                            # = 1/(2n‑1) when sum(a)=1
    diff = conv - avg

    # grad = 4 * conv(diff, rev(a))   (only the central n entries are needed)
    grad_full = np.convolve(diff, a[::-1])       # length 3n‑2
    grad = 4.0 * grad_full[n - 1:n - 1 + n]      # shape (n,)

    # gradient descent step + projection onto simplex
    a_new = a - eta * grad
    a_new = _project_simplex(a_new)
    return a_new

        A list of non‑negative floats (the best sequence found).
    """

    # Initialise a probability vector (sum = 1, non‑negative).
    # A Dirichlet(1) sample gives a neutral random start.
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    a = a / a.sum()

    # Keep the best solution seen so far.
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}  score={best_score:.6f}")

    # Gradient‑descent parameters.
    eta = 0.5                # initial step size
    iter_no_improve = 0      # counter for stagnation detection
    iteration = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # One flattening step.
        a = _flatten_step(a, eta)

        # Decay step size slowly – helps convergence.
        eta = max(eta * 0.9995, 0.01)

            iter_no_improve = 0

            a = _project_simplex(a + jitter)
            eta = 0.5                     # reset step size after a shake‑up
            iter_no_improve = 0
