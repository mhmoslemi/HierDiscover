import numpy as np
import time
import math

def softmax_approx(b, beta):
    maxb = np.max(b)
    exps = np.exp(beta * (b - maxb))
    return exps / np.sum(exps)

def gradient_step(a, beta, eta):
    # compute convolution a*a
    b = np.convolve(a, a)
    # compute softmax weights
    p = softmax_approx(b, beta)
    # compute conv of p with reversed a
    conv = np.convolve(p, a[::-1])
    n = len(a)
    grad = 2 * beta * conv[n-1:n-1+n] - 2.0
    # exponentiated update
    exp_arg = -eta * grad
    # clip for numeric stability
    exp_arg = np.clip(exp_arg, -50, 50)
    a_new = a * np.exp(exp_arg)
    # renormalize to sum 1
    s = a_new.sum()
    if s == 0:
        a_new = np.full_like(a, 1.0/len(a))
    else:
        a_new = a_new / s
    return a_new

def run_descent(n, time_limit, seed):

"""
Gradient‑descent on a smooth approximation of the max‑convolution.

The evaluation score is 2*n*max(conv(a,a))/sum(a)^2.  Because the score
is scale‑invariant we keep the vector on the probability simplex
(sum(a)=1, a_i≥0).  For a fixed β we approximate max(conv) by the
soft‑max (log‑sum‑exp) of the convolution values.  The gradient of the
smooth objective can be written in closed form and a multiplicative
(exponentiated‑gradient) update keeps the iterates non‑negative.
β is increased slowly so that the optimisation first sees a smooth
objective and later focuses on the true maximum.

The routine runs several independent runs with different lengths
(2000, 4000, 6000, 8000) and keeps the best sequence found before the
time budget expires.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------

def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution via FFT (real FFT for speed).  Result length is len(x)+len(y)-1.
    """
    size = _next_pow_two(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[: len(x) + len(y) - 1]
    return conv


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Smooth approximation of the max: soft‑max probabilities."""
    maxb = np.max(b)
    # shifting by maxb prevents overflow
    w = np.exp(beta * (b - maxb))
    return w / np.sum(w)


def _exponential_gradient_step(a: np.ndarray, beta: float, eta: float) -> np.ndarray:
    """
    One mirror‑descent (exponentiated‑gradient) step on the simplex.
    a  – current probability vector (sum=1, a≥0)
    beta – temperature for the soft‑max
    eta – step size (chosen as 0.5/beta in the main loop)
    """
    # ----- 1. convolution a*a -----
    conv_aa = _conv_fft(a, a)                     # length 2n‑1

    # ----- 2. soft‑max probabilities over the convolution entries -----
    p = _softmax_weights(conv_aa, beta)          # same length 2n‑1

    # ----- 3. gradient of the smooth objective -----
    #   grad_i = 2*beta*∑_k p_k * a_{k-i} - 2   (indices out of range give 0)
    #   The sum is a convolution of p with the reversed a.
    conv_pa_rev = _conv_fft(p, a[::-1])          # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * beta * conv_pa_rev[n - 1 : n - 1 + n] - 2.0

    # ----- 4. exponentiated‑gradient update (always stays non‑negative) -----
    # clip the exponent to avoid overflow/underflow
    exp_arg = np.clip(-eta * grad, -50.0, 50.0)
    a_new = a * np.exp(exp_arg)

    # ----- 5. re‑normalise to stay on the simplex -----
    total = a_new.sum()
    if total == 0.0:                     # safety net – should be very rare
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= total
    return a_new


def _run_descent(
    n: int,
    time_limit: float,
    rng_seed: int,
) -> tuple[list[float], float]:
    """
    Perform the mirror‑descent optimisation for a fixed length `n`
    until `time_limit` seconds have elapsed.  Returns the best sequence
    (as a plain list) and its exact evaluation score.
    """
    rng = np.random.default_rng(rng_seed)

    # start from the uniform distribution, add a tiny random jitter
    a = np.full(n, 1.0 / n)
    a += rng.random(n) * 1e-6
    a /= a.sum()

    start = time.time()
    beta = 5.0                     # low temperature → smooth objective
    while True:
        now = time.time()
        if now - start > time_limit:
            break

        # step size is tied to the current temperature
        eta = 0.5 / beta

        a = _exponential_gradient_step(a, beta, eta)

        # slowly raise the temperature (focus on the true maximum)
        beta = min(200.0, beta * 1.001 + 0.01)

        # occasional tiny random perturbation – helps to escape flat regions
        if rng.random() < 0.02:
            jitter = rng.normal(scale=1e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

    # Exact score (the expensive call) – performed only once per run
    score = evaluate_sequence(a.tolist())
    return a.tolist(), float(score)
