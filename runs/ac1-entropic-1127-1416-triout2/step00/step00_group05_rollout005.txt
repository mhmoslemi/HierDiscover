def propose_candidate(seed=0, budget_s=970):
    import numpy as np, time

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    best_seq = [1.0]  # fallback
    best_score = np.inf

    # fallback: if height_sequence_1 exists, use as baseline
    try:
        from __main__ import height_sequence_1
        base = list(height_sequence_1)
        score = evaluate_sequence(base)
        if score < best_score:
            best_score = score
            best_seq = base
    except Exception:
        pass

    ns = [200, 400, 600, 800, 1000, 1500, 2000, 3000, 4000, 6000, 8000]
    cs = np.linspace(0.2, 3.0, 28)  # step ~0.1
    # Additional special set: c=1/n*? maybe use many values near 0.5-1.5.

    for n in ns:
        if time.time() > deadline:
            break
        # precompute indices
        idx = np.arange(n)
        for c in cs:
            if time.time() > deadline:
                break
            delta = c / n
            r = 1.0 - delta
            # ensure r>0
            if r <= 0:
                continue
            # compute sequence
            # Use exponential to avoid underflow issues
            seq_np = np.exp(idx * np.log(r))
            # Evaluate
            try:
                score = evaluate_sequence(seq_np.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = seq_np.tolist()
                print(f"New best: n={n}, c={c:.3f}, r={r:.6f}, score={score:.6f}")

    # Optionally run local refinement: small random perturbations around best_seq
    # Use hill climbing for some steps
    curr_seq = np.array(best_seq)
    n = len(curr_seq)
    while time.time() < deadline:
        # random index modifications
        idx_change = np.random.randint(n)
        epsilon = np.random.randn() * 0.05 * curr_seq[idx_change]  # relative change
        new_seq = curr_seq.copy()
        new_seq[idx_change] = max(0.0, new_seq[idx_change] + epsilon)
        # optionally re-normalize sum to keep same scale (but not needed)
        try:
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_seq.tolist()
                curr_seq = new_seq
                print(f"Refine: new best score {score:.6f}")
            else:
                # maybe accept with small probability for exploration
                if np.random.rand() < 0.01:
                    curr_seq = new_seq
        except Exception:
            pass

    print(f"Final best score {best_score:.6f} with length {len(best_seq)}")
    return best_seq
