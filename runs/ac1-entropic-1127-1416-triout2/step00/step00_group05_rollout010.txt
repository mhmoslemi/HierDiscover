"""
Simple stochastic optimizer for the step‑function evaluation.

The score to minimise is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the score is scale invariant we can normalise the vector to
sum(a) = 1 and simply minimise  max(conv(a, a)).  A simulated‑annealing
scheme is used:

* start with a deterministic “U‑shaped’’ distribution (large values at the
  ends, small in the centre) – this shape tends to make the self‑convolution
  flatter than the uniform distribution.
* repeatedly propose a cheap local move:
    – transfer a tiny amount of mass between two positions,
    – swap two positions, or
    – replace the whole vector by a freshly sampled shape (different exponent).
* accept the move if the score improves, otherwise accept with a
  temperature‑dependent probability (standard Metropolis rule).
* keep the best vector seen so far and return it when the time budget
  (≈1000 s) is exhausted.

All numbers are forced to be non‑negative; the evaluator already clips to
[0, 1000] and discards NaNs/Infs, so no extra checks are required.
"""

import time
import numpy as np

def _initial_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a deterministic U‑shaped probability vector of length n.
    The shape a_i ∝ sqrt((i+1)*(n-i)) concentrates mass at the ends.
    It is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    a = np.sqrt((i + 1) * (n - i))
    a /= a.sum()
    return a


def _score(seq):
    """Convenient thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq)


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Stochastic search for a low‑score coefficient vector.

    Parameters
    ----------
    seed : int, optional
        Random seed to make the run reproducible.
    budget_s : int, optional
        Time budget in seconds (default is the full 1000 s).
    **kwargs
        Ignored – present only to match the expected signature.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) configuration
    # ------------------------------------------------------------------
    lengths = [800, 1600, 3200, 6000]          # a few lengths to try
    # total allowed time, keep a small safety margin
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 2) Main optimisation loop (several independent restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length – change slowly so we can explore a few sizes
        n = int(rng.choice(lengths))

        # start from a nicely shaped distribution
        cur = _initial_shape(n, rng).copy()
        cur_score = _score(cur.tolist())
        if cur_score < best_score:
            best_score, best_seq = cur_score, cur.tolist()

        # simple annealing schedule based on the remaining fraction of time
        inner_iters = 0
        while time.time() < deadline:
            # ------------------------------------------------------------------
            # temperature (decreases linearly with wall‑clock time)
            # ------------------------------------------------------------------
            fraction = (deadline - time.time()) / (deadline - (deadline - budget_s + 0.5))
            T = 0.08 * fraction + 1e-5                     # temperature ∈ [1e‑5, 0.08]

            move = rng.choice(["transfer", "swap", "restart"], p=[0.45, 0.45, 0.10])

            if move == "transfer":
                # move a tiny amount of mass from i to j while keeping the sum constant
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                max_transfer = min(cur[i], 0.5)  # 0.5 is an arbitrary upper bound for a single step
                delta = (rng.random() - 0.5) * 2.0 * max_transfer * T
                # keep both entries non‑negative
                delta = np.clip(delta, -cur[i], cur[j])
                if abs(delta) < 1e-12:
                    continue
                new = cur.copy()
                new[i] += delta
                new[j] -= delta

            elif move == "swap":
                # exchange two positions – sum is preserved automatically
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                new = cur.copy()
                new[i], new[j] = new[j], new[i]

            else:   # "restart" – try a fresh random shape with a random exponent
                exp = rng.uniform(0.2, 5.0)
                i = np.arange(n, dtype=float)
                new = (np.maximum(0.0, n - i) ** exp)
                new_sum = new.sum()
                if new_sum == 0:
                    continue
                new = new / new_sum

            # ------------------------------------------------------------------
            # evaluate the proposal
            # ------------------------------------------------------------------
            new_score = _score(new.tolist())
            # Metropolis acceptance
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # probability to accept a worse move
                prob = np.exp(-(new_score - cur_score) / (T + 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur, cur_score = new, new_score
                if cur_score < best_score:
                    best_score, best_seq = cur_score, cur.tolist()

            inner_iters += 1
            # occasional progress report (without flooding stdout)
            if inner_iters % 4000 == 0:
                print(f"[{time.time():.1f}] n={n} iter={inner_iters} best={best_score:.6f}")

            # small safety break to re‑sample a new length after a while
            if inner_iters >= 12000:
                break

    print("Search finished. Best score:", best_score)
    return best_seq
