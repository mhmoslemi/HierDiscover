def project_onto_simplex(v):
    # algorithm for Euclidean projection onto simplex {x >=0, sum x = 1}
    # v is np.ndarray
    n = len(v)

    cond = u - cssv / ind > 0
    rho = ind[cond].max()

    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    # Euclidean projection onto simplex {x >=0, sum=1}
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]

"""Propose a step‑function (non‑negative sequence) that tries to minimise

    2 * n * max(conv(a,a)) / (sum(a)**2)

using a projected gradient descent on a smooth approximation of the
max‑convolution.  The algorithm works as follows

* Choose a length (usually the length of the known best sequence) and
  initialise a random probability vector (sum = 1).
* Repeatedly:
    - Compute the self‑convolution by FFT.
    - Replace the non‑smooth max by a smooth log‑sum‑exp with temperature
      β (β is increased slowly so the approximation becomes tighter).
    - Compute the gradient of the smooth objective using FFT‑based
      convolutions (this gives ∇a  Σk w_k·(a*a)_k with soft‑max weights w_k).
    - Perform a momentum step and project back onto the simplex
      {a_i ≥ 0, Σa_i = 1}.
    - Keep the best sequence seen so far.
* The whole procedure respects the time budget (≈1000 s) and returns the
  best sequence found.

The method is different from the provided LP‑based scheme and is able
to move mass away from the centre of the step‑function, thereby reducing
the peak of the convolution and improving the evaluation score.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1          # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical errors may violate the sum constraint by a tiny amount
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **_: None) -> list:
    """
    Search for a sequence of non‑negative heights that improves the
    current bound.  The function is allowed to run for `budget_s`
    seconds and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0      # a little safety margin

    # ------------------------------------------------------------------
    # 1. Determine the length to work with.
    #    If the reference sequence is available we reuse its length,
    #    otherwise we pick a length in the 1500–2500 range.
    # ------------------------------------------------------------------
    try:
        base_seq = height_sequence_1          # provided globally
        n = len(base_seq)
    except Exception:
        n = np.random.randint(1500, 2500)
        base_seq = None

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # 2. Run several independent optimisation attempts until time runs out.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        #   Initialise a – a probability vector (sum = 1, non‑negative).
        # ------------------------------------------------------------------
        if base_seq is not None and np.random.rand() < 0.5:
            a = np.array(base_seq, dtype=float)
        else:
            a = np.random.rand(n).astype(float)
        a = np.maximum(a, 0.0)
        a = a / a.sum()               # project onto simplex

        # ------------------------------------------------------------------
        #   Hyper‑parameters (they adapt during the run)
        # ------------------------------------------------------------------
        beta = 5.0                    # temperature for smooth max
        step = 0.05                   # base learning rate
        gamma = 0.9                   # momentum coefficient
        momentum = np.zeros_like(a)

        # Pre‑compute FFT lengths (they never change for a fixed n)
        size_conv = _next_pow2(2 * n - 1)                # a * a
        size_big = _next_pow2(3 * n - 2)                # a * w_rev  and w * 1

        # Limit the number of inner iterations to a short burst; after the
        # burst we restart with a fresh random start.  This keeps the total
        # number of restarts high, which helps escaping local minima.
        inner_deadline = time.time() + min(1.5, deadline - time.time())

        while time.time() < inner_deadline and time.time() < deadline:
            # --------------------------------------------------------------
            #  a) convolution a * a (self‑convolution)
            # --------------------------------------------------------------
            a_fft = np.fft.rfft(a, n=size_conv)
            conv_aa = np.fft.irfft(a_fft * a_fft)[: 2 * n - 1]

            max_conv = conv_aa.max()

            obj = 2.0 * n * max_conv

            # --------------------------------------------------------------
            shift = conv_aa - max_conv            # max_conv is subtracted for stability

            #   raw gradient (before the outer 2*n factor)
            grad_raw = grad1 + a * window_slice
            grad = 2.0 * n * grad_raw

            # --------------------------------------------------------------
            #  d) momentum step and projection onto the simplex
            # --------------------------------------------------------------
            momentum = gamma * momentum + step * grad
            a = a - momentum
            a = _project_simplex(a)

            # --------------------------------------------------------------
            #  e) adapt hyper‑parameters (increase β, slowly decay step)
            # --------------------------------------------------------------
            beta = min(5000.0, beta * 1.015)        # tighten smooth max
            step = max(1e-6, step * 0.997)          # slower learning as we go

            # occasional random perturbation to avoid stagnation
            if np.random.rand() < 0.001:
                a += 0.01 * np.random.randn(n)
                a = _project_simplex(a)

        # ------------------------------------------------------------------
        #  End of one optimisation attempt – start a fresh one if time permits.
        # ------------------------------------------------------------------
        # (no extra work needed; the outer while loop will repeat.)

    # ----------------------------------------------------------------------
    #  Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – a uniform distribution (score = 2.0)
        best_seq = np.full(n, 1.0 / n, dtype=float)
    return best_seq.tolist()
