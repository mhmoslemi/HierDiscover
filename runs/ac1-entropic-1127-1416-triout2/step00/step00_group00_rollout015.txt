                best_seq = seq

            break
    # If still None (unlikely), fallback to uniform of length 10

"""
Propose a step‑function sequence that yields a very low evaluation score.

The evaluation metric is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear convolution of the non‑negative sequence `a`.
For a sequence of length `n` whose entries follow a decreasing power law

        a_i = C * (1 - i/(n-1))**p ,   i = 0 … n‑1

the convolution peak is attained near the centre and scales as
    max(conv) ≈ C² * n * B(p+1, p+1)

with `B` the beta‑function.  Because `C` is chosen so that the
maximum entry equals 1000 (to avoid the built‑in clipping), the score
does **not** depend on the overall scale and tends to

    score → 2 * (p+1)² * B(p+1, p+1)

as `n` grows.  This value drops quickly with the exponent `p`:
p=1 → 1.333…, p=2 → 0.600…, p=3 → 0.229…, p=4 → 0.079…, etc.
Thus a simple power‑law shape with a moderate `p` already beats the
target bound (1.503) by a wide margin.

The function below searches a small grid of lengths and exponents,
evaluates the sequences with the provided `evaluate_sequence`,
keeps the best one, and returns it.  The whole search finishes
well within the 1000 s budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a decreasing power‑law sequence and return the one
    with the lowest evaluation score found within the time budget.
    """
    start_time = time.time()
    best_seq = None
    best_score = float("inf")
    best_info = None

    # Grid of parameters to try – small enough to stay fast,
    # large enough to find a very good solution.
    lengths = [500, 1000, 2000]        # candidate lengths

    for n in lengths:

        positions = (n - 1 - np.arange(n, dtype=np.float64)) / (n - 1)
