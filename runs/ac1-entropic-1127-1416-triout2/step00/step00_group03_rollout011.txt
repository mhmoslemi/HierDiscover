import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Hill‑climbing / simulated‑annealing search for a non‑negative step‑function
    sequence that minimises ``evaluate_sequence``.
    The algorithm repeatedly:
        * computes the self‑convolution and locates the index of its maximum,
        * identifies the slice of indices that contribute to this peak,
        * moves a small amount of mass from a high‑contributing index inside the
          slice to a random index outside the slice (keeping the total sum fixed),
        * accepts the move if the evaluation improves (or with a temperature‑based
          probability for uphill moves).
    Random restarts with different lengths and a short uniform‑plus‑noise
    initialisation are performed until the time budget expires.
    The best sequence found is returned as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a couple of seconds for clean exit
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = np.inf

    # Helper to compute convolution max and related data
    def conv_info(seq):
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        return conv, max_idx, max_val

    # Main outer loop: random restarts while we have time
    while time.time() < deadline:
        # Choose a length (between 500 and 3000) – larger n gives finer granularity
        n = np.random.randint(800, 2500)

        # Start from a uniform distribution plus a tiny random perturbation
        seq = np.full(n, 1.0 / n, dtype=np.float64)
        seq += (np.random.rand(n) - 0.5) * 1e-4   # break symmetry
        seq = np.clip(seq, 0.0, None)
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= seq_sum                     # keep total mass = 1

        # evaluate initial candidate
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[init] n={n} score={best_score:.6f}")

        # inner optimisation loop – hill climbing with occasional annealing
        iter_deadline = deadline
        iter_count = 0
        while time.time() < iter_deadline:
            # compute convolution information
            conv, k_max, _ = conv_info(seq)
            # slice of indices that contribute to this peak
            i_start = max(0, k_max - (n - 1))
            i_end   = min(n - 1, k_max)
            slice_idx = np.arange(i_start, i_end + 1, dtype=np.int64)

            # indices that are *not* in the slice (they do not affect this peak)
            mask = np.ones(n, dtype=bool)
            mask[i_start:i_end + 1] = False
            complement_idx = np.nonzero(mask)[0]

            # simple move – perform a tiny random jitter and continue.
            if complement_idx.size == 0:
                i1, i2 = np.random.choice(n, 2, replace=False)
                delta = seq[i1] * (np.random.rand() * 0.02 + 0.01)
                seq[i1] -= delta
                seq[i2] += delta
                continue

            # contribution of each index in the slice to the peak
            partner = k_max - slice_idx                     # partner index for each i
            contrib = seq[slice_idx] * seq[partner]

            # pick a source index proportional to its contribution (more impact)
            probs = contrib + 1e-12
            probs /= probs.sum()
            i_in = np.random.choice(slice_idx, p=probs)

            i_out = np.random.choice(complement_idx)

            src_val = seq[i_in]
            if src_val <= 0.0:

            seq[i_in] -= delta
            seq[i_out] += delta

            new_score = evaluate_sequence(seq.tolist())

            if new_score < cur_score:
