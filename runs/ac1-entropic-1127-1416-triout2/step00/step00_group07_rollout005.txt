        seq = np.maximum(seq, 0.0)
        # Slight random perturbation to avoid a perfectly flat start.
        rng = np.random.default_rng(seed + 1)
        seq += rng.normal(scale=0.001, size=n)
        seq = np.clip(seq, 0.0, None)
    except NameError:
        # No global sequence – fall back to uniform plus noise.
        rng = np.random.default_rng(seed)
        seq = np.ones(n, dtype=float)
        seq += rng.normal(scale=0.001, size=n)
        seq = np.clip(seq, 0.0, None)

    # Normalise the total sum to 1.0 (scale does not affect the metric).
    total = np.sum(seq)
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n, dtype=float)
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).
    **kwargs : dict
        Ignored – present for compatibility with the runner.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    # ----------------------------------------------------------------------
    # configuration
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Fixed length – the SOTA construction used 2145 heights.
    n = 2145

    # Annealing parameters
    temperature = 1.0           # initial temperature
    cooling_rate = 0.9995       # exponential cooling per iteration
    min_temperature = 1e-4
    max_step_frac = 0.05        # maximal fraction of a donor's mass to move
    stagnation_limit = 5000     # iterations without improvement before a restart

    # ----------------------------------------------------------------------
    # initialise
    # ----------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # keep a few seconds margin

    curr_seq = _initial_sequence(n, seed)          # normalised to sum = 1.0
    best_seq = curr_seq.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")
    curr_score = best_score

    # Counters for restart logic
    iterations_without_improve = 0
    total_iterations = 0

    print(f"Start search: n={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # main optimisation loop
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        total_iterations += 1

        # ------------------------------------------------------------------
        # 1) Compute convolution and locate the current worst peak
        # ------------------------------------------------------------------
        conv = np.convolve(curr_seq, curr_seq)
        peak_idx = int(np.argmax(conv))

        # ------------------------------------------------------------------
        # 2) Pick a donor index that actually contributes to the peak
        #    (i must satisfy 0 <= i < n and 0 <= peak_idx-i < n)
        # ------------------------------------------------------------------
        i_min = max(0, peak_idx - (n - 1))
        i_max = min(n - 1, peak_idx)
        donor_candidates = np.arange(i_min, i_max + 1)
        # filter out positions that currently hold zero mass
        donor_candidates = donor_candidates[curr_seq[donor_candidates] > 1e-12]
        if donor_candidates.size == 0:
            # fall back to any index with non‑zero mass
            donor_candidates = np.where(curr_seq > 1e-12)[0]
            if donor_candidates.size == 0:
                break   # nothing left – abort
        donor = rng.choice(donor_candidates)

        # ------------------------------------------------------------------
        # 3) Choose a recipient (any different index)
        # ------------------------------------------------------------------
        recipient = rng.integers(0, n)
        while recipient == donor:
            recipient = rng.integers(0, n)

        # ------------------------------------------------------------------
        # 4) Transfer a small amount of mass (fraction of donor)
        # ------------------------------------------------------------------
        step_frac = max_step_frac * temperature   # shrink step with temperature
        epsilon = curr_seq[donor] * step_frac
        if epsilon <= 0:
            continue

        # Apply the move (sum is preserved)
        curr_seq[donor] -= epsilon
        curr_seq[recipient] += epsilon

        # ------------------------------------------------------------------
        # 5) Evaluate the new sequence
        # ------------------------------------------------------------------
        try:
            new_score = evaluate_sequence(curr_seq.tolist())
        except Exception:
            # revert the move and skip this iteration
            curr_seq[donor] += epsilon
            curr_seq[recipient] -= epsilon
            continue

        # ------------------------------------------------------------------
        # 6) Acceptance criterion (simulated annealing)
        # ------------------------------------------------------------------
        accepted = False
        if new_score < best_score:
            best_score = new_score
            best_seq = curr_seq.copy()
            accepted = True
            iterations_without_improve = 0
            print(f"Iter {total_iterations:>6d}: new best = {best_score:.6f}")
        if new_score < curr_score:
            accepted = True
        else:
            # probabilistic acceptance
            prob = math.exp(-(new_score - curr_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accepted = True

        if accepted:
            curr_score = new_score
        else:
            # revert move
            curr_seq[donor] += epsilon
            curr_seq[recipient] -= epsilon
            iterations_without_improve += 1

        # ------------------------------------------------------------------
        # 7) Temperature schedule
        # ------------------------------------------------------------------
        temperature = max(min_temperature, temperature * cooling_rate)

        # ------------------------------------------------------------------
        # 8) Occasionally inject random noise or restart if stuck
        # ------------------------------------------------------------------
        if total_iterations % 4000 == 0:
            # small global jitter to escape shallow plateaus
            noise = rng.normal(scale=0.002, size=n)
            curr_seq = np.maximum(curr_seq + noise, 0.0)
            # re‑normalise the sum (preserves scale‑invariance)
            s = np.sum(curr_seq)
            if s > 0:
                curr_seq /= s
            # recompute its score
            try:
                curr_score = evaluate_sequence(curr_seq.tolist())
            except Exception:
                curr_score = float("inf")

        if iterations_without_improve >= stagnation_limit:
            # Re‑initialise from a fresh random Dirichlet‑like draw
            print(f"Restart after {iterations_without_improve} stagnant iters")
            gamma_shape = 0.3   # encourages sparsity
            curr_seq = rng.gamma(shape=gamma_shape, scale=1.0, size=n)
            curr_seq /= np.sum(curr_seq)  # normalise to sum = 1
            curr_score = evaluate_sequence(curr_seq.tolist())
            iterations_without_improve = 0
            temperature = 1.0   # reset temperature

    # ----------------------------------------------------------------------
    # final report
    # ----------------------------------------------------------------------
    print(f"Finished after {total_iterations} iterations. Best score = {best_score:.6f}")
    # Return as plain Python list (evaluation routine expects a list)
    return best_seq.tolist()
