Algorithm for simplex projection (Duchi et al 2008):

    def simplex_projection(v):
        # Projects v onto the probability simplex (sum=1, nonnegative)
        n = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(n) + 1
        cond = u - cssv / ind > 0
        rho = ind[cond].max()
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

def simplex_projection(v):
    """Project vector v onto the probability simplex (nonnegative, sum = 1)."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]

        return np.full_like(v, 1.0 / n)

    """

    { w | w_i >= 0 , sum w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
