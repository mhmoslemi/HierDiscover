def simplex_projection(v):
    # v is numpy array.
    # Implementation via sorting.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def max_conv(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    return np.max(conv)

Propose candidate step heights using a tunable exponential distribution.
The algorithm:
- Determine the length n from the existing construction (height_sequence_1)
  or fallback to a default.
- Search over decay rates c for a distribution a_i = exp(-c*i) (and also the
  reverse order).  After normalising a to sum 1, the evaluation value is
  2*n*max(conv(a)).
- Compute each convolution efficiently with an FFT based autocorrelation.
- Keep the best sequence found within the time budget.
- Optionally perform a few mirror‑descent refinement steps starting from
  the best candidate.
- Return the best sequence as a plain list of floats.

import time
import numpy as np
from numpy.fft import rfft, irfft

def _conv_max(seq):
    """Return max of the linear convolution seq * seq using FFT."""
    n = seq.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()  # next power of two >= m
    # Real FFT of the sequence padded to size
    F = rfft(seq, n=size)
    conv = irfft(F * F, n=size)[:m]
    return conv.max()

def _evaluate_prob(p):
    """Given a probability vector (sum to 1), compute evaluation."""
    n = p.shape[0]

    # --------------------------------------------------------------------

        best_seq = np.full(n, 1.0 / n)

"""
Propose a step function with a low evaluation value.

The algorithm searches a simple one‑parameter family of exponential
distributions a_i = exp(-c*i) (and its reversed version) together with a
Gaussian family.  For a probability vector p (∑p_i = 1) the evaluation
function equals

    score = 2·n·max(p * p)               (p * p = linear convolution)

which depends only on the shape of p.  Using an FFT we can evaluate
max(p * p) very quickly, so a dense grid of decay rates c can be tried
well within the 1000 s budget.  The best candidate is then refined by a
few mirror‑descent steps that move probability mass away from indices
participating in the current maximal convolution value.

The function works with the global variable `height_sequence_1` if it is
present, otherwise falls back to a default length.  The returned list

import time
import numpy as np
from numpy.fft import rfft, irfft


def _max_convolution(seq: np.ndarray) -> float:
    """
    Compute max(conv(seq, seq)) using an FFT based autocorrelation.
    """
    n = seq.shape[0]
    m = 2 * n - 1                      # length of the linear convolution
    size = 1 << (m - 1).bit_length()   # next power of two ≥ m
    F = rfft(seq, n=size)
    conv = irfft(F * F, n=size)[:m]    # linear convolution, real result
    return float(conv.max())


def _score_from_prob(p: np.ndarray) -> float:
    """
    For a probability vector p (∑p = 1) compute the evaluation score,
    which equals 2·n·max(p * p).
    """
    n = p.shape[0]
    return 2.0 * n * _max_convolution(p)
