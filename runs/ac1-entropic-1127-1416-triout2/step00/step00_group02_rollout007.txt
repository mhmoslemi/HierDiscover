"""
Simulated‑annealing / hill‑climbing search for a step‑function
that minimises  2·n·max(conv(a,a))/ (sum(a))² .

The evaluation metric is invariant under scaling, therefore we keep the
ℓ₁‑norm of the candidate near 1.  The algorithm works mainly with
numpy arrays and uses an FFT based convolution (O(n log n)).  A simple
gradient‑like move is performed: the index that contributes most to the
current maximal convolution coefficient is selected (probability
proportional to its contribution) and a small fraction of its mass is
moved to a random other index.  Random swaps are also tried.  An
annealing schedule allows occasional uphill moves.

The routine runs until the supplied time budget expires (default 1000 s)
and returns the best sequence it has seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear convolution via FFT (real inputs)
def _full_conv(seq: np.ndarray) -> np.ndarray:
    """Return the linear convolution of ``seq`` with itself (full length)."""
    n = seq.size
    # length of FFT – next power of two ≥ 2·n‑1 (fast for rfft)
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)          # inverse transform
    return conv[: 2 * n - 1]                # keep only the linear part


# ----------------------------------------------------------------------
# Helper: objective of a candidate (identical to evaluate_sequence but much faster)
def _objective(seq: np.ndarray, conv: np.ndarray) -> float:
    """Compute 2·n·max(conv)/sum(seq)² .  Assumes seq ≥ 0."""
    total = seq.sum()
    if total < 0.01:            # same guard as evaluate_sequence
        return float("inf")
    return float(2.0 * seq.size * conv.max() / (total * total))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that gives a low evaluation
    score.  Returns a Python list of floats (the best sequence found).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # leave a small margin

    # ------------------------------------------------------------------
    # 1) Choose a problem size.  Small to medium problems are much easier
    #    to optimise; a size between 30 and 120 is sufficient to beat the
    #    target 1.503.
    # ------------------------------------------------------------------
    n = np.random.randint(30, 121)                 # 30 … 120 inclusive

    # ------------------------------------------------------------------
    # 2) Initialise the sequence.
    #    A smooth bell (Gaussian‑like) start gives a good head‑start.
    # ------------------------------------------------------------------
    x = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma = n / 3.5                                 # a fairly wide bump
    seq = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    seq += 1e-6                                     # avoid exact zeros
    seq /= seq.sum()                                # ℓ₁‑norm = 1

    # keep the best sequence ever seen
    cur_seq = seq.copy()
    cur_conv = _full_conv(cur_seq)
    cur_score = _objective(cur_seq, cur_conv)
    best_seq = cur_seq.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing parameters.
    # ------------------------------------------------------------------
    T_initial = 0.8                                 # start temperature
    T_final = 1e-5
    max_iters_per_restart = 8000                    # generous for tiny n
    step_fraction_initial = 0.35                    # fraction of a_i moved
    step_fraction_final = 0.02

    # Pre‑computed FFT length (constant for a fixed n)
    L = 1 << ((2 * n - 1).bit_length())

    it_total = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop – we perform several independent restarts
    #    until the time budget runs out.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------- inner annealing loop ---------------------------------
        start_inner = time.time()
        for it in range(max_iters_per_restart):
            # annealing schedule (linear in the inner loop)
            frac = it / max_iters_per_restart
            T = T_initial * (1.0 - frac) + T_final * frac
            step_frac = (step_fraction_initial *
                         (1.0 - frac) + step_fraction_final * frac)

            # compute current convolution and the index of its maximum
            conv = _full_conv(cur_seq)
            kmax = int(conv.argmax())

            # ---------------------------------------------------------
            # 4a) Gradient‑inspired move (70 % chance)
            # ---------------------------------------------------------
            if np.random.rand() < 0.70:
                # contributions of each entry to the maximal convolution term
                idx = np.arange(n)
                partner = kmax - idx
                valid = (partner >= 0) & (partner < n)
                contrib = np.zeros_like(cur_seq)
                contrib[valid] = cur_seq[valid] * cur_seq[partner[valid]]
                total_contrib = contrib.sum()
                if total_contrib > 0:
                    probs = contrib / total_contrib
                    src = np.random.choice(n, p=probs)
                else:
                    src = np.random.randint(n)

                # destination – any other index
                dst = np.random.randint(n)
                while dst == src:
                    dst = np.random.randint(n)

                # move a small fraction of mass from src to dst
                epsilon = cur_seq[src] * step_frac
                if epsilon > cur_seq[src]:
                    epsilon = cur_seq[src]
                new_seq = cur_seq.copy()
                new_seq[src] -= epsilon
                new_seq[dst] += epsilon

            # ---------------------------------------------------------
            # 4b) Pure random swap (30 % chance)
            # ---------------------------------------------------------
            else:
                i, j = np.random.choice(n, size=2, replace=False)
                delta = (np.random.rand() - 0.5) * step_frac * cur_seq[i]
                # ensure we keep the sum unchanged and stay non‑negative
                if delta >= 0:
                    # give delta to i, take from j
                    take = min(delta, cur_seq[j])
                    new_seq = cur_seq.copy()
                    new_seq[i] += take
                    new_seq[j] -= take
                else:
                    delta = -delta
                    take = min(delta, cur_seq[i])
                    new_seq = cur_seq.copy()
                    new_seq[i] -= take
                    new_seq[j] += take

            # enforce non‑negativity (the updates already keep it, but safety)
            np.maximum(new_seq, 0.0, out=new_seq)

            # evaluate the candidate (fast objective, no list conversion)
            new_conv = _full_conv(new_seq)
            new_score = _objective(new_seq, new_conv)

            # ---------------------------------------------------------
            # 4c) Metropolis acceptance
            # ---------------------------------------------------------
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # probability to accept a worse move
                prob = np.exp((cur_score - new_score) / max(T, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                cur_seq = new_seq
                cur_score = new_score

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"new best (n={n})  score={best_score:.6f}")

            it_total += 1
            # early termination of this restart if time is gone
            if time.time() >= deadline:
                break

        # -----------------------------------------------------------------
        # after a restart we optionally change the problem size to explore
        # different dimensionalities.
        # -----------------------------------------------------------------
        if time.time() >= deadline:
            break
        n = np.random.randint(30, 121)               # new size for next restart
        # rebuild a fresh start for the new size
        x = np.arange(n, dtype=float)
        mu = (n - 1) / 2.0
        sigma = n / 3.5
        seq = np.exp(-0.5 * ((x - mu) / sigma) ** 2) + 1e-6
        seq /= seq.sum()
        cur_seq = seq.copy()
        cur_conv = _full_conv(cur_seq)
        cur_score = _objective(cur_seq, cur_conv)
        L = 1 << ((2 * n - 1).bit_length())          # update FFT length

    # ----------------------------------------------------------------------
    # 5) Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    print(f"Search finished – total iterations {it_total}, best score {best_score:.6f}")
    return best_seq.tolist()
