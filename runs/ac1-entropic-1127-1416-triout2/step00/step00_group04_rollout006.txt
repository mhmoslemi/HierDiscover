"""
Algorithm
---------

We treat the objective as  R(a) = 2·n·max(conv(a,a))  because
the sum of the sequence is kept constant (scaled to 1).  The goal is
to minimise the peak of the linear convolution of a non‑negative vector
with itself.

The search works on a single vector `a` of length `n` (taken from the
provided `height_sequence_1` or created at random).  The vector is
normalised to have sum = 1, which makes the objective invariant to
uniform scaling.

A fast O(n) update of the convolution is used when a small amount of
mass `Δ` is transferred from index `i` to index `j`:

    conv[i : i+n] += 2·Δ·a                (linear part)
    conv[2·i]    += Δ²                    (quadratic self‑term)

The update is performed **before** the underlying entry `a[i]` is
changed, so the linear term uses the old value of `a[i]`.

Three inexpensive move types are mixed:

* **Random transfer** – move a small random amount of mass between two
  arbitrary indices.
* **Central → outer transfer** – move mass from the middle half of the
  array to the outer quarters; this tends to flatten the triangular
  autocorrelation of the uniform distribution.
* **Peak‑focused transfer** – locate the current maximal convolution entry
  `k_max`, pick the pair `(i, k_max-i)` that contributes most to this
  peak and shift a fraction of the larger component to a far away index.

A simple simulated‑annealing acceptance rule is applied:
if the move improves the current score it is kept, otherwise it is
accepted with probability `exp(-(Δscore)/T)` where the temperature `T`
decays linearly to zero over the allotted time.

Periodic full recomputation of the convolution eliminates accumulated
floating‑point drift.

The best sequence seen during the run is stored and finally returned.
All numbers stay non‑negative, and the sequence length is the same as the
starting one (≈ 7 000), which is a good trade‑off between attack surface
and speed.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper that updates the full linear convolution when a single entry
# a[idx] is changed by `delta`.  `a` holds the *old* values, `conv` is
# the current convolution (full, length 2*n-1).  The function mutates
# both arrays in place.
def _adjust_convolution(conv: np.ndarray, a: np.ndarray, idx: int, delta: float) -> None:
    """Apply the effect of a change `delta` at position `idx` to `conv`."""
    n = a.shape[0]
    # Linear part: for every j we add 2·delta·a[j] to conv[idx+j]
    conv[idx: idx + n] += 2.0 * delta * a
    # Quadratic self term (the extra delta² in (a_i+delta)²)
    conv[2 * idx] += delta * delta


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The function runs for at most
    `budget_s` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Give a small safety margin so we finish before the hard timeout.
    deadline = start_time + max(0.0, budget_s - 5.0)

    # ------------------------------------------------------------------
    # Choose a starting vector.
    if 'height_sequence_1' in globals() and rng.random() < 0.6:
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        # random Dirichlet‑style vector (different shapes are useful)
        # Length is taken from the known construction if available,
        # otherwise a default of 6000.
        n = len(globals().get('height_sequence_1', np.empty(0)))
        if n == 0:
            n = 6000
        a = rng.random(n)
    # Normalise to have sum = 1 (objective is scale‑invariant)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / a.shape[0]
    else:
        a /= s

    n = a.shape[0]
    conv = np.convolve(a, a)                     # full convolution, length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max                # identical to evaluate_sequence when sum=1
    best_a = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Main simulated‑annealing loop.
    it = 0
    while time.time() < deadline:
        it += 1
        # ------------------------------------------------------------------
        # Choose a move type
        move_type = rng.random()
        if move_type < 0.5:
            # ---- Random transfer -----------------------------------------
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] <= 0.0:
                continue
            max_delta = min(a[i], 0.5 * a[i])
            delta = rng.random() * max_delta
            # apply move
            _adjust_convolution(conv, a, i, -delta)
            a[i] -= delta
            _adjust_convolution(conv, a, j, delta)
            a[j] += delta

        elif move_type < 0.75:
            # ---- Central → outer transfer --------------------------------
            # pick source from the middle half
            lo_mid = int(0.25 * n)
            hi_mid = int(0.75 * n)
            i = rng.integers(lo_mid, hi_mid)
            if a[i] <= 0.0:
                continue
            # pick target from the outer quarters
            if rng.random() < 0.5:
                j = rng.integers(0, int(0.25 * n))
            else:
                j = rng.integers(int(0.75 * n), n)
            if i == j:
                continue
            max_delta = min(a[i], 0.5 * a[i])
            delta = rng.random() * max_delta
            _adjust_convolution(conv, a, i, -delta)
            a[i] -= delta
            _adjust_convolution(conv, a, j, delta)
            a[j] += delta

        else:
            # ---- Peak‑focused transfer ------------------------------------
            k_max = int(conv.argmax())
            # indices that actually contribute to this peak
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idxs = np.arange(i_start, i_end + 1, dtype=int)

            # contribution of each candidate i to the peak
            contrib = a[idxs] * a[k_max - idxs]
            # pick the index with the largest contribution
            best_local = int(np.argmax(contrib))
            i_candidate = int(idxs[best_local])
            j_partner = k_max - i_candidate

            # move mass from the larger of the pair (i_candidate, j_partner)
            if a[i_candidate] >= a[j_partner]:
                src = i_candidate
            else:
                src = j_partner

            if a[src] <= 0.0:
                continue

            # destination far from src (at least n/4 away)
            offset = rng.integers(int(0.25 * n), int(0.5 * n))
            dst = (src + offset) % n
            if dst == src:
                dst = (dst + 1) % n

            max_delta = min(a[src], 0.5 * a[src])
            delta = rng.random() * max_delta
            _adjust_convolution(conv, a, src, -delta)
            a[src] -= delta
            _adjust_convolution(conv, a, dst, delta)
            a[dst] += delta

        # ------------------------------------------------------------------
        # Occasionally recompute the full convolution to avoid drift.
        if it % 2000 == 0:
            conv = np.convolve(a, a)

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # Simulated annealing acceptance
        elapsed = time.time() - start_time
        progress = elapsed / max(1e-6, (deadline - start_time))
        # temperature linearly decays from ~0.02 to 0
        temperature = 0.02 * (1.0 - progress)

        accept = False
        if new_score <= cur_score:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:

            cur_max = new_max
            # record the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{it:8d}] new best score = {best_score:.6f}  max = {cur_max:.8e}")
        else:
            # revert the move (undo the two adjustments)
            # we stored the indices and delta above for each branch,
            # so we simply reverse them.
            # The variables `src`, `dst`, `i`, `j`, `delta` have the last
            # values used – they are always defined in the three branches.
            # The sign reversal restores the previous state.
            # Note: for the random‑transfer branch the names are `i`,`j`.
            # For the central→outer branch they are also `i`,`j`.
            # For the peak‑focused branch they are `src` and `dst`.
            if move_type < 0.5:
                # random transfer
                _adjust_convolution(conv, a, i, delta)
                a[i] += delta
                _adjust_convolution(conv, a, j, -delta)
                a[j] -= delta
            elif move_type < 0.75:
                # central → outer
                _adjust_convolution(conv, a, i, delta)
                a[i] += delta
                _adjust_convolution(conv, a, j, -delta)
                a[j] -= delta
            else:
                # peak‑focused
                _adjust_convolution(conv, a, src, delta)
                a[src] += delta
                _adjust_convolution(conv, a, dst, -delta)
                a[dst] -= delta

        # Safety: keep numerical negativity out of the vector
        a = np.maximum(a, 0.0)

    # Return the best sequence found (as a plain Python list)
    return best_a.tolist()
