def reduce_max(seq, eta=0.01):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    new = np.array(seq)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            # contribution derivative of b_k w.r.t a_i is 2 * a_j (if i != j) else 2 * a_i
            grad = 2 * new[j] if i != j else 2 * new[i]
            new[i] = max(0.0, new[i] - eta * grad)
    # renormalize to sum 1
    s = new.sum()
    if s > 0:
        new = new / s
    return new

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1
    best_score = float('inf')
    best_seq = None
    # try different lengths

        n = np.random.randint(3, 13)  # small lengths

        seq = np.random.dirichlet(alpha * np.ones(n)).tolist()

        if score < best_score:

            # perturb by moving small mass

            score2 = evaluate_sequence(new_seq)
            if score2 < best_score:
                best_score = score2
                best_seq = new_seq

"""candidate_proposal.py
A simple stochastic optimizer for the step‑function evaluation problem.

The evaluation function is scale‑invariant, so we can fix the total sum of the
sequence to 1.  The goal is to minimise

    2 * n * max(convolution(a, a)) / (sum(a)**2)  ==  2 * n * max_b

because sum(a) = 1.

For very short sequences the optimum can be derived analytically (n=3).  For
larger lengths we perform a cheap random‑search / hill‑climbing procedure:
* pick a random length 3‥12,
* draw a Dirichlet vector (the entries are non‑negative and sum to 1),
* repeatedly move a small amount of mass from a random index to another
  (keeping all entries ≥ 0) and keep the change if it improves the score,
* keep the best sequence ever seen.

The whole search runs until the time budget expires (≈ 1000 s).  The best
sequence found is returned.  The algorithm is deliberately lightweight so it
fits in the given limits while still being able to beat the trivial uniform
construction (value ≈ 2.0).  In practice the analytic 3‑term solution
achieves a score ≈ 1.7067, which is the best known simple construction.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def eval_score(seq):
    """Wrapper around the provided evaluate_sequence (global)."""
    return evaluate_sequence(seq)


def analytic_three_term():
    """
    Analytic optimum for n = 3.

    The optimal vector (up to scaling) solves
        a = 0.2,   b = 4/15,   c = 8/15   (sum = 1)
    giving an evaluation score of 384/225 ≈ 1.706666…
    """
    a = 0.2
    b = 4.0 / 15.0
    c = 8.0 / 15.0
    return [a, b, c]


def random_sequence(n, concentration):
    """
    Draw a random Dirichlet vector of dimension n with concentration
    parameter `concentration` (scalar).  The vector is already normalised
    to have sum = 1.
    """
    # Dirichlet with identical concentration for all entries
    vec = np.random.dirichlet([concentration] * n)
    return vec.tolist()


def perturb(seq, eps=0.1):
    """
    Randomly transfer a small amount of mass from one index to another.
    The total sum stays unchanged and all entries remain non‑negative.
    """
    n = len(seq)
    i, j = np.random.choice(n, 2, replace=False)
    # amount to move is a fraction of the source entry
    delta = seq[i] * eps * np.random.rand()
    if delta <= seq[i]:
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # small numerical safety – clamp tiny negatives to zero
        if new_seq[i] < 0:
            new_seq[i] = 0.0
        return new_seq
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Find a sequence of non‑negative heights with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds; the function stops a few seconds before the
        deadline to avoid a forced termination.
    **kwargs : dict
        Ignored – present for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Start with the analytically optimal 3‑term construction.
    # ------------------------------------------------------------------
    best_seq = analytic_three_term()
    best_score = eval_score(best_seq)
    print(f"Initial analytic 3‑term score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stochastic search over short sequences (3‥12 elements).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a random length and a random Dirichlet concentration.
        n = np.random.randint(3, 13)            # lengths 3 – 12
        concentration = np.random.uniform(0.05, 2.0)

        # Initialise a random probability vector (sum = 1).
        seq = random_sequence(n, concentration)

        # Evaluate the fresh random point.
        cur_score = eval_score(seq)

        # Keep it if it improves the global best.
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq
            print(f"[{time.time():.1f}s] New best (rand) {best_score:.6f} n={n}")

        # ------------------------------------------------------------------
        # Short hill‑climbing phase starting from this point.
        # ------------------------------------------------------------------
        for _ in range(25):          # a few local steps per random start
            new_seq = perturb(seq, eps=0.2)
            new_score = eval_score(new_seq)

            # Accept the move if it improves the *local* score.
            if new_score < cur_score:
                seq = new_seq
                cur_score = new_score

                # Propagate any global improvement.
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq
                    print(f"[{time.time():.1f}s] New best (local) {best_score:.6f} n={n}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats).
    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
