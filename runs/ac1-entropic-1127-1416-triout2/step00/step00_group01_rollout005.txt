def project_to_simplex(v):
    # Implementation: from "Efficient Projections onto the l1-ball for Learning in High Dimensions". Standard algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Implementation for project_onto_simplex (algorithm by Duchi et al.):

def project_onto_simplex(v):
    """Projects v onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x>=0, sum=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        # all entries are zero or negative, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_subgradient(a):
    """Return (M, grad) where M = max_k (a*a)[k] and grad = subgradient of M."""
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n-1
    M = conv.max()
    # indices where the max is attained (with a small tolerance)
    tol = M * 1e-12 + 1e-14
    active = np.where(conv >= M - tol)[0]
    # indicator of active k's
    B = np.zeros(2 * n - 1, dtype=a.dtype)
    B[active] = 1.0
    # gradient g_i = 2 * sum_{k in active} a_{k-i}
    # can be obtained via convolution of B with reversed a
    a_rev = a[::-1]
    temp = np.convolve(B, a_rev)                # length 3n-2
    start = n - 1
    grad = 2.0 * temp[start:start + n]
    return M, grad, conv
