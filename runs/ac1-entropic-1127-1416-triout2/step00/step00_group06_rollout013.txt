    # Helper to generate candidates
    def random_dirichlet(alpha):
        # draw from Gamma(alpha,1) for each component.
        vec = np.random.gamma(alpha, 1.0, size=n)
        if np.sum(vec) == 0:
            return None
        vec = (vec / np.sum(vec)) * target_sum
        return vec.tolist()

        mid = (n-1)/2
        idx = np.arange(n)
        vals = np.abs(idx - mid) ** p + 0.01  # add epsilon
        vals = vals / np.sum(vals) * target_sum
        return vals.tolist()

    def exponential(r):

        vals = r**idx
        vals = vals / np.sum(vals) * target_sum
        return vals.tolist()

    # candidate generators
    generators = []

    alphas = [0.5, 1.0, 2.0, 5.0]

    # exponential with r close to 1
    for r in [0.95, 0.98, 0.99, 0.995]:
        generators.append(lambda r=r: exponential(r))

    # Main loop: generate candidates
    while time.time() < deadline:

        seq = gen()
        if seq is None:

        if np.random.rand() < 0.1:

            for _ in range(5):

                seq[i] = max(0.0, seq[i] + np.random.randn()*0.01*target_sum)  # scale perturb
            seq = (seq / np.sum(seq)) * target_sum
            seq = seq.tolist()
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"New best score {best_score:.6f} at time {time.time():.2f}")

        if time.time() < deadline and np.random.rand() < 0.05:

                    # evaluate

                    if new_score < best_score:
                        best_score = new_score
                        best_seq = cur_seq.tolist()
                        print(f"Refined best score {best_score:.6f}")

    # Return best sequence

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

which, after the mandatory normalisation sum(a)=sqrt(2*n), is simply the
maximum value of the auto‑convolution of the sequence.
The implementation builds an analytical “inverse‑square‑root” sequence,
which is known to give an almost flat auto‑convolution (the continuous
analogue of 1/√x convolved with itself is constant).  The generated
vectors are normalised, their score is evaluated and the best one is kept.
After the initial sweep we perform a short stochastic local‑search
(hill‑climbing with occasional random jumps) that moves a small amount
of mass between two positions while preserving the total sum; only moves
that improve the score are accepted.  The whole search respects the
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are defined at top level as required.
# ----------------------------------------------------------------------


def _normalise_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale the non‑negative array so that its L1‑norm equals *target_sum*."""
    total = seq.sum()
    if total == 0:
        # avoid division by zero – return a uniform tiny vector
        return np.full_like(seq, target_sum / len(seq))
    return seq * (target_sum / total)


def _candidate_power_law(n: int, p: float, offset: float, target_sum: float) -> np.ndarray:
    """
    Produce a = (i+offset+1)^{-p} (i = 0 … n‑1) and normalise to *target_sum*.
    The offset avoids a singularity at i=0 and lets us explore nearby shapes.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = 1.0 / np.power(idx + offset + 1.0, p)
    return _normalise_to_target(raw, target_sum)


def _candidate_dirichlet(n: int, alpha: float, target_sum: float) -> np.ndarray:
    """
    Sample a Dirichlet (Gamma) vector with concentration *alpha* and
    normalise to *target_sum*.
    """
    # Gamma(α,1) – the shape parameter α may be <1 (spiky) or >1 (smooth)
    vec = np.random.gamma(alpha, 1.0, size=n)
    return _normalise_to_target(vec, target_sum)


def _hill_climb(seq: np.ndarray,
                target_sum: float,
                deadline: float,
                best_score: float,
                evaluate,
                verbose: bool = True) -> (np.ndarray, float):
    """
    Simple hill‑climbing: repeatedly transfer a small amount of mass from a
    randomly chosen index to another one.  Only improvements are kept.
    """
    n = len(seq)
    cur = seq.copy()
    cur_score = best_score

    # Pre‑compute a tiny fraction that will be moved each step
    # (relative to the current total mass on the chosen source index)
    move_fraction = 0.03  # 3 % of the source entry

    while time.time() < deadline:
        # pick two different indices
        i, j = np.random.choice(n, 2, replace=False)
        if cur[i] == 0:
            continue
        delta = cur[i] * move_fraction

        # propose new vector
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # re‑normalise (tiny numerical drift may appear)
        new = _normalise_to_target(new, target_sum)

        new_score = evaluate(new.tolist())
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if verbose:
                print(f"  hill‑climb improvement: {cur_score:.6f}")
            # keep this as the new global best as well
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                if verbose:
                    print(f"* New global best: {best_score:.6f}")
        # occasional random jitter to escape flat regions
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * target_sum * 0.001
            new = cur + jitter
            new = np.maximum(new, 0.0)
            new = _normalise_to_target(new, target_sum)
            new_score = evaluate(new.tolist())
            if new_score < cur_score:
                cur = new
                cur_score = new_score
                if verbose:
                    print(f"  jitter improvement: {cur_score:.6f}")

    return cur, cur_score

        n = 5208

    target_sum = np.sqrt(2.0 * n)           # required L1‑norm after normalisation

    # ------------------------------------------------------------------
    # 1.  Sweep over analytical power‑law families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    exponents = np.linspace(0.35, 0.65, 7)   # around the optimal 0.5
    offsets   = [0.0, 0.5, 2.0, 5.0]        # small shifts to tune the edge

    for p in exponents:
        for off in offsets:
            cand = _candidate_power_law(n, p, off, target_sum)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[init] power‑law p={p:.3f} off={off:.1f} → score {score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Sample a few Dirichlet vectors (different concentration levels)
    # ------------------------------------------------------------------
    alphas = [0.5, 1.0, 2.0, 5.0]
    for a in alphas:
        for _ in range(5):      # a few draws for each α
            cand = _candidate_dirichlet(n, a, target_sum)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[init] Dirichlet α={a:.1f} → score {score:.6f}")

    # If for any reason no candidate was accepted, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence.")

    # ------------------------------------------------------------------
    # 3.  Stochastic hill‑climbing from the current best sequence.
    # ------------------------------------------------------------------
    # Reserve the remaining time for the local optimisation.
    remaining_deadline = deadline
    if time.time() < remaining_deadline:
        print(f"[search] starting hill‑climb (time left ≈ {remaining_deadline - time.time():.1f}s)")
        best_seq, best_score = _hill_climb(best_seq,
                                           target_sum,
                                           remaining_deadline,
                                           best_score,
                                           evaluate_sequence,
                                           verbose=True)

    # final sanity: ensure we return a plain Python list of floats
    final_seq = [float(x) for x in best_seq]
    print(f"[final] best score achieved: {best_score:.6f} (length {len(final_seq)})")
    return final_seq
