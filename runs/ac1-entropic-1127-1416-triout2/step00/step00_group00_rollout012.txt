        # k from 0 to 2n-2
        for k in range(2 * n - 1):
            # nonzero pairs i where 0<=i<n and 0<=k-i<n
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # for each i, b_j coefficient at position j = k - i
            # So row[j] = f_seq[i]
            # We'll add entries logically: for each i, j = k-i
            # add f_seq[i] at column j
            for i in range(i_min, i_max + 1):
                j = k - i
                rows.append(k)
                cols.append(j)
                data.append(f_seq[i])
            b_ub.append(rhs)
        # Nonnegativity constraints: -b_j <= 0
        # add identity rows
        for i in range(n):
            rows.append(2 * n - 1 + i)
            cols.append(i)
            data.append(-1.0)
            b_ub.append(0.0)
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1 + n, n))
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 1.0})
            if res.success:
                return res.x
        except Exception:
            pass
        return None

    def get_lp_direction(seq):
        # compute normalized seq to have sum sqrt(2n)
        n = len(seq)
        sum_seq = np.sum(seq)
        if sum_seq < 1e-12:
            return None
        norm_seq = seq * np.sqrt(2 * n) / sum_seq
        rhs = max(np.convolve(norm_seq, norm_seq))
        g = solve_lp(norm_seq, rhs)
        if g is None:
            return None
        sum_g = np.sum(g)
        if sum_g == 0:
            return None
        # rescale g to same sum as original seq
        g_norm = g * np.sqrt(2 * n) / sum_g
        return g_norm

    # main loop
    iter_no = 0
    last_improve = time.time()
    while time.time() < deadline:
        iter_no += 1
        # choose operation with some probabilities

                        last_improve = time.time()
                        print(f'Iter {iter_no}: LP mix t={t:.2f} improved score {best_score}')

        if op < 0.5:

            seq = best_seq.copy()

            seq[i] -= amount
            seq[j] += amount
            # enforce non-negativity
            seq = np.maximum(seq, 0.0)
            s = evaluate(seq)
            if s < best_score:
                best_score = s
                best_seq = seq
                last_improve = time.time()
                print(f'Iter {iter_no}: transfer improved score {best_score}')

            idx = rng.integers(0, len(seq))
            scale = 0.05 * (seq[idx] if seq[idx] > 0 else 1.0)
            delta = rng.normal(scale=scale)
            seq[idx] = max(0.0, seq[idx] + delta)
            seq = np.maximum(seq, 0.0)
            s = evaluate(seq)
            if s < best_score:
                best_score = s
                best_seq = seq

"""
Propose a near‑optimal step‑function sequence.

The objective of ``evaluate_sequence`` is
    2 * n * max(conv(a,a)) / (sum(a)**2)

which is invariant under scaling of ``a``.  Hence the task is to find a
non‑negative vector whose *self‑convolution* is as flat as possible.  A
uniform vector gives a triangular convolution and a value of exactly 2.
The algorithm below searches for flatter shapes by

1.  Trying a palette of analytically motivated initialisations
    (uniform, harmonic, exponential, √‑difference, etc.).
2.  Keeping the best sequence found so far.
3.  Repeatedly applying cheap local moves:
        – transfer a random amount of mass between two indices,
        – add a small Gaussian perturbation to a random entry.
4.  Occasionally solving a small linear‑program that maximises the total
    mass while keeping the current convolution bound (the same idea as
    the SOTA code) and mixing the result with a tunable step size.
5.  Occasionally restarting from a fresh random shape.

All moves are evaluated with ``evaluate_sequence``; the current best
sequence is printed whenever it improves.  The loop stops a few seconds
before the allotted budget so that the function always returns a result.

The method is deliberately simple, avoids large dense LP matrices
(only runs the LP when the length is ≤ 800) and works well inside the
1000 s time limit.
"""

import time
import numpy as np
from scipy import optimize, sparse


def _eval(seq: np.ndarray) -> float:
    """Wrapper for the provided evaluate_sequence that accepts ndarray."""
    return evaluate_sequence(seq.tolist())  # evaluate_sequence is pre‑imported


def _solve_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve   maximize sum(b)  subject to  conv(f_seq, b) ≤ rhs,  b ≥ 0.
    Uses a sparse formulation; returns the optimal b as ndarray or None.
    """
    n = len(f_seq)
    # Build sparse matrix for the convolution constraints.
    rows = []
    cols = []
    data = []
    b_ub = []

    # conv constraints
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f_seq[i])
        b_ub.append(rhs)

    # non‑negativity:  -b_i ≤ 0
    offset = 2 * n - 1
    for i in range(n):
        rows.append(offset + i)
        cols.append(i)
        data.append(-1.0)
        b_ub.append(0.0)

    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(offset + n, n))
    c = -np.ones(n)                     # maximise sum(b)  ⇔ minimising -sum(b)

    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": 1.0},
        )
        if res.success:
            return res.x
    except Exception:                  # keep silent – LP may fail for large n
        pass
    return None


def _lp_direction(seq: np.ndarray):
    """
    Return a promising direction g obtained from the LP described in the
    problem statement.  The returned vector has the same L1‐norm as ``seq``.
    If the LP fails (or length is too large) returns None.
    """
    n = len(seq)
    total = seq.sum()
    if total < 1e-12:
        return None
    # Normalise to the convention used by the original code
    norm_seq = seq * np.sqrt(2.0 * n) / total
    rhs = np.max(np.convolve(norm_seq, norm_seq))

    # For speed restrict the LP to modest sizes
    if n > 800:
        return None

    g = _solve_lp(norm_seq, rhs)
    if g is None:
        return None
    sum_g = g.sum()
    if sum_g == 0:
        return None
    # Rescale g to the same L1‑norm as the original sequence
    g_rescaled = g * np.sqrt(2.0 * n) / sum_g
    return g_rescaled
