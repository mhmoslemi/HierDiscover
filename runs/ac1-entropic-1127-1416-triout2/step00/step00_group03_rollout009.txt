"""
Propose a step‑function (non‑negative sequence) with a low value of
evaluate_sequence().  The search combines three cheap heuristics:

1.  Random Dirichlet samples of various lengths and concentration parameters.
    Every sample is normalised to sum 1 (the objective is scale‑invariant).

2.  A cheap greedy “max‑convolution reduction’’ that repeatedly moves a small
    amount of mass away from the pair of indices that contributes most to the
    current maximal convolution value.  The move is kept only if it lowers the
    objective (2·n·max(conv)).

3.  Occasionally (rarely) the LP step from the reference implementation is
    invoked – it can give a strong direction when the simple greedy step stalls.

The loop runs until the time budget (default 1000 s) is exhausted and returns
the best sequence found.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_sequence, rhs):
    """
    Solve the LP:
        maximise   Σ b_j
        subject to  (f * b)[k] ≤ rhs   for all k
                    b_j ≥ 0
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_sequence)
    c = -np.ones(n)                     # maximise sum → minimise –sum
    a_ub = []
    b_ub = []
    # Convolution constraints
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints  -b_j ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # HiGHS has a 'time_limit' option (seconds)
    res = linprog(c, A_ub=a_ub, b_ub=b_ub,
                  bounds=(0, None),
                  method='highs',
                  options={'time_limit': 8.0, 'disp': False})
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(sequence):
    """
    Implements the LP step from the reference code: given a sequence `sequence`,
    find a second sequence `g` with the same convolution bound but larger
    ℓ₁‑norm, then mix a tiny fraction of `g` into `sequence`.
    """
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return None
    # Normalise to the “canonical’’ ℓ₁‑norm √(2n) used in the paper.
    norm_seq = [x * np.sqrt(2 * n) / sum_seq for x in sequence]
    rhs = np.max(np.convolve(norm_seq, norm_seq))
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    # Rescale g back to the same normalisation as the original sequence.
    norm_g = [x * np.sqrt(2 * n) / sum_g for x in g]
    # Small mixing – a larger t often improves convergence.
    t = 0.05
    new_seq = [(1 - t) * x + t * y for x, y in zip(sequence, norm_g)]
    return new_seq

    """
    Simple local optimiser:
    repeatedly identify the index pair (i, k‑i) that contributes most to the
    current maximal convolution value and move a small amount of mass from i
    to another index where the partner value for that k is small.
    The move is kept only if it decreases 2·n·max(conv).
    """
    a = np.array(seq, dtype=float)
    s = a.sum()
    if s == 0:
        a[0] = 1.0
        s = 1.0
    a /= s                     # enforce sum = 1 (scale‑invariant)
    n = len(a)
    best_score = 2.0 * n * np.max(np.convolve(a, a))

    for _ in range(max_iter):
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)

        # contribution of each i to the offending convolution value
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[partner[mask]]

        i_max = int(np.argmax(contrib))

        # partner values for each position (zero where undefined)
        partner_vals = np.zeros(n)
        partner_vals[mask] = a[partner[mask]]

        # choose a destination index with the smallest partner value
        # (avoid i_max itself)
        order = np.argsort(partner_vals)
        j = None
        for cand in order:
            if cand != i_max:
                j = int(cand)
                break
        if j is None:
            j = (i_max + 1) % n

        # amount of mass to shift – keep it tiny to stay smooth
        delta = min(a[i_max] * 0.07, 0.01)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[j] += delta

        # (sum is unchanged; re‑normalise only for numerical safety)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        new_score = 2.0 * n * np.max(np.convolve(a_new, a_new))
        if new_score < best_score:
            a = a_new
            best_score = new_score
        # otherwise discard the move

    return a.tolist()


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Main entry point called by the grader.
    Performs a time‑limited stochastic search for a low‑value sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # start from the provided construction if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        if isinstance(height_sequence_1, list) and height_sequence_1:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"Initial score from height_sequence_1 : {sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # main stochastic loop
    # ------------------------------------------------------------------
    trials = 0
    while time.time() < deadline:
        # 1) draw a random length and a Dirichlet‑type vector
        n = np.random.randint(200, 4000)               # moderate size – fast conv
        alpha = 10 ** np.random.uniform(-0.8, 0.8)      # α ∈ [≈0.16, ≈6.3]
        # Gamma → Dirichlet
        g = np.random.gamma(shape=alpha, scale=1.0, size=n)
        if g.sum() == 0:
            continue
        a = g / g.sum()                               # ℓ₁‑norm = 1

        # 2) quick evaluation (scale invariant → 2·n·max(conv))
        conv = np.convolve(a, a)
        score = 2.0 * n * np.max(conv)

        trials += 1
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"[{trials}] New best = {score:.6f}  (n={n}, α={alpha:.3f})")

        # ------------------------------------------------------------------
        # occasional refinement steps
        # ------------------------------------------------------------------
        # a) gentle greedy local optimisation on promising candidates
        if score < best_score * 1.02 and np.random.rand() < 0.04:
            improved = greedy_improve(a, max_iter=150)
            new_score = 2.0 * n * np.max(np.convolve(improved, improved))
            if new_score < best_score:
                best_score = new_score
                best_seq = improved
                print(f"  Greedy improve -> {new_score:.6f}")

        # b) LP‑direction step (rare – more expensive)
        if np.random.rand() < 0.006:
            direction = get_good_direction_to_move_into(best_seq)
            if direction is not None:
                dir_score = evaluate_sequence(direction)
                if dir_score < best_score:
                    best_score = dir_score
                    best_seq = direction
                    print(f"  LP step improvement -> {dir_score:.6f}")

    print(f"Finished after {trials} samples. Best score = {best_score:.6f}")
    return best_seq
