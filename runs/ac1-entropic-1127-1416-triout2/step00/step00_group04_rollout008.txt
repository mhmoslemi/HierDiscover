"""
Simple sub‑gradient / simulated‑annealing optimizer for the
step‑function objective.

The objective after normalising the sequence to
    sum(a) = sqrt(2·n)
is exactly   max(conv(a, a)),
so we repeatedly try to lower the peak of the convolution.
The algorithm:

1. Initialise a random non‑negative vector of length n
   (or optionally start from the provided ``height_sequence_1``).
2. Normalise it so that its ℓ1‑norm equals √(2·n).
3. While time permits
   * compute the convolution and its maximum M;
   * collect the (small) active set of indices where the maximum is attained;
   * compute a load‑vector g_i = Σ_{k∈active} a_{k‑i} (i.e. the contribution of a_i
     to the active convolutions);
   * move a small amount of mass from the index with the largest load to the
     index with the smallest load (this is a descent direction for the peak);
   * accept the move if it improves the peak or, with a temperature that
     decreases to zero, if it is only slightly worse – this gives the method
     the ability to escape shallow local minima;
   * keep the best sequence ever seen.
4. Return the best sequence (as a plain Python list).

The whole routine respects the 1000‑second budget, uses only NumPy and the
provided ``evaluate_sequence`` for final verification, and never performs any
IO.
"""

import time
import math
import random
import numpy as np

def _normalise(seq: np.ndarray) -> np.ndarray:
    """Scale seq so that its sum equals sqrt(2·len(seq))."""
    target = math.sqrt(2.0 * len(seq))
    s = seq.sum()
    if s == 0:
        # avoid division by zero – just make it uniform
        seq = np.full_like(seq, 1.0 / len(seq))
        s = seq.sum()
    return seq * (target / s)


def _convolution_max(seq: np.ndarray):
    """Return (conv, max value)."""
    conv = np.convolve(seq, seq)
    return conv, conv.max()


def _load_vector(seq: np.ndarray, active):
    """
    For each index i compute
        g_i = Σ_{k∈active} a_{k‑i}    (terms out of range are ignored)
    This is the sub‑gradient of the peak with respect to a_i.
    """
    n = len(seq)
    g = np.zeros(n, dtype=float)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        g[i_range] += seq[j_range]
    return g


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the caller will give us up to 1000).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    # ------------------------------------------------------------------
    # Setup / initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + float(budget_s) - 0.5  # 0.5 s safety margin

    # Decide on the length.  Use the known construction if it exists,
    # otherwise pick a moderate size (4000‑6000) which gives a good trade‑off.
    if 'height_sequence_1' in globals() and np.random.rand() < 0.5:
        base = np.array(list(globals()['height_sequence_1']), dtype=float)
        n = len(base)
        a = _normalise(base)
    else:
        n = np.random.randint(4000, 7000)  # fairly large, still fast enough
        a = np.random.rand(n).astype(float)
        a = _normalise(a)

    # Keep the best sequence seen so far
    conv, cur_max = _convolution_max(a)
    best_seq = a.copy()
    best_val = cur_max
    print(f"[init] n={n}, initial max={best_val:.6f}")

    # Simulated‑annealing schedule
    MAX_TEMP = 0.02           # initial temperature (absolute change allowed)
    iter_no_improve = 0

    iteration = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Identify active set (indices where the current peak is attained)
        # ------------------------------------------------------------------
        tol = 1e-12 * cur_max  # tiny tolerance to catch ties
        active = np.where(conv >= cur_max - tol)[0]

        # ------------------------------------------------------------------
        # Compute load vector (sub‑gradient of the peak)
        # ------------------------------------------------------------------
        g = _load_vector(a, active)

        # Indices with highest / lowest load
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        # ------------------------------------------------------------------
        # Propose a move: shift a small amount of mass from i_max to i_min
        # ------------------------------------------------------------------
        # Temperature shrinks linearly with remaining time
        remaining = max(0.0, deadline - time.time())
        temp = MAX_TEMP * (remaining / float(budget_s))

        step_factor = 0.1 / math.sqrt(iteration + 1.0)   # ≈0.1,0.07,0.05,...

        delta = min(delta, a[i_max])

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta

        # (the scaling is tiny and does not affect the descent direction)
        a_new = _normalise(a_new)

        # Evaluate the candidate (peak of the convolution)
        # ------------------------------------------------------------------
        conv_new, new_max = _convolution_max(a_new)

        accept = False

        elif new_max - cur_max < temp:         # allow a slight worsening
            accept = True

            iter_no_improve = 0
        else:
            iter_no_improve += 1

        # ------------------------------------------------------------------
        # Record the best ever seen
        # ------------------------------------------------------------------
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            print(f"[iter {iteration:7d}] new best max={best_val:.6f}")

        # ------------------------------------------------------------------
        # Occasionally inject a small random perturbation to escape stagnation
        # ------------------------------------------------------------------
        if iteration % 1500 == 0 and iteration > 0:
            # Add a tiny Gaussian noise, then re‑normalise
            noise = np.random.randn(n) * (np.sqrt(2.0 * n) * 0.001)
            a = np.maximum(a + noise, 0.0)
            a = _normalise(a)
            conv, cur_max = _convolution_max(a)
            print(f"[perturb] iter {iteration:7d}, max={cur_max:.6f}")

        # ------------------------------------------------------------------
        # If we have gone a long time without improvement, restart randomly
        # ------------------------------------------------------------------
        if iter_no_improve > 20000:
            a = np.random.rand(n).astype(float)
            a = _normalise(a)
            conv, cur_max = _convolution_max(a)
            iter_no_improve = 0
            print(f"[restart] iter {iteration:7d}, max reset to {cur_max:.6f}")

        iteration += 1

    # ----------------------------------------------------------------------
    # Final sanity check – run the official evaluator on the best sequence
    # ----------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best max (internal)={best_val:.6f}, evaluate_sequence={final_score:.6f}")

    return best_seq.tolist()
