"""
Propose a low‑score step‑function.

The evaluation function that we have to minimise is

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

The score is scale‑invariant, therefore only the *shape* of the
non‑negative sequence matters.  Empirically a gently decaying exponential
profile a_i ≈ exp(‑α i/n) beats the uniform (flat) sequence (which gives
exactly 2.0).  The code

* scans a few lengths (200 … 2000) and several families of parametrised
  shapes (exponential, power‑law and a product of both);
* keeps the best sequence found;
* refines it with a simple stochastic hill‑climber that moves a small
  amount of mass from one index to another;
* finally performs a few LP‑based improvement steps (the same LP that
  the reference implementation uses) with a full step size
  (t = 1.0) – this usually reduces the bound further.

All work is done inside the time budget (default 1000 s).  The returned
list is a plain Python list of floats, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """
    Solve   max  sum(b)   s.t.   conv(f_seq, b) <= rhs,   b >= 0
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f_seq)
    # objective: maximize sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # build inequality matrix for each convolution index
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv index k = sum_i f_i * b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity: -b_i <= 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # linear programme – we give it a modest time limit so the whole
    # optimisation stays inside the global budget.
    opts = {"time_limit": time_limit, "disp": False}
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, bounds=(0, None), method="highs", options=opts)
    if res.success:
        return res.x
    return None


def generate_exponential(n, alpha):
    """a_i = exp(-alpha * i / n)"""
    beta = alpha / n
    a = np.exp(-beta * np.arange(n))
    # keep within the evaluator's 0‑1000 bound
    if a.max() > 1000.0:
        a = a * (999.0 / a.max())
    return a.tolist()


def generate_powerlaw(n, p):
    """a_i = (i+1)^{-p}"""
    i = np.arange(1, n + 1)
    a = 1.0 / (i ** p)
    if a.max() > 1000.0:
        a = a * (999.0 / a.max())
    return a.tolist()


def generate_mixed(n, alpha, p):
    """a_i = exp(-alpha*i/n) * (i+1)^{-p}"""
    beta = alpha / n
    i = np.arange(n)
    a = np.exp(-beta * i) * ((i + 1) ** -p)
    if a.max() > 1000.0:
        a = a * (999.0 / a.max())
    return a.tolist()


def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little margin for returning

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1)  parameter sweep over a few families and a few lengths
    # ------------------------------------------------------------------
    lengths = [200, 500, 1000, 1500, 2000]
    alphas = np.linspace(0.2, 3.0, 28)          # for exponential
    powers  = np.linspace(0.1, 2.0, 20)         # for power‑law
    mixed_params = [(a, p) for a in np.linspace(0.5, 2.5, 10)
                              for p in (0.5, 1.0, 1.5)]

    for n in lengths:
        # uniform (baseline)
        seq = [1.0] * n
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score, best_seq = score, seq[:]

        # exponential family
        for alpha in alphas:
            seq = generate_exponential(n, alpha)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score, best_seq = score, seq[:]

        # power‑law family
        for p in powers:
            seq = generate_powerlaw(n, p)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score, best_seq = score, seq[:]

        # mixed family (exponential * power‑law)
        for alpha, p in mixed_params:
            seq = generate_mixed(n, alpha, p)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score, best_seq = score, seq[:]

    # ------------------------------------------------------------------
    # 2)  simple stochastic hill‑climber starting from the best sequence
    # ------------------------------------------------------------------
    curr_seq = best_seq[:]
    n = len(curr_seq)
    while time.time() < deadline:
        # pick two indices, move a tiny amount of mass from i to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount proportional to the source value (but limited)
        eps = np.random.normal(scale=0.02) * curr_seq[i]
        new_i = max(0.0, curr_seq[i] - eps)
        new_j = max(0.0, curr_seq[j] + eps)
        new_seq = curr_seq[:]
        new_seq[i] = new_i
        new_seq[j] = new_j
        # evaluate
        score = evaluate_sequence(new_seq)
        if score < best_score:
            best_score, best_seq = score, new_seq[:]
            curr_seq = new_seq[:]
        else:
            # occasional random walk to escape plateaus
            if np.random.rand() < 0.005:
                curr_seq = new_seq[:]

    # ------------------------------------------------------------------
    # 3)  LP‑based refinement (few iterations only – each LP is cheap)
    # ------------------------------------------------------------------
    for _ in range(5):
        if time.time() >= deadline:
            break
        n = len(best_seq)
        s = float(np.sum(best_seq))
        if s < 1e-8:
            break
        # normalise to the standard sqrt(2n) without changing the shape
        norm = [x * math.sqrt(2.0 * n) / s for x in best_seq]
        rhs = float(np.max(np.convolve(norm, norm)))
        g = solve_convolution_lp(norm, rhs, time_limit=2.0)
        if g is None:
            break
        # rescale g to have the same total sum as the current sequence
        sum_g = float(np.sum(g))
        if sum_g == 0:
            break
        factor = s / sum_g
        new_seq = [float(x) * factor for x in g]
        # keep the best
        score = evaluate_sequence(new_seq)
        if score < best_score:
            best_score, best_seq = score, new_seq[:]
        else:
            break   # no progress – stop LP loop

    print(f"Best score found: {best_score:.6f}  (length {len(best_seq)})")
    return best_seq
