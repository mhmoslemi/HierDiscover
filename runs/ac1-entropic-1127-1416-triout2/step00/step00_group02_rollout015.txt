import numpy as np, time.

def evaluate_score(seq):
    b = np.convolve(seq, seq)
    maxb = np.max(b)
    n = len(seq)
    sum_a = np.sum(seq)
    return 2.0 * n * maxb / (sum_a * sum_a)

def softmax_weights(b, beta):
    maxb = np.max(b)
    exp_terms = np.exp(beta * (b - maxb))
    w = exp_terms / np.sum(exp_terms)
    return w

def project_to_simplex(v):
    v = np.maximum(v, 0)
    total = v.sum()
    if total > 0:
        v = v / total
    else:
        # random fallback
        v = np.random.rand(len(v))
        v = v / v.sum()
    return v

def gradient_step(a, beta, eta):
    n = len(a)
    b = np.convolve(a, a)
    w = softmax_weights(b, beta)
    a_rev = a[::-1]
    g_full = np.convolve(w, a_rev)
    g = 2.0 * g_full[n-1:n-1+n]
    a_new = a - eta * g
    a_new = project_to_simplex(a_new)
    return a_new, b, maxb

    best_seq = None
    best_score = np.inf
    for seq in candidates:

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # Print initial

def gradient_step(a, eta):
    n = len(a)
    b = np.convolve(a, a)
    maxb = b.max()
    mask = (b == maxb)  # exact equality; may be multiple positions.
    # If multiple indices have identical max due to symmetry, set weights uniform.
    num = mask.sum()
    w = np.zeros_like(b)
    if num > 0:

    # compute gradient: 2 * conv(w, a_rev) slice
    a_rev = a[::-1]
    g_full = np.convolve(w, a_rev)

    a_new = a - eta * grad

        # random reinitialize
        a_new = np.random.rand(n)
        a_new /= a_new.sum()
    return a_new, b, maxb

"""
Simple sub‑gradient search for a step‑function with a small
max‑convolution value.

The evaluation metric is  
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

which is invariant under scaling of `a`.  Hence we keep `a`
normalized to a probability vector (sum = 1) and minimise
`max(conv(a, a))`.  The algorithm uses a sub‑gradient of the
pointwise maximum of the convolution: at each iteration we find
the shift(s) where the convolution attains its current maximum,
use these as the active set and compute the gradient
∂/∂a_i Σ_{k∈active} conv_k = 2 * Σ_{k∈active} a_{k-i}.
A small step is taken opposite to this gradient and the vector
is projected back onto the simplex (non‑negative, sum = 1).
The method starts from several diverse initialisations
(uniform, random Dirichlet, power‑law, arcsine‑shaped) and keeps
the best vector seen during the 1000 s budget.

The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total > 0.0:
        return v / total
    # fallback – uniform random point on the simplex
    v = np.random.rand(v.shape[0])
    return v / v.sum()


def _max_conv(a: np.ndarray) -> float:
    """Return max_k (a * a)[k] for a non‑negative vector a (sum = 1)."""
    b = np.convolve(a, a)
    return float(b.max())


def _initial_vectors(n: int):
    """Yield a few diverse initial probability vectors of length n."""
    # 1. uniform
    yield np.ones(n) / n

    # 2. random Dirichlet (moderately sparse)
    yield np.random.dirichlet(np.ones(n) * 0.3)

    # 3. power‑law a_i ∝ 1/√(i+1)  (α = 0.5)
    p = 1.0 / np.sqrt(np.arange(n) + 1.0)
    yield p / p.sum()

    # 4. arcsine‑shaped  a_i ∝ 1/√[(i+½)(n‑i‑½)]
    i = np.arange(n)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    yield a / a.sum()

    # 5. a slightly perturbed uniform (helps breaking symmetry)
    u = np.ones(n) / n
    u[0] = u[0] + 0.01
    u = _project_to_simplex(u)
    yield u


def _score_from_vector(a: np.ndarray) -> float:
    """Exact score as used by evaluate_sequence (sum(a) == 1)."""
    n = a.shape[0]
    return 2.0 * n * _max_conv(a)


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, eta: float) -> (np.ndarray, float):
    """
    Perform one sub‑gradient step on the max‑convolution objective.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum == 1, non‑negative).
    eta : float
        Step size.

    Returns
    -------
    a_new : np.ndarray
        Updated probability vector.
    maxb : float
        New maximum convolution value after the step.
    """
    n = a.shape[0]

    # 1) compute convolution and locate the active (max) indices
    conv = np.convolve(a, a)
    maxb = float(conv.max())
    # treat all indices whose value is within a tiny tolerance of the max
    eps = maxb * 1e-12 + 1e-15
    active = (conv >= maxb - eps).astype(float)
    if active.sum() == 0:                     # should never happen
        active[conv.argmax()] = 1.0
    active /= active.sum()                    # equal weight on the active set

    # 2) sub‑gradient of the max‑convolution w.r.t. a:
    #    ∂/∂a_i Σ_{k∈active} (a*a)_k = 2 * Σ_{k} active_k * a_{k-i}
    a_rev = a[::-1]                           # reversed a
    grad_full = np.convolve(active, a_rev)    # length 3n‑2
    grad = 2.0 * grad_full[n - 1: n - 1 + n]   # slice of length n

    # 3) take a step opposite to the gradient and project back to simplex
    a_new = a - eta * grad
    a_new = _project_to_simplex(a_new)

    # 4) recompute max‑convolution for the returned value
    new_maxb = _max_conv(a_new)
    return a_new, new_maxb
