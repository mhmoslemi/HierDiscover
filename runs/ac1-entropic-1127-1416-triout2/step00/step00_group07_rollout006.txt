def project_simplex(v, S=1.0):
    # similar to algorithm from [Duchi et al. 2008]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

This ensures sum(w) = S.

def normalize_nonneg(a, target_sum):

    total = np.sum(a)
    if total == 0:
        # fallback uniform distribution
        a = np.full_like(a, target_sum / len(a))
        return a
    a = a * (target_sum / total)

import time
import math

def compute_conv_fft(a, L):
    fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    return conv[:2*len(a)-1]

def normalize_vector(v, target_sum=1.0):
    v = np.maximum(v, 0.0)
    total = np.sum(v)
    if total <= 0:
        # fallback uniform
        v = np.full_like(v, target_sum / v.shape[0])
    else:
        v = v * (target_sum / total)
    return v

    Randomized local search for a non‑negative step‑function sequence that

    most to the current maximum convolution value.  A simple simulated‑annealing
    acceptance rule together with occasional gradient‑like steps is used to
    escape local minima.  The whole optimisation runs until the time budget
    expires and returns the best sequence discovered.

        idx = int(np.argmax(conv))
        return conv[idx], idx, conv
    # Evaluate current state

import numpy as np
import time
import math

def _fft_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * len(a) - 1]


def _project_to_simplex(v: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """Simple scaling projection: clip negatives and renormalise to the required sum."""
    v = np.maximum(v, 0.0)
    total = np.sum(v)
    if total == 0.0:
        # fall back to a uniform distribution
        v = np.full_like(v, target_sum / v.size)
    else:
        v = v * (target_sum / total)
    return v


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Randomised local search for a non‑negative step‑function sequence that
    minimises ``evaluate_sequence``.  The algorithm starts from a random
    probability vector (unit sum) and repeatedly:
      * computes the full self‑convolution,
      * identifies the index where the convolution attains its maximum,
      * selects a source index proportionally to its contribution to that peak,
      * moves a small fraction of its mass to a random destination,
      * accepts the move according to a simple simulated‑annealing schedule,
      * occasionally performs a cheap gradient‑like step.
    The best sequence seen before the time budget expires is returned.
    """
    np.random.seed(seed)

    # -------------------------------------------------
    # 1.  Determine the length of the sequence to work with.
    # -------------------------------------------------
    try:
        n = len(height_sequence_1)       # use the supplied construction if present
    except NameError:                   # fall back to a reasonable default
        n = 2000

    # -------------------------------------------------
    # 2.  Initialise a random positive vector with unit sum.
    # -------------------------------------------------
    a = np.random.rand(n) + 1e-6
    a = _project_to_simplex(a, target_sum=1.0)

    # -------------------------------------------------
    # 3.  Pre‑compute an FFT length (next power of two).
    # -------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # FFT size >= 2*n‑1

    # -------------------------------------------------
    # Helper to obtain max convolution value and its index.
    # -------------------------------------------------
    def _max_convolution(vec):
        conv = _fft_convolution(vec, L)
        idx = int(np.argmax(conv))
        return conv[idx], idx, conv

    # initial evaluation
    B, k_max, _ = _max_convolution(a)
    cur_val = 2.0 * n * B / (np.sum(a) ** 2)

    best_a = a.copy()
    best_val = cur_val

    # -------------------------------------------------
    # 4.  Simulated‑annealing parameters.
    # -------------------------------------------------
    max_temp = 0.05                     # temperature at the start of the run
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # -------------------------------------------------
        # (a)  recompute the convolution and the current score
        # -------------------------------------------------
        B, k_max, conv = _max_convolution(a)
        cur_val = 2.0 * n * B / (np.sum(a) ** 2)

        # -------------------------------------------------
        # (b)  Targeted move: shift mass away from indices that
        #      contribute most to the current peak.
        # -------------------------------------------------
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)

        contribution = np.zeros_like(a)
        contribution[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        total_contrib = np.sum(contribution)

        if total_contrib > 0:
            probs = contribution / total_contrib
            src = np.random.choice(n, p=probs)                # source index
        else:
            src = np.random.randint(n)

        # choose a destination different from src
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # move a modest fraction of the source mass
        eps_factor = 0.10                     # up to 10 % of the source entry
        eps = eps_factor * a[src]
        if eps <= 0.0:
            # nothing to move – try the next iteration
            it += 1
            continue

        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = _project_to_simplex(cand, target_sum=1.0)

        # evaluate candidate
        Bc, _, _ = _max_convolution(cand)
        cand_val = 2.0 * n * Bc / (np.sum(cand) ** 2)

        # -------------------------------------------------
        # (c)  Acceptance rule (simulated annealing)
        # -------------------------------------------------
        remaining = max(deadline - time.time(), 0.0)
        temperature = max_temp * remaining / budget_s   # linear cooling

        accept = False
        if cand_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cand_val - cur_val) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:

            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()

            else:
                no_improve += 1

            no_improve += 1

        # -------------------------------------------------
        if it % 20 == 0:

            a += 1e-4 * np.random.rand(n)
            a = _project_to_simplex(a, target_sum=1.0)
            no_improve = 0

        it += 1

    # -------------------------------------------------
    # 5. Return the best sequence discovered.
    # -------------------------------------------------
    print(f"Finished optimisation after {it} iterations, best value {best_val:.6f}")
    return best_a.tolist()
