```python
def max_conv(seq):
    n = len(seq)
    # next power of 2 at least 2*n
    size = 1 << (2*n-1).bit_length()
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    # conv length = size, but we just consider first 2n-1 entries
    return np.max(conv[:2*n-1])
```

```
def proj_simplex(v):
    # ensures sum of v = 1 and non-negative
    # Code from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np
import time
import math

def proj_simplex(v):
    """
    Projects a vector v onto the probability simplex {x >= 0, sum x = 1}.
    """
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho where u_i > (cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero, return uniform zero vector (should not happen)
        return np.maximum(v, 0)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def max_conv_fft(seq):
    """
    Returns (max_value, argmax_index) of the linear convolution of seq with itself.
    Uses FFT for speed. seq is assumed 1‑D numpy array.
    """
    n = seq.shape[0]
    # Choose a size that is a power of two and at least 2*n-1
    size = 1 << ((2*n - 2).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)
    conv = conv[:2*n-1]  # keep only the valid part
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx

def score_from_max(n, max_val):
    """
    The evaluate_sequence value when the sum of the coefficients is 1.
    """
    return 2.0 * n * max_val

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simulated‑annealing / projected sub‑gradient search for a non‑negative
    step‑function that minimises the provided evaluate_sequence metric.
    The search works on a probability simplex (sum == 1) – scaling does not
    affect the objective – and repeatedly moves mass away from indices that
    contribute most to the current maximal convolution entry.
    """
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a small margin

    # Length of the sequence – use the known construction length when possible
    n = len(globals().get("height_sequence_1", []))
    if n == 0:
        n = 2000  # fallback length

    best_seq = None
    best_score = float('inf')

    # Parameters for the outer restart loop
    n_restarts = 4
    max_iters_per_restart = 15000
    initial_step = 0.5
    min_step = 1e-5
    initial_temp = 0.02   # temperature for occasional uphill moves

    for restart in range(n_restarts):
        # Start from a random point on the simplex (Dirichlet(1,…,1))
        seq = np.random.dirichlet(np.ones(n))
        # Evaluate current score
        cur_max, cur_k = max_conv_fft(seq)
        cur_score = score_from_max(n, cur_max)

        step = initial_step
        it = 0

        while time.time() < deadline and it < max_iters_per_restart:
            # Current convolution statistics
            cur_max, k_max = max_conv_fft(seq)
            cur_score = score_from_max(n, cur_max)

            # Build a sub‑gradient for the active index k_max
            # g_i = 2 * seq[k_max - i] for all i where the index is valid
            idx = np.arange(n)
            j = k_max - idx
            valid = (j >= 0) & (j < n)
            g = np.zeros_like(seq)
            # Use 2*seq[j] for all valid pairs; the diagonal case (i==j) is also covered
            g[valid] = 2.0 * seq[j[valid]]

            # Gradient step
            v = seq - step * g
            new_seq = proj_simplex(v)

            # Energy after the step
            new_max, _ = max_conv_fft(new_seq)
            new_score = score_from_max(n, new_max)

            # Simulated‑annealing acceptance
            remaining = max(0.0, deadline - time.time())

            accept = False
            if new_score < cur_score:
                accept = True
            else:

                if temp > 1e-9 and np.random.rand() < math.exp(-(new_score - cur_score) / temp):
                    accept = True

                seq = new_seq
                cur_score = new_score

                step = max(step * 0.99, min_step)
            else:
                # If we rejected, shrink the step more aggressively to explore locally
                step = max(step * 0.90, min_step)

            # Keep the best solution seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] New best: {best_score:.6f} (restart {restart}, iter {it})")

            # Occasionally perform a small random perturbation to escape flat regions
            if np.random.rand() < 0.002:

                    seq[j] += delta

    # Fallback – uniform distribution if nothing was found (should not happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Return as a plain Python list so the caller can feed it to evaluate_sequence directly
    return best_seq.tolist()

"""
Propose a sequence of step‑function heights that (hopefully) beats the
uniform bound (score ≈ 2.0).  The algorithm works on the probability simplex
(sum = 1 – scaling does not change the objective) and performs a projected
sub‑gradient / simulated‑annealing search:

* start from several random Dirichlet points,
* repeatedly identify the convolution index k where the self‑convolution is
  maximal,
* move probability mass away from indices that contribute heavily to that
  peak (gradient g_i ≈ 2·a_{k‑i}),
* project back onto the simplex,
* accept the move if it improves the objective or with a small
  temperature‑dependent probability,
* keep the best sequence found.

Fast convolution is computed via FFT, and the projection onto the simplex
uses the O(n) algorithm of Duchi et al.  The whole search is bounded by the
given time budget (default 1000 s).  The returned list can be fed directly to
``evaluate_sequence``.
"""

import numpy as np
import time
import math

def _proj_simplex(v):
    """Project vector v onto the probability simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Should never happen with our usage, but guard against it.
        return np.maximum(v, 0.0)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv_fft(seq):
    """
    Compute max value of the linear convolution seq * seq and its index.
    Uses FFT for O(n log n) speed; seq is a 1‑D numpy array.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    conv = conv[: 2 * n - 1]                  # keep only the valid part
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _score_from_max(n, max_val):
    """
    With sum(seq) == 1 the evaluation score reduces to 2·n·max(conv).
    """
    return 2.0 * n * max_val


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The search runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # small safety margin

    # Use the length of the known construction when available, otherwise fall back.
    n = len(globals().get("height_sequence_1", []))
    if n == 0:
        n = 2000  # reasonable default if the global is missing

    best_seq = None
    best_score = float("inf")

    # -------- hyper‑parameters ----------
    n_restarts = 4                # independent random starts
    max_iters = 15000            # per restart
    init_step = 0.5
    min_step = 1e-5
    init_temp = 0.02             # temperature for occasional uphill moves
    # ------------------------------------

    for restart in range(n_restarts):
        # Random point on the simplex (Dirichlet with all parameters = 1)
        seq = np.random.dirichlet(np.ones(n))
        cur_max, cur_k = _max_conv_fft(seq)
        cur_score = _score_from_max(n, cur_max)

        step = init_step
        it = 0

        while time.time() < deadline and it < max_iters:
            cur_max, k_max = _max_conv_fft(seq)
            cur_score = _score_from_max(n, cur_max)

            # ---- sub‑gradient of the active convolution entry -----------------
            # For each i, the derivative of conv[k_max] w.r.t a_i is
            #   2 * a_{k_max - i}   (if the index is inside the array)
            # (the diagonal term i = k_max-i is also covered by the same formula)
            idx = np.arange(n)
            partner = k_max - idx
            valid = (partner >= 0) & (partner < n)
            g = np.zeros_like(seq)
            g[valid] = 2.0 * seq[partner[valid]]
            # ------------------------------------------------------------------

            # Gradient step and projection back onto the simplex
            v = seq - step * g
            new_seq = _proj_simplex(v)

            # New objective
            new_max, _ = _max_conv_fft(new_seq)
            new_score = _score_from_max(n, new_max)

            remaining = max(0.0, deadline - time.time())

            accept = False
            if new_score < cur_score:
                accept = True

    return best_seq.tolist()
