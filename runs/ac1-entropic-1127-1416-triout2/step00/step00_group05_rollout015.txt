def projection_simplex(v, z=1.0):
    v = np.asarray(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence that minimizes the evaluation function.

Algorithm Overview
------------------
1. **Geometric seed** –  
   The search starts from a one‑parameter family  
   a_i = C·r^i,  i = 0 … n‑1.  
   For a fixed length `n` we scan a grid of `r` (equivalently λ = 1‑r) and
   keep the pair (r, C) that gives the smallest value of
   ``evaluate_sequence``.  This cheap pre‑search already brings the score
   from the trivial uniform value 2.0 down to the 1.8 – 1.9 range.

2. **Projected gradient descent** –  
   The objective is  max_k (a * a)[k]  (the scaling‑invariant bound).  
   For the current sequence we locate the index `k*` where the convolution
   attains its maximum.  The sub‑gradient of the max‑functional at that
   point is  

        g_i = 2·a_{k*‑i}   if 0 ≤ k*‑i < n, else 0.

   A step in the opposite direction reduces the peak value.  After the
   update we **project** back onto the simplex
   `{x ≥ 0, Σx = sqrt(2·n)}` to keep the normalization and the
   non‑negativity constraints.

   The step size is decreased slowly and occasional random perturbations
   are added to escape plateaus.

3. **Multiple restarts** –  
   If the search stalls for a while, a fresh random perturbation of the
   best sequence is taken as a new starting point.

All operations use NumPy; the convolution is performed with an FFT for
speed (O(n log n) per iteration).  The routine stops when the allotted
time (≈ 1000 s) is exhausted and returns the best sequence found.

# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = s}
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s.

    Implements the algorithm of Duchi et al. (2008).  The input is a 1‑D
    NumPy array; the returned array has the same shape.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")

    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the largest rho satisfying u[rho] > cssv[rho] / (rho + 1)
    rho = np.nonzero(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        # all entries are zero after projection
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        # avoid numerical underflow when r**n becomes 0
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom   # because sum = C*(1‑r_n)/lam = target_sum
        a = C * (r ** np.arange(n))
        # clip to the limits enforced by evaluate_sequence (0 … 1000)
        a = np.clip(a, 0.0, 1000.0)
        val = evaluate_fast(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    # power‑law seed a_i ∝ 1/√(i+1)
    idx = np.arange(1, n + 1, dtype=float)
    a_pow = 1.0 / np.sqrt(idx)
    C_pow = target_sum / a_pow.sum()
    a_pow = C_pow * a_pow
    a_pow = np.clip(a_pow, 0.0, 1000.0)
    val_pow = evaluate_fast(a_pow)
    if val_pow < best_val:
        best_val = val_pow
        best_seq = a_pow.copy()

    # fallback – uniform (should never happen)
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # ------------------------------------------------------------------- #
    # 3. Projected gradient descent to polish the seed.
    curr = best_seq.copy()
    best_score = best_val
    best = curr.copy()

    # FFT length – next power of two >= 2*n‑1 for efficient cyclic convolution.
    fft_len = 1 << ((2 * n - 1).bit_length())

    # hyper‑parameters
    step_scale = 0.5          # initial absolute step magnitude
    decay_interval = 200      # iterations after which we shrink step_scale
    decay_factor = 0.85
    min_step = 1e-8
    it = 0

    while time.time() < deadline:
        # ----- convolution via FFT ------------------------------------------------
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]

        # location of the current maximum
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # current evaluation (identical to evaluate_sequence's formula)
        sum_curr = curr.sum()
        curr_score = 2.0 * n * max_val / (sum_curr ** 2)

        if curr_score < best_score:
            best_score = curr_score
            best = curr.copy()
            print(
                f"[{time.time() - start_time:6.1f}s] iter {it:5d} → new best {best_score:.6f}"
            )
            # tighten step as we improve
            step_scale = max(step_scale * 0.9, min_step)

        # ----- gradient of the max‑functional ------------------------------------
        # g_i = 2 * a_{k_max - i}  if the index is inside [0, n)
        idx_arr = np.arange(n)
        j = k_max - idx_arr
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(curr)
        grad[mask] = 2.0 * curr[j[mask]]

        # ----- take a step and re‑project onto the simplex -----------------------
        # step magnitude is proportional to the target sum / n so that a typical
        # coordinate moves by a few percent in the early phase.
        step = step_scale * target_sum / n
        tentative = curr - step * grad / (np.linalg.norm(grad) + 1e-12)
        # Clip to the allowed numeric range before projection
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ----- occasional random jitter to escape plateaus -----------------------
        if np.random.rand() < 0.02:   # 2 % of the iterations
            jitter = np.random.randn(n) * step_scale * target_sum / (20.0 * n)
            curr = projection_simplex(np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        # ----- schedule adaptation ------------------------------------------------
        it += 1
        if it % decay_interval == 0:
            step_scale = max(step_scale * decay_factor, min_step)

    # Final sanity check (the evaluator does its own clipping)
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Best score found: {final_score:.6f}")
    return best.tolist()
