def compute_score_fft(a):
    n = a.shape[0]
    # Length for convolution: 2n-1
    size = 1 << ((2*n-1).bit_length())
    A_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A_fft * A_fft, n=size)
    conv = conv[:2*n-1]   # trim
    max_b = np.max(conv)
    sum_a = np.sum(a)
    return 2.0 * n * max_b / (sum_a * sum_a)
```

def compute_score(a):
    nloc = a.shape[0]
    size = 1 << ((2*nloc - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*nloc-1]
    maxb = conv.max()
    s = a.sum()
    if s == 0:
        return np.inf
    return 2.0 * nloc * maxb / (s*s)

"""
propose_candidate
-----------------

A stochastic optimisation that searches for a step‑function (a non‑negative
sequence) with a small value of

    2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the discrete convolution used in ``evaluate_sequence``.
The algorithm works in three stages:

  1.  Generate a few analytic families (uniform, exponential, power‑law,
      linear) and keep the best one as a starting point.
  2.  Perform a targeted hill‑climbing / simulated‑annealing loop.
      In each iteration the convolution is computed (via FFT), the bin
      attaining the current maximum is identified and the two indices that
      contribute most and least to this bin are re‑balanced by moving a
      small amount of mass from the former to the latter.
      Moves that improve the score are always accepted; occasionally a
      worsening move is accepted according to a temperature schedule to
      escape shallow local minima.
  3.  Return the best sequence found (scaled arbitrarily – the objective is
      scale invariant).

The routine respects all problem constraints (non‑negative entries,
finite values) and runs within the supplied time budget (default 1000 s).
"""

import time
import numpy as np

def _fft_conv(a: np.ndarray) -> np.ndarray:
    """Convolution a*a using a power‑of‑two FFT (real inputs)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=size)
    b = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return b


def _score(a: np.ndarray) -> float:
    """Return the value of the evaluation function (no safety checks)."""
    n = a.shape[0]
    b = _fft_conv(a)
    max_b = np.max(b)
    s = np.sum(a)
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _candidate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _candidate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (lam > 0)."""
    w = np.exp(-lam * np.arange(n))
    w /= w.sum()
    return w


def _candidate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^-p  (p ≥ 0)."""
    w = 1.0 / np.power(np.arange(1, n + 1), p)
    w /= w.sum()
    return w


def _candidate_linear(n: int) -> np.ndarray:
    """a_i ∝ (n-i)  (decreasing linear shape)."""
    w = np.arange(n, 0, -1, dtype=float)
    w /= w.sum()
    return w

    Search for a sequence that minimises the evaluation function.
    The returned list will be scored by the external ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # keep a little margin

    # ------------------------------------------------------------------
    # Determine length of the problem (use the supplied construction if
    # available, otherwise fall back to a moderate size).
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:

    # ------------------------------------------------------------------
    # 1) Analytic families – keep the best as starting point.
    # ------------------------------------------------------------------
    best_a = _candidate_uniform(n)
    best_val = _score(best_a)

    families = [
        ("uniform", best_a),
        ("linear", _candidate_linear(n)),
        ("exp_0.5n", _candidate_exponential(n, lam=0.5 / n)),
        ("exp_1n", _candidate_exponential(n, lam=1.0 / n)),
        ("exp_2n", _candidate_exponential(n, lam=2.0 / n)),
        ("exp_5n", _candidate_exponential(n, lam=5.0 / n)),
        ("pow_0.2", _candidate_powerlaw(n, p=0.2)),
        ("pow_0.5", _candidate_powerlaw(n, p=0.5)),
        ("pow_0.8", _candidate_powerlaw(n, p=0.8)),
        ("pow_1.0", _candidate_powerlaw(n, p=1.0)),
        ("pow_1.5", _candidate_powerlaw(n, p=1.5)),
    ]

    for name, cand in families:

            # informative output – will be shown in the log

            if val < best_val:

                print(f"Existing sequence improved initial score to {best_val:.6f}")
        except Exception:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val

    n = a.shape[0]
    # temperature influences acceptance of worse moves
    temperature = 0.02 * cur_val

    iter_cnt = 0

    idx_all = np.arange(n)

    while time.time() < deadline:

        b = _fft_conv(a)

        src = int(np.argmax(contrib))

        # amount to move
        delta = step_fraction * a[src] * np.random.rand()

        # propose new vector
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        # numerical safety – avoid tiny negatives
        if a_new[src] < 0.0:
            a_new[src] = 0.0

        new_val = _score(a_new)

        # ----- acceptance test -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # simulated annealing step
            prob = np.exp(-(new_val - cur_val) / (temperature + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:

            cur_val = new_val
            if new_val < best_val:

                print(f"Iter {iter_cnt:7d}: new best score {best_val:.6f}")

        # occasional progress report (does not affect optimisation)
        if iter_cnt - last_print >= 5000:
            print(f"Iter {iter_cnt:7d}: current score {cur_val:.6f}")
            last_print = iter_cnt

    # ------------------------------------------------------------------
    # 3) Return the best sequence found (scale is irrelevant for the objective)
    # ------------------------------------------------------------------
    # Normalise to unit L1 for cleanliness
    best_a = best_a / best_a.sum()
    return best_a.tolist()
