def step_reduce_max(seq, step=0.1):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k_max = np.argmax(conv)
    grad = np.zeros(n)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] = seq[j]
    # optionally double for symmetric? but fine.
    # compute step size relative to current values
    delta = step * grad / (np.max(grad) + 1e-12)  # scale so max delta = step
    new_seq = seq - delta
    new_seq = np.maximum(new_seq, 0.0)
    # renormalize sum
    total = np.sum(new_seq)
    if total == 0:
        return seq
    new_seq = new_seq * (np.sum(seq) / total)
    return new_seq
```

def perturb_sequence(seq, rng, magnitude=0.05):
    n = len(seq)
    if n < 2:
        return seq
    new_seq = seq.copy()
    i, j = rng.choice(n, size=2, replace=False)
    # amount to transfer (up to magnitude fraction of a_i)
    max_transfer = new_seq[i] * magnitude
    if max_transfer <= 0:
        return new_seq
    delta = rng.random() * max_transfer
    new_seq[i] -= delta
    new_seq[j] += delta
    # apply small Gaussian noise to a random element
    k = rng.integers(n)
    noise = rng.normal(scale=magnitude * np.mean(new_seq))
    new_seq[k] = max(0.0, new_seq[k] + noise)
    # renormalize sum to original sum (scale)
    total = sum(new_seq)
    if total <= 0:
        return seq
    # desired sum = sqrt(2*n)
    target = np.sqrt(2.0 * n)
    factor = target / total
    new_seq = [x * factor for x in new_seq]
    return new_seq

"""
Propose a candidate step‑function sequence.

The algorithm is a very lightweight stochastic optimiser.
It works in three stages:

1.   Initialise a pool of candidates – either the known construction
     ``height_sequence_1`` (if present) or fresh random sequences of
     moderate length (200 – 2000).  Random sequences are created with
     several different distributions (uniform, exponential, power‑law)
     and are normalised so that  Σ a_i = sqrt(2·n) (the normalisation
     used by the reference code – the evaluation function itself is
     scale‑invariant).

2.   For every candidate a short hill‑climbing phase is performed.
     A “perturb” operation moves a small amount of mass from one
     index to another while keeping the total sum fixed (the vector is
     re‑normalised afterwards).  The new sequence is accepted only
     if it improves the evaluation score; a few hundred such moves are
     tried per candidate.

3.   The best sequence found so far is periodically perturbed again
     (larger moves, occasional Gaussian noise) to escape local minima.
     The whole loop runs until the allotted time expires and the best
     sequence is returned.

The optimiser explores many random initialisations (thousands per
minute on a single CPU) and therefore has a good chance of finding a
sequence with a convolution bound smaller than the uniform baseline
(score ≈ 2.0).  All helper functions are top‑level and contain no
closures, satisfying the competition constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Return the normalising sum sqrt(2·n) used by the reference code."""
    return np.sqrt(2.0 * n)


def _generate_random_sequence(rng: np.random.Generator, n: int) -> list:
    """
    Produce a random non‑negative sequence of length *n*.
    Several distribution families are mixed to increase diversity.
    The result is normalised so that its L1‑norm equals sqrt(2·n).
    """
    # pick a distribution type
    typ = rng.integers(0, 3)
    if typ == 0:            # uniform random on [0,1)
        vals = rng.random(n)
    elif typ == 1:          # exponential (scale=1)
        vals = rng.exponential(scale=1.0, size=n)
    else:                   # power‑law / Pareto‑like tail
        shape = rng.uniform(0.3, 3.0)          # shape parameter
        u = rng.random(n)
        # simple Pareto with minimum 1e-6 to avoid zero
        vals = (u ** (-1.0 / shape) - 1.0) + 1e-6

    # apply an extra power transform to vary the skew
    power = rng.uniform(0.5, 2.5)
    vals = vals ** power

    # avoid exact zeros (they are allowed but may hamper hill‑climbing)
    vals = np.maximum(vals, 1e-12)

    # normalise to the required sum
    target = _target_sum(n)
    vals = vals * (target / vals.sum())

    # protect against the hard limits in evaluate_sequence (0 … 1000)
    vals = np.clip(vals, 0.0, 1000.0)

    return vals.tolist()


def _perturb_sequence(seq: list, rng: np.random.Generator,
                      magnitude: float = 0.05) -> list:
    """
    Small stochastic modification of *seq*.
    - Transfer a fraction of mass from one index to another.
    - Add a tiny Gaussian noise to a third index.
    The total sum is re‑normalised to sqrt(2·len(seq)).
    """
    n = len(seq)
    if n < 2:
        return seq

    new_seq = seq.copy()

    # ---- transfer a small amount ----------
    i, j = rng.choice(n, size=2, replace=False)
    max_transfer = new_seq[i] * magnitude
    if max_transfer > 0.0:
        delta = rng.random() * max_transfer
        new_seq[i] -= delta
        new_seq[j] += delta

    # ---- add a little noise ----------
    k = rng.integers(n)
    noise = rng.normal(scale=magnitude * np.mean(new_seq))
    new_seq[k] = max(0.0, new_seq[k] + noise)

    # ---- renormalise to the prescribed L1 ----
    total = sum(new_seq)
    if total <= 0.0:
        # fallback to the original sequence (should never happen)
        return seq
    target = _target_sum(n)
    factor = target / total
    new_seq = [max(0.0, x * factor) for x in new_seq]

    # final clipping to stay inside the evaluator's limits
    new_seq = [min(1000.0, x) for x in new_seq]

    return new_seq


def _hill_climb(start_seq: list, rng: np.random.Generator,
                steps: int = 200,
                start_score: float = None) -> tuple:
    """
    Very light weight hill‑climbing: repeatedly apply a perturbation and
    keep the change only when the evaluation score improves.
    Returns the best sequence and its score.
    """
    seq = start_seq
    try:
        best_score = evaluate_sequence(seq) if start_score is None else start_score
    except Exception:
        # if the starting sequence is somehow invalid, give it a huge score
        best_score = float('inf')

    for _ in range(steps):
        cand = _perturb_sequence(seq, rng, magnitude=0.07)
        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue

        if cand_score < best_score:
            seq = cand
            best_score = cand_score
    return seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.
    The routine respects the time budget, uses random restarts,
    hill‑climbing and occasional perturbations of the current best
    solution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # initialise the best known solution (use the provided construction if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    if 'height_sequence_1' in globals():
        try:
            cand = list(height_sequence_1)          # ensure a plain list
            sc = evaluate_sequence(cand)
            if sc < best_score:
                best_seq = cand
                best_score = sc
                print(f"[init] height_sequence_1 score = {best_score:.6f}")
        except Exception:
            pass

    # If no seed was available, fall back to a modest uniform sequence
    if best_seq is None:
        n0 = 500
        uniform_val = _target_sum(n0) / n0
        best_seq = [uniform_val] * n0
        best_score = evaluate_sequence(best_seq)
        print(f"[init] uniform baseline (n={n0}) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # main stochastic optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) Random fresh start (exploration)
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            n = int(rng.integers(200, 2000))
            seq = _generate_random_sequence(rng, n)
            try:
                cur_score = evaluate_sequence(seq)
            except Exception:
                continue

            # a short hill‑climb on the new candidate
            seq, cur_score = _hill_climb(seq, rng, steps=150, start_score=cur_score)

            if cur_score < best_score:
                best_seq = seq
                best_score = cur_score
                print(f"[{iteration}] New best from scratch: score={best_score:.6f}, n={n}")

        # --------------------------------------------------------------
        # 2) Local perturbation of the current best (exploitation)
        # --------------------------------------------------------------
        else:
            cand = _perturb_sequence(best_seq, rng, magnitude=0.12)
            try:
                cand_score = evaluate_sequence(cand)
            except Exception:
                continue

            # if the perturbation is good, keep it and optionally hill‑climb
            if cand_score < best_score:
                # a brief climb to polish the improvement
                cand, cand_score = _hill_climb(cand, rng, steps=80, start_score=cand_score)
                if cand_score < best_score:
                    best_seq = cand
                    best_score = cand_score
                    print(f"[{iteration}] Improved by perturbation: score={best_score:.6f}")

        # --------------------------------------------------------------
        # occasional aggressive random restart when we seem stuck
        # --------------------------------------------------------------
        if iteration % 400 == 0 and rng.random() < 0.3:
            n = int(rng.integers(300, 1500))
            seq = _generate_random_sequence(rng, n)
            try:
                cur_score = evaluate_sequence(seq)
                seq, cur_score = _hill_climb(seq, rng, steps=200, start_score=cur_score)
                if cur_score < best_score:
                    best_seq = seq
                    best_score = cur_score
                    print(f"[{iteration}] Restart yielded better: score={best_score:.6f}, n={n}")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # final sanity check – make sure the sequence respects the limits
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: a safe uniform construction
        n = 500
        val = _target_sum(n) / n
        best_seq = [val] * n
        best_score = evaluate_sequence(best_seq)

    # ensure we never hand back a list containing NaNs / infinities
    best_seq = [float(x) if np.isfinite(x) else 0.0 for x in best_seq]

    print(f"[final] Best score found = {best_score:.6f} (length {len(best_seq)})")
    return best_seq
