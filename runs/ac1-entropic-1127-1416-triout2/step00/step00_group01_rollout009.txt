"""
Propose a low‑score step‑function by exploiting the scaling‑invariance of
evaluate_sequence.

For a sequence a_i = (i+1)^p (p ≥ 1) the convolution peak is attained at the
middle index and its value grows like n^(2p+1) while the L1‑norm grows like
n^(p+1).  After normalising the maximum coefficient to 1000 (the built‑in clamp)
the ratio 2·n·max(b)/sum(a)^2 depends **only** on the exponent p:

    score(p) = 2·(p+1)^2·B(p+1,p+1)   where   B is the Beta‑function.

For p=2 the score is 0.6 and for p=3 it is ≈0.228 — well below the target
1.503.  The algorithm simply tries a few small exponents and a few lengths,
keeps the best sequence, and returns it.  No heavy optimisation loop is
required; the method finishes far within the 1000 s budget.

All numbers are non‑negative and never exceed 1000, so the security checks in
evaluate_sequence are satisfied.
"""

import time
import numpy as np
from math import beta  # beta(x, y) = Γ(x)Γ(y) / Γ(x+y)

# ----------------------------------------------------------------------
# Helper: generate a monotone polynomial sequence of given length and exponent
def polynomial_sequence(length: int, exponent: int) -> list[float]:
    """Return a list a_i = (i+1)**exponent, normalised so max(a_i) == 1000."""
    if length <= 0:
        return []
    # raw maximum before scaling is length**exponent
    scale = 1000.0 / (float(length) ** exponent)
    # generate scaled values
    return [scale * ((i + 1) ** exponent) for i in range(length)]

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a sequence with a very low evaluate_sequence score.

    The search simply enumerates a handful of lengths and polynomial exponents,
    evaluates each candidate and returns the best one found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # a tiny safety margin

    # In case something goes wrong we fall back to the previously known
    # construction (global variable `height_sequence_1`).
    best_seq = list(globals().get("height_sequence_1", []))
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_score = float("inf")

    # Parameter grids – a few lengths (the problem already works with ~5000)
    # and a few exponents.  Exponents >=2 already beat the target.
    lengths = [500, 1000, 2000, 4000, 5000]
