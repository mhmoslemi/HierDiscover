"""
Search for a step‑function that minimises the evaluation metric.

The construction is based on the observation that for a continuous
density f(x) = (p+1)*(1-x)^p on [0,1] (p>0) the self‑convolution
has supremum  M(p) = (p+1)^2·B(p+1,p+1)  where  B  is the beta function.
After normalising the discrete sequence so that Σ a_i = √(2n) the
evaluation becomes 2·M(p), which decreases super‑exponentially with p.
Hence a simple family

    a_i ∝ (1 - i/(n‑1))^p   ,  i = 0,…,n‑1

produces scores far below the target.  The routine samples several
exponents (both integers and random reals) inside the given time
budget and returns the sequence with the lowest score found.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a decreasing power‑law sequence a_i = (1 - i/(n‑1))^p,
    scaled to have sum √(2n).  Several exponents p are tried (both
    deterministic and random) and the best sequence (lowest
    evaluate_sequence value) is returned.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Seconds allocated for the search.
    **kwargs : dict
        Ignored – kept for compatibility with the caller.

    Returns
    -------
    list[float]
        The best coefficient sequence discovered.
    """
    # Determine the length to use (prefer the provided construction)
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        n = 2000                               # safe fallback

    target_sum = float(np.sqrt(2.0 * n))

    best_seq = None
    best_score = float('inf')
    deadline = time.time() + budget_s - 0.1

    # deterministic integer exponents to start with
    int_exps = list(range(1, 12))
    np.random.seed(seed)
    np.random.shuffle(int_exps)

    # Helper to evaluate a given exponent
    def evaluate_exponent(p):
        # raw decreasing shape
        idx = np.arange(n, dtype=float)
        raw = np.power(1.0 - idx / (n - 1), p)
        raw = np.maximum(raw, 0.0)                     # safety
        sum_raw = raw.sum()
        if sum_raw == 0.0:
            return None, None
        # scale to the required L1 norm
        seq = raw * (target_sum / sum_raw)
        seq_list = seq.tolist()
        try:

        except Exception:
            return None, None
        return seq_list, score

    # Initial pass over a few integer exponents
    for p in int_exps:
