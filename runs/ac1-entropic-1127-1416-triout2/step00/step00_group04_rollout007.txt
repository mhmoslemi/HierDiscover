"""
Propose a near‑optimal step‑function for the given evaluation metric.

The idea is to concentrate the mass at the two ends of the array while
leaving at least one zero entry in the middle.  For a sequence of length
n with L entries equal to 1 at the left and at the right (gap = n‑2L ≥ 1)
the convolution achieves its maximum value exactly L, the total sum
equals 2 L, and the evaluation becomes

    2·n·L / (2 L)² = n / (2 L) = 1 + gap/(2 L)  → 1  as n grows.

Choosing L = ⌊(n‑1)/2⌋ (i.e. a single zero in the centre) gives an
evaluation of about n/(n‑1) ≈ 1.0001 for n≈7000, which is well below the
target 1.503.

The function builds such a sequence using the length of the existing
`height_sequence_1` if it is available; otherwise it defaults to a
moderate odd length (1001).  The resulting sequence is returned directly.
"""

import time

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a step sequence with two blocks of ones at the ends and a
    small gap of zeros in the middle.  This construction yields an
    evaluation close to the theoretical lower bound of 1.

    Parameters
    ----------
    seed : int
        Ignored – kept for compatibility with the runner.
    budget_s : float
        Ignored – the construction is instantaneous.
    **kwargs :
        Additional ignored arguments.

    Returns
    -------
    list[float]
        The constructed sequence.
    """
    # Determine the target length.  Prefer the length of the previously
    # discovered best sequence if it exists; otherwise fall back to a
    # reasonably sized odd length.
    n = None
    try:
        # height_sequence_1 is provided as a global variable in the
        # competition environment.
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = None

    if n is None or n < 3:
        # Use an odd length to make the gap exactly one element.
        n = 1001  # any odd number works

    # Compute block size L and the central gap.
    # We aim for a single zero gap when n is odd.
    L = (n - 1) // 2          # maximal block size while keeping a gap ≥1
    gap = n - 2 * L           # this will be 1 for odd n, 2 for even n

    # If the gap turned out to be zero (possible for even n when L is exact),
    # shrink the blocks a little to ensure a non‑empty gap.
    if gap == 0:
        L = max(1, L - 1)
        gap = n - 2 * L

    # Build the sequence: L ones, gap zeros, L ones.
    sequence = [1.0] * L + [0.0] * gap + [1.0] * L

    # Quick sanity check – evaluate and print the score.
    try:
        score = evaluate_sequence(sequence)
        print(f"Generated block sequence: n={n}, L={L}, gap={gap}, score={score:.6f}")
    except Exception as exc:
        # Should never happen, but if the evaluation crashes we still
        # return the sequence (the runner will then treat it as invalid).
        print(f"Evaluation failed: {exc}")

    return sequence
