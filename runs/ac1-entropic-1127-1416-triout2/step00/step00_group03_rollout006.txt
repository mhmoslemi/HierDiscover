    n = len(f)
    c = -np.ones(n)  # maximize sum(b)
    # Build constraint matrix A_ub shape (2n-1 + n) x n
    # Use list of rows for efficiency.
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        # compute contributions
        # i from max(0, k-(n-1)) to min(n-1,k)
        lo = max(0, k-(n-1))

"""
Propose a sequence of non‑negative heights that (hopefully) beats the current
bound of 2.0 for the evaluation function.

Algorithm summary
-----------------
1.  Start from the best known construction (`height_sequence_1`) if it exists,
    otherwise from a random sequence.
2.  Repeatedly try to *increase the total sum* of the sequence while keeping the
    maximal convolution value unchanged.  This is done by solving the linear
    program

        maximise   Σ b_j
        subject to conv(f, b) ≤ M   (M = max(conv(f, f)))
                    b_j ≥ 0

   where `f` is the current sequence.  The solution `g` is mixed with `f`
   using a relatively large step size `t`.  Because the convolution bound of
   `f` and `g` is the same, the mixed sequence does not increase the maximum
   convolution much, but its total sum grows, which reduces the evaluation
   value `2·n·max(conv) / (Σ a)²`.

3.  If the LP fails, perform a cheap stochastic perturbation (add a small
   Gaussian bump to a random coordinate) and keep the change if it improves
   the score.

4.  Occasionally restart from a fresh random sequence of a random length
   (200–1000) to avoid getting stuck in a local optimum.

5.  Keep the best sequence seen so far and return it when the time budget is
   exhausted.

The method combines the previously published LP‑direction with a more
aggressive mixing factor and random restarts, and it stays well within the
1000 s time limit on a single CPU.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: linear program that maximises the sum of a non‑negative vector `b`
# under the convolution constraint conv(f, b) ≤ M.
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, max_conv):
    """
    Solve:
        maximise   Σ b_j
        subject to Σ_i f_i * b_{k-i} ≤ max_conv   for all k
                    b_j ≥ 0

    Parameters
    ----------
    f_seq : 1‑D np.ndarray, shape (n,)
        Current sequence (non‑negative).
    max_conv : float
        Upper bound on the convolution (the current maximal value).

    Returns
    -------
    np.ndarray or None
        The optimal vector `b` (shape (n,)) or None if LP failed.
    """
    n = f_seq.shape[0]

    # Objective: -1 for every variable to turn maximise into minimise.
    c = -np.ones(n, dtype=float)

    # Build the convolution constraints.
    # For each k = 0 … 2n‑2 we have a row where
    #   Σ_i f_i * b_{k-i} = Σ_{j} f_{k-j} * b_j  (j = k-i)
    # Only indices j that lie in [0, n‑1] appear.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices j = lo … hi  (these are the columns of b)
        # corresponding f‑coefficients are f_{k-j}
        idx = np.arange(lo, hi + 1)
        row[idx] = f_seq[k - idx]
        rows.append(row)

    # Non‑negativity of b.
    rows.append(-np.eye(n))

    A_ub = np.vstack(rows)
    b_ub = np.concatenate([np.full(2 * n - 1, max_conv), np.zeros(n)])

    # Use the HiGHS solver – fast and reliable for modest sizes.
    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"presolve": True, "time_limit": 8.0},
    )
    if res.success:
        return np.maximum(res.x, 0.0)   # safeguard tiny negatives
    return None

                    no_improve_counter = 0
                    print(
                        f"[{time.time():.1f}] Gradient step → new best {sc:.6f} (len={new_seq.size})"
                    )
                else:
                    break

        # ----------------------------------------------------------------
        # Book‑keeping for stagnation detection.
        # ----------------------------------------------------------------
        if best_score < 1.4:
            # Very good – we can stop early.
            break
        if cur_seq is not None and best_score < np.inf:
            # If we haven't improved in this iteration, increment counter.
            if no_improve_counter is not None:
                no_improve_counter += 1

    # ------------------------------------------------------------------
    # Return the best sequence (as a plain Python list, as required).
    # ------------------------------------------------------------------
    print(f"Finished. Best score = {best_score:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
