def subgradient_step(a, step):
    conv = np.convolve(a, a)
    k_max = np.argmax(conv)
    # Compute gradient for each j
    grad = np.zeros_like(a)
    for j in range(len(a)):
        i = k_max - j
        if 0 <= i < len(a):
            grad[j] = 2 * a[i]
    # Step update
    a_new = a - step * grad
    a_new[a_new < 0] = 0
    total = a_new.sum()
    if total == 0:
        a_new = np.random.random(len(a))
        total = a_new.sum()
    a_new = a_new / total
    return a_new
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Determine length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000
    # initialize
    a = np.random.rand(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float('inf')
    # iteration loop
    step0 = 0.5
    iter_count = 0
    while time.time() < deadline:
        # compute convolution and score
        conv = np.convolve(a, a)
        max_b = conv.max()
        score = 2 * n * max_b   # since sum = 1
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iter_count}: new best score {best_score:.6f}")
        # subgradient step
        k_max = int(np.argmax(conv))
        grad = np.zeros_like(a)
        if k_max < n:
            # positions 0..k_max
            grad[:k_max+1] = 2 * a[k_max::-1]
        else:
            start = k_max - (n - 1)
            length = 2 * n - 1 - k_max   # = (2n-1 - k_max)
            # slice of a up to length
            grad[start:] = 2 * a[:length][::-1]
        # step size decay
        eta = step0 / np.sqrt(iter_count + 1)
        a = a - eta * grad
        # projection onto simplex (nonnegative + sum=1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            # reinitialize
            a = np.random.rand(n)
            a = a / a.sum()
        else:
            a = a / s
        # occasional jitter to escape stagnation
        if iter_count % 2000 == 0 and iter_count > 0:
            a = a + np.random.randn(n) * 1e-4
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        iter_count += 1
    # optional final polishing with smaller step
    # return best
    return best_a.tolist()

"""
Propose a candidate sequence for the step‑function evaluation.

The core idea is to start from the classic "constant convolution" sequence
a_k = C(2k,k) / 4^k  (C = central binomial coefficient).  This sequence
satisfies  (a * a)[k] = 1 for all k < n, so after normalising its L1‑norm
to 1 it already improves the bound from 2.0 to about π/2 ≈ 1.57.
We then run a simple projected sub‑gradient descent on the true objective
   F(a) = max_k (a * a)[k]
with the constraints a_i ≥ 0 and Σ a_i = 1.  At each step we locate the
index k̂ attaining the maximal convolution, compute the sub‑gradient
∂F/∂a_i = 2·a_{k̂‑i} (when the index is inside the array), take a small
step opposite to the gradient, clip to non‑negative values and re‑normalise.
Occasional random perturbations help escape plateaus.
The best sequence seen during the run is returned.

The routine respects the 1000 s time budget and works for any length n.
"""

import time
import numpy as np

def _constant_conv_base(n: int) -> np.ndarray:
    """
    Returns the first n terms of the sequence a_k = binom(2k,k) / 4**k.
    Recurrence: a_0 = 1, a_k = a_{k‑1} * (1 - 1/(2k)).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for minimising max(conv(a,a)).
    Returns the updated vector (still non‑negative, not yet normalised).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    kmax = int(np.argmax(conv))

    # build gradient vector g where g[i] = 2 * a[kmax - i] if index valid
    g = np.zeros_like(a)

    if kmax < n:
        # indices 0 … kmax are valid
        g[:kmax + 1] = 2.0 * a[kmax::-1]
    else:
        # kmax ≥ n
        start = kmax - (n - 1)            # first i with kmax‑i < n
        length = 2 * n - 1 - kmax        # number of valid positions
        # a[0:length] are the values that appear in the reversed slice
        g[start:] = 2.0 * a[:length][::-1]

    # gradient step and projection onto the non‑negative orthant
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)

    # re‑normalise to L1‑norm 1 (if sum becomes zero we re‑initialise)
    tot = a_new.sum()
    if tot <= 0.0:
        # very unlikely, but guard against division by zero
        a_new = np.random.rand(n)
        a_new /= a_new.sum()
        return a_new

    return a_new / tot


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------------
    # Choose a length.  If a previous construction exists we reuse its size,
    # otherwise we fall back to a moderate default (2000).
    # --------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000

    # --------------------------------------------------------------------
    # Initialise with the constant‑convolution sequence and normalise.
    # --------------------------------------------------------------------
    base = _constant_conv_base(n)
    a = base / base.sum()                     # sum = 1, all entries ≥ 0

    # keep the best solution seen so far
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"Initial length {n}, initial score {best_score:.6f}")

    # --------------------------------------------------------------------
    # Main optimisation loop: projected sub‑gradient descent with decay.
    # --------------------------------------------------------------------
    it = 0
    step0 = 0.5       # initial step size (will decay like 1/√it)
    while time.time() < deadline:
        # adaptive step size
        step = step0 / np.sqrt(it + 1.0)

        # one sub‑gradient update
        a = _subgradient_step(a, step)

        # occasionally add a tiny random jitter to avoid stagnation
        if it % 3000 == 0 and it > 0:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # evaluate current objective (no need to call evaluate_sequence directly)
        conv = np.convolve(a, a)
        cur_score = 2.0 * n * conv.max()

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {it:6d}: improved score {best_score:.6f}")

        it += 1

    # --------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence (optional,

    # --------------------------------------------------------------------

        final_score = best_score

    print(f"Search finished after {it} iterations, best score {best_score:.6f}")
    return best_seq.tolist()
