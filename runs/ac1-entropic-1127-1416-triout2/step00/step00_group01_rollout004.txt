def project_simplex(v):
    # ensures sum=1, nonnegative
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def exponentiated_update(a, g, eta):
    # a, g arrays length n
    # Compute a_i * exp(-eta * g_i)
    new_a = a * np.exp(-eta * g)
    # renorm to sum 1
    new_a_sum = new_a.sum()
    if new_a_sum <= 0:
        # fallback: uniform
        return np.full_like(a, 1.0 / len(a))
    return new_a / new_a_sum
```

    Subgradient / exponentiated gradient heuristic to minimize the max
    self‑convolution of a non‑negative sequence.  Starts from several
    random Dirichlet vectors (and optionally the known SOTA sequence)
    and iteratively moves mass away from entries that contribute most
    to the current maximal convolution peak.  The best sequence seen
    within the time budget is returned.
    """

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # length of the sequence we will work on
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = np.random.randint(4000, 5000)

    # helper convolution using FFT (real-valued)
    def conv_fft(x, y):
        """Linear convolution of real vectors using rfft."""
        n_total = len(x) + len(y) - 1
        m = 1 << ((n_total - 1).bit_length())   # next power of two
        X = np.fft.rfft(x, m)
        Y = np.fft.rfft(y, m)
        out = np.fft.irfft(X * Y, m)
        return out[:n_total]

    def subgradient(a, conv):
        """Return sub‑gradient of max(conv) w.r.t. a."""
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-15
        active = (conv >= max_val - eps).astype(float)          # shape (2n-1,)
        a_rev = a[::-1]
        corr = conv_fft(active, a_rev)                         # length 3n-2
        # first n entries of corr (indices 0..n-1) correspond to i=n-1..0
        g = 2.0 * corr[:len(a)][::-1]
        return g, max_val

    # projection onto the probability simplex (sum = 1, a_i >= 0)
    def proj_simplex(v):
        """Euclidean projection of v onto the simplex."""
        n_v = len(v)
        u = np.sort(v)[::-1]

        if not np.any(cond):
            # all zeros – return uniform
            return np.full_like(v, 1.0 / n_v)
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    best_seq = None
    best_score = float('inf')
    restart = 0
    # parameters for the inner loop
    max_inner_iters = 5000
    eta0 = 1.0          # base step size for additive update

    total_iters = 0

        # initialise a probability vector

            a = np.array(height_sequence_1, dtype=float)

            a = a / a.sum()

            conv = conv_fft(a, a)               # length 2n-1
            g, max_val = subgradient(a, conv)

            # current score (sum is 1 → evaluate = 2*n*max_val)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:

                best_seq = a.copy()

                if best_score < 1.55:
                    deadline = time.time() + 0.2   # finish soon

    # fallback: uniform if nothing better found
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Return as plain Python list
    return best_seq.tolist()

"""
subgradient_exponential.py

A heuristic that directly minimizes the self‑convolution L∞‑norm of a
non‑negative step‑function.  The evaluation function is

    evaluate_sequence(a) = 2·n·max(a*a) / (sum(a)²)

which is scale‑invariant, so we fix sum(a)=1 and minimise
`max(a*a)`.  The algorithm works as follows:

1. Initialise a probability vector `a` (either the known SOTA sequence
   or a random Dirichlet sample).
2. Compute the linear convolution `c = a*a` using an FFT based routine.
3. Identify the set of indices where `c` attains its maximum.
4. The sub‑gradient of `max(c)` w.r.t. `a` is
   `g_i = 2 * Σ_{k∈active} a_{k-i}`.  This is obtained efficiently via a
   second FFT (correlation of the binary active mask with the reversed `a`).
5. Perform an additive sub‑gradient step `a ← a – η·g`,
   then project back onto the probability simplex (non‑negative, sum = 1).
   Small random jitter is added occasionally to escape plateaus.
6. Keep the best sequence seen (lowest `2·n·max(c)`) and repeat the
   whole inner optimisation from several random restarts until the time
   budget expires.

The method is cheap per iteration (a few FFTs) and can explore many
different candidates within the 1000 s limit, typically improving the
bound well below the uniform value `2.0`.
"""

    the allotted time.
    """

    else:
        n = np.random.randint(4000, 5000)

    # Helper: linear convolution via FFT (real inputs)
    def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
        """Return the linear convolution of real vectors x and y using rFFT."""
        total_len = len(x) + len(y) - 1
        m = 1 << ((total_len - 1).bit_length())  # next power of two
        X = np.fft.rfft(x, m)
        Y = np.fft.rfft(y, m)
        out = np.fft.irfft(X * Y, m)

    # ------------------------------------------------------------------
    # Sub‑gradient of the max‑convolution functional
    def subgradient(a: np.ndarray, conv: np.ndarray):

        Compute a sub‑gradient of max(conv) w.r.t. a and also return the
        current maximum value.
        """

        eps = max_val * 1e-12 + 1e-15

        a_rev = a[::-1]

        return g, max_val

    # Euclidean projection onto the probability simplex (non‑negative, sum=1)
    def proj_simplex(v: np.ndarray) -> np.ndarray:

        n_v = len(v)
