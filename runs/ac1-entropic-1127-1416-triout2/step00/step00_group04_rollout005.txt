def solve_convolution_lp(f, M):
    n = len(f)
    # Build Toeplitz matrix
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([f[:1], np.zeros(n-1)])
    T = scipy.linalg.toeplitz(col, row)[:2*n-1,:]  # shape (2n-1,n)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)]*n
    # Solve
    res = linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'time_limit':5})
    if res.success:
        return np.maximum(0, res.x)  # ensure nonneg
    return None
```

"""
Propose a low‑score step‑function sequence.

The evaluation function is
    score = 2·n·max(conv(a,a)) / (sum(a))² .
If we keep sum(a)=√(2n) then score = max(conv(a,a)).
The algorithm maintains this normalisation and tries to lower the
maximum convolution value.

Strategy
---------
1. Initialise a uniform sequence of length n (∑a = √(2n)).
2. Repeatedly solve a linear programme:
       maximise   ∑b   subject to   conv(f,b) ≤ M ,  b ≥ 0
   where f is the current normalised sequence and M = max(conv(f,f)).
   The LP returns a direction b that can increase the total mass without
   raising the current convolution bound.
3. Scale b back to the normalised sum and blend it with the current f.
   Several blend factors t are tried (0.5 … 0.01); the blend giving the
   best (lowest) score is kept.
4. If the LP fails, a small random perturbation is applied.
5. The process is repeated until the time budget expires or a target
   score (<1.503) is reached.  Occasional restarts avoid local minima.

All helper functions are top‑level; only standard scientific libraries
are used.  The final sequence (list of non‑negative floats) is returned.
"""

import time
import numpy as np
from scipy import optimize, linalg

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise_to_target(seq, target_sum):
    """Scale seq so that its sum equals target_sum, keeping non‑negativity."""
    s = float(np.sum(seq))
    if s <= 0.0:
        # fallback uniform if everything is zero
        n = len(seq)
        return [target_sum / n] * n
    factor = target_sum / s
    # clip negative values to zero after scaling (they should stay non‑negative)
    return [max(0.0, x * factor) for x in seq]


def _build_conv_matrix(f):
    """
    Build the (2n‑1)×n matrix A such that A @ b = conv(f, b).
    Row k (0‑based) contains f[k‑j] for j≤k, otherwise 0.
    """
    n = len(f)
    # first column: f[0], f[1], …, f[n‑1], 0, …, 0   (length 2n‑1)
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    # first row: f[0], 0, …, 0   (length n)
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=float)])
    # Toeplitz matrix gives exactly the required convolution coefficients
    A = linalg.toeplitz(col, row)
    return A  # shape (2n‑1, n)


def solve_convolution_lp(f_norm, max_conv):
    """
    Solve:
        maximise   ∑b
        subject to conv(f_norm, b) ≤ max_conv   (for all offsets)
                   b ≥ 0
    Returns the optimal b as a NumPy array, or None if the LP fails.
    """
    n = len(f_norm)
    # Linear objective: minimise -∑b  → maximise ∑b
    c = -np.ones(n, dtype=float)

    # Convolution constraints
    A_ub = _build_conv_matrix(np.asarray(f_norm, dtype=float))
    b_ub = np.full(2 * n - 1, max_conv, dtype=float)

    # Non‑negativity bounds
    bounds = [(0.0, None) for _ in range(n)]

    # Use the HiGHS interior‑point solver; give a modest time limit
    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": 2.0, "disp": False},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        # Numerical noise may produce tiny negatives; clip to zero
        return np.maximum(0.0, res.x)
    return None


def _direction_via_lp(current_seq):
    """
    Compute a better direction using the LP described in the paper.
    Returns a new normalised sequence (list) or None if the LP fails.
    """
    n = len(current_seq)
    target_sum = (2 * n) ** 0.5

    # Normalise current sequence to the prescribed sum
    f_norm = _normalise_to_target(current_seq, target_sum)

    # Current max convolution (score for the normalised sequence)
    # Because we have the normalisation, evaluate_sequence returns exactly max(conv)
    # The function evaluate_sequence is assumed to be imported in the execution environment.
    try:
        max_conv = evaluate_sequence(f_norm)
    except Exception:  # pragma: no cover
        return None

    # Solve the LP for a candidate direction
    b_opt = solve_convolution_lp(f_norm, max_conv)
    if b_opt is None:
        return None

    # Scale the LP solution back to the required normalisation
    g_norm = _normalise_to_target(b_opt, target_sum)
    return g_norm


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    Uses the LP‑based direction search described above with a simple
    line‑search over blend factors.  Returns the best sequence found
    before the time budget runs out.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a tiny safety margin

    best_seq = None
    best_score = float("inf")
    target_score = 1.503  # early‑stop threshold

    # Parameters for the search
    blend_factors = [0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    restart_interval = 250          # attempts without improvement before restart
    max_local_time = 30.0           # seconds spent on each fresh start

    global_iter = 0
    while time.time() < deadline:
        # ---------- initialise a fresh problem ----------
        n = np.random.randint(200, 801)          # moderate length, good trade‑off
        target_sum = (2 * n) ** 0.5
        current_seq = [target_sum / n] * n       # uniform start (already normalised)
        try:
            current_score = evaluate_sequence(current_seq)
        except Exception:                        # pragma: no cover
            current_score = float("inf")

        # Track the best sequence found in this local run
        local_best_seq = list(current_seq)
        local_best_score = current_score

        # Local optimisation loop
        no_improve = 0
        local_deadline = time.time() + min(max_local_time,
                                           deadline - time.time())
        while time.time() < local_deadline:
            # Obtain a direction from the LP
            g_seq = _direction_via_lp(current_seq)

            improved_this_round = False
            if g_seq is not None:
                # Try several blend factors and keep the best improvement
                for t in blend_factors:
                    cand = [(1.0 - t) * x + t * y for x, y in zip(current_seq, g_seq)]
                    # re‑normalise to stay on the feasible manifold
                    cand = _normalise_to_target(cand, target_sum)

                    try:
                        cand_score = evaluate_sequence(cand)
                    except Exception:
                        continue

                    if cand_score < local_best_score - 1e-9:
                        # Accept this improvement
                        local_best_score = cand_score
                        local_best_seq = list(cand)
                        current_seq = list(cand)
                        improved_this_round = True
                        no_improve = 0
                        break  # no need to test smaller t's
                # end for blend_factors
            else:
                # LP failed – perform a tiny random perturbation
                idx = np.random.randint(n)
                current_seq[idx] = max(0.0,
                                       current_seq[idx] + np.random.randn() * 0.01 * target_sum)

            if not improved_this_round:
                # Small stochastic move to escape possible plateaus
                idx = np.random.randint(n)
                current_seq[idx] = max(0.0,
                                       current_seq[idx] + np.random.randn() * 0.005 * target_sum)
                no_improve += 1

            # Update global best if needed
            if local_best_score < best_score - 1e-9:
                best_score = local_best_score
                best_seq = list(local_best_seq)
                print(f"[{time.time():.1f}] New global best: {best_score:.6f} (n={n})")
                if best_score <= target_score:
                    # reached the desired bound – exit early
                    return best_seq

            # Restart if we have stalled for a while
            if no_improve >= restart_interval:
                # Re‑initialise with uniform distribution again
                current_seq = [target_sum / n] * n
                no_improve = 0

        # end while local optimisation

        global_iter += 1
        if global_iter % 5 == 0:
            # occasional status line
            print(f"[{time.time():.1f}] Iter {global_iter}, best so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (fallback to uniform if nothing else)
    # ------------------------------------------------------------------
    if best_seq is None:
        # Very unlikely, but keep the contract of returning a non‑empty list
        n = 1
        best_seq = [1.0]
    return best_seq
