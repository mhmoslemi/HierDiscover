def project_onto_simplex(v, target):
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    deadline = time.time() + budget_s - 0.5
    n = len(height_sequence_1)  # use existing length maybe else choose length from random

    else:
        a = np.abs(np.random.randn(n))
        a = a / a.sum() * target_sum
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # iteration loop
    iter = 0
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(a, a)
        k_max = np.argmax(conv)
        max_val = conv[k_max]
        # compute gradient of max entry
        # create partner indices
        idx = np.arange(n)

        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad += a  # first term a_i
        grad[mask] += a[partner[mask]]
        # step size

def project_onto_simplex(v, target_sum):
    # v: np.ndarray, target_sum >0
    # Using algorithm from Duchi et al.
    if target_sum <=0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if len(rho)==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid sub‑gradient optimizer for the self‑convolution bound.

The objective (ignoring the scale‑invariant denominator) is to minimise
the maximum entry of a*a, given a_i ≥ 0 and Σ a_i fixed.
We use a projected sub‑gradient descent with several random restarts.
The projection enforces the simplex constraint Σa_i = √(2n).
A simple exponential‑decay step‑size together with occasional Gaussian
noise helps to escape plateaus.  The best sequence seen during the
budget (≈1000 s) is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the simplex Σx_i = target, x_i ≥ 0
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of sum = target."""
    if target <= 0:
        return np.zeros_like(v)

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)          # 1‑based index
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]                # last true index (1‑based)
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: sub‑gradient of the active (max) convolution entry
# ------------------------------------------------------------
def _max_convolution_subgrad(a: np.ndarray):
    """
    Compute the sub‑gradient of the max entry of a * a.
    Returns (grad, max_val, k_max) where:
        grad    – ∂(max_k (a*a)[k]) / ∂a  (sub‑gradient)
        max_val – value of the active entry
        k_max   – index where the maximum occurs
    """
    conv = np.convolve(a, a)                 # length 2n‑1
    k_max = int(np.argmax(conv))
    max_val = conv[k_max]

    n = a.shape[0]
    idx = np.arange(n)
    partner = k_max - idx
    grad = a.copy()
    mask = (partner >= 0) & (partner < n)
    # Each a_i appears twice in the active sum (once as left, once as right term)
    grad[mask] += a[partner[mask]]
    return grad, max_val, k_max

    """

    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    target_sum = np.sqrt(2.0 * n)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Several simple initialisations (uniform, sqrt‑decay, random)
    # ------------------------------------------------------------------
    def _initial_sequence(kind: str) -> np.ndarray:
        if kind == "uniform":
            a = np.full(n, target_sum / n)
        elif kind == "sqrt":
            # a_i ∝ 1/√(i+1)  (decreasing, gives more mass at small indices)
            raw = 1.0 / np.sqrt(np.arange(n) + 1.0)
            a = raw
        else:  # "random"
            a = np.random.rand(n)
        # enforce the simplex constraint
        return _project_onto_simplex(a, target_sum)

    init_kinds = ["uniform", "sqrt", "random"]

    # ------------------------------------------------------------------
    # Main optimisation loop (multiple random restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline:

        a = _initial_sequence(kind)

        # evaluate the starting point
