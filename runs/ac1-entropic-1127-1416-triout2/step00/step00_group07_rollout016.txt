    gradient of the current maximal convolution entry.
    """

    deadline = time.time() + budget_s - 5

    # Determine problem size
    try:
        n = len(height_sequence_1)  # global variable; if not present fallback

    L = 1 << ((2 * n - 1).bit_length())

    a = np.random.rand(n)
    a /= a.sum()
    best_a = a.copy()
    # compute initial score via evaluate_sequence
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial score: {best_score:.6f}")

    # precreate index array for derivative computation
    idx = np.arange(n)

    # annealing parameters
    temp_start = 0.1
    temp_end = 1e-4
    iter_cnt = 0

    while time.time() < deadline:

        Fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[:2 * n - 1]

        M = conv.max()
        score = 2.0 * n * M  # sum(a)=1 so square cancels

        if score < best_score:
            best_score = score
            best_a = a.copy()

def simplex_projection(v):
    """Project v onto the probability simplex."""
    # from https://arxiv.org/pdf/1101.6081.pdf
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
A simple projected sub‑gradient optimiser for the max‑convolution objective.

The evaluation function is scale‑invariant:
    score = 2·n·max_k (a*a)[k] / (sum a)^2 .
Hence we may restrict ourselves to probability vectors (sum = 1) and simply
minimise 2·n·max_k (a*a)[k].

The algorithm works as follows:
1.  Initialise a random probability vector (Dirichlet distribution).
2.  Repeatedly:
    * Compute the linear convolution a*a (via FFT) and locate the index k*
      where it attains its maximum.
    * The sub‑gradient of the maximal convolution entry with respect to a_i is

          g_i = a_{k*-i}          if i == k*-i
          g_i = 2·a_{k*-i}        otherwise (zero if the index is out of range)

      (this follows from d/d a_i  sum_j a_j a_{k*-j} ).
    * Take a small step opposite to the gradient, then project back onto the
      probability simplex (non‑negative, sums to 1).  The projection is performed
      with the standard O(n) algorithm.
    * Keep the best vector found so far (the one with the lowest
      evaluate_sequence score).
    * Occasionally add a tiny random perturbation to escape plateaus.

The method is fast (FFT of length 2·n is performed each iteration) and
fits comfortably in the 1000 s budget.  The final sequence is scaled by a
small constant (10) only to keep the numbers comfortably away from the
0.01‑threshold used by evaluate_sequence; the scaling does **not** change
the score because the objective is homogeneous of degree 0.

The routine prints occasional progress messages and finally returns the
best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex {x>=0, sum x = 1}
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, length a power of two)
def _prepare_fft_length(n: int) -> int:
    """Smallest power‑of‑two >= 2·n‑1 (sufficient for linear convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def conv_max_and_array(a: np.ndarray, L: int):
    """Return (max value, full linear convolution) of a with itself."""
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * a.size - 1]
    return conv.max(), conv


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry
def active_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry at index k.
    For each i, g_i = a_{k-i}       (i == k-i)
                = 2·a_{k-i}       (i != k-i, if the partner index exists)
                = 0               (otherwise)
    """
    n = a.size
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    # self‑pair case
    self_mask = valid & (partner == idx)
    g[self_mask] = a[idx[self_mask]]
    # distinct‑pair case
    off_mask = valid & (partner != idx)
    g[off_mask] = 2.0 * a[partner[off_mask]]
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The search runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a safety margin

    # ------------------------------------------------------------------
    # Determine the length of the problem.
    try:
        n = len(height_sequence_1)          # provided by the evaluation harness
    except NameError:
        # Fallback: a moderately sized vector if the global is missing.
        n = 2000
        print("[INFO] height_sequence_1 not found – using n =", n)

    L = _prepare_fft_length(n)              # FFT length (power of two)

    # ------------------------------------------------------------------
    # Storage for the best solution found.
    best_score = np.inf
    best_vec = None

    # ------------------------------------------------------------------
    # Main outer loop – occasional restarts.
    iteration = 0
    while time.time() < deadline:
        # Random start: Dirichlet(1) (i.e. exponential‑distributed components)
        a = np.random.exponential(scale=1.0, size=n)
        a /= a.sum()                         # probability vector, sum = 1

        # Local optimisation parameters
        step = 0.05                          # initial step size for gradient step
        inner_it = 0

        while time.time() < deadline and inner_it < 5000:
            inner_it += 1
            # 1. current convolution and its active index
            cur_max, conv_full = conv_max_and_array(a, L)
            k_star = int(np.argmax(conv_full))

            # 2. compute the sub‑gradient of the active entry
            g = active_gradient(a, k_star)

            # 3. take a projected sub‑gradient step
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)

            # 4. evaluate the candidate (our own cheap evaluation)
            cand_max, _ = conv_max_and_array(a_candidate, L)
            cand_score = 2.0 * n * cand_max    # sum(a)=1 → denominator = 1

            # 5. accept if improvement; otherwise keep the current a
            if cand_score < 2.0 * n * cur_max:
                a = a_candidate
                cur_max = cand_max
                # shrink step a little after a successful move
                step = max(step * 0.95, 1e-5)
            else:

                step = max(step * 0.99, 1e-5)

            # ----------------------------------------------------------------

            # The cheap score coincides with evaluate_sequence when sum=1,

                    best_score = official_score
                    best_vec = a.copy()
                    print(f"[{time.time():.1f}] New best score = {best_score:.6f}")

            # ----------------------------------------------------------------
            # Random perturbation every few hundred iterations to avoid stagnation
            if inner_it % 500 == 0:
                i1, i2 = np.random.choice(n, 2, replace=False)
                eps = 0.01 * a[i1]                     # move 1 % of mass
                a[i1] -= eps
                a[i2] += eps
                a = simplex_projection(a)

        iteration += 1
        # Occasionally restart even if time remains
        if iteration % 3 == 0 and time.time() < deadline:
            continue

    # ------------------------------------------------------------------
    # If nothing was found (should not happen), fall back to a uniform vector.
    if best_vec is None:

        best_score = evaluate_sequence(best_vec.tolist())
        print("[WARN] No improvement – returning uniform sequence.")

    # Scale the probabilities to avoid the 0.01 sum guard in evaluate_sequence.
    # Scaling does not change the objective because it is homogeneous.
    final_seq = (best_vec * 10.0).tolist()   # values are comfortably >0 and <1000

    # Final verification / reporting
    final_score = evaluate_sequence(final_seq)
    print(f"[DONE] Best score found: {best_score:.6f}, final reported score: {final_score:.6f}")

    return final_seq
