    deadline = time.time() + budget_s - 0.5   # small margin

    # choose length
    n = 500  # adjustable maybe increase if time allows

    # start with uniform distribution
    p = np.full(n, 1.0/n)

    # optionally perturb to random start

    best_p = p.copy()
    best_score = evaluate_sequence(p.tolist())

    # define helper to compute max conv
    def max_conv(v):
        return np.max(np.convolve(v, v))

    # loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Choose random update: select two indices i, j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Determine amount to transfer: delta proportional to min(p[i], p[j]) * random fraction
        delta = (np.random.rand() - 0.5) * 0.02  # small random signed change; we'll adjust to keep sum
        # Actually we propose moving delta from j to i
        delta = np.random.rand() * 0.01 * p[j]  # some fraction of p[j]
        # direction random: with 50% chance move from i to j
        if np.random.rand() < 0.5:
            # move from i to j
            delta = np.random.rand() * 0.01 * p[i]
            new_i = p[i] - delta
            new_j = p[j] + delta
        else:
            # move from j to i
            delta = np.random.rand() * 0.01 * p[j]
            new_i = p[i] + delta
            new_j = p[j] - delta

        # ensure nonnegative
        if new_i < 0 or new_j < 0:
            continue

        new_p = p.copy()
        new_p[i] = new_i
        new_p[j] = new_j

        # compute objective quickly: need max conv only (no scaling)
        # use np.convolve
        maxb = np.max(np.convolve(new_p, new_p))
        # compute f = 2 * n * maxb / (sum(p)^2). sum(p) = 1.
        score = 2.0 * n * maxb  # since sum=1
        # If improvement
        if score < best_score:
            best_score = score
            best_p = new_p.copy()
            p = new_p
            # log

        else:
            # maybe accept with probability depending on temperature?

    # Compute conv
    conv = np.convolve(p, p)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    # Choose one k star
    k = np.random.choice(ks)
    # gradient = 2 * reversed(p) shifted by k
    g = np.zeros_like(p)
    low = max(0, k-(n-1))
    high = min(n-1, k)
    idx = np.arange(low, high+1)
    g[idx] = 2 * p[k - idx]
    # Update p
    p_new = p - step * g
    # Project onto simplex
    p_new = project_simplex(p_new)
    return p_new

def improve_sequence(seq, max_iter, t_start):
    n = len(seq)
    # Normalize sum to sqrt(2n)
    tot = np.sqrt(2 * n)
    seq = np.array(seq)
    if np.sum(seq) == 0:
        seq = np.full(n, tot/n)
    else:
        seq = seq * (tot / np.sum(seq))
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    t = t_start
    for it in range(max_iter):
        # compute normalized version to sum sqrt(2n)
        # Solve LP for g
        g = solve_convolution_lp(seq.tolist(), np.max(np.convolve(seq, seq)))
        if g is None:
            # random perturb
            idx = np.random.randint(n)
            seq[idx] = max(0, seq[idx] + np.random.randn() * 0.01 * seq[idx])
            continue
        # Normalize g to sum sqrt(2n)
        g = np.array(g)
        g = g * (np.sqrt(2 * n) / np.sum(g))
        # mix
        seq = (1 - t) * seq + t * g
        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # decay t
        t *= 0.99
    return best_seq, best_score

"""
propose_candidate
-----------------

A simple simulated‑annealing search for a non‑negative step function that lowers the
evaluation defined in the competition.  The objective
    f(a) = 2·n·max(a∗a) / (sum(a)²)
is invariant under scaling, therefore we keep the ℓ₁‑norm fixed to
√(2·n).  Starting from a uniform distribution we repeatedly move a small amount
of mass from one coordinate to another while keeping the total sum unchanged.
A Metropolis rule with a slowly decreasing temperature accepts improving moves
always and occasionally accepts worsening moves, allowing the search to escape
local minima.  The algorithm is run for several short sequence lengths
(128, 256, 512); the best sequence found in the allotted time is returned.
All numbers are clipped to the interval [0, 1000] so that the provided
`evaluate_sequence` routine never rejects them.

The implementation uses only NumPy and the already‑imported `evaluate_sequence`
function; no file or network access is performed.
"""

import time
import numpy as np

def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper that calls the external evaluate_sequence."""
    # evaluate_sequence expects a plain Python list
    return evaluate_sequence(seq.tolist())

def _anneal_sequence(initial: np.ndarray,
                     max_iters: int,
                     time_deadline: float) -> (np.ndarray, float):
    """
    Perform simulated annealing on a fixed‑size sequence.

    Parameters
    ----------
    initial : np.ndarray
        Starting array (already normalised to sum = sqrt(2·n)).
    max_iters : int
        Upper bound on the number of iterations for this run.
    time_deadline : float
        Unix timestamp when the whole search must stop.

    Returns
    -------
    best_seq : np.ndarray
        The best sequence encountered.
    best_score : float
        Its evaluation value.
    """
    n = len(initial)
    # required ℓ₁‑norm (kept constant during the walk)
    target_sum = np.sqrt(2.0 * n)

    a = initial.copy()
    cur_score = _evaluate(a)
    best_seq = a.copy()
    best_score = cur_score

    # annealing schedule parameters
    temperature = 1.0          # initial temperature
    step_factor = 0.05         # maximal fraction of donor mass to move

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break

        # propose a random transfer of mass
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        donor = a[j]
        if donor <= 0.0:
            continue

        # how much to move from j to i (random fraction of donor)
        delta = step_factor * donor * np.random.rand()
        if delta > donor:
            delta = donor

        a_new = a.copy()
        a_new[i] += delta
        a_new[j] -= delta

        # sanity: all entries must stay non‑negative
        if a_new[j] < 0.0:
            continue

        # quick renormalisation is not needed – the transfer conserves the sum.
        new_score = _evaluate(a_new)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()

        # cooling
        temperature *= 0.9999
        # slowly shrink the step size to fine‑tune later on
        if (it + 1) % 1000 == 0:
            step_factor *= 0.9

    return best_seq, best_score


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Entry point used by the evaluator.  Runs a time‑limited simulated‑annealing
    optimisation and returns the best step function found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget in seconds.

    Returns
    -------
    list[float]
        The optimal sequence discovered within the time budget.
    """
    np.random.seed(seed)

    # leave a small margin for the outer harness
    deadline = time.time() + budget_s - 0.5

    global_best_seq = None
    global_best_score = float("inf")

    # try a few modest lengths – the objective is scale‑invariant,
    # so a short sequence is enough to discover a good shape.
    lengths = [128, 256, 512]

    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a uniform sequence with the required ℓ₁‑norm
        target_sum = np.sqrt(2.0 * n)
        uniform = np.full(n, target_sum / n)

        # run the annealer; we cap the iterations so that the loop stays fast
        max_iters = 20000
        seq, score = _anneal_sequence(uniform, max_iters, deadline)

        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq
            print(f"[{time.time():.2f}] New global best: {global_best_score:.6f} "
                  f"(length {n})")

    # If, for any reason, no sequence was produced, fall back to uniform.
    if global_best_seq is None:
        n = 256
        target_sum = np.sqrt(2.0 * n)
        global_best_seq = np.full(n, target_sum / n)

    # Clip to the limits enforced by evaluate_sequence.
    final = np.clip(global_best_seq, 0.0, 1000.0)

    # Return as a plain Python list of floats.
    return final.tolist()
