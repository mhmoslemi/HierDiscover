def project_simplex(v):
    # Efficient O(n log n) projection.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1,len(v)+1) > (cssv - 1))[0][-1]  # find largest index
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
A very small optimiser for the evaluation function.

The idea is simple:
* start from the uniform step function (which gives a score of exactly 2);
* at each iteration solve a linear programme that finds a direction `g`
  with a larger L1‑norm while keeping the convolution `seq * g` bounded
  by the current maximum `M`.  The LP is linear because each entry of the
  convolution is a linear function of `g`.
* mix the current sequence with the LP solution using a set of mixing
  coefficients and keep the mixture that gives the lowest score.
* repeat until the time budget is exhausted.

The algorithm works on a moderate size (≈ 200) vector – this keeps the LP
small enough to solve many iterations inside the 1000 s budget while still
allowing enough freedom to move away from the uniform solution.
The final sequence is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# -------------------------------------------------------------
# helper: projection of a vector onto the probability simplex
def _project_simplex(v):
    """Project v (1‑D array) onto the simplex {x≥0, sum x = 1}."""
    if np.allclose(v, 0):
        # avoid division by zero – return uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, v.size + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# solve the LP  max 1^T g   s.t.  conv(seq,g) ≤ M , g ≥ 0
def _solve_lp(seq, bound_M):
    """Return a feasible g with maximal L1 norm under the convolution bound.
    The LP is:
        maximize   sum(g_j)
        subject to sum_i seq_i * g_{k-i} ≤ M   for all k
                   g_j ≥ 0
    If the LP fails, ``None`` is returned.
    """
    n = len(seq)
    # objective – maximise sum(g)  → minimise –sum(g)
    c = -np.ones(n)

    # build the convolution constraints (one per possible sum index)
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = seq[i]          # coefficient of g_j in this sum
        a_ub.append(row)
        b_ub.append(bound_M)

    # non‑negativity of g
    a_ub.extend(-np.eye(n))
    b_ub.extend([0.0] * n)

    a_ub = np.vstack(a_ub)
    b_ub = np.array(b_ub)

    # solve (use the fast HiGHS interior‑point solver)
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if not res.success:
        return None
    return res.x


# -------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.

    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found.  No external files are used; everything is computed in‑memory.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # -----------------------------------------------------------------
    # 1.  Choose a modest problem size – large enough to be interesting
    #     but small enough that the LP is cheap.
    #
    # If the global variable ``height_sequence_1`` exists we use its length,
    # otherwise we fall back to a default of 200.
    try:
        n = min(len(height_sequence_1), 300)   # cap to keep the LP cheap
    except Exception:
        n = 200

    # start from the uniform sequence (sum = 1)
    seq = np.full(n, 1.0 / n)

    # keep track of the best solution seen so far
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}  initial score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2.  Main optimisation loop – simple LP‑based descent.
    it = 0
    while time.time() < deadline:
        it += 1

        # current maximum of the convolution
        conv = np.convolve(seq, seq)
        M = conv.max()

        # solve the linear programme for a direction g
        g = _solve_lp(seq, M)
        if g is None:
            # LP failed – give up early
            print("[lp] LP did not return a solution, terminating.")
            break

        # normalise g to have the same L1 norm as seq (optional but keeps numbers nice)
        g = _project_simplex(g)

        # Try a few mixing coefficients and keep the best mixture.
        mix_candidates = [0.0, 0.01, 0.05, 0.1, 0.2, 0.4]   # 0 ⇒ stay where we are
        improved = False
        for t in mix_candidates:
            cand = (1.0 - t) * seq + t * g
            cand = _project_simplex(cand)       # keep it a probability vector
            score = evaluate_sequence(cand.tolist())
            if score < best_score - 1e-9:       # a strict improvement
                best_score = score
                best_seq = cand.copy()
                seq = cand                     # move to the new point
                improved = True
                print(f"[it {it:04d}] t={t:4.2f}  new best = {best_score:.6f}")
                break    # accept the first improvement (larger t are tried later)

        if not improved:
            # no improvement with the tried mixings – shrink the step size
            # (here we simply stop, as further large steps are unlikely to help)
            print(f"[it {it:04d}] no improvement, terminating.")
            break

        # safety break – do not run indefinitely
        if it >= 500:      # more than enough iterations for the given budget
            break

    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()
