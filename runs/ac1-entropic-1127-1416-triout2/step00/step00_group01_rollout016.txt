"""
propose_candidate
-----------------
Generate a step‑function (a list of non‑negative heights) that tries to
minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the discrete convolution used in `evaluate_sequence`.

The algorithm is based on the analytic observation that for a continuous
function f(x)=x^{-1/2} the self‑convolution is (almost) constant:
∫_0^t x^{-1/2}(t-x)^{-1/2}dx = π.  Discretising this idea gives a sequence

    a_i ∝ 1 / sqrt(i+1)

which yields an evaluation around π/2 ≈ 1.57 – substantially better than the
uniform sequence (value 2).  The search explores a few simple parameters
(`α` offset and an exponent `p` around 0.5) to improve the constant a little
further, and then performs cheap random local moves to see if an improvement
can be found within the time budget.

The function respects the 1000‑second limit, prints progress, and finally
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# evaluate_sequence is assumed to be already imported in the execution environment.


def _make_power_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """
    Create the raw (un‑normalised) sequence a_i = (i+1+offset)^{-p}.
    `offset` shifts the denominator to avoid a zero denominator and to
    control the magnitude of the first element.
    """
    idx = np.arange(n, dtype=float)                # i = 0 … n‑1
    raw = 1.0 / np.power(idx + 1.0 + offset, p)    # (i+1+offset)^{-p}
    return raw


def _normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale a non‑negative sequence so that its sum equals `target_sum`.
    """
    current = seq.sum()
    if current == 0.0:
        # Should never happen for our generators, but guard against division by zero.
        return seq
    scale = target_sum / current
    return seq * scale


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a good step‑function using the inverse‑square‑root family.
    The function runs for at most `budget_s` seconds and returns the best
    sequence (as a list of floats) that it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a few seconds margin for cleanup

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Coarse grid search over length, exponent (p) and offset (α).
    # ------------------------------------------------------------------
    lengths = [1000, 1500, 2000, 2500, 3000, 3500, 4000, 4500, 5000]
    # exponents around the analytically optimal 0.5
    p_values = [0.48, 0.49, 0.5, 0.51, 0.52]
    # offset values – small offsets keep a_0 reasonable, larger offsets flatten
    offsets = np.concatenate([
        np.linspace(0.0, 0.5, 11),   # 0 … 0.5
        np.linspace(0.5, 3.0, 11),   # 0.5 … 3
        np.linspace(3.0, 10.0, 8)    # 3 … 10
    ])

    for n in lengths:
        target_sum = np.sqrt(2.0 * n)          # the normalisation used by the original code
        for p in p_values:
            # pre‑compute the denominator part that does not depend on the offset
            base_idx = np.arange(n, dtype=float) + 1.0
            for off in offsets:
                # build, normalise and evaluate the candidate
                raw = 1.0 / np.power(base_idx + off, p)
                seq = _normalise_to_target_sum(raw, target_sum)

                # evaluate – the function internally caps at 1000 and forces non‑negativity
                score = evaluate_sequence(seq.tolist())

                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f'** New best (grid)  n={n:4d}  p={p:.3f}  α={off:.3f}  score={score:.6f}')

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Simple stochastic refinement: move a tiny amount of mass between
    #    two random positions and keep the change if it improves the score.
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback: uniform sequence (always gives 2.0)
        n = 2000
        best_seq = [np.sqrt(2.0 * n) / n] * n
        best_score = evaluate_sequence(best_seq)

    curr = np.array(best_seq, dtype=float)
    n = len(curr)

    # pre‑compute the target sum – it stays constant during the local moves
    target_sum = curr.sum()

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose moving a small amount (up to 1 % of a_i) from i to j
        max_move = 0.01 * curr[i]
        if max_move <= 0.0:
            continue
        delta = np.random.uniform(-max_move, max_move)
        # ensure we stay non‑negative
        if curr[i] - delta < 0.0:
            continue

        cand = curr.copy()
        cand[i] -= delta
        cand[j] += delta
        # (sum is unchanged, no need to renormalise)

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.tolist()
            curr = cand
            print(f'++ Random improvement: score={cand_score:.6f}  n={n}')
        # optional early exit if we already beat the target (1.503)
        if best_score < 1.503:
            break

    print(f'Final best score: {best_score:.6f} (length {len(best_seq)})')
    return best_seq
