def propose_candidate(seed=42, budget_s=1000, **kwargs):

    deadline = time.time() + budget_s - 1
    n = 6000 # maybe use 5000?
    # initialize
    a = np.random.random(n)
    a = a / a.sum()  # Normalize sum to 1
    # ensure no zero entries? allowed.
    # Candidates as numpy array.
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Precompute conv and max via FFT function
    def compute_max(arr):
        L = 1 << ((2*len(arr)-1).bit_length())
        fft = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fft*fft, n=L)[:2*len(arr)-1]
        return conv.max()
    # iterative local search
    iteration = 0
    while time.time() < deadline:
        # compute current max offset and contributions:
        L = 1 << ((2*n-1).bit_length())
        fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fft*fft, n=L)[:2*n-1]
        max_val = conv.max()
        # find indices where conv close to max:
        idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]
        # pick random max index:
        k = int(np.random.choice(idxs))
        # find pairs i,j such that i+j=k:
        # we can compute contributions via a[i]*a[k-i]
        # identify i where both i and k-i in range.
        # Compute products:
        i_low = max(0, k-(n-1))
        i_high = min(n-1, k)
        # compute product list
        i_vals = np.arange(i_low, i_high+1)
        j_vals = k - i_vals
        products = a[i_vals] * a[j_vals]
        # find index with max product
        max_pair_idx = np.argmax(products)
        i_star = i_vals[max_pair_idx]
        j_star = j_vals[max_pair_idx]
        # delta = small fraction of a[i_star] (or a[j_star])
        delta = a[i_star] * 0.01
        # modify: decrease a[i_star] and a[j_star] a bit, increase random other entries
        if a[i_star] < delta:
            delta = a[i_star] * 0.5  # ensure non-neg
        # Decrease both i_star and j_star if they are same then just decrease once
        a_new = a.copy()
        a_new[i_star] -= delta
        if i_star != j_star:
            a_new[j_star] -= delta
        # Distribute the total removed mass (2*delta (or delta if same)) to random indices
        removed = delta if i_star == j_star else 2*delta

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Generate a non‑negative step sequence that tries to minimise the
    `evaluate_sequence` score.  The algorithm performs a simple
    projected‑gradient / simulated‑annealing search on the simplex:
    it repeatedly finds the offset with the maximal convolution value,
    removes a small amount of mass from the pair that contributes most
    to that offset and redistributes the mass elsewhere.  The sum of the
    coefficients is kept equal to one (the objective is scale‑invariant)
    and the best sequence seen during the allotted time is returned.
    """
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # helper: maximal value of self‑convolution via FFT
    def conv_max(arr):
        n = arr.shape[0]
        L = 1 << ((2 * n - 1).bit_length())
        fft = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
        # tiny negative values coming from rounding are set to zero
        conv = np.maximum(conv, 0.0)
        return conv.max(), conv

    # start from a moderately sized sequence – larger than the trivial
    # case but still cheap enough for many FFTs.
    n = 2000
    # initialise a random point on the probability simplex
    a = np.random.rand(n)
    a /= a.sum()

    # keep the best sequence found so far
    best_a = a.copy()
    best_max, _ = conv_max(a)
    best_score = 2.0 * n * best_max   # because sum(a)=1

    iteration = 0
    while time.time() < deadline:
        cur_max, cur_conv = conv_max(a)
        # pick a random index among the maximal entries
        ks = np.where(cur_conv == cur_max)[0]
        k = int(np.random.choice(ks))

        # all (i,j) with i+j = k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_vals = np.arange(i_low, i_high + 1)
        j_vals = k - i_vals

        # contribution of each pair
        products = a[i_vals] * a[j_vals]
        idx = int(np.argmax(products))
        i_star = i_vals[idx]
        j_star = j_vals[idx]

        # a small amount of mass is moved away from the most
        # offending pair and added somewhere else.
        step_frac = 0.02          # 2 % of the source entry
        delta = step_frac * a[i_star]
        # protect against taking more mass than is available
        if i_star == j_star:
            delta = min(delta, a[i_star] * 0.5)
            a_new = a.copy()
            a_new[i_star] -= delta
            # choose a different index to receive the mass
            r = np.random.randint(n)
            while r == i_star:
                r = np.random.randint(n)
            a_new[r] += delta
        else:
            # both entries must stay non‑negative
            delta = min(delta, a[i_star], a[j_star])
            a_new = a.copy()
            a_new[i_star] -= delta
            a_new[j_star] -= delta
            r = np.random.randint(n)
            while r == i_star or r == j_star:
                r = np.random.randint(n)
            a_new[r] += 2.0 * delta

        # project back onto the simplex (non‑negative & sum = 1)
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s

        new_max, _ = conv_max(a_new)
        new_score = 2.0 * n * new_max

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f'Iter {iteration:5d}  n={n}  best score={best_score:.6f}')
        # accept the move if it does not worsen the current state
        if new_score <= 2.0 * n * cur_max:
            a = a_new

"""
Simple projected‑gradient / simulated‑annealing search.

The objective of `evaluate_sequence` is
    score = 2 * n * max( a * a ) / (sum(a)**2)

It is homogeneous: scaling all entries does not change the score.
Therefore we keep the vector on the probability simplex (sum = 1) and
optimise only the shape.

The algorithm repeatedly
  1. computes the self‑convolution of the current sequence (FFT for speed);
  2. finds the offset where the convolution attains its maximum;
  3. among the pairs (i, j) contributing to that offset it selects the
     pair with the largest product a_i * a_j;
  4. removes a small fraction of mass from that pair and redistributes it
     to a random other coordinate, projecting back onto the simplex;
  5. accepts the move if the score does not increase (occasional uphill
     moves are allowed to escape plateaus).

The best sequence encountered before the time budget expires is returned.
"""
import time
import random
import numpy as np


def _conv_max(arr: np.ndarray):
    """
    Return (max_value, full_convolution) of the linear convolution
    of `arr` with itself, computed via FFT (fast for the lengths we use).
    Tiny negative values coming from rounding are clamped to zero.
    """
    n = arr.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)                  # guard against -0.0… due to round‑off
    return conv.max(), conv


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a non‑negative step sequence that (hopefully) lowers the
    value returned by `evaluate_sequence`.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5      # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. choose a problem size.  2000 works well: large enough to improve
    #    over the uniform construction, yet cheap enough for thousands of
    #    FFT‑based evaluations per second.
    # ------------------------------------------------------------------
    n = 2000

    # initialise a random point on the simplex (all entries > 0)
    a = np.random.rand(n)
    a /= a.sum()

    # keep track of the best solution seen
    best_a = a.copy()
    best_max, _ = _conv_max(a)
    best_score = 2.0 * n * best_max               # because sum(a) == 1

    iteration = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 2. evaluate the current state
        # --------------------------------------------------------------
        cur_max, cur_conv = _conv_max(a)

        # pick a random location among the maximal entries
        ks = np.where(cur_conv == cur_max)[0]
        k = int(np.random.choice(ks))

        # all index pairs (i, j) with i + j = k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_vals = np.arange(i_low, i_high + 1)
        j_vals = k - i_vals

        # product contributed by each pair
        products = a[i_vals] * a[j_vals]
        idx = int(np.argmax(products))            # the offending pair
        i_star = i_vals[idx]
        j_star = j_vals[idx]

        # --------------------------------------------------------------
        # 3. propose a small redistribution of mass away from that pair
        # --------------------------------------------------------------
        step_frac = 0.02                          # move ~2 % of the source mass
        delta = step_frac * a[i_star]

        if i_star == j_star:
            # the pair reduces to a single entry (i = j)
            delta = min(delta, a[i_star] * 0.5)
            a_new = a.copy()
            a_new[i_star] -= delta
            # give the removed mass to a random other coordinate
            r = np.random.randint(n)
            while r == i_star:
                r = np.random.randint(n)
            a_new[r] += delta
        else:
            # ensure we do not drive either entry negative
            delta = min(delta, a[i_star], a[j_star])
            a_new = a.copy()
            a_new[i_star] -= delta
            a_new[j_star] -= delta
            r = np.random.randint(n)
            while r == i_star or r == j_star:
                r = np.random.randint(n)
            a_new[r] += 2.0 * delta

        # project back onto the simplex (non‑negative, sum = 1)
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s

        # --------------------------------------------------------------
        # 4. evaluate the proposal
        # --------------------------------------------------------------
        new_max, _ = _conv_max(a_new)
        new_score = 2.0 * n * new_max

        # keep the global best
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f'Iter {iteration:5d} | n={n} | best score = {best_score:.6f}')

        # accept the move if it does not increase the current score;
        # very occasional uphill moves help to escape flat regions.
        if new_score <= 2.0 * n * cur_max:
            a = a_new
        elif np.random.rand() < 0.001:
            a = a_new

        iteration += 1

        # occasional tiny random jitter to avoid stagnation
        if iteration % 5000 == 0:
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            tot = a.sum()
            if tot > 0:
                a /= tot

    # Return the best sequence found (as a plain Python list).
    return best_a.tolist()
