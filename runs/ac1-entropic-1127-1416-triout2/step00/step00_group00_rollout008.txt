import numpy as np
import time
from scipy.optimize import linprog

def normalize_sequence(seq, target_sum=None):
    """Scale seq so that its L1 sum equals target_sum (default sqrt(2*len(seq)))."""
    n = len(seq)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    s = np.sum(seq)
    if s == 0:
        return np.full(n, target_sum / n, dtype=float)
    return seq * (target_sum / s)

def max_conv(seq):
    """Return the maximum value of the autocorrelation (full convolution)."""
    b = np.convolve(seq, seq)
    return float(b.max()), b

def solve_convolution_lp(f_seq, rhs):
    """Linear program: maximize sum(b) s.t. conv(f, b) <= rhs, b >= 0."""
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum(b)
    # Build convolution constraints: for each k from 0 to 2n-2
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'time_limit': 2.0, 'disp': False})
    if res.success:
        return res.x
    return None

def get_lp_direction(seq, t=0.2):
    """Return a new sequence obtained by a convex combination with an LP‑generated direction."""
    n = len(seq)
    seq_sum = np.sum(seq)
    # normalise to the canonical sum sqrt(2 n) – this does not change the objective
    norm_seq = seq * (np.sqrt(2.0 * n) / seq_sum)
    rhs = max_conv(norm_seq)[0]
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None
    # scale g to the same L1 norm as the reference
    g = g * (np.sqrt(2.0 * n) / np.sum(g))
    # convex combination
    new_seq = (1 - t) * seq + t * g
    return normalize_sequence(new_seq)

    n = len(seq)

    mask = (j_candidates >= 0) & (j_candidates < n)

        # fall back to a random tiny redistribution
        i = rng.integers(n)
        delta = seq[i] * rng.uniform(0.0, 0.1)
        l = rng.integers(n)
        while l == i:
            l = rng.integers(n)
        new_seq = seq.copy()
        new_seq[i] = max(0.0, new_seq[i] - delta)
        new_seq[l] += delta
        return normalize_sequence(new_seq, target_sum)
    # Pick a random i from the valid set

    j = int(k - i)

    # keep the total L1 norm fixed
    return normalize_sequence(new_seq, target_sum)

import numpy as np
import time
from scipy.optimize import linprog


def normalize_sequence(seq, target_sum=None):
    """
    Scale a sequence so that its L1 norm equals *target_sum*.
    If *target_sum* is None the canonical value sqrt(2·n) is used.
    """
    n = len(seq)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    s = np.sum(seq)
    if s == 0:
        return np.full(n, target_sum / n, dtype=float)
    return seq * (target_sum / s)


def max_conv(seq):
    """
    Return the maximum value of the full (linear) convolution of *seq* with itself
    together with the whole convolution array.
    """
    b = np.convolve(seq, seq)
    return float(b.max()), b


def solve_convolution_lp(f_seq, rhs):
    """
    Linear programme:
        maximise   sum(b)
        subject to  conv(f_seq, b) <= rhs   (for every shift)
                    b >= 0
    Returns the optimal *b* as a NumPy array or ``None`` on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                       # maximise sum(b)  <=> minimise -sum(b)
    a_ub = []
    b_ub = []
    # convolution constraints – one per possible shift (0 … 2n‑2)
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # ``highs`` is fast and reliable; we restrict the run time per call.
    res = linprog(c, A_ub=a_ub, b_ub=b_ub,
                  method='highs',
                  options={'time_limit': 2.0, 'disp': False})
    if res.success:
        return res.x
    return None


def get_lp_direction(seq, t=0.2):
    """
    Compute a new direction by solving the LP described in the statement.
    Return a convex combination (1‑t)·seq + t·g, renormalised to the canonical L1‑norm.
    If the LP fails, ``None`` is returned.
    """
    n = len(seq)
    # normalise to the standard L1 norm – the objective is scale‑invariant
    ref = normalize_sequence(seq, np.sqrt(2.0 * n))
    rhs, _ = max_conv(ref)
    g = solve_convolution_lp(ref, rhs)
    if g is None:
        return None
    # bring g to the same L1 magnitude as the reference vector
    g = normalize_sequence(g, np.sqrt(2.0 * n))
    new_seq = (1.0 - t) * seq + t * g
    return normalize_sequence(new_seq)


def hill_climb_step(seq, rng, target_sum):
    """
    Perform a tiny mass‑transfer move aimed at reducing the current peak of the
    autocorrelation.  The move is always followed by a renormalisation to *target_sum*.
    """
    n = len(seq)
    max_val, _ = max_conv(seq)
    # shift where the convolution attains its maximum
    k = int(np.argmax(np.convolve(seq, seq)))
    # indices i, j with i + j = k
    i_candidates = np.arange(n)
    j_candidates = k - i_candidates
    mask = (j_candidates >= 0) & (j_candidates < n)
    i_valid = i_candidates[mask]
    if i_valid.size == 0:
        # fallback – random tiny redistribution
        i = rng.integers(n)
        delta = seq[i] * rng.uniform(0.0, 0.1)
        l = rng.integers(n)
        while l == i:
            l = rng.integers(n)
        new_seq = seq.copy()
        new_seq[i] = max(0.0, new_seq[i] - delta)
        new_seq[l] += delta
        return normalize_sequence(new_seq, target_sum)

    i = rng.choice(i_valid)
    j = int(k - i)
    # Choose the larger entry of the pair as donor
    donor = i if seq[i] >= seq[j] else j
    # Pick a receiver different from both i and j
    l = rng.integers(n)
    while l == donor or l == (j if donor == i else i):
        l = rng.integers(n)

    delta = seq[donor] * rng.uniform(0.0, 0.1)   # up to 10 % of the donor mass
    new_seq = seq.copy()
    new_seq[donor] = max(0.0, new_seq[donor] - delta)
    new_seq[l] += delta
    return normalize_sequence(new_seq, target_sum)
