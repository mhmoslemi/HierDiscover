def simplex_projection(v):
    """Project vector v onto the probability simplex {x>=0, sum=1}."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimises
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works on the observation that the denominator is
scale‑invariant (the score does not change if all entries are multiplied
by the same factor).  Hence we can fix the ℓ1‑norm of the vector to 1
and only try to make the autocorrelation (full convolution) as flat as
possible.  For a vector a of length n the average value of the
convolution is

    avg = (sum(a)**2) / (2*n-1) = 1/(2*n-1)                (after projection)

If all convolution entries were equal to this average the score would be
≈ 2*n*avg = 2*n/(2*n-1) ≈ 1, far below the current best bound 2.
The code therefore performs a projected gradient descent on the
objective

    F(a) = ∑_k (conv_k - avg)²                (L2‑variance of the convolution)

Gradient of F can be written in closed form using a correlation,
which we compute efficiently with FFTs.  After each gradient step we
project the vector onto the probability simplex (non‑negative, sum=1).
Multiple random restarts with different lengths are carried out until
the time budget expires; the best vector seen is returned.

The implementation uses only NumPy (no external solver) and respects
the 1 CPU limitation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}.
    Returns a new array with the same shape as v.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays.")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest ρ satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All components are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of real vectors a and b using FFT.
    Result length = len(a) + len(b) - 1.
    """
    n = a.shape[0] + b.shape[0] - 1
    size = _next_pow2(n)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)[:n]
    return conv


def _convolution_and_gradient(a: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the full convolution of a with itself and the gradient of
    the L2 variance of the convolution w.r.t. a (under the simplex
    constraint).  The gradient is returned *before* any projection.
    """
    n = a.shape[0]
    # Full autocorrelation a*a
    conv = _fft_convolve(a, a)               # length 2n‑1
    avg = 1.0 / (2 * n - 1)                  # because sum(a) = 1
    diff = conv - avg                        # shape (2n‑1,)

    # Correlation a ⋆ diff = conv(a, reversed(diff))
    corr = _fft_convolve(a, diff[::-1])      # length 3n‑2
    offset = diff.shape[0] - 1               # = 2n‑2
    # Gradient of Σ (conv‑avg)² is 2 * (a ⋆ diff) shifted appropriately
    grad = 2.0 * corr[offset: offset + n]    # shape (n,)
    return conv, grad

    # The evaluation routine also applies its own clipping and non‑negativity,

    return evaluate_sequence(a.tolist())

        a = _project_onto_simplex(a)

        # --- Gradient descent parameters --------------------------------------------
        lr = 0.6                     # initial learning rate
        decay = 0.9995               # multiplicative decay per iteration
        max_inner = 2000            # safety bound; we break earlier if time is low

        for it in range(max_inner):
            if time.time() >= deadline:
                break

            conv, grad = _convolution_and_gradient(a)

            # Simple gradient step followed by projection onto simplex
            a = a - lr * grad
            a = _project_onto_simplex(a)

            # Schedule learning‑rate decay
            lr *= decay

            # Occasionally evaluate the true score
            if (it & 31) == 0:        # every 32nd iteration
                score = _evaluate(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.2f}] New best: score={score:.6f} "
                        f"len={n} iter={iteration}"
                    )
                # Early exit if we already beat the target
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    return best_seq.tolist()
            iteration += 1

        # small safety pause to avoid overly long inner loops that starve the outer loop
        if time.time() + 0.1 >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should not happen) use a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0 / n)
        best_score = _evaluate(best_seq)
        print("Fallback to uniform sequence, score =", best_score)

    print(f"Finished search. Best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
