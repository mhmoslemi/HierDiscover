def exponential_sequence(n, lam):
    # lam must be positive and less than n
    r = 1.0 - lam / n
    if r <= 0:
        return None
    # generate raw decreasing seq r**i
    raw = np.power(r, np.arange(n, dtype=np.float64))
    seq = raw / raw.sum()  # normalize sum to 1
    # enforce non-negative, returns list of floats
    return seq.tolist()
```

Implementation of simplex projection:

```
def simplex_projection(v):
    # based on algorithm by Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Ensures sum=1.

Now subgradient descent loop:

```
def refine_sequence(seq, iterations=200, eta_start=0.1, eta_decay=0.99):
    a = np.array(seq, dtype=np.float64)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    eta = eta_start
    for it in range(iterations):
        conv = np.convolve(a, a)
        k_star = np.argmax(conv)
        grad = np.zeros_like(a)
        # For each i, if 0 <= k_star - i < n:
        # assign grad[i] = a[k_star - i]
        # Use vectorized: indices = np.arange(n); partner = k_star - indices; mask = (partner>=0) & (partner<n); grad[mask] = a[partner[mask]]
        indices = np.arange(len(a))
        partner = k_star - indices
        mask = (partner >= 0) & (partner < len(a))
        grad[mask] = a[partner[mask]]
        # Multiply by 2 as derivative equals 2*a[partner], but scaling constant not needed for descent direction
        grad = 2 * grad
        a = a - eta * grad
        # Simplex projection
        a = simplex_projection(a)
        # Evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # Decay step
        eta *= eta_decay
    return best_a, best_score
```

def project_simplex(v):
    # Using stable algorithm
    if np.all(v >=0) and abs(v.sum() - 1) < 1e-9:
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho)==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def optimize_sequence_via_softmax(n, iterations, lr, beta_start, beta_end):
    a = np.random.rand(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    beta = beta_start
    for it in range(iterations):
        conv = np.convolve(a, a)
        # Compute softmax weights
        exponents = beta * conv
        max_exp = np.max(exponents)
        w = np.exp(exponents - max_exp)  # shift
        w = w / w.sum()
        # gradient = 2 * sum_i w_i * shifted a
        grad = np.zeros_like(a)
        # Vectorized loops:
        for i in range(2*n - 1):
            wi = w[i]
            if wi == 0:
                continue
            # contributions to grad for j where 0 <= i - j < n
            # Means j in [i-n+1, i] intersect [0, n-1]
            j_start = max(0, i - (n - 1))
            j_end = min(i, n - 1)
            # For j in that range, k = i - j
            # a[k] element used
            # grad[j] += 2 * a[i - j] * wi
            # we can slice arrays
            js = np.arange(j_start, j_end + 1)
            ks = i - js
            grad[js] += 2 * a[ks] * wi
        a = a - lr * grad
        a = project_simplex(a)
        # evaluate periodically
        if (it+1) % 10 == 0 or it == iterations-1:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
        # slowly increase beta
        if beta < beta_end:
            beta *= (beta_end / beta_start) ** (1.0 / iterations)  # exponential schedule
    return best_a, best_score
```

    np.random.seed(seed)
    a = np.random.rand(n)

    best_a = a.copy()

    beta = 1.0

        conv = np.convolve(a, a)

"""
Propose a good step‑function sequence.

The core idea is a fast stochastic search that explores a few
parameterised families (single‑exponential, two‑exponential mixtures)
and then refines the best candidate with a simple simulated‑annealing
procedure.  All candidates are normalised to have a unit sum – the
evaluation function is invariant under scaling, so this guarantees a
valid (non‑zero) input while keeping the search space bounded.

The algorithm works completely within the given time budget,
keeps the best sequence found and finally returns it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees: v >= 0 and sum(v) == 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ r^i with r = 1 - lam/n (lam>0, r>0)."""
    if lam <= 0 or lam >= n:
        return None
    r = 1.0 - lam / n
    if r <= 0:
        return None
    i = np.arange(n, dtype=np.float64)
    raw = r ** i
    if raw.sum() == 0:
        return None
    return raw / raw.sum()


def _two_exponential_sequence(n: int, lam1: float, lam2: float, w: float) -> np.ndarray:
    """
    a_i = w * r1^i + (1-w) * r2^i  with r_k = 1 - lam_k/n.
    The vector is normalised to sum to 1.
    """
    if not (0 < w < 1):
        return None
    if lam1 <= 0 or lam2 <= 0 or max(lam1, lam2) >= n:
        return None
    r1 = 1.0 - lam1 / n
    r2 = 1.0 - lam2 / n
    if r1 <= 0 or r2 <= 0:
        return None
    i = np.arange(n, dtype=np.float64)
    raw = w * (r1 ** i) + (1.0 - w) * (r2 ** i)
    if raw.sum() == 0:
        return None
    return raw / raw.sum()


def _beta_sequence(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta‑like shape:  (i+1)^(a-1) * (n-i)^(b-1)
    (both a,b > 0). Normalised to unit sum.
    """
    if a <= 0 or b <= 0:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)
    raw = (i ** (a - 1)) * ((n - i + 1) ** (b - 1))
    if raw.sum() == 0:
        return None
    return raw / raw.sum()


def _anneal_sequence(init_seq: np.ndarray,
                     max_iter: int,
                     start_temp: float,
                     end_temp: float) -> np.ndarray:
    """
    Simple simulated annealing that perturbs two coordinates while
    keeping the vector on the simplex.  The objective is the
    `evaluate_sequence` value (lower is better).
    """
    n = init_seq.shape[0]
    cur = init_seq.copy()
    best = cur.copy()
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:  # safety fallback
        best_score = np.inf

    for it in range(1, max_iter + 1):
        # temperature schedule (geometric cooling)
        temp = start_temp * (end_temp / start_temp) ** (it / max_iter)

        # propose a move: transfer a small amount from i to j
        i, j = np.random.choice(n, 2, replace=False)
        delta = cur[i] * 0.05 * (np.random.rand() - 0.5)  # up to ±2.5 % of a_i
        if delta == 0.0:
            continue

        new = cur.copy()
        new[i] = max(0.0, new[i] - delta)
        new[j] = max(0.0, new[j] + delta)

        # re‑project onto the simplex (guarantees sum==1 & non‑negative)
        new = _project_to_simplex(new)

        try:
            new_score = evaluate_sequence(new.tolist())
        except Exception:
            continue

        # Metropolis acceptance
        if new_score < best_score or np.random.rand() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            cur = new
            if new_score < best_score:
                best_score = new_score
                best = new
        # occasional hard reset to escape very bad regions
        if it % 200 == 0 and np.random.rand() < 0.1:
            cur = np.random.rand(n)
            cur = _project_to_simplex(cur)

    return best

# Main search routine (exposed as required)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value step‑function sequence within the given time budget.
    Returns a Python list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 3.0   # keep a small safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 1) Try known constructive families (single/dual exponential,
    #    beta‑like shapes).  Simple grid / random sweep.
    # --------------------------------------------------------------
    lengths = [200, 400, 800, 1200, 2000]          # moderate sizes, fast evaluation
    for n in lengths:
        # single exponential sweep
        for lam in np.linspace(0.5, 5.0, 12):
            if time.time() > deadline:
                break
            seq = _exponential_sequence(n, lam)
            if seq is None:
                continue
            try:
                sc = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
                print(f"[exp] n={n} lam={lam:.2f} score={sc:.6f}")

        # beta‑like shapes (a few hand‑picked parameters)
        for a_par, b_par in [(1.0, 1.0), (0.7, 1.2), (1.5, 0.8), (2.0, 2.0)]:
            if time.time() > deadline:
                break
            seq = _beta_sequence(n, a_par, b_par)
            if seq is None:
                continue
            try:
                sc = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
                print(f"[beta] n={n} a={a_par} b={b_par} score={sc:.6f}")

        # random two‑exponential mixtures
        while time.time() < deadline:
            lam1 = np.random.uniform(0.2, 4.0)
            lam2 = np.random.uniform(lam1 + 0.1, 8.0)
            w = np.random.rand()
            seq = _two_exponential_sequence(n, lam1, lam2, w)
            if seq is None:
                continue
            try:
                sc = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
                print(f"[mix] n={n} lam1={lam1:.2f} lam2={lam2:.2f} w={w:.2f} score={sc:.6f}")
            # limit the number of random samples per length
            if np.random.rand() < 0.01:   # roughly 1 in 100 iterations exits
                break

    # --------------------------------------------------------------
    # 2) Refine the current best candidate with a tiny annealing run.
    # --------------------------------------------------------------
    if best_seq is not None:
        init = np.array(best_seq, dtype=np.float64)
        # keep the length modest for the annealer – very fast iterations
        refined = _anneal_sequence(
            init_seq=init,
            max_iter=800,
            start_temp=1.0,
            end_temp=0.001,
        )
        try:
            refined_score = evaluate_sequence(refined.tolist())
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined.tolist()
                print(f"[anneal] improved to {refined_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3) Fallback – if nothing succeeded (unlikely), return a short uniform.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 100
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print("[fallback] using uniform sequence")

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq
