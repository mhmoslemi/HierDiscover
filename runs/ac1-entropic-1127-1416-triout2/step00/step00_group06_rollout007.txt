"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the ordinary discrete convolution (numpy.convolve).
The evaluation function normalises the sum to √(2n), therefore the
objective is simply the maximal convolution entry.

The algorithm consists of two phases:

1. **Global initialisation** – try a family of exponentially decaying
   sequences  a_i = C·r**i  (r∈(0,1)), compute the objective for a fine
   grid of r and keep the best one.

2. **Local hill‑climbing** – repeatedly pick the index that contributes
   most to the current maximal convolution coefficient and move a small
   amount of mass to an index that contributes the least.  After each
   transfer the sequence is re‑normalised to keep the exact sum
   √(2n).  Only moves that strictly decrease the maximal convolution
   coefficient are accepted; otherwise the move is undone.
   Random jitter is injected occasionally to escape plateaus.

The method uses only numpy (FFT for fast convolution) and runs until the
time budget (≈1000 s) is exhausted, always returning the best sequence
found.  The code is deliberately written without closures or lambdas,
as required by the competition framework.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _max_convolution(seq: np.ndarray) -> float:
    """
    Return max(conv(seq, seq)) using an FFT of size power‑of‑two.
    The sequence is assumed to be non‑negative and sum‑normalised,
    therefore this value equals the evaluation score.
    """
    n = len(seq)
    # size is next power of two ≥ 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    # keep only the first 2*n-1 entries
    return float(conv[: 2 * n - 1].max())


def _exponential_sequence(n: int, r: float, target_sum: float) -> np.ndarray:
    """
    Generate a_i = C·r**i with sum = target_sum.
    """
    # avoid r very close to 1 that would cause division by zero
    if r <= 0.0 or r >= 1.0:
        raise ValueError("r must be in (0,1)")
    # geometric series sum = (1 - r**n)/(1 - r)
    geom = (1.0 - r ** n) / (1.0 - r)
    C = target_sum / geom
    return C * (r ** np.arange(n))


def _improve_by_pairwise_transfer(
    seq: np.ndarray, target_sum: float, deadline: float, best_obj: float,
    best_seq: np.ndarray,
):
    """
    Perform the hill‑climbing described in the module docstring.
    Returns possibly improved (best_seq, best_obj).
    """
    n = len(seq)
    rng = np.random.default_rng()
    no_improve = 0
    iteration = 0
    size = 1 << ((2 * n - 1).bit_length())  # FFT size (constant)

    while time.time() < deadline:
        iteration += 1

        # ----- current max convolution -----
        fft_seq = np.fft.rfft(seq, n=size)
        conv = np.fft.irfft(fft_seq * fft_seq, n=size)
        conv = conv[: 2 * n - 1]
        cur_max = float(conv.max())
        k_max = int(conv.argmax())

        # ----- contributions to the maximal coefficient -----
        idx = np.arange(n)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        contributions = np.zeros(n, dtype=seq.dtype)
        contributions[valid] = seq[valid] * seq[j[valid]]

        # pick source (largest contribution) and destination (smallest)
        i_src = int(contributions.argmax())
        i_dst = int(contributions.argmin())
        if i_src == i_dst:
            i_dst = int(rng.integers(0, n - 1))
            if i_dst == i_src:
                i_dst = (i_dst + 1) % n

        # amount to move – a small fraction of the source entry
        delta = min(0.1 * seq[i_src], 0.02 * target_sum)
        if delta <= 0.0:
            break

        # propose move
        seq[i_src] -= delta
        seq[i_dst] += delta

        # re‑normalise to keep the exact sum (protect against round‑off)
        cur_sum = seq.sum()
        if cur_sum != 0.0:
            seq *= (target_sum / cur_sum)

        # evaluate new max
        fft_seq = np.fft.rfft(seq, n=size)
        conv_new = np.fft.irfft(fft_seq * fft_seq, n=size)
        new_max = float(conv_new[: 2 * n - 1].max())

        if new_max < cur_max:          # accept improvement
            no_improve = 0
            if new_max < best_obj:
                best_obj = new_max
                best_seq = seq.copy()
                print(f"[{iteration:6d}] new best max={best_obj:.6f}")
        else:                           # reject – revert
            no_improve += 1
            seq[i_dst] -= delta
            seq[i_src] += delta
            # keep the sum exactly correct
            cur_sum = seq.sum()
            if cur_sum != 0.0:
                seq *= (target_sum / cur_sum)

        # occasional random jitter to escape local minima
        if iteration % 2000 == 0:
            jitter = rng.normal(0.0, target_sum * 0.001, size=n)
            seq = np.clip(seq + jitter, 0.0, None)
            cur_sum = seq.sum()
            if cur_sum != 0.0:
                seq *= (target_sum / cur_sum)

        # stop early if the target is already reached
        if best_obj <= 1.503:
            break

    return best_seq, best_obj
