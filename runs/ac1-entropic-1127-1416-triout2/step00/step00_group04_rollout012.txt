"""
Propose a candidate step‑function using a hybrid of a linear‑programming
direction finder (the same idea as the published SOTA) together with a
lightweight stochastic local search.  

* Start from either the provided `height_sequence_1` or a random
  distribution of length 200–600 (all values are non‑negative).
* Re‑scale the current sequence so that its ℓ₁ norm equals √(2·n)
  – this does not change the evaluation score but improves the numerical
  stability of the LP.
* Solve the LP  

      max   Σ b_j
      s.t.  (f * b)(k) ≤ M   (for every convolution index k)
            b_j ≥ 0

  where `f` is the normalized current sequence and `M` is its
  current worst‑case convolution value.  The LP returns a new direction
  `g`.  A convex combination ` (1‑t)·f + t·g ` reduces the
  max‑convolution for a suitable step size `t`.

* Try several step sizes (`t` in {0.01,0.05,0.1,0.2,0.3}) and keep the
  best candidate.
* If the LP fails, perform a small random perturbation of a few
  coefficients.
* Keep the best sequence found so far; occasionally accept a worse
  candidate with a tiny probability to escape local minima.
* Stop after the allotted time budget (default 1000 s) and return the
  best sequence.

The algorithm is intentionally lightweight so that many LP solves can be
performed inside the time limit, typically driving the score well below
the naive value `2.0`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the linear program that finds a good direction g.
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs):
    """
    Solve   max Σ b_j   s.t.   (f * b)(k) ≤ rhs   for all k,
    b_j ≥ 0.
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # objective: maximize sum(b)  →  minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i + j = k, j = k‑i
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        for i in range(i_low, i_high + 1):
            j = k - i
            row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity of b
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # Use HiGHS (the default) – limit the LP time so we never block.
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method='highs',
        options={'presolve': True, 'time_limit': 5.0, 'disp': False},
    )
    if res.success:
        # Numerical noise can produce tiny negatives; clip to zero.
        return np.clip(res.x, 0.0, None)
    return None

# ----------------------------------------------------------------------
def move_towards_direction(seq, g, t_vals):
    """
    Return the best convex combination (1‑t)·seq + t·g among the
    supplied step sizes `t_vals`.  The function evaluates each candidate
    with `evaluate_sequence` and picks the one with the lowest score.
    """
    best_seq = None
    best_score = float('inf')
    for t in t_vals:
        cand = [(1 - t) * a + t * b for a, b in zip(seq, g)]
        try:
            sc = evaluate_sequence(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        except Exception:
            # Should never happen – the evaluate routine is robust.
            continue
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple random perturbation (used when the LP does not return a direction).
# ----------------------------------------------------------------------
def random_perturb(seq, sigma=0.02):
    """
    Add a small Gaussian noise to a few random entries and re‑normalise
    the ℓ₁ norm to keep the scale unchanged.
    """
    n = len(seq)
    new = np.array(seq, copy=True)
    # perturb up to three indices (more would be wasteful)
    idx = np.random.choice(n, size=min(3, n), replace=False)
    for i in idx:
        # relative noise – keep the sign positive after clipping
        new[i] = max(0.0, new[i] + np.random.randn() * sigma * (new[i] + 1e-6))
    # re‑scale to original ℓ₁ norm (scale does not affect the score)
    total = new.sum()
    if total > 0:
        new *= (np.sum(seq) / total)
    return new.tolist()


# ----------------------------------------------------------------------
# Core search routine required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑score step‑function sequence.
    The function respects the given time budget and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Initialise – either from the provided global or random.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` may be defined by the harness.
        cur_seq = list(height_sequence_1)
        print("Starting from provided height_sequence_1 (len={})".format(len(cur_seq)))
    except Exception:
        # Random start with a length between 250 and 500.
        n = np.random.randint(250, 501)
        cur_seq = np.random.rand(n) + 0.1   # avoid exact zeros
        cur_seq = cur_seq.tolist()
        print("Starting from a fresh random sequence (len={})".format(n))

    # Normalise ℓ₁ norm to √(2·n); scaling does not affect the final score.
    cur_seq = np.array(cur_seq, dtype=float)
    cur_seq = cur_seq * (np.sqrt(2 * len(cur_seq)) / cur_seq.sum())
    cur_seq = cur_seq.tolist()

    best_seq = cur_seq.copy()
    best_score = evaluate_sequence(best_seq)
    print(f"Initial score = {best_score:.6f}")

    # step‑size list used in the convex combination
    step_list = [0.01, 0.05, 0.1, 0.2, 0.3]

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 2️⃣  Find a better direction with the LP.
        # --------------------------------------------------------------
        n = len(cur_seq)
        total = np.sum(cur_seq)
        if total == 0:
            # safeguard – should never happen
            cur_seq = np.random.rand(n).tolist()
            total = np.sum(cur_seq)

        # Normalise so that the LP works with a modest scale.
        f_norm = (np.array(cur_seq) * (np.sqrt(2 * n) / total)).tolist()
        rhs = float(np.max(np.convolve(f_norm, f_norm)))

        g_dir = solve_convolution_lp(f_norm, rhs)

        if g_dir is not None:
            # Scale the direction to the same ℓ₁ norm as `cur_seq`.
            g_norm = g_dir * (np.sqrt(2 * n) / g_dir.sum())
            # Try several step sizes and keep the best.
            cand_seq, cand_score = move_towards_direction(cur_seq, g_norm.tolist(), step_list)
        else:
            # LP failed – fall back to a tiny random tweak.
            cand_seq = random_perturb(cur_seq, sigma=0.02)
            cand_score = evaluate_sequence(cand_seq)

        # --------------------------------------------------------------
        # 3️⃣  Accept / remember the best candidate.
        # --------------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            cur_seq = cand_seq   # continue from the improvement
            print(f"Iter {iteration:5d}: new best {best_score:.6f}  (len={len(best_seq)})")
        else:
            # occasionally move to a worse point to escape plateaus
            if np.random.rand() < 0.02:
                cur_seq = cand_seq
            # otherwise keep the current sequence unchanged.

        # ----------------------------------------------------------------
        # 4️⃣  Occasionally mutate the whole vector to avoid stagnation.
        # ----------------------------------------------------------------
        if iteration % 120 == 0:
            # a modest global jitter
            cur_seq = random_perturb(cur_seq, sigma=0.03)

    print("Search finished – best score found:", best_score)
    return best_seq
