"""
Uniform step function generator.

Mathematical insight:
For any non‑negative sequence a_0 … a_{n‑1},
the convolution value at shift 0 is
    C_0 = Σ a_i² .
By the QM–AM inequality,
    Σ a_i² ≥ (Σ a_i)² / n .
Because the evaluation function is
    2·n·max_k C_k / (Σ a_i)²
and max_k C_k ≥ C_0,
the value can never be smaller than 2.
A uniform sequence (all entries equal) attains C_0 = (Σ a_i)² / n
and therefore yields exactly the lower bound 2.
Thus the best possible score is 2.  This routine simply returns
a uniform sequence (length 1000, value 1.0) which achieves that optimum.
"""

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Returns a uniform step‑function sequence.
    
    The function ignores the time budget because the optimum (evaluation = 2)
    is already achieved analytically.  A sequence of length 1000 with all
    entries equal to 1.0 respects the clipping limits (≤1000) and yields the
    minimal possible evaluation score of 2.0.
    
    Parameters
    ----------
    seed : int, optional
        Random seed (unused, kept for compatibility).
    budget_s : int, optional
        Time budget in seconds (unused).
    **kwargs : dict
        Additional ignored keyword arguments.
    
    Returns
    -------
    list[float]
        Uniform sequence of length 1000.
    """
    # The exact length does not affect the score (uniform gives 2 for any n),
    # but we keep it moderate to avoid unnecessary computation.
    n = 1000
    value = 1.0  # any positive value ≤1000 works; 1.0 is simple.
    uniform_seq = [value] * n

    # Optional: quick sanity check (can be removed in production).

        from evaluate_sequence import evaluate_sequence  # type: ignore

        print(f"[INFO] Uniform sequence (n={n}) evaluation score: {score}")
    except Exception:

        pass

    return uniform_seq
