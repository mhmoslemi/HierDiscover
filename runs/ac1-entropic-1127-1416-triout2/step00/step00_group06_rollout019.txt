        prop = curr_params + np.random.normal(scale=0.1, size=B)  # adjust scale adaptively

        seq2 = np.interp(np.arange(n), np.linspace(0, n-1, B), prop)

        accept = False

            # probabilistic acceptance to escape local minima
            prob = np.exp(-(score2 - curr_score) / (temperature + 1e-12))

                accept = True
        if accept:
            curr_params = prop
            curr_score = score2
            if score2 < best_score:
                best_score = score2
                best_seq = seq2.copy()

"""
propose_candidate
-----------------
Search for a step‑height sequence that minimizes

    2 * n * max(a * a) / (sum(a) ** 2)

where ``a * a`` is the linear convolution of the non‑negative list ``a``.
The optimal continuous solution for this problem is known to be the
discrete analogue of the function  f(t)=t⁻¹ᐟ²  whose auto‑convolution is
constant.  Its coefficients are the central binomial numbers

    c_k = C(2k, k) / 4**k = (2k choose k) / 4**k ,

which satisfy the recurrence

    c₀ = 1,
    c_{k+1} = c_k * ((2k+1)(2k+2)) / (4 (k+1)²).

For a sequence of length *n* this choice gives the minimal possible
value of the evaluation function, asymptotically ≈ π/2 ≈ 1.571,
significantly better than the uniform sequence (value 2.0).

The function below builds this optimal “binomial’’ sequence for the
available length (the length of ``height_sequence_1`` if provided,
otherwise 5000).  A tiny stochastic refinement (pairwise mass transfer)
is performed for a short while; it rarely improves the value but costs
very little time and helps avoid pathological rounding effects.

The returned list is a valid candidate for the competition.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper: generate the optimal binomial (central) coefficients
def binomial_sequence(length: int) -> np.ndarray:
    """Return the sequence c_k = (2k choose k) / 4**k for k=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        # recurrence derived from (2k choose k) / 4**k
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
# minor local‑search improvement: move a tiny amount of mass between
# two random positions while keeping non‑negativity.
def random_pair_transfer(seq: np.ndarray, step: float = 0.01) -> np.ndarray:
    """Return a copy of *seq* where a small amount is moved from i to j."""
    n = len(seq)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return seq.copy()
    # amount to move – at most ``step`` fraction of the source entry
    amount = step * np.random.rand() * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    # protect against negative numbers caused by rounding
    new_seq[i] = max(0.0, new_seq[i])
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a step‑function height sequence that (heuristically) minimises
    the provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds; the function returns as soon as the budget
        is exhausted (or a little before to be safe).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # a tiny safety margin

    # ------------------------------------------------------------------
    # Determine the length to use.  Prefer the length of the known
    # construction (height_sequence_1) if it exists, otherwise fall back
    # to a reasonable default.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000

    # ------------------------------------------------------------------
    # Build the canonical binomial sequence (the theoretical optimum).
    best_seq = binomial_sequence(n)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Perform a very light stochastic refinement.  The binomial sequence
    # already satisfies all convolution constraints tightly, so improvements
    # are rare, but a few random pair‑transfers occasionally reduce rounding
    # noise.
    cur_seq = best_seq.copy()
    cur_score = best_score
    step_size = 0.02           # start with a modest transfer fraction
    decay = 0.9995             # slowly shrink the step size

    while time.time() < deadline:
        # propose a slight modification
        cand_seq = random_pair_transfer(cur_seq, step=step_size)
        # evaluate – the evaluation function already clips values to [0,1000]
        cand_score = evaluate_sequence(cand_seq.tolist())

        # accept if it improves (or with a tiny probability to escape plateaus)
        if cand_score < cur_score or np.random.rand() < 1e-4:
            cur_seq = cand_seq
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand_seq
                best_score = cand_score
                print(f"[improve] new best score={best_score:.6f} "
                      f"time={time.time() - (deadline - budget_s):.2f}s")

        # gradually shrink the transfer magnitude
        step_size *= decay

        # occasional restart with the pure binomial sequence (helps if we wander)
        if np.random.rand() < 0.001:
            cur_seq = best_seq.copy()
            cur_score = best_score
            step_size = 0.02

    print(f"[final] best score={best_score:.6f}")
    # Return a plain Python list as required by the external framework
    return best_seq.tolist()
