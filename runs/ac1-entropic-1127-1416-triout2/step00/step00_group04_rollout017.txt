Implement simplex_projection as described:

def simplex_projection(v):
   """Project onto probability simplex (nonnegative sum=1)."""
   n = len(v)
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   rho = np.where(u - cssv / np.arange(1,n+1) > 0)[0][-1]
   theta = cssv[rho] / (rho + 1)
   w = np.maximum(v - theta, 0)
   return w

def propose_candidate(budget_s=980, seed=42):
   np.random.seed(seed)
   start = time.time()
   deadline = start + budget_s

   best_seq = np.array(height_sequence_1, dtype=float)
   best_seq = simplex_projection(np.maximum(best_seq,0.0))
   best_score = evaluate_sequence(best_seq.tolist())

   # Also track maybe uniform.
   n = len(best_seq)
   uniform = np.full(n, 1.0/n)
   uniform_score = evaluate_sequence(uniform.tolist())
   if uniform_score < best_score:
        best_seq = uniform; best_score = uniform_score

   # Now iterative search
   seq = best_seq.copy()
   it = 0
   while time.time() < deadline:
        # compute conv, gradient
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        # maybe early stop if close to avg
        idx_max = np.where(conv == max_val)[0]
        # compute gradient
        n = len(seq)
        grad = np.zeros(n)
        for k in idx_max:
            lo = k - (n-1)
            if lo < 0: lo = 0
            hi = k
            if hi > n-1: hi = n-1
            # j range lo..hi
            # vectorize: neighbor indices = k - np.arange(lo, hi+1)
            j = np.arange(lo, hi+1)
            neighbor = k - j
            grad[lo:hi+1] += 2 * seq[neighbor]  # but need shape match
        # step size
        lr = 0.1 / np.sqrt(it + 1)
        seq_new = seq - lr * grad
        seq_new = simplex_projection(seq_new)
        # optionally momentum or random jitter where needed
        seq = seq_new

        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f'Iter {it}, new best {best_score:.6f}, max_conv {max_val:.6e}')
        it += 1

        # occasional random perturbation
        if it % 200 == 0:

def project_simplex(v):
    """Projects a vector onto the probability simplex (nonnegative, sum=1)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that minimises the given evaluation
function.  The algorithm builds an initial exponential (geometric) shape,
normalises it to a probability simplex and then performs a projected gradient
descent that tries to flatten the self‑convolution (make all convolution
entries close to their average).  The projected gradient step uses the exact
gradient of the squared error
    ∑_k ( (a*a)[k] - avg )²
which is
    4 * (a_rev  *  ((a*a) - avg) )  (restricted to the original length).
After each gradient step the vector is projected back onto the simplex
(non‑negative, sum = 1).  The best sequence seen during the timed search is
returned.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence for a vector that already sums to 1."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = np.convolve(a, a)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def _initial_exponential(n: int, c: float = 1.2) -> np.ndarray:
    """
    Return a normalised exponentially decreasing sequence of length n:
        a[i] ∝ exp(-c * i / n)
    """
    i = np.arange(n, dtype=float)
    r = np.exp(-c / n)
    a = r ** i
    a = a / a.sum()
    return a


def propose_candidate(budget_s: float = 990.0, seed: int = 42, **kwargs):
    """
    Search for a good height sequence within the allotted time.
    The routine builds an exponential start sequence and runs a
    projected gradient descent that flattens the autocorrelation.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + float(budget_s) - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose the length of the sequence.  If the global `height_sequence_1`
    # exists we reuse its length (≈ 7 000); otherwise we fall back to 2000.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))          # noqa: F821 (global variable)
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Initial guess – a mild exponential (c≈1.2) gives a modest improvement
    # over the uniform distribution and is a good starting point for the
    # flattening optimisation.
    # ------------------------------------------------------------------
    a = _initial_exponential(n, c=1.2)
    best_a = a.copy()
    best_score = _fast_score(best_a)

    # ------------------------------------------------------------------
    # Parameters for the gradient descent.
    # ------------------------------------------------------------------
    iteration = 0
    base_lr = 0.5          # initial learning rate (will decay)

    while time.time() < deadline:
        # Convolution of the current sequence with itself
        conv = np.convolve(a, a)                 # length 2n‑1
        avg = 1.0 / (2 * n - 1)                  # target flat value
        diff = conv - avg                         # how far each entry is

        # Gradient of  ∑ (conv - avg)²  w.r.t. a
        # = 4 * (a_rev * diff)   (restricted to the original n positions)
        a_rev = a[::-1]
        grad_full = np.convolve(a_rev, diff)     # length 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]   # slice to length n

        # Simple diminishing step size
        lr = base_lr / (1.0 + np.sqrt(iteration))
        a_next = a - lr * grad
        a_next = _project_simplex(a_next)

        # Evaluate the new candidate
        cur_score = _fast_score(a_next)

        # Keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_next.copy()
            print(f"Iter {iteration:5d} – new best score {best_score:.6f}")

        # Accept the step (the projection already keeps a feasible point)
        a = a_next
        iteration += 1

        # Occasionally inject a small random perturbation to escape plateaus
        if iteration % 2000 == 0:
            # random shuffle of a small amount of mass
            idx = np.random.randint(n, size=2)
            delta = (np.random.rand() - 0.5) * 0.02
            a[idx[0]] = max(0.0, a[idx[0]] - delta)
            a[idx[1]] = max(0.0, a[idx[1]] + delta)
            a = _project_simplex(a)

    # Return the best sequence found as a plain Python list
    return best_a.tolist()
