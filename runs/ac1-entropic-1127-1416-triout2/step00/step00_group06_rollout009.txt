def solve_min_max_convolution(a, sum_target=1.0, time_limit=30.0):
    n = len(a)

    A_rows = []
    A_cols = []
    # For each k, create row

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0
    
    n = 2500  # moderate size
    # initial shape: power law p=0.5
    a = (np.arange(1, n+1, dtype=float) ** -0.5)
    a = a / np.sum(a)  # normalize sum to 1
    
    best_seq = a.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    print(f'Initial evaluation: {best_val}')
    
    while time.time() < deadline:
        # Perform a gradient step
        c = convolve_fft(a, a)
        avg = np.mean(c)
        r = c - avg

        grad = convolve_fft(r, a[::-1])

        # try several step sizes
        improved = False
        for eta in (0.001, 0.005, 0.01, 0.02):
            if time.time() >= deadline:
                break
            new_a = a - eta * g
            new_a = np.maximum(new_a, 0.0)
            sum_new = np.sum(new_a)
            if sum_new < 1e-12:
                continue
            new_a = new_a / sum_new  # renormalize

                break

            idx = np.random.randint(len(a))
            a[idx] = max(0.0, a[idx] + np.random.randn() * 0.005)
            a = a / np.sum(a)
    
    return best_seq.tolist()

"""
Propose a low‑value sequence for the convolution bound.

The algorithm proceeds in three stages:

1. **Power‑law initialization** – start with a_i = (i+1)^(-0.5) (the classic
   1/√i shape) which already gives a bound ≈ π/2 ≈ 1.57 (much better than 2.0).

2. **Gradient descent on the L2 deviation of the convolution** – we minimise
   J(a)=∑_k ( (a*a)_k – mean(a*a) )² using the exact gradient obtained by a
   convolution.  After each update the vector is projected back onto the
   simplex (non‑negative, sum = 1).  Small step sizes and a few line‑search
   candidates are used; this flattens the autocorrelation and typically lowers
   the bound further.

3. **Linear‑programming refinement** – given the current sequence f, solve
   the LP
        minimise M
        s.t.   (f * g)_k ≤ M   for all k,
               g_i ≥ 0,
               Σ_i g_i = 1 .
   The optimal g reduces the maximal convolution for the fixed L1‑norm.
   One LP solve (subject to a short time limit) is enough for the final
   improvement.

The best sequence found during the allotted time is returned.
"""

import time
import numpy as np
from scipy import optimize, sparse

# ----------------------------------------------------------------------
# Helper: convolution using FFT (exact up to floating‑point noise)
def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of x and y using real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    fz = fx * fy
    conv = np.fft.irfft(fz, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Initialise a power‑law sequence a_i = (i+offset)^(-p) and normalise.
def _initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    # prevent division by zero
    idx = np.arange(n, dtype=float) + offset + 1.0
    a = idx ** (-p)
    a /= a.sum()                # normalise sum to 1 (scale invariant for the score)
    return a


# ----------------------------------------------------------------------
# One gradient‑descent step that tries a few step sizes and keeps the best.
def _gradient_step(a: np.ndarray, step_candidates) -> np.ndarray:
    n = len(a)
    # self‑convolution and its mean
    conv = _convolve_fft(a, a)                 # length 2n‑1
    avg = conv.mean()
    residual = conv - avg                       # zero‑mean

    # gradient of J = ∑ (c‑avg)²  w.r.t. a  (ignoring the tiny diagonal correction)
    #   ∂J/∂a_i = 2 * Σ_k residual_k * a_{k‑i}
    # This is a cross‑correlation; we compute it via convolution with a reversed.
    grad_full = _convolve_fft(residual, a[::-1])   # length 3n‑2
    grad = grad_full[n - 1 : n - 1 + n]            # slice of length n

    best_a = a
    best_val = float('inf')
    for eta in step_candidates:
        cand = a - eta * grad
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:            # avoid division by zero
            continue
        cand /= s                # renormalise to sum = 1 (scale does not matter for the score)
        # cheap estimate using the same J; we still call evaluate_sequence later
        val_est = ((_convolve_fft(cand, cand) - _.mean()) ** 2).sum()
        if val_est < best_val:
            best_val = val_est
            best_a = cand
    return best_a


# ----------------------------------------------------------------------
# Solve   minimise   M
#   s.t.   (f * g)_k ≤ M   for all k,
#          g_i ≥ 0,
#          Σ_i g_i = S   (S is the current L1‑norm, here taken = 1)
def _lp_minimise_max(f: np.ndarray, sum_target: float = 1.0,
                    time_limit: float = 15.0):
    n = len(f)
    rows = 2 * n - 1

    # Build the (rows) × (n+1) constraint matrix A_ub * [g ; M] ≤ 0
    # Each row k contains coefficients f[i] for variable g[j] where i + j = k.
    data = []
    row_ind = []
    col_ind = []

    for i in range(n):
        fi = f[i]
        # positions k = i + j  with j = 0 … n‑1
        k_start = i
        k_end = i + n - 1
        # the corresponding column for g_j is j = k - i
        for k in range(k_start, k_end + 1):
            j = k - i
            data.append(fi)
            row_ind.append(k)
            col_ind.append(j)

    # column for M (last column) has coefficient -1 in each row
    data.extend([-1.0] * rows)
    row_ind.extend(list(range(rows)))
    col_ind.extend([n] * rows)          # index of M is n

    A_ub = sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n + 1))

    # Equality Σ g_i = sum_target
    A_eq = sparse.csr_matrix(([[1.0] * n + [0.0]]))   # 1×(n+1)
    b_eq = np.array([sum_target])

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]   # M ≥ 0 as well

    res = optimize.linprog(c,
                           A_ub=A_ub,
                           b_ub=np.zeros(rows),
                           A_eq=A_eq,
                           b_eq=b_eq,
                           bounds=bounds,
                           method='highs',
                           options={'time_limit': time_limit, 'presolve': True})
    if res.success:
        g = res.x[:n]
        return g
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises the evaluation function.

    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found.  The returned list contains non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0    # keep a few seconds for a clean exit
    print("Search started – time budget:", budget_s, "seconds")

    # ------------------------------------------------------------------
    # 1. Choose a moderate length.
    n = 2500                       # large enough for a good approximation,
                                   # small enough for the LP to stay tractable
    print("Using length n =", n)

    # ------------------------------------------------------------------
    # 2. Power‑law start (p = 0.5 gives the classic 1/√i shape)
    a = _initial_sequence(n, p=0.5, offset=0.0)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print("Initial score (1/√i shape):", best_score)

    # ------------------------------------------------------------------
    # 3. Simple grid search over a small offset to see if a shifted power law helps
    offsets = np.linspace(0, 100, 21)   # 0,5,10,…,100
    for off in offsets:
        cand = _initial_sequence(n, p=0.5, offset=off)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"Improved by offset={off:.1f} → score {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Gradient‑descent refinement (run while time permits)
    step_sizes = [0.001, 0.003, 0.006, 0.01, 0.02]   # tried in order
    while time.time() < deadline:
        a = best_seq.copy()          # start from the current best
        a = _gradient_step(a, step_sizes)

        # evaluate only occasionally (every ~0.2 s) to keep the loop cheap
        if time.time() > deadline:
            break
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print("Gradient descent improvement → score", sc)
        else:
            # small random jitter to try to escape a flat region
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + np.random.randn() * 0.005)
            a /= a.sum()
            # keep the jittered version as the next start point
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 5. One LP refinement – minimise the maximal convolution for the current sum
    #    (the LP is relatively expensive, so we run it at most once)
    if time.time() < deadline:
        print("Running final LP refinement (may take a few seconds)…")
        lp_seq = _lp_minimise_max(best_seq, sum_target=1.0,
                                  time_limit=max(5.0, deadline - time.time()))
        if lp_seq is not None:
            lp_seq = np.maximum(lp_seq, 0.0)
            lp_seq /= lp_seq.sum()
            lp_score = evaluate_sequence(lp_seq.tolist())
            print("LP refinement score:", lp_score)
            if lp_score < best_score:
                best_score = lp_score
                best_seq = lp_seq.copy()

    # ------------------------------------------------------------------
    print("Best score found:", best_score)
    return best_seq.tolist()
