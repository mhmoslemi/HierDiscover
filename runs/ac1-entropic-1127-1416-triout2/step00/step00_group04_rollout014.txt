def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, math, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    
    # Parameter choices
    n = 500  # length, we try moderate
    # Optional schedule: start with uniform then random perturbations
    # Number of restarts
    restart_interval = 10.0  # after this many seconds, reinitialize random

    init_time = time.time()
    while time.time() < deadline:
        # If we have no current sequence, initialize
        if best_seq is None or time.time() - init_time > restart_interval:
            # random Dirichlet
            a = np.random.rand(n)  # uniform random
            a = a / a.sum()
            init_time = time.time()
        # run a few optimization steps
        # Use Adam parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        # number of inner steps limited by remaining time
        steps = 0
        while time.time() < deadline and steps < 2000:
            steps += 1
            # compute convolution via FFT
            size = 1 << ((2*n - 1).bit_length())
            A = np.fft.rfft(a, n=size)
            conv = np.fft.irfft(A*A, n=size)[:2*n-1]
            max_idx = int(np.argmax(conv))
            max_val = conv[max_idx]
            # compute gradient
            grad = np.zeros_like(a)
            # valid i range where j = max_idx - i within [0,n)
            i_min = max(0, max_idx - (n-1))
            i_max = min(n-1, max_idx)
            # vectorized:
            i_vals = np.arange(i_min, i_max+1)
            j_vals = max_idx - i_vals
            grad[i_vals] = 2.0 * a[j_vals]
            # Adam update
            t = steps
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.05 / (np.sqrt(t) + 1)  # decreasing step size
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # enforce non-negativity and renorm
            a = np.maximum(a, 0)
            sum_a = a.sum()
            if sum_a < 1e-12:
                a = np.random.rand(n)
                a = a / a.sum()
                continue
            a = a / sum_a

            # evaluate
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best score {best_score:.6f} at time {time.time():.2f}")

                if best_score < 1.503:
                    return best_seq.tolist()

        # continue outer loop
    if best_seq is None:
        # fallback: return uniform distribution
        return [1.0] * n
    return best_seq.tolist()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Return a step function sequence (list of non‑negative heights) that tries
    to minimize the evaluation function. The algorithm starts from a random
    Dirichlet‑distributed vector and performs a projected Adam optimisation
    that reduces the maximum entry of the self‑convolution, which directly
    lowers the evaluation score.
    """
    import numpy as np, time, math, random

    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5.0   # leave a little margin

    # Choose problem size; 300 gives a good trade‑off between speed and
    # theoretical lower bound (≈1.01).
    n = 300

    # Helper to compute evaluation quickly (calls the provided evaluate_sequence)
    def eval_seq(arr):
        return evaluate_sequence(arr.tolist())

    # Initialise with a random distribution (Dirichlet with concentration 0.5 for sparsity)
    a = np.random.rand(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = eval_seq(best_a)
    print(f"Initial score {best_score}")

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    step = 0
    # Pre‑compute FFT length (next power of two for speed)
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()

    while time.time() < deadline:
        step += 1
        # Compute linear convolution via FFT (real‑valued)
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]

        # Find the k indices with highest convolution values (top‑k)
        top_k = 5
        if conv_len <= top_k:
            idxs = np.arange(conv_len)
        else:
            # argpartition gives unsorted top‑k; we sort them for determinism
            idxs = np.argpartition(-conv, top_k-1)[:top_k]
            idxs = idxs[np.argsort(-conv[idxs])]

        # Build gradient that tries to reduce all large peaks simultaneously
        grad = np.zeros_like(a)
        for k in idxs:
            # i runs over all positions where j = k - i lies in [0, n)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_vals = np.arange(i_min, i_max + 1, dtype=int)
            j_vals = k - i_vals
            # derivative of sum_i a_i a_{k-i} w.r.t a_i is 2 a_{k-i}
            grad[i_vals] += 2.0 * a[j_vals]

        # Adam update (step‑size scheduled)
        lr = 0.05 / (math.sqrt(step) + 1.0)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project back to the feasible set: non‑negative and sum = 1
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total < 1e-12:
            # In the unlikely event that everything vanished, re‑initialise.
            a = np.random.rand(n)
            a = a / a.sum()
            m[:] = 0.0
            v[:] = 0.0
            continue
        a = a / total

        # Occasionally (every 500 steps) evaluate the current sequence.
        if step % 200 == 0:
            cur_score = eval_seq(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Step {step}: new best {best_score:.6f}")

            # Simple random perturbation to escape flat regions
            if step % 2000 == 0:
                noise = 0.02 * np.random.randn(n) * a
                a = np.clip(a + noise, 0.0, None)
                a = a / a.sum()

        # Early exit if we already beat the target
        if best_score < 1.503:
            break

    print(f"Finished optimisation – best score {best_score:.6f}")
    return best_a.tolist()

"""
Propose a step‑function (list of non‑negative heights) that tries to minimise
the evaluation function defined in the statement.

The algorithm is a simple projected Adam optimiser that directly reduces the
largest entry of the self‑convolution `a*a`.  Because the evaluation score is

    score = 2 * n * max(conv) / (sum(a)^2)

minimising `max(conv)` while keeping the sum fixed also minimises the score.
At each iteration we

1. compute the convolution via an FFT,
2. pick the few largest convolution entries,
3. build a sub‑gradient of the (smoothed) maximum,
4. take an Adam step, clip to non‑negative values and renormalise the vector.

Occasionally a small random perturbation is added to avoid getting stuck.
The procedure stops after the allotted time budget (default 1000 s) and
returns the best sequence found so far.
"""

import time
import math
import random
import numpy as np

# Helper: compute the evaluation score (wrapper around the provided function)
def _score(seq):
    """Return the evaluation score for a numpy array `seq`."""
    # evaluate_sequence expects a plain Python list of floats
    return evaluate_sequence(seq.tolist())

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a candidate sequence of step heights.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds.  The function stops a few seconds before this
        limit to avoid being killed.
    **kwargs

    Returns

        The best sequence found within the time budget.
    """
    np.random.seed(seed)

        A = np.fft.rfft(a, n=fft_len)
