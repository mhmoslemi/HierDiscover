def project_to_simplex(v):
    # returns v_proj onto simplex {x>=0, sum x=1}
    # algorithm:
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    """Projects a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if len(rho) == 0:
        # all non-positive => return uniform? Actually we'll set all zero.
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‐function (non‑negative sequence) that minimizes the
evaluation function:

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the score is scale‑invariant we can restrict ourselves to probability
vectors (sum = 1).  The problem then becomes

    minimise  max( a * a ) over the simplex a_i ≥ 0 , Σ a_i = 1.

The algorithm is a projected (sub)gradient descent on a smooth approximation
of the L∞‑norm.  For a given a we compute the convolution c = a*a and a
soft‑max weighting w_k ∝ exp(β·c_k); the gradient of the smooth max is

    g_i = 2 * Σ_k w_k a_{k‑i}       (cross‑correlation).

We take a step opposite to g, project back onto the probability simplex and
repeat.  The soft‑max temperature β is kept moderate (β≈10) so that many
convolution entries influence the descent direction.  After the main phase a
short hill‑climbing pass with random local moves is performed to polish the
solution.

All convolutions are carried out by FFT (real‑valued) which makes each
iteration O(n log n) and allows thousands of iterations within the 1000 s
budget.  The routine keeps the best sequence seen and finally returns it
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero (should not happen); return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using FFT (fast for lengths up to ~10⁴)."""
    n = len(a) + len(b) - 1
    # next power of two for efficient FFT
    nfft = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    fc = fa * fb
    conv = np.fft.irfft(fc, n=nfft)[:n]
    # Numerical noise may produce tiny imaginary parts; keep real part only
    return np.real(conv)


def smooth_max_score(p: np.ndarray, beta: float = 10.0) -> (float, np.ndarray, np.ndarray):
    """
    Return (score, convolution, soft‑max weights) for a probability vector p.
    score = 2 * n * max(conv(p,p))    (the exact objective)
    The weights are w_k ∝ exp(beta * (c_k - max_c)) and sum to 1.
    """
    n = p.shape[0]
    c = convolve_fft(p, p)                     # length 2n‑1
    max_c = np.max(c)
    # soft‑max that is numerically stable
    exp_vals = np.exp(beta * (c - max_c))
    w = exp_vals / np.sum(exp_vals)
    score = 2.0 * n * max_c
    return score, c, w


def gradient_softmax(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max (with weights w) w.r.t. p.
    g_i = 2 * Σ_k w_k * p_{k-i}          (cross‑correlation)
    """
    n = p.shape[0]
    # cross‑correlation can be obtained via convolution with the reversed weight vector
    w_rev = w[::-1]
    conv = convolve_fft(p, w_rev)                # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)            # pick the required slice
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    The function respects the given time budget (budget_s seconds) and
    returns the best sequence found as a plain list of Python floats.
    """
    np.random.seed(seed)
    # reserve a few seconds for finalisation/logging
    deadline = time.time() + budget_s - 5.0

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Parameters of the optimisation
    # ------------------------------------------------------------------
    # choose a problem size – larger n gives a lower theoretical bound.
    # 5000 works well in practice (FFT‑based convolutions are cheap).
    n = 5000

    # initialise a random probability vector (break symmetry from the start)
    p = np.random.rand(n)
    p = project_to_simplex(p)

    # hyper‑parameters (they adapt during the run)
    beta = 10.0                       # temperature for the soft‑max
    it = 0

    # ------------------------------------------------------------------
    # Main projected (sub)gradient descent loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # exact score and soft‑max weights
        score, conv, w = smooth_max_score(p, beta=beta)

        # keep the best solution seen so far
        if score < best_score:
            best_score = score
            best_seq = p.copy()
            print(f"[{it:6d}] new best score = {best_score:.6f}")

        # compute gradient of the smooth max
        g = gradient_softmax(p, w)

        # simple step‑size schedule (decreases like 1/√it)
        step = 0.1 / np.sqrt(it)

        # gradient step and projection onto the simplex
        p_new = p - step * g
        p = project_to_simplex(p_new)

        # occasional tiny random jitter to avoid stagnation
        if it % 200 == 0:
            jitter = np.random.randn(n) * 1e-5
            p = project_to_simplex(p + jitter)

    # ------------------------------------------------------------------
    # Post‑processing: short stochastic hill‑climbing around the best point
    # ------------------------------------------------------------------
    if best_seq is None:          # should never happen, but be defensive
        best_seq = np.full(n, 1.0 / n)

    p = best_seq.copy()
    it_hc = 0
    while time.time() < deadline:
        it_hc += 1
        # pick a random index and propose a tiny move
        idx = np.random.randint(n)
        delta = np.random.randn() * 5e-4      # small perturbation
        q = p.copy()
        q[idx] = max(q[idx] + delta, 0.0)
        # renormalise to keep sum = 1 (simple projection)
        q = project_to_simplex(q)

        q_score, _, _ = smooth_max_score(q, beta=beta)
        if q_score < best_score:
            best_score = q_score
            best_seq = q.copy()
            p = q
            print(f"[HC {it_hc:6d}] improved score = {best_score:.6f}")
        else:
            # keep the current point for the next trial
            p = q

    # Return the best sequence as a plain Python list
    return best_seq.tolist()
