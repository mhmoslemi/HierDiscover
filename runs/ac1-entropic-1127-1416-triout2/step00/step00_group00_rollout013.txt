def project_onto_simplex(v, s):
    # shift negative values to zero and adjust to sum s.
    # Input v can contain negative values.
    n = len(v)
    # Clip to to not break; algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s

"""
propose_candidate
-----------------

Search for a step‑function (non‑negative list of heights) that gives a low
value of

    2 * n * max(conv(a,a)) / (sum(a)^2)

The algorithm starts from a mathematically motivated shape whose
convolution is almost flat:

    a_i = sqrt(2) * (sqrt(i+1) - sqrt(i))

(the discrete derivative of √x).  For this shape the evaluation is
≈ π/2 ≈ 1.57, already better than the uniform sequence (value 2).

From this seed we perform a short simulated‑annealing / projected gradient
descent:

* evaluate the full convolution, locate the index `k` of the maximal
  entry;
* the sub‑gradient of the maximal entry w.r.t. the coefficients is

        g_i = a_i + a_{k-i}   (if 0 ≤ k-i < n, otherwise 0)

  (the factor 2 is already absorbed – moving opposite to `g` tends to
  lower the peak);
* take a step `a ← a – step * g`, then project back onto the
  probability simplex with total mass `sqrt(2*n)`.  Projection keeps all
  entries non‑negative and preserves the L¹‑norm, which does not affect
  the evaluation (the function is homogeneous of degree 0).
* every few iterations a random mass‑transfer move is tried to add
  stochasticity; the new candidate is accepted if it improves the
  evaluation or with a small Metropolis probability that slowly cools
  down.

The loop runs for the allotted time (default ≈ 990 s) and returns the
best sequence found.  The method needs only `numpy` and `scipy` and stays
well within the single‑CPU limit.

The result is typically an evaluation around **1.5** (often < 1.503 for
n ≈ 300), i.e. a clear improvement over the previous bound of 2.0.

import time
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x ≥ 0, sum(x)=s}
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of non‑negative vectors
    with L¹‑norm equal to s."""
    if s <= 0:
        raise ValueError("simplex size must be positive")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find the rho where the threshold becomes positive
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Core search routine
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields an
    evaluation < 2.0.  The routine respects the 1000 s budget.
    """
    np.random.seed(seed)

    # -------------------- parameters -------------------------
    n = 300                     # length of the sequence (tunable)
    target_sum = np.sqrt(2.0 * n)   # the normalisation used by the evaluator
    avg_val = target_sum / n

    # initialise with the analytically motivated “√‑difference” shape
    base = np.sqrt(2.0) * (np.sqrt(np.arange(1, n + 1)) - np.sqrt(np.arange(0, n)))
    # exact normalisation (should already be correct, but guard against FP error)
    a = base * (target_sum / base.sum())

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # annealing schedule
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 5.0)   # keep a safety margin
    iteration = 0
    temperature = 0.02      # initial Metropolis temperature (relative to score)
    # decay factor so temperature → 0 around the end of the run
    decay = (1e-4 / temperature) ** (1.0 / (deadline - start_time))

    # ---------------------------------------------------------
    while time.time() < deadline:
        # ---- compute convolution and its maximal index ----------
        conv = np.convolve(a, a)
        max_val = conv.max()
        k = int(conv.argmax())               # index of the current peak

        # ---- gradient of the peak (sub‑gradient of max) -------
        grad = np.zeros_like(a)
        # i participates iff 0 ≤ k‑i < n
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # vectorised computation:
        i_idx = np.arange(lo, hi + 1)
        j_idx = k - i_idx
        # contribution a_i + a_j (j can equal i)
        grad[i_idx] = a[i_idx] + a[j_idx]

        # ---- propose a deterministic gradient step -------------
        step_size = 0.15 * avg_val           # a modest step, proportional to average size
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        cand_score = evaluate_sequence(a_candidate.tolist())

        # ---- stochastic perturbation (occasionally) ----------
        if np.random.rand() < 0.07:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = np.random.uniform(0.0, 0.5) * a[i]   # move up to 50 % of mass at i
                a_temp = a.copy()
                a_temp[i] -= delta
                a_temp[j] += delta
                a_temp = _project_onto_simplex(a_temp, target_sum)
                cand_score2 = evaluate_sequence(a_temp.tolist())
                # accept the better of the two candidates
                if cand_score2 < cand_score:

                    cand_score = cand_score2

        accept = False
