"""
Propose a candidate step‑function heights sequence.

The algorithm starts from a (long) uniform sequence (or from the
previously discovered `height_sequence_1`).  It then performs a simple
gradient‑descent / simulated‑annealing loop that tries to flatten the
self‑convolution `a * a`.  In each iteration we:

1. Compute the full convolution and locate the index `k_max` attaining
   the maximal value.
2. The contribution of entry `i` to this maximal entry is
   `a[i] * a[k_max - i]`.  The entry with the largest contribution is
   taken as a “source” of mass.
3. Two random “target’’ indices that do **not** contribute to `k_max`
   are chosen, and a small amount of mass `eps` is moved from the source
   (and its counterpart) to the targets, keeping the total sum constant.
4. The new sequence is evaluated with `evaluate_sequence`.  Better
   sequences are always kept; worse ones are occasionally accepted
   according to a simple temperature schedule (simulated annealing).

The loop runs until the allocated time budget (≈ 1000 s) is exhausted,
and the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _initial_sequence(seed: int, length: int) -> np.ndarray:
    """
    Return an initial non‑negative sequence.
    If a global `height_sequence_1` exists we use it (converted to float);
    otherwise we start from a uniform vector of the given length.
    """
    np.random.seed(seed)
    if "height_sequence_1" in globals():
        seq = np.array([float(x) for x in globals()["height_sequence_1"]], dtype=float)
        # if the provided sequence is shorter than the desired length,
        # pad with small random values; if longer, truncate.
        if len(seq) < length:
            extra = np.random.rand(length - len(seq)) * 0.01
            seq = np.concatenate([seq, extra])
        elif len(seq) > length:
            seq = seq[:length]
        return seq
    # pure uniform start – all entries identical
    return np.ones(length, dtype=float)


def _select_source_and_counterpart(a: np.ndarray, k_max: int) -> (int, int):
    """
    Choose the index `i` that contributes most to the convolution entry
    `k_max`.  Its counterpart is `j = k_max - i`.
    Returns (i, j).
    """
    n = a.shape[0]
    low = max(0, k_max - (n - 1))
    high = min(n - 1, k_max)
    idxs = np.arange(low, high + 1)
    counter = k_max - idxs
    # contribution = a[i] * a[j] (ordered pairs count both directions,
    # but for the purpose of picking a source the product suffices)
    contrib = a[idxs] * a[counter]
    # pick the largest contribution; if all zero pick a random valid index
    if contrib.max() == 0:
        i = np.random.randint(low, high + 1)
    else:
        probs = contrib / contrib.sum()
        i = np.random.choice(idxs, p=probs)
    j = k_max - i
    return int(i), int(j)


def _pick_two_targets(a_len: int, forbid_range: np.ndarray) -> (int, int):
    """
    Choose two distinct indices that are *not* inside `forbid_range`.
    If the complement set is too small we fall back to any two distinct
    indices.
    """
    all_idx = np.arange(a_len)
    complement = np.setdiff1d(all_idx, forbid_range, assume_unique=True)
    if complement.size >= 2:
        t = np.random.choice(complement, size=2, replace=False)
        return int(t[0]), int(t[1])
    # fallback – just pick any two distinct indices
    t = np.random.choice(all_idx, size=2, replace=False)
    return int(t[0]), int(t[1])


def _make_candidate(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Build a new candidate sequence by moving a small amount of mass from
    the dominant contributing pair (i, j) for `k_max` to two target
    positions that do not affect that maximal entry.
    The total sum of the vector stays unchanged.
    """
    n = a.shape[0]

    # source pair
    i, j = _select_source_and_counterpart(a, k_max)

    # indices that *do* contribute to k_max
    low = max(0, k_max - (n - 1))
    high = min(n - 1, k_max)
    forbid = np.arange(low, high + 1)

    # two target indices that are outside the contributing range
    p, q = _pick_two_targets(n, forbid)

    # amount to move – a modest fraction of the smallest entry among the source pair
    if i == j:
        eps = a[i] * 0.05  # 5 % of the value (i==j means a single entry produces the max)
        eps = min(eps, a[i] * 0.5)  # never remove more than half
        if eps <= 0:
            return a.copy()
        cand = a.copy()
        # subtract 2·eps from the single source (to keep the sum constant after adding eps twice)
        cand[i] -= 2.0 * eps
        cand[p] += eps
        cand[q] += eps
    else:
        # remove eps from each side of the pair, add eps to each target
        eps = min(a[i], a[j]) * 0.05  # 5 % of the smaller partner
        eps = min(eps, a[i] * 0.5, a[j] * 0.5)
        if eps <= 0:
            return a.copy()
        cand = a.copy()
        cand[i] -= eps
        cand[j] -= eps
        cand[p] += eps
        cand[q] += eps

    # safety – enforce non‑negativity (the security checks in evaluate_sequence
    # also do this, but keeping numbers non‑negative helps the optimisation)
    cand[cand < 0] = 0.0
    return cand


def _temperature(remaining: float, total: float) -> float:
    """
    Simple linear cooling schedule used for occasional uphill moves.
    Returns a positive temperature; tiny values still allow a minute
    probability of acceptance.
    """
    # end temperature ≈ 1e‑4 to keep a non‑zero chance at the very end
    return max(1e-4, remaining / total)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a step‑function (non‑negative height sequence) that minimises
    ``evaluate_sequence``.  The routine runs for roughly ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # Choose a starting length – longer is theoretically better.
    # The problem statement mentions a construction length of 5146,
    # we keep the same order of magnitude.
    # ------------------------------------------------------------------
    n = 5200                     # a little larger than the known 5146
    a = _initial_sequence(seed, n)

    # Ensure we are not starting with an almost‑zero vector (the security
    # checks reject sums < 0.01).  Scale to a reasonable total mass.
    total = a.sum()
    if total < 0.1:
        a *= (0.1 / total)

    # Evaluate the initial candidate
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Main optimisation loop -------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # 1. compute current convolution and locate the maximal entry
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # 2. propose a new sequence by moving a small chunk of mass
        candidate = _make_candidate(a, k_max)

        # 3. Occasionally do a pure random perturbation to escape plateaus
        if np.random.rand() < 0.04:   # 4 % random step
            i = np.random.randint(n)
            delta = np.random.randn() * 0.02 * max(1.0, a[i])
            # keep the sum constant by moving the opposite amount from a random other index
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)
            delta = np.clip(delta, -a[i] + 1e-12, a[j] - 1e-12)  # keep non‑negative
            candidate[i] += delta
            candidate[j] -= delta

        # 4. Evaluate the candidate
        cand_score = evaluate_sequence(candidate.tolist())

        # 5. Acceptance rule (always keep improvements; sometimes accept worsening)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            a = candidate
            print(f"[{time.time():.2f}] Iter {iteration:6d} – NEW BEST = {best_score:.6f}")
        else:
            # simulated‑annealing style acceptance
            remaining = deadline - time.time()
            temp = _temperature(remaining, budget_s)
            # Metropolis criterion
            prob = np.exp(-(cand_score - best_score) / temp)
            if np.random.rand() < prob:
                a = candidate

        # occasional full restart if we have stagnated too long
        if iteration % 2500 == 0:
            # keep the best found so far but re‑initialise the working vector
            a = best_seq.copy()
            # add a tiny random jitter to all entries to explore a new neighbourhood
            jitter = np.random.randn(n) * 0.005 * np.mean(best_seq)
            a = np.maximum(0.0, a + jitter)
            # re‑normalise to keep the total mass roughly unchanged
            a_sum = a.sum()
            if a_sum > 0:
                a *= best_seq.sum() / a_sum

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    return best_seq.tolist()
