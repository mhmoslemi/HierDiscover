def refine_with_adam(a_start, betas, iters_per_beta, deadline):
    z = np.log(a_start + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_a = a_start.copy()
    best_score = compute_score(best_a)
    for beta in betas:
        for i in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1
            f, grad_z = smooth_max_and_grad(z, beta, n)
            # Adam update
            m = 0.9 * m + 0.1 * grad_z
            v = 0.999 * v + 0.001 * (grad_z ** 2)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            lr = 0.02 / (np.sqrt(v_hat) + 1e-8)
            z = z - lr * m_hat
            # compute a and evaluate
            a = softmax(z)
            # optional symmetrize?
            # Actually enforce symmetry: a = (a+a[::-1])/2
            a = (a + a[::-1]) * 0.5
            # project to simplex (if needed), but softmax already sum to 1.
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

def adam_optimize(a_start, betas, iters_per_beta, deadline):
    n = a_start.shape[0]
    # convert to log domain
    z = np.log(a_start + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t_iter = 0
    best_a = a_start.copy()
    best_score = compute_score(best_a)
    for beta in betas:
        for i in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t_iter += 1
            f, grad = smooth_max_and_grad(z, beta, n)  # includes projection constraints automatically
            # adam update
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1 - 0.9**t_iter)
            v_hat = v / (1 - 0.999**t_iter)
            lr = 0.05 / (np.sqrt(v_hat) + 1e-8)   # adapt step
            z = z - lr * m_hat
            # compute a
            a = softmax(z)
            # symmetrize
            a = (a + a[::-1]) * 0.5
            # maybe project to simplex just in case (softmax ensures sum=1 and nonneg)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                # optionally tighten betas or reset m,v?
    return best_a, best_score
```

def shifted_product_family(n, alpha, delta):
    i = np.arange(n, dtype=np.float64)
    # to avoid zero at ends: i+delta
    w = np.exp(-alpha * np.log(i + delta) - alpha * np.log(n - delta - i))
    w = np.where(np.isfinite(w), w, 0)
    return w / w.sum()

"""
Hybrid search for the step‑function that minimises

        2·n·max( a * a ) / (∑a)²

The algorithm proceeds in three phases:

1. **Very fast analytic seeding** – a dense scan of a few
   low‑dimensional families (product, beta‑power, exponential‑U,
   symmetric‑Beta) on a relatively large length (2048‑8192).  The best
   candidate so far is kept.

2. **Adam‑based smooth‑max optimisation** – the objective is
   approximated by a log‑sum‑exp “soft‑max” (parameter β) and the
   gradient w.r.t. the soft‑max parameters  `z` (a = softmax(z)) is
   computed analytically.  A schedule of increasing β drives the
   solution towards the true max while an Adam optimiser performs many
   cheap iterations (FFT‑based convolutions).

3. **Short simulated‑annealing hill‑climb** – a final stochastic walk
   on the simplex refines any remaining asymmetry.

All phases respect the global wall‑clock budget (≈1000 s) and the best
sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize  # only needed for the fallback LP idea

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for cheap seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} (the classic arcsine‑type family)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def shifted_product_family(n: int, alpha: float, delta: float) -> np.ndarray:
    """Generalised product family with a shift δ∈(0,1):
       a_i ∝ [(i+δ)(n‑δ‑i)]^{‑α}.
    """
    i = np.arange(n, dtype=np.float64)
    # avoid hitting zero at the ends: δ>0 and δ<1
    term1 = i + delta
    term2 = n - delta - i
    # guard against non‑positive entries (should not happen for 0<δ<1)
    ok = (term1 > 0) & (term2 > 0)
    logprod = np.empty_like(i)
    logprod[ok] = np.log(term1[ok]) + np.log(term2[ok])
    logprod[~ok] = -np.inf          # will be turned into zero weight
    w = np.exp(-alpha * logprod)
    w[~ok] = 0.0
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def symmetric_beta_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{p‑1}·(n‑i)^{p‑1}  (Beta(p,p) on a discrete grid)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** (p - 1) * (n - i) ** (p - 1)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its exact gradient
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z   (gradient w.r.t. the soft‑max parameters)

    The gradient is derived analytically (no finite differences).
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights over the convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta                 # smooth approximation of max
    w = expw / sumexp                                # normalised soft‑max over positions

    # gradient of max(conv) w.r.t. a:
    #   ∂ conv_k / ∂ a_j = a_{k‑j} + a_{k‑j} (the two symmetric terms),
    #   but we can obtain the weighted sum via a convolution with w.
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]        # central slice, shape (n,)

    # back‑propagation through the soft‑max (see “softmax‑grad” identity)
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  betas,
                  iters_per_beta,
                  deadline: float):
    """
    Adam optimiser for the smooth‑max objective.

    Parameters
    ----------
    a_start : np.ndarray
        Starting probability vector (already normalised, sum=1).
    betas : list[float]
        Sequence of smoothing parameters (increasing).
    iters_per_beta : list[int]
        Number of Adam steps for each β.
    deadline : float
        Absolute time after which the optimiser must stop.

    Returns
    -------
    best_a : np.ndarray
        The best probability vector found.
    best_score : float
        Corresponding exact score (2·n·max(conv)).
    """
    n = a_start.shape[0]
    # initialise Adam state in the log‑softmax space
    z = np.log(a_start + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_a = a_start.copy()
    best_score = compute_score(best_a)

    for beta, nit in zip(betas, iters_per_beta):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Adam stage β={beta}, steps={nit}")
        for _ in range(nit):
            if time.time() >= deadline:
                break
            t += 1
            f, grad_z = smooth_max_and_grad(z, beta, n)

            # Adam update with classic coefficients
            m = 0.9 * m + 0.1 * grad_z
            v = 0.999 * v + 0.001 * (grad_z ** 2)

            m_hat = m / (1.0 - 0.9 ** t)
            v_hat = v / (1.0 - 0.999 ** t)

            # step size – a modest global scale (tuned experimentally)
            lr = 0.02 / (np.sqrt(v_hat) + 1e-8)

            z = z - lr * m_hat

            # recover the probability vector; enforce symmetry explicitly
            a = softmax(z)
            a = (a + a[::-1]) * 0.5    # keep the vector perfectly symmetric
            # (softmax already guarantees sum=1)

            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                # shrink the learning‑rate a little when we improve
                # (helps to settle near a plateau)
                # (not mandatory, merely a heuristic)
    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated‑annealing hill‑climber (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a linear temperature schedule.
    50 % swaps, 50 % mass‑transfer moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature chosen as a small fraction of the initial score
    T0 = best_score * 5e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # pure swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta_score = cand_score - cur_score
        temperature = T0 * (1 - step / max_steps)

        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (temperature + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score
