"""
Non‑negative step‑function optimiser (Adam on a smoothed max‑convolution).

Main ideas
----------
1. Build many analytical seeds (uniform, central‑binomial, power‑law,
   geometric, Beta‑shaped) and keep the best.

2. Refine the best seed with a *continuation* Adam optimiser:
   minimise 2·n·max(conv(a,a)) on the probability simplex.
   The nonsmooth max is replaced by a smooth log‑sum‑exp
   ϕ_β(a)=2·n·β⁻¹·log Σ_k exp(β·conv_k).
   β is increased geometrically (5 → 160 → …) so the optimiser first
   sees a smooth landscape then gradually focuses on the true peak.
   Gradients are computed efficiently via FFT (cross‑correlation).

3. After each Adam step the iterate is Euclidean‑projected onto the
   simplex and symmetrised (a←(a+a[::-1])/2); the optimum is known to be
   symmetric, this halves the effective search space.

4. A cheap stochastic hill‑climb (swap / fractional mass transfer) is
   performed at the end to polish any remaining tiny improvement.

The whole search respects the 1000 s budget (a 5 s safety margin) and
keeps the best sequence seen.  The final list of floats is returned
ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Symmetric and heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (continuation on β)
# ----------------------------------------------------------------------


def softmax_stable(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    shift = x.max()
    ex = np.exp(x - shift)
    return ex / ex.sum()


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    beta_schedule: list[float],
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective.
    Returns the best vector seen (on the simplex) and its exact score.
    """
    a = a0.astype(np.float64).copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = compute_score(a)

    # iterate over beta values (continuation)
    for stage, beta in enumerate(beta_schedule):
        print(f"Adam stage {stage+1}/{len(beta_schedule)}: beta = {beta:.3g}")
        # allocate a fraction of remaining time per stage (simple heuristic)
        stage_deadline = deadline - (len(beta_schedule) - stage - 1) * 2.0  # keep 2 s per later stage
        while time.time() < stage_deadline:
            # ---- gradient of smoothed max ----
            conv = convolution_fft(a, a)                     # length 2n‑1
            # smooth weights (softmax over β·conv)
            w = softmax_stable(beta * conv)                 # sum(w)=1
            # gradient = 2 * cross‑correlation(a, w)
            a_rev = a[::-1]
            corr = convolution_fft(a_rev, w)                # length 3n‑2
            grad = 2.0 * corr[n - 1:n - 1 + n]               # slice the central n entries

            # ---- Adam update ----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # ---- enforce simplex & symmetry ----
            a = (a + a[::-1]) * 0.5        # symmetrisation
            a = project_to_simplex(a)

            # ---- occasional exact evaluation ----
            if (t % 2000) == 0:
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"  New best score {best_score:.6f} at iter {t}")

            # safety break if too close to deadline
            if time.time() >= deadline:
                break

    # final check
    final_score = compute_score(best_a)
    if final_score < best_score:
        best_score, best_a = final_score, best_a.copy()
    return best_a, best_score

# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    deadline,
    max_steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb: swaps and fractional mass moves.
    Keeps the best sequence encountered.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07
    rng = np.random.default_rng()

    step = 0
    while step < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if rng.random() < 0.5:
            i, j = rng.integers(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = rng.integers(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5          # keep symmetry
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.96

        step += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Safety margin for the harness (a few seconds)
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        print(f"Using external height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        print(f"No external sequence – default length set to {n}.")

    # ------------------------------------------------------------------
    # 1. Build a collection of analytic starters
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # central binomial (classic)
    init_candidates.append(generate_central_binomial(n))

    # power‑law family
    for p in np.linspace(0.3, 1.0, 13):
        init_candidates.append(generate_powerlaw(n, float(p)))

    # geometric family (r close to 1)
    for r in [0.96, 0.97, 0.98, 0.985, 0.99, 0.992, 0.995, 0.998]:
        init_candidates.append(generate_geometric(n, float(r)))

    # Beta‑shaped family (α in (0.1,0.6])
    for alpha in np.linspace(0.10, 0.60, 11):
        init_candidates.append(generate_beta_shape(n, float(alpha)))

    # possibly include the previously found best construction
    try:
        init_candidates.append(np.array(height_sequence_1, dtype=np.float64))  # type: ignore[name-defined]
        print("Added provided height_sequence_1 as a starter.")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Choose the best starter by exact evaluation
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimiser with continuation on β
    # ------------------------------------------------------------------
    # allocate most of the remaining time to the continuous optimiser
    adam_deadline = deadline - 30.0          # keep ~30 s for polishing
    if time.time() < adam_deadline:
        # beta schedule – start smooth and finish very close to max
        beta_schedule = [5, 20, 80, 320, 1280]
        print(f"Starting Adam optimisation (deadline in {adam_deadline - time.time():.1f}s).")
        a_opt, opt_score = adam_optimize(
            best_seq,
            time_deadline=adam_deadline,
            beta_schedule=beta_schedule,
            lr=0.12,
        )
        if opt_score < best_val:
            best_val, best_seq = opt_score, a_opt.copy()
            print(f"Adam phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional short random‑restart bursts (cheap local Adam)
    # ------------------------------------------------------------------
    restart_deadline = deadline - 12.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(3):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.018 * np.random.randn(n)
            pert = project_to_simplex(pert)
            # a very short Adam run with a modest beta
            a_tmp, s_tmp = adam_optimize(
                pert,
                time_deadline=time.time() + 7.0,
                beta_schedule=[8, 32, 128],
                lr=0.10,
            )
            if s_tmp < best_val:
                best_val, best_seq = s_tmp, a_tmp.copy()
                print(f"Restart {r} yielded improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Final hill‑climb (up to {steps} steps).")
        polished_seq, hill_score = hill_climb(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val, best_seq = hill_score, polished_seq.copy()
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard – fall back to uniform if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 7. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq.tolist()
