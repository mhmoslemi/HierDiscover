"""
Step‑function optimizer (≈ 1000 s budget).

Overview
--------
1. Choose a moderate length (2048 or 4096 depending on time).
2. Produce a rich set of seeds:
   * deterministic sinusoidal and polynomial shapes,
   * Dirichlet draws of various concentration,
   * sparse spikes,
   * random‑phase Fourier‑inverse magnitudes (often give flat auto‑correlation).
3. Keep the best seed (exact max‑convolution score).
4. Four optimisation phases (all respect the probability simplex):
   – (A) smooth surrogate (soft‑max) gradient descent,
   – (B) variance‑flattening on the auto‑convolution,
   – (C) sub‑gradient descent on the true ‖·‖∞‑objective,
   – (D) simulated‑annealing / stochastic hill‑climb.
5. Final greedy polishing and a tiny smoothing pass.
6. Return the best sequence found (list of floats).

All heavy loops stop as soon as the global deadline is reached, guaranteeing
termination before the 1000 s timeout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    # if all entries are non‑positive, return the uniform distribution
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution density on the discrete set {0,…,n‑1}."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` distinct positions with random weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def random_phase_abs(n: int) -> np.ndarray:
    """
    Take the absolute value of an inverse‑FFT of a unit‑magnitude spectrum
    (random phases, Hermitian‑symmetric → real signal).  Normalised to sum 1.
    """
    # Hermitian symmetry for a real signal
    # frequencies 0 … n‑1
    phases = np.random.rand(n // 2 - 1) * 2.0 * np.pi
    # construct full spectrum
    spec = np.empty(n, dtype=np.complex128)
    spec[0] = 1.0  # DC component = 1
    spec[1:n // 2] = np.exp(1j * phases)
    spec[n // 2] = 1.0  # Nyquist (real) if n even
    spec[n // 2 + 1:] = np.conj(spec[1:n // 2][::-1])
    signal = np.fft.ifft(spec).real
    a = np.abs(signal)
    if a.sum() == 0:
        a = np.ones_like(a)
    a /= a.sum()
    return a


def random_phase_squared(n: int) -> np.ndarray:
    """
    Same as `random_phase_abs` but square the signal before taking absolute.
    This tends to give a more spread‑out mass.
    """
    phases = np.random.rand(n // 2 - 1) * 2.0 * np.pi
    spec = np.empty(n, dtype=np.complex128)
    spec[0] = 1.0
    spec[1:n // 2] = np.exp(1j * phases)
    spec[n // 2] = 1.0
    spec[n // 2 + 1:] = np.conj(spec[1:n // 2][::-1])
    signal = np.fft.ifft(spec).real
    a = np.abs(signal) ** 2
    if a.sum() == 0:
        a = np.ones_like(a)
    a /= a.sum()
    return a

# Helper for the variance‑flattening phase
# ----------------------------------------------------------------------


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Phase A – soft‑max surrogate gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2500.0,
    step_coef: float = 0.003,
) -> tuple[np.ndarray, float]:
    """Projected GD on Lβ(a)=log Σ exp(β·conv(a,a))."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    it = 0
    start = time.time()
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # anneal β
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights (shift for numerical stability)
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of the surrogate (see analysis)
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 4000 == 0:
            print(
                f"[soft‑max GD] it {it:6d} | time {elapsed:6.1f}s | "
                f"best {best_score:.6f} | β {beta:.1f}"
            )
    return best_a, best_score

def variance_gradient_descent(

    it = 0

        conv = convolution_fft(cur_a, cur_a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        e = conv - avg
        grad = 4.0 * correlation_valid(e, cur_a)

        it += 1
        step = step0 / np.sqrt(it)

        nxt = cur_a - step * grad
        cur_a = project_to_simplex(nxt)

        if it % 3000 == 0:
            print(
                f"[var‑GD] it {it:6d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase C – sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # small jitter to escape flat regions
        if t % 1200 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase D – simulated‑annealing / stochastic hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    start_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic moves accepted with a Metropolis criterion.
    The temperature decays linearly to near‑zero.
    Moves are of two types:
        – swap two entries,
        – transfer a small fraction from i → j.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule (linear)
    total = max(deadline - time.time(), 1e-6)
    T_start = max(0.1 * cur_score, 1e-4)
    T_end = 1e-7 * cur_score
    it = 0

    while time.time() < deadline:
        it += 1
        # linear schedule based on elapsed fraction
        elapsed = time.time() - (deadline - total)
        frac = elapsed / total
        T = T_start * (1.0 - frac) + T_end * frac

        # propose move
        cand = cur_a.copy()
        if np.random.rand() < 0.55:            # transfer move
            i, j = np.random.randint(n, size=2)
            if i == j or cand[i] == 0.0:
                continue
            # fraction of mass to move, shrinking with temperature
            max_frac = 0.1 * (T / T_start) + 0.001
            delta = max_frac * cand[i] * np.random.rand()
            cand[i] -= delta
            cand[j] += delta
        else:                                   # swap move
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # sum is preserved, only non‑negativity may be violated by rounding
        if np.any(cand < 0):
            continue

        new_score = compute_score(cand)
        diff = new_score - cur_score
        # Metropolis acceptance
        if diff < 0 or np.random.rand() < np.exp(-diff / max(T, 1e-12)):
            cur_a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

        # occasional logging
        if it % 50000 == 0:
            print(
                f"[SA] it {it:7d} | time {time.time():6.1f}s | "
                f"best {best_score:.6f} | cur {cur_score:.6f} | T {T:.2e}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase E – final greedy polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb that only accepts improvements."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.07

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:                     # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # directional transfer
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # renormalise numerically
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.09, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

        if steps % 300000 == 0:
            print(
                f"[hill‑climb] step {steps:7d} | time {time.time():6.1f}s | best {best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny smoothing pass (optional)
# ----------------------------------------------------------------------


def smoothing_pass(a: np.ndarray, passes: int = 2) -> np.ndarray:
    """Apply a 3‑point averaging filter a few times, keeping the simplex constraint."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)
    cur = a.copy()
    for _ in range(passes):
        conv = np.convolve(cur, kernel, mode="same")
        cur = project_to_simplex(conv)
    return cur


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.8
    total_time = deadline - start_time
    print(f"Total budget ≈ {total_time:.1f}s")

    # ------------------------------------------------------------------
    # Choose length (larger length gives a lower theoretical bound)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 4096
    print(f"Working with sequence length n = {n}")

    # ------------------------------------------------------------------
    # 0 – optional fallback (previous best)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        if len(fallback) == n:
            fb = evaluate_sequence(fallback)
            print(f"[fallback] score = {fb:.6f}")
            if fb < best_val:
                best_val = fb
                best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1 – generate a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # beta pdf families
    for alpha, beta in [
        (0.2, 0.2),
        (0.5, 0.5),
        (1.0, 1.0),
        (2.0, 2.0),
        (5.0, 5.0),
        (0.5, 2.0),
        (2.0, 0.5),
    ]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # random Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # sparse spikes (often good starting points)
    for k in (30, 50, 80):
        seeds.append(random_spike(n, k=k))

    # random‑phase based seeds
    seeds.append(random_phase_abs(n))
    seeds.append(random_phase_squared(n))

    # include fallback if it matches length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float))

    # ------------------------------------------------------------------
    # 2 – evaluate seeds and keep the best
    # ------------------------------------------------------------------
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] new best → score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – initialise optimisation from the best seed
    # ------------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)
    print(f"Starting point score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4 – allocate time to the successive phases
    # ------------------------------------------------------------------
    t0 = time.time()
    phase_a_dead = t0 + total_time * 0.30
    phase_b_dead = phase_a_dead + total_time * 0.20
    phase_c_dead = phase_b_dead + total_time * 0.25
    phase_d_dead = phase_c_dead + total_time * 0.20
    # remaining for final polishing
    phase_e_dead = deadline - 0.3

    # ------------------------------------------------------------------
    # Phase A – soft‑max GD
    # ------------------------------------------------------------------
    print("\n[Phase A] soft‑max gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a,
        cur_score,
        phase_a_dead,
        beta_start=5.0,
        beta_end=2500.0,
        step_coef=0.003,
    )
    if cur_score < best_val:
        best_val = cur_score
        print(f"[after A] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – variance‑flattening
    # ------------------------------------------------------------------
    if time.time() < phase_b_dead:
        print("\n[Phase B] variance‑flattening gradient descent")
        cur_a, cur_score = variance_gradient_descent(
            cur_a,
            cur_score,
            phase_b_dead,
            step0=0.07,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after B] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase_c_dead:
        print("\n[Phase C] sub‑gradient refinement")
        cur_a, cur_score = subgradient_refine(
            cur_a,
            cur_score,
            phase_c_dead,
            max_iter=800_000,
            step0=0.09,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after C] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase_d_dead:
        print("\n[Phase D] simulated‑annealing")
        cur_a, cur_score = simulated_annealing(
            cur_a,
            cur_score,
            phase_d_dead,
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after D] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase E – final greedy polishing + tiny smoothing
    # ------------------------------------------------------------------
    if time.time() < phase_e_dead:
        print("\n[Phase E] hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, phase_e_dead, max_steps=1_500_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"[after hill‑climb] best score = {best_val:.6f}")

        # optional smoothing pass
        smoothed = smoothing_pass(cur_a, passes=2)
        sm_score = compute_score(smoothed)
        if sm_score < best_val:
            best_val = sm_score
            cur_a = smoothed
            print(f"[after smoothing] best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion & official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
