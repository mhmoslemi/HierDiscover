def exponentiated_gradient_descent(a0, start_score, deadline, beta_start=5.0, beta_end=5000.0, step0=0.1):
    n = len(a0)
    a = a0.copy()
    u = np.log(np.maximum(a, 1e-12))
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t = 0
    best_a = a.copy()
    best_score = start_score
    while time.time() < deadline:
        # compute conv and present score for logging
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score; best_a = a.copy()
        # anneal beta
        frac = (time.time() - start) / (deadline - start)
        beta = beta_start + frac * (beta_end - beta_start)
        # compute w (softmax of conv)
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        # gradient wrt a
        grad_a = 2.0 * beta * np.correlate(w, a, mode='valid')
        # Compute grad_u using formula:
        scalar = np.dot(grad_a, a)
        grad_u = a * (grad_a - scalar)
        # Adam update
        t += 1
        lr = step0 / (np.sqrt(t))   # perhaps constant step?
        # using typical Adam parameters: b1=0.9, b2=0.999, eps=1e-8
        m = 0.9 * m + (1-0.9) * grad_u
        v = 0.999 * v + (1-0.999) * (grad_u ** 2)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        u -= (lr / (np.sqrt(v_hat) + 1e-8)) * m_hat
        # recompute a from u via softmax
        a = np.exp(u - np.max(u))
        a /= a.sum()
    return best_a, best_score
```

def simulated_annealing(a, deadline, start_temp=1e-2, end_temp=1e-6, steps_per_temp=500):
   cur = a.copy()
   cur_score = compute_score(cur)
   best = cur.copy()
   best_score = cur_score
   total_iters = 0
   while time.time() < deadline:
       # schedule temperature
       progress = (time.time() - start) / (deadline - start)
       temp = start_temp * (end_temp/start_temp) ** progress
       # propose move: select random i,j; transfer fraction
       i, j = np.random.randint(n, size=2)
       if i == j: continue
       if cur[i] > 0:
           delta = temp * cur[i]  # magnitude scaled with temperature
           new = cur.copy()
           new[i] = max(0, new[i] - delta)
           new[j] += delta
           # renorm for numeric drift
           new /= new.sum()
           new_score = compute_score(new)
           if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score)/temp):
               cur = new; cur_score = new_score
               if new_score < best_score:
                   best = new.copy(); best_score = new_score
       total_iters += 1
   return best, best_score

def softmax_gradient_exponential(a0, start_score, deadline, beta_start=5., beta_end=3000., step0=0.3):
    n = len(a0)
    a = a0.copy()
    eps = 1e-12
    u = np.log(np.maximum(a, eps))
    best_a = a.copy()
    best_score = start_score
    it = 0
    start = time.time()
    while time.time() < deadline:
        # compute score for logging
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # schedule beta
        if deadline > start:
            frac = (time.time() - start) / (deadline - start)
        else:
            frac = 0
        beta = beta_start + frac * (beta_end - beta_start)
        # compute softmax w
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        # gradient wrt a
        grad_a = 2.0 * beta * np.correlate(w, a, mode='valid')
        # natural gradient wrt u
        scalar = np.dot(grad_a, a)
        grad_u = a * (grad_a - scalar)
        # step size
        it += 1

        u -= step * grad_u
        # new a
        a = np.exp(u - np.max(u))
        a /= a.sum()
        # logging occasional
        if it % 2000 == 0:
            print(f"[exp‑GD] it {it} | best {best_score:.6f} | β {beta:.0f}")
    return best_a, best_score

def simulated_annealing(a, deadline, init_temp=0.1, final_temp=1e-5):
    n = len(a)
    cur = a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    while time.time() < deadline:
        # compute temperature schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = init_temp * (final_temp / init_temp) ** frac
        # propose move: choose i, j
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # decide delta fraction of cur[i]
        max_delta = cur[i]  # can't exceed current mass
        # sample delta: maybe uniform up to step_scale*cur[i]
        # step_scale related to temp
        delta = min(max_delta, temp * cur[i] * 5.0)  # scaling factor
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)  # ensure non-negative
        cand /= cand.sum()
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score)/temp):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                # optional print
                if best_score < best_score - 0.0001:
                    print(f"[SA] new best {best_score:.6f}")
    return best, best_score

"""
Hybrid optimizer for the step‑function minimisation problem.

The algorithm consists of three main stages:
1.   Generate a diverse pool of seed distributions (deterministic,
     beta‑family, Gaussian, arcsine, random Dirichlet/spikes, etc.).
2.   Refine each seed independently using a *soft‑max* surrogate of the
     max‑convolution objective.  The optimisation is performed on the
     log‑parameters (exponentiated gradient / natural gradient) with an
     Adam‑style step size and an annealed temperature β.  This stage
     tends to produce very flat autocorrelations.
3.   Take the best result from stage 2 and run a simulated‑annealing
     hill‑climb that accepts occasional uphill moves.  The temperature
     decays linearly, allowing large exploratory moves early and fine
     tuning later.

All updates stay on the probability simplex, so the objective
`2·n·max(conv(a,a))` (identical to ``evaluate_sequence`` for a
normalised vector) can be computed exactly with an FFT‑based
convolution.  The code adapts the sequence length to the allowed run‑time
(bigger lengths when the budget permits) and returns the best sequence
found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_valid(e: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (e ⋆ a)[i] = Σ_m e[i+m] * a[m].
    Implemented via one FFT‑convolution with the reversed vector.
    """
    conv = convolution_fft(e, a[::-1])
    n = a.shape[0]
    return conv[n - 1 : n - 1 + n]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_arcsine(n: int) -> np.ndarray:
    """
    Approximation of the arcsine density on {0,…,n‑1}:
        a_i ∝ 1 / √(x·(1−x)),  x = (i+0.5)/n .
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = 1.0 / np.sqrt(np.clip(x * (1.0 - x), 1e-20, None))
    a = w / w.sum()
    return a


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the interval."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    w = np.exp(-((i - mid) ** 2) / (2.0 * sigma ** 2))
    a = w / w.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n−i))**p.
    Positive p concentrates mass in the centre,
    negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a

# Optimisation primitives

def softmax_exponential_gradient(
    a0: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 3000.0,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth log‑sum‑exp surrogate
        Lβ(a) = log Σ exp(β·conv(a,a))

    The optimisation is performed on the log‑parameters u (i.e. a_i = softmax(u))
    using a natural‑gradient (projected) step with a simple Adam‑like schedule.
    """
    n = a0.shape[0]
    a = a0.copy()
    eps = 1e-12
    u = np.log(np.maximum(a, eps))

    # Adam bookkeeping
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = a.copy()
    best_score = start_score
    it = 0
    start = time.time()

    while time.time() < deadline:
        # exact score for book‑keeping
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule
        frac = (time.time() - start) / max(deadline - start, 1e-12)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights over the convolution entries
        shifted = conv - max_conv               # max(conv) → 0
        exp_vals = np.exp(beta * shifted)      # all ≤ 1, numerically stable
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. a (smooth surrogate)
        grad_a = 2.0 * beta * correlation_valid(w, a)

        # natural gradient on the log‑parameters (simple mirror map)
        scalar = np.dot(grad_a, a)
        grad_u = a * (grad_a - scalar)

        # Adam update on u
        it += 1
        lr = step0 / np.sqrt(it)               # decaying learning rate
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u -= (lr / (np.sqrt(v_hat) + eps_adam)) * m_hat

        # recover a from u (soft‑max)
        a = np.exp(u - np.max(u))
        a /= a.sum()

        # occasional reporting
        if it % 2000 == 0:
            print(
                f"[exp‑GD] it {it:6d} | best {best_score:.6f} | β {beta:.0f}"
            )

    return best_a, best_score


def simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    init_temp: float = 0.01,
    final_temp: float = 1e-6,
    max_fraction: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a linear temperature schedule.
    Moves consist of transferring a random fraction of mass from one index
    to another; uphill moves are occasionally accepted according to a
    Metropolis criterion.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / max(deadline - start, 1e-12)
        temp = init_temp * (1.0 - frac) + final_temp * frac

        # propose a transfer move
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue

        # magnitude of transfer – up‑to `max_fraction` of the source weight,
        # scaled by temperature so that early moves are larger.
        max_move = cur[i] * max_fraction * (temp / init_temp + 0.1)
        delta = np.random.rand() * max_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # Clip any tiny negative values caused by rounding
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()                     # re‑normalise

        cand_score = compute_score(cand)

        # accept if improvement or with Metropolis probability
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[SA] new best {best_score:.6f} (it {it})")

        it += 1

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a list of floats (a probability vector).  The routine adapts the
    sequence length to the available time budget and combines a
    gradient‑based refinement with a simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the external harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0 – choose a workable sequence length
    # ------------------------------------------------------------------
    if total_time > 850.0:
        n = 4096
    else:
        n = 2048
    print(f"[setup] using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 1 – optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        # if the length differs, interpolate to the current size
        if len(fallback) != n:
            fallback = np.interp(
                np.linspace(0, len(fallback) - 1, n),
                np.arange(len(fallback)),
                fallback,
            ).tolist()
        fb_arr = np.array(fallback, dtype=float)
        fb_arr = np.maximum(fb_arr, 0.0)
        fb_arr /= fb_arr.sum()
        fb_val = compute_score(fb_arr)
        print(f"[fallback] score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fb_arr.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2 – generate a diverse set of seeds
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(generate_deterministic(n))
    seeds.append(generate_arcsine(n))
    seeds.append(generate_gaussian(n, sigma=n / 8.0))
    seeds.append(np.full(n, 1.0 / n))  # uniform

    # symmetric polynomial family
    for p in np.linspace(-3.0, 3.0, 13):
        seeds.append(generate_beta_family(n, p))

    # Beta pdf family
    for alpha, beta in [(0.2, 0.2), (0.5, 0.5), (1.0, 1.0), (2.0, 2.0),
                        (5.0, 5.0), (0.5, 2.0), (2.0, 0.5)]:
        seeds.append(generate_beta_pdf(n, alpha, beta))

    # random Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # sparse spikes – often good starting points
    for k in (30, 60, 100):
        seeds.append(random_spike(n, k=k))

    # include fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float))

    # ------------------------------------------------------------------
    # 3 – keep the K best seeds (K small to allow deeper optimisation)
    # ------------------------------------------------------------------
    K = 5
    scored_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)
        val = compute_score(s)
        scored_seeds.append((val, s))
        if val < best_val:
            best_val = val
            best_seq = s.tolist()
            print(f"[seed] new global best score = {best_val:.6f}")

    scored_seeds.sort(key=lambda x: x[0])
    top_seeds = [seq for (_, seq) in scored_seeds[:K]]

    # ------------------------------------------------------------------
    # 4 – time budgeting for the two optimisation phases
    # ------------------------------------------------------------------
    cur = time.time()
    remaining = deadline - cur
    grad_budget = remaining * 0.65            # bulk of the time for gradient descent
    anneal_budget = remaining - grad_budget - 0.5  # leave a small safety margin

    per_seed_budget = grad_budget / max(len(top_seeds), 1)

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent on the soft‑max surrogate
    # ------------------------------------------------------------------
    best_a = np.array(best_seq, dtype=float) if best_seq is not None else np.full(n, 1.0 / n)
    best_a /= best_a.sum()
    best_score = compute_score(best_a)

    for idx, seed_a in enumerate(top_seeds):
        phase_deadline = time.time() + per_seed_budget
        if time.time() >= deadline - 0.5:
            break
        print(f"\n[Phase 1] gradient refinement on seed {idx + 1}/{len(top_seeds)}")
        a_opt, sc_opt = softmax_exponential_gradient(
            seed_a,
            compute_score(seed_a),
            phase_deadline,
            beta_start=5.0,
            beta_end=3000.0,
            step0=0.3,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase 1] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("\n[Phase 2] simulated‑annealing polishing")
        a_final, sc_final = simulated_annealing(
            best_a,
            deadline,
            init_temp=best_score * 0.02,
            final_temp=best_score * 1e-6,
            max_fraction=0.10,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"[Phase 2] new best after SA → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5 – final conversion, safety clamp and official evaluation
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, x)) for x in best_a.tolist()]
    total = sum(final_seq)
    if total == 0.0:
        final_seq = [1.0 / n] * n
    else:
        final_seq = [x / total for x in final_seq]

    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
