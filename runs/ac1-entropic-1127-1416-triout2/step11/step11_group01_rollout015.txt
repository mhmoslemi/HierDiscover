"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

using a diversified search:
1.  Quick analytic and random scans over several families (multiple lengths).
2.  A continuation Adam optimisation with a smooth‑max surrogate and a large
    final β (≈5 000) – this directly attacks the max‑convolution objective.
3.  Simple mixing with the uniform distribution and with the reversed vector
    to flatten the autocorrelation further.
4.  A short stochastic‑annealing hill‑climb for final fine‑tuning.
All phases respect the global 1000 s wall‑clock budget and keep the
sequence normalised to sum = 1 (the true objective is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast for many iterations)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def double_exp_family(n: int, gamma: float) -> np.ndarray:
    """Sum of two exponentials decaying from both ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    a = w / w.sum()
    return a


def cosine_family(n: int, freq: float = 1.0) -> np.ndarray:
    """Cosine‑squared bump – always non‑negative."""
    i = np.arange(n, dtype=np.float64)
    w = np.cos(np.pi * freq * (i - (n - 1) / 2) / n) ** 2
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 2.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # numerically stable soft‑max over β·conv
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (see analysis notes)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:                     # modest progress reports
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# Simple stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Mixing helpers (uniform / reverse smoothing)
# ----------------------------------------------------------------------


def mix_with_uniform(a: np.ndarray,
                     max_t: float = 0.3,
                     steps: int = 6) -> tuple[np.ndarray, float]:
    """Convexly mix a with the uniform distribution to flatten the autocorrelation."""
    n = a.size
    uniform = np.full_like(a, 1.0 / n)
    best_a = a.copy()
    best_sc = compute_score(a)
    for i in range(1, steps + 1):
        t = (i / steps) * max_t
        cand = (1 - t) * a + t * uniform
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    return best_a, best_sc


def mix_with_reverse(a: np.ndarray,
                     max_t: float = 0.2,
                     steps: int = 5) -> tuple[np.ndarray, float]:
    """Mix a with its reversal – often reduces peak autocorrelation."""
    rev = a[::-1]
    best_a = a.copy()
    best_sc = compute_score(a)
    for i in range(1, steps + 1):
        t = (i / steps) * max_t
        cand = (1 - t) * a + t * rev
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence minimising `evaluate_sequence`.  The routine
    respects the global wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Quick multi‑length analytic / random scan
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]                 # a few sizes to diversify
    candidate_pool = []                         # (score, n, vector)

    for n in lengths:
        if time.time() >= deadline - 30:
            break
        print(f"[{time.time():.1f}] scanning length n={n}")

        # uniform baseline
        vec = np.full(n, 1.0 / n, dtype=np.float64)
        candidate_pool.append((compute_score(vec), n, vec))

        # families with modest parameter sweeps
        for alpha in np.linspace(0, 20, 11):
            if time.time() >= deadline - 30:
                break
            vec = symmetric_product_family(n, alpha)
            candidate_pool.append((compute_score(vec), n, vec))

        for beta in np.linspace(0, 20, 11):
            if time.time() >= deadline - 30:
                break
            vec = left_heavy_power_family(n, beta)
            candidate_pool.append((compute_score(vec), n, vec))

        for gamma in np.linspace(0, 12, 7):
            if time.time() >= deadline - 30:
                break
            vec = exp_u_family(n, gamma)
            candidate_pool.append((compute_score(vec), n, vec))

        for gamma in np.linspace(0, 8, 6):
            if time.time() >= deadline - 30:
                break
            vec = double_exp_family(n, gamma)
            candidate_pool.append((compute_score(vec), n, vec))

        for freq in (0.5, 1.0, 1.5):
            if time.time() >= deadline - 30:
                break
            vec = cosine_family(n, freq)
            candidate_pool.append((compute_score(vec), n, vec))

        # random monotone / U‑shape draws
        for _ in range(15):
            if time.time() >= deadline - 30:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            candidate_pool.append((compute_score(vec), n, vec))

        # Dirichlet draws – give truly random simplex points
        for _ in range(20):
            if time.time() >= deadline - 30:
                break
            vec = np.random.dirichlet(np.ones(n))
            candidate_pool.append((compute_score(vec), n, vec))

    # Keep the best few distinct seeds for intensive optimisation
    candidate_pool.sort(key=lambda x: x[0])
    top_k = 8
    selected = candidate_pool[:top_k]

    # ------------------------------------------------------------------
    # 2.  Continuation Adam optimisation on the best seeds
    # ------------------------------------------------------------------
    for init_sc, n, vec in selected:
        if time.time() >= deadline - 20:
            break
        a = np.array(vec, dtype=np.float64)
        if a.sum() == 0:
            continue
        a = a / a.sum()
        remaining = deadline - time.time()
        # allocate at most 150 s (or a proportion of what is left) per candidate
        sub_deadline = time.time() + min(remaining * 0.8, 150.0)
        print(f"[{time.time():.1f}] Adam start: n={n}, init_sc={init_sc:.6f}, time left≈{remaining:.1f}s")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            sub_deadline,
            max_iters=300_000,
            beta_start=2.0,
            beta_end=5000.0,
            lr=0.018,
        )
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] ** New global best (Adam) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Simple mixing refinements (uniform & reverse)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        a = np.array(best_seq, dtype=np.float64)
        a, sc = mix_with_uniform(a, max_t=0.3, steps=6)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] mixed with uniform → {best_score:.6f}")

        a, sc = mix_with_reverse(a, max_t=0.2, steps=5)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] mixed with reverse → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final short stochastic hill‑climb (SA)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        a = np.array(best_seq, dtype=np.float64)
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=500_000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after SA best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq if best_seq is not None else [1.0]
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
