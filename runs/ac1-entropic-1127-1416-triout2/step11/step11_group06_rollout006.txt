    m = len(a)+len(b)-1

    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)

    conv = np.fft.irfft(C, n=n_fft)[:m]

def smooth_gradient(a, p):
    conv = convolution_fft(a, a)  # length m
    # compute smoothing weights w
    pow_p = np.power(conv, p)
    S = pow_p.sum()
    if S == 0:

    conv_rev = convolution_fft(a[::-1], w)
    n = a.shape[0]
    # extract slice
    start = n-1
    g = 2.0 * conv_rev[start:start+n]

"""
Non‑negative step‑function optimiser (smooth mirror descent).

Main ideas
----------
* Generate a diverse portfolio of analytic seeds (uniform, central‑binomial,
  power‑law, geometric, Beta‑shaped, symmetrised variants).
* Pick the best seed w.r.t. the exact objective
  `2·n·max(conv(a,a))` (with `sum(a)=1`).
* Refine the seed with a *smooth* mirror‑descent:
     – minimise a smoothed Lₚ‑norm of the self‑convolution
       (p starts at 4 and is increased up to 12).
     – use an exponentiated‑gradient (KL‑mirror) update.
     – optionally enforce symmetry a = (a + a[::-1])/2 after each step.
* Perform a few cheap random‑restart bursts to escape shallow local minima.
* Finish with a stochastic hill‑climb (swap / tiny mass moves) to polish the
  result.
* All phases respect the global 1000 s wall‑clock deadline and keep the
  best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry by averaging with the reversed copy."""
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Symmetric and heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth‑gradient mirror descent
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, p: int) -> np.ndarray:
    """
    Gradient of the smoothed L_p norm of the self‑convolution:
        f_p(a) = (∑_k (conv(a,a)[k])^p)^{1/p}
    Returns the (sub‑)gradient ∇f_p(a) of shape (n,).
    """
    n = a.shape[0]

    # 1. self‑convolution
    conv = convolution_fft(a, a)                     # length 2n‑1

    # 2. smoothing weights w_k ∝ conv_k^{p‑1}
    pow_p = np.power(conv, p, out=np.empty_like(conv))
    S = pow_p.sum()
    if S <= 0.0:
        # degenerate case – fall back to uniform weights
        w = np.zeros_like(conv)
    else:
        # w_k = conv_k^{p‑1} / (∑ conv^p)^{(p‑1)/p}
        w = np.power(conv, p - 1, out=np.empty_like(conv))
        denom = S ** ((p - 1) / p)
        w /= denom

    # 3. correlation a_rev ⋆ w  (a_rev is a reversed)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad


def smooth_mirror_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.8,
    sym: bool = False,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) optimisation of the smoothed
    Lₚ‑norm of the self‑convolution.  ``sym`` enforces a = (a+a[::-1])/2
    after each update.
    Returns the best vector seen and its exact score.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # scheduling of p (smoothness) – slowly increase to sharpen the max
    phase_len = max(1, max_iters // 5)   # five phases of equal length

    for t in range(1, max_iters + 1):
        # time guard
        if time.time() >= time_deadline:
            break

        # choose p according to iteration count
        phase = (t - 1) // phase_len
        p = 4 + 2 * phase          # 4,6,8,10,12
        if p > 12:
            p = 12

        # gradient of the smooth surrogate
        g = smooth_gradient(a, p)

        # zero‑mean the gradient (helps with the KL‑mirror dynamics)
        avg = np.dot(a, g)

        # diminishing stepsize
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update
        exp_arg = -eta * (g - avg)
        # keep the exponent numerically safe
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # optional symmetry enforcement
        if sym:
            a = symmetrize(a)

        # project back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % 800 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps: int = 300_000):
    """
    Simple local stochastic moves (swap or fractional mass transfer)
    with simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a.tolist(), best_score
