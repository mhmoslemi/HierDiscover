"""
Propose a step‑function sequence that minimises the evaluation function.

Algorithm
---------
* Choose a problem size n (4096 by default, larger if time permits).
* Scan several analytic families on a fine grid (product, power, U‑shape)
  and keep the best.
* Generate a few random Dirichlet seeds (some sparse, some uniform).
* For every seed run a multi‑stage L‑BFGS optimisation on a smooth
  log‑sum‑exp approximation of the max‑convolution.  The smoothing
  parameter β is increased gradually (β = 5 … 30000) and each stage
  receives a generous number of L‑BFGS iterations (≈300).
* After the continuous optimisation enforce exact symmetry and re‑project
  onto the simplex.
* Finish with an extended simulated‑annealing hill‑climb that performs
  mass‑transfer and swap moves on the simplex.  The temperature schedule
  is linear and the step‑size is adapted on‑the‑fly.
* The best sequence seen during the whole run is returned as a plain
  Python list of floats.

All heavy operations (convolutions, gradients) use FFT‑based
implementations that scale O(n log n).  The routine respects the 1000 s
wall‑clock budget and prints intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive -> uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1).
    = 2·n·max(conv) where conv = a * a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (fast seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} (U‑shaped, symmetric)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow when alpha = 0
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = smooth approximation of max(conv)
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)

    The smoothing is the log‑sum‑exp of the convolution values.
    """
    a = softmax(z)                              # a is on the simplex
    conv = convolution_fft(a, a)                # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights over the convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta

    w = expw / sumexp                           # normalized soft‑max over positions
    a_rev = a[::-1]

    # weighted sum of the convolution gradients (via a second convolution)
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # gradient w.r.t. a (size n)

    # back‑propagation through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation wrapper (multi‑stage smoothing)
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    Returns the best a (still on the simplex) and its exact score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        # initialise the optimisation variables in the soft‑max space
        z0 = np.log(a + 1e-12)

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun, z0, method="L-BFGS-B", jac=True, options=opts)
        except Exception as exc:
            print(f"  L‑BFGS failure (β={beta}): {exc}")
            continue

        a = softmax(res.x)

        # optional symmetry enforcement (helps numerical stability)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved (β={beta}) → {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 4_000_000) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex.
    Moves: (50 % swap, 50 % transfer a fraction of mass).
    Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature roughly 0.1 % of the current score
    T0 = max(best_score * 1e-4, 1e-9)
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # Metropolis acceptance
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Random seed generators
# ----------------------------------------------------------------------


def random_dirichlet_seed(n: int, concentration: float = 1.0) -> np.ndarray:
    """Sample a Dirichlet(α) distribution; α < 1 yields sparsity."""
    x = np.random.gamma(concentration, 1.0, size=n)
    return x / x.sum()


def random_heavy_ends_seed(n: int) -> np.ndarray:
    """
    Produce a U‑shaped vector by allocating most mass near the ends
    and a tiny uniform background.
    """
    # base heavy‑ends shape
    base = symmetric_product_family(n, alpha=2.0)
    # blend with a very small uniform component to keep full support
    eps = 0.02
    vec = (1 - eps) * base + eps / n
    return project_to_simplex(vec)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` (i.e. 2·n·max(conv) with sum=1).
    The routine respects the wall‑clock budget and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Optional fallback using a previously discovered sequence
    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose problem size
    # -----------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 650 else 2048
    print(f"[{time.time():.1f}] using length n = {n}")

    # -----------------------------------------------------------
    # 2. Fine scan of analytic families (cheap)
    # -----------------------------------------------------------
    # product family (α)
    alphas = np.arange(0.0, 4.01, 0.08)   # ~51 points
    for alpha in alphas:
        if time.time() >= deadline - 150:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    # left‑heavy power (β)
    betas_fam = np.arange(0.0, 4.01, 0.08)
    for beta in betas_fam:
        if time.time() >= deadline - 150:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    # exponential U‑shape (γ)
    gammas = np.arange(0.0, 6.01, 0.12)
    for gamma in gammas:
        if time.time() >= deadline - 150:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Generate a handful of random seeds for L‑BFGS
    # -----------------------------------------------------------
    random_seeds = []
    # 3 Dirichlet with concentration 0.5 (sparse)
    for _ in range(3):
        random_seeds.append(random_dirichlet_seed(n, concentration=0.5))
    # 2 Dirichlet with concentration 1 (moderately uniform)
    for _ in range(2):
        random_seeds.append(random_dirichlet_seed(n, concentration=1.0))
    # 1 heavy‑ends seeded vector
    random_seeds.append(random_heavy_ends_seed(n))

    # add the current best (if any) as a seed
    if best_seq is not None:
        random_seeds.append(np.array(best_seq, dtype=np.float64))

    # -----------------------------------------------------------
    # 4. L‑BFGS refinement (multi‑start)
    # -----------------------------------------------------------
    betas_schedule = [5, 30, 150, 500, 2000, 8000, 20000]
    max_iter_stage = 300

    for idx, seed_vec in enumerate(random_seeds):
        if time.time() >= deadline - 120:  # reserve time for final SA
            break
        print(f"[{time.time():.1f}] L‑BFGS start #{idx + 1}")
        a_opt, sc_opt = l_bfgs_optimize(seed_vec,
                                        betas=betas_schedule,
                                        max_iter_per_stage=max_iter_stage,
                                        deadline=deadline - 80)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}]  *** new global best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Enforce perfect symmetry and re‑project onto simplex
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        sc_sym = compute_score(a_arr)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_arr.tolist()
            print(f"[{time.time():.1f}] symmetry projection improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final simulated‑annealing polishing (until time runs out)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=4_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Fallback if nothing found (very unlikely)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return best_seq
