        a_new[i] -= delta
        a_new[j] += delta

        # compute new score
        new_score = compute_score(a_new)

        # Accept with probability
        if new_score < cur_score:

            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # else reject

"""
Simple simulated‑annealing + sub‑gradient optimiser.

1.  A quick parametric sweep (power‑law / geometric / mixed) finds a
    decent starting shape.
2.  Repeated random‑restart simulated‑annealing hill climbs explore the
    simplex of non‑negative vectors.  Each iteration moves a small
    fraction of mass from a random source index to a random target
    index; worsening moves are accepted with a temperature‑controlled
    probability, allowing the search to escape shallow local minima.
3.  The best vector discovered is finally polished by a few
    sub‑gradient steps (the same routine used in the reference code).
4.  The whole search respects the supplied wall‑clock budget
    (default 1000 s) and returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

    for p in np.linspace(0.30, 1.00, 71):
        if time.time() >= deadline:
            return best
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best = val, a

    # geometric family
    for r in np.linspace(0.50, 0.99, 81):
        if time.time() >= deadline:
            return best
        a = (r ** np.arange(n, dtype=np.float64))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best = val, a

    # mixed power‑law + uniform head
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= deadline:
            return best
        L = int(L_frac * n)
        if L == 0 or L >= n:
            continue
        for p in np.linspace(0.4, 0.8, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head = np.full(L, tail.max())
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best = val, a
    return best


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing hill climbing (many random restarts)
# ----------------------------------------------------------------------
def annealing_hill_climb(a_start: np.ndarray,
                         time_deadline: float,
                         max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing search on the probability simplex."""
    n = a_start.shape[0]
    a = a_start.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    # annealing schedule parameters
    T0 = 0.012          # initial temperature
    init_step = 0.12    # initial fraction of mass moved each step

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # decay temperature and step size linearly
        frac = it / max_iters
        T = T0 * (1.0 - frac)
        step_scale = max(0.004, init_step * (1.0 - frac))

        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount to move (cannot exceed source mass)
        delta = step_scale * a[i]
        if delta <= 0.0:
            continue

        # propose new vector (mass transfer)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        new_score = compute_score(a_new)

        # acceptance test
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis rule
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # occasional jitter to keep motion lively
        if it % 10000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.03)
            a += jitter
            a = project_to_simplex(a)
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – sub‑gradient polishing (as in the reference)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray,
                       deadline: float,
                       step0: float = 0.08) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        a -= step * g
        a = project_to_simplex(a)

        # tiny jitter every 5000 steps
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a safety margin for harness bookkeeping
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 4096                       # length of the step function

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # 0. Optional fallback – use a previously discovered construction if
    #    it exists in the global namespace.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Global parametric sweep for a quick seed
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 300.0
    if time.time() < sweep_deadline:
        a0 = best_initial_vector(n, sweep_deadline)
        if a0 is not None:
            val = compute_score(a0)
            print(f"[global sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a0.tolist()

    # ------------------------------------------------------------------
    # 2. Random‑restart simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    while time.time() < deadline - 200.0:
        # pick an initial vector
        if best_seq is not None and np.random.rand() < 0.35:
            a_start = np.array(best_seq, dtype=float)
        else:
            # random Dirichlet‑like draw
            a_start = np.random.rand(n)
            a_start /= a_start.sum()

        # occasionally inject a random parametric shape
        if np.random.rand() < 0.20:
            p = np.random.uniform(0.3, 1.0)
            r = np.random.uniform(0.5, 0.99)
            a_start = ((np.arange(1, n + 1, dtype=np.float64) ** (-p)) *
                       (r ** np.arange(n, dtype=np.float64)))
            a_start = np.maximum(a_start, 0.0)
            if a_start.sum() > 0:
                a_start /= a_start.sum()
            else:
                a_start = np.ones(n) / n

        # allocate a chunk of time for this restart
        remaining = deadline - time.time()
        if remaining < 20.0:
            break
        restart_deadline = time.time() + min(30.0, remaining)

        a_best, val_best = annealing_hill_climb(a_start,
                                                restart_deadline,
                                                max_iters=150_000)
        if val_best < best_val:
            best_val, best_seq = val_best, a_best.tolist()
            print(f"[restart] new best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sub‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, deadline, step0=0.07)
        val = compute_score(a_polished)
        print(f"[polish] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # ------------------------------------------------------------------
    # Safeguard – fall back to uniform if we didn't manage to find anything
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final report using the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
