"""
Non‑negative step‑function optimiser (≈ 1000 s budget)

Algorithm
---------
1. **Global exploration** – sweep several parametric families
   (power‑law, exponential, beta) and a large pool of random Dirichlet
   draws.  The best few candidates (by the exact objective
   ``2·n·max(conv(a,a))``) are kept.

2. **Linear‑programming refinement** – for each retained candidate we
   repeatedly solve the LP

       maximise   1ᵀb
       subject to  T_f · b ≤ M·1 ,   b ≥ 0

   where ``T_f`` is the (2·n‑1)×n convolution matrix of the current
   vector ``f`` and ``M = max(conv(f,f))``.  The solution ``b`` is
   normalised to the probability simplex and mixed with ``f`` by a
   small convex step.  Each iteration provably does not increase the
   objective, and in practice drives the max‑convolution towards the
   uniform lower bound.

3. **Projected sub‑gradient polishing** – a few hundred thousand
   sub‑gradient steps (with Nesterov momentum) fine‑tune the best LP
   result.

4. **Simulated annealing** – a final stochastic hill‑climb with
   decreasing temperature to escape possible shallow minima.

All phases respect the overall 1000 s wall‑clock budget.  The function
returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

# ------------------------------------------------------------
# Imports
# ------------------------------------------------------------
import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ------------------------------------------------------------
# Numerical utilities
# ------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Parametric families
# ------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}  (discrete Beta‑shaped density)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ------------------------------------------------------------
# Global baseline exploration
# ------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """Explore a few parametric families and many random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []                     # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law (wide sweep, including steep tails)
    for p in np.linspace(0.2, 1.2, 101):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.0001, 0.05, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep (coarse grid)
    al_vals = np.linspace(0.1, 5.0, 19)          # step ≈0.26
    be_vals = np.linspace(0.1, 5.0, 19)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Random Dirichlet draws – diverse concentration parameters
    concs = [0.1, 0.3, 0.5, 1.0, 2.0]
    samples_per = 200                          # ≈ 1000 draws total
    for conc in concs:
        for _ in range(samples_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ------------------------------------------------------------
# Linear‑programming refinement
# ------------------------------------------------------------


def build_conv_indices(n: int):
    """Pre‑compute row/column index arrays for the (2·n‑1)×n convolution matrix."""
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    return rows.astype(np.int32), cols.astype(np.int32)


def lp_refine(a0: np.ndarray, rows, cols, deadline: float,
              max_iter: int = 25, mix_factor: float = 0.5):
    """Iteratively improve ``a0`` by solving the LP described in the paper.
    Returns the best vector found and its score."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current maximal autocorrelation value
        conv = full_convolution(a)
        M = conv.max()

        # Build sparse convolution matrix T_f (shape (2n-1, n))
        data = np.repeat(a, n)                # every row uses the same a_i
        T = sp.csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Linear programme: maximise 1ᵀb  subject to T·b ≤ M, b ≥ 0
        c = -np.ones(n)                       # linprog minimises, so -1ᵀb
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[LP] iteration {it} – linprog failed: {res.message}")
            break

        b = res.x
        sb = b.sum()
        if sb == 0.0:
            break
        g = b / sb                               # re‑normalise to simplex

        # convex combination – small step guarantees non‑increase of the max
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = project_to_simplex(a)               # safety projection

        cur_score = compute_score(a)
        print(f"[LP] it {it:02d} | M={M:.6e} | score={cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ------------------------------------------------------------
# Projected sub‑gradient polishing (with momentum)
# ------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30, momentum: float = 0.9) -> np.ndarray:
    """Projected sub‑gradient descent with simple momentum."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    vel = np.zeros_like(a)          # momentum buffer
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate gradient from all active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        vel = momentum * vel - step * g
        a += vel
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if t % 3000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb)
# ------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Simple Metropolis hill‑climb with temperature decay."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

    best_seq = None

        a0 = np.array(best_seq, dtype=float)

    # -------------------------------------------------

    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)

    return best_seq
