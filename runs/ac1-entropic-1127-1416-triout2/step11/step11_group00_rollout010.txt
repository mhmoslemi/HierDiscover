def grad_max_smoothed(a, lam):
    conv = full_convolution(a)
    max_val = np.max(conv)
    # compute w stable
    scaled = lam * (conv - max_val)  # subtract max to avoid overflow
    exp_scaled = np.exp(scaled)
    w = exp_scaled / np.sum(exp_scaled)
    # compute gradient
    w_rev = w[::-1]
    conv_rev = full_convolution(w_rev, a)
    # gradient index mapping
    n = a.shape[0]
    grad_a = 2.0 * conv_rev[2*n - 2 - np.arange(n)]
    return grad_a, max(conv)  # also return current max conv for evaluation
```

def smooth_gradient_descent(a0, deadline, max_iters=5000, init_lambda=5.0, lambda_factor=1.05):
   u = np.log(np.maximum(a0, 1e-12))
   m = np.zeros_like(u)  # Adam first moment
   v = np.zeros_like(u)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   t = 0
   lam = init_lambda
   best_a = a0.copy()
   best_val = compute_score(a0)
   while time.time() < deadline and t < max_iters:
       t += 1
       # compute a
       exp_u = np.exp(u - np.max(u))
       a = exp_u / np.sum(exp_u)
       grad_a, cur_max = grad_max_smoothed(a, lam)
       # transform to grad_u
       dot = np.dot(a, grad_a)
       grad_u = a * (grad_a - dot)
       
       # Adam update
       m = beta1 * m + (1 - beta1) * grad_u
       v = beta2 * v + (1 - beta2) * (grad_u ** 2)
       m_hat = m / (1 - beta1 ** t)
       v_hat = v / (1 - beta2 ** t)
       lr = 0.5 / (np.sqrt(t) + 1)  # decreasing learning rate
       u -= lr * m_hat / (np.sqrt(v_hat) + eps)
       
       # occasionally renormalize u to avoid overflow
       if t % 1000 == 0:
           u -= np.max(u)  # shift to keep values manageable
       
       # compute score using exact objective
       if t % 200 == 0:
           val = compute_score(a)
           if val < best_val:
               best_val = val
               best_a = a.copy()
       
       # increase lambda slowly
       lam = min(200.0, lam * lambda_factor)
   return best_a, best_val

def grad_max_smoothed(a, lam):
    conv = full_convolution(a)
    max_val = conv.max()
    # Compute softmax weights
    # to prevent overflow, subtract max conv from conv before scaling
    # Actually we want exp(lam * (conv - max_val)) not lam*conv - lam*max_val
    scaled = lam * (conv - max_val)  # note lam*(conv - max)
    exp_scaled = np.exp(scaled)
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp
    # compute gradient 2 * sum_k w_k a_{k-i}
    w_rev = w[::-1]
    conv_rev = full_convolution(w_rev, a)
    n = a.shape[0]
    grad_a = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]
    return grad_a, max_val

def smooth_gradient_descent(a0, deadline, init_lambda=5.0, lambda_factor=1.001,
                            max_iters=5000, lr0=0.3):
    a = a0.copy()
    n = a.shape[0]
    # initialize parameters u = log(a)
    eps = 1e-12
    u = np.log(a + eps)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0
    lam = init_lambda
    best_a = a.copy()
    best_val = compute_score(a)
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute softmax
        max_u = np.max(u)     # for numeric stability
        exp_u = np.exp(u - max_u)
        a = exp_u / np.sum(exp_u)
        # compute gradient of smoothed max
        grad_a, max_conv = grad_max_smoothed(a, lam)
        # transform to gradient w.r.t u (softmax)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        # step size schedule
        lr = lr0 / np.sqrt(t)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # update lambda gradually
        lam = min(150.0, lam * lambda_factor)
        # occasionally evaluate actual objective
        if t % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
        # check time
        if t % 1000 == 0 and time.time() >= deadline:
            break
    # final check
    val = compute_score(a)
    if val < best_val:
        best_val = val
        best_a = a.copy()
    return best_a, best_val
```

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The optimiser combines several phases:
1.  A diverse global search over parametric families (power‑law,
    exponential, Beta‐shaped, sinusoidal) and many random Dirichlet draws.
    The best few candidates (by the exact objective 2·n·max(conv))
    are retained.

2.  A smooth‑max gradient descent.  The non‑differentiable max is
    approximated by a log‑sum‑exp with a temperature λ that is slowly
    increased.  The gradient of the smoothed objective is computed via
    FFT‑based convolutions, and an Adam optimiser updates a soft‑max
    parametrisation of the probability vector (guaranteeing non‑negativity
    and ∑a = 1).

3.  An optional short simulated‑annealing hill‑climb for final polishing.

All phases respect a wall‑clock deadline (default 1000 s).  The
function returns a plain Python list of floats that minimises
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def sort_key(item):
    """Key function for sorting (item is a tuple (score, vector))."""
    return item[0]


def vector_score(vec):
    """Return the exact objective for a normalised probability vector."""
    return compute_score(vec)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Guarantees non‑negative entries that sum to one.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    If ``b`` is omitted, convolve ``a`` with itself.
    """
    if b is None:
        b = a
    n = a.shape[0]
    m = b.shape[0]
    size = n + m - 1
    n_fft = 1 << ((size - 1).bit_length())          # next power of two ≥ size
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:size]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
    2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)                # a*a, length 2·n‑1
    return float(2.0 * n * conv.max())

# Parametric families for the global search
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} (discrete Beta density)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_sine(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^{power} – a smooth bell‑shape."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total

def try_candidate(vec: np.ndarray, best: list, max_keep: int):
    """Validate a candidate, project it onto the simplex and possibly store it."""
    if not np.isfinite(vec).all():
        return
    vec = np.maximum(vec, 0.0)
    vec = project_to_simplex(vec)
    sc = compute_score(vec)
    if len(best) < max_keep:
        best.append((sc, vec.copy()))
        best.sort(key=sort_key)
    elif sc < best[-1][0]:
        best[-1] = (sc, vec.copy())
        best.sort(key=sort_key)


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """Explore several families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []                     # list of (score, vector)

    # 1) power‑law sweep
    for p in np.linspace(0.4, 1.5, 141):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p), best, max_keep)

    # 2) exponential sweep
    for lam in np.linspace(0.0001, 0.05, 250):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam), best, max_keep)

    # 3) Beta sweep (moderately fine grid)
    al_vals = np.linspace(0.1, 5.0, 21)
    be_vals = np.linspace(0.1, 5.0, 21)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta), best, max_keep)

    # 4) Sinusoidal family
    for power in np.linspace(0.5, 4.0, 15):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine(n, power), best, max_keep)

    # 5) Random Dirichlet draws – varied concentration parameters
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per = 30                              # total ≈ 180 draws
    for conc in concs:
        for _ in range(draws_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a, best, max_keep)

    best.sort(key=sort_key)
    return [vec for sc, vec in best]

# Smoothed‑max gradient descent (Adam optimiser)
# ----------------------------------------------------------------------


def grad_max_smoothed(a: np.ndarray, lam: float):
    """
    Gradient of the smoothed max (log‑sum‑exp) of the convolution.
    Returns (grad_a, current_max_conv).
    """
    conv = full_convolution(a)               # length 2·n‑1
    max_val = conv.max()

    # Soft‑max weights with temperature λ.
    scaled = lam * (conv - max_val)          # shift for numerical stability
    exp_scaled = np.exp(scaled)
    w = exp_scaled / np.sum(exp_scaled)      # w sums to 1, length 2·n‑1

    # Gradient: ∂/∂a_i  (1/λ) log Σ exp(λ·conv_k) = Σ_k w_k ∂conv_k/∂a_i
    # and ∂conv_k/∂a_i = 2·a_{k‑i} (zero when out of range).
    w_rev = w[::-1]
    conv_rev = full_convolution(w_rev, a)   # length 3·n‑2
    n = a.shape[0]
    grad_a = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]
    return grad_a, max_val


def smooth_gradient_descent(a0: np.ndarray, deadline: float,
                           init_lambda: float = 5.0,
                           lambda_factor: float = 1.001,
                           max_iters: int = 5000,
                           lr0: float = 0.3) -> tuple:
    """
    Adam optimiser on the smoothed max objective.
    Returns (best_vector, best_score) where the score is the exact
    2·n·max(conv) value.
    """
    n = a0.shape[0]
    eps = 1e-12
    u = np.log(a0 + eps)                     # soft‑max parameters
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    lam = init_lambda
    best_a = a0.copy()
    best_val = compute_score(best_a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # Soft‑max to obtain a valid probability vector
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / np.sum(exp_u)

        # Smoothed‑max gradient of the convolution
        grad_a, _ = grad_max_smoothed(a, lam)

        # Chain rule through the soft‑max
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)               # decreasing step size
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Increase the smoothing temperature slowly
        lam = min(150.0, lam * lambda_factor)

        # Periodic exact evaluation
        if t % 200 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()

        # occasional re‑centering to avoid overflow
        if t % 1000 == 0:
            u -= np.max(u)

    # Final check
    final_val = compute_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val

def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climb that can accept occasional worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # reject
            step_scale = max(step_scale * 0.98, step_scale_min)

        # temperature schedule
        T = max(final_temp, T * cooling_factor)

        # occasional global jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
