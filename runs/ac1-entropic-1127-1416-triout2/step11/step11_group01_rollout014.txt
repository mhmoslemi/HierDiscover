def lp_refine_step_sparse(f, M, mix_weight=0.2):
    n = f.shape[0]
    m = 2*n-1
    # Build sparse matrix list
    from scipy.sparse import coo_matrix
    rows = []
    cols = []
    data = []
    for i in range(n):
        row_start = i
        row_end = i + n
        # rows i...i+n-1, column i
        rows.extend(range(row_start, row_end))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b = np.full(m, M)
    # use linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10})
    if not res.success:
        return None
    g = np.maximum(res.x, 0)
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    g_norm = g / sum_g
    f_new = (1 - mix_weight) * f + mix_weight * g_norm
    # project onto simplex
    f_new = project_to_simplex(f_new)
    return f_new

"""
Propose a rectangular non‑negative step‑function (list of heights) that
minimises the evaluation‐function

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages:

1. **Very extensive analytic scan** – families (symmetric products,
   left‑heavy power, U‑shaped exponential, monotone exponential,
   logistic) are swept on a fine grid (including fractional parameters)
   and many random mixtures are tried.  The 20 best vectors are retained.

2. **Continuous optimisation** – starting from the best retained vector
   we run an exponentiated‑gradient (mirror‑descent) optimisation on the
   true max‑convolution objective, followed by a smoothed‑max Adam
   refinement and a cheap projected sub‑gradient descent.  The time
   budget is split dynamically between the three phases.

3. **Final stochastic hill‑climb** – a simulated‑annealing style walk
   refines the solution for the remaining budget.

All phases respect a global wall‑clock limit (default 1000 s).  The
returned Python list is ready for ``evaluate_sequence``.
"""

import time
import math

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-0.5-i))^{-alpha} (symmetric, ends‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-beta}  – monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential: w_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def monotone_exp_family(n: int, theta: float) -> np.ndarray:
    """Pure exponential decay from the left side."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-theta * i)
    a = w / w.sum()
    return a


def logistic_family(n: int, steep: float, offset: float) -> np.ndarray:
    """Logistic decay (sigmoidal)."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / (1.0 + np.exp(steep * (i - offset)))
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and U‑shaped exponential."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random monotone decreasing start (power‑law like)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shape via exp_u_family."""
    gamma = np.random.uniform(0.0, 30.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimisation
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """Minimise the smooth max surrogate via Adam."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (via convolution with w)
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# LP‑based refinement (sparse implementation)
# ----------------------------------------------------------------------


def lp_refine_step_sparse(f: np.ndarray, M: float, mix_weight: float = 0.2):
    """One LP refinement step using a sparse convolution matrix."""
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (size m×n) such that A·g = f * g
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows i .. i+n-1 receive f[i] in column i
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()

    c = -np.ones(n)                     # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None)] * n
    b = np.full(m, M)

    # Solve with a modest time limit per LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5})
    if not res.success:
        return None

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return None
    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    """

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use at least 2048 entries)
    # ------------------------------------------------------------------
    fallback_len = 0
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_len = len(fb_vec)
    except Exception:
        fb_vec = None

    n = max(fallback_len, 2048)
    print(f"[{time.time():.1f}] problem size n = {n}")

    # ------------------------------------------------------------------
    # Helper to keep the best K candidates
    # ------------------------------------------------------------------
    top_k = 20
    best_candidates = []   # list of tuples (score, vector)

        if len(best_candidates) < top_k:
            best_candidates.append((sc, vec.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:

                best_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if any)
    # ------------------------------------------------------------------
    if fb_vec is not None:
        fb = np.array(fb_vec, dtype=np.float64)
        fb = np.maximum(fb, 0.0)
        if fb.sum() == 0.0:
            fb = np.full_like(fb, 1.0 / fb.size)
        else:
            fb = fb / fb.sum()
        track(fb)
        print(f"[{time.time():.1f}] fallback score = {compute_score(fb):.6f}")

    # ------------------------------------------------------------------
    # 1.  Fine‑grained analytic families scan
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] starting fine analytic scan")
    # symmetric product (alpha from 0.0 to 5.0 in steps of 0.05)
    for alpha in np.arange(0.0, 5.001, 0.05):
        if time.time() >= deadline - 30:
            break
        vec = symmetric_product_family(n, alpha)
        track(vec)

    # left‑heavy power (beta from 0.0 to 5.0, step 0.05)
    for beta in np.arange(0.0, 5.001, 0.05):
        if time.time() >= deadline - 30:
            break
        vec = left_heavy_power_family(n, beta)
        track(vec)

    # exponential U‑shape (gamma 0–30 step 0.3)
    for gamma in np.arange(0.0, 30.001, 0.3):
        if time.time() >= deadline - 30:
            break
        vec = exp_u_family(n, gamma)
        track(vec)

    # monotone exponential (theta 0–12 step 0.2)
    for theta in np.arange(0.0, 12.001, 0.2):
        if time.time() >= deadline - 30:
            break
        vec = monotone_exp_family(n, theta)
        track(vec)

    # logistic family (coarse sweep)
    steep_vals = np.linspace(0.2, 8.0, 16)
    offset_vals = np.linspace(0.0, n - 1, 16)
    for steep in steep_vals:
        for offset in offset_vals:
            if time.time() >= deadline - 30:
                break
            vec = logistic_family(n, steep, offset)
            track(vec)

    # mixed uniform + U‑shape random
    for _ in range(80):
        if time.time() >= deadline - 30:
            break
        lam = np.random.rand()
        gamma = np.random.uniform(0.0, 30.0)
        vec = mixed_uniform_u_family(n, lam, gamma)
        track(vec)

    # Random monotone & random U‑shape starts
    for _ in range(300):
        if time.time() >= deadline - 30:
            break
        if np.random.rand() < 0.5:
            vec = random_monotone_start(n)
        else:
            vec = random_u_start(n)
        track(vec)

    # ------------------------------------------------------------------
    # 2.  Pick the current best and start continuous optimisation
    # ------------------------------------------------------------------
    best_vec = best_candidates[0][1].copy()
    best_score = compute_score(best_vec)
    print(f"[{time.time():.1f}] best after scanning = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2a. Exponentiated‑gradient phase (major share of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 250:
        eg_deadline = min(deadline - 150, time.time() + 150)
        print(f"[{time.time():.1f}] EG optimisation (deadline {eg_deadline:.1f})")
        best_vec, eg_score = eg_optimize(best_vec, eg_deadline,
                                         max_iters=400_000, eta0=0.6)
        if eg_score < best_score:
            best_score = eg_score
            print(f"[{time.time():.1f}] EG new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2b. Adam‑based smoothed‑max refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        adam_deadline = min(deadline - 80, time.time() + 80)
        print(f"[{time.time():.1f}] Adam refinement (deadline {adam_deadline:.1f})")
        best_vec, adam_score = adam_smooth_optimize(best_vec, adam_deadline,
                                                    max_iters=200_000,
                                                    beta_start=4.0,
                                                    beta_end=2000.0,
                                                    lr=0.018)
        if adam_score < best_score:
            best_score = adam_score
            print(f"[{time.time():.1f}] Adam new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2c. Very cheap projected sub‑gradient descent (long tail)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        sub_deadline = min(deadline - 30, time.time() + 30)
        print(f"[{time.time():.1f}] Sub‑gradient descent (deadline {sub_deadline:.1f})")
        best_vec, sub_score = subgrad_optimize(best_vec, sub_deadline,
                                               max_iters=4_000_000,
                                               step0=0.07)
        if sub_score < best_score:
            best_score = sub_score
            print(f"[{time.time():.1f}] Sub‑grad new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Optional LP refinement loop (sparse, limited iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        print(f"[{time.time():.1f}] LP refinement loop (sparse)")
        mix_weight = 0.15
        lp_iters = 0
        while time.time() < deadline - 10 and lp_iters < 25:
            conv = convolution_fft(best_vec, best_vec)
            M = float(conv.max())
            new_vec = lp_refine_step_sparse(best_vec, M, mix_weight=mix_weight)
            if new_vec is None:
                break
            new_score = compute_score(new_vec)
            if new_score < best_score - 1e-7:
                best_score = new_score
                best_vec = new_vec
                print(f"[{time.time():.1f}] LP step {lp_iters} improved to {best_score:.6f}")
                mix_weight = max(0.05, mix_weight * 0.9)
            else:
                mix_weight = min(0.3, mix_weight * 1.1)
            lp_iters += 1

    # ------------------------------------------------------------------
    # 4.  Symmetrisation test (often beneficial)
    # ------------------------------------------------------------------
    a_sym = (best_vec + best_vec[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sym_score = compute_score(a_sym)
    if sym_score < best_score:
        best_score = sym_score
        best_vec = a_sym
        print(f"[{time.time():.1f}] Symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        best_vec, sa_score = hill_climb_sa(best_vec, deadline, max_steps=2_000_000)
        if sa_score < best_score:
            best_score = sa_score
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Return the best-found sequence
    # ------------------------------------------------------------------
    final_seq = best_vec.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
