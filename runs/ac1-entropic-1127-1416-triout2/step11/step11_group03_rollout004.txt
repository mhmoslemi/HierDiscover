"""Optimising a non‑negative step‑function for the max‑auto‑convolution objective.

The algorithm works in three phases:
  1. A quick scan of several analytic families (product‑power, power‑law,
     exponential‑U, beta, Gaussian, edge‑bimodal) on a medium‑size grid.
     The best parameter of each family is refined with golden‑section search.
  2. Heavy continuous optimisation:
     – Adam on a smooth‐max surrogate (temperature β slowly raised up to 2·10⁴)
       with symmetry enforcement and projection onto the probability simplex.
     – Projected sub‑gradient descent on the true max‑objective.
     – A second Adam pass with an even higher temperature for fine tuning.
  3. A final simulated‑annealing hill‑climb for last polishing.

All vectors stay on the simplex (∑a_i = 1) and are kept symmetric after
each major step.  The routine respects the 1000 s wall‑clock budget and
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
      a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the grid: (i+½)^{p‑1}(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (symmetrised automatically)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    if w.sum() == 0.0:
        w[:] = 1.0
    w /= w.sum()
    return w


def bimodal_edge_family(n: int, sigma: float) -> np.ndarray:
    """Two Gaussian bumps glued to the edges (symmetrised)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * (i / sigma) ** 2) + np.exp(-0.5 * ((n - 1 - i) / sigma) ** 2)
    if w.sum() == 0.0:
        w[:] = 1.0
    w /= w.sum()
    return w

# Optimisers
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 800_000,
                         beta_start: float = 5.0,
                         beta_end: float = 20_000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    start_a : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    deadline : float
        Unix timestamp after which the optimisation must stop.
    max_iters : int
        Upper bound on the number of Adam iterations.
    beta_start / beta_end : float
        Temperature schedule for the smooth‑max (exponential annealing).
    lr : float
        Base learning rate.
    sym_interval : int
        Every ``sym_interval`` steps the iterate is symmetrised and re‑projected.

    Returns
    -------
    best_a, best_score
        Best vector found and its exact score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # logits for the simplex (softmax re‑parameterisation)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        # self‑convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a of the smooth‑max objective
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective check
        if it % 3000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection and evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient (indices that realise the maximum)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature roughly 1e‑4 of the current score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section refinement (single‑parameter families)
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 15) -> Tuple[float, float]:
    """
    Minimises ``func`` on the interval [a,b] using golden‑section search.
    Returns the best parameter and its function value.
    """
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)

    for _ in range(max_iter):
        if fc < fd:          # we are minimising
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)

    if fc < fd:
        return c, fc
    else:
        return d, fd

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback (previous best)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working length (larger N gives finer discretisation)
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 2. Coarse scan of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan (n={n})")
    best_alpha: float | None = None
    best_sigma: float | None = None
    best_sigma_bi: float | None = None

    # product‑power family
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 20.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")

    # exponential‑U family
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")

    # beta family (grid)
    for p in np.linspace(0.1, 3.0, 13):
        if time.time() >= deadline - 30.0:
            break
        for q in np.linspace(0.1, 3.0, 13):
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (beta p={p:.2f},q={q:.2f}) = {best_score:.6f}")

    # mixture uniform + U shape
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

    # Gaussian family (sigma in absolute points)
    sigma_vals = np.linspace(0.05 * n, 0.5 * n, 20)
    for sigma in sigma_vals:
        if time.time() >= deadline - 30.0:
            break
        vec = gaussian_family(n, sigma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_sigma = sigma
            print(f"[{time.time():.1f}] new best (gauss σ={sigma:.1f}) = {best_score:.6f}")

    # Bimodal edge family
    sigma_vals_bi = np.linspace(0.03 * n, 0.4 * n, 18)
    for sigma in sigma_vals_bi:
        if time.time() >= deadline - 30.0:
            break
        vec = bimodal_edge_family(n, sigma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_sigma_bi = sigma
            print(f"[{time.time():.1f}] new best (bimodal σ={sigma:.1f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Golden‑section refinements of promising one‑parameter families
    # ------------------------------------------------------------------
    # product family refinement
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product‑α around {best_alpha:.3f}")

        def score_alpha(a_val: float) -> float:
            return compute_score(symmetric_product_family(n, a_val))

        lo = max(0.0, best_alpha - 0.6)
        hi = min(12.0, best_alpha + 0.6)
        best_alpha, _ = golden_section_optimize(score_alpha, lo, hi, max_iter=15)
        vec = symmetric_product_family(n, best_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] refined product‑α={best_alpha:.6f} → score {best_score:.6f}")

    # Gaussian refinement
    if best_sigma is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining Gaussian σ around {best_sigma:.1f}")

        def score_sigma(s: float) -> float:
            return compute_score(gaussian_family(n, s))

        lo = max(0.02 * n, best_sigma - 0.2 * n)
        hi = min(0.6 * n, best_sigma + 0.2 * n)
        best_sigma, _ = golden_section_optimize(score_sigma, lo, hi, max_iter=15)
        vec = gaussian_family(n, best_sigma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] refined Gaussian σ={best_sigma:.3f} → score {best_score:.6f}")

    # Bimodal refinement
    if best_sigma_bi is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining bimodal σ around {best_sigma_bi:.1f}")

        def score_sigma_bi(s: float) -> float:
            return compute_score(bimodal_edge_family(n, s))

        lo = max(0.02 * n, best_sigma_bi - 0.2 * n)
        hi = min(0.5 * n, best_sigma_bi + 0.2 * n)
        best_sigma_bi, _ = golden_section_optimize(score_sigma_bi, lo, hi, max_iter=15)
        vec = bimodal_edge_family(n, best_sigma_bi)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] refined bimodal σ={best_sigma_bi:.3f} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. First Adam refinement (smooth‑max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] first Adam refinement (smooth‑max, β up to 2e4)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 150.0,
            max_iters=800_000,
            beta_start=5.0,
            beta_end=20_000.0,
            lr=0.015,
            sym_interval=2000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 50.0,
            max_iters=2_500_000,
            step0=0.045,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second Adam pass (higher β, finer tuning)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass (smooth‑max, β up to 8e4)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=400_000,
            beta_start=20.0,
            beta_end=80_000.0,
            lr=0.012,
            sym_interval=1500,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Explicit symmetrisation test
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=5_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Fallback to a safe uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # Final evaluation through the official wrapper (includes safety checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
