"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Global exploration** – sample a handful of structured families
   (power‑law, exponential, beta) plus random Dirichlet draws.
   Keep the best few candidates (by the exact objective
   ``2·n·max(conv(a,a))``).

2. **Smooth‐max Adam optimisation** – starting from the best candidate,
   minimise a smooth approximation of the max‑convolution using the
   Adam optimiser.  The soft‑max temperature (parameter ``α``) is ramped
   from a small to a large value, turning the surrogate from a smooth
   average into a sharp approximation of the true maximum.
   Gradients are obtained by a single FFT‑based correlation, the update
   is projected onto the probability simplex after each step.

3. **Final evaluation** – the best sequence found is fed to the official
   ``evaluate_sequence`` function and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector with non‑negative entries summing to 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    N = next_pow2(m)
    A = np.fft.rfft(a, n=N)
    C = A * A
    conv = np.fft.irfft(C, n=N)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for the initial global search
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑shaped density."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Baseline global exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """
    Explore a few parametric families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays).
    """
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) sweep power‑law (extended range)
    for p in np.linspace(0.1, 2.0, 191):  # step ≈0.01
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) sweep exponential decay (extended range)
    for lam in np.linspace(1e-4, 0.05, 501):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta family (moderately fine grid)
    al_vals = np.linspace(0.2, 5.0, 25)
    be_vals = np.linspace(0.2, 5.0, 25)
    for alpha in al_vals:
        for beta_param in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta_param))

    # 4) Random Dirichlet draws – diverse concentrations
    concs = [0.2, 0.5, 1.0, 2.0, 5.0]
    samples_per = 100
    for conc in concs:
        for _ in range(samples_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam optimiser on a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  max_iter: int = 200_000,
                  lr: float = 0.20,
                  alpha_start: float = 10.0,
                  alpha_end: float = 400.0,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser that minimises a smoothed max‑convolution.
    The smoothing parameter `α` grows linearly from ``alpha_start`` to
    ``alpha_end``.  The iterate is projected onto the simplex after each
    update.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam moment estimates
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # exponential decay of the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)

        # full linear convolution a*a
        conv = full_convolution(a)                # length 2n‑1
        max_conv = conv.max()

        # soft‑max probabilities (stable implementation)
        #   p_k ∝ exp(α·conv_k)
        shifted = conv - max_conv                 # avoid overflow
        exps = np.exp(alpha * shifted)
        p = exps / exps.sum()                     # length 2n‑1, sums to 1

        # ---- gradient of the soft‑max max‑convolution ----
        # grad_j = 2 * Σ_k p_k * a_{k‑j}
        # This is a correlation of p with a:
        #   g_j = Σ_i a_i * p_{i+j}
        # We'll compute it via FFT (linear correlation).
        len_corr = p.shape[0] + a.shape[0] - 1   # = 3n‑2
        N = next_pow2(len_corr)

        P = np.fft.rfft(p, n=N)
        A = np.fft.rfft(a, n=N)

        # linear correlation = irfft(P * conj(A))
        corr = np.fft.irfft(P * np.conj(A), n=N)

        # slice the needed n entries (indices n‑1 … n‑1+n‑1)
        g = corr[n - 1: n - 1 + n]

        grad = 2.0 * g

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        # current objective (exact, not the surrogate)
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny jitter to help escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (lr * 0.005)
            a = a + jitter
            a = project_to_simplex(a)

    return best_a

    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
