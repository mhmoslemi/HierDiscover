def lp_refine(a_init: np.ndarray, deadline: float, max_iters: int = 10, t: float = 0.2) -> np.ndarray:
    a = a_init.copy()
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, 1.0 / a.shape[0])
        a_sum = 1.0
    a = a / a_sum   # normalize sum=1
    n = a.shape[0]
    # precompute convolution matrix A (2n-1 x n)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a  # each column
    # We might need to recompute A each iteration because a changes.
    for it in range(max_iters):
        if time.time() >= deadline - 5: break
        conv = convolution_fft(a, a)
        M = conv.max()
        # build A based on current a
        # we can reuse code vectorized: via toeplitz approach? But simpler: fill row.
        # Instead we rebuild each time:
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = a
        # solve LP: maximize sum(b) subject to A @ b <= M
        c = -np.ones(n)
        ub = M * np.ones(rows)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP solve failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b < 1e-12:
            break
        # normalize b to sum 1
        g = b / sum_b
        # mix with a
        a_new = (1 - t) * a + t * g
        # enforce symmetry and simplex
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = project_to_simplex(a_new)
        # compute new objective
        new_score = compute_score(a_new)
        cur_score = compute_score(a)

            best_score = new_score

        else:

        # optional early exit

"""
Hybrid search for a non‑negative step‑function minimizing the auto‑convolution peak.

The algorithm proceeds in stages:
  1) Coarse scan of several analytic families (product‑power, power‑law,
     exponential‑U, beta, uniform+U mix) on a chosen length n.
  2) An LP‑based refinement step (SciPy HiGHS) that implements the classic
     “maximize sum(g) subject to f∗g ≤ M” update described in the state‑of‑the‑art.
  3) Smooth‑max Adam optimisation of the true objective.
  4) Projected sub‑gradient descent on the exact max‑objective.
  5) A final simulated‑annealing hill‑climb for polishing.

All vectors are kept on the probability simplex (∑a = 1) and are symmetrised
after each major step.  The routine respects a 1000 s wall‑clock budget and
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Optional SciPy import for the linear‑programming step
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)^2)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    maxc = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * maxc / (sum_a * sum_a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the coarse scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric product family: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the grid: (i+½)^{p‑1}(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w

def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ~5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 10.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.015,
                         sym_interval: int = 2500) -> Tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of max(conv)."""
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal smooth‑max temperature
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived from smooth‑max surrogate)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective evaluation
        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score

# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.045) -> Tuple[np.ndarray, float]:
    """Classical sub‑gradient descent for the exact max‑objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> Tuple[np.ndarray, float]:
    """Long random walk on the simplex with a tiny temperature schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def lp_refine(start_a: np.ndarray,
              deadline: float,
              max_outer_iters: int = 6,
              mix_factor: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Implements the classic LP update:
        maximize Σ b   subject to   b ≥ 0,   f ∗ b ≤ M
    where M = max(f ∗ f).  The solution b is mixed with f
    (controlled by ``mix_factor``) and the process repeats.
    """
    if not SCIPY_AVAILABLE:
        print(f"[{time.time():.1f}] SciPy not available – skipping LP refinement.")
        return start_a, compute_score(start_a)

    a = start_a.copy()
    a = project_to_simplex(a)          # ensure sum = 1
    n = a.shape[0]
    rows = 2 * n - 1

    best_score = compute_score(a)
    best_a = a.copy()

    for outer in range(max_outer_iters):
        if time.time() >= deadline:
            break

        # current peak
        conv_ff = convolution_fft(a, a)
        M = conv_ff.max()

        # build convolution matrix A (rows x n) : (A @ b)[k] = (a ∗ b)[k]
        # A has a Toeplitz‐like structure: column j contributes a shifted copy of a.
        A = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        # linear program: maximize Σ b  <=> minimize -Σ b
        c = -np.ones(n, dtype=np.float64)
        ub = M * np.ones(rows, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # solve using HiGHS (fast for modest sizes)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP failed (iter {outer}): {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b < 1e-12:
            print(f"[{time.time():.1f}] LP returned near‑zero mass – stop.")
            break

        # Normalise the LP solution to a probability vector
        g = b / sum_b

        # Mix and re‑project onto the simplex
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        a_new = (a_new + a_new[::-1]) * 0.5            # enforce symmetry
        a_new = project_to_simplex(a_new)

        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] LP iteration {outer} improved score to {best_score:.6f}")
            a = a_new
        else:
            # No improvement – maybe reduce mixing strength
            mix_factor = max(mix_factor * 0.6, 0.05)
            print(f"[{time.time():.1f}] LP iteration {outer} no improvement, mix_factor→{mix_factor:.4f}")

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence minimising ``evaluate_sequence``.
    Respects the wall‑clock budget and returns the best sequence as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. Optional fallback on a previously discovered construction
    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose a working length (power‑of‑two for FFT efficiency)
    # -----------------------------------------------------------
    n = 2048
    # If we have a fallback with a different length, adopt it
    if best_seq is not None and len(best_seq) != n:
        n = len(best_seq)

    # -----------------------------------------------------------
    # 2. Coarse scan of analytic families
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] starting coarse scan (n={n})")
    # product‑power family
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")

    # exponential U shape
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")

    # beta family
    for p in np.linspace(0.1, 3.0, 13):
        if time.time() >= deadline - 30.0:
            break
        for q in np.linspace(0.1, 3.0, 13):
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (beta p={p:.2f},q={q:.2f}) = {best_score:.6f}")

    # mixture uniform + U shape
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Linear‑programming refinement (alternating LP)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting LP‑based refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_refined, sc_refined = lp_refine(a_start, deadline - 120.0,
                                          max_outer_iters=6, mix_factor=0.25)
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after LP refinement best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. First Adam (smooth‑max) optimisation
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] first Adam optimisation")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 150.0,
            max_iters=600_000,
            beta_start=15.0,
            beta_end=6000.0,
            lr=0.012,
            sym_interval=2500,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Projected sub‑gradient descent (exact max)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 50.0,
            max_iters=2_000_000,
            step0=0.04,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Second Adam pass (higher β, finer tuning)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=300_000,
            beta_start=60.0,
            beta_end=9000.0,
            lr=0.010,
            sym_interval=1500,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Explicit symmetrisation test
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Final hill‑climb (simulated annealing)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=5_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 9. Fallback to a safe uniform vector if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] fallback to uniform, score = {best_score:.6f}")

    # final evaluation through the official wrapper (includes all safety checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
