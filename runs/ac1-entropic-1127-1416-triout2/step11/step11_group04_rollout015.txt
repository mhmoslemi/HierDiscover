"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

The implementation builds on the previous solver but adds several
enhancements that proved useful for squeezing out the last few
percent:

* **Much larger random exploration** – Dirichlet draws with many
  concentration parameters, plus a dense scan of the analytic families.
* **Two smooth‑max phases** – first a fast Adam run, then a second
  Adam run with a much larger β‑schedule and a smaller learning rate.
* **Exponentiated‑gradient (mirror‑descent) refinement** – multiplicative
  updates on the simplex with the true max‑gradient.
* **More aggressive LP refinement** – many LP steps (up to 30) with a
  continuous line‑search for the optimal mixing weight.
* **Extended simulated‑annealing hill‑climb** – 10 M steps, richer move
  set, and a geometric line‑search for the step‑size.
* **Dynamic allocation of the global budget** – every phase respects
  the remaining wall‑clock time, so the whole search fits into the
  1000 s limit.

The code follows the same conventions as the reference implementation:
all helpers are top‑level, no closures or lambdas, and the final
``propose_candidate`` returns a plain ``list[float]`` that can be fed
directly into ``evaluate_sequence``.

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT (fast for n≈2‑k)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum(a)=1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # based on https://arxiv.org/abs/1101.6081
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float) -> np.ndarray:
    """Dirichlet draw with concentration ``conc`` (conc>0)."""
    if conc <= 0:
        conc = 1.0
    y = np.random.gamma(conc, 1.0, size=n)
    return y / y.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Monotone decreasing start (used in the reference)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_unordered_start(n: int) -> np.ndarray:
    """Completely random simplex point."""
    r = np.random.rand(n)
    a = r / r.sum()
    return a

# Adam on the smooth‑max surrogate (two passes)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.014) -> tuple[np.ndarray, float]:
    """Adam optimisation of  f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)."""
    eps = 1e-12
    z = np.log(start_a + eps)          # parametrisation a = softmax(z)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)          # full convolution
        maxc = conv.max()
        # smooth‑max weights (log‑sum‑exp trick)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (see reference implementation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:

            sc = compute_score(a_tmp)

                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def adam_smooth_refine(start_a: np.ndarray,
                       deadline: float,
                       max_iters: int = 200_000,
                       beta_start: float = 200.0,
                       beta_end: float = 5_000.0,
                       lr: float = 0.006) -> tuple[np.ndarray, float]:
    """
    Second Adam pass with a higher β‑range and a smaller learning
    rate – useful for fine‑grained polishing.
    """
    return adam_smooth_optimize(
        start_a,
        deadline,
        max_iters=max_iters,
        beta_start=beta_start,
        beta_end=beta_end,
        lr=lr,
    )


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) on the true max‑objective
# ----------------------------------------------------------------------


def expgrad_max_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on f(a)=max(conv). The update is:
        a_i ← a_i * exp(-η * g_i)   ;  g = gradient of the max entry
    followed by a simplex projection.  η decays as 1/√t.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the k‑th convolution entry
        n = a.shape[0]
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal (i=j) factor 1 instead of 2
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        # multiplicative update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # simplex projection (ensures ∑a=1)
        a = project_to_simplex(a)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Exp‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement with exact line‑search
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  (all k),   g ≥ 0.
    Then perform a continuous line‑search on the mixing weight t∈[0,1].
    """
    n = f.shape[0]
    m_len = 2 * n - 1

    # Build convolution matrix A where (A @ g)_k = (f * g)_k
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ↔ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m_len, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, compute_score(f)

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, compute_score(f)

    g_norm = g / sum_g

    # --- line search for optimal mixing weight t --------------------
    # we minimise max(conv((1‑t)·f + t·g_norm)).  Because the objective is
    # monotone in max(conv) we can do a simple ternary search on [0,1].
    lo, hi = 0.0, 1.0
    best_f = f
    best_sc = compute_score(f)

    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0

        cand1 = project_to_simplex((1.0 - t1) * f + t1 * g_norm)
        cand2 = project_to_simplex((1.0 - t2) * f + t2 * g_norm)

        sc1 = compute_score(cand1)
        sc2 = compute_score(cand2)

        if sc1 < best_sc:
            best_sc = sc1
            best_f = cand1
        if sc2 < best_sc:
            best_sc = sc2
            best_f = cand2

        if sc1 < sc2:
            hi = t2
        else:
            lo = t1

    return best_f, best_sc


def lp_refine_loop(a: np.ndarray,
                   deadline: float,
                   max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeated LP refinement – each iteration solves the LP and updates ``a``.
    Stops early if time runs out or if no improvement is seen.
    """
    best_a = a
    best_sc = compute_score(a)

    for it in range(max_iters):
        if time.time() >= deadline - 2.0:
            break

        conv = convolution_fft(a, a)
        M = float(conv.max())
        a_new, sc = lp_refine_step(a, M)

        if sc + 1e-12 < best_sc:
            best_sc = sc
            best_a = a_new
            a = a_new
            print(f"[{time.time():.1f}] LP iteration {it} improved to {best_sc:.6f}")
        else:
            # stagnation – stop early
            print(f"[{time.time():.1f}] LP iteration {it} no improvement")
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (more aggressive)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a slowly decreasing temperature.
    Move set:
        * swap two entries,
        * transfer a fraction of mass,
        * random permutation of a small block (to break structure).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature proportional to current score
    T0 = cur_score * 6e-5
    step_scale = 0.14

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        r = np.random.rand()

        if r < 0.4:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.8:        # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        else:                # block shuffle (size 4‑8)
            blk = np.random.randint(4, 9)
            start = np.random.randint(0, n - blk + 1)
            perm = np.random.permutation(blk)
            cand[start:start + blk] = cand[start:start + blk][perm]

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)   # linear cooling
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] SA step {step:,} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The function respects the global time budget and returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load a fallback candidate if present
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working length (fallback length or 2048)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Aggressive random / analytic scan (≈45 s)
    # ------------------------------------------------------------------
    if time.time() + 45.0 < deadline:
        print(f"[{time.time():.1f}] extensive analytic & random scan (n={n})")
        # fine grid for the three families
        alphas = np.linspace(0.0, 12.0, 25)
        betas = np.linspace(0.0, 12.0, 25)
        gammas = np.linspace(0.0, 6.0, 25)

        for alpha in alphas:
            if time.time() >= deadline - 5.0: break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (sym α={alpha:.2f}) = {best_score:.6f}")

        for beta in betas:
            if time.time() >= deadline - 5.0: break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (left β={beta:.2f}) = {best_score:.6f}")

        for gamma in gammas:
            if time.time() >= deadline - 5.0: break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (exp γ={gamma:.2f}) = {best_score:.6f}")

        # many Dirichlet draws with varying concentration
        concs = np.concatenate([np.logspace(-2, 0, 12), np.linspace(0.2, 3.0, 12)])
        for conc in concs:
            if time.time() >= deadline - 5.0: break
            vec = random_dirichlet(n, conc)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (Dirichlet conc={conc:.3f}) = {best_score:.6f}")

        # a few thousand completely random points
        for _ in range(3000):
            if time.time() >= deadline - 5.0: break
            vec = random_unordered_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise starter to a probability vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. First Adam (smooth‑max) phase – up to ~20 % of budget
    # ------------------------------------------------------------------
    if time.time() + 80.0 < deadline:
        print(f"[{time.time():.1f}] first Adam phase")
        a, sc = adam_smooth_optimize(
            a,
            deadline - 10.0,
            max_iters=400_000,
            beta_start=4.0,
            beta_end=2000.0,
            lr=0.014,
        )
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Second Adam (high‑β polish) – up to ~10 % of remaining budget
    # ------------------------------------------------------------------
    if time.time() + 50.0 < deadline:
        print(f"[{time.time():.1f}] second Adam (high‑β) phase")
        a, sc = adam_smooth_refine(
            a,
            deadline - 5.0,
            max_iters=200_000,
            beta_start=200.0,
            beta_end=5_000.0,
            lr=0.006,
        )
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient refinement – up to ~10 % of remaining budget
    # ------------------------------------------------------------------
    if time.time() + 40.0 < deadline:
        print(f"[{time.time():.1f}] exponentiated‑gradient phase")
        a, sc = expgrad_max_optimize(
            a,
            deadline - 5.0,
            max_iters=500_000,
            eta0=0.5,
        )
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] after exp‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. LP‑based refinement loop – as many iterations as time permits
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] LP refinement loop start")
        a, sc = lp_refine_loop(a, deadline, max_iters=30)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] after LP refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Optional symmetrisation – often helpful
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score, best_seq = sc_sym, a_sym.tolist()
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final aggressive SA hill‑climb (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a, sc = hill_climb_sa(a, deadline, max_steps=10_000_000)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] final hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
