"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

The algorithm departs from the previous symmetric‑only approach:
1. **Unsymmetric L‑BFGS** – we optimise the smooth‑max surrogate of
   ``max(conv(a,a))`` directly on the soft‑max parametrisation
   (ensuring non‑negativity and ∑a=1) *without* forcing symmetry.
2. **Multiple diverse seeds** – left‑heavy, right‑heavy, product,
   exponential‑U families and a few random Dirichlet vectors are
   launched in parallel; each seed runs a short schedule of increasing
   smoothing parameters β.
3. **Final simulated‑annealing hill‑climb** – a lightweight SA walk
   refines the best solution found by the gradient phases.
4. **Time‑aware** – the routine constantly checks the 1000 s wall‑clock
   budget and stops early if necessary.

All helper routines are top‑level (no closures) and only ``numpy``,
``scipy`` and the provided ``evaluate_sequence`` are used.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT.  Returns a vector of length
    ``len(a)+len(b)-1``.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        # degenerate case – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
    ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (cheap seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Smooth approximation of ``max(conv(a,a))`` and its gradient w.r.t.
    the soft‑max parameters ``z``.
    Returns ``(f, grad_z)`` where ``f`` is the surrogate value.
    """
    a = softmax(z)                     # a ≥ 0, ∑a = 1
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()

    # soft‑max weights over the convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta   # smooth max
    w = expw / sumexp                  # normalised weights, sum = 1

    # weighted gradient of the convolution w.r.t. a
    # g_a = 2 * conv_rev[n-1 : n-1+n] where conv_rev = a_rev * w
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # shape (n,)

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimiser (unsymmetric)
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Run a cascade of L‑BFGS optimisations on the smooth‑max surrogate.
    No symmetry is imposed – only the simplex constraint is respected.
    Returns the best vector (on the simplex) and its exact score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS β={beta}")

        z0 = np.log(a + 1e-14)   # initialise soft‑max parameters

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        try:
            res = minimize(fun,
                           z0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter_per_stage,
                                    'disp': False})
        except Exception as e:
            print(f"  L‑BFGS failed (β={beta}): {e}")
            continue

        a = softmax(res.x)               # stays on the simplex automatically
        a = project_to_simplex(a)        # tiny numerical drift correction
        sc = compute_score(a)

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    A lightweight SA walk that respects the simplex.
    Moves are either a swap or a small mass transfer.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # Initial temperature – a few percent of the current objective
    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                 # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # Metropolis acceptance
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # gradually enlarge step size when we improve
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            # slowly shrink step size on rejection
            step_scale *= 0.97

    return best_a, best_score

        a_opt, sc_opt = l_bfgs_optimize(a0,
                                        betas=betas_schedule,
                                        max_iter_per_stage=max_iter_stage,
                                        deadline=deadline - 10.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new global best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. final simulated‑annealing refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. safety fallback (should never be needed)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return best_seq
