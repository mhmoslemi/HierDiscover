```python
def exponentiated_gradient_opt(a0, deadline, eta0=0.7):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-10
        active = np.where(conv >= max_val - eps)[0]

        # compute gradient
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1)
            j = k - i
            g[i] += 2.0 * a[j]

        eta = eta0 / np.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * g)
        # avoid underflow
        total = a.sum()
        if total <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # occasional jitter to escape stagnation
        if t % 500 == 0:
            # apply small perturbation
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(t))
            a = a * np.exp(jitter)
            a = a / a.sum()
    return best_a
```

"""
non‑negative step‑function optimiser for ``evaluate_sequence``

Strategy
--------
1.  Global exploration – dense sweeps of three parametric families
    (power‑law, exponential, beta) and a large pool of random
    Dirichlet draws.  The top‐k candidates (by the exact objective
    2·n·max(conv(a,a))) are retained.

2.  Exponentiated‑gradient (EG) polishing – the objective is
    f(a)=max_k (a*a)_k.  The sub‑gradient at the active lag(s) k is
    g_i = 2·∑_{k active} a_{k-i}.  An EG update
        a_i ← a_i·exp(‑η·g_i)  (followed by renormalisation to the simplex)
    is performed with a decreasing step‑size η.  The EG scheme respects
    the simplex constraints and empirically descends much faster than
    plain sub‑gradient descent.

3.  Optional simulated‑annealing hill‑climb – a few short Metropolis
    runs are applied to the best vector to escape possible local minima.

The whole search respects the 1000 s wall‑clock budget, allocating roughly
120 s to the global sweep, 700 s to EG polishing (including many random
re‑starts), and the remainder to a final annealing pass.  The best
sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape:
       a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    """
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global exploration (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 8):
    """Sweep families and many random Dirichlet samples.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []   # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0:
            return
        vec = vec / s                     # project onto the simplex
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep – dense grid
    for p in np.linspace(0.1, 2.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep – dense grid
    for lam in np.linspace(0.0001, 0.05, 300):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep – medium resolution
    al_vals = np.linspace(0.2, 5.0, 25)
    be_vals = np.linspace(0.2, 5.0, 25)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) massive random Dirichlet draws with several concentration values
    concs = [0.2, 0.5, 1.0, 2.0, 4.0]
    draws_per_conc = 1200   # total ≈ 6 000 draws (fast on modern hardware)
    for conc in concs:
        alpha_vec = np.full(n, conc)
        for _ in range(draws_per_conc):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(alpha_vec)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (sub‑gradient descent on max‑conv)
# ----------------------------------------------------------------------


def eg_polish(a0: np.ndarray, deadline: float,
              eta0: float = 0.7, tol: float = 1e-10) -> np.ndarray:
    """
    Exponentiated‑gradient minimisation of  max_k (a*a)_k  on the simplex.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active set (indices attaining the maximum up to a tiny relative tolerance)
        eps = max_val * tol
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: g_i = 2 * sum_{k in active} a_{k-i}
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # decreasing step size
        eta = eta0 / np.sqrt(it)

        # EG update (multiplicative)
        a = a * np.exp(-eta * g)

        # renormalise to the simplex (protect against exact zero)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional random jitter to escape plateaux
        if it % 500 == 0:
            jitter = np.random.randn(n) * (eta0 * 0.001 / np.sqrt(it))
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (optional final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis‑style hill‑climb that may accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep on simplex
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 4000 == 0:
            # global jitter
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a = a * np.exp(jitter)
            a = a / a.sum()
            cur_val = compute_score(a)

    return best_a
