def smoothed_mirror_optimize(start_a, time_deadline, max_iters=..., eta0=0.5, gamma_start=2.0, gamma_end=200.0):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    L = 2*n - 1

    for t in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        # compute convolution
        conv = convolution_fft(a, a)
        # compute smoothing parameter gamma gradually increase
        gamma = gamma_start + (gamma_end - gamma_start) * min(1.0, t / max_iters)
        # compute weights w via softmax
        # subtract max for stability
        maxc = np.max(conv)
        exp_vals = np.exp(gamma * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        w_rev = w[::-1]
        grad_conv = convolution_fft(a, w_rev)  # full conv length n+L-1 = 3n-2
        # extract slice for gradient: indices (L-1)-np.arange(n)
        # L-1 = 2n-2
        idx_start = (2*n - 2) - (n-1) # = n-1
        # Actually we need to sample conv at positions (2n-2) - j for j=0..n-1.
        # So we can take grad = grad_conv[(2*n-2) - np.arange(n)]
        grad = 2.0 * grad_conv[(2*n-2) - np.arange(n)]

        # Compute average dot product a ⋅ grad
        avg = np.dot(a, grad)

        # step size schedule
        eta = eta0 / np.sqrt(t)

        # exponentiated update
        exp_arg = -eta * (grad - avg)
        exp_arg = np.clip(exp_arg, -50, 50)
        a = a * np.exp(exp_arg)
        # project onto simplex
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Strategy
--------
1. Build a rich library of analytic seeds (uniform, binomial, power‑law,
   geometric, Beta, sinusoidal, cosine, logistic, triangular) and also
   a set of random Dirichlet vectors.  Keep the best one seen.

2. Refine the best seed with a *smoothed* KL‑mirror descent.  Instead of
   using only the active peak of the convolution we employ a soft‑max
   (log‑sum‑exp) approximation of the max – this yields a smoother
   sub‑gradient and helps to escape shallow local minima.  The smoothing
   parameter γ is increased linearly during optimisation.

3. Perform a short stochastic hill‑climb (swap / fractional mass transfer)
   to polish the final result.

All phases respect a wall‑clock deadline (default 1000 s) and keep the
best sequence encountered.  The function returns a plain list of floats
ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Symmetric and heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    return project_to_simplex(a)


def generate_cosine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ cos(π·i/(2·(n‑1)))^p, normalised (decreasing shape)."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / (2.0 * (n - 1))) ** p
    return project_to_simplex(a)


def generate_logistic(n: int, k: float) -> np.ndarray:
    """Logistic CDF shape centred at the middle, steeper with larger k."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(k * (i - mid)))
    return project_to_simplex(a)


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed KL‑mirror optimiser
# ----------------------------------------------------------------------


def smoothed_mirror_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.5,
    gamma_start: float = 5.0,
    gamma_end: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on the smooth approximation
        f_γ(a) = (1/γ)·log∑_k exp(γ·conv_k(a)),
    which converges to the original max‑convolution as γ→∞.
    The update is exponentiated‑gradient with a diminishing stepsize.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    L = 2 * n - 1                     # length of a*a convolution

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # 1) current convolution
        conv = convolution_fft(a, a)
        maxc = float(conv.max())

        # 2) smoothing parameter (linear schedule)
        ratio = t / max_iters
        gamma = gamma_start + (gamma_end - gamma_start) * ratio

        # 3) soft‑max weights (stable)
        exp_vals = np.exp(gamma * (conv - maxc))
        w = exp_vals / exp_vals.sum()               # length 2n‑1

        # 4) gradient of f_γ:  g = 2·a * w (cross‑correlation)
        w_rev = w[::-1]
        grad_full = convolution_fft(a, w_rev)       # length 3n‑2
        idx = (L - 1) - np.arange(n)                # map to cross‑correlation entries
        grad = 2.0 * grad_full[idx]

        # 5) mirror‑descent (KL) update
        avg = np.dot(a, grad)
        eta = eta0 / np.sqrt(t)
        exp_arg = -eta * (grad - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # 6) project back onto the probability simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional progress check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final verification
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score

def hill_climb_permutation(seq, time_deadline, max_steps: int = 300_000) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
