def mirror_smooth_optimize(start_a, time_deadline, max_iters=500000, eta0=0.8, alpha=20.0):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline: break
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        # compute weight vector w of length 2n-1
        # Use softmax with temperature parameter alpha
        # Compute diff = conv - max_conv (all <=0) -> large negative for non-max
        diff = conv - max_conv
        # avoid overflow: exponent = alpha * diff (<=0)
        w = np.exp(alpha * diff)
        w_sum = w.sum()
        w = w / w_sum  # normalized
        # compute gradient g = 2 * sum_k w_k * a shifted
        # using convolution with reversed w
        rev_w = w[::-1]
        conv_g = convolution_fft(a, rev_w)  # length 3n-2
        # g_i = 2 * conv_g[2*n-2 - i]
        g = 2.0 * conv_g[2*n-2 - np.arange(n)]

        avg = np.dot(a, g)

        eta = eta0 / np.sqrt(t)
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

"""
Optimiser for the “step‑function” attack.

The algorithm consists of three stages:

1.  Build a diverse pool of analytic seeds (uniform, central‑binomial,
    power‑law, geometric, symmetric Beta, tent, sine, random Dirichlet).

2.  Refine the best seed with a *multi‑peak* KL‑mirror descent.
    Instead of using the sub‑gradient of only the maximal convolution
    entry we form a convex combination of the sub‑gradients belonging to
    the K largest coefficients (K≈5–7).  This gives a smoother descent
    direction and helps escape plateaus that trap the single‑peak
    optimiser used previously.

3.  A cheap stochastic hill‑climb (swap / fractional mass move) finally
    polishes the result.

All updates keep the vector on the probability simplex (sum = 1,
non‑negative).  The code respects a wall‑clock deadline (default 1000 s)
and always returns the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i (decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution (1‑r)·r^i."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (alpha - 1.0))
    return project_to_simplex(a)


def generate_tent(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    return project_to_simplex(a)


def generate_sine_shape(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n) – zero at the ends, peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    return project_to_simplex(a)


def generate_random_dirichlet(n: int) -> np.ndarray:
    """Random point on the simplex via exponential sampling."""
    a = np.random.rand(n).astype(np.float64) + 1e-12
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Multi‑peak KL‑mirror descent optimiser
# ----------------------------------------------------------------------


def mirror_topk_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
    K: int = 7,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent using the *average* sub‑gradient of the K
    largest convolution coefficients.  The vector stays on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    idx = np.arange(n)  # reused index vector

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # full convolution and the K biggest entries
        conv = convolution_fft(a, a)
        # choose K distinct indices with largest values
        if K >= conv.size:
            top_idxs = np.arange(conv.size)
        else:
            top_idxs = np.argpartition(conv, -K)[-K:]

        # average sub‑gradient over the chosen peaks
        g = np.zeros_like(a)
        weight = 1.0 / len(top_idxs)
        for k in top_idxs:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += weight * 2.0 * a[j[mask]]

        # KL‑mirror step
        avg = np.dot(a, g)                     # scalar = ⟨a,g⟩
        eta = eta0 / np.sqrt(t)                # diminishing stepsize
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)  # safety
        a = a * np.exp(exp_arg)

        # re‑normalise onto the simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # occasional progress check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer)
    with simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
