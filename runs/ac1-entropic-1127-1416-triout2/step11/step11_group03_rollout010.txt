"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2).

The algorithm works in three stages:

1. **Coarse analytic scan** – try a few classic symmetric families
   (product‑power, left‑heavy power, exponential‑U) for several lengths.
   The best parameters are kept.

2. **Local refinement** – from the best candidate run a two‑phase
   optimiser:
   * Adam on a smooth log‑sum‑exp surrogate of max(conv) (the smoothing
     parameter β is annealed from low to high values).
   * Projected sub‑gradient descent on the exact max‑objective.

3. **Final polishing** – a lightweight simulated‑annealing hill‑climb.

All vectors are kept on the probability simplex (∑a_i = 1) and are
symmetrised after each major block.  The routine respects a 1000 s
wall‑clock budget and returns a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence, but without safety checks.
    a may be any non‑negative vector; the function returns
        2·n·max(a*a) / (sum(a)^2)
    where * denotes linear convolution.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max()) / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all produce vectors that already sum to 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    """
    i = np.arange(n, dtype=np.float64)
    # log of the product to avoid overflow/underflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow for extreme α
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 3000.0,
                         lr: float = 0.018,
                         sym_interval: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise log‑probabilities (soft‑max will enforce simplex)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule in log‑space for the smooth‑max temperature β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # occasional symmetry enforcement
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived from the smooth‑max surrogate)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # cheap true‑objective evaluation every few thousand steps
        if it % 4000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.07) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 4_000_000) -> Tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing hill‑climb on the simplex.
    Moves: swap two entries (50 %) or transfer a fraction of mass (50 %).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature proportional to current score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

_global_n_for_gs = 256  # dummy initial value


def _gs_score(alpha: float) -> float:
    """Score for the symmetric product family at the globally‑set n."""
    vec = symmetric_product_family(_global_n_for_gs, alpha)
    return compute_score(vec)


def golden_section_optimize(a: float, b: float,
                            max_iter: int = 14) -> Tuple[float, float]:
    """
    Minimises the global function _gs_score on [a,b] using golden‑section.
    Returns (best_alpha, best_score).
    """
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = _gs_score(c)
    fd = _gs_score(d)
    for _ in range(max_iter):
        if fc < fd:          # minimising
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = _gs_score(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = _gs_score(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

        # ----------------------------------------------------------------
        # 6. Projected sub‑gradient descent (exact max)
        # ----------------------------------------------------------------
        if best_seq and time.time() < deadline - 80.0:
            print(f"[{time.time():.1f}] sub‑gradient descent (n={n})")
            a_sub, sc_sub = subgrad_optimize(
                np.array(best_seq, dtype=np.float64),
                deadline - 40.0,
                max_iters=2_500_000,
                step0=0.07,
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.tolist()
                print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

        # ----------------------------------------------------------------
        # 7. Final simulated‑annealing polish (short)
        # ----------------------------------------------------------------
        if best_seq and time.time() < deadline - 5.0:
            print(f"[{time.time():.1f}] final SA polish (n={n})")
            a_final, sc_final = hill_climb_sa(
                np.array(best_seq, dtype=np.float64),
                deadline,
                max_steps=2_500_000,
            )
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.tolist()
                print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

        # ----------------------------------------------------------------
        # 8. Symmetrisation test (cheap)
        # ----------------------------------------------------------------
        if best_seq is not None:
            a_arr = np.array(best_seq, dtype=np.float64)
            a_sym = (a_arr + a_arr[::-1]) * 0.5
            a_sym = project_to_simplex(a_sym)
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = a_sym.tolist()
                print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

        # End of this length – move to next if time permits
        print(f"[{time.time():.1f}] finished length n={n}, best so far = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: if we never found a candidate, return a tiny uniform vector
    # ------------------------------------------------------------------
    if not best_seq:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] fallback to uniform length {n_small}")

    # Final official evaluation (includes all safety checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
