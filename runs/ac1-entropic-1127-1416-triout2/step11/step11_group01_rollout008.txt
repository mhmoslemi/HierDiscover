"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

using a multi‑phase search:
  1. quick scan of several analytic families (including new linear, logistic,
     Gaussian and sparse supports);
  2. many random starts;
  3. exponentiated‑gradient (mirror‑descent) on the true max‑convolution;
  4. Adam optimisation of a smooth log‑sum‑exp surrogate with a very
     aggressive β‑schedule;
  5. LP‑based refinement with an *optimal* mixing weight obtained by a
     cheap 1‑D line‑search;
  6. optional symmetrisation and a final simulated‑annealing hill‑climb.
All phases respect the global wall‑clock budget (default 1000 s) and
the best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def linear_decreasing_family(n: int) -> np.ndarray:
    """Deterministic linear decrease a_i ∝ (n-i)."""
    w = np.arange(n, 0, -1, dtype=np.float64)
    return w / w.sum()


def logistic_family(n: int, slope: float) -> np.ndarray:
    """Monotone decreasing logistic shape."""
    i = np.arange(n, dtype=np.float64)
    shift = (n - 1) / 2.0
    w = 1.0 / (1.0 + np.exp(slope * (i - shift)))
    return w / w.sum()


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian (truncated) shape."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


def random_sparse_start(n: int) -> np.ndarray:
    """Random support of size k (10‑%–30‑% of n) with Dirichlet weights."""
    k = np.random.randint(int(0.1 * n), int(0.3 * n) + 1)
    idx = np.random.choice(n, size=k, replace=False)
    vals = np.random.rand(k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimisation on the true max
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the exact max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[EG] iter {t} improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Smoothed‑max optimisation (Adam on the soft‑max parametrisation)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """Minimise the smooth surrogate (log‑sum‑exp) with Adam."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable log‑sum‑exp
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (see original code)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[Adam] iter {it} improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement with optimal mixing weight
# ----------------------------------------------------------------------


def best_mix_weight(f: np.ndarray,
                    g: np.ndarray,
                    conv_f: np.ndarray,
                    conv_fg: np.ndarray,
                    conv_g: np.ndarray) -> float:
    """
    Find t∈[0,1] minimising max_k ((1−t)² conv_f + 2t(1−t) conv_fg + t² conv_g).
    Simple grid + local refinement.
    """
    best_t = 0.0
    best_M = conv_f.max()

    # coarse grid
    ts = np.linspace(0.0, 1.0, 21)
    for t in ts:
        combo = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_g
        M = combo.max()
        if M < best_M:
            best_M = M
            best_t = t

    # local refinement around best_t
    lo = max(0.0, best_t - 0.05)
    hi = min(1.0, best_t + 0.05)
    ts2 = np.linspace(lo, hi, 21)
    for t in ts2:
        combo = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_g
        M = combo.max()
        if M < best_M:
            best_M = M
            best_t = t

    return best_t


def lp_refine_step_optimal(f: np.ndarray,
                           M: float,
                           mix_weight: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then mix with the optimal coefficient t (found by a cheap line‑search)
    and return the new f together with its max‑convolution value.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build the convolution matrix A (size m×n)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)      # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # Solve the LP (short time limit)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 10.0})

    if not res.success:
        # fallback to simple convex combination
        return f, M

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, M

    g_norm = g / sum_g

    # Pre‑compute required convolutions
    conv_f = convolution_fft(f, f)
    conv_fg = convolution_fft(f, g_norm)
    conv_g = convolution_fft(g_norm, g_norm)

    # Optimal mixing coefficient
    t_opt = best_mix_weight(f, g_norm, conv_f, conv_fg, conv_g)

    # Form the new candidate
    f_new = (1.0 - t_opt) * f + t_opt * g_norm
    f_new = project_to_simplex(f_new)

    # New max‑convolution value
    M_new = convolution_fft(f_new, f_new).max()
    return f_new, float(M_new)


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two coordinates
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  Returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin for harness

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[fallback] score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Determine problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = fallback_len if fallback_len >= 1024 else 2048
    print(f"[info] using length n={n}")

    # ------------------------------------------------------------------
    # 2. Quick analytic / random scan
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print("[phase] analytic families scan")
        # sweep parameters
        alphas = np.linspace(0.0, 30.0, 31)
        betas = np.linspace(0.0, 30.0, 31)
        gammas = np.linspace(0.0, 15.0, 31)
        slopes = [0.02, 0.05, 0.1, 0.2, 0.5]
        sigmas = [n / 12.0, n / 8.0, n / 5.0, n / 3.0]

        # symmetric product family
        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[analytic] sym‑α={alpha:.2f} improved to {best_score:.6f}")

        # left‑heavy power family
        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[analytic] left‑β={beta:.2f} improved to {best_score:.6f}")

        # exponential U‑shape family
        for gamma in gammas:
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[analytic] exp‑γ={gamma:.2f} improved to {best_score:.6f}")

        # linear decreasing (no param)
        vec = linear_decreasing_family(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[analytic] linear decreasing improved to {best_score:.6f}")

        # logistic family
        for slope in slopes:
            if time.time() >= deadline - 30.0:
                break
            vec = logistic_family(n, slope)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[analytic] logistic slope={slope:.3f} improved to {best_score:.6f}")

        # Gaussian family
        for sigma in sigmas:
            if time.time() >= deadline - 30.0:
                break
            vec = gaussian_family(n, sigma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[analytic] gaussian sigma={sigma:.1f} improved to {best_score:.6f}")

        # mixed uniform / U‑shape
        for _ in range(30):
            if time.time() >= deadline - 30.0:
                break
            lam = np.random.rand()
            gamma = np.random.uniform(0.0, 15.0)
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[analytic] mixed improved to {best_score:.6f}")

        # random draws
        for _ in range(300):
            if time.time() >= deadline - 30.0:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[analytic] random start improved to {best_score:.6f}")

        # random sparse starts
        for _ in range(100):
            if time.time() >= deadline - 30.0:
                break
            vec = random_sparse_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[analytic] sparse start improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise the working vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Exponentiated‑gradient (mirror descent) phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print("[phase] EG mirror‑descent")
        a_opt, sc_opt = eg_optimize(
            a,
            deadline - 180.0,
            max_iters=350_000,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[EG] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam‑based smooth‑max refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 180.0:
        print("[phase] Adam smooth‑max")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline - 120.0,
            max_iters=250_000,
            beta_start=4.0,
            beta_end=2000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[Adam] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based refinement with optimal mixing
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("[phase] LP refinement (optimal mix)")
        mix_weight = 0.2
        lp_iter = 0
        while time.time() < deadline - 30.0 and lp_iter < 30:
            f = a
            conv = convolution_fft(f, f)
            M = float(conv.max())
            f_new, M_new = lp_refine_step_optimal(f, M, mix_weight=mix_weight)
            sc = 2.0 * n * M_new  # same as compute_score for sum=1
            if sc < best_score - 1e-7:
                best_score = sc
                a = f_new
                print(f"[LP] iter {lp_iter} improved to {best_score:.6f}")
            else:
                # gradually reduce mixing weight if no improvement
                mix_weight = max(0.05, mix_weight * 0.85)
            lp_iter += 1

    # ------------------------------------------------------------------
    # 6. Symmetrisation test
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[sym] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final stochastic hill‑climb (SA)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[phase] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=1_800_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[SA] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Return the best sequence
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[finished] final reported score = {final_score:.6f}")
    return final_seq
