def softmax_optimize(start_a, time_deadline, beta_schedule, eta0):
    a = start_a.copy()
    best_seq = a.copy()
    best_score = compute_score(a)
    n = len(a)
    t=0
    while time.time() < time_deadline:
        t += 1
        # compute conv
        conv = convolution_fft(a, a)
        # compute beta (maybe increase over time)
        beta = beta_schedule(t)
        # compute w = exp(beta*conv - logsumexp)
        # stable: w = np.exp(beta*conv - logZ)
        logZ = np.log(np.sum(np.exp(beta*conv)))
        w = np.exp(beta*conv - logZ)
        # gradient
        # correlation: g_raw = 2 * correlation(w, a)
        # correlation via FFT: convolve(w, a_rev) and slice
        a_rev = a[::-1]
        corr = convolution_fft(w, a_rev)[n-1 : n-1+n]  # length n
        g = 2.0 * corr

        # Mirror update: EG relative to avg = dot(a,g)
        avg = np.dot(a, g)
        eta = eta0 / np.sqrt(t)
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50, 50)
        a = a * np.exp(exp_arg)
        # simplex projection
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a / sum_a

        # keep best
        if t % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Iter {t}: best_score={best_score:.6f}")

        # optional: break if epoch too many steps.

    # final compute
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_seq = final_score, a.copy()
    return best_seq, best_score

def lp_refine(a, time_deadline, max_iters=30, initial_t=0.5):
    best_a = a.copy()
    best_score = compute_score(a)
    t = initial_t
    n = len(a)
    for i in range(max_iters):
        if time.time() >= time_deadline:
            break
        # compute current M
        conv = convolution_fft(a, a)
        M = conv.max()
        # build A
        A = build_convolution_matrix(a)
        # solve LP
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs', options={'presolve':True})
        if not res.success:
            print(f"LP solve failed at iter {i}, reason: {res.message}")
            break
        b = res.x
        # normalize to sum=1
        b_sum = b.sum()
        if b_sum == 0:
            # fallback to uniform
            b = np.full(n, 1.0/n)
        else:
            b = b / b_sum
        # convex combination
        a_new = (1 - t) * a + t * b
        # compute new score
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            # optionally increase step size t (more aggressive) if improvement
        else:
            # if no improvement, reduce t
            t = max(t * 0.5, 0.01)
        a = a_new
        # optionally increase t gradually
    return best_a, best_score

"""
Non‑negative step‑function optimiser.

Main ideas
----------
1. Build a diverse set of analytic seeds (uniform, central‑binomial,
   power‑law, geometric, Beta‑shaped).  Keep the best.

2. Refine the best seed with a *linear‑program* (LP) based direction
   finder.  For a given vector a we maximise Σb subject to the
   self‑convolution bound a * b ≤ max(a*a).  The solution pushes mass
   into positions that are “least saturated” and tends to flatten the
   autocorrelation drastically.

3. Continue optimisation with the KL‑mirror (exponentiated‑gradient)
   sub‑gradient descent on the true max‑convolution objective.

4. Polish the result with a cheap stochastic hill‑climb (swap / mass‑move).

All phases respect a wall‑clock deadline (default 1000 s) and keep the
best sequence encountered.  The final sequence is returned as a plain
list of floats (already on the simplex) and can be fed directly into
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Symmetric and heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# LP‑based flattening step
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ x) = f * x (full convolution).
    Row k, column j contains f[k‑j] when that index is in range, otherwise 0.
    """
    n = f.shape[0]
    # each diagonal offset = -i carries the constant value f[i]
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = [-i for i in range(n)]
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def lp_refine(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 15,
    initial_t: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    LP‑based direction finder.
    Replaces a by a convex combination (1‑t)·a + t·g where g is the LP optimum.
    Repeats a few times while time permits.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t = initial_t
    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # current maximal autocorrelation
        conv = convolution_fft(a, a)
        M = conv.max()

        # linear program: maximise Σb   s.t.  A @ b ≤ M, b ≥ 0
        A = build_convolution_matrix(a)
        bounds = [(0.0, None)] * n
        # linprog minimises c·x → we give -1 to maximise Σb
        res = linprog(
            -np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M, dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # if the LP fails (rare) fall back to the current vector
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        b = res.x
        b_sum = b.sum()
        if b_sum == 0.0:
            g = np.full_like(a, 1.0 / n)
        else:
            g = b / b_sum

        # convex update
        a_new = (1.0 - t) * a + t * g
        cur_score = compute_score(a_new)

        # keep best
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
            # if we made progress we can be a little bolder
            t = min(t * 1.1, 0.9)
            print(f"[LP] iter {it:02d}: improved score = {cur_score:.6f}")
        else:
            # otherwise damp the step size
            t = max(t * 0.5, 0.05)

        a = a_new

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑descent optimiser on the true max‑convolution objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).  a is kept on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        avg = np.dot(a, g)                     # average = 2·max(conv)
        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)  # keep numbers sane
        a = a * np.exp(exp_arg)

        # renormalise onto the simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= sum_a

        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very small random moves (swap or fractional mass transfer) with simplex
    projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
