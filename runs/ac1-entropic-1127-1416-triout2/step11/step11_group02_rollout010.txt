    # Softmax weights:
    # To avoid overflow, subtract maxv.

    a = a0.copy()
    n = len(a)
    best_a = a.copy()

        conv = full_convolution(a, a)   # length 2n-1

"""
Search for a non‑negative step‑function (vector of heights) that minimizes
``evaluate_sequence``.  The algorithm combines:

*  Broad parametric sweep (power‑law, geometric, triangular, Gaussian,
   sinusoidal) plus a few Dirichlet random starts.
*  A smooth‐max gradient descent (soft‑max approximation of the max‑convolution)
   with simplex projection.  The soft‑max temperature (α) is increased
   gradually so the optimisation follows the true non‑smooth objective.
*  A cheap stochastic hill‑climb for final polishing.

All phases respect the overall wall‑clock budget (default 1000 s).  The
returned list sums to one, stays inside the [0,1000] range, and therefore
produces a valid score via the official ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # fallback
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using a real FFT.
    Length of the result is len(a) + len(b) - 1.
    """
    n = len(a) + len(b) - 1
    n_fft = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:n]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale invariant) for a normalised vector (Σa = 1)."""
    n = a.shape[0]
    conv = full_convolution(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families for initialisation
# ----------------------------------------------------------------------
def make_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def make_geometric(n: int, r: float) -> np.ndarray:
    a = (r ** np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def make_triangular(n: int) -> np.ndarray:
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def make_sinusoidal(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def make_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) - n / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient descent on the soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def _gradient_weighted_conv(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k * (a * a)[k]   w.r.t. a,
    where w is a probability vector (soft‑max weights) of length 2·n‑1.
    Returns a vector of length n.
    """
    rev_a = a[::-1]
    conv_wr = full_convolution(w, rev_a)        # length = (2n‑1) + n‑1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_wr[n - 1:n - 1 + n]        # slice the appropriate block
    return grad


def smooth_max_gradient_descent(a0: np.ndarray,
                                deadline: float,
                                max_iter: int = 2000,
                                alpha_start: float = 5.0,
                                alpha_end: float = 200.0,
                                eta0: float = 0.5) -> np.ndarray:
    """
    Projected gradient descent on  f(a)=max(a*a) approximated by a
    soft‑max.  The iterate stays on the simplex (Σa = 1, a ≥ 0).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # initialise with the current objective (exact max, not smooth)
    best_val = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linearly increase temperature α
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)

        # full self‑convolution and its maximum
        conv = full_convolution(a, a)
        max_conv = conv.max()

        # soft‑max weights (stable by subtracting the max)
        shifted = conv - max_conv                     # ≤ 0
        w_exp = np.exp(alpha * shifted)
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / w_sum

        # gradient of the smooth approximation
        grad = _gradient_weighted_conv(a, w)

        # optional gradient clipping (avoid huge steps)
        norm = np.linalg.norm(grad)
        if norm > 1.0:
            grad = grad / norm

        # diminishing step size
        step = eta0 / np.sqrt(it)

        # update and project back onto the simplex
        a = a - step * grad
        a = project_to_simplex(a)

        # exact objective (no smoothing) – cheap because we already have conv
        val = float(2.0 * n * max_conv)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (greedy local moves)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 300_000) -> np.ndarray:
    """
    Tiny stochastic moves with greedy acceptance.
    Moves a fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    conv = full_convolution(a, a)
    best_M = conv.max()
    step_scale = 0.05
    no_improve = 0
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce non‑negativity (tiny negative numbers can appear due to fp errors)
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand, cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve > 100_000:
            # escape stagnation by a small random jitter
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
