def smooth_mirror_optimize(start_a, time_deadline, max_iters=1_000_000, eta0=0.6, alpha0=0.1, alpha_factor=1.0005, alpha_max=20.0):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        # compute conv
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        # compute weights
        # exponent scaling
        w = np.exp(alpha0 * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.full_like(w, 1.0 / w.shape[0])
        else:
            w_norm = w / w_sum
        # compute gradient via convolution: weighted sum of g_k. For each k, gradient_k = 2*a_{k-i} (as earlier)
        # Weighted sum gradient = sum_k w_norm[k] * g_k = 2 * sum_k w_norm[k] * shift a.
        # Equivalent to 2 * convolution of a with w_rev weighted appropriately
        w_rev = w_norm[::-1]
        grad_full = convolution_fft(a, w_rev)  # length 3n-2 maybe; we need entries at index (len(w)-1 - i)
        # length of w: 2n-1, so len(w)-1 = 2n-2
        offset = len(w) - 1
        grad = 2.0 * grad_full[offset - np.arange(n)]   # this extracts proper slice.
        # optionally clamp gradient tiny values to avoid numerical issues
        avg = np.dot(a, grad)  # expectation under a
        # step size diminishing
        eta = eta0 / np.sqrt(t)
        exp_arg = -eta * (grad - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)
        # simplex projection
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        # optionally update alpha for annealing
        if alpha0 < alpha_max:
            alpha0 *= alpha_factor
        # best tracking
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

        max_conv = conv.max()

        frac = t / max_iters
        alpha = alpha_start + (alpha_target - alpha_start) * frac
        # compute weights
        w = np.exp(alpha * (conv - max_conv))  # values <=1
        w_sum = w.sum()

            w_norm = np.full_like(w, 1.0 / w.shape[0])
        else:
            w_norm = w / w_sum
        # compute weighted gradient via convolution with reversed w_norm
        w_rev = w_norm[::-1]
        grad_full = convolution_fft(a, w_rev)
        offset = m - 1  # = 2n-2

        if a_sum == 0.0:

"""
Non‑negative step‑function optimizer (target: lower evaluation score).

Main ideas
----------

* Build a rich palette of analytic “seed” vectors (uniform, power‑law,
  geometric, central‑binomial, Beta‑shaped, linear, quadratic,
  Gaussian, sine‑shaped).  Pick the best seed by the exact objective.

* Refine the best seed with a *smooth KL‑mirror* descent:
  – Instead of the raw sub‑gradient of the active convolution peak,
    use a soft‑max (log‑sum‑exp) approximation to the max.  This yields a
    weighted sum of all convolution gradients, encouraging the algorithm
    to flatten the whole convolution profile rather than only the
    currently active index.
  – Gradient of the smoothed objective can be obtained with a single
    extra convolution (via the FFT routine).  The update is an
    exponentiated‑gradient step with a diminishing stepsize and a
    temperature (α) that is annealed from a small to a larger value.
  – After each step the iterate is projected onto the probability simplex.

* A short polishing phase with the classic max‑peak sub‑gradient
  (mirror‑descent) to fine‑tune the final vector.

* Final cheap stochastic hill‑climb (swap / fractional mass move) to
  squeeze out any remaining improvement.

All phases respect a wall‑clock deadline (default 1000 s) and keep the
best vector encountered.  The returned list is ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Symmetric and heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_linear_decrease(n: int) -> np.ndarray:
    """Linear decreasing a_i = n-i, normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return project_to_simplex(a)


def generate_quadratic_decrease(n: int) -> np.ndarray:
    """Quadratic decreasing a_i = (n-i)^2, normalised."""
    i = np.arange(n, 0, -1, dtype=np.float64)
    a = i ** 2
    return project_to_simplex(a)


def generate_gaussian_decrease(n: int, sigma: float) -> np.ndarray:
    """Gaussian‑shaped decreasing tail a_i = exp(-i²/(2σ²))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i ** 2) / (2.0 * sigma ** 2))
    return project_to_simplex(a)


def generate_sine_decrease(n: int) -> np.ndarray:
    """Sine‑shaped decreasing a_i = sin(π·(n‑i)/(2·n))."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (n - i) / (2.0 * n))
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Mirror‑descent on the true max‑convolution objective (used for polishing)
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).  a is kept on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        avg = np.dot(a, g)

        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if t % 500 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    cur = compute_score(a)
    if cur < best_score:
        best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth KL‑mirror optimiser (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_mirror_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.6,
    alpha_start: float = 0.3,
    alpha_target: float = 8.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth approximation of
    max(conv(a,a)).  The smoothing parameter α is increased linearly
    from `alpha_start` to `alpha_target` during the run.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    m = 2 * n - 1                     # length of the full convolution

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- convolution & smoothing weight ----
        conv = convolution_fft(a, a)
        max_conv = conv.max()

        # linear schedule for α
        frac = t / max_iters
        alpha = alpha_start + (alpha_target - alpha_start) * frac

        # soft‑max weights (all ≤ 1)
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.shape[0])
        else:
            w_norm = w / w_sum

        # ---- weighted gradient (via an extra convolution) ----
        w_rev = w_norm[::-1]                         # reversed weight vector
        grad_full = convolution_fft(a, w_rev)        # length 3n‑2
        offset = m - 1                               # = 2n‑2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # ---- KL‑mirror update ----
        avg = np.dot(a, grad)                        # mean gradient under a
        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (grad - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        # ---- keep the best seen vector ----
        if t % 1000 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    cur = compute_score(a)
    if cur < best_score:
        best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Simple stochastic moves (swap or fractional mass transfer) with
    projection onto the simplex.  Keeps the best encountered sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                       # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
