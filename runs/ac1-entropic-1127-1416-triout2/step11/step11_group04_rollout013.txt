"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(convolve(a,a)) / (sum(a)**2)

The algorithm is a mixture of:
  * quick scan over a few analytic families (including a new Beta‑family);
  * a low‑dimensional parametrisation (piecewise‑linear exponent) that
    captures monotone decreasing shapes;
  * simulated‑annealing on the low‑dimensional parameters (many cheap
    evaluations thanks to the FFT based exact objective);
  * a short projected sub‑gradient descent starting from the best
    candidate found;
  * a final aggressive hill‑climb (simulated‑annealing) on the full
    vector.

All phases respect a global wall‑clock budget (default 1000 s).  The
search works with any length ≤≈4000; we use 4096 points (larger than
the fallback 2048) because the objective is essentially scale‑free and a
finer discretisation often yields a smaller peak in the autocorrelation.

The returned list is a plain Python list of floats (already normalised
to sum = 1, which leaves the objective unchanged).
"""
import time, math, numpy as np
from scipy.optimize import linprog  # only for the optional LP refinement

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑like density:
        f(i) ∝ ((i+0.5)/n)**alpha * (1 - (i+0.5)/n)**beta
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    # keep values non‑negative even if alpha or beta is zero
    w = (t ** max(alpha, 0.0)) * ((1.0 - t) ** max(beta, 0.0))
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Low‑dimensional parametrisation (piecewise‑linear exponent)
# ----------------------------------------------------------------------


def seq_from_piecewise_exp(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a monotone‑decreasing probability vector from a set of
    sorted parameters.

    params – sorted (ascending) values of the exponent at a few knots.
    The function interpolates the exponent linearly over the index range,
    then applies exp(-exponent) and finally normalises.
    """
    params = np.sort(params)                     # enforce monotonicity
    knots = len(params)
    positions = np.linspace(0, n - 1, knots)     # knot positions
    idx = np.arange(n)
    exp_vals = np.interp(idx, positions, params)   # linear interpolation
    a = np.exp(-exp_vals)
    a[a < 0] = 0.0
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Small optimisation helpers (sub‑gradient & hill‑climb)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal terms (i=j) counted only once
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb on the full vector.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature tuned to the current score magnitude
    T0 = cur_score * 4e-5
    scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                scale = min(scale * 1.07, 0.5)
        else:
            scale *= 0.97

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a little safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load fallback sequence if present
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose working length (larger than the fallback if possible)
    # ------------------------------------------------------------------
    n = 4096
    if best_seq is not None:
        # if fallback is longer, respect it; otherwise keep 4096
        n = max(len(best_seq), n)

    # ------------------------------------------------------------------
    # 2. Quick analytic / random scan (≈30 s)
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] quick analytic scan (n={n})")
        # families already defined above
        alphas = np.linspace(0.0, 12.0, 13)
        betas = np.linspace(0.0, 12.0, 13)
        gammas = np.linspace(0.0, 6.0, 13)

        for a_ in alphas:
            if time.time() >= deadline - 5.0:
                break
            vec = symmetric_product_family(n, a_)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (sym α={a_:.2f}) = {best_score:.6f}")

        for b_ in betas:
            if time.time() >= deadline - 5.0:
                break
            vec = left_heavy_power_family(n, b_)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (left β={b_:.2f}) = {best_score:.6f}")

        for g_ in gammas:
            if time.time() >= deadline - 5.0:
                break
            vec = exp_u_family(n, g_)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (exp γ={g_:.2f}) = {best_score:.6f}")

        # additional Beta‑family sweep (α,β)
        alphas2 = np.linspace(0.0, 8.0, 9)
        betas2 = np.linspace(0.0, 8.0, 9)
        for a_ in alphas2:
            for b_ in betas2:
                if time.time() >= deadline - 5.0:
                    break
                vec = beta_family(n, a_, b_)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best (beta a={a_:.1f},b={b_:.1f}) = {best_score:.6f}")

        # a few purely random vectors (still normalised)
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            r = np.random.rand(n)
            r = np.sort(r)[::-1]            # monotone decreasing random start
            r /= r.sum()
            sc = compute_score(r)
            if sc < best_score:
                best_score, best_seq = sc, r.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Normalise the current best to a probability vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4. Low‑dimensional piecewise‑exp parametrisation + SA
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        print(f"[{time.time():.1f}] parametric SA start (n={n})")
        M = 15  # number of knots
        # initialise params roughly matching the current best shape
        # we take the log‑inverse of the current a to obtain a sensible start
        init_w = -np.log(np.maximum(a, 1e-12))
        # compress to M points by simple averaging
        bins = np.array_split(init_w, M)
        init_params = np.array([b.mean() for b in bins])
        cur_params = np.sort(init_params)

        cur_vec = seq_from_piecewise_exp(cur_params, n)
        cur_score = compute_score(cur_vec)
        if cur_score < best_score:
            best_score, best_seq = cur_score, cur_vec.tolist()
            print(f"[{time.time():.1f}] improvement from param init: {best_score:.6f}")

        # simple SA on the parameters
        T0 = cur_score * 3e-5
        sigma0 = 0.8
        step = 0
        while time.time() < deadline - 2.0:
            step += 1
            progress = (time.time() - start_time) / (budget_s - 5.0)
            T = max(1e-9, T0 * (1.0 - progress))
            sigma = sigma0 * (1.0 - progress) + 0.02

            # propose a new point
            new_params = cur_params.copy()
            # perturb a random subset (≈30 % of the knobs)
            mask = np.random.rand(M) < 0.3
            new_params[mask] += np.random.normal(scale=sigma, size=mask.sum())
            new_params = np.clip(new_params, 0.0, None)
            new_params.sort()

            new_vec = seq_from_piecewise_exp(new_params, n)
            new_score = compute_score(new_vec)

            if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
                cur_params, cur_score, cur_vec = new_params, new_score, new_vec
                if new_score < best_score:
                    best_score, best_seq = new_score, new_vec.tolist()
                    print(f"[{time.time():.1f}] param‑SA improved to {best_score:.6f}")

            # occasional large jump to avoid stagnation
            if step % 2000 == 0 and np.random.rand() < 0.1:
                cur_params = np.sort(np.random.uniform(0.0, 5.0, size=M))

    # ------------------------------------------------------------------
    # 5. Short projected sub‑gradient descent from the best found vector
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        print(f"[{time.time():.1f}] sub‑gradient refinement start")
        a = np.array(best_seq, dtype=np.float64)
        a, sc = subgrad_optimize(a, deadline - 5.0, max_iters=2_000_000,
                                 step0=0.07)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final aggressive hill‑climb (full‑dimensional SA)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a = np.array(best_seq, dtype=np.float64)
        a, sc = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] final hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
