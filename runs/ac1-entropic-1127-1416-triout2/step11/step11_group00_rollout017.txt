    best_a = a.copy()

def generate_gaussian(n, sigma):
    i = np.arange(n) - (n-1)/2.0
    a = np.exp(-0.5 * (i/sigma)**2)
    total = a.sum()
    if total == 0:
        return np.full(n, 0.0)
    return a / total

def generate_sine(n, cycles=1):
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) * cycles / n)
    a = np.maximum(a, 0)  # sine partially positive, but for cycles=1 it's positive.
    total = a.sum()
    return a / total if total>0 else np.full(n,0.0)

def generate_triangular(n):
    mid = n // 2
    a = np.concatenate([np.arange(1, mid+1), np.arange(mid, 0, -1)][:n])
    total = a.sum()
    return a / total

def baseline_search(n, deadline, max_keep=20):
    best = [] # list of (score, vector)
    def try_candidate(vec):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # parametric families sweeps
    # powerlaw
    for p in np.linspace(0.1, 2.0, 100):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # exponential
    for lam in np.linspace(0.00001, 0.05, 150):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # beta
    for alpha in np.linspace(0.1, 5.0, 25):
        for beta in np.linspace(0.1, 5.0, 25):
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # gaussian
    for sigma_factor in np.linspace(0.05, 0.5, 20):
        sigma = sigma_factor * n
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # sine
    for cycles in [1, 2, 3]:
        if time.time() >= deadline:
            break
        try_candidate(generate_sine(n, cycles))

    # triangular
    try_candidate(generate_triangular(n))

    # random Dirichlet
    for conc in [0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(60):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]
```

def full_convolution_general(u, v):
    m = len(u) + len(v) - 1
    n_fft = 1 << ((m - 1).bit_length())
    U = np.fft.rfft(u, n=n_fft)
    V = np.fft.rfft(v, n=n_fft)
    C = U * V
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv
```

def smooth_descent(a0, deadline, beta_start=5.0, beta_end=200.0, step0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    beta = beta_start
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        maxc = conv.max()
        exp_term = np.exp(beta * (conv - maxc))
        w = exp_term / exp_term.sum()
        w_rev = w[::-1]
        grad_full = full_convolution_general(a, w_rev)
        idx = 2 * len(a) - 2 - np.arange(len(a))
        grad = 2.0 * grad_full[idx]
        step = step0 / np.sqrt(t)
        a -= step * grad
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
        # increase beta gradually
        if beta < beta_end:
            beta = min(beta_end, beta * 1.001)
        # jitter occasionally
        if t % 4000 == 0:
            jitter = np.random.randn(len(a)) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
    return best_a

"""
Minimal‑max‑autocorrelation step‑function optimiser.

Strategy
--------
1. **Exploration** – sweep a richer collection of parametric families
   (power‑law, exponential, beta, Gaussian, sine, inverted‑triangular)
   and a pool of random Dirichlet draws.  The best few candidates (by the
   exact objective 2·n·max(conv(a,a))) are kept.

2. **Smooth max‑descent** – starting from each retained candidate we run a
   projected gradient descent on a smooth approximation of the max‑operator:
   f₍β₎(a) = (1/β)·log Σₖ exp(β·(a*a)ₖ).  For growing β the gradient converges
   to a sub‑gradient of the true max.  After each step we project back onto
   the probability simplex.

3. **Fine polishing** – a short sub‑gradient walk (as in the reference code)
   tightens the final result.

4. **Stochastic hill‑climb** – a simulated‑annealing phase with mass‑transfer
   moves gives a last chance to escape shallow minima.

All phases respect the 1000 s wall‑clock budget.  The returned list of
floats is directly usable by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def full_convolution_general(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (possibly different) vectors."""
    m = len(u) + len(v) - 1
    n_fft = 1 << ((m - 1).bit_length())
    U = np.fft.rfft(u, n=n_fft)
    V = np.fft.rfft(v, n=n_fft)
    C = U * V
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a=1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_sine(n: int, cycles: int = 1) -> np.ndarray:
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) * cycles / n)
    a = np.maximum(a, 0.0)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


def generate_triangular(n: int) -> np.ndarray:
    # inverted triangle (U‑shaped) – large at the ends, small in the centre
    idx = np.arange(1, n + 1)
    a = np.maximum(idx, n - idx + 1).astype(float)
    total = a.sum()
    return a / total if total > 0 else np.full(n, 0.0)


# ----------------------------------------------------------------------
# Global baseline exploration
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 20):
    """Explore parametric families and many random Dirichlet draws.
    Returns up to ``max_keep`` vectors with the smallest scores."""
    best = []                     # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.1, 2.0, 120):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.00001, 0.05, 130):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) beta sweep (coarse grid)
    for alpha in np.linspace(0.1, 5.0, 20):
        for beta in np.linspace(0.1, 5.0, 20):
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep
    for sigma_factor in np.linspace(0.05, 0.5, 20):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma_factor * n))

    # 5) Sine (single cycle is usually enough)
    try_candidate(generate_sine(n, cycles=1))

    # 6) Inverted triangular shape
    try_candidate(generate_triangular(n))

    # 7) Random Dirichlet draws with many concentration parameters
    for conc in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(70):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (softened max)
# ----------------------------------------------------------------------


def smooth_descent(a0: np.ndarray, deadline: float,
                   beta_start: float = 5.0,
                   beta_end: float = 200.0,
                   step0: float = 0.5) -> np.ndarray:
    """Projected gradient descent on the smooth approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    beta = beta_start
    while time.time() < deadline:
        t += 1
        # full autocorrelation (a*a)
        conv = full_convolution(a)

        # soft‑max weights – numerically stable
        maxc = conv.max()
        exp_term = np.exp(beta * (conv - maxc))          # max term becomes 1
        w = exp_term / exp_term.sum()                   # length 2n‑1, sums to 1

        # gradient of (1/β)·log∑exp(β·conv) = Σ w_k ∂conv_k/∂a
        # ∂conv_k/∂a_i = 2·a_{k‑i} (if index in range)
        w_rev = w[::-1]                                 # reverse for convolution trick
        grad_full = full_convolution_general(a, w_rev)  # length 3n‑2
        idx = 2 * n - 2 - np.arange(n)                  # pick the slice that matches Σ w_{i+j} a_j
        grad = 2.0 * grad_full[idx]

        # projected gradient step
        step = step0 / np.sqrt(t)
        a -= step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # increase β slowly – the problem turns into the true max as β→∞
        if beta < beta_end:
            beta = min(beta_end, beta * 1.001)

        # occasional tiny jitter to avoid stagnation
        if t % 4000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Classical sub‑gradient polishing (as in the reference)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climb with decreasing temperature."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Smooth‑max gradient descent on several seeds
    # -----------------------------------------------------------------

            a0 = np.array(vec, dtype=float)
            a_refined = smooth_descent(a0, cand_deadline)
            sc = compute_score(a_refined)
            print(f"[smooth] refined score = {sc:.6f}")
            if sc < best_val:

        # a brief sub‑gradient polish of the current best

            a_refined = subgradient_polish(a0, sub_deadline)

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

        a_final = simulated_annealing(a_start, anneal_deadline)
        sc = compute_score(a_final)
        print(f"[anneal] final score = {sc:.6f}")
        if sc < best_val:

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
