"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Broad global sweep** – generate many candidates from several
   parametric families (power‑law, exponential, beta, Hann window,
   sinusoid) plus a large pool of random Dirichlet draws.
   The best few (by the true objective 2·n·max(conv(a,a))) are kept.

2. **Fast Adam‑style polishing** – from each elite candidate run a
   projected Adam optimiser that uses the sub‑gradient of the
   ``max``‑convolution (all active peaks contribute).  The iterate is
   projected back onto the probability simplex after every step.
   This often pushes the score noticeably lower than a plain
   sub‑gradient descent.

3. **Stochastic annealing** – a final Metropolis hill‑climb with a
   slowly decaying temperature and adaptive step‑scale, starting from
   the best polished vector.

All phases respect a wall‑clock deadline (≈ 1000 s).  The function
``propose_candidate`` returns a plain ``list`` of floats suitable for
the official ``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, length 2·n‑1)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}  (discrete Beta‑shaped density)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_hann(n: int) -> np.ndarray:
    """Hann (raised cosine) window, normalised to sum 1."""
    i = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    total = a.sum()
    return a / total


def generate_sine(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^{power}, normalised."""
    i = np.arange(n, dtype=float) + 0.5
    a = np.sin(np.pi * i / n) ** power
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Baseline global exploration
# ----------------------------------------------------------------------
def baseline_search(n: int, deadline: float, max_keep: int = 10):
    """Explore many families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # Power‑law sweep (0.1 … 2.0)
    for p in np.linspace(0.1, 2.0, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # Exponential sweep (0.0001 … 0.08)
    for lam in np.linspace(0.0001, 0.08, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # Beta sweep (α,β from 0.1 … 5.0, step ≈0.2)
    al_vals = np.linspace(0.1, 5.0, 26)
    be_vals = np.linspace(0.1, 5.0, 26)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # Hann window
    if time.time() < deadline:
        try_candidate(generate_hann(n))

    # Sine window (different powers)
    for powr in [0.5, 1.0, 2.0, 3.0]:
        if time.time() >= deadline:
            break
        try_candidate(generate_sine(n, power=powr))

    # Random Dirichlet draws with diverse concentrations
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    samples_per = 150
    for conc in concs:
        for _ in range(samples_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Adam‑style polishing (projected onto simplex)
# ----------------------------------------------------------------------
def adam_polish(a0: np.ndarray, deadline: float,
                lr: float = 0.15,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8) -> np.ndarray:
    """Projected Adam optimiser on the exact max‑convolution objective.
    Returns the best vector seen before the deadline."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # active set (indices attaining the max up to a tiny tolerance)
        eps_tol = max_val * 1e-9
        active = np.where(conv >= max_val - eps_tol)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if t % 4000 == 0:
            jitter = np.random.randn(n) * (lr * 0.002 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (Metropolis hill‑climb)
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-3,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9999,
                        step_scale_init: float = 0.04,
                        step_scale_min: float = 5e-6) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves.
    Small mass‑transfer steps are proposed; the current temperature decays."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices, transfer a fraction of mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_scale * a[i]          # amount to move from i to j
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.015, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # reject
            step_scale = max(step_scale * 0.985, step_scale_min)

        # temperature cooling
        T = max(final_temp, T * cooling_factor)

        # global jitter every few thousand steps
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a
