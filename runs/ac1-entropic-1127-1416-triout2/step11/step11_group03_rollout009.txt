"""
Propose a step‑function (non‑negative vector) minimizing the
`evaluate_sequence` objective.

Algorithm
---------
1. **Setup** – fix a modest length (n = 2048) and utilities for FFT‑based
   convolution, simplex projection and fast score evaluation.
2. **Fallback** – if a previous construction `height_sequence_1` is present,
   evaluate it first.
3. **Analytic families** – scan a few parametric families
   (symmetric product, left/right exponential, left/right power,
   uniform+U mixture).  The best few candidates are kept.
4. **Random Dirichlet starts** – generate a few hundred random
   probability vectors (`a = exp(-Exp(1)) / sum`).  Keep the top‑k.
5. **Smooth‑max Adam** – for each of the kept candidates run an Adam
   optimiser on the log‑sum‑exp surrogate of the max‑convolution.
   No artificial symmetrisation is performed, allowing fully
   asymmetric solutions.  The temperature `β` is annealed from a small
   value to a very large one (≈1e5) so the surrogate converges to the
   true ℓ∞‑norm.
6. **Projected sub‑gradient descent** – a short sub‑gradient phase
   refines the best Adam output.
7. **Simulated‑annealing hill‑climb** – a final cheap random walk on the
   simplex with a slowly decreasing temperature polishes the result.
8. **Return** the best raw list of floats found before the deadline.

All time‑consuming loops check the wall‑clock deadline (`budget_s` seconds
minus a small safety margin) and abort early if needed.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, simplex projection, score)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):

        score = 2·n·max(a*a)

    where the convolution is the full aperiodic convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (some symmetric, some intentionally asymmetric)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} (symmetric)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_exponential_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(‑γ·i) (monotonically decreasing, asymmetric)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    return w / w.sum()


def right_exponential_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(‑γ·(n‑1‑i)) (mirror of left)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * (n - 1 - i))
    return w / w.sum()


def left_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (asymmetric)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def right_power_family(n: int, beta: float) -> np.ndarray:
    """mirror of left_power_family."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(n - 0.5 - i))
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and a U‑shaped exponential."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = left_exponential_family(n, gamma) * 0.0
    # construct true symmetric “U” shape
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    u_shape = np.exp(-gamma * dist)
    u_shape /= u_shape.sum()
    w = lam * uniform + (1.0 - lam) * u_shape
    return w / w.sum()


# ----------------------------------------------------------------------
# Random starters
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int) -> np.ndarray:
    """Generate a random probability vector (Dirichlet(1) = exponential)."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate (no forced symmetry)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 1.0,
                         beta_end: float = 1e5,
                         lr: float = 0.01) -> Tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp approximation of max(conv).

    The iterate is kept on the simplex via a soft‑max parametrisation.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise logits; add eps to avoid log(0)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for temperature β (smooth‑max annealing)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)                     # probability vector
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 5000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improvement → {best_score:.6f}")

    # final projection & score
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement → {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     step0: float = 0.05) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the exact max‑convolution objective.
    No symmetry is enforced.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for max at shift k:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improvement → {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (purely on the simplex)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> Tuple[np.ndarray, float]:
    """
    Simple SA walk: swap or transfer a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # clipping handled by projection
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
