def subgradient_momentum(a0, deadline, step0=0.4, beta=0.9):
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-8
        active = np.where(conv >= max_val - eps)[0]
        # Compute gradient aggregated over active lags
        grad = np.zeros_like(a)
        idx = np.arange(a.shape[0])
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < a.shape[0])
            grad[mask] += 2.0 * a[j[mask]]
        # adjust for diagonal entries (i=j case)
        # Actually above counting duplicates counts each pair twice; but that's fine as used previously.
        step = step0 / np.sqrt(t)
        v = beta * v + (1 - beta) * grad
        a -= step * v
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        if t % 2000 == 0:
            # jitter
            jitter = (np.random.randn(a.shape[0]) *
                      (step0 * 0.01 / np.sqrt(t)))
            a += jitter
            a = project_to_simplex(a)
    return best_a
```

def generate_general(n, params):
    # params = [alpha, beta, lam] with log transforms.
    alpha, beta, lam = np.exp(params[0]), np.exp(params[1]), np.exp(params[2])
    i = np.arange(1, n+1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    exp_factor = np.exp(-lam * (i - 1))
    a = left * right * exp_factor
    total = a.sum()
    if total <=0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total
```

"""
Hybrid optimiser for ``evaluate_sequence``.

The algorithm combines several ideas:

1. **Extensive global exploration** – sweep three classical families
   (power‑law, exponential, beta‑shaped) on a fine grid and supplement
   them with a large set of random Dirichlet draws.  The ``max_keep``
   best vectors (by the exact objective 2·n·max(conv)) are retained.

2. **Local continuous optimisation** – each parametric family is
   refined with a small‐budget derivative‑free optimiser
   (SciPy Nelder‑Mead) on the logarithmic parameters, guaranteeing
   positivity.  The best result over all families is added to the pool.

3. **Momentum projected sub‑gradient polishing** – starting from the
   best candidate, a projected sub‑gradient descent with exponential
   momentum is run until the time budget expires.  The active set
   (indices attaining the current convolution maximum) provides the
   sub‑gradient.

4. **Stochastic annealing** – a final Metropolis hill‑climb with a
   decreasing temperature helps to escape shallow local minima.

The search respects the 1000 s wall‑clock limit; intermediate results
are printed for monitoring.  The returned list of non‑negative floats
can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α-1}·(n-i)^{β-1} (discrete Beta)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Book‑keeping utilities
# ----------------------------------------------------------------------


def update_best(best, score, vec, max_keep):
    """Maintain a sorted list of the best ``max_keep`` entries."""
    if len(best) < max_keep:
        best.append((score, vec.copy()))
        best.sort(key=lambda x: x[0])
    elif score < best[-1][0]:
        best[-1] = (score, vec.copy())
        best.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Global baseline search (extended)
# ----------------------------------------------------------------------


def baseline_search_ext(n: int, deadline: float, max_keep: int = 10):
    """
    Exhaustive exploration of a few parametric families and many
    random Dirichlet draws.  Returns up to ``max_keep`` best vectors.
    """
    best = []  # list of (score, vector)

    # 1) Power‑law sweep (fine grid)
    for p in np.linspace(0.3, 1.5, 241):
        if time.time() >= deadline:
            break
        vec = generate_powerlaw(n, p)
        sc = compute_score(vec)
        update_best(best, sc, vec, max_keep)

    # 2) Exponential sweep (fine grid)
    for lam in np.linspace(1e-5, 0.05, 301):
        if time.time() >= deadline:
            break
        vec = generate_exponential(n, lam)
        sc = compute_score(vec)
        update_best(best, sc, vec, max_keep)

    # 3) Beta‑shaped sweep (moderately dense grid)
    al_vals = np.linspace(0.1, 6.0, 31)
    be_vals = np.linspace(0.1, 6.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            vec = generate_beta(n, alpha, beta)
            sc = compute_score(vec)
            update_best(best, sc, vec, max_keep)

    # 4) Random Dirichlet draws (large sample)
    concs = [0.1, 0.2, 0.5, 1.0, 2.0]
    total_draws = 6000
    for _ in range(total_draws):
        if time.time() >= deadline:
            break
        conc = np.random.choice(concs)
        vec = np.random.dirichlet([conc] * n)
        sc = compute_score(vec)
        update_best(best, sc, vec, max_keep)

    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Local continuous optimisation of families (Nelder–Mead)
# ----------------------------------------------------------------------


class PowerLawObjective:
    def __init__(self, n):
        self.n = n

    def __call__(self, logp):
        p = np.exp(logp[0])
        a = generate_powerlaw(self.n, p)
        return compute_score(a)


class ExponentialObjective:
    def __init__(self, n):
        self.n = n

    def __call__(self, loglam):
        lam = np.exp(loglam[0])
        a = generate_exponential(self.n, lam)
        return compute_score(a)


class BetaObjective:
    def __init__(self, n):
        self.n = n

    def __call__(self, log_params):
        alpha = np.exp(log_params[0])
        beta = np.exp(log_params[1])
        a = generate_beta(self.n, alpha, beta)
        return compute_score(a)


class CombinedObjective:
    """
    Three‑parameter family: beta‑shape multiplied by a decaying exponential.
    a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}·exp(-λ·i)
    """
    def __init__(self, n):
        self.n = n

    def __call__(self, log_params):
        alpha = np.exp(log_params[0])
        beta = np.exp(log_params[1])
        lam = np.exp(log_params[2])
        i = np.arange(1, self.n + 1, dtype=float)
        left = np.power(i, alpha - 1.0)
        right = np.power(self.n - i + 1.0, beta - 1.0)
        exp_factor = np.exp(-lam * (i - 1.0))
        a = left * right * exp_factor
        total = a.sum()
        if total == 0 or not np.isfinite(total):
            return 1e6
        a = a / total
        return compute_score(a)


def optimise_families(n, deadline, current_best_vec, current_best_score):
    """
    Run a short Nelder–Mead optimisation for each parametric family.
    Return the best vector found (may be the original one).
    """
    try:
        import scipy.optimize as opt
    except Exception:
        # SciPy not available – skip this phase
        return current_best_vec, current_best_score

    # store global best
    best_vec = current_best_vec
    best_score = current_best_score

    # ---- power law ----
    if time.time() < deadline:
        obj = PowerLawObjective(n)
        res = opt.minimize(
            obj,
            x0=np.log(np.array([0.7])),
            method="Nelder-Mead",
            options={"maxfev": 250, "disp": False},
        )
        vec = generate_powerlaw(n, np.exp(res.x[0]))
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec

    # ---- exponential ----
    if time.time() < deadline:
        obj = ExponentialObjective(n)
        res = opt.minimize(
            obj,
            x0=np.log(np.array([0.001])),
            method="Nelder-Mead",
            options={"maxfev": 250, "disp": False},
        )
        vec = generate_exponential(n, np.exp(res.x[0]))
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec

    # ---- beta ----
    if time.time() < deadline:
        obj = BetaObjective(n)
        res = opt.minimize(
            obj,
            x0=np.log(np.array([1.0, 1.0])),
            method="Nelder-Mead",
            options={"maxfev": 300, "disp": False},
        )
        vec = generate_beta(n, np.exp(res.x[0]), np.exp(res.x[1]))
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec

    # ---- combined (alpha,beta,lam) ----
    if time.time() < deadline:
        obj = CombinedObjective(n)
        res = opt.minimize(
            obj,
            x0=np.log(np.array([1.0, 1.0, 0.001])),
            method="Nelder-Mead",
            options={"maxfev": 400, "disp": False},
        )
        vec = generate_beta(n, np.exp(res.x[0]), np.exp(res.x[1]))
        # apply the exponential factor after the beta shape
        lam = np.exp(res.x[2])
        i = np.arange(1, n + 1, dtype=float)
        exp_factor = np.exp(-lam * (i - 1.0))
        vec = vec * exp_factor
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec

    return best_vec, best_score


# ----------------------------------------------------------------------
# Momentum projected sub‑gradient polishing
# ----------------------------------------------------------------------


def subgradient_momentum(a0: np.ndarray, deadline: float,
                         step0: float = 0.35, beta: float = 0.9) -> np.ndarray:
    """
    Projected sub‑gradient descent with exponential momentum.
    The sub‑gradient is built from all lag indices attaining the current
    convolution maximum (within a tiny relative tolerance).
    """
    a = a0.copy()
    v = np.zeros_like(a)
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # tolerance – a few extra indices help to smooth the descent
        eps = max_val * 1e-8
        active = np.where(conv >= max_val - eps)[0]

        # accumulate (sub‑)gradient over the active lags
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        v = beta * v + (1.0 - beta) * g
        a -= step * v
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = (np.random.randn(n) *
                      (step0 * 0.005 / np.sqrt(t)))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        cooling_factor: float = 0.9998,
                        step_scale_init: float = 0.03,
                        step_scale_min: float = 1e-6) -> np.ndarray:
    """Metropolis hill‑climb with a decaying temperature."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        # pick a source index with non‑zero mass
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept move
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # reject – shrink the step scale slightly
            step_scale = max(step_scale * 0.98, step_scale_min)

        # temperature schedule
        T = max(final_temp, T * cooling_factor)

        # global jitter every few thousand iterations
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            fallback_seq = [float(x) for x in fallback]
            val = evaluate_sequence(fallback_seq)
            print(f"[fallback] score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback_seq
        except Exception:
            pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0
    if time.time() < baseline_deadline:
        candidates = baseline_search_ext(n, baseline_deadline, max_keep=10)
        for vec in candidates:
            seq = vec.tolist()
            val = evaluate_sequence(seq)
            if val < best_val:
                best_val, best_seq = val, seq
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Local continuous optimisation of parametric families
    # -----------------------------------------------------------------
    opt_deadline = deadline - 150.0
    if best_seq is not None and time.time() < opt_deadline:
        cur_vec = np.array(best_seq, dtype=float)
        cur_score = compute_score(cur_vec)
        vec_opt, score_opt = optimise_families(n, opt_deadline, cur_vec, cur_score)
        if score_opt < best_val:
            best_val, best_seq = score_opt, vec_opt.tolist()
            print(f"[param_opt] improved score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Momentum sub‑gradient polishing
    # -----------------------------------------------------------------
    polish_deadline = deadline - 50.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_momentum(a0, polish_deadline,
                                         step0=0.35, beta=0.9)
        val = evaluate_sequence(a_refined.tolist())
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()
        print(f"[polish] refined score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 4. Simulated annealing / stochastic hill‑climb
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline,
                                      init_temp=5e-4,
                                      final_temp=1e-7,
                                      cooling_factor=0.9998,
                                      step_scale_init=0.03,
                                      step_scale_min=1e-6)
        val = evaluate_sequence(a_final.tolist())
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()
        print(f"[anneal] final score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback uniform] score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
