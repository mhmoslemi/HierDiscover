def beta_family(n, p, q):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # to avoid zero, clamp
    x = np.clip(x, 1e-12, 1-1e-12)
    w = np.power(x, p-1) * np.power(1-x, q-1)
    # w may be infinite if p<1 and x->0 but we clip.
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0.0:
        # fallback uniform
        w = np.ones(n)
    return w / w.sum()

"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

under the simplex constraint.  

The search proceeds in three stages  

1.  A broad multi‑length scan of several analytic families (including new
    Beta‑distribution, Gaussian and mixed families) and many random
    Dirichlet draws.  The best vector over all tested lengths is kept.
2.  A refinement cascade – exponentiated‑gradient (mirror descent), a
    smooth‑max Adam optimiser, cheap projected sub‑gradient descent and a
    state‑of‑the‑art LP‑based mixing step – all respecting the remaining
    wall‑clock budget.
3.  A final stochastic hill‑climb (simulated‑annealing style) and a
    symmetrisation test.

All heavy work uses FFT‑based convolutions for speed and the code is
careful to stop before the global deadline (default 1000 s).  The
returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    where * denotes full convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (including new ones)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family using product of distances to the ends."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power‑law family."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential “U” shape."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential U‑shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type distribution on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid exact 0/1 which would create infinities for p<1 or q<1
    eps = 1e-15
    x = np.clip(x, eps, 1 - eps)
    w = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0.0:
        w = np.ones(n, dtype=np.float64)
    return w / w.sum()


def symmetric_beta_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta (p = q = alpha) – convenient one‑parameter family."""
    return beta_family(n, alpha, alpha)


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        w = np.ones(n, dtype=np.float64)
    return w / w.sum()


def sin_squared_family(n: int, power: float = 1.0) -> np.ndarray:
    """Shape proportional to sin(pi * x)^{power}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.sin(np.pi * x) ** power
    w = np.where(w > 0, w, 0.0)
    if w.sum() == 0.0:
        w = np.ones(n, dtype=np.float64)
    return w / w.sum()


def random_dirichlet_start(n: int) -> np.ndarray:
    """Dirichlet draw with a random concentration."""
    # concentration in [0.05, 5] on a log‑scale
    log_alpha = np.random.uniform(np.log(0.05), np.log(5.0))
    alpha = math.exp(log_alpha)
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimisation on the true max
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]          # factor 1 for the diagonal term

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)

        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """Minimise the smooth surrogate of max‑convolution using Adam."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule the smoothing temperature β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement step (the classic state‑of‑the‑art iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (size m×n) so that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)      # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)

    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two coordinates
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function respects a global time budget and
    returns a plain Python list ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine problem sizes to explore
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # candidate lengths (avoid extremely huge sizes early on)
    cand_lengths = [1024, 2048, 4096, 8192]
    # make sure we at least try the fallback length if it is larger
    if fallback_len and fallback_len not in cand_lengths:
        cand_lengths.append(fallback_len)
    cand_lengths = sorted(set(cand_lengths))

    # ------------------------------------------------------------------
    # 2. Quick analytic / random scan over all candidate lengths
    # ------------------------------------------------------------------
    for n in cand_lengths:
        if time.time() >= deadline - 200.0:
            break
        print(f"[{time.time():.1f}] scanning length n={n}")

        # Choose grid granularity based on size
        fine = n <= 4096
        alphas = np.linspace(0.0, 30.0, 31 if fine else 16)
        betas = np.linspace(0.0, 30.0, 31 if fine else 16)
        gammas = np.linspace(0.0, 15.0, 31 if fine else 16)
        alphas_beta = np.linspace(0.1, 5.0, 41 if fine else 21)
        sigmas = np.linspace(n / 20.0, n / 2.0, 20 if fine else 10)

        # 2.1 symmetric product family (U‑shape)
        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # 2.2 left‑heavy power law
        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        # 2.3 exponential U‑shape
        for gamma in gammas:
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        # 2.4 Mixed uniform + exponential
        for _ in range(20):
            if time.time() >= deadline - 30.0:
                break
            lam = np.random.rand()
            gamma = np.random.uniform(0.0, 15.0)
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix) = {best_score:.6f}")

        # 2.5 Symmetric Beta family (new)
        for alpha in alphas_beta:
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_beta_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (beta‑α={alpha:.2f}) = {best_score:.6f}")

        # 2.6 Gaussian centred family (new)
        for sigma in sigmas:
            if time.time() >= deadline - 30.0:
                break
            vec = gaussian_family(n, sigma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (gauss‑σ={sigma:.1f}) = {best_score:.6f}")

        # 2.7 Random Dirichlet draws
        for _ in range(80):
            if time.time() >= deadline - 30.0:
                break
            vec = random_dirichlet_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑dir) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Normalise the current best vector (ensuring sum = 1)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(2048, 1.0 / 2048, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 4. Refinement pipeline (respect remaining budget)
    # ------------------------------------------------------------------
    # 4.1 Exponentiated‑gradient mirror descent
    if time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] starting EG mirror‑descent")
        a_opt, sc_opt = eg_optimize(
            a,
            deadline - 180.0,
            max_iters=400_000,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] after EG best = {best_score:.6f}")

    # 4.2 Adam smooth‑max refinement
    if time.time() < deadline - 180.0:
        print(f"[{time.time():.1f}] starting Adam refinement")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline - 120.0,
            max_iters=250_000,
            beta_start=4.0,
            beta_end=2000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # 4.3 Projected sub‑gradient descent (cheap long tail)
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_opt, sc_opt = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=4_000_000,
            step0=0.07,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # 4.4 LP‑based refinement loop (more aggressive)
    if time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] starting LP refinement loop")
        mix_weight = 0.2
        lp_iters = 0
        while time.time() < deadline - 20.0:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-7:
                best_score = sc
                a = a_new
                print(f"[{time.time():.1f}] LP step {lp_iters} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.05, mix_weight * 0.85)
            lp_iters += 1
            if lp_iters >= 30:
                break

    # ------------------------------------------------------------------
    # 5. Symmetrisation test (often helps)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
