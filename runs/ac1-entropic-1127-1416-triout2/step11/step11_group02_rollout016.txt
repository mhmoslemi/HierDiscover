"""
Step‑function optimiser for ``evaluate_sequence``.

Algorithm Overview
------------------
* Choose a moderate length *n* (2048) – large enough to give a good
  lower bound but still cheap to convolve many times.
* **Exploratory sweep** – evaluate many simple parametric families
  (power‑law, exponential, symmetric “V” shape and edge‑plateau
  constructions).  Keep the best normalised vector.
* **Random restarts** – generate a few Dirichlet samples to avoid a
  poor local basin.
* **Mirror‑descent refinement** – the objective
  ``M = max(conv(a,a))`` is convex in the probability simplex.
  We use an exponentiated‑gradient (KL‑mirror) update:

        a ← a ⊙ exp(−η·g)   (⊙ element‑wise)

  where *g* is the sub‑gradient of the max‑convolution obtained from
  the currently active convolution entries.
  The step‑size η decays as 1/√t.  After each update we enforce
  symmetry (a = (a + a[::-1])/2) and project onto the simplex.
* **Additive sub‑gradient polishing** – a short Euclidean‑projected
  sub‑gradient phase smooths the remaining kinks.
* **Stochastic simulated‑annealing hill‑climb** – a cheap greedy
  mass‑transfer neighbourhood with a temperature schedule for escaping
  tiny plateaus.
* The whole search respects the 1000 s wall‑clock budget; the final
  sequence is returned as a plain Python list.

All helper functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The objective is scale‑invariant, therefore we keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def geometric(n: int, r: float) -> np.ndarray:
    a = (r ** np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def v_shape(n: int, p: float) -> np.ndarray:
    """Symmetric V‑shape: (i+1)^(-p) + (n‑i)^(-p)."""
    left = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    right = np.arange(n, 0, -1, dtype=np.float64) ** (-p)
    a = left + right
    a /= a.sum()
    return a


def plateau_edges(n: int, L_frac: float, ratio: float) -> np.ndarray:
    """High plateau on both ends of length ~L_frac·n, ratio = edge/mid."""
    L = max(1, int(L_frac * n))
    if 2 * L >= n:
        L = n // 2 - 1
    if L <= 0:
        return None
    # solve 2*L*we + (n-2L)*wm = 1, we = ratio*wm
    wm = 1.0 / (2 * L * ratio + (n - 2 * L))
    we = ratio * wm
    a = np.full(n, wm, dtype=np.float64)
    a[:L] = we
    a[-L:] = we
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient utilities
# ----------------------------------------------------------------------
def active_indices(conv: np.ndarray, max_ks: int = 30) -> np.ndarray:
    """Return up to ``max_ks`` indices of the largest convolution entries."""
    if conv.size <= max_ks:
        return np.arange(conv.size)
    # ``np.argpartition`` gives the top‑k indices without full sort
    idx = np.argpartition(-conv, max_ks)[:max_ks]
    # sort them for deterministic behaviour (optional)
    return np.sort(idx)


def subgradient_from_active(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv(a,a))`` using the active
    convolution indices ``ks``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g[idx] += a[k - idx]
    g *= 2.0
    return g


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) refinement
# ----------------------------------------------------------------------
def exp_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    KL‑mirror descent on the simplex using an exponentiated‑gradient
    update.  The iteration stops when the wall‑clock deadline is hit.
    Returns the best vector seen (still Σ = 1, non‑negative, symmetric).
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        M = conv.max()
        ks = active_indices(conv, max_ks=30)
        g = subgradient_from_active(a, ks)

        # step size decays as 1/√t  (η₀ = 0.5 works well)
        eta = 0.5 / np.sqrt(it)
        # clip gradients to avoid overflow in the exponent
        g_clip = np.clip(g, 0.0, 10.0)

        # exponentiated‑gradient update (move opposite to the gradient)
        a = a * np.exp(-eta * g_clip)

        # enforce symmetry (helps the objective)
        a = (a + a[::-1]) * 0.5

        # projection onto the probability simplex
        a = project_to_simplex(a)

        # track best
        conv = full_convolution(a)
        M_new = conv.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

        # occasional early exit if improvement stalls
        if it % 2000 == 0 and time.time() > deadline - 5:
            break

    return best_a


# ----------------------------------------------------------------------
# Additive projected sub‑gradient polishing (Euclidean)
# ----------------------------------------------------------------------
def additive_subgradient(polish_start: np.ndarray, deadline: float,
                         step0: float = 0.12) -> np.ndarray:
    """
    Classic Euclidean projected sub‑gradient descent (with decreasing
    step size) for a short polishing phase.
    """
    a = polish_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        M = conv.max()
        ks = active_indices(conv, max_ks=30)
        g = subgradient_from_active(a, ks)

        step = step0 / np.sqrt(it)
        a = a - step * g
        a = (a + a[::-1]) * 0.5            # keep symmetry
        a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

        if it % 2000 == 0 and time.time() > deadline - 5:
            break

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def hill_climb_sa(start_seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Tiny stochastic mass‑transfer moves with Metropolis acceptance.
    The temperature decays linearly with the remaining time.
    """
    a = start_seq.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]

    start_time = time.time()
    while time.time() < deadline:
        remaining = deadline - time.time()
        # linear temperature schedule (starts at ~0.01)
        T = max(1e-12, remaining / (deadline - start_time)) * 0.02

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose a transfer – a random fraction of the source mass
        delta = np.random.rand() * 0.5 * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(M_cand - best_M) / T):
                a = cand

    return best_a


# ----------------------------------------------------------------------
# Global parametric sweep (cheap exploration)
# ----------------------------------------------------------------------
def sweep_initial(n: int, deadline: float):
    best_a = None
    best_val = float('inf')

    # power‑law
    for p in np.linspace(0.20, 1.50, 66):
        if time.time() >= deadline:
            break
        a = power_law(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # geometric decay
    for r in np.linspace(0.50, 0.99, 80):
        if time.time() >= deadline:
            break
        a = geometric(n, r)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # symmetric V‑shape
    for p in np.linspace(0.25, 2.00, 71):
        if time.time() >= deadline:
            break
        a = v_shape(n, p)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # edge‑plateau families
    for L_frac in np.linspace(0.01, 0.30, 10):
        if time.time() >= deadline:
            break
        for ratio in np.linspace(2.0, 60.0, 20):
            if time.time() >= deadline:
                break
            a = plateau_edges(n, L_frac, ratio)
            if a is None:
                continue
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # a few random Dirichlet samples (sparse candidates)
    for _ in range(25):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(alpha=np.ones(n))
        a = (a + a[::-1]) * 0.5            # enforce symmetry early
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence minimising
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # Optional fallback (previous best known construction)
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 0. Global parametric sweep (allocates ~30 % of the budget)
    # -----------------------------------------------------------------
    sweep_deadline = start_time + budget_s * 0.30
    a0, val0 = sweep_initial(2048, sweep_deadline)
    if a0 is not None:
        print(f"[sweep] best score = {val0:.6f}")
        if val0 < best_val:
            best_val, best_seq = val0, a0.tolist()

    # -----------------------------------------------------------------
    # 1. Random‑restart mirror‑descent refinement (≈50 % of the budget)
    # -----------------------------------------------------------------
    refine_deadline = start_time + budget_s * 0.80
    while time.time() < refine_deadline:
        # give each run at most 20 s (or less if we are close to the limit)
        run_deadline = min(time.time() + 20.0, refine_deadline)
        # 80 % probability: start from the current best,
        # 20 % probability: start from a fresh random Dirichlet point.
        if np.random.rand() < 0.2:
            a_start = np.random.dirichlet(alpha=np.ones(2048))
            a_start = (a_start + a_start[::-1]) * 0.5
            a_start = project_to_simplex(a_start)
        else:
            a_start = np.array(best_seq, dtype=float) if best_seq is not None else a0

        a_refined = exp_gradient_descent(a_start, run_deadline)
        val_refined = compute_score(a_refined)
        if val_refined < best_val:
            best_val, best_seq = val_refined, a_refined.tolist()
            print(f"[exp‑grad] improved score = {val_refined:.6f}")

        # stop early if we are out of time
        if time.time() > refine_deadline - 5.0:
            break

    # -----------------------------------------------------------------
    # 2. Additive sub‑gradient polishing (≈10 % of the budget)
    # -----------------------------------------------------------------
    polish_deadline = start_time + budget_s * 0.90
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = additive_subgradient(a0, polish_deadline, step0=0.12)
        val_polished = compute_score(a_polished)
        if val_polished < best_val:
            best_val, best_seq = val_polished, a_polished.tolist()
            print(f"[sub‑grad] polished score = {val_polished:.6f}")

    # -----------------------------------------------------------------
    # 3. Final stochastic hill‑climb (remaining time)
    # -----------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb_sa(a_start, hill_deadline)
        val_final = compute_score(a_final)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.tolist()
            print(f"[hill‑climb] final score = {val_final:.6f}")

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 2048
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final official evaluation (the harness will call the same)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
