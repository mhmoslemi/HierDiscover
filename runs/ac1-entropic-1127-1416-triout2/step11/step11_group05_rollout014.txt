"""
Propose a step‑function sequence that minimises the provided
evaluate_sequence() objective.

Main ideas (beyond the previous runner):
* 5 analytic families (including a new “exp‑product” family) sampled very
  densely on a moderately large length (4096 or 8192).  The best
  candidate of this cheap scan seeds the optimisation.
* A multi‑restart, multi‑stage L‑BFGS optimiser on the smooth log‑sum‑exp
  surrogate of the max‑convolution.  The β‑schedule is much longer and
  each stage is allowed many more internal iterations (up to 500) so
  the optimiser can “see” deeper minima.  Random Dirichlet seeds with
  various concentration parameters are also tried.
* A final simulated‑annealing hill‑climb with a slightly richer move set
  (swap, mass transfer, and targeted mass removal from indices that
  participate in the current max‑convolution) tightens the result.

All steps respect the 1000 s wall‑clock budget, continuously update the
best sequence seen so far and print progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
          2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (cheap seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """ a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex. """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """ a_i ∝ (i+½)^{‑β}, monotone decreasing. """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


def exp_product_family(n: int, gamma: float) -> np.ndarray:
    """
    Very steep U‑shape: a_i ∝ exp(-γ·norm((i+½)*(n‑½‑i))),
    where the product is normalised to [0,1] before applying the
    exponential.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    max_prod = ((n - 0.5) / 2.0) ** 2
    norm = prod / max_prod               # ∈[0,1]
    w = np.exp(-gamma * norm)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


# ----------------------------------------------------------------------
# Smooth‑max surrogate (gradient available analytically)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) for the β‑smoothed max of the convolution.

    f   = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
    grad = ∂f/∂z (gradient w.r.t. the soft‑max parameters)
    """
    a = softmax(z)                 # simplex, non‑negative
    conv = convolution_fft(a, a)   # length 2n‑1
    maxc = conv.max()
    # stable soft‑max over the 2n‑1 entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta

    w = expw / sumexp               # weights summing to 1

    # gradient of max‑conv w.r.t. a :
    a_rev = a[::-1]
    # conv_rev = a_rev * w  (full convolution)
    conv_rev = convolution_fft(a_rev, w)          # length 3n‑2
    # central slice corresponds to indices where a participates
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]       # shape (n,)

    # back‑prop through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Run L‑BFGS on the sequence a_start (assumed ∑a=1) using a schedule
    of smoothing parameters `betas`.  Returns the best a found (still on
    the simplex) and its exact score (2·n·max(conv)).
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        z0 = np.log(a + 1e-15)                     # initialise soft‑max logits

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as exc:
            print(f"    L‑BFGS failed: {exc}")
            continue

        a = softmax(res.x)

        # optional symmetrisation (helps numerical stability)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Final simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_500_000) -> tuple[np.ndarray, float]:
    """
    Simple annealing on the simplex.  Moves:
      * swap two entries,
      * shift a fraction of mass between two positions,
      * targeted reduction of mass from the most “responsible”
        indices (those that appear in the current max‑convolution).
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = max(cur_score, 1e-6) * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # ----- propose candidate -----
        cand = cur_a.copy()
        r = np.random.rand()
        if r < 0.33:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.66:  # simple mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        else:  # targeted move away from the current max‑conv contributor
            conv = convolution_fft(cand, cand)
            k = int(conv.argmax())
            # pick a random index that participates in this sum
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            i = np.random.randint(low, high + 1)
            j = k - i
            # move a small fraction from the larger of the two to a random other slot
            src = i if cand[i] > cand[j] else j
            dst = np.random.randint(n)
            if src != dst:
                delta = step_scale * cand[src]
                cand[src] -= delta
                cand[dst] += delta
                if cand[src] < 0.0:
                    cand[src] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # ----- acceptance test -----
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

    best_score = float("inf")

    # 0. Choose a problem size.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()

    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------

        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}]   new best (scan) = {best_score:.6f}")

    # uniform starter
    record_candidate(np.full(n, 1.0 / n))

    # family scans – generous parameter ranges,
    # break early if we run out of time.
    families = []
    # symmetric product (α)

    # left‑heavy power (β)

        families.append(left_heavy_power_family(n, beta))
    # exponential U (γ)
    for gamma in np.linspace(0.0, 12.0, 13):
        families.append(exp_u_family(n, gamma))
    # exponential product (δ)

    for vec in families:

        start_candidates.append(rnd / rnd.sum())

    betas = [0.5, 2, 5, 20, 80, 300, 1000, 3000, 10000, 30000, 80000]

        if time.time() >= deadline - 70:

                                        deadline=deadline - 50)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}]   L‑BFGS improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional symmetry projection (often helpful).
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        sc_sym = compute_score(a_arr)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_arr.tolist()
            print(f"[{time.time():.1f}] symmetry projection improved best = {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        print(f"[{time.time():.1f}] final SA polishing")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=2_500_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Safety net – uniform if everything else failed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = compute_score(np.array(best_seq))

    # ------------------------------------------------------------------
    # 7. Final evaluation (the real wrapper may clip/clamp values).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
