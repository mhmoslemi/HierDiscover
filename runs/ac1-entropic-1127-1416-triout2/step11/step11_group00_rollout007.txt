def full_conv_fft(x, y):
    n = len(x)
    m = len(y)
    L = n + m - 1
    n_fft = 1 << ((L - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:L]
    return conv
```

def smooth_max_and_weights(conv, alpha):
    # conv: np.ndarray length M
    # compute weights w_i = exp(alpha * (conv_i - max_conv)) / sum...
    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    smooth = max_c + np.log(sum_exp) / alpha
    return smooth, w

def conv_fft(x, y):

"""
Smooth‑max gradient descent on the ℓ∞‑norm of the self‑convolution.

The objective (for a vector a with sum(a)=1) is 2·n·max(a*a).
We minimise a smooth approximation of the max:
    f_α(a) = (1/α)·log( Σ_i exp(α·(a*a)_i) )
Its gradient is:
    ∇f_α(a) = 2·cross‑correlation( a , w )
where w_i ∝ exp(α·(a*a)_i) (weights sum to 1).

We use Adam optimisation with projection onto the probability simplex.
A schedule slowly increases α so the smoothed objective converges to the
true max.  Several random restarts (Dirichlet draws, exponential,
Beta‑shaped and random‑phase Fourier seeds) are explored.
The best vector found (converted to a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y using a real FFT."""
    L = len(x) + len(y) - 1
    n_fft = 1 << ((L - 1).bit_length())          # next power of two ≥ L
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    C = X * Y
    conv = np.fft.irfft(C, n=n_fft)[:L]
    return conv

def self_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a."""
    n = a.shape[0]
    L = 2 * n - 1
    n_fft = 1 << ((L - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:L]
    return conv

# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective for a (sum = 1): 2·n·max(a*a)."""
    n = a.shape[0]
    conv = self_conv_fft(a)
    return float(2.0 * n * conv.max())


def smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max weights w_i = exp(alpha·conv_i)/Σ_j exp(alpha·conv_j).
    The computation is numerically stabilised by subtracting max(conv).
    """
    cmax = conv.max()
    exp_vals = np.exp(alpha * (conv - cmax))
    sum_exp = exp_vals.sum()
    return exp_vals / sum_exp


def gradient_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max objective with respect to a.
    w : weights of length 2·n‑1 (same length as a*a)
    a : current vector (length n)
    Returns ∇f = 2·cross‑correlation(a, w) as a length‑n vector.
    """
    # cross‑correlation a ↔ w can be obtained via convolution with w reversed
    w_rev = w[::-1]
    h = conv_fft(a, w_rev)                     # length = 3·n‑2
    M = len(w)                                 # = 2·n‑1
    start = M - len(a)                         # = n‑1
    grad = 2.0 * h[start:start + len(a)]
    return grad

# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------
def random_dirichlet_init(n: int, conc: float) -> np.ndarray:
    return np.random.dirichlet([conc] * n)


def exponential_init(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a / a.sum()


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if a.sum() == 0 or not np.isfinite(a).all():
        return np.full(n, 1.0 / n)
    return a / a.sum()


def random_phase_fft_init(n: int, seed: int = None) -> np.ndarray:
    """Generate a non‑negative vector from a random Fourier phase spectrum."""
    if seed is not None:
        np.random.seed(seed)
    L = 1 << ((n - 1).bit_length())            # next power of two ≥ n
    # random complex spectrum (Hermitian symmetry is not required for irfft)
    Z = np.random.randn(L) + 1j * np.random.randn(L)
    # obtain a real signal, take absolute square to enforce non‑negativity
    a = np.real(np.fft.ifft(Z))
    a = np.abs(a) ** 2
    a = a[:n]
    a = a / a.sum()
    return a

# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # A larger problem size gives a lower theoretical bound (≈1 for n≫1).
    n = 8192

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to record improvements
    # ------------------------------------------------------------------
    def try_update(a_vec):
        nonlocal best_seq, best_score
        # a_vec is already on the simplex (sum=1)
        cur = compute_score(a_vec)
        if cur < best_score:
            best_score, best_seq = cur, a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Build a modest pool of diverse starting points
    # ------------------------------------------------------------------
    init_pool = []

    # uniform start – useful baseline
    init_pool.append(np.full(n, 1.0 / n))

    # Dirichlet draws with several concentrations
    for conc in [0.2, 0.5, 0.9, 1.5, 2.5]:
        init_pool.append(random_dirichlet_init(n, conc))

    # exponential families
    for lam in [0.0003, 0.0007, 0.0015]:
        init_pool.append(exponential_init(n, lam))

    # beta‑shaped families
    for a, b in [(0.3, 0.3), (1.0, 1.0), (2.0, 0.5), (0.5, 2.0), (3.0, 3.0)]:
        init_pool.append(beta_init(n, a, b))

    # random Fourier phase seeds
    for i in range(4):
        init_pool.append(random_phase_fft_init(n))

    # keep only a reasonable number to stay within the budget
    np.random.shuffle(init_pool)
    init_pool = init_pool[:12]

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05            # initial learning rate for Adam
    alpha_start = 4.0
    alpha_max = 200.0
    alpha_factor = 1.15       # geometric increase each 500 iterations

    # ------------------------------------------------------------------
    # Main optimisation loops (one per restart)
    # ------------------------------------------------------------------
    for restart_id, a0 in enumerate(init_pool):
        if time.time() >= deadline:
            break

        a = project_to_simplex(a0.astype(float))
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        lr = base_lr
        alpha = alpha_start
        iteration = 0

        while time.time() < deadline:
            iteration += 1

            # self‑convolution (needed for the weights)
            conv = self_conv_fft(a)

            # soft‑max weights for the smooth max
            w = smooth_weights(conv, alpha)

            # gradient of the smooth objective
            grad = gradient_from_weights(w, a)

            # Adam update (unconstrained) followed by projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_to_simplex(a)

            # periodic checks / parameter adjustments
            if iteration % 100 == 0:
                # evaluate the true (non‑smoothed) objective
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score, best_seq = cur_score, a.copy()
                    print(f"[restart {restart_id}] iter {iteration:5d} "
                          f"score = {best_score:.6f} (α={alpha:.1f})")

            if iteration % 500 == 0:
                alpha = min(alpha * alpha_factor, alpha_max)

            if iteration % 2000 == 0:
                lr *= 0.9

        # end of this restart
        print(f"[restart {restart_id}] finished after {iteration} iterations; "
              f"best so far = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded, fall back to uniform (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # Final reporting using the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print("\nSearch completed.")
    print(f"Best reported score = {final_score:.6f}")
    return best_seq.tolist()
