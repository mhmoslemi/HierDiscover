"""
Propose a step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

subject to a_i ≥ 0.  The algorithm expands on the previous
state‑of‑the‑art pipeline:

1.  Choose a problem size (n) – if no good fallback is supplied we
    use n = 4096 to give the optimiser more freedom.
2.  Scan a rich collection of analytic families (including several
    new ones) with fine‑grained parameter grids and a few random
    starts – keep the top 5 candidates.
3.  Run a short Exponentiated‑Gradient (mirror descent) optimisation
    from each of those seeds (the loop stops automatically on the
    global deadline).
4.  Refine the best survivor with a smooth‑max Adam optimiser that
    gradually increases the smoothing parameter β.
5.  Perform several LP‑based refinement steps (the classic
    “state‑of‑the‑art” iteration) – the mixing weight is slowly
    decayed.
6.  Finish with a simple simulated‑annealing hill‑climb.

All stages respect a common wall‑clock deadline (default 1000 s) and
return a plain Python list ready for ``evaluate_sequence``.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def arcsin_family(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = w / w.sum()
    return a


def semicircle_family(n: int) -> np.ndarray:
    # shape proportional to sqrt(1 - (2x-1)^2) on (0,1)
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    w = np.sqrt(np.maximum(0.0, 1.0 - (2 * x - 1) ** 2))
    a = w / w.sum()
    return a


def double_exponential_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * (i + 0.5)) + np.exp(-gamma * (n - i - 0.5))
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 500_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution
    objective.  Stops automatically when the global deadline is hit.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]          # diagonal term has weight 1

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)

        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of the max‑convolution using Adam
    on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for β in log‑space
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)

        maxc = conv.max()
        # smooth max via log‑sum‑exp trick
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of smooth max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # map to gradient on z (softmax Jacobian)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Very cheap projected sub‑gradient descent on the true max‑convolution.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a bit of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A such that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)      # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 12})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)

    return f_new

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimizes
    ``evaluate_sequence`` under the simplex constraint.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # 0.  Attempt to use a fallback (if the harness provides one)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem dimension
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # use a larger n if we have nothing strong yet
    n = fallback_len if fallback_len >= 1024 else 4096
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Analytic / random scan (fine‑grained)
    # ------------------------------------------------------------------
    top_k = 5
    top_candidates = []   # list of (score, a)

    def consider(vec: np.ndarray, label: str = ""):
        nonlocal best_score, best_seq, top_candidates
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new global best ({label}) = {best_score:.6f}")
        # keep a pool of the best few distinct candidates
        if len(top_candidates) < top_k:
            top_candidates.append((sc, vec.copy()))
        else:
            # replace worst if better
            worst_idx = max(range(len(top_candidates)),
                            key=lambda i: top_candidates[i][0])
            if sc < top_candidates[worst_idx][0]:
                top_candidates[worst_idx] = (sc, vec.copy())

    # Scan symmetric_product_family (alpha ∈ [0,5] step 0.05)
    alphas = np.arange(0.0, 5.0001, 0.05)
    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        consider(vec, f"symα={alpha:.2f}")

    # Scan left_heavy_power_family (beta ∈ [0,5] step 0.05)
    betas = np.arange(0.0, 5.0001, 0.05)
    for beta in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        consider(vec, f"lhβ={beta:.2f}")

    # Scan exp_u_family (gamma ∈ [0,15] step 0.2)
    gammas = np.arange(0.0, 15.0001, 0.2)
    for gamma in gammas:
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        consider(vec, f"expγ={gamma:.1f}")

    # Scan double exponential family (γ ∈ [0,10] step 0.2)
    for gamma in np.arange(0.0, 10.0001, 0.2):
        if time.time() >= deadline - 30.0:
            break
        vec = double_exponential_family(n, gamma)
        consider(vec, f"deγ={gamma:.1f}")

    # New families without extra parameters
    for vec, label in [(arcsin_family(n), "arcsin"),
                       (semicircle_family(n), "semicircle")]:
        consider(vec, label)

    # Random mixed families
    for _ in range(30):
        if time.time() >= deadline - 30.0:
            break
        lam = np.random.rand()
        gamma = np.random.uniform(0.0, 15.0)
        vec = mixed_uniform_u_family(n, lam, gamma)
        consider(vec, "mix")

    # Pure random starts
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        if np.random.rand() < 0.5:
            vec = random_monotone_start(n)
        else:
            vec = random_u_start(n)
        consider(vec, "rand")

    # ------------------------------------------------------------------
    # Normalise current best vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] score after analytic scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Mirror‑descent (EG) from the pool of good seeds
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] Starting EG from top candidates")
        for idx, (sc, seed_vec) in enumerate(sorted(top_candidates)[:3]):
            if time.time() >= deadline - 200.0:
                break
            print(f"   EG run {idx+1} (seed score {sc:.6f})")
            a_opt, sc_opt = eg_optimize(
                seed_vec,
                deadline - 150.0,
                max_iters=300_000,
                eta0=0.6,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                a = a_opt
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] EG run {idx+1} improved global best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam smooth‑max refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 180.0:
        print(f"[{time.time():.1f}] Adam smooth‑max refinement")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline - 120.0,
            max_iters=150_000,
            beta_start=4.0,
            beta_end=2500.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Cheap projected sub‑gradient descent (long tail)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] Sub‑gradient refinement")
        a_opt, sc_opt = subgrad_optimize(
            a,
            deadline - 60.0,
            max_iters=3_000_000,
            step0=0.07,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  LP‑refinement loop (more iterations, adaptive mixing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] LP refinement loop")
        mix_weight = 0.2
        lp_iters = 0
        while time.time() < deadline - 20.0 and lp_iters < 80:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-7:
                best_score = sc
                a = a_new
                print(f"[{time.time():.1f}] LP step {lp_iters} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.05, mix_weight * 0.85)
            lp_iters += 1

    # ------------------------------------------------------------------
    # 7.  Symmetrisation test (often beneficial)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final stochastic hill‑climb (annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final evaluation & return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
