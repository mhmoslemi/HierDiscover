```
def adam_l2_optimize(start_a, deadline, max_iters=200_000, lr=0.01):
    eps = 1e-12
    z = np.log(start_a + eps)
    m = np.zeros_like(z); v = np.zeros_like(z)
    beta1=0.9; beta2=0.999; eps_adam=1e-12
    best_a = softmax(z)
    best_score = compute_score(best_a)
    n = start_a.size
    target = 1.0/(2*n-1)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        a = softmax(z)
        conv = convolution_fft(a, a)
        # compute gradient of L2 loss: L = sum (conv_k - target)^2
        r = conv - target
        # gradient w.r.t. a: g = 4 * (conv of a with r) (valid mode)
        grad_a = 4 * np.convolve(a, r, mode='valid')
        # relative to L2: derivative of sum r^2 is 2 r * d conv/d a, but earlier we considered factor 4? Let's verify:
        L = sum_i r_i^2. dL/d a = 2* sum_i r_i * d conv_i / d a. And d conv_i/da_j = 2*a_{i-j} as earlier (if valid). So g_j = 2 * sum_i r_i * 2*a_{i-j} = 4 * sum_i r_i a_{i-j} = 4 * (a convolve r)_valid. So factor 4 is correct.
        grad_z = a * grad_a - np.dot(a, grad_a) * a  # backprop through softmax
        # Adam update
        m = beta1 * m + (1-beta1) * grad_z
        v = beta2 * v + (1-beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc; best_a = a_tmp.copy()
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin; best_a = a_fin.copy()
    return best_a, best_score
```

assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function (a probability vector) that
minimises  

\[
\text{score}(a)=\frac{2\,n\; \max (a*a)}{(\sum a)^2},
\]

where `*` denotes the aperiodic convolution.
The objective is invariant under scaling, therefore we restrict ourselves to
vectors with unit sum (*simplex*).

The optimiser consists of the following stages (all respect the global
time‑budget):

1. **Fallback / quick analytic scan** – a few analytically‑derived families
   (`symmetric_product_family`, `left_heavy_power_family`,
   `exp_u_family`) and many random Dirichlet starts are evaluated and the
   best one is kept.

2. **Adam on a smooth‑max surrogate** – we minimise the
   \(\frac{1}{\beta}\log\sum_k e^{\beta\, (a*a)_k}\) using Adam on the
   soft‑max parametrisation \(a=\operatorname{softmax}(z)\).  
   The temperature \(\beta\) grows from 4 up to 5000, allowing a very tight
   approximation of the true max.

3. **L₂‑flattening (Adam)** – the squared‑error
   \(L(a)=\sum_k\bigl((a*a)_k-1/(2n-1)\bigr)^2\) is minimised.  
   Its gradient can be computed by a single FFT‑based convolution,
   and Adam on the soft‑max parameters is used again.  
   Flattening the convolution drives the maximum towards the average value
   \(1/(2n-1)\).

4. **Final Adam smooth‑max refinement** – a short second pass of the smooth‑max
   optimiser cleans up any remaining peaks.

5. **Symmetrisation** – averaging the vector with its reversal never harms the
   score and often improves it.

6. **Aggressive simulated‑annealing hill‑climb** – the remaining budget is spent
   on a simple SA that performs swaps or mass‑transfer moves on the simplex.

All phases keep the best vector found so far and finally return it.
The code works for vectors up to a few‑thousand entries (default 4096); the
FFT‑based steps scale as *O(n log n)* and comfortably fit the 1000 s budget.

```python
"""
Propose a non‑negative step‑function (probability vector) that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.  The algorithm combines:
  * a quick analytic/random scan,
  * Adam optimisation of a smooth‑max surrogate,
  * Adam optimisation of an L2‑flattening surrogate,
  * a second short smooth‑max pass,
  * optional symmetrisation,
  * a final simulated‑annealing hill‑climb.
All phases respect the global time limit (default 1000 s) and never use
LPs (FFT‑based convolutions are O(n log n)).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # only required for fallback handling

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT (real -> real)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum(a)=1):
       score = 2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+0.5)^{-α}·(n‑0.5‑i)^{-α} (heavy at the edges)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+0.5)^{-β} (left‑heavy, mirror‑symmetrised later)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·dist(i, centre)) where dist is distance to the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def random_dirichlet_start(n: int) -> np.ndarray:
    """Random Dirichlet (exponential) start, sum to 1."""
    a = np.random.exponential(1.0, size=n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.012) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max surrogate
          (1/β)·log Σ_k exp(β·(a*a)_k)
    with Adam on the soft‑max parametrisation a = softmax(z).
    The temperature β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights over the convolution entries
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived from the smooth‑max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam‑smooth improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser for the L2‑flattening surrogate
# ----------------------------------------------------------------------


def adam_l2_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 250_000,
                     lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the L2 penalty
          L(a) = Σ_k ( (a*a)_k – 1/(2n‑1) )²
    with Adam on the soft‑max parametrisation.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    n = start_a.shape[0]
    target = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = softmax(z)
        conv = convolution_fft(a, a)          # full convolution, length 2n‑1
        r = conv - target                     # residual vector

        # gradient of L w.r.t. a:
        #   ∂L/∂a = 4 * (a_rev * r)_{valid}
        a_rev = a[::-1]
        grad_full = convolution_fft(a_rev, r)  # length 3n‑2
        grad_a = 4.0 * grad_full[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam‑L2 improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a linear temperature schedule.
    Moves are either swaps or a fractional mass transfer between two indices.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 5e-5   # small initial temperature
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # simple projection onto simplex (clip‑then‑renorm)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / n)

        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)   # linear cooling
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine respects the global wall‑clock limit (default 1000 s) and
    returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load fallback candidate if present
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Decide on working lengths (try a few, increasing if time permits)
    # ------------------------------------------------------------------
    base_n = len(best_seq) if best_seq is not None else 2048
    lengths = [max(1024, base_n)]
    if time.time() + 30 < deadline:
        lengths.append(min(4096, base_n * 2))
    if time.time() + 80 < deadline:
        lengths.append(min(8192, base_n * 4))

    # ------------------------------------------------------------------
    # 2. Iterate over chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        print(f"[{time.time():.1f}] ---- working length = {n} ----")
        # ---- quick analytic + random scan ---------------------------------
        candidates = []

        # analytic families (coarse grid)
        for alpha in np.linspace(0, 12, 7):
            candidates.append(symmetric_product_family(n, alpha))
        for beta in np.linspace(0, 10, 6):
            candidates.append(left_heavy_power_family(n, beta))
        for gamma in np.linspace(0, 6, 7):
            candidates.append(exp_u_family(n, gamma))

        # random Dirichlet starts
        for _ in range(80):
            candidates.append(random_dirichlet_start(n))

        # evaluate all candidates, keep the best
        for vec in candidates:
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (scan) = {best_score:.6f}")

        # ----- initialise working vector -----------------------------------
        a = np.array(best_seq, dtype=np.float64)
        if a.size != n:
            # simple linear interpolation to the new size
            old_n = a.size
            a = np.interp(
                np.linspace(0, old_n - 1, n),
                np.arange(old_n),
                a,
            )
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        best_seq = a.tolist()
        best_score = compute_score(a)

        # ------------------------------------------------------------------
        # 3. Adam smooth‑max optimisation (first pass)
        # ------------------------------------------------------------------
        if time.time() + 30 < deadline:
            phase_deadline = min(deadline - 5, time.time() + (deadline - time.time()) * 0.30)
            print(f"[{time.time():.1f}] Adam‑smooth first pass (n={n})")
            a, sc = adam_smooth_optimize(
                a,
                deadline=phase_deadline,
                max_iters=250_000,
                beta_start=4.0,
                beta_end=5000.0,
                lr=0.012,
            )
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] after Adam‑smooth1 best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4. Adam L2‑flattening optimisation
        # ------------------------------------------------------------------
        if time.time() + 30 < deadline:
            phase_deadline = min(deadline - 5, time.time() + (deadline - time.time()) * 0.30)
            print(f"[{time.time():.1f}] Adam‑L2 flattening (n={n})")
            a, sc = adam_l2_optimize(
                a,
                deadline=phase_deadline,
                max_iters=200_000,
                lr=0.015,
            )
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] after Adam‑L2 best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 5. Short second Adam smooth‑max pass (fine‑tuning)
        # ------------------------------------------------------------------
        if time.time() + 20 < deadline:
            phase_deadline = min(deadline - 5, time.time() + (deadline - time.time()) * 0.20)
            print(f"[{time.time():.1f}] Adam‑smooth second pass (n={n})")
            a, sc = adam_smooth_optimize(
                a,
                deadline=phase_deadline,
                max_iters=120_000,
                beta_start=4.0,
                beta_end=8000.0,
                lr=0.008,
            )
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] after Adam‑smooth2 best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 6. Symmetrisation (often beneficial)
        # ------------------------------------------------------------------
        a_sym = (a + a[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            a = a_sym
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 7. Final aggressive hill‑climb (simulated annealing)
        # ------------------------------------------------------------------
        if time.time() < deadline - 2.0:
            print(f"[{time.time():.1f}] final SA hill‑climb (n={n})")
            a, sc = hill_climb_sa(a, deadline, max_steps=4_000_000)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
```
