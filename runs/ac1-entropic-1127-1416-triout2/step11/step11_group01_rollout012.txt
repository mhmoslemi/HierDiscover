"""
Search for a non‑negative step‑function (list of heights) that minimises

        2·n·max(convolve(a,a)) / (sum(a)²)

The algorithm performs:
  1.   Load a fallback candidate if present.
  2.   Exhaustive exploration of several analytic families (including new
       U‑shaped, power‑law and sinusoidal families) on a moderately large
       length (default 4096).  The best shape found becomes the seed.
  3.   Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
       objective to escape coarse‑grid local minima.
  4.   Adam optimisation on a smooth soft‑max surrogate for fine‑tuning.
  5.   A final long stochastic hill‑climb (simulated‑annealing style) that
       respects the remaining wall‑clock budget.
The routine respects a global 1000 s wall‑clock limit and returns a plain
Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_conv = float(conv.max())
    return 2.0 * n * max_conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def power_dist_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped power law: w_i ∝ (dist+1)^(-alpha)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = (dist + 1.0) ** (-alpha)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def exp_pow_dist_family(n: int, gamma: float, p: float) -> np.ndarray:
    """U‑shaped exponential of a power of the distance."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i) + 1.0
    w = np.exp(-gamma * (dist ** p))
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def asymmetric_product_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """A_i ∝ (i+0.5)^(-alpha) * (n-0.5-i)^(-beta).  Allows asymmetric U‑shapes."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * np.log(i + 0.5) - beta * np.log(n - 0.5 - i))
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def sin_power_family(n: int, p: float) -> np.ndarray:
    """Sinusoidal power family: w_i ∝ sin(π*(i+0.5)/n)^p."""
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 0.5) / n)
    eps = 1e-12
    w = np.power(np.maximum(s, eps), p)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


def random_dirichlet_start(n: int, concentration: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(np.full(n, concentration))
    return a

# Mirror‑descent (exponentiated‑gradient) optimisation on true max
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry (max entry)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]  # diagonal term counted once

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Smoothed‑max optimisation (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 800.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (smooth max of convolution)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# LP‑based refinement (the “state‑of‑the‑art” iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A so that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)      # maximize Σ g  → minimize −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 20})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)

    return f_new

def propose_candidate(seed: int = 123456, budget_s: int = 1000, **kwargs):

    # ------------------------------------------------------------------
    # Normalise the working vector (ensure simplex)

    # ------------------------------------------------------------------
    # 4. Adam (smooth‑max) refinement

        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent (cheap many‑iteration phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] sub‑gradient phase start")
        a_opt, sc_opt = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=4_000_000,
            step0=0.07,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. LP‑based refinement loop
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] LP refinement loop start")
        mix_weight = 0.2
        lp_iters = 0
        while time.time() < deadline - 20.0:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-8:
                best_score = sc
                a = a_new
                print(f"[{time.time():.1f}] LP step {lp_iters} improved to {best_score:.6f}")
                # keep mix weight modestly large when we improve
                mix_weight = min(0.25, mix_weight * 1.05)
            else:
                # slowly decay the mixing coefficient if not improving
                mix_weight = max(0.05, mix_weight * 0.90)
            lp_iters += 1
            if lp_iters >= 40:   # safeguard against infinite loop
                break

    # ------------------------------------------------------------------
    # 7. Symmetrisation test (often reduces the max‑convolution)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        # allocate remaining time to SA
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_500_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Return the final sequence
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
