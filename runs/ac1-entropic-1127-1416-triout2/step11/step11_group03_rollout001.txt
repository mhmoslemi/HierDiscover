def exponential_mirror_descent(start_a, deadline, max_iters=500_000, eta0=0.1, sym_interval=500):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute subgradient g: for i in [0,n-1], index j = k-i
        n = a.shape[0]
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # step size:
        eta = eta0 / np.sqrt(t)
        w = a * np.exp(-eta * g)   # elementwise
        # projection onto simplex:
        a = project_to_simplex(w)
        # symmetrize occasionally:
        if sym_interval and (t % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
        # evaluate occasionally:
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror descent improved to {best_score:.6f}")
    # final evaluation:
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score

"""
Propose a step‑function sequence that minimizes the auto‑convolution peak.
The search combines several ideas:

1.  Scan a handful of analytic families (product‑power, power‑law,
    exponential‑U, beta, uniform+U mixture) on a medium size n≈4096.
2.  Refine the best product‑power parameter by golden‑section search.
3.  Perform a **mirror‑descent** on the exact max‑convolution objective
    using an exponentiated‑gradient (KL‑mirror) update.  This is a
    sub‑gradient descent on the simplex that often beats plain projected
    sub‑gradient descent.
4.  Run a few random restarts of the mirror descent to escape local minima.
5.  Finish with a short simulated‑annealing hill‑climb and an optional
    symmetrisation step.

All vectors are kept on the probability simplex (∑a=1, a_i≥0) and the
routine respects the 1000 s wall‑clock budget.  The final sequence is
returned as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a probability vector (∑a = 1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    The implementation follows the algorithm from
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        # uniform distribution is the projection of a non‑positive vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric product family:
        a_i ∝ [(i+½)(n‑½‑i)]^{‑α}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # guard against overflow for extreme α
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the grid: (i+½)^{p‑1}(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Random initialisers (fallback exploration)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start: ~5 % non‑zero entries, uniform mass on them."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sort a powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the true max‑objective
# ----------------------------------------------------------------------


def mirror_descent_max(start_a: np.ndarray,
                       deadline: float,
                       max_iters: int = 500_000,
                       eta0: float = 0.5,
                       sym_interval: int = 2500) -> Tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) descent on max(convolution).
    Parameters
    ----------
    start_a : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    deadline : float
        Unix timestamp after which the optimisation must stop.
    max_iters : int
        Upper bound on number of updates.
    eta0 : float
        Base step size (scaled as eta0 / sqrt(t)).
    sym_interval : int
        Frequency of explicit symmetrisation.
    Returns
    -------
    best_a, best_score
        Best vector encountered and its exact score.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    n = a.shape[0]
    eps = 1e-12

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full auto‑convolution
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        eta = eta0 / math.sqrt(t)

        # exponentiated‑gradient update (mirror descent)
        w = a * np.exp(-eta * g)

        # enforce strict positivity before projection (helps numerical stability)
        w = np.maximum(w, eps)

        # projection onto simplex
        a = project_to_simplex(w)

        # occasional symmetrisation
        if sym_interval and (t % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)

        # cheap true‑objective check
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror descent improved to {best_score:.6f}")

    # final evaluation
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
        print(f"[{time.time():.1f}] mirror descent final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (kept from previous version)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny temperature schedule.
    Moves:
      – 50 % swap two entries,
      – 50 % transfer a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section refinement of a single‑parameter family
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 12) -> Tuple[float, float]:
    """
    Minimises `func` on [a,b] using golden‑section search.
    Returns the best parameter and its function value.
    """
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:          # we are minimising
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

# Main driver

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence minimising ``evaluate_sequence``.
    Respects the wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback (previous best known)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working length
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 2. Coarse scan of analytic families
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan (n={n})")
    best_alpha = None

    # symmetric product family (α)
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")

    # left‑heavy power family (β)
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")

    # exponential U family (γ)
    for gamma in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")

    # beta family (p,q)
    for p in np.linspace(0.1, 3.0, 13):
        if time.time() >= deadline - 30.0:
            break
        for q in np.linspace(0.1, 3.0, 13):
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (beta p={p:.2f},q={q:.2f}) = {best_score:.6f}")

    # mixture uniform + U
    for lam in np.linspace(0.0, 1.0, 9):
        for gamma in np.linspace(0.0, 8.0, 33):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine golden‑section optimisation of the product family
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product‑α around {best_alpha:.3f}")

        def score_of_alpha(alpha_val: float) -> float:
            return compute_score(symmetric_product_family(n, alpha_val))

        lo = max(0.0, best_alpha - 0.6)
        hi = min(12.0, best_alpha + 0.6)
        refined_alpha, refined_score = golden_section_optimize(
            func=score_of_alpha,
            a=lo,
            b=hi,
            max_iter=14,
        )
        vec = symmetric_product_family(n, refined_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] refined product‑α={refined_alpha:.6f} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Mirror‑descent optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting mirror‑descent phase")
        # First run from the current best
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = mirror_descent_max(
            a_start,
            deadline - 80.0,
            max_iters=400_000,
            eta0=0.6,
            sym_interval=2000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] mirror descent (from best) improved to {best_score:.6f}")

        # Random restarts while time permits
        restarts = 0
        while time.time() < deadline - 80.0:
            remaining = deadline - time.time()
            # decide a modest iteration budget for each restart
            max_iters = int(min(200_000, remaining * 250))  # approx 250 iters per second
            if max_iters < 5_000:
                break

            # pick a random starter
            starter = random_sparse_start(n)
            a_opt_r, sc_opt_r = mirror_descent_max(
                starter,
                deadline - 60.0,
                max_iters=max_iters,
                eta0=0.6,
                sym_interval=2000,
            )
            restarts += 1
            if sc_opt_r < best_score:
                best_score = sc_opt_r
                best_seq = a_opt_r.tolist()
                print(f"[{time.time():.1f}] mirror descent restart {restarts} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final explicit symmetrisation test
    # ------------------------------------------------------------------
    if best_seq:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Short hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) phase")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=3_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Fallback to a simple uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if not best_seq:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 8. Final official evaluation (includes security checks)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
