    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    eps = 1e-8
    t = 0

    while time.time() < deadline:
        t += 1
        # schedule alpha linearly with iteration count or time
        progress = (t * 1.0) / 10000.0
        alpha = min(alpha_max, alpha0 + (alpha_max - alpha0) * progress)
        # compute gradient of smooth max
        _, grad = smooth_max_derivative(a, alpha)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = init_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # projection onto simplex
        a = project_to_simplex(np.maximum(a, 0.0))
        # evaluate actual objective occasionally
        if t % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a

def convolve_fft(x, y):
    size = len(x) + len(y) - 1
    n_fft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:size]
    return conv

def subgradient_polish(a0, deadline, step0=0.3):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            # convolution indices where a_i * a_{k - i}
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
    return best_a

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Candidate seeding** – generate a diverse pool of seed vectors using
   several parametric families (power‑law, exponential, beta, sine,
   triangular and Gaussian) together with a handful of random Dirichlet
   draws and an optional fallback from a previously discovered construction.

2. **Smooth‑max Adam optimisation** – minimise a smooth approximation of
   ``max(conv(a,a))`` using the log‑sum‑exp surrogate.  The gradient of the
   surrogate is obtained analytically via two FFT‑based convolutions.
   An Adam optimiser with a learning‑rate schedule and an ``α`` schedule
   (the temperature of the soft‑max) drives the vector towards a flatter
   autocorrelation while staying on the probability simplex.

3. **Hard‑max sub‑gradient polishing** – after the smooth phase the exact
   ``max`` is used and a projected sub‑gradient descent (the same idea as
   the previous state‑of‑the‑art) finishes the optimisation.

4. **Optional simulated‑annealing fine‑tuning** – a cheap Metropolis
   hill‑climb with mass‑transfer moves is performed if time permits.

All steps respect the global 1000 s wall‑clock budget.  The best sequence
found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of real vectors x and y using a real FFT."""
    size = len(x) + len(y) - 1
    n_fft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:size]
    return conv


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    return convolve_fft(a, a)


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum=1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for seeding
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return a / a.sum()


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a / a.sum()


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    return a / a.sum()


def generate_sine(n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1)), strictly positive on (0,n+1)."""
    a = np.sin(np.pi * np.arange(1, n + 1, dtype=float) / (n + 1))
    return a / a.sum()


def generate_triangle(n: int) -> np.ndarray:
    """Symmetric triangle: linearly rising then falling."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=float)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=float)
        right = left[:-1][::-1]
    a = np.concatenate([left, right])
    return a / a.sum()


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑μ)^2/(2σ^2)), centred at the middle."""
    mu = (n - 1) / 2.0
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def smooth_max_and_grad(a: np.ndarray, alpha: float):
    """
    Returns a tuple (smooth_max, gradient) where ``smooth_max`` equals
    log‑sum‑exp_{i}(alpha·conv_i)/alpha and ``gradient`` is the gradient of
    this smooth surrogate w.r.t. a.
    """
    conv = full_convolution(a)                     # length 2n‑1
    maxc = conv.max()
    # stable softmax weights
    exp_term = np.exp(alpha * (conv - maxc))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                         # shape (2n‑1,)

    # gradient := 2 * (w * a_rev)_{j}  (cross‑correlation)
    rev_a = a[::-1]
    conv_full = convolve_fft(w, rev_a)             # length 3n‑2
    n = a.shape[0]
    mid = n - 1                                    # (len(w)//2)
    grad = 2.0 * conv_full[mid:mid + n]

    smooth_val = maxc + np.log(sum_exp) / alpha
    return smooth_val, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, deadline: float,
                  init_lr: float = 0.05,
                  alpha0: float = 5.0,
                  alpha_max: float = 200.0) -> np.ndarray:
    """
    Adam optimiser that minimises the smooth‑max surrogate.
    Parameters
    ----------
    a0 : np.ndarray
        Starting probability vector (sum≈1, non‑negative).
    deadline : float
        Absolute time (Unix timestamp) after which the routine must stop.
    init_lr : float
        Base learning rate.
    alpha0, alpha_max : float
        Temperature schedule for the log‑sum‑exp: it starts at ``alpha0`` and
        linearly grows to ``alpha_max`` as time progresses.
    Returns
    -------
    np.ndarray
        The best vector found (projected onto the simplex).
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)

    start_time = time.time()
    total_budget = max(deadline - start_time, 1e-9)

    while time.time() < deadline:
        t += 1
        # linearly increase alpha with elapsed fraction
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / total_budget)
        alpha = alpha0 + (alpha_max - alpha0) * frac

        # smooth‑max value (unused) and gradient
        _, grad = smooth_max_and_grad(a, alpha)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update (descent)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (keeps sum=1, non‑negative)
        a = project_to_simplex(a)

        # occasional evaluation of the true objective
        if t % 400 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient polishing (identical to the earlier approach)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on the exact max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Light simulated annealing (optional final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Gather an initial pool of candidates
    # ------------------------------------------------------------------
    candidates = []

    # optional fallback from a previously discovered construction
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback = np.array(fallback, dtype=float)
        fallback = project_to_simplex(np.maximum(fallback, 0.0))
        candidates.append(fallback)
        print("[info] fallback vector added.")
    except Exception:
        pass

    # uniform distribution (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # parametric families (coarse sweeps, cheap)
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(generate_powerlaw(n, p))
    for lam in np.linspace(0.0, 0.2, 9):
        candidates.append(generate_exponential(n, lam))
    for a in [0.5, 1.0, 2.0]:
        for b in [0.5, 1.0, 2.0]:
            candidates.append(generate_beta(n, a, b))
    candidates.append(generate_sine(n))
    candidates.append(generate_triangle(n))
    for sigma in [n / 6.0, n / 4.0, n / 3.0]:
        candidates.append(generate_gaussian(n, sigma))

    # random Dirichlet draws with a few concentration values
    for conc in [0.1, 0.3, 0.7, 1.2]:
        for _ in range(5):
            a = np.random.dirichlet([conc] * n)
            candidates.append(a)

    # ------------------------------------------------------------------
    # 1. Quick scan of the pool (purely by the fast exact score)
    # ------------------------------------------------------------------
    for vec in candidates:
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
    print(f"[init] best initial exact score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the smooth surrogate (major work)
    # ------------------------------------------------------------------
    if best_seq is not None:
        now = time.time()
        remaining = deadline - now
        # allocate ~60 % of the remaining budget to the Adam phase
        adam_deadline = now + remaining * 0.60
        refined = adam_optimize(best_seq, adam_deadline)
        refined_score = compute_score(refined)
        print(f"[adam] refined score = {refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()

    # ------------------------------------------------------------------
    # 3. Sub‑gradient polishing on the true max (fine‑tune)
    # ------------------------------------------------------------------
    if best_seq is not None:
        now = time.time()
        remaining = deadline - now
        polish_deadline = now + remaining * 0.25  # ~25 % of the leftover time
        polished = subgradient_polish(best_seq, polish_deadline)
        polished_score = compute_score(polished)
        print(f"[polish] polished score = {polished_score:.6f}")
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished.copy()

    # ------------------------------------------------------------------
    # 4. Optional simulated annealing for a final push
    # ------------------------------------------------------------------
    if best_seq is not None:
        now = time.time()
        remaining = deadline - now
        if remaining > 2.0:  # tiny safeguard
            anneal_deadline = now + remaining * 0.10  # use the rest
            final_seq = simulated_annealing(best_seq, anneal_deadline)
            final_score = compute_score(final_seq)
            print(f"[anneal] final annealed score = {final_score:.6f}")
            if final_score < best_score:
                best_score = final_score
                best_seq = final_seq.copy()

    # ------------------------------------------------------------------
    # 5. Fallback to a uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_score(best_seq)
        print("[fallback] uniform vector used as last resort.")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official function and return
    # ------------------------------------------------------------------
    final_report = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Reported score = {final_report:.6f}")
    return best_seq.tolist()
