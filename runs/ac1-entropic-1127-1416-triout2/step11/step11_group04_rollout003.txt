    n = len(f)
    m_len = 2*n -1
    # build A matrix using f values.
    A = np.zeros((m_len, n), dtype=np.float64)

    c = -np.ones(n)
    bounds = [(0, None)]*n

"""
Multi‑phase heuristic for minimizing

        2·n·max(conv(a,a)) / (sum(a)**2)

under a non‑negative simplex constraint.

The algorithm works with any sequence length up to ~3000.  It consists of
   1) a fast random / analytic scan for a good starter,
   2) Adam optimisation of a smooth max‑surrogate (β grows to 5 000),
   3) a projected sub‑gradient descent on the true max‑objective,
   4) several LP‑refinement steps; the LP solves
          max Σ g_j   s.t.   (f*g)_k ≤ M,
      and the resulting g is mixed with f using an exact line‑search
      (the max‑convolution of the mixture is a quadratic function of the
      mixing weight, which we minimise by golden‑section search),
   5) a long simulated‑annealing hill‑climb with swap and mass‑transfer
      moves.
All phases respect a global wall‑clock budget (default 1000 s) and the
best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5_000.0,
                         lr: float = 0.012) -> tuple[np.ndarray, float]:
    """Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
       using Adam on the soft‑max parametrisation a = softmax(z)."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t a (derived from smooth‑max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 800 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap iterations)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.12) -> tuple[np.ndarray, float]:
    """Sub‑gradient method for the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (shift k)
        g = np.zeros_like(a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] = a[j[mask]] + a[idx[mask]]
        diag = (j == idx) & mask
        g[diag] = 2.0 * a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑refinement with exact line‑search on the mixing weight
# ----------------------------------------------------------------------


def optimal_mix(f: np.ndarray, g: np.ndarray) -> tuple[np.ndarray, float]:
    """
    For two probability vectors f,g compute the convex mixture
    (1‑t)f + t g that minimises max(conv) using a golden‑section search.
    Returns the mixture and its score.
    """
    n = f.shape[0]
    conv_f = convolution_fft(f, f)
    conv_g = convolution_fft(g, g)
    conv_fg = convolution_fft(f, g)

    A = conv_f + conv_g - 2.0 * conv_fg      # quadratic coeff.
    B = 2.0 * (conv_fg - conv_f)             # linear coeff.
    C = conv_f                               # constant term.

    def max_at(t):
        return np.max(A * t * t + B * t + C)

    # golden‑section search on [0,1]
    lo, hi = 0.0, 1.0
    phi = (math.sqrt(5) - 1) / 2  # 0.618...
    for _ in range(30):
        t1 = lo + (1 - phi) * (hi - lo)
        t2 = lo + phi * (hi - lo)
        if max_at(t1) < max_at(t2):
            hi = t2
        else:
            lo = t1
    t_opt = 0.5 * (lo + hi)
    mix = (1.0 - t_opt) * f + t_opt * g
    mix = project_to_simplex(mix)
    score = compute_score(mix)
    return mix, score


def lp_refine_step_opt(f: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M   (all k),   g ≥ 0,
    then mix f with the normalised solution using the optimal mixing weight.
    """
    n = f.shape[0]
    m_len = 2 * n - 1

    # Build convolution matrix A where A @ g = f * g (full convolution)
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ↔ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m_len, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return f, compute_score(f)

    g = np.maximum(res.x, 0.0)
    if g.sum() <= 0.0:
        return f, compute_score(f)

    g_norm = g / g.sum()
    return optimal_mix(f, g_norm)


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_500_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a linearly cooled temperature."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 6e-5
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The function respects the global time budget and returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Load fallback (if present)
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Choose working length (do not exceed 3000)
    # --------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048
    n = min(max(n, 1024), 3000)

    # --------------------------------------------------------------
    # 2.  Quick random / analytic scan (time‑bounded)
    # --------------------------------------------------------------
    scan_deadline = min(deadline - 30.0, start_time + 120.0)  # at most ~2 min
    families = ['sym', 'left', 'exp', 'mono', 'rand']
    while time.time() < scan_deadline:
        kind = np.random.choice(families)
        if kind == 'sym':
            alpha = np.random.uniform(0.0, 15.0)
            vec = symmetric_product_family(n, alpha)
        elif kind == 'left':
            beta = np.random.uniform(0.0, 15.0)
            vec = left_heavy_power_family(n, beta)
        elif kind == 'exp':
            gamma = np.random.uniform(0.0, 6.5)
            vec = exp_u_family(n, gamma)
        elif kind == 'mono':
            vec = random_monotone_start(n)
        else:  # 'rand'
            vec = random_u_start(n)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (scan) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Normalise starter to a probability vector
    # --------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] starter score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Adam optimisation (smooth‑max) – up to ~30 % of remaining time
    # --------------------------------------------------------------
    if time.time() + 70.0 < deadline:
        print(f"[{time.time():.1f}] Adam optimisation start")
        a, sc = adam_smooth_optimize(
            a,
            deadline - 10.0,
            max_iters=500_000,
            beta_start=4.0,
            beta_end=5_000.0,
            lr=0.012,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Heavy sub‑gradient descent – bulk of remaining budget
    # --------------------------------------------------------------
    if time.time() + 140.0 < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient descent start")
        a, sc = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=3_000_000,
            step0=0.12,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  LP‑based refinement with exact line‑search – many iterations
    # --------------------------------------------------------------
    if time.time() + 40.0 < deadline:
        print(f"[{time.time():.1f}] LP refinement loop start")
        lp_iters = 20
        for it in range(lp_iters):
            if time.time() >= deadline - 5.0:
                break
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new, sc = lp_refine_step_opt(a, M)
            if sc < best_score - 1e-9:
                best_score = sc
                a = a_new
                print(f"[{time.time():.1f}] LP iter {it} improved to {best_score:.6f}")
            else:
                # if stuck, tighten the bound a little by shrinking M
                # (not strictly necessary but can help escape plateaus)
                pass

    # --------------------------------------------------------------
    # 7.  Symmetrisation (often beneficial)
    # --------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 8.  Final aggressive hill‑climb (use all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a, sc = hill_climb_sa(a, deadline, max_steps=5_000_000)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 9.  Final official evaluation and return
    # --------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
