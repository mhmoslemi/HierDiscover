def anti_symmetrify(v):
    # v: np.ndarray
    # Returns a new arrangement with large values in left half reversed, small values right.
    sorted_vals = np.sort(v)  # ascending
    mid = len(v) // 2
    left = sorted_vals[mid:][::-1]  # larger half reversed
    right = sorted_vals[:mid]
    return np.concatenate([left, right])

1. **Exploratory scan** – analytic families, fine‑grained parameter sweeps
   and many random Dirichlet draws.  Every candidate is also tried in an
   *anti‑symmetric* ordering (large values on the left, small on the
   right) which strongly reduces the centre autocorrelation.
2. **Local refinement** – a long simulated‑annealing (SA) hill‑climb on
   the simplex.  The SA runs until the global wall‑clock budget is
   exhausted, always projecting back onto the simplex.
3. **Return** – the best sequence found is evaluated with the official
   ``evaluate_sequence`` and returned as a plain Python list.

All helper functions are top‑level; no ``lambda`` expressions or closures
are used.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # kept for completeness; not used here

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Anti‑symmetric reordering
# ----------------------------------------------------------------------


def anti_symmetrify(v: np.ndarray) -> np.ndarray:
    """
    Return a permutation of v where the large values occupy the left
    half (in decreasing order) and the small values the right half
    (in increasing order).  This arrangement minimises the centre
    autocorrelation term (a·reverse(a)).
    """
    n = v.shape[0]
    sorted_vals = np.sort(v)          # ascending
    mid = n // 2
    left = sorted_vals[mid:][::-1]    # larger half, reversed → decreasing
    right = sorted_vals[:mid]         # smaller half → increasing
    return np.concatenate((left, right))


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    The routine stops when ``deadline`` is reached.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` under the simplex constraint.
    The function respects the given wall‑clock budget and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Determine problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = fallback_len if fallback_len >= 1024 else 4096
    print(f"[{time.time():.1f}] using length n={n}")

    # ------------------------------------------------------------------
    # 2.  Quick analytic and random scan (including anti‑symmetrisation)
    # ------------------------------------------------------------------
    def try_candidate(vec: np.ndarray):
        nonlocal best_seq, best_score
        # safeguard – normalise to the probability simplex
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] NEW BEST = {best_score:.6f}")

    # ----- analytic families (coarse grid) -----
    alphas = np.linspace(0.0, 30.0, 61)       # step 0.5
    betas = np.linspace(0.0, 30.0, 61)
    gammas = np.linspace(0.0, 15.0, 61)

    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        try_candidate(vec)
        try_candidate(anti_symmetrify(vec))

    for beta in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        try_candidate(vec)
        try_candidate(anti_symmetrify(vec))

    for gamma in gammas:
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        try_candidate(vec)
        try_candidate(anti_symmetrify(vec))

    # ----- mixed uniform‑U family (random parameters) -----
    for _ in range(30):
        if time.time() >= deadline - 30.0:
            break
        lam = np.random.rand()
        gamma = np.random.uniform(0.0, 15.0)
        vec = mixed_uniform_u_family(n, lam, gamma)
        try_candidate(vec)
        try_candidate(anti_symmetrify(vec))

    # ----- random Dirichlet draws (with and without reordering) -----
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        # concentration drawn log‑uniformly between 0.01 and 10
        log_alpha = np.random.uniform(-2.0, 1.0)
        alpha = 10.0 ** log_alpha
        vec = np.random.dirichlet(np.full(n, alpha))
        try_candidate(vec)
        try_candidate(anti_symmetrify(vec))

    # ----- random monotone / U‑shape generators -----
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        if np.random.rand() < 0.5:
            vec = random_monotone_start(n)
        else:
            vec = random_u_start(n)
        try_candidate(vec)
        try_candidate(anti_symmetrify(vec))

    # ------------------------------------------------------------------
    # 3.  Local refinement with long SA hill‑climb
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = a / a.sum()

    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] starting long SA hill‑climb")
        # allow a very large number of steps; the deadline will stop us
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=10_000_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
