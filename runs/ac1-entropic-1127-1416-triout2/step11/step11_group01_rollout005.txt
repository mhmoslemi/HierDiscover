def solve_lp_get_g(f, M, time_limit=10.0):
    n = f.shape[0]; m = 2*n-1
    A = np.zeros((m,n), dtype=np.float64)
    for i in range(n):
        A[i:i+n,i] = f[i]

    c = -np.ones(n)
    bounds = [(0.0, None)]*n
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    return g

"""
Propose a step‑function (list of non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The implementation improves upon the reference by

* extending the analytic families (very large α/β for edge‑heavy shapes,
  two‑parameter beta‑like families, and many random draws),
* using a more aggressive LP‑based refinement that solves a linear program
  at every iteration and performs a small line‑search over the mixing weight,
* running many LP iterations (instead of a single fixed‑weight step),
* keeping a simple simulated‑annealing hill‑climb as a final polish,
* careful time‑budget handling (wall‑clock budget ≈ 1000 s).

All helper functions are top‑level; the only required external import is
`evaluate_sequence`, which is supplied by the competition harness.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """Edge‑heavy family a_i ∝ ((i+½)(n‑½‑i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+½)^{-β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def beta_shape_family(n: int, a_param: float, b_param: float) -> np.ndarray:
    """
    Two‑parameter Beta‑like shape:
        a_i ∝ (i+½)^{a‑1}·(n‑½‑i)^{b‑1}.
    With a,b∈(0,∞). Values <1 concentrate mass at the ends.
    """
    i = np.arange(n, dtype=np.float64)
    log_term = (a_param - 1.0) * np.log(i + 0.5) + (b_param - 1.0) * np.log(n - 0.5 - i)
    w = np.exp(log_term)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape a_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mix of uniform and a U‑shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (power‑law style)."""
    exponent = np.random.uniform(0.5, 12.0)
    r = np.random.rand(n) ** exponent
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shape."""
    gamma = np.random.uniform(0.0, 20.0)
    return exp_u_family(n, gamma)


def random_beta_start(n: int) -> np.ndarray:
    """Random Beta‑shape."""
    a = np.random.uniform(0.1, 5.0)
    b = np.random.uniform(0.1, 5.0)
    return beta_shape_family(n, a, b)


# ----------------------------------------------------------------------
# Mirror‑descent (Exponentiated‑gradient) – optional short boost
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 200_000,
                eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the exact max‑convolution objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)

        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def solve_lp_get_g(f: np.ndarray, M: float, time_limit: float = 8.0) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    g ≥ 0.
    Returns the optimal g (not normalised) or None on failure.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (size m×n) where (A·g)_k = (f*g)_k
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': time_limit})

    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    return g


def lp_refine(a: np.ndarray,
              deadline: float,
              max_outer_iters: int = 150,
              no_improve_limit: int = 20) -> tuple[np.ndarray, float]:
    """
    Aggressive LP‑based improvement loop.
    At each iteration we solve the LP for the current max‑convolution M,
    then try several mixing weights t∈[0,0.5] to minimise the objective.
    """
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_outer_iters and no_improve < no_improve_limit:
        conv = convolution_fft(best_a, best_a)
        M = float(conv.max())

        g = solve_lp_get_g(best_a, M, time_limit=6.0)
        if g is None:
            print(f"[{time.time():.1f}] LP solve failed – terminating refinement")
            break
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g

        # try a handful of mixing weights
        improved = False
        for t in (0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.40):
            a_new = (1.0 - t) * best_a + t * g_norm
            a_new = project_to_simplex(a_new)
            sc = compute_score(a_new)
            if sc < best_score - 1e-9:
                best_score, best_a = sc, a_new.copy()
                improved = True
                print(f"[{time.time():.1f}] LP iter {it} t={t:.2f} → {best_score:.6f}")

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Simple SA on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a good step‑function under the given time budget.
    Returns a plain Python list ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from the harness
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Determine problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = fallback_len if fallback_len >= 1024 else 4096
    print(f"[{time.time():.1f}] using length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Broad analytic / random scan (extended ranges)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] starting extended analytic scan")
        # symmetric product family – huge α range
        for alpha in np.linspace(0.0, 120.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # left–heavy power family – extended β range
        for beta in np.linspace(0.0, 120.0, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        # beta‑shape (two‑parameter) sweep
        a_vals = np.linspace(0.1, 5.0, 9)
        b_vals = np.linspace(0.1, 5.0, 9)
        for a_param in a_vals:
            for b_param in b_vals:
                if time.time() >= deadline - 30.0:
                    break
                vec = beta_shape_family(n, a_param, b_param)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score, best_seq = sc, vec.tolist()
                    print(f"[{time.time():.1f}] new best (beta a={a_param:.2f},b={b_param:.2f}) = {best_score:.6f}")

        # mixed uniform / U‑shape
        for _ in range(20):
            if time.time() >= deadline - 30.0:
                break
            lam = np.random.rand()
            gamma = np.random.uniform(0.0, 20.0)
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (mix) = {best_score:.6f}")

        # random monotone and U‑shape draws
        for _ in range(200):
            if time.time() >= deadline - 30.0:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

        # random beta‑shape draws
        for _ in range(100):
            if time.time() >= deadline - 30.0:
                break
            vec = random_beta_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (rand‑beta) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise the best candidate found so far
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Short EG boost (optional, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] brief EG boost")
        a, sc = eg_optimize(a,
                            deadline - 150.0,
                            max_iters=150_000,
                            eta0=0.5)
        if sc < best_score:
            best_score, a = sc, a.copy()
            print(f"[{time.time():.1f}] after EG boost best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Aggressive LP refinement loop
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] LP refinement loop")
        a, best_score = lp_refine(a, deadline - 5.0,
                                  max_outer_iters=200,
                                  no_improve_limit=30)
        print(f"[{time.time():.1f}] LP loop finished with best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Symmetrisation test (helps for many shapes)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score, a = sc_sym, a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final lightweight SA hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA hill‑climb")
        a, sc = hill_climb_sa(a, deadline, max_steps=300_000)
        if sc < best_score:
            best_score, a = sc, a.copy()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Return result
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] search finished – final reported score = {final_score:.6f}")
    return final_seq
