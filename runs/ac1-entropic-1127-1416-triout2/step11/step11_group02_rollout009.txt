"""
Step‑function optimiser (2026 edition).

The optimiser follows a multi‑phase pipeline:

0. **Fallback** – if a previously discovered good vector (`height_sequence_1`)
   is available we start from it.

1. **Global sweep** – cheap exploration of simple parametric families
   (power‑law, geometric, plateau+tail) to obtain a reasonable seed.

2. **Convex‑LP refinement** – conditional‑gradient (Frank‑Wolfe) step
   (optional, uses cvxpy if present).

3. **Smooth‑max Adam optimisation** – a fast first‑order method that
   minimises a smooth approximation of the objective
   `max_k (a*a)[k]`.  The surrogate uses a soft‑max with an
   annealing temperature, giving a gradient that can be computed via a
   single FFT per iteration.  Adam with a decaying learning‑rate and
   projection onto the probability simplex drives the iterate toward a
   lower maximum convolution.

4. **Projected sub‑gradient polishing** – short clean‑up using the
   classic non‑smooth sub‑gradient method.

5. **Stochastic hill‑climb** – inexpensive greedy mass‑transfer moves
   to squeeze out the last remaining improvement.

All phases respect the overall 1000 s wall‑clock budget.  The returned
list consists of non‑negative floats; `evaluate_sequence` will see the
same objective as the internal `compute_score` (the sum is kept equal to 1).

The key new component is phase 3, which replaces the previous sub‑gradient
only approach by a smoothed‑max Adam optimiser; this typically yields a
few‐percent improvement on the best known bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional CVXPY – used only for the Frank‑Wolfe LP step
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAVE_CVX = True
except Exception:   # pragma: no cover
    _HAVE_CVX = False

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Phase 1 – global parametric sweep

def best_initial_vector(n: int, time_deadline: float) -> np.ndarray:
    """Search simple families (power‑law, geometric, plateau+tail) and return the best."""
    best_a = None
    best_val = float('inf')

    # ----- power‑law sweep -----
    for p in np.linspace(0.30, 1.00, 71):
        if time.time() >= time_deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ----- geometric (exponential) decay -----
    for r in np.linspace(0.50, 0.99, 81):
        if time.time() >= time_deadline:
            break
        a = (r ** np.arange(n, dtype=np.float64))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ----- mixed (plateau + power‑law tail) -----
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= time_deadline:
            break
        L = int(L_frac * n)
        if L == 0 or L >= n:
            continue
        for p in np.linspace(0.4, 0.8, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head = np.full(L, tail.max())
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – Frank‑Wolfe (LP) refinement (optional)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the matrix A such that A @ g = conv(f, g) (full linear convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   maximise Σ g_i   s.t.  conv(f,g) ≤ M ,  g ≥ 0."""
    if not _HAVE_CVX:
        return None
    n = f.shape[0]
    A = _build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=2000, abstol=1e-9, reltol=1e-9)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.asarray(g.value).flatten()


def frank_wolfe_refine(f0: np.ndarray, time_deadline: float,
                       max_iter: int = 200) -> np.ndarray:
    """Conditional‑gradient improvement of a probability vector."""
    f = f0.copy()
    best_f = f.copy()
    best_M = full_convolution(f).max()
    it = 0
    t_candidates = np.array([0.02, 0.05, 0.10, 0.15, 0.20, 0.25, 0.30])

    while time.time() < time_deadline and it < max_iter:
        it += 1
        cur_M = full_convolution(f).max()

        g = _solve_lp_for_g(f, cur_M)
        if g is None or g.sum() <= 0:
            break
        g_hat = g / g.sum()

        best_t = 0.0
        best_new_M = cur_M
        for t in t_candidates:
            f_candidate = (1.0 - t) * f + t * g_hat
            M_candidate = full_convolution(f_candidate).max()
            if M_candidate < best_new_M:
                best_new_M = M_candidate
                best_t = t

        if best_t == 0.0:
            break

        f = (1.0 - best_t) * f + best_t * g_hat
        cur_M = best_new_M

        if cur_M < best_M:
            best_M = cur_M
            best_f = f.copy()

    return best_f


# ----------------------------------------------------------------------
# Phase 3 – Smooth‑max Adam optimisation
# ----------------------------------------------------------------------
def gradient_smooth_max(a: np.ndarray, alpha: float):
    """
    Returns the gradient of the soft‑max approximation of max(conv(a,a))
    together with the actual convolution and the weight vector.
    """
    n = a.shape[0]
    conv = full_convolution(a)                     # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    exp_term = np.exp(alpha * (conv - max_c))
    w = exp_term / exp_term.sum()                  # length 2n‑1

    # gradient = 2 * (a ⋆ w) where ⋆ denotes cross‑correlation
    # This can be obtained via a convolution with the reversed weight vector.
    w_rev = w[::-1]
    L = conv.shape[0]                              # = 2n‑1
    size = 1 << ((n + L - 1 - 1).bit_length())    # next power‑of‑2 ≥ n+L‑1
    A = np.fft.rfft(a, n=size)
    W_rev = np.fft.rfft(w_rev, n=size)
    conv_rev = np.fft.irfft(A * W_rev, n=size)     # length ≥ n+L‑1
    # slice the relevant part: indices n‑1 … 2n‑2 (inclusive), then reverse
    segment = conv_rev[n - 1: 2 * n - 1]           # length n
    grad = 2.0 * segment[::-1]
    return grad, conv, w


def adam_optimize(a0: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Adam‑based optimisation of the smooth max objective.
    Returns the best vector seen (still Σ = 1, a_i ≥ 0).
    """
    n = a0.shape[0]
    a = a0.copy()
    conv = full_convolution(a)
    best_a = a.copy()
    best_val = 2.0 * n * conv.max()

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.5
    t = 0
    start = time.time()

    while time.time() < time_deadline:
        t += 1
        elapsed = time.time() - start
        total = time_deadline - start
        # linear annealing of the soft‑max temperature
        alpha = 5.0 + 195.0 * min(1.0, elapsed / total)

        grad, conv, w = gradient_smooth_max(a, alpha)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = step0 / np.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasibility
        a = project_to_simplex(a)
        # optional symmetry – helps the landscape
        a = 0.5 * (a + a[::-1])

        cur_val = 2.0 * n * conv.max()   # value before the update (conv is from previous iter)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] iteration {t:5d} new best {best_val:.6f}")

        # compute the new convolution for the next loop (cheap – reuse next grad)
        # (the next iteration will recompute it, so we skip an extra call here)

    return best_a

def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

def hill_climb(a_start: np.ndarray, time_deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Greedy stochastic moves with acceptance when the max‑convolution improves."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a
