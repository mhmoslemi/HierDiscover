"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Broad exploration** – generate candidates from a wide variety of
   parametric families (power‑law, exponential, beta, Gaussian,
   sine‑power, etc.) and a large pool of random Dirichlet draws.
   The best few are kept.

2. **Soft‑max gradient descent** – starting from each kept candidate,
   minimise a smooth approximation of the max‑convolution
   ``f(a)=τ·log Σ exp(conv(a)/τ)``.  The gradient can be obtained by a
   single FFT‑based convolution, the parameters τ are annealed
   automatically, and after each step the vector is projected onto the
   probability simplex.

3. **Projected sub‑gradient polishing** – a classic sub‑gradient
   optimiser (active‑set gradient) is run for a short time to fine‑tune
   the solution.

4. **Simulated annealing** – a stochastic hill‑climb with temperature
   decay provides a final escape from shallow local minima.

All phases respect the 1000 s wall‑clock budget; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑mid)²/(2σ²))."""
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=float) - mid
    a = np.exp(-0.5 * (x / sigma) ** 2)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    angles = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(angles) ** p
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global search (baseline)
# ----------------------------------------------------------------------


def baseline_search(
    n: int, deadline: float, max_keep: int = 8, random_draws: int = 2000
):
    """Explore many families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []  # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) Power‑law sweep
    for p in np.linspace(0.1, 2.0, 101):
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) Exponential sweep
    for lam in np.linspace(0.0001, 0.2, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep (dense grid)
    al_vals = np.linspace(0.2, 4.0, 31)
    be_vals = np.linspace(0.2, 4.0, 31)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.5 * n, 25):
        if time.time() >= deadline:
            break
        try_candidate(generate_gaussian(n, sigma))

    # 5) Sine‑power sweep
    for p in np.linspace(0.1, 6.0, 31):
        if time.time() >= deadline:
            break
        try_candidate(generate_sine_power(n, p))

    # 6) Random Dirichlet draws – many concentrations
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for conc in concs:
        for _ in range(random_draws // len(concs)):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def smooth_max_optim(a0: np.ndarray, deadline: float,
                     tau_start: float = 0.30,
                     tau_end: float = 1e-5,
                     tau_decay: float = 0.98,
                     step0: float = 0.20) -> np.ndarray:
    """Minimise a smooth approximation of max(conv) using an
    FFT‑based gradient and Euclidean projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    tau = tau_start
    while time.time() < deadline:
        # anneal temperature for the soft‑max
        if t % 100 == 0 and tau > tau_end:
            tau = max(tau_end, tau * tau_decay)

        # convolution a*a
        conv = full_convolution(a)            # length m
        # soft‑max weights
        shift = conv.max()
        exp_vals = np.exp((conv - shift) / tau)
        w = exp_vals / exp_vals.sum()         # length m

        # gradient: 2 * (a ⋆ w)   (cross‑correlation)
        # we compute via FFT: g = 2 * conv_full[(m‑1)‑i]
        rev_w = w[::-1]
        size = 1 << ((n + rev_w.shape[0] - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(rev_w, n=size)
        conv_full = np.fft.irfft(A * B, n=size)[: n + rev_w.shape[0] - 1]

        # extract the gradient for each coordinate
        idx = (m - 1) - np.arange(n)          # indices in conv_full
        grad = 2.0 * conv_full[idx]

        # step size (decreasing)
        step = step0 / np.sqrt(t + 1.0)
        a = a - step * grad
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 1500 == 0 and t > 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t + 1.0))
            a = a + jitter
            a = project_to_simplex(a)

        t += 1

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (as in the original code)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient from all active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        T = max(final_temp, T * cooling_factor)

        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    n = 2048
    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Global baseline exploration (≈ 250 s)
    # -----------------------------------------------------------------
    baseline_deadline = deadline - 300.0
    if time.time() < baseline_deadline:
        candidates = baseline_search(
            n, baseline_deadline, max_keep=8, random_draws=2000
        )
        for vec in candidates:
            val = compute_score(vec)
            if val < best_val:
                best_val, best_seq = val, vec.tolist()
        print(f"[baseline] best known score = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Soft‑max gradient descent from the best candidates
    # -----------------------------------------------------------------
    if best_seq is not None:
        # allocate about 350 s for this phase
        grad_deadline = min(deadline - 150.0,
                            start_time + 250.0 + (deadline - start_time) * 0.35)
        # Running the optimiser on a few top candidates
        top_vectors = []
        # re‑project the current best to guarantee feasibility
        best_vec = np.array(best_seq, dtype=float)
        best_vec = project_to_simplex(best_vec)
        top_vectors.append(best_vec)

        # we also try a uniform vector and a few random draws
        top_vectors.append(np.full(n, 1.0 / n))
        for _ in range(3):
            top_vectors.append(project_to_simplex(np.random.rand(n)))

        refined_best = best_vec.copy()
        refined_val = best_val

        for idx, start_vec in enumerate(top_vectors):
            if time.time() >= grad_deadline:
                break
            local_deadline = grad_deadline - (time.time() - start_time) - 1.0
            if local_deadline <= time.time():
                break
            a_ref = smooth_max_optim(start_vec, local_deadline)
            val = compute_score(a_ref)
            print(
                f"[smooth‑opt {idx}] refined score = {val:.6f}"
            )
            if val < refined_val:
                refined_val, refined_best = val, a_ref.copy()

        best_val, best_seq = refined_val, refined_best.tolist()
        print(f"[smooth‑opt] best after smooth optimisation = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (focus on active constraints)
    # -----------------------------------------------------------------
    polish_deadline = deadline - 120.0
    if best_seq is not None and time.time() < polish_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, polish_deadline)
        val = compute_score(a_refined)
        print(f"[polish] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Simulated annealing / stochastic hill‑climb
    # -----------------------------------------------------------------
    anneal_deadline = deadline - 5.0
    if best_seq is not None and time.time() < anneal_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = simulated_annealing(a_start, anneal_deadline)
        val = compute_score(a_final)
        print(f"[anneal] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report using the official evaluation function
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
