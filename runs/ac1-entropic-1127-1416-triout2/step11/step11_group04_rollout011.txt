"""
Propose a non‑negative step sequence that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

under the simplex constraint ∑a = 1.
The algorithm consists of three stages:

1. **Fast analytic start** – a one‑parameter symmetric “product” family
   a_i ∝ ((i+½)(n‑½‑i))^{-α}.  The optimal α is found by a cheap
   golden‑section search (≈30 s).

2. **Greedy max‑flattening** – while time permits we repeatedly move a
   small amount of mass from the index that most influences the current
   maximal convolution entry to a location that influences it the least.
   The convolution is updated *incrementally* (O(n) per step) so that
   many tiny improving moves can be performed in the remaining budget.

3. **Optional symmetrisation** – averaging the vector with its reverse
   (and projecting back onto the simplex) sometimes reduces the peak
   further.

The code respects the global 1000 s wall‑clock limit and returns a plain
list of floats ready for ``evaluate_sequence``.  No external files or
network access are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def score_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective directly from a convolution array (∑a = 1)."""
    return float(2.0 * n * float(conv.max()))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic family – symmetric product (edge‑heavy) shape
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{-α}, then normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    # avoid overflow/underflow: work in log‑space
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # w = exp(-α * logprod)
    w = np.exp(-alpha * logprod)
    # protect against all zeros (should not happen for reasonable α)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    a = w / w.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------

            conv_cand[i_src:i_src + n] -= 2.0 * delta * a
            # quadratic correction (exact)
            conv_cand[2 * i_src] += delta * delta
            conv_cand[2 * j_tgt] += delta * delta
            conv_cand[i_src + j_tgt] -= 2.0 * delta * delta

            new_M = conv_cand.max()
            if new_M < M - 1e-12:               # strict improvement
                # accept move
                a[i_src] -= delta
                a[j_tgt] += delta
                conv = conv_cand
                cur_score = score_from_conv(conv, n)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.tolist()
                    print(
                        f"[{time.time():.1f}] iter {iter_count} improved to {best_score:.6f}"
                    )
                # adapt step size gently
                step_scale = min(step_scale * 1.02, 0.2)
                max_no_improve = 0
                accepted = True
                break

        if not accepted:
            # no improvement – shrink step size
            step_scale = max(step_scale * 0.7, min_step)
            max_no_improve += 1
            # if we have been stuck for a long time, try a random perturbation
            if max_no_improve >= 2000:
                # random small swap to escape flat regions
                i, j = np.random.randint(n, size=2)
                if i != j:
                    # exchange a fraction of mass
                    eps = step_scale * min(a[i], a[j])
                    if eps > 0:
                        a[i] -= eps
                        a[j] += eps
                        # recompute full convolution (still cheap)
                        conv = convolution_fft(a, a)
                        cur_score = score_from_conv(conv, n)
                max_no_improve = 0

        iter_count += 1
        if step_scale <= min_step:
            # step has become too tiny – break to avoid endless tiny moves
            break

    # ------------------------------------------------------------------
    # 4. Optional symmetrisation (often helpful)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    conv_sym = convolution_fft(a_sym, a_sym)
    sc_sym = score_from_conv(conv_sym, n)
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_sym.tolist()
        print(f"[{time.time():.1f}] symmetrisation improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return final_seq
