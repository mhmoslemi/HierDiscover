"""
Propose a step‑function (non‑negative sequence) with a small evaluation score.

Algorithm outline
-----------------
1.  Choose a moderate length `n = 8192`.  The objective 2·n·max(conv) /
    (∑a)² is invariant to a global scaling, so we keep `∑a = 1`
    (probability simplex) which simplifies the objective to
    `2·n·max(conv)`.
2.  Scan a few analytic families (product‑power, left‑heavy power,
    exponential‑U) on a coarse grid to obtain a decent starting point.
3.  Perform a **random‑mixture** search:
    generate convex combinations of 2–3 analytic families with random
    parameters, keep the best candidate.
4.  Refine the best candidate with a **mirror‑subgradient** method
    (entropy‑mirror descent on the simplex) that directly minimizes
    `max(conv)`.  The update is
        a ← a·exp(−η·g) ,   a ← a/∑a
    where `g_i = 2·a_{k−i}` and `k` is the index of the current maximal
    convolution entry.
5.  A short **Adam** optimisation on the smooth log‑sum‑exp surrogate
    (as in the reference implementation) gives a final fine‑tuning.
6.  A lightweight **simulated‑annealing hill‑climb** is applied as a
    polish step.
7.  The best sequence found is returned as a plain Python list.
   All steps respect the wall‑clock budget (default 1000 s).

The code contains only top‑level functions, no closures or file I/O,
and uses only `numpy`, `math` and the already‑imported
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return vectors normalised to sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}  (symmetric U‑shape)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the grid: (i+½)^{p‑1}(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


def uniform_family(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Mirror‑subgradient optimiser (entropy‑mirror descent)
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 0.12) -> Tuple[np.ndarray, float]:
    """
    Entropy‑mirror descent for the exact max‑convolution objective.

    Parameters
    ----------
    start_a   – probability vector (sum = 1, non‑negative).
    deadline  – Unix timestamp after which optimisation must stop.
    max_iters – hard cap on iteration count.
    eta0      – base step size (decays as eta0 / sqrt(t)).

    Returns
    -------
    best_a, best_score
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the max‑convolution entry w.r.t. a
        g = np.zeros_like(a)
        if k < n:
            g[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            g[start:] = 2.0 * a[:n - start][::-1]

        eta = eta0 / math.sqrt(t)
        # Mirror (exponential) update, then re‑normalise onto simplex
        a = a * np.exp(-eta * g)
        # renorm (exact simplex projection is cheap because a stays positive)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = project_to_simplex(a)
        else:
            a = a / a_sum

        # occasional symmetry enforcement & evaluation
        if t % 5_000 == 0:
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad improved to {best_score:.6f}")

    # final evaluation
    a = (a + a[::-1]) * 0.5
    a = a / a.sum()
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
        print(f"[{time.time():.1f}] mirror sub‑grad final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (kept from the reference implementation)
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 20.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.013,
                         sym_interval: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient through the smooth‑max
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true objective evaluation
        if it % 3000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score

# Simulated‑annealing hill‑climb (lightweight final polish)

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_500_000) -> Tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: swap or transfer a small chunk of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 5e-5          # tiny initial temperature
    T = T0
    step_scale = 0.08

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                 # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a chunk
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def random_mixture_search(num_samples: int,
                          n: int,
                          deadline: float) -> Tuple[np.ndarray, float]:
    """
    Generate random convex combinations of a few base families,
    keep the best candidate according to `compute_score`.
    """
    best_a = None
    best_score = float("inf")
    families = [
        ("prod", lambda: symmetric_product_family(n, np.random.uniform(0.0, 8.0))),
        ("left", lambda: left_heavy_power_family(n, np.random.uniform(0.0, 6.0))),
        ("expU", lambda: exp_u_family(n, np.random.uniform(0.0, 6.0))),
        ("beta", lambda: beta_family(
            n,
            np.random.uniform(0.1, 3.0),
            np.random.uniform(0.1, 3.0))),
        ("unif", lambda: uniform_family(n)),
    ]

    for s in range(num_samples):
        if time.time() >= deadline:
            break

        # decide how many components (2 or 3) and draw them
        k = np.random.choice([2, 3])
        chosen = np.random.choice(families, size=k, replace=False)

        # random Dirichlet weights
        weights = np.random.dirichlet(np.ones(k))

        vec = np.zeros(n, dtype=np.float64)
        for w, (name, gen) in zip(weights, chosen):
            vec += w * gen()
        # normalise to simplex (tiny numerical drift possible)
        vec = project_to_simplex(vec)

        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_a = vec.copy()
            print(f"[{time.time():.1f}] mixture #{s} improved to {best_score:.6f}")

    return best_a, best_score

            best_seq = mix_a.tolist()
            print(f"[{time.time():.1f}] mixture improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Mirror‑subgradient refinement (exact max‑objective)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] mirror sub‑gradient refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = mirror_subgrad_optimize(
            a_start,
            deadline - 120.0,
            max_iters=800_000,
            eta0=0.15,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Adam smooth‑max optimisation (fine‑tuning)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] Adam fine‑tuning")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 40.0,
            max_iters=500_000,
            beta_start=30.0,
            beta_end=9000.0,
            lr=0.011,
            sym_interval=1500,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam fine‑tune best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final symmetry enforcement (may shave a tiny bit)
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetry improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Final hill‑climb (light SA) for polishing
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA polish")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=3_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA polished best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Fallback to uniform if nothing succeeded
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final evaluation through the official wrapper (incl. safety checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
