def lp_refine(start_a: np.ndarray, deadline: float, max_iters: int = 8, t_init: float = 0.2) -> Tuple[np.ndarray, float]:
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(a)
    # Precompute row/col indices
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    rows = rows.astype(np.int64)
    cols = cols.astype(np.int64)
    # LP parameters
    c = -np.ones(n)  # maximize sum(g)
    for it in range(max_iters):
        if time.time() >= deadline - 5:  # keep some buffer
            break
        conv = convolution_fft(a, a)
        M = conv.max()
        data = np.repeat(a, n)
        A_ub = coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP failed: {res.message}")
            break
        g = res.x
        # mixing step
        t = t_init * (0.5 ** it)  # decreasing t
        a = (1 - t) * a + t * g
        a = project_to_simplex(a)
        sc = compute_score(a)

            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it+1} improved to {best_score:.6f}")
    return best_a, best_score

"""
Search for a near‑optimal step‑function minimizing
``evaluate_sequence`` (i.e. 2·n·max(conv) with ∑a_i=1).

The algorithm combines several complementary ideas:
  1. A very quick coarse sweep of a few analytic families
  2. Adam optimisation of a smooth‐max surrogate (no forced symmetry)
  3. A small number of linear‑programming refinement steps
     (maximise Σg subject to (f·g)≤M, then mix f←(1‑t)f+t·g)
  4. A short exact sub‑gradient descent
  5. A low‑temperature simulated‑annealing hill‑climb for polishing.
All phases respect the global 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution using real FFT (O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector (∑a=1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the initial coarse scan)
# ----------------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, then normalised.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser of the smooth‑max surrogate
# ----------------------------------------------------------------------
def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 10.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.015,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth approximation of max(conv).
    sym_interval == 0 disables forced symmetrisation.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)      # initialise in log‑softmax space

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # anneal smooth‑max temperature
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()               # smooth‑max weights

        # gradient of the smooth max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through the soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 3000 == 0:
            a_tmp = softmax(z)
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final evaluation
    a_fin = softmax(z)
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score

# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------
def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.04) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on f(a)=2·n·max(conv) with Euclidean projections.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))           # index of the current peak

        # sub‑gradient of max(conv) (see analysis notes)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a chunk of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling schedule
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (max‑convolution bound reduction)
# ----------------------------------------------------------------------
def lp_refine(start_a: np.ndarray,
              deadline: float,
              max_iters: int = 6,
              t_start: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Iteratively improve a by solving
        maximise Σ g_j  subject to  g≥0  and (a * g)_k ≤ M,
    where M = max(a*a).  After each LP we mix
        a ← (1‑t)·a + t·g   and project onto the simplex.
    The mixing weight t is geometrically decayed.
    """
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(a)

    # constant index pattern for the Toeplitz matrix of the convolution a*g
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    rows = rows.astype(np.int64)
    cols = cols.astype(np.int64)

    c = -np.ones(n)                         # maximise Σ g  => minimise -Σ g

    for it in range(max_iters):
        if time.time() >= deadline - 2.0:
            break

        conv = convolution_fft(a, a)
        M = conv.max()

        # Build the (sparse) constraint matrix A_ub·g ≤ M
        data = np.repeat(a, n)               # each column j gets a_i as coefficient
        A_ub = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=(0, None), method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP failed at iter {it+1}: {res.message}")
            break

        g = res.x
        t = t_start * (0.5 ** it)            # decreasing mixing weight
        a = (1.0 - t) * a + t * g
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it+1} improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section helper (used for a tiny local α‑refinement)
# ----------------------------------------------------------------------
def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 12) -> Tuple[float, float]:
    """Minimise ``func`` on [a,b] via golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Main driver required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights (a probability vector) that
    minimises ``evaluate_sequence`` within the given wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback (previous best attempt)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. choose a working length (2048 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # 2. coarse scan of a few analytic families (no symmetry enforcement)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] coarse scan (n={n})")
    best_alpha = None

    # product‑power family (U‑shaped)
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_alpha = alpha
            print(f"[{time.time():.1f}] new best (α={alpha:.3f}) = {best_score:.6f}")

    # left‑heavy power law (monotone decreasing)
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (β={beta:.3f}) = {best_score:.6f}")

    # exponential U‑shape
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (γ={gamma:.3f}) = {best_score:.6f}")

    # beta‑type family (grid)
    for p in np.linspace(0.1, 3.0, 13):
        if time.time() >= deadline - 30.0:
            break
        for q in np.linspace(0.1, 3.0, 13):
            vec = beta_family(n, p, q)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (β={p:.2f},{q:.2f}) = {best_score:.6f}")

    # mixture uniform + U‑shape
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (λ={lam:.2f},γ={gamma:.2f}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. fine golden‑section refinement of the product family (if we have α)
    # ------------------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] golden‑section refinement around α={best_alpha:.3f}")

        def score_alpha(a_val: float) -> float:
            return compute_score(symmetric_product_family(n, a_val))

        lo = max(0.0, best_alpha - 0.6)
        hi = min(12.0, best_alpha + 0.6)
        best_alpha, best_alpha_score = golden_section_optimize(score_alpha, lo, hi, max_iter=12)
        vec = symmetric_product_family(n, best_alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] refined α={best_alpha:.6f} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam optimisation (smooth max) – no forced symmetry
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam smooth optimisation (no symmetrisation)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 150.0,
            max_iters=350_000,
            beta_start=10.0,
            beta_end=5000.0,
            lr=0.015,
            sym_interval=0,            # keep asymmetry
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming refinement (few quick iterations)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] LP refinement")
        a_lp, sc_lp = lp_refine(
            np.array(best_seq, dtype=np.float64),
            deadline - 100.0,
            max_iters=6,
            t_start=0.2,
        )
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.tolist()
            print(f"[{time.time():.1f}] after LP best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second Adam pass (higher β, optional light symmetrisation)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=250_000,
            beta_start=50.0,
            beta_end=8000.0,
            lr=0.012,
            sym_interval=0,            # still asymmetric
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 5.0,
            max_iters=1_500_000,
            step0=0.04,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=4_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Fallback to a safe uniform vector if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (includes all security checks)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
