"""
Propose a step‑function that (empirically) improves the auto‑convolution peak.
The search consists of three phases:

1. Initialise a symmetric probability vector (either the previously found
   ``height_sequence_1`` or a product‑family shape).
2. Repeatedly solve a small linear programme:
      maximise Σ g_j  subject to  g ≥ 0  and  f∗g ≤ M,
   where M = max(f∗f). The solution g is scaled so that also max(g∗g) ≤ M.
   The best convex combination f←(1‑t)f + t·g (t found by a tiny line‑search)
   reduces the peak M while the vector stays on the simplex.
   Only a handful of LP‑iterations are needed (≈ 8) and each iteration is
   fast (n≈2048, ~0.1 s).
3. A final smooth‑max Adam optimisation (≈ 2⋅10⁵ steps) refines the result.

All intermediate vectors remain non‑negative and are re‑projected onto the
probability simplex; the final list is returned as plain Python floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to ``x``."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def _compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = _convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used only for the very first guess)
# ----------------------------------------------------------------------


def _symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    # log‑product to avoid overflow
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (identical to the one from the reference)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _adam_smooth_optimize(start_a: np.ndarray,
                          deadline: float,
                          max_iters: int = 200_000,
                          beta_start: float = 10.0,
                          beta_end: float = 5000.0,
                          lr: float = 0.015,
                          sym_interval: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).  Returns
    the best vector found (as a numpy array) and its exact score.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # parameters before the soft‑max

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = _softmax(z)
    best_score = _compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (smooth‑max becomes sharper)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = _softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = _project_to_simplex(a)
            z = np.log(a + eps)

        conv = _convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max surrogate w.r.t. a
        a_rev = a[::-1]
        conv_rev = _convolution_fft(a_rev, w)          # length 2n‑1
        grad_a = 2.0 * conv_rev[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # periodic true‑objective check
        if it % 3000 == 0:
            a_tmp = _softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = _project_to_simplex(a_tmp)
            sc = _compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final evaluation
    a_fin = _softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = _project_to_simplex(a_fin)
    sc_fin = _compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Golden‑section helper (used for the line‑search of the LP step)
# ----------------------------------------------------------------------


def _golden_section_optimize(func, a: float, b: float, max_iter: int = 12) -> float:
    """Minimise ``func`` on the interval [a,b] using golden‑section search."""
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:          # we are minimising
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    return c if fc < fd else d

# Main driver – the function required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a small auto‑convolution peak.
    The routine respects the wall‑clock budget and returns the best sequence
    found as a Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    # --------------------------------------------------------------
    # 0. Initialise a symmetric probability vector
    # --------------------------------------------------------------
    try:
        # Prefer the previously discovered construction if it exists.
        init_seq = np.array([float(x) for x in height_sequence_1],
                            dtype=np.float64)
        n = init_seq.shape[0]
        # keep it as a probability vector
        f = init_seq / init_seq.sum()
        print(f"[{time.time():.1f}] Using height_sequence_1 of length {n}")
    except Exception:
        # Fallback: a symmetric product family (α≈5 is a good starter)
        n = 2048
        f = _symmetric_product_family(n, alpha=5.0)
        print(f"[{time.time():.1f}] No previous sequence – start with product family n={n}")

    best_score = _compute_score(f)
    best_seq = f.copy()
    print(f"[{time.time():.1f}] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Iterative LP‑mixing phase
    # --------------------------------------------------------------
    import scipy.sparse as sp
    from scipy.optimize import linprog

    max_lp_iters = 12                # usually enough to converge
    lp_deadline = deadline - 250.0   # keep some budget for final polishing
    iteration = 0

    while time.time() < lp_deadline and iteration < max_lp_iters:
        iteration += 1
        # compute the current peak M
        conv_f = _convolution_fft(f, f)
        M = conv_f.max()

        # Build the (2n‑1)×n convolution matrix A where (A·g)[k] = (f∗g)[k]
        # Each diagonal i (offset i) is filled with f[i].
        diag_data = np.repeat(f[:, np.newaxis], n, axis=1)   # shape (n,n)
        A = sp.diags(diag_data, offsets=np.arange(n),
                     shape=(2 * n - 1, n), format='csc')

        # Solve   max Σ g_j   s.t.   A·g ≤ M ,  g ≥ 0
        c = -np.ones(n)                     # linprog minimises cᵀx
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=M * np.ones(2 * n - 1),
                      bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP failed – terminating LP phase")
            break
        g = res.x
        sum_g = g.sum()
        sum_f = f.sum()                     # = 1 after normalisation

        # No increase in total mass → cannot improve further
        if sum_g <= sum_f + 1e-12:
            print(f"[{time.time():.1f}] LP gave no mass increase – stop LP loop")
            break

        # Ensure the new candidate also respects the peak bound
        conv_g = _convolution_fft(g, g)
        max_cgg = conv_g.max()
        if max_cgg > M:
            gamma = math.sqrt(M / max_cgg)
            g = g * gamma
            conv_g = _convolution_fft(g, g)   # recompute after scaling
            max_cgg = conv_g.max()

        # Compute the mixed convolutions needed for the line‑search
        conv_fg = _convolution_fft(f, g)

        # ----------------------------------------------------------
        # 1a. Tiny line‑search on t∈[0,1] to minimise the resulting peak
        # ----------------------------------------------------------
        def _peak_for_t(t: float) -> float:
            """Maximum convolution of (1‑t)·f + t·g."""
            coeff_f = (1.0 - t) ** 2
            coeff_g = t ** 2
            coeff_fg = 2.0 * t * (1.0 - t)
            conv_t = coeff_f * conv_f + coeff_g * conv_g + coeff_fg * conv_fg
            return conv_t.max()

        # coarse grid to locate a promising interval
        ts = np.linspace(0.0, 1.0, 21)
        peaks = [_peak_for_t(t) for t in ts]
        best_idx = int(np.argmin(peaks))
        t_lo = max(0.0, ts[max(0, best_idx - 1)])
        t_hi = min(1.0, ts[min(len(ts) - 1, best_idx + 1)])

        # refined golden‑section search on that interval
        t_opt = _golden_section_optimize(_peak_for_t, t_lo, t_hi, max_iter=12)
        new_M = _peak_for_t(t_opt)

        # Build the convex combination and re‑normalise to the simplex
        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = _project_to_simplex(f_new)   # guarantees sum = 1

        new_score = _compute_score(f_new)
        print(f"[{time.time():.1f}] LP‑iter {iteration:2d}: t={t_opt:.4f}, "
              f"M={new_M:.6e}, score={new_score:.6f}")

        if new_score < best_score - 1e-9:
            best_score = new_score
            best_seq = f_new.copy()
            f = f_new
        else:
            # No improvement – stop the LP phase early
            print(f"[{time.time():.1f}] No improvement after LP‑mix; exiting LP loop")
            break

    # --------------------------------------------------------------
    # 2. Smooth‑max Adam refinement (fine tuning)
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] Starting final Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = _adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=250_000,
            beta_start=30.0,
            beta_end=8000.0,
            lr=0.012,
            sym_interval=1500,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Adam refinement improved score to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Final safety check via the official wrapper
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)   # the provided evaluator
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return final_seq
