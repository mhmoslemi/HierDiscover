"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.
--------------------------------------------------------------

This version builds on the previous mirror‑gradient optimiser and adds
a *smooth‑gradient* phase that minimises a soft–max approximation of
the convolution peak.  The soft‑max (log‑sum‑exp) yields a differentiable
objective whose gradient is a weighted correlation of the current
sequence with a probability distribution derived from the convolution
values.  By performing projected Euclidean gradient descent on this
smoothed objective (with a slowly decreasing step‑size) we can escape
plateaus that the pure sub‑gradient method gets stuck in and typically
shave a few thousandths off the final bound.

The optimiser proceeds in three timed stages:

1. **Analytic seeds** – a diverse collection of uniform, power‑law,
   geometric and Beta‑shaped families (including very heavy‑edge
   parameters) plus a few random sparse seeds.  The best seed is kept.

2. **Mirror‑subgradient descent** – the tried‑and‑tested KL‑mirror
   optimiser runs for a configurable portion of the budget.

3. **Smooth‑gradient descent** – a log‑sum‑exp smoothed version of the
   max‑convolution objective is minimised with Euclidean projected
   gradient descent.  The gradient is computed efficiently by two FFT‑
   based convolutions per iteration.

4. **Stochastic hill‑climb** – a final cheap local search (swap / mass
   move) polishes the best solution found.

All phases respect a global wall‑clock deadline (default 1000 s) and
the best sequence encountered is finally evaluated by the provided
``evaluate_sequence`` function.  The returned value is a plain Python
list of floats already on the probability simplex (∑a_i = 1, a_i ≥ 0).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, automatically normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}.
    For α<1 this concentrates mass near the two ends.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_random_sparse(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose m indices (without replacement) and assign equal mass.
    """
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Mirror‑subgradient optimiser (KL‑mirror / exponentiated‑gradient)
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).  a is kept on the simplex.
    Returns the best sequence seen and its score.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # Obtain the active convolution peak and its sub‑gradient.
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Average sub‑gradient (equals 2·max_conv)
        avg = np.dot(a, g)

        # Diminishing stepsize
        eta = eta0 / np.sqrt(t)

        # Exponentiated‑gradient update (KL‑mirror)
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)   # avoid overflow
        a = a * np.exp(exp_arg)

        # Projection onto the simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # Periodic progress check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # Final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑gradient optimiser (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    eps: float = 5.0,
    eta0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Projected Euclidean gradient descent on a smooth approximation
    f_ε(a) = (1/ε)·log∑_k exp(ε·conv_k).

    The gradient can be written as a weighted correlation of the current
    sequence with a probability vector derived from the softmax over
    convolution values.  Two FFT‑based convolutions give the gradient
    in O(n log n) time per iteration.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # 1) convolution of a with itself
        conv = convolution_fft(a, a)                 # length 2n‑1

        # 2) softmax weights (numerically stable)
        shift = np.max(conv)
        exp_vals = np.exp(eps * (conv - shift))
        w = exp_vals / np.sum(exp_vals)              # length 2n‑1, sums to 1

        # 3) gradient = 2 * correlation(a, w)
        #    correlation can be obtained via a convolution with the reversed w.
        w_rev = w[::-1]                               # reverse order
        grad_full = convolution_fft(a, w_rev)        # length 3n‑2
        offset = len(w_rev) - 1                       # = 2n‑2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # 4) Euclidean step with diminishing stepsize
        eta = eta0 / np.sqrt(t)
        cand = a - eta * grad
        cand = project_to_simplex(cand)

        # 5) bookkeeping
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
        a = cand

        # Optional early exit: if improvement under a tiny epsilon for many steps
        # (not strictly necessary)

    # Final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (swap / fractional mass move)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Tiny local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

    restart_deadline = deadline - 60.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(6):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.02 * rng.standard_normal(n)
            pert = project_to_simplex(pert)
            sub_dead = time.time() + 10.0
            a_tmp, s_tmp = mirror_subgrad_optimize(
                pert,
                time_deadline=sub_dead,
                max_iters=150_000,
                eta0=0.6,
            )
            if s_tmp < best_val:
                best_val, best_seq = s_tmp, a_tmp.copy()
                print(f"Restart {r} yielded improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb (tiny local moves)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Final hill‑climb for up to {steps} steps.")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val, best_seq = hill_score, np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard – fall back to uniform if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 8. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
