def softmax_gradient(a, T):
    conv = convolution_fft(a, a)   # length 2n-1
    maxc = conv.max()
    # subtract maxc for numerical stability
    scaled = (conv - maxc) / T
    # exponentials
    exp_scaled = np.exp(np.clip(scaled, -700, 700))   # avoid overflow
    sum_exp = exp_scaled.sum()
    w = exp_scaled / sum_exp   # shape (2n-1)
    # compute gradient
    a_rev = a[::-1]
    c = convolution_fft(w, a_rev)   # length (2n-1)+(n)-1=3n-2
    # slice
    g = 2.0 * c[len(a)-1 : len(a)-1 + len(a)]   # because len(a) = n

def hill_climb(seq, deadline, max_steps=500000):
   a = np.array(seq, dtype=np.float64)
   n = len(a)
   best = a.copy()
   best_score = compute_score(best)
   step_scale = 0.05   # start
   steps = 0
   while steps < max_steps and time.time() < deadline:
       # propose a move; with probability 0.5, swap; else move mass
       cand = best.copy()
       if np.random.rand() < 0.5:
           i, j = np.random.randint(n, size=2)
           cand[i], cand[j] = cand[j], cand[i]
       else:
           i, j = np.random.randint(n, size=2)
           if i != j:
               delta = step_scale * best[i]
               cand[i] -= delta
               cand[j] += delta
               if cand[i] < 0:
                   cand[i] = 0.0
       cand = project_to_simplex(cand)
       cur_score = compute_score(cand)
       if cur_score < best_score:
           best_score = cur_score
           best = cand
           step_scale = min(step_scale * 1.05, 0.5)
       else:
           step_scale *= 0.97
       steps += 1
   return best.tolist(), best_score

"""
Non‑negative step‑function optimizer.

Algorithm
---------
1. Build a wide collection of analytic “seed’’ sequences (uniform,
   central‑binomial, power‑law, geometric, asymmetric Beta shapes) and
   evaluate them exactly.  Keep the three best seeds.

2. For each of these seeds run a *soft‑max‑based exponentiated‑gradient*
   optimisation on the true objective
        F(a) = 2·n·max(conv(a,a))          (sum(a)=1, a≥0).

   The gradient of the smooth approximation
        φ_T(a)=T·log Σ_k exp(conv_k/T)
   is computed analytically via a single FFT‑based convolution.
   An exponential‑gradient (KL‑mirror) update with a diminishing step‑size
   is applied and the iterate is projected back onto the probability simplex.
   The temperature T is annealed (exponential decay) so the method initially
   follows a smooth landscape and gradually focuses on the true maximum.

3. After the annealing phase a cheap stochastic hill‑climb (swap / mass‑move
   moves with simplex projection) is performed to polish the best vector.

All steps respect a wall‑clock deadline (default 1000 s) and keep the best
vector seen.  The returned list of floats can be fed directly to the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector on the simplex (sum(a)=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial seeds
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum = 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(−p), normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    Allows asymmetry (α≠β).
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (beta - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Soft‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def softmax_eg_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 1_000_000,
    base_lr: float = 0.6,
    T0: float = 0.004,
    decay_rate: float = 5e-5,
    T_min: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) with an exponentiated‑gradient descent on the
    temperature‑smoothed objective φ_T(a)=T·log Σ exp(conv/T).

    The update is
        a_i ← a_i·exp( -η·(g_i - ⟨g⟩) )
    where g is the gradient of φ_T and ⟨g⟩ its mean (to keep the scale stable).
    After each update the vector is projected onto the simplex.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # ---- current convolution and score ---------------------------------
        conv = convolution_fft(a, a)               # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- temperature ----------------------------------------------------
        T = max(T_min, T0 * np.exp(-decay_rate * t))

        # ---- soft‑max weights (numerically safe) ---------------------------
        scaled = (conv - cur_max) / T                # max ⇒ 0
        scaled = np.clip(scaled, -50.0, 0.0)          # avoid overflow
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:                            # should not happen
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # ---- gradient of φ_T (using convolution) ---------------------------
        a_rev = a[::-1]                              # reversed a
        c = convolution_fft(w, a_rev)                # length 3n‑2
        g = 2.0 * c[n - 1 : 2 * n - 1]                # slice of length n

        # ---- stepsize (diminishing) ---------------------------------------
        eta = base_lr / np.sqrt(t)

        # ---- exponentiated‑gradient (KL mirror) update --------------------
        exp_arg = -eta * (g - g.mean())
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # ---- simplex projection --------------------------------------------
        a = project_to_simplex(a)

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb for polishing
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline, max_steps: int = 800_000):
    """
    Tiny stochastic local search (swap / fractional mass transfer) with
    simplex projection.  Keeps the best vector seen.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a.tolist(), best_score
