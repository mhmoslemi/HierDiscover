def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}   (U‑shape for α>0)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-β} (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def centre_heavy_exp_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape centred at the middle: w_i ∝ exp(-γ·dist_to_centre)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.abs(i - (n - 1) / 2.0)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def edge_heavy_exp_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape with mass at the edges: w_i ∝ exp(γ·dist_to_centre)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.abs(i - (n - 1) / 2.0)
    # to avoid overflow we shift the exponent by its maximum
    exp_arg = gamma * dist
    exp_arg -= exp_arg.max()
    w = np.exp(exp_arg)
    a = w / w.sum()
    return a


def beta_family(n: int, a_param: float, b_param: float) -> np.ndarray:
    """Discrete Beta‑distribution (a,b>0)."""
    i = np.arange(n, dtype=np.float64)
    # use (i+0.5) to avoid zero at the boundaries
    w = np.power(i + 0.5, a_param - 1.0) * np.power(n - 0.5 - i, b_param - 1.0)
    # guard against NaNs when a_param or b_param < 1
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    a_vec = w / w.sum()
    return a_vec


def random_dirichlet_start(n: int, concentration: float = 0.5) -> np.ndarray:
    """Sample from Dirichlet(α) and enforce symmetry."""
    g = np.random.gamma(concentration, 1.0, size=n)
    a = g / g.sum()
    a = (a + a[::-1]) * 0.5      # symmetrisation
    a = a / a.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Monotone decreasing random vector (used previously)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the true max objective
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 200_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (EG) on  f(a)=2·n·max(conv(a,a))."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (same as in the reference code)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        a = a / a.sum()                     # simplex (EG preserves positivity)

        # enforce symmetry – this empirical trick often reduces the peak
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_sc:.6f}")

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on the log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 4.0,
                         beta_end: float = 3000.0,
                         lr: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The β schedule is exponential from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linearly interpolate in log‑space between start and end β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # log‑sum‑exp trick for numerical stability
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()          # softmax on the convolution entries

        # gradient w.r.t. a (derivation of the smoothed max)
        # conv = a * a  ⇒  ∂conv/∂a = 2 * (a * reverse(w))
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # backprop through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_sc:
                best_sc = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_sc:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_sc:
        best_sc = sc_fin
        best_a = a_fin.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (very cheap iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient on the exact max‑convolution."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        # enforce symmetry
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_sc:.6f}")

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_sc = compute_score(cur_a)

    best_a = cur_a.copy()
    best_sc = cur_sc

    # initial temperature scaled to the current score
    T0 = cur_sc * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                           # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # clipping negative values
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)

        delta = cand_sc - cur_sc
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_sc


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  Returns a plain Python list ready for the harness.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback candidate if supplied by the harness
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]      # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose working length
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = fallback_len if fallback_len >= 1024 else 4096
    print(f"[{time.time():.1f}] working length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Analytic families + random starts (quick scan)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] starting analytic/random scan")
        # 2a) a few systematic families
        alphas = np.linspace(0.0, 25.0, 26)
        betas = np.linspace(0.0, 25.0, 26)
        gammas = np.linspace(0.0, 12.0, 25)

        # symmetric product / U‑shape centred
        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            a = symmetric_product_family(n, alpha)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # left‑heavy power
        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            a = left_heavy_power_family(n, beta)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        # centre‑heavy exponential
        for gamma in gammas:
            if time.time() >= deadline - 30.0:
                break
            a = centre_heavy_exp_family(n, gamma)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[{time.time():.1f}] new best (centre‑γ={gamma:.2f}) = {best_score:.6f}")

        # edge‑heavy exponential (new family)
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            a = edge_heavy_exp_family(n, gamma)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[{time.time():.1f}] new best (edge‑γ={gamma:.2f}) = {best_score:.6f}")

        # beta family sweep (coarse grid)
        a_params = np.linspace(0.2, 4.0, 9)
        b_params = np.linspace(0.2, 4.0, 9)
        for a_p in a_params:
            for b_p in b_params:
                if time.time() >= deadline - 30.0:
                    break
                a = beta_family(n, a_p, b_p)
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.tolist()
                    print(f"[{time.time():.1f}] new best (beta a={a_p:.2f} b={b_p:.2f}) = {best_score:.6f}")

        # random Dirichlet / monotone draws
        for _ in range(300):
            if time.time() >= deadline - 30.0:
                break
            if np.random.rand() < 0.5:
                a = random_dirichlet_start(n, concentration=np.random.uniform(0.1, 1.0))
            else:
                a = random_monotone_start(n)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise the current best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    # enforce symmetry from the start
    a = (a + a[::-1]) * 0.5
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3.  Multiple restarts of (EG → Adam) refinement
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 200.0 and restarts < 5:
        restarts += 1
        print(f"[{time.time():.1f}] restart #{restarts}")

        # start from a fresh random perturbation of the current best
        perturb = a + 0.02 * np.random.randn(n)
        perturb = np.abs(perturb)                     # keep non‑negative
        perturb = project_to_simplex(perturb)
        perturb = (perturb + perturb[::-1]) * 0.5
        perturb = perturb / perturb.sum()

        # EG phase
        eg_deadline = min(deadline - 120.0, time.time() + 50.0)
        a_eg, sc_eg = eg_optimize(perturb,
                                  deadline=eg_deadline,
                                  max_iters=250_000,
                                  eta0=0.7)
        if sc_eg < best_score:
            best_score, a = sc_eg, a_eg
            print(f"[{time.time():.1f}] after EG (restart) best = {best_score:.6f}")

        # Adam smooth‑max phase
        adam_deadline = min(deadline - 70.0, time.time() + 70.0)
        a_adam, sc_adam = adam_smooth_optimize(a,
                                                deadline=adam_deadline,
                                                max_iters=300_000,
                                                beta_start=4.0,
                                                beta_end=2500.0,
                                                lr=0.014)
        if sc_adam < best_score:
            best_score, a = sc_adam, a_adam
            print(f"[{time.time():.1f}] after Adam (restart) best = {best_score:.6f}")

        # short sub‑gradient polishing (very cheap)
        sub_deadline = min(deadline - 20.0, time.time() + 20.0)
        a_sub, sc_sub = subgrad_optimize(a,
                                         deadline=sub_deadline,
                                         max_iters=800_000,
                                         step0=0.05)
        if sc_sub < best_score:
            best_score, a = sc_sub, a_sub
            print(f"[{time.time():.1f}] after sub‑grad (restart) best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final symmetrisation test (just in case)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score, a = sc_sym, a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Simulated‑annealing hill‑climb for the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_sa, sc_sa = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_sa < best_score:
            best_score, a = sc_sa, a_sa
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
