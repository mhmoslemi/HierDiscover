"""
Propose a non‑negative step‑function (list of heights) that minimizes

        2·n·max(convolve(a,a)) / (sum(a)²)

by a fast multi‑phase heuristic:

1.  Load a fallback candidate (if any) and evaluate it.
2.  Scan several analytically‑parameterised families (U‑shape, power‑law,
    symmetric product, mixtures) on a relatively large problem size
    (≥4096).  Keep the best vector.
3.  Generate many random monotone / U‑shaped starts and keep any improvement.
4.  Refine the current best using a long Adam optimisation of the smooth
    log‑sum‑exp surrogate of the max‑convolution (high β → tight approximation).
5.  Optionally mix the refined solution with the uniform distribution and keep
    the best mixture.
6.  Symmetrise the solution (a ← (a + a[::-1])/2) and keep any improvement.
7.  A final short stochastic‑hill‑climb (simulated‑annealing) polish.
8.  Return the best sequence found before the global 1000 s deadline.

The implementation relies only on NumPy and the provided ``evaluate_sequence``
function; no external files or network access are used.  All time‑critical
loops check the deadline and stop early if necessary.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (initialisation)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a decreasing vector with random power–law shape."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """U‑shape with random decay."""
    gamma = np.random.uniform(0.0, 15.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimisation of the smooth-max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-14
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for β (exponential schedule works well)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max over the β‑scaled convolution values
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()          # weights sum to 1

        # gradient of f_β w.r.t. a : 2 * (a_rev * w)   (slice the appropriate part)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # chain rule through the soft‑max parametrisation
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update on z
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional reporting
        if it % 5000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – start relatively low because we already have a good point
    T0 = cur_score * 2e-4
    T = T0
    step_scale = 0.1

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # move a little mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback candidate
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = compute_score(np.array(fb_vec, dtype=np.float64))
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        # no fallback supplied
        pass

    # ------------------------------------------------------------------
    # 1.  Determine problem size
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    # Use at least 4096 points (larger n gives more freedom)
    n = max(fallback_len, 4096)
    print(f"[{time.time():.1f}] problem size set to n = {n}")

    # ------------------------------------------------------------------
    # 2.  Analytic families scan (quick but broad)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting analytic family sweep")
        # expand the parameter ranges a little compared with the previous run
        alphas = np.linspace(0.0, 80.0, 17)   # 0 … 80 step 5
        betas = np.linspace(0.0, 80.0, 17)
        gammas = np.linspace(0.0, 30.0, 13)

        # symmetric product family
        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # left heavy power family
        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        # exponential U‑shape family
        for gamma in gammas:
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        # mixed uniform‑U family (random sweeps)
        for _ in range(40):
            if time.time() >= deadline - 30.0:
                break
            lam = np.random.rand()
            gamma = np.random.uniform(0.0, 30.0)
            vec = mixed_uniform_u_family(n, lam, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (mix) = {best_score:.6f}")

        # random monotone / random U‑shape starts
        for _ in range(800):
            if time.time() >= deadline - 30.0:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Normalise the current best vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform fallback
    else:
        a = np.array(best_seq, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] after initial phase best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam smoothing (long run – consumes most of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        remaining = deadline - time.time() - 10.0   # keep a small safety buffer
        max_iters = int(6e5)                      # generous upper bound
        # adapt max_iters to the time we actually have
        est_time_per_iter = 0.00025               # rough estimate (seconds)
        max_iters = min(max_iters, int(remaining / (est_time_per_iter * 2)))
        print(f"[{time.time():.1f}] starting Adam smoothing, max_iters={max_iters}")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline - 5.0,
            max_iters=max_iters,
            beta_start=4.0,
            beta_end=5000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] Adam finished – new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Small mixture with the uniform distribution (helps flatten peaks)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        for lam in (0.02, 0.05, 0.10, 0.15):
            cand = (1.0 - lam) * a + lam * uniform
            cand = project_to_simplex(cand)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                a = cand
                print(f"[{time.time():.1f}] uniform mix λ={lam:.2f} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Symmetrisation test
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final short hill‑climb (simulated annealing) for polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        remaining_steps = 200_000
        print(f"[{time.time():.1f}] final hill‑climb (SA) – {remaining_steps} steps")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=remaining_steps)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] search completed – final reported score = {final_score:.6f}")
    return final_seq
