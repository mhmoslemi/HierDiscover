def softmax_adam(a, deadline, beta_start=1.0, beta_end=5000.0, lr=0.01, beta1=0.9, beta2=0.999):
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = compute_score(best)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0
    start = time.time()
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(cur, cur)
        maxc = conv.max()
        cur_score = 2.0 * n * float(maxc)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
        # temperature schedule based on elapsed fraction
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)
        # compute softmax weights
        shifted = conv - maxc
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        # gradient of surrogate
        grad = 2.0 * beta * np.correlate(w, cur, mode='valid')
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr / (np.sqrt(v_hat) + 1e-8)
        cur = cur - step * m_hat
        cur = project_to_simplex(cur)
    return best, best_score

def hill_climb_anneal(a, deadline, init_step=0.05):
    cur = a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    # initial temperature relative to current score
    T0 = cur_score * 0.01   # allow up to 1% worse moves early
    t_start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - t_start
        # linear cooling
        T = max(0.0, T0 * (1 - elapsed/(deadline - t_start)))
        i, j = np.random.randint(len(cur), size=2)
        # random move: transfer fraction
        if cur[i] > 0:
            delta = init_step * cur[i] * (np.random.rand()*2-1)  # can increase or decrease?
            # Let's choose transfer from i to j if delta>0 else opposite
            if delta > 0:
                delta = min(delta, cur[i])
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                delta = min(-delta, cur[j])
                cand = cur.copy()
                cand[i] += delta
                cand[j] -= delta
            cand = project_to_simplex(cand)  # ensure sum=1, nonnegative
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                cur = cand
                cur_score = cand_score
            else:
                # accept with probability
                if T > 0 and np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                    cur = cand
                    cur_score = cand_score
    return best, best_score

            i, j = np.random.randint(n, size=2)
            if cand[i] > 0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            cur = cand.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            # accept uphill with small probability decreasing over time
            elapsed = time.time() - start
            total = deadline - start
            # temperature linearly decays from 0.01*best_score to 0
            T = max(0.0, (1 - elapsed/total) * best_score * 0.01)
            if T > 0 and np.random.rand() < np.exp(-(cand_score - cur_score)/T):
                cur = cand
        # possibly adjust step_scale down if no improvement for a while
        # Not needed
    return best, best_score

"""
Advanced optimiser for the step‑function evaluation problem.

Strategy
--------
* Choose a moderate length (2048 or 4096) according to the available time.
* Build a diverse portfolio of candidate shapes (deterministic, beta‑family,
  beta‑pdf, random Dirichlet, sparse spikes, optional fallback).
* Keep the few best seeds (by the exact objective) and improve each of them
  with three optimisation phases:
    1. Soft‑max (log‑sum‑exp) surrogate minimisation using Adam.
    2. Projected sub‑gradient descent on the true max‑objective.
    3. Stochastic hill‑climb with simulated‑annealing‑style uphill moves.
* Return the best sequence found (as a plain Python list).

All updates are projected onto the probability simplex, so the objective
2·n·max(conv) is directly comparable with ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (zero‑padded).
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def softmax_adam(
    a0: np.ndarray,
    deadline: float,
    beta_start: float = 1.0,
    beta_end: float = 5000.0,
    lr: float = 0.01,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate Lβ(a)=log Σ exp(β·conv(a,a))
    using Adam with an annealed temperature β.
    """
    n = a0.shape[0]
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(best)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    start = time.time()

    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        maxc = conv.max()
        cur_score = 2.0 * n * float(maxc)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # annealed β
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - maxc          # max → 0
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of Lβ
        grad = 2.0 * beta * np.correlate(w, a, mode="valid")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + epsilon)
        a = a - step * m_hat

        a = project_to_simplex(a)

    return best, best_score


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    step_start: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    """
    n = a0.shape[0]
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(best)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # derivative of max‑convolution (2·a_{k-i})
        g[mask] = 2.0 * a[j[mask]]

        step = step_start / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = np.random.randn(n) * (step_start * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best, best_score


def hill_climb_anneal(
    a0: np.ndarray,
    deadline: float,
    init_step: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with occasional uphill moves (simulated annealing).
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    # initial temperature as a small fraction of the current score
    T0 = best_score * 0.02

    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = T0 * (1.0 - fraction)               # linear cooling

        cand = cur.copy()

        # two elementary moves
        if np.random.rand() < 0.5:               # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = init_step * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            cur = cand
            cur_score = cand_score
            # enlarge step a little after a success
            init_step = min(init_step * 1.07, 0.5)
        else:
            # accept uphill with probability exp(-Δ/T)
            if T > 0.0 and np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                cur = cand
                cur_score = cand_score
            # shrink step gradually when failing
            init_step *= 0.96

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Choose sequence length (more variables → more freedom)
    # ------------------------------------------------------------------
    n = 2048
    if budget_s > 800:
        n = 4096
    print(f"[init] sequence length = {n}")

    # ------------------------------------------------------------------
    # Build a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic shape
    seeds.append(generate_deterministic(n))

    # symmetric polynomial family
    for p in np.linspace(-3.0, 3.0, 61):
        seeds.append(generate_beta_family(n, p))

    # beta pdf family (log‑space parameters)
    alphas = np.logspace(-0.7, 0.7, 15)   # ≈ 0.2 … 5
    betas = np.logspace(-0.7, 0.7, 15)
    for a in alphas:
        for b in betas:
            seeds.append(generate_beta_pdf(n, a, b))

    # random Dirichlet draws
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(random_dirichlet(n, conc=conc))

    # sparse spikes – often a good starting point
    for k in (30, 50, 80, 120):
        seeds.append(random_spike(n, k=k))

    # optional fallback from previous run (if available)
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore

            vec = np.array(fallback[:n], dtype=float)
        else:
            vec = np.zeros(n, dtype=float)

    top_k = 5
    best_overall = evaluated[0][1].copy()

    print(f"[seed] best initial score = {best_val:.6f}")

            a_soft,

    final_seq = best_overall.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"\n[done] final reported score = {final_eval:.6f}")

    return final_seq
