def gradient_from_weights(a, w):
    n = a.shape[0]
    conv_len = n + len(w) - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    G = A * W
    conv_full = np.fft.irfft(G, n=fft_len)[:conv_len]
    # gradient slice:
    grad = 2 * conv_full[n-1:n-1+n]  # length n
    return grad

    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    iteration = 0
    beta = 5.0
    # we will gradually increase beta
    while time.time() < deadline:
        iteration += 1
        # update beta
        beta = min(beta * 1.005, 200.0)  # increase slowly
        conv = full_convolution(a)
        # compute softmax weights
        # subtract max conv for numerical stability
        maxc = conv.max()
        exps = np.exp(beta * (conv - maxc))
        w = exps / np.sum(exps)
        # gradient:
        grad = gradient_from_weights(a, w)
        # step size schedule
        step_factor = 0.5 / np.sqrt(iteration)  # adjust
        # compute candidate step size based on gradient norm
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        # base step size ensuring step length ~ step_factor
        base_t = step_factor / norm
        # evaluate few steps
        t_candidates = [0, base_t, base_t*0.5, base_t*2.0, base_t*5.0]
        best_t = 0
        best_M = conv.max()
        for t in t_candidates:
            if t == 0:
                continue
            cand = a - t * grad
            cand = project_to_simplex(cand)
            M_cand = full_convolution(cand).max()
            if M_cand < best_M:
                best_M = M_cand
                best_t = t
        if best_t == 0:
            # maybe try smaller step
            # try random jitter as fallback
            jitter = np.random.randn(len(a)) * 0.001
            cand = a + jitter
            cand = project_to_simplex(cand)
            M_cand = full_convolution(cand).max()
            if M_cand < best_M:
                best_t = 0  # we accept cand but we track best externally
                best_M = M_cand
                a = cand

def anneal_optimize(a0, time_deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    iteration = 0
    beta = 5.0
    max_beta = 200.0
    while time.time() < time_deadline:
        iteration += 1
        # increase beta gradually
        if iteration % 100 == 0:
            beta = min(beta * 1.05, max_beta)

        conv = full_convolution(a)
        maxc = conv.max()
        # softmax weights: w_i = exp(beta * (conv_i - maxc))
        # This emphasizes max indices more as beta grows.
        exps = np.exp(beta * (conv - maxc))
        w = exps / np.sum(exps)

        # gradient
        grad = gradient_from_weights(a, w)

        # step size based on iteration
        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # base step
        step_scale = 0.5 / np.sqrt(iteration)  # decreasing
        step = step_scale / norm

        # try several multipliers for step
        best_local_a = None
        best_local_score = None
        for factor in (0.5, 1.0, 2.0, 5.0):
            t = step * factor
            cand = a - t * grad
            cand = project_to_simplex(cand)
            val = compute_score(cand)
            if best_local_score is None or val < best_local_score:
                best_local_score = val
                best_local_a = cand

        # Accept if improved
        if best_local_score < best_val:
            a = best_local_a
            best_val = best_local_score
            best_a = a.copy()
        else:
            # maybe accept if improvement moderate or random
            # Add small jitter occasionally
            if iteration % 200 == 0:
                jitter = np.random.randn(len(a)) * 0.001
                cand = a + jitter
                cand = project_to_simplex(cand)
                val = compute_score(cand)
                if val < best_val:
                    a = cand
                    best_val = val
                    best_a = a.copy()
    return best_a

"""
Step‑function optimiser (≈ 1 500 ms budget)

Algorithm
---------
1. **Initial families** – Exhaustively test a few low‑dimensional families
   (geometric, power‑law, symmetric “double‑tail” and a flat‑head‑tail mix)
   plus a few random Dirichlet draws. The best candidate is kept.

2. **Entropic‑gradient annealing** – minimise a smooth surrogate of the
   ℓ∞‑norm of the full convolution:

       Lβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k])

   The gradient of Lβ is   ∇L = 2·a * w  where w is the softmax of the
   convolution entries.  β is increased slowly (annealing) so that the
   optimisation first follows a gentle landscape and finally focuses on
   the true maximum.  An adaptive step‑size  η = c/√t /‖∇L‖ and a tiny
   line‑search over a few multiples of η are used.  After each step the
   iterate is projected onto the probability simplex (non‑negative,
   Σ=1).

3. **Stochastic hill‑climb** – a cheap greedy neighbour search transfers a
   small amount of mass between two random positions, accepts it if the
   objective improves and adapts the transfer scale.

All phases respect the remaining wall‑clock budget (default 1000 s).  The
returned list contains non‑negative floats whose sum is 1; the official
``evaluate_sequence`` sees exactly the same objective as ``compute_score``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n−1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σa = 1):
       2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate families (phase 1)
# ----------------------------------------------------------------------
def generate_initial_candidates(n: int, deadline: float) -> np.ndarray | None:
    """Search simple parametric families + a few random draws."""
    best_a = None
    best_val = float('inf')

    # uniform (always a valid baseline)
    a = np.full(n, 1.0 / n)
    val = compute_score(a)
    if val < best_val:
        best_val, best_a = val, a

    # power‑law a_i ∝ i^{‑p}
    for p in np.linspace(0.30, 1.50, 49):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # symmetric double‑power‑law a_i ∝ i^{‑p} + (n‑i)^{‑p}
    for p in np.linspace(0.30, 1.50, 49):
        if time.time() >= deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-p) + (n - idx + 1) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # geometric decay a_i ∝ r^i
    for r in np.linspace(0.50, 0.99, 81):
        if time.time() >= deadline:
            break
        a = r ** np.arange(n, dtype=np.float64)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # symmetric double‑geometric a_i ∝ r^i + r^{n‑1‑i}
    for r in np.linspace(0.50, 0.99, 81):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = r ** i + r ** (n - 1 - i)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # flat head + power‑law tail (as in the reference)
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= deadline:
            break
        L = int(L_frac * n)
        if L == 0 or L >= n:
            continue
        for p in np.linspace(0.40, 0.80, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head = np.full(L, tail.max())
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    # a few Dirichlet random draws (heavy‑tailish)
    for _ in range(20):
        if time.time() >= deadline:
            break
        # concentration 0.5 creates a lot of variance
        a = np.random.dirichlet(alpha=np.full(n, 0.5))
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    return best_a


# ----------------------------------------------------------------------
# Entropic‑gradient annealing (phase 2)
# ----------------------------------------------------------------------
def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇L where w is a probability vector over the 2·n‑1 convolution
    indices. Uses FFT to evaluate
        g = 2 * (rev(a) * w)   (central slice of the convolution)
    """
    n = a.shape[0]
    rev_a = a[::-1]
    conv_len = rev_a.shape[0] + w.shape[0] - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    A = np.fft.rfft(rev_a, fft_len)
    W = np.fft.rfft(w, fft_len)
    G = A * W
    full = np.fft.irfft(G, fft_len)[:conv_len]
    grad = 2.0 * full[n - 1:n - 1 + n]          # length n
    return grad


def anneal_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Entropic‑gradient descent with annealing (soft‑max approximation of max).
    Returns the best vector found before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    iteration = 0
    beta = 5.0                     # start with a gentle smoothing
    max_beta = 250.0               # final temperature (high β ≃ max)

    while time.time() < deadline:
        iteration += 1

        # slowly raise the inverse temperature
        if iteration % 100 == 0 and beta < max_beta:
            beta = min(beta * 1.05, max_beta)

        conv = full_convolution(a)
        maxc = conv.max()
        # numerically stable soft‑max centred at the current maximum
        exps = np.exp(beta * (conv - maxc))
        w = exps / np.sum(exps)                     # length 2·n‑1, sums to 1

        grad = gradient_from_weights(a, w)

        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break

        # step size ~ 0.5 / sqrt(t) normalized by the gradient magnitude
        base_step = 0.5 / np.sqrt(iteration) / grad_norm

        # try a handful of multiples of the base step and keep the best
        local_best = None
        local_best_val = None
        for factor in (0.5, 1.0, 2.0, 5.0):
            step = base_step * factor
            cand = a - step * grad
            cand = project_to_simplex(cand)
            val = compute_score(cand)
            if (local_best_val is None) or (val < local_best_val):
                local_best_val = val
                local_best = cand

        # accept only if we have an improvement
        if local_best_val is not None and local_best_val < best_val:
            a = local_best
            best_a = a.copy()
            best_val = local_best_val
        else:
            # occasional jitter to escape tiny plateaus
            if iteration % 250 == 0:
                jitter = np.random.randn(a.shape[0]) * 0.001
                cand = a + jitter
                cand = project_to_simplex(cand)
                val = compute_score(cand)
                if val < best_val:
                    a = cand
                    best_a = a.copy()
                    best_val = val

    return best_a


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climb (phase 3)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 500_000) -> np.ndarray:
    """Tiny stochastic moves with greedy acceptance."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``. Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for outer harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # 0. optional fallback (if a previous construction is provided)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. parametric + random sweep (phase 1)
    # ------------------------------------------------------------------
    # use the length of the best known sequence, otherwise a reasonable default
    n = len(best_seq) if best_seq is not None else 2048
    sweep_deadline = deadline - 500.0
    if time.time() < sweep_deadline:
        cand = generate_initial_candidates(n, sweep_deadline)
        if cand is not None:
            val = compute_score(cand)
            print(f"[sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand.tolist()

    # ------------------------------------------------------------------
    # 2. entropic‑gradient annealing (phase 2)
    # ------------------------------------------------------------------
    anneal_deadline = deadline - 150.0
    if best_seq is not None and time.time() < anneal_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_opt = anneal_optimize(a0, anneal_deadline)
        val = compute_score(a_opt)
        print(f"[anneal] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.tolist()

    # ------------------------------------------------------------------
    # 3. final stochastic hill‑climb (phase 3)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (uses the harness's security wrapper)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
