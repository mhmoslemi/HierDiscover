"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm
---------
0.  Initialise the random seed, set a time budget and a working length n
    (default 2048).  A small safety margin is kept for the harness.

1.  **Very fine analytic scan** – explore three flexible parametric families
    (symmetric product, left‑heavy power, centred exponential) on a dense
    grid of parameters (step ≈0.05).  For each candidate we normalise to a
    probability vector and keep the best one (by the exact objective).

2.  **Adam on a smoothed‑max surrogate** – starting from the best candidate
    we minimise the log‑sum‑exp approximation of the max‑convolution.
    The temperature β is increased from 4 to 5000, the learning‑rate is
    schedule‑free and we use the Adam optimiser on the soft‑max
    parametrisation `a = softmax(z)`.  The best point seen is retained.

3.  **Projected sub‑gradient descent** – a cheap but robust optimiser for the
    true max‑objective.  The step size follows the classical `c/√t` rule.
    The maximum over the convolution is recomputed each iteration and the
    resulting sub‑gradient is projected onto the simplex.

4.  **Aggressive simulated‑annealing hill‑climb** – a final stochastic search
    that performs both swaps and mass‑transfer moves.  The temperature is
    linearly cooled to zero, the step‑scale is adapted on the fly and a
    simple one‑dimensional line‑search on the mixing weight is performed
    after each successful move.

5.  The best sequence found (as a plain Python list of floats) is returned.
    All steps respect the global wall‑clock limit (default 1000 s).

The code works for any n ≤ ~3000 (the FFT‑based convolution is O(n log n)
and the LP‑refinement from the reference solution is omitted here in favour
of a faster pure‑gradient search).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Parametric families (dense grid versions)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+0.5)^(-alpha)·(n-0.5-i)^(-alpha)  (symmetric heavy‑tail)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-alpha) * (n - 0.5 - i) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+0.5)^(-beta)  (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp( -gamma * distance to centre )."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp( -(i-center)² / (2·sigma²) )."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random monotone decreasing distribution."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_symmetric_start(n: int) -> np.ndarray:
    """Random symmetric decreasing shape."""
    half = random_monotone_start((n + 1) // 2)
    if n % 2 == 0:
        a = np.concatenate([half, half[::-1]])
    else:
        a = np.concatenate([half, half[-2::-1]])
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.012) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate of max(conv) with Adam.
    The vector is parametrised via soft‑max to enforce the simplex.
    """
    eps = 1e-12
    # Initialise logits from the current distribution (add eps for safety)
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # geometric schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # log‑sum‑exp weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derivation from the smooth‑max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 1000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# Projected sub‑gradient descent (cheap but robust)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for the true max‑objective.
    Step size = step0 / sqrt(t), projection onto the simplex each iteration.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (shift k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # derivative of a_i * a_{k-i} w.r.t a_i is a_{k-i}
        g[mask] = 2.0 * a[j[mask]]
        # diagonal (i == k-i) contributes only once
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Simulated‑annealing hill‑climb (aggressive)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 6_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    Temperature is linear from T0 down to zero.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 4e-5                 # modest initial temperature
    step_scale = 0.14

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries – cheap but occasionally useful
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # occasionally perform a more radical “re‑randomise” of a tiny chunk
        if np.random.rand() < 0.001:
            block = np.random.choice(n, size=3, replace=False)
            mass = cand[block].sum()
            rand_vec = np.random.rand(3)
            rand_vec /= rand_vec.sum()
            cand[block] = mass * rand_vec

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        T = T0 * (1.0 - step / max_steps)      # linear cooling
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step_scale when we are improving
                step_scale = min(step_scale * 1.06, 0.6)
        else:
            # otherwise shrink step_scale a bit
            step_scale *= 0.97

        if step % 500_000 == 0:
            print(f"[{time.time():.1f}] SA step {step:,}  best={best_score:.6f}")

    return best_a, best_score

    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load fallback candidate if present
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a working length (fallback length or 2048)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048
    print(f"[{time.time():.1f}] Working length = {n}")

    # ------------------------------------------------------------------
    # 2. Fine analytic scan (≈30 s)
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] Starting fine analytic scan (n={n})")
        # parameter grids – note the dense steps
        alphas = np.linspace(0.05, 3.0, 60)          # 0.05 … 3.0
        betas  = np.linspace(0.2, 5.0, 48)           # 0.2 … 5.0
        gammas = np.linspace(0.0, 6.0, 61)           # 0 … 6
        sigmas = np.linspace(0.08 * n, 0.35 * n, 30) # Gaussian width

        # helper to test a vector
        def try_candidate(vec: np.ndarray, comment: str):
            nonlocal best_seq, best_score
            if np.any(np.isnan(vec)) or np.any(np.isinf(vec)):
                return
            vec = np.maximum(vec, 0.0)
            if vec.sum() == 0.0:
                return
            vec /= vec.sum()
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best ({comment}) = {best_score:.6f}")

        # symmetric product family (dense α)
        for a in alphas:
            if time.time() >= deadline - 5.0:
                break
            try_candidate(symmetric_product_family(n, a), f"sym α={a:.2f}")

        # left‑heavy power family (dense β)
        for b in betas:
            if time.time() >= deadline - 5.0:
                break
            try_candidate(left_heavy_power_family(n, b), f"left β={b:.2f}")

        # centred exponential family
        for g in gammas:
            if time.time() >= deadline - 5.0:
                break
            try_candidate(exp_u_family(n, g), f"exp γ={g:.2f}")

        # Gaussian‑shaped family
        for s in sigmas:
            if time.time() >= deadline - 5.0:
                break
            try_candidate(gaussian_family(n, s), f"gauss σ={s:.0f}")

        # a few random starts (both monotone and symmetric)
        for _ in range(50):
            if time.time() >= deadline - 5.0:
                break
            if np.random.rand() < 0.5:
                try_candidate(random_monotone_start(n), "rand mono")
            else:
                try_candidate(random_symmetric_start(n), "rand sym")

    # ------------------------------------------------------------------
    # 3. Normalise the current best to a probability vector
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)
    print(f"[{time.time():.1f}] After scan best_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam optimisation on the smoothed‑max surrogate
    # ------------------------------------------------------------------
    if time.time() + 70.0 < deadline:
        print(f"[{time.time():.1f}] Adam optimisation start")
        a, sc = adam_smooth_optimize(
            a,
            deadline - 12.0,
            max_iters=300_000,
            beta_start=4.0,
            beta_end=5000.0,
            lr=0.012,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected sub‑gradient descent (main bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() + 140.0 < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient descent start")
        a, sc = subgrad_optimize(
            a,
            deadline - 50.0,
            max_iters=3_500_000,
            step0=0.15,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation test (often gives a tiny boost)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        best_seq = a.tolist()
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final aggressive hill‑climb (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a, sc = hill_climb_sa(a, deadline, max_steps=6_000_000)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
