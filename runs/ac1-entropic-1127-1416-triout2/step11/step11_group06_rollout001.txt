assistantfinal**Solution Overview**

The objective to minimise is  

\[
F(a)=\frac{2\,n\;\max_k (a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge0 ,
\]

where \((a*a)[k]=\sum_i a_i a_{k-i}\) is the self‑convolution.
Because the ratio is invariant under a global scaling of the vector,
the problem can be reduced to  

* minimise  \(2n\max_k (a*a)[k]\)  subject to  \(\sum_i a_i = 1,\;a_i\ge0\).

The function is the maximum of many bilinear forms – non‑smooth and non‑convex.
A pure sub‑gradient method (the “mirror‑descent” used before) converges,
but gets stuck close to a local optimum.

**New ideas**

1. **Soft‑max smoothing** – replace the hard max by a smooth
   log‑sum‑exp (temperature \(θ\)).  The gradient of the smoothed
   objective is a *weighted* average of the gradients of all
   convolution entries, which gives a much richer direction early on.
2. **Annealing the temperature** – start with a low temperature (very smooth)
   and increase it slowly, so the optimisation first moves in a well‑behaved
   landscape and finally refines the true max‑objective.
3. **Exponentiated‑gradient (KL‑mirror) update** – the same update as in the
   previous code, but now using the smoothed gradient.
4. **Two‑stage optimisation** –  
   *Stage 1* : smoothed mirror‑descent (large number of cheap iterations).  
   *Stage 2* : standard sub‑gradient mirror‑descent to polish the solution.  
   *Stage 3* : a tiny stochastic hill‑climb.
5. **Diverse analytic seeds** – besides the uniform and central‑binomial
   vectors we also try Beta‑shaped, geometric, power‑law and a sinusoidal
   (half‑sine) shape.  The best seed is taken as the start for the optimiser.

All phases respect the wall‑clock deadline (default 1000 s) and keep the
best sequence seen so far.

```python
"""
Non‑negative step‑function optimiser (improved).

The algorithm consists of three phases:
1. Analytic seeds → best starter.
2. Smoothed KL‑mirror descent (soft‑max temperature annealing).
3. Classic sub‑gradient KL‑mirror descent + short stochastic hill‑climb.

The smoothed phase uses a temperature θ that is increased during the run,
turning the max‑convolution objective into a smooth log‑sum‑exp.
Both mirror steps keep the vector on the probability simplex.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}.
    Heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


def generate_sine_shape(n: int) -> np.ndarray:
    """
    Half‑sine shape (zero at the ends, maximal in the centre):
        a_i ∝ sin(π·(i+0.5)/n).
    """
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / n)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed KL‑mirror descent (log‑sum‑exp temperature)
# ----------------------------------------------------------------------


def smooth_mirror_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.5,
    theta0: float = 5.0,
    theta_max: float = 200.0,
    theta_mul: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on the *smoothed* objective
        f_θ(a) = (1/θ)·log Σ_k exp(θ·(a*a)[k])

    The temperature θ is increased multiplicatively each iteration.
    Returns the best vector seen and its (unsmoothed) score.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    theta = theta0

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # 1) full self‑convolution
        conv = convolution_fft(a, a)                # length 2n‑1
        max_conv = np.max(conv)

        # 2) soft‑max weights w_k ∝ exp(θ·conv_k)
        #    we subtract max_conv for numerical stability
        exp_vals = np.exp(theta * (conv - max_conv))
        w = exp_vals / np.sum(exp_vals)             # length 2n‑1, sums to 1

        # 3) gradient of the smoothed objective:
        #    g_i = 2· Σ_k w_k a_{k-i}
        rev_a = a[::-1]
        conv_w = convolution_fft(w, rev_a)           # length 3n‑2
        g = 2.0 * conv_w[n - 1 : n - 1 + n]           # keep only n components

        # 4) centre the gradient (mirror‑descent trick)
        avg = np.dot(a, g)

        # 5) stepsize
        eta = eta0 / np.sqrt(t)

        # 6) exponentiated‑gradient update, safe clipping
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # 7) project back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # 8) anneal temperature
        if theta < theta_max:
            theta = min(theta_max, theta * theta_mul)

        # occasional progress check
        if t % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score

# Classic (hard‑max) mirror‑subgradient optimiser (unchanged)

def mirror_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).  a stays on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        avg = np.dot(a, g)

        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score

    """
    Very local stochastic moves (swap or fractional mass transfer)
    with simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size ----------------------------------------------------
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        print(f"No external sequence provided – using length {n}.")

    # ------------------------------------------------------------------
    # 1. Build a collection of analytic starters
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform (baseline)
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # classic central‑binomial
    init_candidates.append(generate_central_binomial(n))

    # power‑law family
    for p in np.linspace(0.35, 1.0, 9):
        init_candidates.append(generate_powerlaw(n, float(p)))

    # geometric family (r close to 1)
    for r in [0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:
        init_candidates.append(generate_geometric(n, float(r)))

    # Beta‑shaped family (α in (0.1,0.5])
    for alpha in np.linspace(0.10, 0.55, 10):
        init_candidates.append(generate_beta_shape(n, float(alpha)))

    # Half‑sine shape
    init_candidates.append(generate_sine_shape(n))

    # Possibly include the previously found best construction
    try:
        init_candidates.append(np.array(height_sequence_1, dtype=np.float64))
        print("Added provided height_sequence_1 as a starter.")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Choose the best starter by exact evaluation
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed KL‑mirror descent (large portion of the budget)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 250.0            # leave ~250 s for polishing
    if time.time() < smooth_deadline:
        remaining = smooth_deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 1300))
        print(f"Smoothed mirror‑descent phase – up to {max_iters} iterations.")
        a_smooth, score_smooth = smooth_mirror_optimize(
            best_seq,
            time_deadline=smooth_deadline,
            max_iters=max_iters,
            eta0=0.5,
            theta0=4.0,
            theta_max=180.0,
            theta_mul=1.002,
        )
        if score_smooth < best_val:
            best_val, best_seq = score_smooth, a_smooth.copy()
            print(f"Smoothed phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Classic sub‑gradient mirror descent (refinement)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 70.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_iters = int(min(1_500_000, remaining * 1200))
        print(f"Classic sub‑gradient phase – up to {max_iters} iterations.")
        a_refine, score_refine = mirror_subgrad_optimize(
            best_seq,
            time_deadline=subgrad_deadline,
            max_iters=max_iters,
            eta0=0.7,
        )
        if score_refine < best_val:
            best_val, best_seq = score_refine, a_refine.copy()
            print(f"Sub‑gradient phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Short random‑restart bursts (optional)
    # ------------------------------------------------------------------
    restart_deadline = deadline - 50.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(4):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.02 * np.random.randn(n)
            pert = project_to_simplex(pert)
            sub_dead = time.time() + 9.5
            a_tmp, s_tmp = mirror_subgrad_optimize(
                pert,
                time_deadline=sub_dead,
                max_iters=120_000,
                eta0=0.6,
            )
            if s_tmp < best_val:
                best_val, best_seq = s_tmp, a_tmp.copy()
                print(f"Restart {r} yielded improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Final hill‑climb for up to {steps} steps.")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val, best_seq = hill_score, np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard – fall back to uniform if something went wrong
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 8. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
```
