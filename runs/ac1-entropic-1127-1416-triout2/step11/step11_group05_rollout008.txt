Multi‑start L‑BFGS on a smooth‑max surrogate + final simulated annealing.

The algorithm:
1. Choose a length n (4096 if time permits, otherwise 2048).

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic full convolution using real FFT (fast for large n)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector with sum(a)=1:
        score = 2·n·max(a∗a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (cheap seeds)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w = np.maximum(w, 0.0)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w = np.maximum(w, 0.0)
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i)), normalised."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w = np.maximum(w, 0.0)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)

    The gradient is derived analytically (no finite differences).
    """
    a = softmax(z)                     # a lies on the simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights over the convolution entries
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta

    w = expw / sumexp                  # soft‑max weights, sum to 1
    # gradient w.r.t. a: weighted sum of derivatives of each conv entry
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # shape (n,)

    # back‑prop through the soft‑max (Jacobian = diag(a) - a aᵀ)
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


class SmoothMaxObjective:
    """Callable required by scipy.optimize.minimize (returns f and grad)."""

    __slots__ = ("beta", "n")

    def __init__(self, beta: float, n: int):
        self.beta = beta
        self.n = n

    def __call__(self, z: np.ndarray):
        return smooth_max_and_grad(z, self.beta, self.n)


# ----------------------------------------------------------------------
# L‑BFGS optimisation driver
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Run a sequence of L‑BFGS optimisations on increasing smooth‑max
    parameters `betas`.  After each stage the vector is symmetrised and
    projected onto the simplex.  Returns the best vector found and its
    exact score.
    """
    n = a_start.shape[0]
    best_a = a_start.copy()
    best_score = compute_score(best_a)

    a = best_a.copy()
    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage β={beta}")

        z0 = np.log(a + 1e-12)                     # initialise soft‑max params
        obj = SmoothMaxObjective(beta, n)

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(obj,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure (β={beta}): {e}")
            continue

        a = softmax(res.x)

        # enforce symmetry (helps both numerics and the true objective)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Final simulated‑annealing hill‑climb (light polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Very short SA walk: random swaps / mass transfers on the simplex,
    accepting uphill moves with a linearly decaying temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4   # initial temperature (tiny but non‑zero)
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        temperature = T0 * (1.0 - step / max_steps)
        if delta < 0 or np.random.rand() < math.exp(-delta / (temperature + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    `evaluate_sequence` function.  Returns the best sequence found as a
    plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 2.0

    # -----------------------------------------------------------
    # 0. Attempt to use the globally supplied fallback, if any
    # -----------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose problem size (larger n ⇒ potentially lower bound)
    # -----------------------------------------------------------
    remaining = deadline - time.time()
    n = 4096 if remaining > 650 else 2048
    print(f"[{time.time():.1f}] using length n = {n}")

    # -----------------------------------------------------------
    # 2. Build a pool of seed vectors (cheap analytic families + random)
    # -----------------------------------------------------------
    seed_pool = []

    # a) Analytic families
    for alpha in np.linspace(0.0, 4.0, 11):
        if time.time() >= deadline - 120.0:
            break
        v = symmetric_product_family(n, alpha)
        seed_pool.append(v)

    for beta in np.linspace(0.0, 4.0, 11):
        if time.time() >= deadline - 120.0:
            break
        v = left_heavy_power_family(n, beta)
        seed_pool.append(v)

    for gamma in np.linspace(0.0, 5.0, 11):
        if time.time() >= deadline - 120.0:
            break
        v = exp_u_family(n, gamma)
        seed_pool.append(v)

    # b) Random symmetric Dirichlet draws
    for _ in range(12):
        if time.time() >= deadline - 120.0:
            break
        v = rng.random(n)
        v = (v + v[::-1]) * 0.5                 # enforce symmetry immediately
        v = project_to_simplex(v)
        seed_pool.append(v)

    # -----------------------------------------------------------
    # 3. Quick scan to keep the most promising candidates
    # -----------------------------------------------------------
    # keep top K by exact score (K modest to limit later work)
    top_K = 5
    scored = []
    for vec in seed_pool:
        sc = compute_score(vec)
        scored.append((sc, vec))
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new overall best (pre‑L‑BFGS) = {sc:.6f}")

    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:top_K]]

    # -----------------------------------------------------------
    # 4. Multi‑start L‑BFGS with a dense β‑schedule
    # -----------------------------------------------------------
    betas_schedule = [0.5, 1, 5, 20, 80, 300, 1000,
                      3000, 10000, 30000, 60000]
    # generous iteration budget per stage – we have lots of time
    max_iter_stage = 500

    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 200.0:
            break
        print(f"[{time.time():.1f}] L‑BFGS multi‑start #{idx+1}")
        a_opt, sc_opt = l_bfgs_optimize(a0,
                                        betas=betas_schedule,
                                        max_iter_per_stage=max_iter_stage,
                                        deadline=deadline - 150.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS multi‑start improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Ensure perfect symmetry & project once more
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        sc_sym = compute_score(a_arr)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_arr.tolist()
            print(f"[{time.time():.1f}] symmetry projection improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final short simulated‑annealing polishing
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final SA polishing")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=2_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Fallback (should never be needed)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # return plain Python list as required
    return best_seq
