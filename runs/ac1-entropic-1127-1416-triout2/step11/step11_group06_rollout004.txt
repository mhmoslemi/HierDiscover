def smooth_mirror_optimize(start_a, time_deadline, max_iters, beta0, beta_factor, eta0):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)   # actual objective using max conv

    beta = beta0
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline: break
        # compute conv
        conv = convolution_fft(a, a)
        # compute softmax weights
        # Use stable trick: maxc = conv.max()
        maxc = conv.max()
        # w = exp(beta*(conv - maxc))   # all <=1
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        w = w / w_sum
        # compute gradient g = 2 * (a * w_rev)[...]
        # compute conv_rev = convolution_fft(a, w[::-1])
        conv_rev = convolution_fft(a, w[::-1])
        # g = 2 * conv_rev[n-1:2n-1][::-1]  (as derived)
        g_slice = conv_rev[n-1:2*n-1][::-1]
        g = 2.0 * g_slice

        # center gradient: avg = np.dot(a, g)
        avg = np.dot(a, g)
        # update
        eta = eta0 / np.sqrt(t)
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50, 50)
        a = a * np.exp(exp_arg)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / a_sum

        # optionally increase beta gradually
        if t % (max_iters // 10 + 1) == 0:
            beta *= beta_factor

        # track best using actual objective (max conv)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    # return best
    return best_a, best_score
```

            seeds.append(v)

"""
Non‑negative step‑function optimiser (enhanced).

Idea
----
1.  Build a wide library of analytic “seed’’ sequences (uniform,
    central‑binomial, power‑law, geometric, Beta‑shaped) and also a
    collection of random Dirichlet draws.  Keep the best seed (by the
    exact objective 2·n·max(conv) for a sum‑1 vector).

2.  Run many short mirror‑descent bursts from diverse starts.
    Two flavours are offered:
      • standard sub‑gradient descent on the true max‑convolution
        (the same exponentiated‑gradient update used before);
      • a continuation method that minimises a *smooth* approximation
        of the max using a soft‑max (β‑log‑sum‑exp) and the same
        KL‑mirror update.  β is slowly increased so the optimisation
        first sees a smooth landscape and then focuses on the true
        maximum.

    Restarts are cheap (a few seconds each) and are limited only by
    the overall wall‑clock budget.

3.  The best vector obtained so far is polished with a cheap stochastic
    hill‑climb that swaps or moves small fractions of mass while
    staying on the simplex.

All phases respect a hard deadline (default 1000 s).  The function
returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1).

    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Mirror‑descent on the *true* max‑convolution objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).  a is kept on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active peak
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        avg = np.dot(a, g)

        eta = eta0 / np.sqrt(t)

        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑descent on a *smooth* (log‑sum‑exp) approximation of max
# ----------------------------------------------------------------------


def smooth_mirror_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta0: float = 0.2,
    beta_factor: float = 1.15,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Optimises fβ(a)= (1/β)·log Σ_i exp(β·conv_i(a,a)).
    The smooth objective converges to the true max as β → ∞.
    Uses the same KL‑mirror (exponentiated‑gradient) updates.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)           # true objective (max‑based)

    beta = beta0
    beta_increase_step = max(1, max_iters // 10)

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # convolution a*a
        conv = convolution_fft(a, a)

        # soft‑max weights
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))      # stability: maxc subtracted
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient of the smooth objective:
        #   g = Σ_i w_i ∇ conv_i  = 2·a * (w reversed) (cross‑correlation)
        conv_rev = convolution_fft(a, w[::-1])
        # slice of length n, then reverse to obtain g_k for k=0..n‑1
        g = 2.0 * conv_rev[n - 1 : 2 * n - 1][::-1]

        avg = np.dot(a, g)

        eta = eta0 / np.sqrt(t)
        exp_arg = -eta * (g - avg)
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # increase β periodically (continuation)
        if t % beta_increase_step == 0:
            beta *= beta_factor

        # track true objective occasionally
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climb (local polishing)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer)
    with simplex projection. Returns the best seen list and its score.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
