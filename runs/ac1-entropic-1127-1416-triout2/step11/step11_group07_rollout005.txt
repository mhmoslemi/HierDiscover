"""
candidate_generator.py
---------------------

Optimisation of a non‑negative step function for the challenge
`evaluate_sequence`.  The algorithm mixes three ideas:

1.  **Diverse deterministic / random seeds** – a quick scan over a
    handful of analytic families (sin², beta‑like, exponential) and a
    few Dirichlet draws.  The best seed is kept.

2.  **Alternating linear programme (ALP)** – given a current vector `f`
    we compute `M = max(conv(f,f))`.  The linear programme

         maximise   Σ g_i
         subject to conv(f,g) ≤ M     (element‑wise)
                     g_i ≥ 0

    finds the most mass that can be placed without breaking the
    convolution bound.  After normalising `g` to a probability vector
    we replace `f` by the new shape whenever the objective improves.
    Repeating this step (≈5 – 10 times) drives the max‑convolution
    down dramatically.

3.  **Stochastic hill‑climb polishing** – a cheap random‑swap / mass‑transfer
    local search that accepts only improvements.  It is run on the
    remaining time budget to squeeze out the last few hundredths.

All vectors are kept on the probability simplex (sum = 1) so that
`compute_score(a) = 2·n·max(conv(a,a))` matches the value returned by
`evaluate_sequence`.  The code is fully self‑contained, uses only
NumPy / SciPy and respects the 1000 s time limit.

# Low‑level utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast, exact for doubles)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a_i = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ ((i+1)*(n-i))**p.
    Positive p concentrates in the centre, negative pushes to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_exp_family(n: int, r: float) -> np.ndarray:
    """
    Truncated exponential / geometric shape:
        a_i ∝ r**i  (0 < r < 1)
    """
    i = np.arange(n, dtype=float)
    w = np.power(r, i)
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` positions, assign random weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming helper (alternating LP step)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) such that
        conv(f, g) = A @ g
    for any vector g of length n.
    """
    n = f.shape[0]
    # first column = f padded with zeros, first row = [f0, 0, 0, …]
    c = np.r_[f, np.zeros(n - 1)]
    r = np.r_[f[0], np.zeros(2 * n - 2)]
    T = toeplitz(c, r)          # (2n‑1) × (2n‑1)
    return T[:, :n].astype(np.float64)   # keep only the first n columns


def lp_improve(f: np.ndarray, deadline: float) -> tuple[np.ndarray, bool]:
    """
    One alternating‑LP step.
    Returns a new probability vector (or the original one if the LP fails)
    and a flag indicating whether the step succeeded.
    """
    n = f.shape[0]
    conv_ff = convolution_fft(f, f)
    M = float(conv_ff.max())          # current bound for the LP

    # Build constraint matrix A_ub such that A_ub @ g ≤ M
    A = build_conv_matrix(f)
    b = np.full(A.shape[0], M * 1.0000001)   # tiny slack for numerical safety

    # maximise Σ g_i  → minimise –Σ g_i
    c_obj = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    # Solve using HiGHS (fast interior‑point/simplex hybrid)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return f, False

    g = res.x
    if g.sum() <= 0:
        return f, False

    # Normalise to a probability vector
    g_norm = g / g.sum()
    return g_norm, True


# ----------------------------------------------------------------------
# Simple hill‑climb polishing (fast local search)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, deadline: float, max_steps: int = 500_000) -> np.ndarray:
    """
    Stochastic moves (swap or fractional transfer) that are kept only
    when they improve the exact objective.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:          # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()          # correct numeric drift
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The method:
        * generate a diverse pool of seeds,
        * run a few alternating LP refinements,
        * finish with a stochastic hill‑climb.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.8

    # --------------------------------------------------------------
    # 0 – pick a moderate length (2048 works well for the LP)
    # --------------------------------------------------------------
    n = 2048
    print(f"[info] sequence length chosen = {n}")

    # --------------------------------------------------------------
    # 1 – collect a handful of diverse seeds
    # --------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(generate_sin2(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))
    # exponential/geometric family – sweep a few ratios
    for r in (0.90, 0.94, 0.97, 0.99):
        seeds.append(generate_exp_family(n, r))

    # random Dirichlet draws with various concentrations
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))

    # very sparse spikes – often a good starting point
    for k in (30, 50, 80):
        seeds.append(random_spike(n, k))

    # include the historic fallback if available
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        if len(fallback) == n:
            fb = np.asarray(fallback, dtype=float)
            fb = project_to_simplex(fb / fb.sum())
            seeds.append(fb)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2 – evaluate seeds (exact objective) and keep the best one
    # --------------------------------------------------------------
    best_vec = None
    best_score = float("inf")
    for s in seeds:
        a = project_to_simplex(np.asarray(s, dtype=float))
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_vec = a
            print(f"[seed] new best score = {best_score:.6f}")

    if best_vec is None:                     # should never happen
        best_vec = np.full(n, 1.0 / n, dtype=float)

    # --------------------------------------------------------------
    # 3 – alternating linear‑programming refinement
    # --------------------------------------------------------------
    cur = best_vec.copy()
    cur_score = compute_score(cur)
    print(f"[info] start LP phase – initial score = {cur_score:.6f}")

    # allocate roughly 40 % of the remaining budget to the LP loop
    lp_deadline = start_time + (budget_s * 0.40)
    lp_iter = 0
    while time.time() < lp_deadline:
        lp_iter += 1
        new_vec, ok = lp_improve(cur, lp_deadline)
        if not ok:
            print("[lp] optimisation failed – exiting LP loop")
            break
        new_score = compute_score(new_vec)
        if new_score < cur_score - 1e-9:
            cur = new_vec
            cur_score = new_score
            print(f"[lp] iter {lp_iter:02d} → score = {cur_score:.6f}")
        else:
            # No improvement – stop early to avoid wasted LP solves
            print("[lp] no improvement, terminating LP phase")
            break
        # safety break if we approach the global time limit
        if time.time() > deadline - 10.0:
            break

    # --------------------------------------------------------------
    # 4 – final stochastic hill‑climb polishing on remaining time
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[info] hill‑climb polishing")
        cur = hill_climb(cur, deadline)

    # --------------------------------------------------------------
    # 5 – final conversion and official evaluation
    # --------------------------------------------------------------
    final_seq = cur.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\n[final] reported score = {final_val:.6f}")

    return final_seq
