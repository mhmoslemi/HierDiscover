def piecewise_uniform_family(n, w, mass_edge):
    """Edge half-width w, total mass at edges = mass_edge (0-1). Interior gets remaining mass."""
    if w == 0:
        # no edges, uniform
        return np.full(n, 1.0 / n)
    # allocate edge weight per entry = mass_edge/(2*w)
    edge_weight = mass_edge / (2 * w) if mass_edge > 0 else 0.0
    interior_mass = 1.0 - mass_edge
    interior_len = n - 2 * w
    if interior_len <= 0:
        # all edges
        interior_weight = 0.0
    else:
        interior_weight = interior_mass / interior_len
    a = np.empty(n, dtype=np.float64)
    a[:w] = edge_weight
    a[-w:] = edge_weight
    if interior_len > 0:
        a[w:n-w] = interior_weight
    # ensure positivity and normalization
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

"""
Search for a non‑negative step‑function minimizing the auto‑convolution peak.

The strategy extends the previous multi‑stage approach:
  1.  Exhaustive scan of a richer collection of analytic families
      (product‑power, left‑heavy power, exponential‑U, beta, uniform‑mixes,
      plus new mixtures of uniform+product and product+U, and a simple
      two‑level piecewise‑uniform family).
  2.  Golden‑section refinement of the best product‑power exponent.
  3.  Several restarts of Adam optimisation on the smooth log‑sum‑exp
      surrogate of the max‑convolution, *without* forced symmetrisation –
      this allows asymmetric minima.
  4.  Projected sub‑gradient descent on the exact max‑objective.
  5.  A final simulated‑annealing hill‑climb for polishing.

All vectors are kept on the probability simplex (∑a = 1) via soft‑max
or Euclidean projection.  The routine observes a 1000 s wall‑clock
budget and returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast, O(N log N))."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} (symmetric, U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β} (monotone decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(‑γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    w = lam * uniform + (1.0 - lam) * u_shape
    w /= w.sum()
    return w


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density on the grid."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w /= w.sum()
    return w


def mixed_uniform_product_family(n: int, lam: float, alpha: float) -> np.ndarray:
    """Uniform + symmetric product mixture."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    prod = symmetric_product_family(n, alpha)
    w = lam * uniform + (1.0 - lam) * prod
    w /= w.sum()
    return w


def mixed_product_exp_family(n: int, lam: float, alpha: float, gamma: float) -> np.ndarray:
    """Product‑family mixed with exponential‑U shape."""
    prod = symmetric_product_family(n, alpha)
    u = exp_u_family(n, gamma)
    w = lam * prod + (1.0 - lam) * u
    w /= w.sum()
    return w


def piecewise_uniform_family(n: int, w_half: int, mass_edge: float) -> np.ndarray:
    """
    Two‑level shape:
      – first w_half entries and last w_half entries share `mass_edge` total.
      – the middle region receives the remaining mass uniformly.
    Parameters:
        w_half : half‑width of each edge block (int)
        mass_edge : total probability mass allocated to both edges (0 ≤ mass_edge ≤ 1)
    Returns a probability vector of length n.
    """
    if w_half <= 0 or mass_edge <= 0.0:
        # pure uniform
        return np.full(n, 1.0 / n, dtype=np.float64)

    edge_per_entry = mass_edge / (2 * w_half)
    interior_len = n - 2 * w_half
    if interior_len <= 0:
        interior_per_entry = 0.0
    else:
        interior_per_entry = (1.0 - mass_edge) / interior_len

    a = np.empty(n, dtype=np.float64)
    a[:w_half] = edge_per_entry
    a[-w_half:] = edge_per_entry
    if interior_len > 0:
        a[w_half:n - w_half] = interior_per_entry
    # guard against numerical drift
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 10.0,
                         beta_end: float = 8000.0,
                         lr: float = 0.012,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    start_a : np.ndarray (probability vector)
    deadline : float (unix timestamp after which optimisation stops)
    max_iters : int
    beta_start / beta_end : temperature schedule for the smooth‑max
    lr : learning rate
    sym_interval : if >0, enforce symmetry every `sym_interval` steps
    """
    eps = 1e-12
    z = np.log(start_a + eps)           # log‑weights for soft‑max
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for smooth‑max temperature
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights over all convolution entries
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth‑max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional true‑objective check
        if it % 5000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improvement to {best_score:.6f}")

    # final projection & evaluation
    a_fin = softmax(z)
    if sym_interval:
        a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
        print(f"[{time.time():.1f}] Adam final improvement to {best_score:.6f}")

    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.04) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> Tuple[np.ndarray, float]:
    """
    A long‑run simulated‑annealing hill‑climber on the true objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:   # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                        # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# One‑dimensional golden‑section search
# ----------------------------------------------------------------------


def golden_section_optimize(func, a: float, b: float,
                            max_iter: int = 15) -> Tuple[float, float]:
    """Locate the minimum of `func` on [a,b] using golden‑section search."""
    phi = (1 + 5 ** 0.5) / 2.0
    invphi = 1.0 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
    if fc < fd:
        return c, fc
    else:
        return d, fd

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence minimizing ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. optional fallback (previous best)
    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Choose a working length (trade‑off between resolution and speed)
    # -----------------------------------------------------------
    n = 4096

    # -----------------------------------------------------------
    # 2. Coarse scan of analytic families (including new mixtures)
    # -----------------------------------------------------------
    print(f"[{time.time():.1f}] start coarse scan (n={n})")

    def consider(vec: np.ndarray):
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}]   new best = {best_score:.6f}")

    # symmetric product family
    best_alpha = None
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        consider(symmetric_product_family(n, alpha))
        if best_seq is not None and best_score == compute_score(symmetric_product_family(n, alpha)):
            best_alpha = alpha

    # left‑heavy power
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        consider(left_heavy_power_family(n, beta))

    # exponential‑U
    for gamma in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        consider(exp_u_family(n, gamma))

    # beta family (grid)
    for p in np.linspace(0.1, 3.0, 13):
        for q in np.linspace(0.1, 3.0, 13):
            if time.time() >= deadline - 30.0:
                break
            consider(beta_family(n, p, q))

    # mixture uniform + exponential‑U
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for gamma in np.linspace(0.0, 6.0, 13):
            if time.time() >= deadline - 30.0:
                break
            consider(mixed_uniform_u_family(n, lam, gamma))

    # NEW: mixture uniform + product
    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        for alpha in np.linspace(0.0, 12.0, 31):
            if time.time() >= deadline - 30.0:
                break
            consider(mixed_uniform_product_family(n, lam, alpha))

    # NEW: mixture product + exponential‑U (coarse)
    for lam in (0.0, 0.5, 1.0):
        for alpha in np.linspace(0.0, 12.0, 13):
            for gamma in np.linspace(0.0, 6.0, 7):
                if time.time() >= deadline - 30.0:
                    break
                consider(mixed_product_exp_family(n, lam, alpha, gamma))

    # NEW: two‑level piecewise‑uniform family
    max_w = n // 2
    step_w = max(1, max_w // 40)                     # ~40 different widths
    for w_half in range(1, max_w + 1, step_w):
        for mass_edge in np.linspace(0.0, 0.9, 19):
            if time.time() >= deadline - 30.0:
                break
            consider(piecewise_uniform_family(n, w_half, mass_edge))

    # -----------------------------------------------------------
    # 3. Fine golden‑section optimisation of the product family
    # -----------------------------------------------------------
    if best_alpha is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] refining product‑α around {best_alpha:.3f}")

        def score_alpha(a_val: float) -> float:
            return compute_score(symmetric_product_family(n, a_val))

        lo = max(0.0, best_alpha - 0.6)
        hi = min(12.0, best_alpha + 0.6)
        best_alpha, _ = golden_section_optimize(score_alpha, lo, hi, max_iter=15)
        consider(symmetric_product_family(n, best_alpha))

    # -----------------------------------------------------------
    # 4. First Adam (smooth‑max) refinement – unsymmetric passes
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Adam refinement (unsymmetric, long run)")
        a_start = np.array(best_seq, dtype=np.float64)

        # a handful of restarts (best + a few random mixtures)
        restarts = [a_start]
        while len(restarts) < 4 and time.time() < deadline - 200.0:
            lam = np.random.rand()
            alpha = np.random.uniform(0.0, 12.0)
            restarts.append(mixed_uniform_product_family(n, lam, alpha))

        for idx, a0 in enumerate(restarts):
            if time.time() >= deadline - 150.0:
                break
            a_opt, sc_opt = adam_smooth_optimize(
                a0,
                deadline - 150.0,
                max_iters=300_000,
                beta_start=10.0,
                beta_end=8000.0,
                lr=0.012,
                sym_interval=0,          # keep asymmetry
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] Adam restart {idx} improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Projected sub‑gradient descent on the exact max‑objective
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 50.0,
            max_iters=2_000_000,
            step0=0.035,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Second Adam pass (higher temperature, finer tuning)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] second Adam pass (very high β)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt2, sc_opt2 = adam_smooth_optimize(
            a_start,
            deadline - 30.0,
            max_iters=300_000,
            beta_start=50.0,
            beta_end=15000.0,
            lr=0.008,
            sym_interval=0,
        )
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.tolist()
            print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 7. Final hill‑climb (simulated annealing)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated‑annealing hill‑climb")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=3_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 8. Fallback to a safe uniform vector if everything failed
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final official evaluation (security wrapper)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
