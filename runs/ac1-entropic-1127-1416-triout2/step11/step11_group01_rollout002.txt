"""
Search for a non‑negative step‑function minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

under the simplex constraint.  The algorithm proceeds as follows:

1.  Choose a problem size (default 4096, larger if a fallback sequence
    is provided).
2.  Exhaustively scan several analytic families (symmetric “product”,
    exponential U‑shape, left‑heavy power law and mixtures) on a fine
    grid of parameters.  The best vector from this cheap scan is used as
    a warm‑start.
3.  Refine the warm‑start with a short exponentiated‑gradient (mirror
    descent) phase.
4.  Run a long Adam optimisation on the smooth‐max surrogate of the
    true objective (β slowly increased to a very large value).
5.  Apply a cheap stochastic hill‑climb (simulated‑annealing style) for
    the remaining time budget.
6.  Symmetrise the final vector and return it as a plain list.

All steps are time‑aware: they stop as soon as the global deadline
approaches, guaranteeing that `propose_candidate` finishes before the
1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective (sum(a) assumed = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ [(i+0.5)(n‑0.5‑i)]^{‑α}   (symmetric, arcsine‑like)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{‑β}   (monotonically decreasing left‑to‑right)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential: w_i ∝ exp(‑γ·dist(i,center))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential U‑shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing sequence (used only for fallback)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random U‑shape."""
    gamma = np.random.uniform(0.0, 12.0)
    return exp_u_family(n, gamma)

# Optimisers
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 300_000,
                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the true max‑convolution."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / math.sqrt(t)
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 800.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth max surrogate
        f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    by Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for β on a log‑scale
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # softmax over β‑scaled entries to obtain weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of Σ w_k (a*a)_k  w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # convert to gradient w.r.t. logits z (softmax Jacobian)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing style random walk on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that

    wall‑clock budget and returns a plain Python list ready for
    ``evaluate_sequence``.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback candidate if available
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Determine problem size (larger gives more expressive power)
    # ------------------------------------------------------------------
    fallback_len = len(best_seq) if best_seq is not None else 0
    n = fallback_len if fallback_len >= 1024 else 4096
    print(f"[{time.time():.1f}] using length n={n}")

    # ------------------------------------------------------------------
    # 2.  Fast exhaustive scan of analytic families (fine grids)
    # ------------------------------------------------------------------
    # we keep track of the current best vector as a NumPy array
    if best_seq is None:
        cur_best = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur_best = np.array(best_seq, dtype=np.float64)
        cur_best = np.maximum(cur_best, 0.0)
        if cur_best.sum() == 0.0:
            cur_best = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            cur_best = cur_best / cur_best.sum()

    # helper to possibly update the global best
    def try_update(vec: np.ndarray, label: str) -> None:
        nonlocal cur_best, best_score, best_seq
        if time.time() >= deadline:
            return
        a = np.maximum(vec, 0.0)
        if a.sum() == 0.0:
            return
        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            cur_best = a.copy()
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] new best {label} = {sc:.6f}")

    # --- symmetric product family (arcsine‑like) ---
    alphas = np.linspace(0.05, 2.5, 241)   # step ≈0.01
    for alpha in alphas:
        if time.time() >= deadline - 30.0:

    # --- exponential U‑shape ---
    gammas = np.linspace(0.0, 12.0, 241)
    for gamma in gammas:
        if time.time() >= deadline - 30.0:
            break
        vec = exp_u_family(n, gamma)
        try_update(vec, f"γ={gamma:.3f}")

    # --- left‑heavy power law ---
    betas = np.linspace(0.0, 5.0, 251)
    for beta in betas:
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        try_update(vec, f"β={beta:.3f}")

    # --- mixed uniform / U‑shape ---
    lam_vals = np.linspace(0.0, 1.0, 21)
    for lam in lam_vals:
        for gamma in gammas[::5]:   # coarse inner loop to keep time low
            if time.time() >= deadline - 30.0:
                break
            vec = mixed_uniform_u_family(n, lam, gamma)
            try_update(vec, f"mix λ={lam:.2f} γ={gamma:.2f}")

    # --- random starts (few) ---
    for _ in range(30):
        if time.time() >= deadline - 30.0:
            break
        if np.random.rand() < 0.5:
            vec = random_monotone_start(n)
        else:
            vec = random_u_start(n)
        try_update(vec, "rand")

    # ------------------------------------------------------------------
    # 3.  Short exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting EG refinement")
        a_opt, sc_opt = eg_optimize(
            cur_best,
            deadline - 80.0,
            max_iters=250_000,
            eta0=0.8,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            cur_best = a_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam optimisation on smooth‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 50.0:
        print(f"[{time.time():.1f}] starting Adam refinement")
        # start with a modest β and finish at a very large β
        a_opt, sc_opt = adam_smooth_optimize(
            cur_best,
            deadline - 20.0,
            max_iters=400_000,
            beta_start=2.0,
            beta_end=8000.0,
            lr=0.012,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            cur_best = a_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final stochastic hill‑climb (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # allocate roughly remaining seconds to steps (≈0.001 s per step)
        max_steps = int(min(2_000_000, remaining * 800))
        print(f"[{time.time():.1f}] final hill‑climb (SA) up to {max_steps} steps")
        a_opt, sc_opt = hill_climb_sa(cur_best, deadline, max_steps=max_steps)
        if sc_opt < best_score:
            best_score = sc_opt
            cur_best = a_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Symmetrisation (often beneficial)
    # ------------------------------------------------------------------
    a_sym = (cur_best + cur_best[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        cur_best = a_sym
        best_seq = a_sym.tolist()
        print(f"[{time.time():.1f}] symmetrisation improved best = {sc_sym:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = cur_best.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
