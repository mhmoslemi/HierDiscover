"""
Propose a non‑negative step‑function (probability vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

under the simplex constraint.  The algorithm is a fast multi‑phase
search:

0.  Load a fallback candidate if available.
1.  Exhaustive scan of several analytic families (power, beta, exponential,
    monotone, random) – this quickly finds a decent shape.
2.  Refine the best candidate with a cheap projected sub‑gradient descent.
3.  Aggressive simulated‑annealing hill‑climb that uses the remaining time.
   The move set consists of swaps and small mass‑transfer steps; temperature
   is linearly cooled to zero.
4.  Return the best sequence (as a plain list of floats).

All phases respect the global time budget (default 1000 s) and keep the
vector on the probability simplex after every change.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                # decreasing
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


def power_family(n: int, p: float, reverse: bool = False) -> np.ndarray:
    """a_i ∝ (i+1)^p  (or reversed if reverse=True)."""
    i = np.arange(n, dtype=np.float64)
    if reverse:
        w = (n - i) ** p          # decreasing from left to right
    else:
        w = (i + 1.0) ** p        # increasing from left to right
    a = w / w.sum()
    return a


def beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+0.5)^{α‑1}·(n‑0.5‑i)^{β‑1}."""
    i = np.arange(n, dtype=np.float64)
    # add a tiny epsilon to avoid zero to the power of a negative number
    eps = 1e-12
    left = (i + 0.5 + eps) ** (alpha - 1.0)
    right = (n - 0.5 - i + eps) ** (beta - 1.0)
    w = left * right
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheap iterations)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     step0: float = 0.15) -> Tuple[np.ndarray, float]:
    """Sub‑gradient method for the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal case where i=j (j==idx) should have factor 1
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 5_000_000) -> Tuple[np.ndarray, float]:
    """Random walk on the simplex with occasional uphill moves."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature – scaled to the current score
    T0 = cur_score * 5e-5
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        # two kinds of elementary moves
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                           # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # linear cooling
        T = T0 * (1.0 - step / max_steps)
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

# High‑level driver

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The function respects the global time budget and returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback (if present)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine working length (fallback length or 2048)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Quick analytic / random scan (≈30 s)
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] quick analytic scan (n={n})")

        # a) power families (both increasing and decreasing)
        for p in np.linspace(0.2, 5.0, 30):
            if time.time() >= deadline - 5.0:
                break
            a_inc = power_family(n, p, reverse=False)
            sc = compute_score(a_inc)
            if sc < best_score:
                best_score, best_seq = sc, a_inc.tolist()
                print(f"[{time.time():.1f}] new best (power p={p:.2f} inc) = {best_score:.6f}")

            a_dec = power_family(n, p, reverse=True)
            sc = compute_score(a_dec)
            if sc < best_score:
                best_score, best_seq = sc, a_dec.tolist()
                print(f"[{time.time():.1f}] new best (power p={p:.2f} dec) = {best_score:.6f}")

        # b) beta‑like families – random sampling of (α,β)
        for _ in range(300):
            if time.time() >= deadline - 5.0:
                break
            α = np.random.uniform(0.2, 5.0)
            β = np.random.uniform(0.2, 5.0)
            a_beta = beta_family(n, α, β)
            sc = compute_score(a_beta)
            if sc < best_score:
                best_score, best_seq = sc, a_beta.tolist()
                print(f"[{time.time():.1f}] new best (beta α={α:.2f},β={β:.2f}) = {best_score:.6f}")

        # c) existing families from the reference code
        alphas = np.linspace(0.0, 12.0, 13)
        betas = np.linspace(0.0, 12.0, 13)
        gammas = np.linspace(0.0, 6.0, 13)

        for α in alphas:
            if time.time() >= deadline - 5.0:
                break
            vec = symmetric_product_family(n, α)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (sym α={α:.2f}) = {best_score:.6f}")

        for β in betas:
            if time.time() >= deadline - 5.0:
                break
            vec = left_heavy_power_family(n, β)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (left β={β:.2f}) = {best_score:.6f}")

        for γ in gammas:
            if time.time() >= deadline - 5.0:
                break
            vec = exp_u_family(n, γ)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (exp γ={γ:.2f}) = {best_score:.6f}")

        # d) a few random starts
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            vec = random_monotone_start(n) if np.random.rand() < 0.5 else random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise starter to a probability vector (∑a = 1)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent – allocate a decent slice of time
    # ------------------------------------------------------------------
    if time.time() + 120.0 < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient descent start")
        a, sc = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=2_000_000,
            step0=0.15,
        )
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final aggressive hill‑climb (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA) start")
        a, sc = hill_climb_sa(a, deadline, max_steps=8_000_000)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
