"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a list of floats) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

under the simplex constraint (the vector is scaled to sum = 1, scaling does
not affect the objective).  The routine consists of several stages,
each respecting the global wall‑clock limit (default 1000 s).

0.  Load a fallback candidate (if the variable ``height_sequence_1`` exists).

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT (fast for large n)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)(n‑0.5‑i))^{‑α}  (edges heavy when α>0)."""
    i = np.arange(n, dtype=np.float64)
    logw = -alpha * (np.log(i + 0.5) + np.log(n - 0.5 - i))
    # subtract max for numerical stability
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{‑β} (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    logw = -beta * np.log(i + 0.5)
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shaped exponential: w_i ∝ exp(-γ·dist_to_center)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)  # distance to nearer end
    logw = -gamma * dist
    logw -= np.max(logw)
    w = np.exp(logw)
    return w / w.sum()


def beta_shape_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Generalised Beta‑type shape:
        w_i ∝ (i+0.5)^{p} · (n‑0.5‑i)^{q}
    Allows independent control of the two tails.
    """
    i = np.arange(n, dtype=np.float64)
    # avoid log(0) by never evaluating exactly at boundary (i+0.5>0)
    logw = p * np.log(i + 0.5) + q * np.log(n - 0.5 - i)
    logw -= np.max(logw)
    w = np.exp(logw)
    # numerical safety – replace NaNs (if any) by zeros
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        # fall back to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / total


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape (sorted powered uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential U‑shape."""
    gamma = np.random.uniform(0.0, 8.0)
    return exp_u_family(n, gamma)


def random_beta_start(n: int) -> np.ndarray:
    """Random Beta‑type shape with exponents drawn from a wide range."""
    p = np.random.uniform(-20.0, 20.0)
    q = np.random.uniform(-20.0, 20.0)
    return beta_shape_family(n, p, q)

def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 400_000,
    beta_start: float = 4.0,
    beta_end: float = 5_000.0,
    lr: float = 0.014,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate:
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    Uses the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # log‑sum‑exp with temperature β
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a:   ∂f/∂a = 2 * (a * w) convolved with reversed a
        # Using convolution property:
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize_polyak(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 8_000_000,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    Uses a Polyak step size: (f - f_best) / ||g||².
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal term (i = j) gets factor 1 instead of 2
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        cur_score = 2.0 * n * conv.max()
        # Polyak step size (guard against division by zero)
        norm_sq = np.dot(g, g) + eps
        step = (cur_score - best_score + eps) / norm_sq
        # Clip step to a reasonable range to avoid wild jumps
        step = max(min(step, 0.5), eps)

        a = a - step * g
        a = project_to_simplex(a)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad (Polyak) improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def lp_refine_step(
    f: np.ndarray, M: float, mix_weights: list[float]
) -> tuple[np.ndarray, float]:
    """
    Solve ``max Σ g_j`` s.t. ``(f * g)_k ≤ M`` (all k) and ``g ≥ 0``.
    Return the best convex mixture (1‑t)·f + t·g_norm with t taken from
    *mix_weights* (line‑search).  Both f and g_norm are normalised to
    sum = 1 before mixing.
    """
    n = f.shape[0]
    m_len = 2 * n - 1

    # Build convolution matrix A where (A @ g) = f * g .
    # Each column i of A contains the vector f shifted by i.
    A = np.zeros((m_len, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ↔ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m_len, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return f, compute_score(f)

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, compute_score(f)

    g_norm = g / sum_g

    best_f = f
    best_sc = compute_score(f)
    for t in mix_weights:
        cand = (1.0 - t) * f + t * g_norm
        cand = project_to_simplex(cand)
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_f = cand
    return best_f, best_sc


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 3_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.
    Moves:
        * swap two entries,
        * transfer a fraction of mass from i to j,
        * scale a random contiguous block by a factor s>0.
    The temperature is linearly cooled; step scale adapts to the acceptance
    ratio.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = cur_score * 2e-5          # small initial temperature
    step_scale = 0.15

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        move_type = np.random.rand()
        if move_type < 0.4:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.8:          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        else:                          # block scaling
            # pick a random block length between 1 and n//8
            length = np.random.randint(1, max(2, n // 8))
            start = np.random.randint(0, n - length + 1)
            s = 1.0 + step_scale * (np.random.rand() - 0.5)  # factor ≈ 1±scale/2
            cand[start : start + length] *= s
            # ensure non‑negativity before projection
            cand = np.clip(cand, 0.0, None)

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # linear cooling
        T = T0 * (1.0 - step / max_steps)
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step_scale when we improve
                step_scale = min(step_scale * 1.08, 0.8)
        else:
            # on rejection shrink the step scale a little
            step_scale *= 0.96

        if step % 50000 == 0:
            print(
                f"[{time.time():.1f}] SA step {step}/{max_steps}, best={best_score:.6f}"
            )

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain list of floats (the candidate sequence).
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Load fallback (if present)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_val = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine working length (fallback length or default 2048)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Fast analytic / random scan (≈30 s)
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] analytic scan (n={n})")
        # extended ranges
        alphas = np.linspace(0.0, 30.0, 31)
        betas = np.linspace(0.0, 30.0, 31)
        gammas = np.linspace(0.0, 12.0, 13)

        for alpha in alphas:
            if time.time() >= deadline - 4.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym α={alpha:.2f}) = {best_score:.6f}")

        for beta in betas:
            if time.time() >= deadline - 4.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left β={beta:.2f}) = {best_score:.6f}")

        for gamma in gammas:
            if time.time() >= deadline - 4.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp γ={gamma:.2f}) = {best_score:.6f}")

        # random draws from the Beta‑type family
        for _ in range(200):
            if time.time() >= deadline - 4.0:
                break
            vec = random_beta_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand β‑type) = {best_score:.6f}")

        # a handful of generic random starts
        for _ in range(80):
            if time.time() >= deadline - 4.0:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand generic) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise the current best to a probability vector (sum = 1)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Adam optimisation on smooth‑max (≈45 s if time permits)
    # ------------------------------------------------------------------
    if time.time() + 45.0 < deadline:
        print(f"[{time.time():.1f}] Adam smooth‑max start")
        a, sc = adam_smooth_optimize(
            a,
            deadline - 10.0,
            max_iters=400_000,
            beta_start=4.0,
            beta_end=5_000.0,
            lr=0.014,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient descent with Polyak step (bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() + 120.0 < deadline:
        print(f"[{time.time():.1f}] Sub‑gradient (Polyak) start")
        a, sc = subgrad_optimize_polyak(
            a,
            deadline - 30.0,
            max_iters=8_000_000,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based refinement (multiple passes)
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print(f"[{time.time():.1f}] LP‑refinement loop start")
        mix_weights = [0.3, 0.15, 0.07, 0.03, 0.01, 0.005]
        for it in range(15):
            if time.time() >= deadline - 4.0:
                break
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new, sc = lp_refine_step(a, M, mix_weights)
            if sc < best_score - 1e-9:
                best_score = sc
                a = a_new
                print(f"[{time.time():.1f}] LP step {it} improved to {best_score:.6f}")
            else:
                # shrink mixing weights if no improvement
                mix_weights = [max(w * 0.6, 0.001) for w in mix_weights]

    # ------------------------------------------------------------------
    # 6. Optional symmetrisation (often helps)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final aggressive hill‑climb (use all remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_steps = int(5_000_000 * remaining / 400.0)  # scale steps with remaining time
        print(f"[{time.time():.1f}] final SA hill‑climb start (≈{max_steps} steps)")
        a, sc = hill_climb_sa(a, deadline, max_steps=max_steps)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(
        f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}"
    )
    return final_seq
