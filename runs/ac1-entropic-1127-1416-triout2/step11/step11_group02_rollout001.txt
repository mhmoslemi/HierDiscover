"""
Advanced optimiser for ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Initialisation** – run a cheap parametric sweep (power‑law,
   geometric, mixed) and, if time permits, a few random Dirichlet
   draws.  Keep the best vector (Σa = 1, a_i ≥ 0).

2. **Continuation with a smoothed max** – minimise a smooth
   approximation of the objective
       L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
   For a fixed α the gradient is
       ∇L = Σ_k w_k ∇(a*a)_k   with   w_k = softmax(α·c_k) .
   Using the explicit form ∇(a*a)_k = 2·a_{k‑i},
   the gradient can be obtained by a single FFT‑based convolution.
   Starting with a small α (very smooth) we run projected gradient
   descent, then increase α (continuation) to focus on the true
   max‑norm.  After each step we project onto the probability simplex.

3. **Aggressive stochastic hill‑climb** – from the best smoothed‑optimum
   we perform a long greedy mass‑transfer search.  Moves that lower the
   exact objective are accepted; otherwise the step‑size is slowly
   decreased.  Random jitteres are injected to escape tiny plateaus.

All phases respect the global 1000 s budget and the function always
returns a feasible non‑negative list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        # the projection of the zero vector is the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution (any lengths)
# ----------------------------------------------------------------------
def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y using a real FFT."""
    m = len(x) + len(y) - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact objective (Σa = 1)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector a."""
    n = a.shape[0]
    conv = conv_fft(a, a)               # length 2·n‑1
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families for a quick warm start
# ----------------------------------------------------------------------
def best_initial_vector(n: int, time_deadline: float) -> np.ndarray | None:
    """Search a few basic families (power‑law, geometric, mixed)."""
    best_a = None
    best_val = float("inf")

    # power‑law
    for p in np.linspace(0.30, 1.00, 71):
        if time.time() >= time_deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # geometric decay
    for r in np.linspace(0.50, 0.99, 81):
        if time.time() >= time_deadline:
            break
        a = (r ** np.arange(n, dtype=np.float64))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # mixed: flat head + power‑law tail
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= time_deadline:
            break
        L = int(L_frac * n)
        if L == 0 or L >= n:
            continue
        for p in np.linspace(0.4, 0.8, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head_val = tail.max()
            head = np.full(L, head_val)
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    return best_a


# ----------------------------------------------------------------------
# Smoothed‑max continuation optimisation (projected gradient descent)
# ----------------------------------------------------------------------
def smoothed_optimize(a0: np.ndarray,
                      deadline: float,
                      alphas: list[float],
                      lr0: float = 0.05) -> np.ndarray:
    """
    Minimise a smoothed version of max(conv(a,a)) by continuation.
    ``alphas`` is an increasing list (e.g. [1,2,5,10,20,40]).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_obj = compute_score(a)

    for alpha in alphas:
        it = 0
        # shrink step-size schedule for each α
        while time.time() < deadline:
            it += 1
            # full self‑convolution
            c = conv_fft(a, a)                     # length 2n‑1
            c_max = c.max()

            # soft‑max weights (stable)
            exp_args = alpha * (c - c_max)         # ≤ 0
            w = np.exp(exp_args)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.ones_like(c) / c.size
                w_sum = 1.0
            w /= w_sum

            # gradient: 2 * conv(w, a_rev)[n‑1 : n‑1+n]
            a_rev = a[::-1]
            grad_full = conv_fft(w, a_rev)         # length 3n‑2
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # adaptive step – inversely proportional to the gradient norm
            grad_norm = np.linalg.norm(grad) + 1e-12
            step = lr0 / np.sqrt(it) / grad_norm

            a_candidate = a - step * grad
            a_candidate = project_to_simplex(a_candidate)

            cand_obj = compute_score(a_candidate)

            if cand_obj < best_obj - 1e-12:
                best_obj = cand_obj
                best_a = a_candidate.copy()
                a = a_candidate
                print(f"[smooth] α={alpha:.1f} it={it} obj={cand_obj:.6f}")

            # optional early stop for this α if improvement stalls
            if it % 500 == 0:
                # re‑evaluate current a (might be unchanged)
                cur_obj = compute_score(a)
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                # if we have not improved for many steps we can break
                # (simple heuristic – can be tuned)
                # Here we just continue until time runs out.

            # keep a small safety margin for the next phases
            if time.time() > deadline - 0.2:
                break
        # start next α from the best found so far
        a = best_a.copy()

    return best_a


# ----------------------------------------------------------------------
# Stochastic hill‑climb (greedy mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_iter: int = 1_000_000) -> np.ndarray:
    """Simple greedy hill‑climb with adaptive step‑scale."""
    a = a_start.copy()
    best_a = a.copy()
    best_M = compute_score(a)
    n = a.shape[0]
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a small fraction of mass from i to j
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cand_obj = compute_score(cand)
        if cand_obj < best_M:
            best_M = cand_obj
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        # occasional random jitter to keep the search alive
        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine follows the stages described
    in the module docstring and respects the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for harness bookkeeping
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # 0. Optional fallback sequence (if ``height_sequence_1`` exists)
    # -----------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Problem size (the original run used n = 2048)
    # -----------------------------------------------------------------
    n = 2048

    # -----------------------------------------------------------------
    # 1. Global parametric sweep (≈200 s)
    # -----------------------------------------------------------------
    sweep_deadline = min(deadline, start_time + 200.0)
    a0 = best_initial_vector(n, sweep_deadline)
    if a0 is not None:
        val = compute_score(a0)
        print(f"[global sweep] best score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a0.tolist()

    # -----------------------------------------------------------------
    # 2. A few random Dirichlet initial vectors (optional, time‑permitting)
    # -----------------------------------------------------------------
    rand_deadline = deadline - 100.0  # reserve the last 100 s for optimisation
    while time.time() < rand_deadline:
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        val = compute_score(a_rand)
        if val < best_val:
            best_val, best_seq = val, a_rand.tolist()
            print(f"[random] improved score = {val:.6f}")

    # -----------------------------------------------------------------
    # 3. Smoothed‑max continuation optimisation
    # -----------------------------------------------------------------
    if best_seq is None:
        a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.array(best_seq, dtype=float)

    cont_deadline = deadline - 30.0   # keep ~30 s for the final hill‑climb
    alphas = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
    a_opt = smoothed_optimize(a_start, cont_deadline, alphas, lr0=0.08)
    val = compute_score(a_opt)
    print(f"[smooth opt] best score after continuation = {val:.6f}")
    if val < best_val:
        best_val, best_seq = val, a_opt.tolist()

    # -----------------------------------------------------------------
    # 4. Aggressive hill‑climb (use whatever time is left)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_hill_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_hill_start, deadline, max_iter=2_000_000)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
