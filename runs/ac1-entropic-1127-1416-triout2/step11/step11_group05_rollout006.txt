smooth_max_and_grad(z,beta,n):
    a=softmax(z)
    conv = convolution_fft(a,a)
    maxc=conv.max()
    expw = np.exp(beta*(conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp)/beta
    w = expw / sumexp   # shape 2n-1
        
    # compute gradient w.r.t a (partial derivative of smooth max)
    # derivative of conv * w w.r.t a: Since d conv = 2 * conv_{lin} (?), For each position k, conv[k] = sum_i a_i a_{k-i}. gradient wrt a_j is a_{k-j} + a_{k-j} (2a). Weighted sum w_k gives grad_j = sum_k w_k * 2 a_{k-j} if index valid.
    # That is 2 * (a_rev convolved with w) central slice.
    a_rev = a[::-1]
    # conv_rev = convolution_fft(a_rev, w) gives length n + (2n-1) -1 = 3n-2. The central slice from n-1 to n-1 + n gives length n vector.
    conv_rev = convolution_fft(a_rev, w)
    grad_a = 2 * conv_rev[n-1:n-1+n]  # shape (n,)
    # backprop to z via softmax: grad_z = a * (grad_a - dot(grad_a,a))
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return f, grad_z

"""
Multi‑start smooth‑max L‑BFGS + simulated annealing for minimizing the
auto‑convolution peak (the 2·n·max(a*a)/sum(a)² objective).  The algorithm
generates several analytic and random seeds, refines each with a schedule of
log‑sum‑exp smoothed objectives (increasing β) using L‑BFGS on the
soft‑max parametrisation (which enforces non‑negativity and the simplex
constraint), keeps the best candidate, forces exact symmetry, projects onto
the simplex, and finally performs a long simulated‑annealing hill‑climb.
All steps respect the 1000‑second wall‑clock budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----- Basic utilities -------------------------------------------------

def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    # Based on algorithm of Wang & Carreira‑Perpiñán (2013)
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector that already sums to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

# ----- Analytic families ------------------------------------------------

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, normalized."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()

def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalized."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()

def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i)), normalized."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()

def mixed_family(n: int, alpha: float, beta: float, gamma: float,
                 w1: float, w2: float, w3: float) -> np.ndarray:
    """Weighted mixture of the three basic families."""
    a1 = symmetric_product_family(n, alpha)
    a2 = left_heavy_power_family(n, beta)
    a3 = exp_u_family(n, gamma)
    mix = w1 * a1 + w2 * a2 + w3 * a3
    # bring back to simplex
    return project_to_simplex(mix)

# ----- Smooth‑max surrogate and its gradient ---------------------------

def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where f is a smooth approximation of max(conv) and
    grad is the gradient w.r.t. the soft‑max parameters z.
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                 # length 2n‑1
    maxc = conv.max()
    # log‑sum‑exp weighting (stable)
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta
    w = expw / sumexp                            # shape (2n‑1,)

    # gradient of the smooth max w.r.t a
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # (n,)

    # back‑prop through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return f, grad_z

# ----- L‑BFGS driver -----------------------------------------------------

def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Refine a_start using L‑BFGS on the smoothed max objective.
    ``betas`` is an increasing list of smoothing parameters.
    Returns the best vector (still on the simplex) and its exact score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}] L‑BFGS stage β={beta}")

        z0 = np.log(a + 1e-12)

        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun, z0, method="L-BFGS-B", jac=True, options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure: {e}")
            continue

        a = softmax(res.x)
        # enforce symmetry (helps convergence)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.2f}]    improved to {best_score:.6f}")

    return best_a, best_score

# ----- Simple simulated‑annealing hill‑climb ---------------------------

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on the simplex.
    50 % swaps, 50 % mass transfers, linear temperature decay.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score

# ----- Random Dirichlet generator -------------------------------------

def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point on the simplex using Gamma(1,1)."""
    x = rng.gamma(1.0, size=n)
    return x / x.sum()

    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i)), normalised."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def mixed_family(n: int, alpha: float, beta: float, gamma: float,
                 w1: float, w2: float, w3: float) -> np.ndarray:
    """Weighted mixture of the three basic families."""
    a1 = symmetric_product_family(n, alpha)
    a2 = left_heavy_power_family(n, beta)
    a3 = exp_u_family(n, gamma)
    mix = w1 * a1 + w2 * a2 + w3 * a3
    return project_to_simplex(mix)


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = smooth approximation of max(conv)
        grad = ∂f/∂z   (gradient w.r.t. soft‑max parameters)
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                 # length 2n‑1
    maxc = conv.max()

    # stable soft‑max weighting over the entries of conv
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta

    w = expw / sumexp                            # weights, sum to 1

    # gradient of the smooth max w.r.t. a
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # shape (n,)

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation driver
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS on the smooth‑max surrogate.
    ``betas`` – increasing smoothing parameters.
    Returns the best probability vector (still on the simplex) and its
    exact (non‑smoothed) score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}]   L‑BFGS stage β={beta}")

        z0 = np.log(a + 1e-12)

        def fun(z):
            return smooth_max_and_grad(z, beta, n)

        opts = {"maxiter": max_iter_per_stage, "disp": False}
        try:
            res = minimize(fun, z0, method="L-BFGS-B", jac=True, options=opts)
        except Exception as e:
            print(f"      L‑BFGS failure: {e}")
            continue

        a = softmax(res.x)
        # enforce symmetry (helps stability)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.2f}]       improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    50 % swap, 50 % mass transfer, linear temperature decay.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Random Dirichlet generator
# ----------------------------------------------------------------------


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using Gamma(1,1)."""
    x = rng.gamma(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Main entry point – public API
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` under the simplex constraint.  The routine
    respects the wall‑clock budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0.  Optional fallback using a previously discovered sequence
    # -----------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.2f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1. Choose problem size (larger n gives a potentially better bound)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 720:
        n = 8192
    elif remaining > 480:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.2f}] using length n = {n}")

    # -----------------------------------------------------------------
    # 2. Quick scan of simple analytic families (very cheap)
    # -----------------------------------------------------------------
    # symmetric product family
    for alpha in np.linspace(0.0, 4.0, 15):
        if time.time() >= deadline - 120.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.2f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    # left‑heavy power family
    for beta in np.linspace(0.0, 4.0, 15):
        if time.time() >= deadline - 120.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.2f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    # exponential U‑shape family
    for gamma in np.linspace(0.0, 5.0, 13):
        if time.time() >= deadline - 120.0:
            break
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.2f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # random mixtures of the three families
    for _ in range(12):
        if time.time() >= deadline - 120.0:
            break
        alpha = rng.uniform(0.0, 5.0)
        beta = rng.uniform(0.0, 5.0)
        gamma = rng.uniform(0.0, 5.0)
        w1, w2, w3 = rng.dirichlet([1.0, 1.0, 1.0])
        vec = mixed_family(n, alpha, beta, gamma, w1, w2, w3)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.2f}] new best (mixed) = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Random Dirichlet seeds (coarse L‑BFGS warm‑up)
    # -----------------------------------------------------------------
    warmup_candidates = []
    for _ in range(20):
        if time.time() >= deadline - 120.0:
            break
        vec = random_dirichlet(n, rng)
        sc = compute_score(vec)
        warmup_candidates.append((sc, vec))
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.2f}] new best (rand) = {best_score:.6f}")

    # keep the best few for deeper optimisation
    warmup_candidates.sort(key=lambda x: x[0])
    top_candidates = [v for _, v in warmup_candidates[:5]]

    # -----------------------------------------------------------------
    # 4. Deep L‑BFGS refinement on the most promising candidates
    # -----------------------------------------------------------------
    beta_schedule = [5, 30, 150, 500, 2000, 8000, 50000, 200000]
    max_iter_stage = 250                     # generous per‑stage budget

    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline - 120.0:
            break
        print(f"[{time.time():.2f}] deep L‑BFGS start on warm‑up candidate #{idx+1}")
        a_opt, sc_opt = l_bfgs_optimize(cand,
                                        betas=beta_schedule,
                                        max_iter_per_stage=max_iter_stage,
                                        deadline=deadline - 30.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.2f}] candidate #{idx+1} improved best to {best_score:.6f}")

    # also run a deep L‑BFGS from the current best (in case it wasn't in the list)
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.2f}] extra deep L‑BFGS from current best")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = l_bfgs_optimize(a_start,
                                        betas=beta_schedule,
                                        max_iter_per_stage=max_iter_stage,
                                        deadline=deadline - 30.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.2f}] extra L‑BFGS improved best to {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5. Enforce perfect symmetry and re‑project onto the simplex
    # -----------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        sc_sym = compute_score(a_arr)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_arr.tolist()
            print(f"[{time.time():.2f}] symmetry projection improved best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb (longer walk)
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.2f}] final hill‑climb (SA) – extended walk")
        a_final, sc_final = hill_climb_sa(np.array(best_seq, dtype=np.float64),
                                          deadline=deadline,
                                          max_steps=5_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.2f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 7. Final safety check – evaluate with the official function
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback to a uniform distribution of modest size
        n_small = 256
        best_seq = [1.0 / n_small] * n_small

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.2f}] finished – final reported score = {final_score:.6f}")

    return best_seq
